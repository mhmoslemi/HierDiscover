def build_set_greedy(n, max_collisions, target_size=None, rng):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)   # or Python list
    S = []
    indices = rng.permutation(n)
    for idx in indices:
        # compute pair sums with existing S and self sum
        new_sums = [idx + i for i in S] + [2*idx]  # i >=0
        # check if any sum_counts[sum] + 1 > max_collisions

            S.append(idx)
            for s in new_sums:
                sum_counts[s] += 1
            # optional break if reached target size
            if target_size and len(S) >= target_size:
                break
    return S, sum_counts

def _low_overlap_set(n, max_ordered, rng):
    sum_counts = np.zeros(2*n-1, dtype=np.int16)
    S = []
    perm = rng.permutation(n)
    for idx in perm:
        # Check self term:
        if sum_counts[2*idx] + 1 > max_ordered:
            continue
        # Check cross sums
        if S:
            ys = np.array(S, dtype=np.int32)
            sums = idx + ys
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # Accept idx
        # Update self
        sum_counts[2*idx] += 1
        # Update cross
        if S:
            sum_counts[sums] += 2
        S.append(idx)
    return S, sum_counts

"""
Improved search for the step‑function minimising

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The core idea:

1. **Low‑overlap support** – build a random set of indices whose
   pairwise sums collide only a few times (max 4 ordered collisions).
   A uniform distribution on such a set already gives a much lower
   convolution maximum than the uniform full‑length vector.

2. **Warm‑start** – keep the best of the standard warm‑starts
   (previous solution, uniform, Dirichlet, anti‑pair, etc.).

3. **Continuous refinement** – run a short Adam phase on the smooth‑max
   surrogate and a sub‑gradient (exponentiated‑gradient) phase,
   starting from the best low‑overlap candidate.

4. **Long simulated‑annealing (SA)** – finish with the incremental SA
   from the original reference.  All steps respect the global time
   budget.

The algorithm never uses closures, only top‑level functions and
standard libraries (numpy, time).  It returns a list of non‑negative
heights that can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return float(2.0 * a.shape[0] * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Candidate generators used by the original code (kept for completeness)
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives (identical to the reference implementation)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # soft‑max → simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c               # denominator is 1 (sum=1)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental convolution update (used by the long SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.02,
    max_frac: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing that uses the incremental
    convolution update.  Moves are transfers of a random fraction of
    mass from one position to another.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[SA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Low‑overlap support construction (the new ingredient)
def _low_overlap_set(
    n: int, max_ordered: int, rng: np.random.Generator
) -> tuple[list, np.ndarray]:
    """
    Greedy construction of a set S⊂{0,…,n‑1} such that every possible sum
    appears at most ``max_ordered`` times as an *ordered* pair.
    Returns the list of indices and the full sum‑count array.
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)   # ordered pair counts
    S = []
    perm = rng.permutation(n)
    for idx in perm:
        # self‑pair contribution
        if sum_counts[2 * idx] + 1 > max_ordered:
            continue
        # contributions with already‑chosen elements
        if S:
            ys = np.array(S, dtype=np.int32)
            sums = idx + ys
            if np.any(sum_counts[sums] + 2 > max_ordered):
                continue
        # accept this index
        sum_counts[2 * idx] += 1               # self term
        if S:
            sum_counts[sums] += 2               # ordered cross terms
        S.append(idx)
    return S, sum_counts


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The search consists of:
        * a diverse pool of warm‑starts,
        * a greedy low‑overlap support generator,
        * a short Adam + sub‑gradient refinement,
        * a long simulated‑annealing local search.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – keep the original size used by the reference code
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) Gather warm‑starts (previous best, uniform, simple patterns)
    candidates = []

    # previous best, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
            candidates.append(warm.copy())
            # a few rotations – sometimes useful
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(warm, shift))

    # a few generic patterns
    candidates.append(_uniform_candidate(n))
    for alpha in (0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
    for alpha in (0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # --------------------------------------------------------------
    # 1) Pick the best warm‑start (cheap exact objective)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init warm {idx:02d}] obj {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 2) Search for a low‑overlap support set
    # keep 2 min for the final SA
    support_deadline = deadline - 120.0
    max_ordered = 4                     # allow up to 4 ordered collisions
    attempt = 0
    while time.time() < support_deadline:
        S, _ = _low_overlap_set(n, max_ordered, rng)
        k = len(S)
        if k < 80:                      # too small to be useful
            attempt += 1
            continue
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / k
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Low‑overlap {attempt}] size {k}, obj {val:.6f}")
            # stop early if we already beat the target
            if val < 1.503:
                break
        attempt += 1
        if attempt >= 5000:            # safety cap
            break

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # reserve a small window for the final SA
    remaining_time = deadline - time.time()
    if remaining_time > 30.0:
        # ---- Adam phase (smooth‑max)
        adam_deadline = min(time.time() + 180.0, deadline - 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # ---- Sub‑gradient refinement
        sub_deadline = min(time.time() + 250.0, deadline - 5.0)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=300_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long incremental simulated‑annealing local search
    if time.time() < deadline - 5.0:
        sa_seq, sa_val = _incremental_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.03,
            max_frac=0.4,
        )
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[SA] improved to {sa_val:.6f}")

    # --------------------------------------------------------------
    # 5) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq /= s

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
