def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096
    conv_len = _next_pow2(2*n - 1)

    best_seq = None
    best_val = float('inf')

    def consider(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] new best {val:.6f}")

    # warm start
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:

                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))

            warm = None

            warm = warm / warm.sum()

"""
search for a low‑value step‑function.
The algorithm mixes:

* a broad random‑spike/Dirichlet/Gaussian‑mixture warm‑start,
* a long Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate
  with a very slow temperature schedule,
* a short simulated‑annealing move‑based refinement,
* an exact‑max exponentiated‑gradient (sub‑gradient) pass,
* a cheap final random hill‑climb.

All phases respect the supplied time budget (default 1000 s) and the
best sequence found so far is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr: float = 0.05,
    lam_start: float = 0.02,
    lam_factor: float = 1.01,
    noise_sigma: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)                 # avoid log(0)
    logits = np.log(a0)                         # on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ---------- soft‑max to stay on simplex ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        # temperature = 1/lam  (lam grows → temperature drops)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (entropy mirror map)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ---------- Adam update on logits ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e6)

        # gentle decay of learning rate
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake close to deadline
        if it % 5_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against under‑/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local it={it:07d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing refinement: random mass transfers between
    coordinates are accepted with probability exp(-Δ/T). The temperature
    decays geometrically.
    """
    a = a0.copy()
    cur_val = 2.0 * n * _full_convolution(a, conv_fft_len).max() / (a.sum() ** 2)
    best_val = cur_val
    best_a = a.copy()

    # initialise temperature proportionally to current value
    T = max(1e-4, cur_val * 0.1)
    decay = 0.9995
    min_T = 1e-9

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 40 %) of source mass
        delta = a[i] * (np.random.rand() * 0.4)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = 2.0 * n * _full_convolution(a2, conv_fft_len).max() / (a2.sum() ** 2)
        d = val2 - cur_val

        # Metropolis acceptance
        if d < 0 or np.random.rand() < np.exp(-d / T):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 20000 == 0:
                    print(f"[SA it={it:07d}] new best {val2:.6f}")

        T = max(min_T, T * decay)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] new best {val:.6f}")

    # ------------------------------------------------------------------
    # warm start (if available)
    warm = globals().get("height_sequence_1")

        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    # ------------------------------------------------------------------
    # fast random families (cheap, many)
    # uniform baseline
    _maybe_update(np.full(n, 1.0 / n))

    # Dirichlet / Gamma families with varied concentration
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for alpha in alphas:
        for _ in range(3):
            a = np.random.gamma(alpha, 1.0, size=n)
            a /= a.sum()
            _maybe_update(a)

    # spike sets – random supports of varying size
    for k in (32, 64, 128, 256, 512):
        for _ in range(6):

    for _ in range(12):

        sigma2 = np.random.rand() * 0.12 + 0.02
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        _maybe_update(a)

    # a simple squared sinusoid (zero at the borders)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** 2
    a /= a.sum()
    _maybe_update(a)

    # ------------------------------------------------------------------
    # time allocation for the heavy phases
    now = time.time()
    remaining = deadline - now
    if remaining <= 0:
        return best_seq.tolist()

    # ------------------------------------------------------------------
    # Adam optimisation (main workhorse)
    adam_deadline = time.time() + 0.55 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,

        lr=0.05,
        lam_start=0.02,
        lam_factor=1.01,
        noise_sigma=0.015,

    if val_adam < best_val:

        best_seq = a_adam.copy()

    if time.time() < deadline - 40:

        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, sa_deadline, max_moves=150_000
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (exact max)
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=500_000,
            eta0=0.8,
        )

            best_seq = a_sub.copy()

            best_seq, n, conv_fft_len, deadline, max_moves=120_000

            print(f"[Local] improved best {val_loc:.6f}")

    # fallback – ensure a valid non‑empty return

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
