def _project_onto_simplex(v):
    # v: numpy array
    # returns w s.t. w>=0, sum(w)=1 and w is closest to v in Euclidean norm.
    # Implementation:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The strategy is a multi‑phase search:

1.  Fast parametric family sweep (beta‑shapes, exponentials, Gaussian,
    sine‑powers, cosine‑powers, plus a few random Dirichlet draws).  The
    best raw candidate is kept.

2.  A long Adam optimisation of the smooth‑max (log‑sum‑exp) surrogate.
    The Adam run uses a soft‑max (logits) representation to stay on the
    simplex.

3.  A noisy‑restart Adam loop to escape shallow basins.

4.  An exponentiated‑gradient sub‑gradient phase (the same scheme as the
    reference implementation, but with a slightly larger initial step).

5.  A projected‑gradient descent phase that directly follows the true
    max‑convolution sub‑gradient and projects onto the probability simplex.

6.  A cheap random hill‑climb that moves a random fraction of mass
    between coordinates.

All phases are time‑bound; the current best sequence is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.05,
    lam_start: float = 0.04,
    lam_factor: float = 1.01,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(np.clip(a0, eps, None))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _eval_objective(a0, n, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        # ---- simplex projection (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional noise
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ---- Adam step ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, 1e6)

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _proj_grad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    step0: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the true max‑convolution objective.
    Uses the sub‑gradient (uniform weight on all active indices).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[ProjGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = step0 / np.sqrt(it + 1)
        a_new = a - step * grad_a
        a_new = _project_simplex(a_new)

        a = a_new
        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.5)   # up to 50 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Shape factories -------------------------------------------------------
def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Beta‑type shape a_i ∝ (x·(1‑x))^α  (zero at both ends)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    a = (x * (1 - x)) ** alpha
    a_sum = a.sum()
    if a_sum < 1e-12:
        return np.full(n, 1.0 / n)
    a /= a_sum
    return a


def _exp_shape(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from the centre."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-gamma * np.abs(idx - center))
    a_sum = a.sum()
    a /= a_sum
    return a


def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Discrete Gaussian centred in the middle."""
    idx = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    a = np.exp(-0.5 * ((idx - center) / sigma) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _sin_pow_shape(n: int, p: float) -> np.ndarray:
    """Half‑wave sinusoid raised to power p (always non‑negative)."""
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    a_sum = a.sum()
    a /= a_sum
    return a


def _cos_pow_shape(n: int, p: float) -> np.ndarray:
    """Centered cosine raised to power p (non‑negative on the whole interval)."""
    a = np.cos(np.pi * (np.arange(n) - (n - 1) / 2.0) / n) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    a /= a_sum
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start ---------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # helper to update the global best -----------------------------------
    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1.  fast random / parametric sweep ---------------------------------
    # baseline
    if warm is not None:
        _try_candidate(warm, "Warm")
    else:
        _try_candidate(np.full(n, 1.0 / n), "Uniform")

    # random Dirichlet (full support) with various concentrations
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        if time.time() >= deadline - 5:
            break
        a = np.random.dirichlet(np.full(n, conc))
        _try_candidate(a, f"Dirichlet(c={conc})")

    # beta‑shapes
    for alpha in np.linspace(0.3, 3.0, 20):
        if time.time() >= deadline - 5:
            break
        a = _beta_shape(n, alpha)
        _try_candidate(a, f"Beta(alpha={alpha:.2f})")

    # exponential shapes
    for gamma in np.logspace(-3, 0, 25):   # 0.001 … 1.0
        if time.time() >= deadline - 5:
            break
        a = _exp_shape(n, gamma)
        _try_candidate(a, f"Exp(g={gamma:.4f})")

    # Gaussian shapes
    for sigma_frac in np.linspace(0.02, 0.25, 20):
        if time.time() >= deadline - 5:
            break
        sigma = sigma_frac * n
        a = _gaussian_shape(n, sigma)
        _try_candidate(a, f"Gauss(s={sigma_frac:.3f})")

    # sine‑power family
    for p in np.linspace(0.5, 10.0, 20):
        if time.time() >= deadline - 5:
            break
        a = _sin_pow_shape(n, p)
        _try_candidate(a, f"Sine(p={p:.2f})")

    # cosine‑power family
    for p in np.linspace(0.5, 10.0, 20):
        if time.time() >= deadline - 5:
            break
        a = _cos_pow_shape(n, p)
        _try_candidate(a, f"Cos(p={p:.2f})")

    # ------------------------------------------------------------------
    # 2.  Adam optimisation ------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        adam_deadline = time.time() + 0.55 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=600_000,
            lr=0.05,
            lam_start=0.04,
            lam_factor=1.01,
            noise_sigma=0.02,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam main] improved best {val_adam:.6f}")

        # ---- noisy‑restart Adam (few short bursts) --------------------
        for restart in range(3):
            if time.time() >= deadline - 20:
                break
            eps = 1e-12
            logits = np.log(np.clip(best_seq, eps, None))
            logits += np.random.normal(scale=0.25, size=n)
            a0 = np.exp(logits - np.max(logits))
            a0 /= a0.sum()
            r_deadline = time.time() + 0.12 * (deadline - time.time())
            a_tmp, val_tmp = _adam_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                r_deadline,
                max_iters=200_000,
                lr=0.04,
                lam_start=0.04,
                lam_factor=1.01,
                noise_sigma=0.02,
            )
            if val_tmp < best_val:
                best_val = val_tmp
                best_seq = a_tmp.copy()
                print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # 3.  Sub‑gradient refinement -----------------------------------------
    if time.time() < deadline - 15:
        sg_deadline = deadline - 12
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=300_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4.  Projected‑gradient descent on true max convolution -------------
    if time.time() < deadline - 8:
        pg_deadline = deadline - 5
        a_pg, val_pg = _proj_grad_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pg_deadline,
            max_iters=250_000,
            step0=0.15,
        )
        if val_pg < best_val:
            best_val = val_pg
            best_seq = a_pg.copy()
            print(f"[ProjGrad] improved best {val_pg:.6f}")

    # ------------------------------------------------------------------
    # 5.  cheap local random hill‑climb ------------------------------------
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
