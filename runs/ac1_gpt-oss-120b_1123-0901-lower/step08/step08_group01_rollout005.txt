"""
Spiky‑set search for the autocorrelation bound

The objective to minimise is  

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,

with a ≥ 0, a_i ≤ 1000.  For a vector that consists only of equal‑weight
spikes (a_i = 1/k on k distinct indices) we have  

    sum(a) = 1 ,            max(a∗a) = Mmax/k²

where Mmax is the maximum number of ordered pairs (i,j) with the same
sum i+j.  Randomly chosen spike locations give Mmax ≈ k²/(2n) + O(√(k²·log n/n)),
so the objective behaves roughly like  

    F ≈ 1 + 2·√(n·log n) / k .

Hence taking a moderately large number of spikes (k ≈ 1500 … 2500 for n=4096)
already drives the bound well below the previous best 1.5077.
The algorithm therefore:

1. **Random spiky initialisation** – many random subsets of size k are drawn,
   the best one (smallest F) is kept.
2. **Simple hill‑climbing** – repeatedly move a spike to a new empty index;
   keep the change only if the objective improves.
3. **Optional short gradient refinement** – a few steps of the Adam optimiser
   from the original code are applied (only if time permits).

All work is done with FFT‑based convolution ( O(n log n) ) which is fast for
n=4096.  The routine respects the 1000 s budget and always returns a feasible
sequence (non‑negative, sum≥0.01, entries≤1000).  The final vector is printed
together with its value.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2*len(a)-1 entries (the valid linear part)
    return conv[: 2 * a.shape[0] - 1]

def _objective_from_conv(conv: np.ndarray, n: int, sum_a: float) -> float:
    """Exact objective value from a pre‑computed convolution."""
    return float(2.0 * n * conv.max() / (sum_a ** 2))

def _random_spike_vector(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """Return a length‑n vector with k equal‑weight spikes (sum = 1)."""
    a = np.zeros(n, dtype=np.float64)
    idx = rng.choice(n, size=k, replace=False)
    a[idx] = 1.0 / k
    return a

# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The method concentrates on sparse “spike” vectors and a tiny hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem size – keep the same as the reference implementation
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – massive random spiky draws (≈ 40 % of the time)
    phase1_deadline = start_time + 0.40 * (deadline - start_time)
    while time.time() < phase1_deadline:
        # pick a number of spikes; a good region is 1200 … 2500
        k = rng.integers(1200, 2501)
        a = _random_spike_vector(n, k, rng)

        conv = _full_convolution(a, fft_len)
        val = _objective_from_conv(conv, n, a.sum())

        if val < best_val:
            best_val = val
            best_seq = a
            print(f"[Random] k={k:4d}  val={val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – cheap hill‑climbing on the current best (≈ 50 % of time)
    if best_seq is not None:
        a = best_seq.copy()
        mask = a > 0                      # boolean mask of spike positions
        k = int(mask.sum())

        # pre‑compute the set of indices for fast “empty‑slot” sampling
        all_indices = np.arange(n, dtype=np.int64)

        it = 0
        while time.time() < deadline - 2.0:   # keep a couple of seconds for final tweak
            # choose a spike to move
            src = rng.choice(np.flatnonzero(mask))
            # pick a new position that is currently empty
            dst = rng.choice(all_indices)
            while mask[dst]:
                dst = rng.choice(all_indices)

            # propose the swap
            a2 = a.copy()
            a2[src] = 0.0
            a2[dst] = 1.0 / k               # keep equal weight, sum stays 1

            conv2 = _full_convolution(a2, fft_len)
            val2 = _objective_from_conv(conv2, n, a2.sum())

            if val2 < best_val:
                best_val = val2
                best_seq = a2
                a = a2
                mask = a > 0
                k = int(mask.sum())
                print(f"[Hill] it={it:06d}  improved to {val2:.6f}")

            it += 1
            # stop early if we already have a very good bound
            if best_val < 1.20:
                break

    # ------------------------------------------------------------------
    # Phase 3 – a very short Adam refinement (≤ 5 % of remaining time)
    # This uses the same smooth‑max surrogate as the reference code.
    # It is optional; if time is already exhausted the block is skipped.
    remaining = deadline - time.time()
    if remaining > 5.0 and best_seq is not None:
        # --------------------------------------------------------------
        def _adam_optimize(a0: np.ndarray, max_iters: int = 30_000) -> tuple[np.ndarray, float]:
            """Tiny Adam on the log‑sum‑exp surrogate (as in the reference)."""
            eps = 1e-15
            logits = np.log(a0 + eps)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.02
            lam_factor = 1.02
            best_local = a0.copy()
            best_local_val = float("inf")
            it2 = 0
            while time.time() < deadline and it2 < max_iters:
                # soft‑max to stay on the simplex
                mx = logits.max()
                exp_log = np.exp(logits - mx)
                a = exp_log / exp_log.sum()

                conv = _full_convolution(a, fft_len)
                cur_val = _objective_from_conv(conv, n, a.sum())
                if cur_val < best_local_val:
                    best_local_val = cur_val
                    best_local = a.copy()

                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0 or np.isnan(w_sum):
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the smooth‑max surrogate
                grad = 2.0 * _cross_correlation(w, a, _next_pow2(3 * n - 2))

                # mirror‑descent (softmax) update
                avg = np.dot(a, grad)
                grad_log = a * (grad - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** (it2 + 1))
                v_hat = v / (1.0 - beta2 ** (it2 + 1))
                step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
                logits -= step

                lam = min(lam * lam_factor, 1e7)
                lr *= 0.9999
                it2 += 1
            return best_local, best_local_val

        # auxiliary – cross‑correlation needed only by the tiny Adam step.
        def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len_corr: int) -> np.ndarray:
            rev_a = a[::-1]
            fw = np.fft.rfft(w, fft_len_corr)
            fr = np.fft.rfft(rev_a, fft_len_corr)
            corr = np.fft.irfft(fw * fr, fft_len_corr)
            start = a.shape[0] - 1
            return corr[start : start + a.shape[0]]

        a_adj, val_adj = _adam_optimize(best_seq, max_iters=25_000)
        if val_adj < best_val:
            best_val = val_adj
            best_seq = a_adj
            print(f"[Adam‑refine] new best {val_adj:.6f}")

    # ------------------------------------------------------------------
    # Final safeguard – ensure a feasible vector is returned
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
