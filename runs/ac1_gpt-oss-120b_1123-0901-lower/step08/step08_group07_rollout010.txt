"""
propose_candidate
-----------------
Search for a non‑negative step function minimizing

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

using a multi‑stage strategy:

1.  Warm‑starts + a diverse pool of analytically‑motivated candidates
    (uniform, power‑law, U‑shape, Gaussian edges, triangular,
    linear, sinusoidal, Dirichlet samples and their anti‑pair
    permutations).

2.  Fast exact evaluation (FFT‑based convolution) and keep the best
    few candidates.

3.  Continuous refinement of each top candidate:
        • Adam on a smooth‑max surrogate (log‑sum‑exp) – many
          iterations, temperature (λ) increasing to the true max.
        • Exact‑max exponentiated‑gradient (sub‑gradient) phase.

   The two phases are run for a large fraction of the remaining time,
   repeatedly updating the global best.

4.  Stochastic hill‑climbing (swap / mass‑transfer moves) on the
   current best vector – a simulated‑annealing loop that continues
   until the time budget is exhausted.

5.  Optional up‑sampling: if time permits the solution is linearly
   interpolated to a finer grid (8192 points) and refined once more
   with the same Adam / sub‑gradient pipeline.

All operations respect the safety checks of ``evaluate_sequence``.
The function returns a plain Python list of floats ready for the
scoring script.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (fast, no security checks) ---------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:                     # degenerate – should never happen
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑half left, smallest‑half right (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]                # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]                  # biggest half left
    a[-half:] = sorted_vals[-half:][::-1]          # smallest half right
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = _random_dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape: 1,2,…,⌈n/2⌉,…,2,1 (normalised)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape = shape / shape.sum()
    return shape


def _linear_desc_candidate(n: int) -> np.ndarray:
    """Linearly decreasing from left to right."""
    shape = np.arange(n, 0, -1, dtype=np.float64)
    shape = shape / shape.sum()
    return shape


def _linear_asc_candidate(n: int) -> np.ndarray:
    """Linearly increasing (mirror of descending)."""
    shape = np.arange(1, n + 1, dtype=np.float64)
    shape = shape / shape.sum()
    return shape


def _sinusoidal_candidate(n: int) -> np.ndarray:
    """Positive half‑sine: sin(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = np.sin(np.pi * i / n)
    shape = shape.clip(min=0.0)
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = vals / vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = vals / vals.sum()
    return vals

# Optimisation primitives ------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000,
                   lr: float = 0.07,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.05) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # initial log‑odds
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # convert to gradient w.r.t. logits (projection onto simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape shallow plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 400_000,
                          eta0: float = 0.6) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient step using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size decays like 1/sqrt(t)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
        * swap two entries,
        * or transfer a fraction of mass from i to j.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # temperature linearly decays to 0
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


def _warm_start(n: int):
    """Return a normalised warm‑start from the global variable if it exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    # Parameters
    n = 4096

    candidates.append(_linear_asc_candidate(n))
    candidates.append(_sinusoidal_candidate(n))

    # power‑law and U‑shapes with a range of exponents
    for p in (0.1, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0):
        candidates.append(_powerlaw_candidate(n, p))
        candidates.append(_ushape_candidate(n, p))

    # Gaussian‑edge families
    for gamma in (0.1, 0.3, 0.5, 0.8, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair Dirichlet draws
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # random permutations of a strong anti‑pair starter
    base_ap = _anti_pair_candidate(n, alpha=0.2)
    for _ in range(3):
        perm = np.random.permutation(n)
        candidates.append(base_ap[perm])

    # ------------------------------------------------------------------
    # 2) Quick exact evaluation, keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 10 == 0:
            print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5                         # number of candidates to refine intensively
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuous refinement of the top candidates
    remaining = max(0.0, deadline - time.time())
    per_candidate_budget = remaining * 0.65 / top_k

    for cid, (init_val, cand) in enumerate(evaluated[:top_k]):
        print(f"\n--- Refining candidate {cid} (init val {init_val:.6f}) ---")
        # Adam phase
        adam_dead = time.time() + per_candidate_budget * 0.55
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=500_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # Sub‑gradient phase
        sub_dead = time.time() + per_candidate_budget * 0.45
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=500_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing on the current best
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        print("\n--- Starting local stochastic hill‑climbing ---")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=400_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional upscale to a finer grid (8192) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        print("\n--- Upscaling to finer grid and refining ---")
        # linear interpolation of the current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        big_adam_dead = min(deadline - 2.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=300_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # sub‑gradient refinement on the large grid
        if time.time() < deadline - 1.0:
            a_big_sub, val_big_sub = _subgradient_optimize(
                a_big_adam,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                deadline,
                max_iters=300_000,
                eta0=0.6,
            )
            if val_big_sub < best_val:
                best_val = val_big_sub
                best_seq = a_big_sub.copy()
                print(f"[Big] improved to {val_big_sub:.6f}")
        else:
            # if time ran out, keep the Adam result
            if val_big_adam < best_val:
                best_val = val_big_adam
                best_seq = a_big_adam.copy()
                print(f"[Big] improved to {val_big_adam:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final reporting (using the provided security‑checked evaluator)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final score (via evaluate_sequence) {final_score:.6f}")
    except Exception as e:
        print(f"Failed to evaluate final sequence: {e}")

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
