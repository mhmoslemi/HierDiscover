"""
Propose a low‑objective step‑function.

The algorithm works in three phases under a 1000‑second budget.

Phase 1 – large, cheap exploration.
    * deterministic families (edge‑exponential, edge‑power‑law, beta‑pdf,
      double‑Gaussian, sinusoid, uniform, a reciprocal‑edge shape);
    * many random Dirichlet / smoothed Dirichlet samples;
    * sparse‑spike vectors.
    All candidates are symmetrised, normalised to sum 1 and evaluated
    with the exact objective.  The best ~10 candidates are kept.

Phase 2 – deep refinement of the survivors.
    * a long Adam optimisation on a smooth‑max surrogate of the
      convolution‑maximum (soft‑max with a slowly increasing temperature);
    * a sub‑gradient (exponentiated‑gradient) phase on the exact objective;
    * a projected‑gradient phase.
    Symmetry is enforced after every update (a ↔ a[::-1]) and the vector
    is re‑normalised.

Phase 3 – cheap random hill‑climbing for the remaining seconds.

The whole search respects the global deadline and always returns the best
vector found (as a Python list).  All numbers are non‑negative, finite and
well within the range accepted by ``evaluate_sequence``.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Objective
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective F(a)=2·n·max(a∗a)/(sum a)^2."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Shape generators (all return a *probability* vector, i.e. sum≈1)
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape = shape / shape.sum()
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _reciprocal_edge(n: int) -> np.ndarray:
    """U‑shape a_i ∝ 1/(i+1)+1/(n-i) (large at the ends)."""
    i = np.arange(n, dtype=np.float64)
    shape = 1.0 / (i + 1.0) + 1.0 / (n - i)
    shape /= shape.sum()
    return shape


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k spikes of equal weight placed uniformly at random."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=min(k, n), replace=False)
    a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, alpha))
    return a


def _smooth_random(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random vector smoothed by a small triangular kernel."""
    a = rng.random(n)
    # power transform to bias either to edges or to centre
    exp = rng.uniform(0.2, 5.0)
    a = a ** exp
    kernel = np.array([0.25, 0.5, 0.25])
    a = np.convolve(a, kernel, mode='same')
    a = np.maximum(a, 0.0)
    a /= a.sum()
    # enforce symmetry
    a = (a + a[::-1]) / 2.0
    return a


# ----------------------------------------------------------------------
# Optimisers (all maintain sum==1 and non‑negativity)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.04,
    lam_start: float = 0.5,
    lam_factor: float = 1.01,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start

    best_a = a0.copy()
    best_val = np.inf
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → current iterate on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # enforce exact symmetry after soft‑max
        a = (a + a[::-1]) / 2.0
        a /= a.sum()

        # exact objective (used only for tracking)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (see derivation in the description)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # keep logits symmetric
        logits = (logits + logits[::-1]) / 2.0

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # small random jitter near the deadline
        if it % 5_000 == 0 and time.time() > deadline - 30.0:
            logits += (rng.random(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step‑size
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # enforce symmetry and renormalise
        a = (a + a[::-1]) / 2.0
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = np.inf
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * grad_a

        # projection onto simplex + enforce symmetry
        a = _project_onto_simplex(a)
        a = (a + a[::-1]) / 2.0   # symmetry preserves the simplex property

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climber: move a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of the source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Initial warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        warm = (warm + warm[::-1]) / 2.0
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – cheap wide exploration
    phase1_deadline = start_time + 0.30 * (deadline - start_time)

    # pool of candidates (symmetrised, normalised)
    candidates = []

    # deterministic families
    def add_candidate(arr):
        arr = np.maximum(arr, 0.0)
        arr = (arr + arr[::-1]) / 2.0
        s = arr.sum()
        if s > 0:
            arr = arr / s
        else:
            return
        candidates.append(arr)

    # warm start and its few shifts
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            w = np.roll(warm, shift)
            add_candidate(w)

    # uniform and sinusoid
    add_candidate(_uniform(n))
    add_candidate(_sinusoid(n))

    # edge‑exponential (log‑spaced gamma)
    for g in np.logspace(-2, 2, num=20):
        add_candidate(_edge_exponential(n, float(g)))

    # edge‑power‑law (negative alphas)
    for a in np.linspace(-0.2, -8.0, num=20):
        add_candidate(_edge_power_law(n, float(a)))

    # beta shapes (log‑spaced a,b)
    for a in np.logspace(-1, 1, num=10):
        for b in np.logspace(-1, 1, num=10):
            add_candidate(_beta_shape(n, float(a), float(b)))

    # double‑Gaussian
    for sigma in np.logspace(-2, -0.3, num=15):
        add_candidate(_double_gaussian(n, float(sigma)))

    # reciprocal‑edge shape
    add_candidate(_reciprocal_edge(n))

    # random Dirichlet draws (various concentrations)
    for _ in range(120):
        alpha = rng.uniform(0.01, 5.0) ** 2   # more weight on smaller alphas
        add_candidate(_random_dirichlet(n, alpha, rng))

    # smoothed random draws
    for _ in range(80):
        add_candidate(_smooth_random(n, rng))

    # sparse spikes
    for k in (16, 32, 64, 128, 256, 512, 1024):
        add_candidate(_sparse_spikes(n, k, rng))

    # Keep only the best few (use a heap of size 12)
    best_heap = []   # (value, vector)
    for arr in candidates:
        if time.time() >= phase1_deadline:
            break
        val = _objective(arr, n, conv_fft_len)
        if len(best_heap) < 12:
            heapq.heappush(best_heap, (-val, arr))
        else:
            # heap stores the worst of the current best (negative value)
            if -val > best_heap[0][0]:
                heapq.heapreplace(best_heap, (-val, arr))

    # extract top‑k survivors
    survivors = [item[1] for item in best_heap]

    # ------------------------------------------------------------------
    # Phase 2 – deep refinement of survivors
    # allocate the bulk of remaining time but keep a safety buffer for phase 3
    phase2_deadline = deadline - 25.0
    for idx, cand in enumerate(survivors):
        if time.time() >= phase2_deadline:
            break
        # each survivor gets a slice of the remaining time
        remaining = phase2_deadline - time.time()
        if remaining <= 5.0:
            break
        # Adam optimisation
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=deadline - 10.0,          # reserve a few seconds for later phases
            max_iters=200_000,
            lr_start=0.04,
            lam_start=0.5,
            lam_factor=1.01,
        )
        if val_adam < float('inf'):
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Survivor {idx}] Adam best {val_adam:.6f}")

        # Sub‑gradient refinement (longer)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=deadline - 5.0,
            max_iters=300_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Survivor {idx}] Sub‑grad best {val_sub:.6f}")

        # Projected‑gradient phase (optional but cheap)
        if time.time() < deadline - 3.0:
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline=deadline - 2.0,
                max_iters=200_000,
                eta0=0.4,
            )
            if val_pgd < best_val:
                best_val = val_pgd
                best_seq = a_pgd.copy()
                print(f"[Survivor {idx}] PGD best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – short random local hill‑climb (last few seconds)
    if 'best_seq' not in locals():
        # fallback uniform if nothing succeeded
        best_seq = _uniform(n)
        best_val = _objective(best_seq, n, conv_fft_len)

    if time.time() < deadline - 1.0:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline=deadline - 0.5,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final safety checks & return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
