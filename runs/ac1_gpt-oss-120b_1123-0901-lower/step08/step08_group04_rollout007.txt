def subgradient_opt(a, n, conv_len, fft_len, time_limit):
    best_a = a.copy()
    best_max = _full_convolution(a, fft_len).max()
    it = 0
    while time.time() < time_limit:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        if max_c < best_max:
            best_max = max_c
            best_a = a.copy()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if not np.any(active):
            active = (conv == max_c) # fallback
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, fft_len)  # cross-corr req bigger len
        # step
        eta = 0.5 / np.sqrt(it+1)
        a = a * np.exp(-eta * grad)
        # projection onto simplex
        s = a.sum()
        if s == 0:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        it += 1
    return best_a, best_max
```

"""
Propose a step‑function (non‑negative heights) with a very small
value of F(a)=2·n·max(a∗a)/(∑a)².

The search consists of three stages:

1. **Warm‑starts** – a small pool of diverse candidates (uniform,
   Dirichlet, anti‑pair, triangular, Gaussian‑edge, and the previous
   best solution if present) is built and the best one (according to
   the exact objective) is kept.

2. **Continuous polishing** – the chosen start is refined by a short
   Adam phase on a smooth‑max surrogate (log‑sum‑exp) followed by a
   longer sub‑gradient (exponentiated‑gradient) phase.  Both phases
   work on the simplex (∑a=1) and use FFT‑based convolutions, so they
   are fast even for n=4096.

3. **Long simulated‑annealing (SA)** – the polished vector is further
   improved by an incremental O(n) update of the full convolution.
   Random mass transfers are accepted according to a linear temperature
   schedule; the move size (fraction of transferred mass) is large at
   the beginning and shrinks later, allowing both exploration and
   fine‑grained exploitation.

The algorithm automatically adapts the time spent in each stage to
the global budget (default 1000 s).  The final vector is clipped to the
allowed range [0,1000] and renormalised to sum≈1, which matches the
behaviour of ``evaluate_sequence``.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (assumes ∑a=1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) for a vector a with sum≈1."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()

# Candidate generators -------------------------------------------------
def _warm_start(n: int):
    """Return a copy of the previously found solution (if any) adapted to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place large numbers opposite small ones."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    v = np.random.dirichlet(alpha * np.ones(n))
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return _anti_pair_arrangement(v)


def _triangle_candidate(n: int) -> np.ndarray:
    """Large at the two edges, linearly decreasing to the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape

# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # softmax → simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # build weight vector for the active set (all indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Incremental convolution update (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (index 2*idx in the full convolution)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.05,
    max_frac: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing with incremental convolution updates.
    Moves are transfers of a random fraction of the mass from one position
    to another.  The temperature decays linearly.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        # pick a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 100_000 == 0:
            print(f"[SA it={it:06d}] best max_c {best_val:.9f}")

    return best_a, best_val

    with Adam + sub‑gradient optimisation and finishes with a long
    incremental simulated‑annealing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # problem size – 4096 is a good trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a pool of initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))
    for alpha in (0.2, 0.5, 0.8, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
    for alpha in (0.2, 0.6, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))
    for gamma in (0.2, 0.5, 0.8, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) Pick the best starter (exact objective)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init {idx:02d}] objective {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Continuous optimisation (Adam + Sub‑gradient)
    # Allocate up to 30 % of the remaining budget to Adam, the rest
    # (except the final SA reserve) to sub‑gradient.
    now = time.time()
    remaining = deadline - now
    adam_time = min(150.0, remaining * 0.3)   # at most 30 % or 150 s
    if remaining > 30.0:
        adam_deadline = now + adam_time
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient (use the time that is left after Adam, keeping ~30 s
    # for the final SA)
    now = time.time()
    remaining = deadline - now
    sg_time = max(0.0, remaining - 35.0)       # leave ~35 s for SA
    if sg_time > 5.0:
        sg_deadline = now + sg_time
        a_sg, val_sg = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=500_000,
            eta0=0.8,
        )
        if val_sg < best_val:
            best_val = val_sg
            best_seq = a_sg.copy()
            print(f"[Subgrad] improved to {val_sg:.6f}")

    # ------------------------------------------------------------------
    # 4) Long simulated‑annealing local search (uses whatever time is left)
    now = time.time()
    if now < deadline - 5.0:
        sa_seq, max_c_sa = _incremental_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.05,
            max_frac=0.5,
        )
        val_sa = 2.0 * n * max_c_sa
        if val_sa < best_val:
            best_val = val_sa
            best_seq = sa_seq.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / total

    final_obj = _objective(best_seq)
    print(f"Final objective {final_obj:.6f} (max_c {final_obj/(2*n):.9f}) length {len(best_seq)}")
    return best_seq.tolist()
