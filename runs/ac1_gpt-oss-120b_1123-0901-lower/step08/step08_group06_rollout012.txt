            it += 1

"""
Propose a non‑negative step‑function (list of floats) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of several cheap exploration stages followed by
a fast “Adam” optimisation of a smooth‑max surrogate and a short
projected‑gradient refinement.  It starts from any warm‑start stored in
the global variable ``height_sequence_1`` (if present) and always
returns a valid list before the 1000 s budget expires.

Key ideas
---------

* **Warm‑start & diverse random pool** – Dirichlet samples with a
  wide range of concentration parameters, reshaped by the “anti‑pair”
  ordering (largest opposite smallest).  Additional handcrafted families
  (exponential decay, geometric, triangular, Gaussian‑edges) are also
  tried.

* **Smooth‑max surrogate** –  max(conv) is replaced by a log‑sum‑exp
  approximation controlled by a temperature λ.  Optimising the surrogate
  with Adam on the logits (soft‑max parametrisation) gives a fast descent
  direction while respecting the simplex constraint.

* **Projected sub‑gradient fine‑tuning** – after the Adam phase a few
  additive updates on the exact max are performed (Nesterov‑type momentum
  + Euclidean projection onto the probability simplex).

* **Local annealing** – stochastic mass‑transfer moves are applied as a
  last polishing step.

* **Optional up‑sampling** – if time permits the best vector is linearly
  interpolated to 8192 points and a short Adam refinement is run on the
  finer grid.

All FFT‑based convolutions are performed with NumPy’s real FFT for speed.
The code is written without closures or lambdas and respects the 1000 s
global time limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise to a probability vector (sum = 1)
    warm = warm / warm.sum()
    return warm


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]                       # biggest half, left side
    a[-half:] = s[-half:][::-1]               # smallest half, right side reversed
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _exponential_decay_candidate(n: int, rate: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-rate * i)
    shape /= shape.sum()
    return shape


def _geometric_candidate(n: int, r: float) -> np.ndarray:
    shape = (1 - r) * (r ** np.arange(n))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Soft‑max surrogate (log‑sum‑exp) and gradient w.r.t. logits
def _smooth_max_and_grad(logits: np.ndarray, n: int,
                         conv_fft_len: int, corr_fft_len: int,
                         lam: float) -> tuple[float, np.ndarray]:
    """Return smooth‑max surrogate value and gradient w.r.t. logits."""
    # soft‑max
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    # convolution
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # smooth‑max (log‑sum‑exp) around max_c
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    # gradient w.r.t. a (softmax output)
    w = z / sum_z                     # soft‑max weights (sum to 1)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    return surrogate, grad_logits.astype(np.float64)


def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    return exp_l / exp_l.sum()


# ----------------------------------------------------------------------
# Adam optimiser for the smooth‑max surrogate
def _adam_optimize(logits: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   lam_schedule: tuple[float, ...] = (5.0, 15.0, 40.0, 120.0, 300.0),
                   lr_base: float = 0.12,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps_adam: float = 1e-8) -> tuple[np.ndarray, float]:
    """Run Adam on the smooth‑max surrogate with increasing λ."""
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_logits = logits.copy()
    best_val = _objective(_softmax(logits))

    it_total = 0
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # learning‑rate scaled with λ (larger λ → sharper, need smaller steps)
        lr = lr_base / (lam**0.5 + 1e-12)
        # allocate at most 60 s per λ (or whatever is left)
        stage_deadline = min(deadline, time.time() + 60.0)
        while time.time() < stage_deadline:
            surrogate, grad = _smooth_max_and_grad(logits, n,
                                                   conv_fft_len, corr_fft_len, lam)
            it_total += 1
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** it_total)
            v_hat = v / (1.0 - beta2 ** it_total)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

            # occasional exact evaluation
            if it_total % 200 == 0:
                a = _softmax(logits)
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_logits = logits.copy()
                    print(f"[Adam λ={lam:.1f} it={it_total}] new best {val:.6f}")

    # final best vector
    best_seq = _softmax(best_logits)
    best_val = _objective(best_seq)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Projected additive sub‑gradient optimisation (with simple momentum)
def _additive_subgradient_optimize(a0: np.ndarray,
                                  n: int,
                                  conv_fft_len: int,
                                  corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 200_000,
                                  eta0: float = 0.5,
                                  beta: float = 0.9) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    momentum = np.zeros_like(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        # simple Nesterov‑type momentum on the gradient
        momentum = beta * momentum + (1.0 - beta) * grad_a
        a_new = a - eta * momentum
        a = _project_onto_simplex(a_new)

        if it % 1000 == 0:
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Local stochastic annealing (mass‑transfer moves)
def _local_annealing(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.6)   # move up to 60 % of mass at i
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Anneal it={it:06d}] best {best_val:.6f}")
        it += 1

    return best_a, best_val

    # ------------------------------------------------------------------
    # 3) Exponential‑decay families (provide monotone shapes)
    for rate in np.linspace(0.0003, 0.003, 8):
        if time.time() >= deadline:
            break
        cand = _exponential_decay_candidate(n, rate)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Exp decay] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Geometric families
    for r in np.linspace(0.90, 0.9995, 8):
        if time.time() >= deadline:
            break
        cand = _geometric_candidate(n, r)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Geometric] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Adam optimisation of smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 20.0:
        logits = np.log(best_seq + 1e-12)
        seq_adam, val_adam = _adam_optimize(
            logits, n, conv_fft_len, corr_fft_len, deadline,
            lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0),
            lr_base=0.12
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 6) Additive projected sub‑gradient fine‑tuning (with momentum)
    if best_seq is not None and time.time() < deadline - 10.0:
        seq_add, val_add = _additive_subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000, eta0=0.5, beta=0.9
        )
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # ------------------------------------------------------------------
    # 7) Local stochastic annealing
    if best_seq is not None and time.time() < deadline - 5.0:
        seq_loc, val_loc = _local_annealing(
            best_seq, n, conv_fft_len, deadline,
            max_moves=350_000
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Anneal] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 8) Optional up‑sampling to a finer grid (8192 points)
    if time.time() < deadline - 5.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        if time.time() < deadline - 2.0:
            logits_big = np.log(a_big + 1e-12)
            seq_big, val_big = _adam_optimize(
                logits_big, n_big,
                conv_fft_len_big, corr_fft_len_big,
                deadline,
                lam_schedule=(5.0, 20.0, 80.0, 200.0),
                lr_base=0.09
            )
            if val_big < best_val:
                best_val, best_seq = val_big, seq_big.copy()
                print(f"[BigAdam] improved to {val_big:.6f}")

    # ------------------------------------------------------------------
    # 9) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
