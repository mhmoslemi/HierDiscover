def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    candidates = []
    # warm start
    warm = _warm_start(n)
    if warm is not None: candidates.append(warm)

    # base candidates
    candidates.append(_uniform_candidate(n))
    # Dirichlet
    for alpha in [0.1,0.3,0.5,0.7,1.0,2.0]:
        candidates.append(_random_dirichlet_candidate(n, alpha))
    # anti-pair
    for alpha in [0.1,0.3,0.5,0.7,1.0]:
        candidates.append(_anti_pair_candidate(n, alpha))
    # gaussian edges
    for gamma in [0.1,0.2,0.4,0.8,1.2]:
        candidates.append(_gaussian_edges_candidate(n, gamma))
    # triangle
    candidates.append(_triangle_candidate(n))
    # Chebyshev
    try:
        from scipy.signal import chebwin
        for atten in [40, 70, 100, 130]:
            w = chebwin(n, at=atten, sym=True)
            candidates.append(w.astype(np.float64) / w.sum())
    except Exception:
        pass
    # Kaiser
    try:
        from scipy.signal import kaiser
        for beta in [4,6,8,10,12]:
            w = kaiser(n, beta, sym=True)
            candidates.append(w.astype(np.float64) / w.sum())
    except Exception:
        pass

    # Ensure each candidate is non-negative and sum>0.01 and normalized
    cleaned = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        c = c / s
        cleaned.append(c)

    best_seq = None
    best_val = np.inf
    for idx,c in enumerate(cleaned):
        val = _objective(c)
        print(f"[Init {idx:02d}] val {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = c.copy()
    print(f"Best after init {best_val:.6f}")

    # Define sub-deadline for heavy phases
    # Use fraction of total time
    time_left = deadline - time.time()
    # invariants: allocate ~20% to Adam, 30% to subgrad, rest to SA
    adam_time = min(200.0, time_left * 0.2)
    subgrad_time = min(300.0, time_left * 0.3)
    # Expect SA remaining
    # If time left is insufficient skip phases

    # Adam phase
    if time.time() < deadline - (subgrad_time + 30): # keep some for SA

"""
Propose a low‑value step function.

Algorithm
---------
1. Build a diverse pool of warm‑starts:
   * previous best (if any)
   * uniform, Dirichlet, anti‑pair, edge‑focused and cosine‑shaped samples
   * a few classic windows (Chebyshev, Kaiser) when SciPy is available.
2. Pick the best initial candidate using the exact objective.
3. Refine it with two gradient‑based phases:
   * Adam on a smooth‑max (log‑sum‑exp) surrogate.
   * Exact‑max sub‑gradient (exponentiated‑gradient) updates.
   Both phases run until a time budget slice is exhausted.
4. Long simulated‑annealing (SA) local search.
   Moves transfer a random fraction of mass from one position to another.
   The convolution is updated incrementally in O(n).  Linear temperature
   schedule and occasional large‑fraction moves help escape shallow minima.
5. Final safety: clip, normalise (sum ≈ 1) and return as a plain list.

The routine respects the global time limit (default 1000 s) and prints
progress.  All helper functions are top‑level and use only NumPy (and
SciPy optionally for extra windows).
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant) ------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a)/(sum(a)²); assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Warm start -----------------------------------------------------------
def _warm_start(n: int):
    """Adapt a previously found sequence (global variable) to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None

def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Large values on left, mirrored small values on right."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)
    return _anti_pair_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    """Classic triangular window (largest in centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _cosine_abs_candidate(n: int) -> np.ndarray:
    """V‑shaped window: |cos|, high at the edges, zero centre."""
    i = np.arange(n, dtype=np.float64)
    shape = np.abs(np.cos(np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _cosine_squared_candidate(n: int) -> np.ndarray:
    """Raised‑cosine (square) – also high at the edges, low centre."""
    i = np.arange(n, dtype=np.float64)
    shape = np.cos(np.pi * i / (n - 1)) ** 2
    shape /= shape.sum()
    return shape


def _chebwin_candidate(n: int, atten: float = 100.0) -> np.ndarray:
    try:
        from scipy.signal import chebwin
        w = chebwin(n, at=atten, sym=True).astype(np.float64)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        return w
    except Exception:
        return _uniform_candidate(n)   # fallback


def _kaiser_candidate(n: int, beta: float = 8.0) -> np.ndarray:
    try:
        from scipy.signal import kaiser
        w = kaiser(n, beta, sym=True).astype(np.float64)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        return w
    except Exception:
        return _uniform_candidate(n)

# Optimisation primitives -----------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.06,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (fast, no extra scaling)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient optimisation (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """
    Increment a[idx] by `delta` (delta may be negative) and
    update the full linear convolution a*a in O(n) using slice ops.
    """
    old = a[idx]
    conv[idx: idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.08,
    max_frac: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing.
    Moves: transfer a random fraction of mass from one index to another.
    Returns the best vector found and its *convolution maximum* (not yet scaled).
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        # choose a different index j
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # random fraction of the mass at i to move to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear cooling schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert move
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[SA it={it:07d}] best conv‑max {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the allocated time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – 4096 offers a good speed–quality trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a rich pool of diverse warm‑starts
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotations of the warm start – sometimes useful
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # basic shapes
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))
    candidates.append(_cosine_abs_candidate(n))
    candidates.append(_cosine_squared_candidate(n))

    # Dirichlet draws
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair draws
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # edge‑focused Gaussians
    for gamma in (0.1, 0.2, 0.4, 0.8, 1.2):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # classic windows (if SciPy is present)
    for atten in (40, 70, 100, 130):
        candidates.append(_chebwin_candidate(n, atten))
    for beta in (4, 6, 8, 10, 12):
        candidates.append(_kaiser_candidate(n, beta))

    # --------------------------------------------------------------
    # Normalise & filter candidates
    clean = []
    for c in candidates:
        c = np.maximum(c, 0.0)
        s = c.sum()
        if s < 0.01:
            c = np.random.dirichlet(np.ones(n))
            s = c.sum()
        c = c / s
        clean.append(c)

    # --------------------------------------------------------------
    # 2) Pick the best starter (exact objective)
    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(clean):
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] objective {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"[Init] best objective {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # Reserve a few seconds for the final SA
    sa_deadline = deadline - 30.0
    # ---------- Adam phase ----------
    if time.time() < sa_deadline - 70.0:
        adam_deadline = min(sa_deadline - 70.0, time.time() + 200.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            adam_deadline, max_iters=500_000,
            lr=0.06, lam_start=0.5, lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ---------- Sub‑gradient phase ----------
    if time.time() < sa_deadline - 20.0:
        sub_deadline = min(sa_deadline - 20.0, time.time() + 300.0)
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_deadline, max_iters=600_000, eta0=0.5,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long simulated‑annealing local search
    if time.time() < deadline - 5.0:
        sa_seq, sa_conv_max = _incremental_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.08,
            max_frac=0.45,
        )
        sa_obj = 2.0 * n * sa_conv_max
        if sa_obj < best_val:
            best_val = sa_obj
            best_seq = sa_seq.copy()
            print(f"[SA] improved to objective {sa_obj:.6f} (conv‑max {sa_conv_max:.6f})")

    # --------------------------------------------------------------
    # 5) Final safety checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / total

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
