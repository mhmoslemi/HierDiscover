def _lp_optimize_sequence(a0, n, fft_len, deadline, max_outer=15, tol=1e-12):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)  # compute objective (2*n*max)
    while time.time() < deadline and max_outer>0:
        conv = _full_convolution(a, fft_len)
        M = conv.max()
        # get index of max
        max_idx = int(conv.argmax())
        # initialize active set
        active = set([max_idx])
        # inner cut-plane loop
        while True:
            # Build LP matrices
            active_list = sorted(active)
            m = len(active_list)
            A_ub = np.zeros((m, n+1), dtype=np.float64)
            for row_i, k in enumerate(active_list):
                i_low = max(0, k - (n - 1))
                i_high = min(k, n - 1)
                # row_i for g variables
                A_ub[row_i, i_low:i_high+1] = a[k - np.arange(i_low, i_high+1)]
                A_ub[row_i, -1] = -1.0
            # equality for sum(g) = 1
            A_eq = np.zeros((1, n+1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])
            c = np.zeros(n+1, dtype=np.float64)
            c[-1] = 1.0  # minimize M
            bounds = [(0.0, None)] * n + [(0.0, None)]
            res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=np.zeros(m), A_eq=A_eq, b_eq=b_eq, bounds=bounds, method='highs')
            if not res.success:
                print("[LP] solve failed: ", res.message)
                break
            g = res.x[:n]
            M_opt = res.x[-1]
            # compute convolution a * g
            conv_ag = _conv_ab(a, g, fft_len)
            max_val = conv_ag.max()
            max_idx_new = int(conv_ag.argmax())
            # if max_val <= M_opt + tol: feasible for all constraints
            if max_val <= M_opt + 1e-12:
                # feasible solution found
                break
            # else add violating constraint
            active.add(int(max_idx_new))
            # continue inner loop
        # now we have candidate g
        # evaluate objective for g
        val_g = _objective(g)
        if val_g < best_val - 1e-12:
            best_val = val_g
            best_a = g.copy()
            print(f"[LP iter] improved bound to {val_g:.6f}")
        # check if improvement significant: if M_opt >= M - tol: break
        if M_opt >= M - 1e-12:
            # no improvement
            break
        # else update a = g (maybe mix)
        a = g.copy()
        # next outer iteration
        max_outer -= 1

    return best_a, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096  # can optionally increase but choose 4096 to be safe.

    conv_fft_len = _next_pow2(2 * n - 1)

    # Build list of initial candidates
    init_candidates = []

    # warm start from previous global
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            warm = np.resize(warm, n)  # unsaid but we can trim/pad
            if warm.shape[0] < n:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        warm /= warm.sum()
        init_candidates.append(warm)

    # add a few random Dirichlet candidates
    for alpha in (0.3, 0.7, 1.0, 2.0):
        cand = np.random.dirichlet(alpha * np.ones(n))
        init_candidates.append(cand)

    # some edge-biased shapes
    init_candidates.append(_gaussian_edges_candidate(n, gamma=0.5))
    init_candidates.append(_anti_pair_candidate(n, alpha=0.7))
    init_candidates.append(_triangle_candidate(n))

    # keep track of best overall
    best_seq = None
    best_val = float("inf")

    for idx, cand in enumerate(init_candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            cand_sum = cand.sum()
        cand = cand / cand_sum

        # run LP based refinement
        a_refined, val_refined = _lp_refine(cand, n, conv_fft_len, deadline)

        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[candidate {idx}] LP improved to {val_refined:.6f}")

    # If still time, do a final SA fine-tune on best_seq
    if best_seq is not None and time.time() < deadline - 5.0:
        remaining = deadline - time.time() - 5.0
        a_sa, val_sa = _incremental_sa(best_seq, time_limit=deadline - 5.0, start_temp=0.02, max_frac=0.3)
        conv = _full_convolution(a_sa, conv_fft_len)
        val = 2.0 * n * conv.max()   # exact objective
        if val < best_val:
            best_val = val
            best_seq = a_sa.copy()
            print(f"[final SA] improved to {best_val:.6f}")

    # final safety
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq /= best_seq.sum()
    print(f"Final best objective {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()

A direct linear‑programming refinement for the min‑max convolution

The LP step solves

    minimise   M

               (f * g)[k] ≤ M   for a (small) active set of indices k,

The routine respects the 1000 s budget, never returns a sequence with
negative entries, and always normalises the sum to 1 before returning.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic FFT utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) via FFT.
    Used for the exact objective.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_ab(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*b (length len(a)+len(b)‑1) via FFT.
    Used inside the LP refinement.
    """
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: a.shape[0] + b.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (no security checks) ---------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the centre, small at the edges."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Incremental convolution update (used by the long SA) -----------------
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(a0: np.ndarray,
                    time_limit: float,
                    start_temp: float = 0.02,
                    max_frac: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing that uses the incremental
    convolution update.  Moves transfer a random fraction of mass from one
    index to another.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[SA it={it:06d}] best conv‑max {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑based refinement ----------------------------------------------------
def _lp_refine(a0: np.ndarray,
               n: int,
               fft_len: int,
               deadline: float,
               max_outer: int = 12,
               tol: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Repeatedly solves the LP

        minimise   M
        subject to  Σ_i a_i g_{k‑i} ≤ M   (k in an active set)
                    Σ_i g_i = 1 , g_i ≥ 0

    where `a` is the current sequence.  The active set is enlarged
    with any newly violated convolution constraint.  The procedure stops
    when no improvement is observed or the time budget expires.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    while time.time() < deadline and max_outer > 0:
        # current convolution and its maximum
        conv = _full_convolution(a, fft_len)
        M_curr = conv.max()
        k_max = int(conv.argmax())

        # initialise active set with the (currently) worst index
        active = {k_max}
        # ------------------------------------------------------------------
        # inner cutting‑plane loop (adds constraints until the solution is feasible)
        while True:
            rows = sorted(active)
            m = len(rows)

            # build A_ub * [g, M] ≤ 0  (i.e. Σ a_i g_{k‑i} – M ≤ 0)
            A_ub = np.zeros((m, n + 1), dtype=np.float64)
            for r, k in enumerate(rows):
                i_low = max(0, k - (n - 1))
                i_high = min(k, n - 1)
                idxs = np.arange(i_low, i_high + 1)
                A_ub[r, i_low:i_high + 1] = a[k - idxs]          # coeff. = a_{k‑i}
                A_ub[r, -1] = -1.0                                 # –M

            # equality Σ g_i = 1
            A_eq = np.zeros((1, n + 1), dtype=np.float64)
            A_eq[0, :n] = 1.0
            b_eq = np.array([1.0])

            # objective: minimise M
            c = np.zeros(n + 1, dtype=np.float64)
            c[-1] = 1.0

            bounds = [(0.0, None)] * n + [(0.0, None)]

            res = scipy.optimize.linprog(c,
                                         A_ub=A_ub,
                                         b_ub=np.zeros(m),
                                         A_eq=A_eq,
                                         b_eq=b_eq,
                                         bounds=bounds,
                                         method='highs')

            if not res.success:
                print("[LP] solve failed:", res.message)
                return best_a, best_val

            g = res.x[:n]
            M_opt = res.x[-1]

            # check feasibility of the candidate against the *full* set of constraints
            full_conv = _conv_ab(a, g, fft_len)
            new_max = full_conv.max()
            k_new = int(full_conv.argmax())

            if new_max <= M_opt + 1e-12:          # feasible for all k
                break
            # otherwise add the most violated index and repeat
            active.add(k_new)

        # At this point g is feasible and achieves a bound M_opt ≤ M_curr
        val_g = _objective(g)

        if val_g < best_val - 1e-12:
            best_val = val_g
            best_a = g.copy()
            print(f"[LP] improved objective to {best_val:.6f}")

        # stop if the bound did not improve appreciably
        if M_opt >= M_curr - tol:
            break

        # replace a by the newly found feasible vector and continue
        a = g.copy()
        max_outer -= 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the allotted time.
    The method combines:
      * a small pool of diverse warm‑starts,
      * a cutting‑plane LP refinement that directly lowers the peak of the
        autocorrelation, and
      * a final incremental simulated‑annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 is a good compromise for speed / quality
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) build a pool of initial candidates
    candidates = []

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
            candidates.append(warm)

    # a few generic starters
    candidates.append(_uniform_candidate(n))
    for a in (0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, a))
    for a in (0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, a))
    for g in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma=g))
    candidates.append(_triangle_candidate(n))

    # ------------------------------------------------------------------
    # 2) pick and improve the best starter
    best_seq = None
    best_val = float("inf")

    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()

        init_val = _objective(cand)
        print(f"[Init {idx:02d}] objective {init_val:.6f}")

        # LP refinement (uses the remaining time)
        refined_seq, refined_val = _lp_refine(cand, n, fft_len, deadline)

        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[Best after LP] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) final simulated‑annealing polish (if time permits)
    if best_seq is not None and time.time() < deadline - 5.0:
        sa_seq, sa_conv_max = _incremental_sa(best_seq,
                                               time_limit=deadline - 5.0,
                                               start_temp=0.025,
                                               max_frac=0.35)
        # compute exact objective for the SA result
        conv = _full_convolution(sa_seq, fft_len)
        sa_val = 2.0 * n * conv.max()
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[SA] improved objective to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) safety & return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq /= best_seq.sum()            # ensure sum ≈ 1

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
