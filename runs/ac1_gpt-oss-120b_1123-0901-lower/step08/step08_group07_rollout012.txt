"""
Propose a step‑function with a low value of
    F(a)=2·n·max(a∗a)/(sum(a)²)

The algorithm mixes three ideas:

1.  **Monotone anti‑pair shape** – we keep the values sorted in
    decreasing order and place the largest half on the left,
    the smallest half (reversed) on the right.  For a given multiset
    this arrangement is provably optimal for the centre convolution.

2.  **Linear‑programming improvement** – with a fixed sequence `a`
    we maximise the total mass of a new sequence `g` under the
    linear constraints `a⋆g ≤ max(a⋆a)`.  This is a small LP
    (`≈2·n` constraints, `n` variables) solved with SciPy’s HiGHS.
    Mixing `a` and `g` (and renormalising) strictly reduces the
    objective, exactly as in the “alternating optimisation” described
    in the statement.

3.  **Fine‑grained continuous optimisation** – after each LP step we
    run a short Adam phase on a smooth–max (log‑sum‑exp) surrogate,
    followed by a few exact‑max sub‑gradient steps.  Finally the best
    sequence is interpolated to a larger grid (4096 points) and refined
    once more.

The whole routine respects the 1000 s budget, prints progress, and
returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports (LP solver)
try:
    import scipy.optimize as _opt
    import scipy.sparse as _sp
except Exception:  # pragma: no cover
    _opt = _sp = None

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact (no‑security‑checks) objective ----------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a)/(sum(a)²)  –  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple shape generators ------------------------------------------------
def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{‑p}, p>0."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest half on the left, smallest half (reversed) on the right."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]                     # biggest half
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    a[-half:] = sorted_vals[-half:][::-1]             # smallest half reversed
    return a


# ----------------------------------------------------------------------
# Adam on the smooth‑max surrogate --------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """Adam on log‑sum‑exp (smooth max) of the convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact value (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient phase ------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Linear‑programming step -------------------------------------------------
def _lp_max_sum(a: np.ndarray, M: float, time_limit: float) -> np.ndarray | None:
    """
    Solve   max  sum(g)   s.t.   a ⋆ g ≤ M,   g ≥ 0.
    Returns the optimal g (normalised to sum = 1) or ``None`` on failure.
    """
    if _opt is None:                     # SciPy not available
        return None

    n = a.shape[0]
    m = 2 * n - 1

    # Build sparse matrix for the constraints a⋆g ≤ M.
    # Row k corresponds to shift k, column j to g_j.
    # (a⋆g)[k] = Σ_i a_i * g_{k‑i}  with the convention g_{*}=0 out of range.
    col_arr = np.arange(n, dtype=np.int32)
    rows = np.empty(n * n, dtype=np.int32)
    cols = np.empty(n * n, dtype=np.int32)
    data = np.empty(n * n, dtype=np.float64)
    pos = 0
    for i in range(n):
        rows[pos:pos + n] = i + col_arr          # k = i + j
        cols[pos:pos + n] = col_arr              # j
        data[pos:pos + n] = a[i]                 # coefficient a_i
        pos += n

    A_ub = _sp.csr_matrix((data, (rows, cols)), shape=(m, n))
    b_ub = np.full(m, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)            # minimise –sum(g)  ⇒ maximise sum(g)

    # HiGHS respects a time limit via the 'time_limit' option (seconds).
    # If the SciPy version does not support it we simply rely on the outer timer.
    opts = {"presolve": True}
    if hasattr(_opt, "LinprogOptions"):
        opts["time_limit"] = max(0.0, time_limit)

    try:
        res = _opt.linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=[(0, None)] * n,
            method="highs",
            options=opts,
        )
    except Exception as exc:                           # pragma: no cover
        print(f"LP exception: {exc}")
        return None

    if not res.success:
        # occasionally the LP may be infeasible with a too‑small M; fall back to a tiny step
        print("LP failed – returning None")
        return None

    g = res.x
    s = g.sum()
    if s < 1e-12:
        return None
    return g / s

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises

        F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine mixes an LP‑based improvement with Adam / sub‑gradient
    refinements and finally interpolates to a finer grid (4096 points).
    It respects the supplied time budget and returns the best sequence
    discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # 0.5 s safety margin

    ######################################################################
    # 1️⃣  Small‑scale LP / mixing stage
    ######################################################################
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # start from a power‑law, then enforce anti‑pair ordering
    a = _powerlaw_candidate(n_small, p=0.6)
    a = _anti_pair_arrangement(a)
    a = a / a.sum()

    best_seq = a.copy()
    best_val = _objective(a)
    print(f"[Init] n={n_small}  value={best_val:.6f}")

    # schedule – we keep roughly a third of the total budget for the LP loop
    lp_budget = (deadline - time.time()) * 0.35
    max_lp_iters = 6                # we only need a handful of LP solves
    mix_factor = 0.30               # how strongly to mix the LP solution

    for it_lp in range(max_lp_iters):
        if time.time() > deadline - 30:        # reserve time for final up‑sampling
            break

        # current bound M = max(a⋆a)
        conv = _full_convolution(a, conv_fft_len_small)
        M = conv.max()

        # solve LP, give it at most a few seconds
        lp_time_left = max(0.0, deadline - time.time() - 5.0)
        g = _lp_max_sum(a, M, time_limit=lp_time_left)
        if g is None:
            # LP failed – fall back to a tiny random perturbation
            g = np.random.dirichlet(np.ones(n_small))
            g = g / g.sum()

        # convex mixing, then re‑enforce anti‑pair ordering
        a = (1 - mix_factor) * a + mix_factor * g
        a = np.sort(a)[::-1]                     # descending
        a = _anti_pair_arrangement(a)
        a = a / a.sum()

        val = _objective(a)
        print(f"[LP‑mix {it_lp}] value={val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"   *** new best {best_val:.6f}")

        # a short Adam run to clean up the freshly mixed vector
        sub_dead = time.time() + min(5.0, deadline - time.time() - 5.0)
        a_ad, val_ad = _adam_optimize(
            a,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            sub_dead,
            max_iters=30_000,
            lr=0.05,
            lam_start=1.0,
            lam_factor=1.08,
        )
        a = a_ad
        val = val_ad
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"   >>> Adam refined best {best_val:.6f}")

    ######################################################################
    # 2️⃣  Upsample to a larger grid and final continuous refinement
    ######################################################################
    if time.time() < deadline - 5.0:
        n_big = 4096
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation from the small grid to the big one
        xs = np.linspace(0.0, 1.0, n_small, endpoint=False)
        xb = np.linspace(0.0, 1.0, n_big,   endpoint=False)
        a_big = np.interp(xb, xs, best_seq)
        a_big = np.clip(a_big, 0.0, None)
        a_big = a_big / a_big.sum()

        # longer Adam phase (≈30 % of the remaining budget)
        remaining = deadline - time.time()
        adam_dead = time.time() + remaining * 0.30
        a_big, val_big = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_dead,
            max_iters=200_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big‑Adam] improved to {best_val:.6f}")

        # sub‑gradient refinement on the big grid (remaining time)
        sub_dead = deadline
        a_big, val_big = _subgradient_optimize(
            best_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead,
            max_iters=300_000,
            eta0=0.6,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
            print(f"[Big‑Sub] improved to {best_val:.6f}")

    ######################################################################
    # 3️⃣  Final safety clamp and return
    ######################################################################
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
