def project_simplex(v):
    # Returns projection of v onto probability simplex.
    # v may already sum to >0; algorithm:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv/ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    if a.sum() == 0:

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=300_000):
    eps=1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr=0.04
    lam=0.01
    lam_factor=1.03
    best_a = a0.copy()
    best_val = float('inf')
    it=0
    while time.time()<deadline and it<max_iters:
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it%10000==0:
                print(f"[Adam it={it}] new best {val:.6f}")
        # compute smooth max weights
        w = np.exp(lam*(conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # compute gradient in log-space
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam updates
        m = beta1 * m + (1- beta1) * grad_log
        v = beta2 * v + (1- beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase lambda
        lam = min(lam * lam_factor, 1e8)

        # optional decay of lr
        if it % 50000 == 0 and it>0:
            lr *= 0.92

        it +=1
    return best_a, best_val
```

"""
Propose a step‑function that minimizes

      F(a) = 2·n·max(a∗a) / (sum(a)²)

using a multi‑stage search:

1.  Build a diverse pool of **parametric** candidates (edge‑exponential,
    edge‑polynomial, inverse‑product, double‑Gaussian, logistic, random
    Dirichlet).  All candidates are normalised to sum = 1 and evaluated
    with a fast FFT‑based convolution.  The best of this cheap sweep
    seeds the refinement.

2.  Refine the current best with three complementary optimisers, each
    respecting the simplex constraint:
    * Adam on a smooth‑max (log‑sum‑exp) surrogate – slow temperature increase.
    * Projected‑gradient descent on the exact max (sub‑gradient) using an
      explicit Euclidean projection onto the probability simplex.
    * A cheap random hill‑climber that moves a random fraction of mass
      between two indices and keeps improving moves.

3.  The loop runs until the 1000 s budget expires, constantly keeping the
    best feasible sequence seen so far.

All heavy convolution / cross‑correlation work uses FFT (O(n log n)).
The final sequence is clipped to the evaluator’s admissible range
[0, 1000] and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) :
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1 }.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _edge_exp(n: int, gamma: float) -> np.ndarray:
    """Mass concentrated at the two edges: exp(-γ·i) + exp(-γ·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_poly(n: int, alpha: float) -> np.ndarray:
    """Power‑law edge family: (i·(n‑i+1))^α , α<0 concentrates at ends."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = np.where(prod > 0, prod ** alpha, 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    shape /= shape.sum()
    return shape


def _inv_product(n: int, beta: float) -> np.ndarray:
    """Inverse product shape: 1/[(i+1)(n‑i)]^β   (β>0 → edge‑heavy)."""
    i = np.arange(n, dtype=np.float64)
    prod = (i + 1.0) * (n - i)
    shape = prod ** (-beta)
    shape /= shape.sum()
    return shape


def _double_gauss(n: int, sigma: float) -> np.ndarray:
    """Two narrow Gaussians centred at the boundaries."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(
        -0.5 * ((i - (n - 1)) / (sigma * n)) ** 2
    )
    shape /= shape.sum()
    return shape


def _logistic(n: int, s: float) -> np.ndarray:
    """Symmetric logistic (sigmoid) – edge‑heavy for small s."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 / (1.0 + np.exp((i - n / 2.0) / s))
    shape = base + base[::-1]
    shape -= shape.min()          # ensure zero at the centre
    if shape.sum() == 0:
        return np.full(n, 1.0 / n)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp (smooth‑max) surrogate.
    Returns the best vector (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                # a bit smaller than the original
    lam = 0.01                               # start temperature (softer)
    lam_factor = 1.03                        # slower annealing

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ------------------------------------------------------------------
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum is 1 by construction)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.92

        # small random shake when close to deadline
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    # convert back to probability simplex (numerical drift)
    best_a = _project_simplex(best_a)
    best_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()
    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max projected gradient descent on the simplex.
    Uses the sub‑gradient of the true max (indicator on the active set).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of the (non‑smooth) objective
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size – diminishing like 1/√t
        eta = eta0 / np.sqrt(it + 1)

        # Euclidean gradient step then simplex projection
        a = a - eta * grad
        a = _project_simplex(a)

        it += 1

    # final exact value
    best_val = 2.0 * n * _full_convolution(best_a, conv_fft_len).max()
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of the mass from i→j
    and keep the move if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 30 % of the source mass (randomised)
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Generates a pool of parametrised step‑functions,
    picks the best cheap candidate, and then refines it with several
    optimiser phases until the time budget expires.
    Returns a Python list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start ---------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # 1.  Build a modest but diverse pool of *parametric* candidates.
    pool = []

    if warm is not None:
        pool.append(warm.copy())

    # Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # Edge‑exponential
    for gamma in np.logspace(-3, 0.5, 30):
        pool.append(_edge_exp(n, gamma))

    # Edge‑polynomial (negative power)
    for alpha in np.linspace(-3.0, -0.1, 30):
        pool.append(_edge_poly(n, alpha))

    # Inverse‑product family
    for beta in np.linspace(0.2, 2.0, 25):
        pool.append(_inv_product(n, beta))

    # Double‑Gaussian
    for sigma in np.linspace(0.02, 0.4, 20):
        pool.append(_double_gauss(n, sigma))

    # Logistic (sigmoid) edge‑heavy
    for s in np.linspace(0.01, 0.5, 15):
        pool.append(_logistic(n, s))

    # Random Dirichlet draws (different concentrations)
    for _ in range(20):
        alpha = np.random.rand() * 5.0 + 0.1   # α ∈ [0.1,5.1]
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # 2.  Quick sweep: evaluate all candidates with the exact objective
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        total = a.sum()
        if total < 1e-12:
            continue
        a = a / total
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] candidate {idx:03d} value {val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # 3.  Refinement phases – allocate the remaining budget.
    #       a) Adam on smooth‑max
    #       b) Projected‑gradient descent on exact max
    #       c) Random perturbations + short Adam runs
    #       d) Cheap local random hill‑climber
    #
    # The loop repeats the cycle while time permits; each stage gets a
    # slice of the remaining budget.
    while time.time() < deadline - 5.0:
        remaining = deadline - time.time()

        # ---- Adam (smooth max) ---------------------------------------
        sub_deadline = time.time() + min(remaining * 0.35, 120.0)
        best_seq, best_val = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=250_000
        )
        print(f"[Stage Adam] best value {best_val:.6f}")

        if time.time() >= deadline - 5.0:
            break

        # ---- Projected gradient descent (exact max) ------------------
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.30, 110.0)
        best_seq, best_val = _pgd_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=250_000
        )
        print(f"[Stage PGD] best value {best_val:.6f}")

        if time.time() >= deadline - 5.0:
            break

        # ---- Random perturbation + short Adam -----------------------
        remaining = deadline - time.time()
        # perturb logits (soft‑max representation)
        logits = np.log(best_seq + 1e-12)
        logits += np.random.randn(n) * 0.06
        pert = np.exp(logits - np.max(logits))
        pert = pert / pert.sum()
        sub_deadline = time.time() + min(remaining * 0.20, 80.0)
        best_seq, best_val = _adam_optimize(
            pert, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=150_000
        )
        print(f"[Stage Perturb] best value {best_val:.6f}")

        if time.time() >= deadline - 5.0:
            break

        # ---- Very cheap local random search -------------------------
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.15, 60.0)
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, sub_deadline, max_moves=60_000
        )
        print(f"[Stage Local] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4.  Return a valid, clipped sequence.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
