        nonlocal best_seq, best_val
        # ensure nonnegative and sum>0.01
        if a.min()<0:
            a = np.maximum(a,0)

        if s < 1e-3:

        a = a / s
        val = _objective(a)
        if val < best_val:

            best_seq = a.copy()

        return val

    warm = _warm_start(n)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, lam_schedule=(5,15,40,120,400,800), steps_per_lam=2000, lr=0.1):
    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = a0.copy()
    best_val = _objective(best_a)
    for lam in lam_schedule:
        for step in range(steps_per_lam):
            if time.time() >= deadline:
                break
            surrogate, grad = _lbfgs_obj(logits, n, conv_fft_len, corr_fft_len, lam)  # we can reuse this function for gradient
            t += 1
            m = 0.9 * m + 0.1 * grad
            v = 0.999 * v + 0.001 * (grad**2)
            m_hat = m / (1 - 0.9**t)
            v_hat = v / (1 - 0.999**t)
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # maybe clip logits to reasonable range
            if t % 200 == 0:
                a = np.exp(logits - np.max(logits))
                a = a / a.sum()
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam λ={lam} step={step}] new best {val:.6f}")
    return best_a, best_val

def _interleaved_arrangement(vals):
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n, dtype=np.float64)
    left = 0
    right = n-1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a

"""
Improved step‑function search for the evaluation function

F(a) = 2·n·max(a*a) / (sum(a)²)

The algorithm consists of

1️⃣  Warm‑starts – a large random Dirichlet pool (wide α‑range) together
    with two deterministic orderings:
    • anti‑pair (the “large opposite small” trick)
    • interleaved (largest, smallest, second‑largest, …)
2️⃣  Smooth‑max L‑BFGS refinement – minimise a log‑sum‑exp surrogate of
    max(a*a) while increasing the temperature λ; a few extra λ‑levels
    (up to 400) give a tighter approximation.
3️⃣  Adam optimisation of the same surrogate – a few hundred Adam steps
    after L‑BFGS provide a different trajectory that can escape shallow
    minima.
4️⃣  An optional local stochastic hill‑climb (swap / fractional transfer)
    if time permits.
5️⃣  (If time remains) upscale to 8192 points via linear interpolation and
    repeat the L‑BFGS + Adam refinement.

All stages respect the global 1000 s budget and keep the best sequence
found so far.  The returned list is clipped to the limits enforced by
``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling ----------------------------------------------------
def _warm_start(n: int):
    """Return a stored sequence (global ``height_sequence_1``) if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    return shape / shape.sum()


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    return shape / shape.sum()


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _interleaved_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑first, then smallest‑first, alternating from the ends."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    for i, v in enumerate(s):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _anti_pair_arrangement(vals)


def _interleaved_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _interleaved_arrangement(vals)


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS & Adam) -------------------------
def _smoothmax_obj(logits: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, lam: float):
    """Return surrogate value and gradient w.r.t. logits."""
    # soft‑max to obtain a (guaranteed non‑negative, sum‑1)
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp around the max
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate -----------------------
def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 400.0),
                    max_iter_stage: int = 500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, raising λ progressively."""
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # Remaining time caps the number of iterations
        remaining = deadline - time.time()
        stage_iters = min(max_iter_stage, int(remaining * 20))  # generous bound
        res = _opt.minimize(_smoothmax_obj, logits,
                            args=(n, conv_fft_len, corr_fft_len, lam),
                            method='L-BFGS-B', jac=True,
                            options={'maxiter': stage_iters, 'disp': False})
        logits = res.x
        # evaluate the exact objective for the current a
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimisation (smooth‑max) ----------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   lam_schedule=(5.0, 15.0, 40.0, 120.0, 400.0, 800.0),
                   steps_per_lam: int = 1000,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999) -> tuple[np.ndarray, float]:
    """Simple Adam optimiser on the smooth‑max surrogate."""
    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = a0.copy()
    best_val = _objective(best_a)

    for lam in lam_schedule:
        for _ in range(steps_per_lam):
            if time.time() >= deadline:
                break
            surrogate, grad = _smoothmax_obj(logits, n, conv_fft_len, corr_fft_len, lam)
            t += 1
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + 1e-12)

            # occasional exact check
            if t % 250 == 0:
                mx = np.max(logits)
                exp_l = np.exp(logits - mx)
                a = exp_l / exp_l.sum()
                val = _objective(a)
                if val < best_val:
                    best_val, best_a = val, a.copy()
                    print(f"[Adam λ={lam:.1f} it={t}] new best {val:.6f}")
        if time.time() >= deadline:
            break
    return best_a, best_val

# Local stochastic hill‑climbing -----------------------------------------
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """Swap‑or‑transfer moves with a simple annealing acceptance rule."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                    # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up‑to‑80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a)=2·n·max(a∗a)/(sum(a)²)

    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    def _register(a: np.ndarray):
        """Normalise, evaluate and keep the best."""
        nonlocal best_seq, best_val
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        val = _objective(a)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Update] new best {val:.6f}")

    # --------------------------------------------------------------
    # 0) Warm‑start (if provided by harness)
    warm = _warm_start(n)
    if warm is not None:
        _register(warm)

    # --------------------------------------------------------------
    # 1) Deterministic starters
    starter_fns = [
        _uniform_candidate,
        _triangle_candidate,
        lambda n: _gaussian_edges_candidate(n, 0.2),
        lambda n: _gaussian_edges_candidate(n, 0.5),
        lambda n: _gaussian_edges_candidate(n, 0.8),
        lambda n: _gaussian_edges_candidate(n, 1.2),
        lambda n: _gaussian_edges_candidate(n, 2.0),
    ]
    for fn in starter_fns:
        _register(fn(n))

    # --------------------------------------------------------------

    n_samples = 5000
    print(f"Generating {n_samples} random candidates …")
    for _ in range(n_samples):
        # α drawn from a broad log‑uniform range
        log_alpha = np.random.uniform(-2.5, 2.0)   # α ∈ [~0.003, ~100]
        alpha = 10.0 ** log_alpha

        cand1 = _anti_pair_arrangement(vals)
        _register(cand1)

    # --------------------------------------------------------------

    if time.time() < deadline - 150:

        seq_lb, val_lb = _lbfgs_optimize(start, n, conv_fft_len,
                                         corr_fft_len, deadline,
                                         lam_schedule=(5, 15, 40, 120, 400),
                                         max_iter_stage=500)
        _register(seq_lb)

    # --------------------------------------------------------------
    # 4) Adam fine‑tuning on the smooth‑max surrogate
    if time.time() < deadline - 80:

        seq_adam, val_adam = _adam_optimize(start, n, conv_fft_len,

    if time.time() < deadline - 30:

    # --------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192) -------------------
    if time.time() < deadline - 15:

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        a_big = np.maximum(a_big, 0.0)

        # quick L‑BFGS on the larger grid
        seq_big, val_big = _lbfgs_optimize(a_big, n_big,
