def alt_end_arrangement(vals):
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1] # descending
    res = np.empty(n, dtype=vals.dtype)
    left = 0
    right = n-1
    for i, v in enumerate(sorted_vals):
        if i % 2 == 0:
            res[left] = v; left += 1
        else:
            res[right] = v; right -=1
    return res

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                  lam_schedule=None, iters_per_stage=3000, lr=0.5,
                  beta1=0.9, beta2=0.999, eps=1e-8):
    if lam_schedule is None:
        lam_schedule = [2.0,5.0,10.0,20.0,40.0,80.0,150.0,300.0]
    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_a = a0.copy()
    best_val = _objective(a0)
    t_global = 0  # global step count for bias correction

    for lam in lam_schedule:
        # number of iterations for this lambda might be limited but also can break early if time.
        for _ in range(iters_per_stage):
            if time.time() >= deadline:
                break
            # compute a
            mx_log = np.max(logits)
            exp_l = np.exp(logits - mx_log)
            a = exp_l / exp_l.sum()
            # compute convolution
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            # compute w probabilities
            if lam < 100.0:
                # safe compute using max_c shift to avoid overflow
                w_exp = np.exp(lam * (conv - max_c))
                w_sum = w_exp.sum()
                # avoid division by zero
                if w_sum == 0.0:
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0
                else:
                    w = w_exp / w_sum
            else:
                # use hard max as subgradient
                w = np.zeros_like(conv)
                w[np.isclose(conv, max_c, atol=1e-12)] = 1.0 / np.sum(np.isclose(conv, max_c, atol=1e-12))
            # gradient w.r.t a
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            # chain rule to logits
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            # Adam update
            t_global += 1
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log ** 2)
            m_hat = m / (1 - beta1 ** t_global)
            v_hat = v / (1 - beta2 ** t_global)
            # update logits
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)
            # ensure logits not too large (prevent overflow)
            # optional: clip logits to reasonable range
            # compute objective periodically
            if t_global % 500 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam λ={lam:.1f}] new best {val:.6f}")
        # check time
        if time.time() >= deadline:
            break
    # final recompute a from logits
    mx_log = np.max(logits)
    exp_l = np.exp(logits - mx_log)
    a = exp_l / exp_l.sum()
    val = _objective(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()
    return best_a, best_val

"""
Propose a step‑function (non‑negative list) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The strategy consists of three stages:

1. **Broad warm‑start generation** – many simple families (uniform,
   power‑law, exponential, Beta‑shaped, Dirichlet samples with several
   deterministic rearrangements) are evaluated very quickly using FFTs.
   The best shape found is kept.

2. **Adam optimisation of a smooth‑max surrogate** – we minimise a
   differentiable approximation of the max‑convolution,
   `max_c + (1/λ)·log Σ exp(λ·(conv‑max_c))`,
   using the Adam optimiser on the logits of a soft‑max (which guarantees
   a simplex‑constrained non‑negative vector.  The temperature λ is increased
   in stages (2 → 300) so the surrogate approaches the true max.

3. **Final projected sub‑gradient polishing** – a few hundred thousand
   additive sub‑gradient steps with Euclidean projection onto the simplex
   clean up any remaining slack.

All work respects the global 1000 s budget.  The returned sequence is
clipped to the limits accepted by `evaluate_sequence`.  The algorithm is
pure NumPy (optional SciPy for a tiny L‑BFGS refinement) and therefore
portable.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no smoothing) ----------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) -----------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling ----------------------------------------------------
def _warm_start(n: int):
    """Return a stored global sequence (if any) resized to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()

# Simple deterministic candidates -----------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape

# Parameterised families --------------------------------------------------
def _beta_decreasing_candidate(n: int, p: float) -> np.ndarray:
    """f(i) ∝ (1 - i/(n‑1))**p, a decreasing Beta‑like shape."""
    i = np.arange(n, dtype=np.float64)
    shape = (1.0 - i / (n - 1)) ** p
    shape /= shape.sum()
    return shape


def _exponential_candidate(n: int, r: float) -> np.ndarray:
    """f(i) ∝ exp(-r·i) (geometric decay)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-r * i)
    shape /= shape.sum()
    return shape


def _power_law_candidate(n: int, p: float) -> np.ndarray:
    """f(i) ∝ (i+1)⁻ᵖ (heavy‑tail decay)."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + 1.0) ** (-p)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Rearrangement helpers ---------------------------------------------------
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest half at the start, smallest half reversed at the end."""
    n = len(vals)
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=vals.dtype)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _alternating_ends_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place sorted values alternately at left‑most and right‑most positions."""
    s = np.sort(vals)[::-1]                 # descending
    n = len(s)
    out = np.empty(n, dtype=vals.dtype)
    left, right = 0, n - 1
    for i, v in enumerate(s):
        if i % 2 == 0:
            out[left] = v
            left += 1
        else:
            out[right] = v
            right -= 1
    return out


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate ---------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   lam_schedule=None,
                   iters_per_stage: int = 3000,
                   lr: float = 0.5,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate:
        S_λ(a) = max(conv) + (1/λ)·log Σ exp(λ·(conv‑max(conv))).
    The parameters a are represented as soft‑max logits to keep a on the simplex.
    """
    if lam_schedule is None:
        lam_schedule = [2.0, 5.0, 10.0, 20.0, 40.0, 80.0, 150.0, 300.0]

    logits = np.log(a0 + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_a = a0.copy()
    best_val = _objective(best_a)
    global_step = 0

    for lam in lam_schedule:
        for _ in range(iters_per_stage):
            if time.time() >= deadline:
                break

            # ---- Current probabilities ----
            mx = np.max(logits)
            exp_l = np.exp(logits - mx)
            a = exp_l / exp_l.sum()

            # ---- Convolution and soft‑max weights ----
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()

            if lam < 100.0:               # smooth regime
                w_exp = np.exp(lam * (conv - max_c))
                w_sum = w_exp.sum()
                if w_sum == 0.0:
                    w = np.zeros_like(conv)
                    w[np.argmax(conv)] = 1.0
                else:
                    w = w_exp / w_sum
            else:                         # hard max (sub‑gradient)
                active = np.isclose(conv, max_c, atol=1e-12)
                cnt = active.sum()
                w = np.zeros_like(conv)
                if cnt > 0:
                    w[active] = 1.0 / cnt

            # ---- Gradient w.r.t. a ----
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # ---- Chain rule to logits ----
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # ---- Adam update ----
            global_step += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** global_step)
            v_hat = v / (1.0 - beta2 ** global_step)

            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # ---- Periodic evaluation ----
            if global_step % 500 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam λ={lam:.1f}] new best {val:.6f}")

        if time.time() >= deadline:
            break

    # final vector from latest logits
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()
    val = _objective(a)
    if val < best_val:
        best_val, best_a = val, a.copy()
        print(f"[Adam final] improved to {val:.6f}")

    return best_a, best_val

# Additive projected sub‑gradient optimisation ----------------------------
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)        # length needed for a*a
    corr_fft_len = _next_pow2(3 * n - 2)        # length needed for w★a

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Helper to update the current best
    def _maybe_update(cand: np.ndarray, tag: str):
        nonlocal best_seq, best_val
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[{tag}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 1) Warm‑start from possible harness
    warm = _warm_start(n)
    if warm is not None:
        _maybe_update(warm, "WarmStart")

    # 2) Simple deterministic starters
    _maybe_update(_uniform_candidate(n), "Uniform")
    _maybe_update(_triangle_candidate(n), "Triangle")
    _maybe_update(_gaussian_edges_candidate(n, 0.3), "Gauss0.3")
    _maybe_update(_gaussian_edges_candidate(n, 0.7), "Gauss0.7")

    # 3) Parameterised families
    for p in [0.2, 0.5, 1.0, 2.0, 3.0, 5.0]:
        _maybe_update(_beta_decreasing_candidate(n, p), f"Beta(p={p})")
    for r in [0.05, 0.1, 0.2, 0.4]:
        _maybe_update(_exponential_candidate(n, r), f"Exp(r={r})")
    for p in [0.5, 1.0, 2.0, 3.0, 4.0]:
        _maybe_update(_power_law_candidate(n, p), f"Power(p={p})")

    # 4) Large random Dirichlet pool with three deterministic arrangements
    n_samples = 2500
    print(f"Generating {n_samples} random Dirichlet candidates …")
    for _ in range(n_samples):
        if time.time() >= deadline:
            break
        # concentration α spanning several orders of magnitude
        log_alpha = np.random.uniform(-2.0, 1.5)     # α ∈ [0.01, 31.6]
        alpha = 10.0 ** log_alpha
        vals = np.random.dirichlet(alpha * np.ones(n))

        # pick a deterministic rearrangement
        r = np.random.rand()
        if r < 0.33:
            cand = _anti_pair_arrangement(vals)
            tag = "AntiPair"
        elif r < 0.66:
            cand = _alternating_ends_arrangement(vals)
            tag = "AltEnds"
        else:
            cand = np.sort(vals)[::-1]               # descending order
            tag = "Desc"

        cand = cand / cand.sum()
        _maybe_update(cand, f"Random-{tag}")

    # --------------------------------------------------------------
    # Fallback if nothing was generated (should not happen)
    if best_seq is None:
        best_seq = _uniform_candidate(n)
        best_val = _objective(best_seq)

    # --------------------------------------------------------------
    # 5) Adam optimisation on the smooth‑max surrogate
    adam_deadline = min(deadline - 30.0, time.time() + 280.0)   # roughly 5 min max
    if time.time() < adam_deadline:
        seq_adam, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 6) Final short additive sub‑gradient polishing (if time left)
    sub_deadline = deadline - 5.0
    if time.time() < sub_deadline:
        seq_sub, val_sub = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          sub_deadline,
                                                          max_iters=300_000,
                                                          eta0=0.4)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[AddSubgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 7) Final safeguard: ensure positivity, reasonable scale and clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # tiny safeguard against an (unlikely) near‑zero sum after clipping
    if best_seq.sum() < 0.01:
        best_seq = best_seq + 1e-6
    best_seq = best_seq / best_seq.sum()   # renormalise (objective is homogeneous)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
