    # generate a random vector of length n with normal distribution
    freq = np.random.randn(n) + 1j * np.random.randn(n)

def _mixed_gaussian_candidate(n, k=4):

"""
Improved step‑function search for the minimisation of

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The programme expands the original multi‑stage pipeline:

1.  Warm‑starts from any previously found sequence.
2.  A large pool of analytically‑motivated candidates (Dirichlet,
    anti‑pair, power‑law, U‑shapes, Gaussian edges, random phase,
    logistic, mixed Gaussians, triangles, and many random permutations).
3.  The pool is evaluated quickly (FFT‑based convolution) and the
    best *k* candidates are retained.
4.  Each retained candidate is refined by two gradient‑based stages:
        • Adam optimisation of a smooth‑max (log‑sum‑exp) surrogate.
        • Exact‑max sub‑gradient optimisation using exponentiated
          updates (mirror‑descent on the simplex).
5.  A global stochastic hill‑climber (swap / mass‑transfer moves)
    finishes the search.
6.  If time permits the best solution is linearly interpolated to a
    finer grid (8192 points) and refined once more.

All operations respect the security checks of ``evaluate_sequence``.
The code is fully self‑contained, uses only NumPy and runs within the
given 1000‑second budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, without safety checks) -----------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start ------------------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Simple deterministic / random generators -----------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑opposite‑smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p).  p>0 concentrates mass on the first entries."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    """
    Valley in the centre, larger at the two ends:
    a_i ∝ ( i·(n‑1‑i) + 1 )^{-p}.
    """
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _random_phase_candidate(n: int) -> np.ndarray:
    """
    Random Fourier phases, absolute value taken to enforce non‑negativity.
    """
    # Generate a Hermitian–symmetric spectrum for a real sequence.
    spec = np.random.randn(n) + 1j * np.random.randn(n)
    spec[0] = spec[0].real + 0j
    if n % 2 == 0:
        spec[n // 2] = spec[n // 2].real + 0j
    # Enforce conjugate symmetry.
    for k in range(1, n // 2):
        spec[-k] = np.conj(spec[k])
    a = np.real(np.fft.ifft(spec))
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a


def _logistic_candidate(n: int) -> np.ndarray:
    """A smooth S‑shape controlled by random steepness and shift."""
    i = np.arange(n, dtype=np.float64)
    steep = np.random.uniform(0.5, 5.0)
    shift = np.random.uniform(-0.2, 0.2) * n
    shape = 1.0 / (1.0 + np.exp(steep * (i - n / 2 + shift) / n))
    shape = shape / shape.sum()
    return shape


def _mixed_gaussian_candidate(n: int, k: int = 4) -> np.ndarray:
    """Sum of a few Gaussian bumps with random centres, widths and amplitudes."""
    i = np.arange(n, dtype=np.float64)
    a = np.zeros_like(i)
    for _ in range(k):
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.02 * n, 0.2 * n)
        amp = np.random.rand()
        a += amp * np.exp(-0.5 * ((i - mu) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / s
    return a


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.09,
    lam_start: float = 0.3,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    ``a`` is represented as a soft‑max of logits to keep it on the simplex.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        val_exact = _objective(a, conv_fft_len)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e8)

        # occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / math.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
        - swap two entries OR transfer a fraction of mass.
        - accept if improvement or with probability exp(-Δ/Temp).
        Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap move
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer move
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.6)   # up to 60 % transfer
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < math.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a very diverse initial pool
    candidates = []

    # warm start (if present)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # deterministic shapes
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))

    # random families – we generate many instances to increase diversity
    for _ in range(5):
        candidates.append(_random_dirichlet_candidate(n, alpha=np.random.uniform(0.05, 2.0)))
    for _ in range(5):
        candidates.append(_anti_pair_candidate(n, alpha=np.random.uniform(0.05, 2.0)))
    for _ in range(5):
        candidates.append(_powerlaw_candidate(n, p=np.random.uniform(0.05, 3.0)))
    for _ in range(5):
        candidates.append(_ushape_candidate(n, p=np.random.uniform(0.05, 3.0)))
    for _ in range(5):
        candidates.append(_gaussian_edges_candidate(n, gamma=np.random.uniform(0.1, 2.0)))
    for _ in range(5):
        candidates.append(_random_phase_candidate(n))
    for _ in range(5):
        candidates.append(_logistic_candidate(n))
    for _ in range(5):
        candidates.append(_mixed_gaussian_candidate(n, k=np.random.randint(2, 6)))

    # random permutations of a strong anti‑pair starter (explore layout)
    base_ap = _anti_pair_candidate(n, alpha=0.7)
    for _ in range(5):
        perm = np.random.permutation(n)
        candidates.append(base_ap[perm])

    # ------------------------------------------------------------------
    # 2) Quick evaluation, keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, conv_fft_len)
        evaluated.append((val, cand))
        if idx % 15 == 0:
            print(f"[Init cand {idx:03d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_val, best_seq = evaluated[0][0], evaluated[0][1].copy()
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Refine the top candidates (more time per candidate than before)
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.6 / top_k   # 60 % of the time for refinement

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # Adam‑phase (smooth‑max)
        adam_dead = time.time() + per_candidate_budget * 0.45
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=800_000,
            lr=0.09,
            lam_start=0.3,
            lam_factor=1.02,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {best_val:.6f}")

        # Sub‑gradient phase (exact max)
        sub_dead = time.time() + per_candidate_budget * 0.55
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=800_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Global stochastic hill‑climbing on the current best
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Global local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        big_ad_dead = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_ad_dead,
            max_iters=500_000,
            lr=0.09,
            lam_start=0.3,
            lam_factor=1.02,
        )
        # sub‑gradient refinement on the large grid
        big_sub_dead = deadline
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_dead,
            max_iters=500_000,
            eta0=0.7,
        )
        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, a_big_sub.copy()
            print(f"[Big refine] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
