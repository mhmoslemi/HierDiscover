"""
Step‑function optimizer for the objective

    F(a) = 2·n·max(a∗a) / (∑a)²

The method works on a moderately large grid (default n=8192) and
combines three complementary phases:

1️⃣  A fast *exponentiated‑gradient* sub‑gradient descent that
    directly minimises the exact L∞‑norm of the convolution.
    The iterate stays on the probability simplex, so the
    denominator ∑a stays equal to 1 and the objective simplifies to
    2·n·max(conv).

2️⃣  An intensive *simulated‑annealing* local search that repeatedly
    transfers a random fraction of mass from a source index (chosen
    proportionally to its current weight) to a random target.
    Moves that increase the objective are accepted with a
    temperature‑dependent probability, allowing the algorithm to
    escape plateaus.

3️⃣  An optional *upsampling* step: if time permits the best
    solution is linearly interpolated onto a finer grid (n=16384)
    and refined again with the two phases above.

All phases respect the global time budget (default 1000 s) and the
routine always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the exact sub‑gradient.
    w has length 2·len(a)-1, a has length n.
    Returns a vector of length n (the gradient of Σ_k w_k·(a*a)[k]).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation using exponentiated‑gradient updates.
    The iterate stays on the probability simplex (∑a = 1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set -> uniform weight on all indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # exact sub‑gradient
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule ηₜ = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        # we clip the exponent to avoid overflow/underflow
        upd = -eta * grad
        upd = np.clip(upd, -50.0, 50.0)
        a = a * np.exp(upd)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
    start_temp: float = 0.06,
    end_temp: float = 1e-4,
    max_frac: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search.
    Moves consist of transferring a random fraction of the mass from a
    source index (chosen proportionally to its weight) to a random target.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()

    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_moves:
        # linear temperature schedule
        T = max(end_temp, start_temp * (1.0 - it / float(max_moves)))

        # select source proportional to current mass (so we move mass from big entries)
        src = np.random.choice(n, p=a)
        if a[src] < 1e-15:
            it += 1
            continue

        tgt = np.random.randint(n)
        if tgt == src:
            it += 1
            continue

        # transfer a random fraction of src's mass
        frac = np.random.rand() * max_frac
        delta = a[src] * frac

        a2 = a.copy()
        a2[src] -= delta
        a2[tgt] += delta

        # evaluate
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # metropolis accept
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest but diverse pool of starter vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet with several concentration values
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # exponential edges (mass concentrated at the boundary)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # symmetric power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.4, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double Gaussian bumps (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a small number of equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    # random‑position sparse spikes
    for k in (64, 128, 256):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation onto the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short sub‑gradient on the large grid
        sub_dead_big = min(deadline, time.time() + 8.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=120_000,
            eta0=0.8,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            n = n_big
            conv_fft_len = conv_fft_len_big
            print(f"[Upsample] subgrad best {val_big_sub:.6f}")

        # final annealing on the large grid (if still time)
        if time.time() < deadline - 5:
            a_big_ann, val_big_ann = _anneal_optimize(
                best_seq,
                n,
                conv_fft_len,
                deadline,
                max_moves=120_000,
                start_temp=0.06,
                end_temp=1e-4,
            )
            if val_big_ann < best_val:
                best_val = val_big_ann
                best_seq = a_big_ann.copy()
                print(f"[Upsample] anneal best {val_big_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty, valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure compliance with the evaluator’s safety checks
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
