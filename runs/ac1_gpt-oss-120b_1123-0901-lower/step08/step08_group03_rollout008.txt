"""
Simulated‑annealing optimizer for the max‑convolution objective.

The routine starts from a warm solution (if the global variable
`height_sequence_1` exists) or from a uniform distribution.
It computes the full linear convolution once and then performs a very
large number of random mass‑transfer moves.  Each move transfers a
fraction of the mass from one index to another and the convolution is
updated *exactly* using the algebraic formula

    (a–δ·e_i+δ·e_j)*(a–δ·e_i+δ·e_j)
    = a*a – 2δ·a shifted by i + 2δ·a shifted by j
      + δ²·δ_{2i}+δ²·δ_{2j}–2δ²·δ_{i+j} .

The update costs O(n) and avoids a full FFT at every step.
Moves are accepted with the Metropolis criterion, allowing uphill
steps early on (high temperature) and gradually freezing (temperature
decays to a tiny value).  After the annealing phase a short greedy
hill‑climb cleans up the final solution.  The best sequence found is
always returned before the time budget expires.
"""

import time
import numpy as np

def _move_convolution(conv: np.ndarray, a: np.ndarray,
                      i: int, j: int, delta: float) -> np.ndarray:
    """Return a new convolution vector after moving `delta` mass from i to j."""
    n = a.shape[0]
    conv_new = conv.copy()
    # shift updates
    conv_new[i:i + n] -= 2.0 * delta * a
    conv_new[j:j + n] += 2.0 * delta * a
    # diagonal (self‑product) corrections
    conv_new[2 * i] += delta * delta
    conv_new[2 * j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta
    return conv_new


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The function respects the given time budget and always returns the
    best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (the same size used in earlier runs)
    n = 4096

    # ------------------------------------------------------------------
    # warm start – use the previous best if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.shape[0] > n:
            a = a[:n]
        elif a.shape[0] < n:
            a = np.pad(a, (0, n - a.shape[0]))
        a = np.maximum(a, 0.0)
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # normalise to sum = 1 (the objective is scale‑invariant)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    # ------------------------------------------------------------------
    # initial convolution (exact, using NumPy)
    conv = np.convolve(a, a)                # length 2·n‑1
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max

    best_a = a.copy()
    best_val = cur_val

    # ------------------------------------------------------------------
    # Simulated‑annealing parameters
    T_start = 0.5        # initial temperature
    T_end = 1e-6         # final temperature
    max_move_frac = 0.25  # maximal fraction of source mass moved in one step
    # total time available for the annealing stage
    anneal_time = deadline - time.time()
    if anneal_time <= 0:
        anneal_time = 0.1

    it = 0
    # ------------------------------------------------------------------
    # Main annealing loop
    while time.time() < deadline:
        # ----- pick source (biased towards larger entries) ----- #
        i = np.random.choice(n, p=a)   # always picks a position with a[i] > 0
        ai = a[i]
        if ai <= 1e-16:
            continue

        # ----- pick target (any other index) ----- #
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # ----- propose a mass transfer ----- #
        delta = ai * np.random.rand() * max_move_frac
        if delta <= 0.0:

        # ----- evaluate the move using an incremental convolution update ----- #
        conv_new = _move_convolution(conv, a, i, j, delta)
        new_max = conv_new.max()
        new_val = 2.0 * n * new_max

        elapsed = time.time() - start_time
        frac = min(elapsed / anneal_time, 1.0)
        T = T_start * (1.0 - frac) + T_end * frac

        # ----- Metropolis acceptance ----- #
        if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
            # accept the move
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_val = new_val

        if (it % 50000 == 0) and (np.random.rand() < 0.15):
            i2 = np.random.choice(n, p=a)
            if a[i2] > 1e-12:
                j2 = np.random.randint(n - 1)
                if j2 >= i2:
                    j2 += 1
                # move a large fraction (50‑100 %) of the source mass
                delta_big = a[i2] * (0.5 + 0.5 * np.random.rand())
                conv_test = _move_convolution(conv, a, i2, j2, delta_big)
                val_test = 2.0 * n * conv_test.max()
                if val_test < cur_val or np.random.rand() < np.exp(-(val_test - cur_val) / max(T, 1e-12)):
                    a[i2] -= delta_big
                    a[j2] += delta_big
                    conv = conv_test
                    cur_val = val_test
                    if cur_val < best_val:
                        best_val = cur_val
                        best_a = a.copy()
                        print(f"[SA big move it={it}] new best {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final greedy refinement (always downhill moves)
    a = best_a.copy()
    conv = np.convolve(a, a)
    cur_val = 2.0 * n * conv.max()
    for _ in range(60000):
        i = np.random.choice(n, p=a)
        if a[i] <= 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        delta = a[i] * np.random.rand() * 0.08   # small local tweaks
        conv_new = _move_convolution(conv, a, i, j, delta)
        new_val = 2.0 * n * conv_new.max()
        if new_val < cur_val:
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[Refine] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return the best sequence found (clipped to the evaluator limits)
    best_seq = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
