def _anneal_optimize(a0, n, conv_fft_len, deadline, max_moves=20000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0*n*conv.max()
    best = a.copy()
    best_val = val
    T_start = 0.1
    T_end = 0.001
    move = 0
    while time.time() < deadline and move < max_moves:
        T = max(T_end, T_start * (1 - move / max_moves))
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0*n*conv2.max()
        if val2 < best_val:
            best_val = val2
            best = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        move += 1
    return best, best_val
```

"""
Fast step‑function optimizer.

The goal is to minimise

    F(a) = 2·n·max(a*a) / (∑a)²

which is scale‑invariant.  We therefore work on the probability simplex
(∑a = 1) and minimise the maximum entry of the linear convolution a*a.
The optimiser consists of three stages:

1. **Broad parametric sweep** – generate a few hundred analytically
   motivated shapes (sinⁿ, exponentials, inverse‑powers, Gaussians,
   spikes, Dirichlet draws, …) and keep the best one.

2. **Projected (Euclidean) gradient descent on the exact max‑convolution**
   using the sub‑gradient 2·(w★a) where w is the uniform weight on the
   active set of maximal entries.  The update is a simple
   a ← a – η·grad followed by Euclidean projection onto the simplex.
   Step size η = η₀/√t decays slowly.

3. **Light simulated‑annealing polishing** – random mass transfers
   between two coordinates; uphill moves are accepted with a
   temperature that decays linearly.

All phases respect the global 1000 s time budget and the routine always
returns the best sequence found (as a plain Python list).  The code
contains no closures, uses only NumPy and the provided `evaluate_sequence`
function, and prints occasional progress messages.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# 1. FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Both inputs are real, w has length 2·len(a)-1, a has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    (non‑negative entries that sum to 1).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# 2. Objective (sum of a is kept equal to 1)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(conv) (exact value for ∑a=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# 3. Gradient‑descent optimiser (exact max)
def _gd_optimize(a0: np.ndarray,
                 n: int,
                 conv_fft_len: int,
                 corr_fft_len: int,
                 deadline: float,
                 max_iters: int = 300_000,
                 eta0: float = 0.30) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean gradient descent on the true max‑convolution.
    Returns the best vector seen (sum=1) and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = active.astype(float) / active.sum()
        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size ηₜ = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * grad
        a = _project_simplex(a)               # keep on the simplex

        # evaluate
        cur_val = 2.0 * n * conv.max()
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {best_val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# 4. Light simulated‑annealing polish
def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Random mass transfers with a linear cooling schedule.
    Returns the best vector encountered.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    T_start, T_end = 0.08, 1e-4
    move = 0
    while time.time() < deadline and move < max_moves:
        T = max(T_end, T_start * (1.0 - move / max_moves))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            move += 1
            continue
        j = np.random.randint(n)
        if i == j:
            move += 1
            continue

        # transfer a random fraction (≤30 %) of the source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a, conv = a2, conv2
        else:
            # uphill move with Metropolis probability
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a, conv = a2, conv2

        move += 1
    return best_a, best_val

# 5. Candidate generators (analytic families)
def _generate_candidates(n: int) -> list[np.ndarray]:
    """Return a modest but diverse list of deterministic candidates."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0))

    # linear ramps
    cand.append(np.arange(1.0, n + 1.0))
    cand.append(np.arange(n, 0.0, -1.0))

    # half‑sine with various powers
    x = np.arange(n) + 0.5
    sin_base = np.sin(np.pi * x / n)
    for p in (0.2, 0.5, 0.7, 1.0, 1.5, 2.0, 3.0, 4.0):
        cand.append(sin_base ** p)

    # pure exponentials
    for gamma in np.logspace(-3, -0.3, 5):
        cand.append(np.exp(-gamma * np.arange(n)))

    # double‑Gaussian (peaks at both ends)
    for sigma_factor in (0.02, 0.05, 0.1, 0.2, 0.3):
        sigma = sigma_factor * n
        i = np.arange(n, dtype=float)
        a = np.exp(-0.5 * (i / sigma) ** 2) + np.exp(-0.5 * ((i - (n - 1)) / sigma) ** 2)
        cand.append(a)

    # symmetric inverse‑power (U‑shaped) families
    for beta in (0.3, 0.5, 0.7, 1.0, 1.2, 1.5, 2.0):
        idx = np.arange(1, n + 1, dtype=float)
        a = (idx ** -beta) + ((n - idx + 1) ** -beta)
        cand.append(a)

    # sparse equally‑spaced spikes
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=float)
        if k:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    return cand


def _random_dirichlet_candidate(n: int) -> np.ndarray:
    """Draw a Dirichlet vector with a random concentration."""
    # log‑uniform draw for concentration α ∈ [10⁻³, 10²]
    log_alpha = np.random.uniform(-3, 2)
    alpha = 10.0 ** log_alpha
    return np.random.dirichlet(alpha * np.ones(n))

    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size (fixed for the whole run)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

        warm = np.clip(warm, 0.0, None)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – deterministic sweep
    best_a = None
    best_val = float("inf")

    candidates = _generate_candidates(n)
    if warm is not None:
        candidates.append(warm)

    for raw in candidates:
        if raw.sum() == 0:
            continue
        a = raw.astype(float)
        a = a / a.sum()               # put on the simplex (∑a = 1)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Sweep] new best {best_val:.6f}")

    # ------------------------------------------------------------------

        val = _objective(a, n, conv_fft_len)
        if val < best_val:

    # ------------------------------------------------------------------

    if best_a is not None and time.time() < deadline - 20.0:
        print("[GD] Starting projected gradient descent")

        a_gd, val_gd = _gd_optimize(best_a, n,
                                    conv_fft_len, corr_fft_len,

                                    eta0=0.30)
        if val_gd < best_val:

            print(f"[GD] improved best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – short simulated‑annealing polish
    if best_a is not None and time.time() < deadline - 5.0:

    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Finished – best objective {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
