"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm is a multi‑phase search that respects a 1000 s budget:

1. **Warm start** – reuse a previously found sequence if it exists.
2. **Diverse random families** – many analytic families (Dirichlet,
   exponentials, power‑law, rational, sinusoid, two‑Gaussian mixtures,
   sparse spikes, block‑histograms) are sampled and the best is kept.
3. **Smooth‑max Adam** – minimise a temperature‑smoothed version of the
   max‑convolution using Adam (log‑sum‑exp surrogate).  A long run is
   performed, followed by a few noisy restarts.
4. **Exact‑max exponentiated‑gradient** – sub‑gradient descent on the
   true max‑convolution (mirror‑descent with entropy regularisation).
   The number of iterations is increased until the time budget is almost
   exhausted.
5. **Simulated annealing** – a cheap “move‑mass” hill‑climb that also
   accepts worsening moves with a slowly decaying temperature.  This
   explores the neighbourhood of the best vector found so far.
6. **Final greedy hill‑climb** – only accepting improvements, for a few
   more moves.

All phases keep track of the best vector seen so far and always return
it before the deadline.  The code uses only NumPy (no external LP/MI
solvers) and cheap FFT‑based convolutions, so it fits comfortably into the
time limit while often improving the previous best value (~1.5077) down
towards the target 1.503.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.001,
    lam_factor: float = 1.02,
    noise_sigma: float = 0.025,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # soft‑max projection onto the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient optimisation on the exact‑max objective.
    The step‑size decays as η₀/√t.  Returns the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the max entry
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing mover: transfer a random fraction of mass
    from one coordinate to another; accept worsening moves with a
    temperature that linearly decays to zero.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    start = time.time()
    move = 0
    while move < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            move += 1
            continue
        j = np.random.randint(n)
        if i == j:
            move += 1
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
        else:
            # temperature linearly from 1.0 to 0.0
            T = max(0.0, (deadline - time.time()) / (deadline - start))
            if np.random.rand() < np.exp(-(val2 - best_val) / (T + 1e-12)):
                a = a2
                conv = conv2
        move += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    n = 4096
    fft_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    def _try(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # baseline
    if warm is not None:
        _try(warm)
    else:
        _try(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Diverse random families

    # 1) Dirichlet (full support)
    for _ in range(30):
        a = np.random.dirichlet(np.ones(n))
        _try(a)

    # 2) One‑sided exponential families
    for gamma in (0.15, 0.3, 0.6, 1.2, 2.5):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        a = left / left.sum()
        _try(a)

    # 3) Symmetric power‑law
    for p in (-1.5, -1.0, -0.5, 0.2, 0.5, 1.0):
        idx = np.arange(n, dtype=np.float64) + 1.0
        left = idx ** p
        right = left[::-1]
        a = left + right
        a = np.maximum(a, 0.0)
        a /= a.sum()
        _try(a)

    # 4) Rational shapes a_i = 1/(i+1)^p + 1/(n-i)^p
    for p in (0.3, 0.6, 1.0, 1.5):
        i = np.arange(n, dtype=np.float64) + 1.0
        a = 1.0 / (i ** p) + 1.0 / ((n - i + 1) ** p)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        _try(a)

    # 5) Half‑wave sinusoid (already used before, but include different phases)
    phases = (0.0, 0.25, 0.5)
    for phi in phases:
        a = np.sin(np.pi * (np.arange(n) + 0.5 + phi * n) / n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        _try(a)

    # 6) Two‑Gaussian mixtures (more draws)
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(30):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.12 + 0.03
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        _try(a)

    # 7) Sparse random spikes (support sizes up to 512)
    for k in (64, 128, 256, 512, 1024):
        for _ in range(6):
            idx = np.random.choice(n, size=k, replace=False)
            w = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = w
            _try(a)

    # 8) Block‑histogram: random number of contiguous blocks with random heights
    for _ in range(15):
        nb = np.random.randint(2, 7)          # number of blocks
        cuts = sorted(np.random.choice(range(1, n), nb - 1, replace=False))
        cuts = [0] + cuts + [n]
        a = np.empty(n, dtype=np.float64)
        for i in range(nb):
            start, end = cuts[i], cuts[i + 1]
            height = np.random.rand()
            a[start:end] = height
        a = np.maximum(a, 0.0)
        a /= a.sum()
        _try(a)

    # ------------------------------------------------------------------
    # Adam optimisation (main long run)
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.55 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        fft_len,
        corr_len,
        adam_deadline,
        max_iters=400_000,
        lr=0.07,
        lam_start=0.001,
        lam_factor=1.018,
        noise_sigma=0.028,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Noisy‑restart Adam (several short runs)
    for restart in range(5):
        if time.time() >= deadline - 30:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.4, size=n)
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()
        r_deadline = min(deadline - 5,
                         time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            fft_len,
            corr_len,
            r_deadline,
            max_iters=220_000,
            lr=0.065,
            lam_start=0.001,
            lam_factor=1.015,
            noise_sigma=0.025,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient (exponentiated‑gradient) refinement – longest remaining phase
    if time.time() < deadline - 20:
        sub_deadline = deadline - 10
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            fft_len,
            corr_len,
            sub_deadline,
            max_iters=800_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Simulated‑annealing local search
    if time.time() < deadline - 8:
        sa_deadline = deadline - 4
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            fft_len,
            sa_deadline,
            max_moves=500_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SimAnn] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Final greedy hill‑climb (only accepting improvements)
    if time.time() < deadline - 2:
        # cheap moves: transfer up to 20 % of source mass
        a = best_seq.copy()
        conv = _full_convolution(a, fft_len)
        best_val_hc = 2.0 * n * conv.max() / (a.sum() ** 2)
        for it in range(120_000):
            if time.time() >= deadline - 1:
                break
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.2)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, fft_len)
            val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
            if val2 < best_val_hc:
                best_val_hc = val2
                best_seq = a2.copy()
                a = a2
                if it % 10_000 == 0:
                    print(f"[HillClimb it={it:06d}] improved to {val2:.6f}")

        if best_val_hc < best_val:
            best_val = best_val_hc
            print(f"[HillClimb] final best {best_val_hc:.6f}")

    # ------------------------------------------------------------------
    # final safety
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
