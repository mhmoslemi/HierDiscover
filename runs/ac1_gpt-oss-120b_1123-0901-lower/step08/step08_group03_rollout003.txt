def build_sidonic_set(max_mult=2, target_k=None):
    S=[]
    sum_counts=np.zeros(2*n-1, dtype=np.int16)
    # permutation of indices random
    idx = np.random.permutation(n)
    for i in idx:
        # compute sums with existing S
        if len(S)==0:
            # self sum only
            s_self=2*i
            if sum_counts[s_self] + 1 <= max_mult:
                S.append(i)
                sum_counts[s_self]+=1
                # no other sums
        else:
            sums = i + np.array(S, dtype=np.int32)
            # Check self sum
            s_self = 2*i
            # compute max after increments
            if (sum_counts[sums] + 1).max() > max_mult:
                continue
            if sum_counts[s_self] + 1 > max_mult:
                continue
            # Accept
            S.append(i)
            sum_counts[sums] += 1
            sum_counts[s_self] += 1

    # return S

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start=time.time()
    deadline=start+budget_s-0.5
    n=4096
    conv_fft_len=_next_pow2(2*n-1)
    # warm start optional may be used later

    best_a=None
    best_val=inf

    # Add warm if exists:
    if 'height_sequence_1' in globals():
        ... (like before)
        try candidate.

    # Stage 1: Random Sidon-like sets.
    max_mult=2
    attempts=0
    while time.time() < deadline - 20 and attempts < 2000:
        # build set
        S=build_set(max_mult)
        k=len(S)
        if k < 50: # skip small sets
            continue
        a=np.zeros(n)
        a[S]=1.0/k
        val=_eval_objective(a,n,conv_fft_len)
        if val<best_val:
            best_val=val
            best_a=a.copy()
            print(f"[Sidon mult={max_mult} size={k}] new best {val:.6f}")
        attempts+=1
        # occasionally increase max_mult to 3 if not sufficient
        if attempts%200==0 and max_mult<3:
            max_mult+=1

    # If after this best_val < 1.5 (target), we may skip further steps.
    # Else Continue with gradient refinement.

    # Stage 2: Adam from best
    if best_a is not None and time.time() < deadline - 5:
        remaining=deadline - time.time()
        # allocate half for Adam
        adam_deadline=time.time() + 0.5 * remaining
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            _next_pow2(3*n-2),
            adam_deadline,
            max_iters=200_000,
            lr=0.06,
            lam_start=0.05,
            lam_factor=1.03,
            noise_sigma=0.02,
        )
        if val_adam < best_val:
            best_val=val_adam
            best_a=a_adam.copy()
            print(f"[Adam post-sidon] new best {val_adam:.6f}")

        # Subgradient refinement
        sg_deadline=time.time() + 0.3 * (deadline - time.time())
        a_sg, val_sg = _subgradient_optimize(
            best_a,
            n,
            conv_fft_len,
            _next_pow2(3*n-2),
            sg_deadline,
            max_iters=200_000,
            eta0=0.8,
        )
        if val_sg < best_val:
            best_val=val_sg
            best_a=a_sg.copy()
            print(f"[Subgrad after Adam] new best {val_sg:.6f}")

        # Local random search final
        if time.time() < deadline - 2:
            a_loc, val_loc = _local_random_search(
                best_a,
                n,
                conv_fft_len,
                deadline,
                max_moves=50_000,
            )
            if val_loc < best_val:
                best_val=val_loc
                best_a=a_loc.copy()
                print(f"[Local final] new best {val_loc:.6f}")

    # fallback if no best
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_val = _eval_objective(best_a,n,conv_fft_len)

    # Clip and return list
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f}")
    return best_a.tolist()

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The method consists of three phases

1. **Low‑collision “Sidon‑like” sampling** – we build random sets of
   indices whose pairwise sums collide at most *c* times (c=2 or 3).  With
   uniform weights on such a set the convolution peak is bounded by
   c/k², where k is the set size.  For n=4096 this easily yields
   F<1.5 when k≈110–130.

2. **Smooth‑max Adam optimisation** – the best set from phase‑1 is used
   as a warm‑start for Adam on a softmax‑parameterised simplex.
   The temperature λ grows quickly so that the surrogate quickly
   concentrates on the true maximum.

3. **Exact‑max exponentiated‑gradient refinement + cheap hill‑climb** –
   a final sub‑gradient loop on the true max, followed by a few random
   mass‑transfer moves that only accept improvements.

All phases respect the overall 1000 s budget; the best vector found so
far is always returned.  The code uses only NumPy and is completely
self‑contained.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact (no‑security‑checks) evaluation of the objective."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:                    # should never happen
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _build_low_collision_set(n: int,
                             max_mult: int = 2,
                             target_k: int | None = None) -> np.ndarray:
    """
    Greedy construction of a set S⊂{0,…,n‑1} such that every sum
    i+j (i≤j) appears at most `max_mult` times.
    Returns the index array S (unsorted).
    """
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)
    S = []                     # Python list for incremental growth

    # Random ordering – many tries will give different sizes
    cand = np.random.permutation(n)
    for i in cand:
        if not S:                                   # first element
            if sum_counts[2 * i] < max_mult:
                S.append(i)
                sum_counts[2 * i] += 1
        else:
            cur = np.array(S, dtype=np.int32)
            sums = i + cur                          # vectorised
            # check all future counts
            if (sum_counts[sums] >= max_mult).any():
                continue
            if sum_counts[2 * i] >= max_mult:
                continue
            # accept the index
            S.append(i)
            sum_counts[sums] += 1
            sum_counts[2 * i] += 1

        if target_k is not None and len(S) >= target_k:
            break
    return np.asarray(S, dtype=np.int32)


def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.07,
                   lam_start: float = 0.05,
                   lam_factor: float = 1.03,
                   noise_sigma: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector seen and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                 # soft‑max coordinates on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")

    it = 0
    while it < max_iters and time.time() < deadline:
        # ---- simplex projection (soft‑max) -----------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective --------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights -----------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient w.r.t. a (surrogate) -----------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma,
                                         size=grad_log.shape)

        # ---- Adam update on logits -------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule --------------------------
        lam = min(lam * lam_factor, 1e6)

        # gentle learning‑rate decay every 50k iters
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # renormalise (protect against under/overflow)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 70_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------- problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----------------------------------------------------- warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    def try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Candidate] new best {val:.6f}")

    # ---------------------------------------------------- baseline candidates
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # some simple analytic families (cheap, only for diversity)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # half‑wave sinusoid (the old code used this)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # --------------------------------------------------- low‑collision sampling
    # We target multiplicity 2 first; if we cannot reach decent size we
    # allow multiplicity 3.
    max_mult = 2
    attempts = 0
    while time.time() < deadline - 30 and attempts < 2000:
        S = _build_low_collision_set(n, max_mult=max_mult)
        k = len(S)
        if k < 80:                 # too small to be useful
            attempts += 1
            continue
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / k
        try_candidate(a)

        attempts += 1
        # occasionally try a slightly higher multiplicity if progress stalls
        if attempts % 300 == 0 and max_mult < 3:
            max_mult += 1

    # --------------------------------------------------- Adam refinement (if time)
    if best_seq is not None and time.time() < deadline - 20:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.55 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
            lr=0.06,
            lam_start=0.05,
            lam_factor=1.03,
            noise_sigma=0.02,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam main] improved best {val_adam:.6f}")

    # --------------------------------------------------- Sub‑gradient refinement
    if best_seq is not None and time.time() < deadline - 15:
        sg_deadline = time.time() + 0.30 * (deadline - time.time())
        a_sg, val_sg = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=250_000,

        )

    return best_seq.tolist()
