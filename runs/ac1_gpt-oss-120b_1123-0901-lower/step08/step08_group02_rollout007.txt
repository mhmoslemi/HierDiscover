"""
Linear-programming + local refinement search for minimizing
    F(a)=2·n·max(a∗a)/(sum a)^2

The algorithm:
  1. Build a small pool of diverse seeds (uniform, edge‑exponential,
     random Dirichlet, the previous solution if any).
  2. For each seed run a Frank‑Wolfe–type loop:
        * compute current M = max(a∗a)
        * solve a linear program   maximise 1ᵀg   s.t. g≥0 , (a∗g) ≤ M
        * mix the old vector with the LP solution using a line‑search
          that guarantees a strict decrease of M.
     The loop stops when no improvement is possible or the global time
     budget is exhausted.
  3. The best vector obtained is refined with a cheap exponentiated‑gradient
     sub‑gradient method and a very light random‑move hill‑climber.
  4. The routine always returns the best feasible sequence encountered.

All helper functions are defined at module level; the entry point is
`propose_candidate`.
"""

import time
import numpy as np
import cvxpy as cp
import scipy.sparse as sp

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max(a: np.ndarray) -> float:
    """Return max of the linear convolution a∗a."""
    # using FFT is faster for long vectors
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * n - 1].max()


def _make_toeplitz(f: np.ndarray) -> sp.csc_matrix:
    """
    Build the (2n‑1)×n Toeplitz matrix T_f such that
        (T_f @ g)[k] = (f∗g)[k] .
    """
    n = f.shape[0]
    # each column i contributes the vector f shifted by i positions
    # we can create a sparse matrix with n diagonals, all equal to f.
    data = np.tile(f, (n, 1))          # shape (n, n) – each row = one diagonal
    offsets = np.arange(n)             # 0,1,…,n‑1
    T = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
    return T


def _lp_step(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve the LP   maximise 1ᵀg   s.t.   g≥0 , (f∗g) ≤ M .
    Returns the optimal g (numpy array) or None on failure.
    """
    n = f.shape[0]
    T = _make_toeplitz(f)

    g = cp.Variable(n, nonneg=True)
    # element‑wise bound (broadcast scalar M)
    constraints = [T @ g <= M * np.ones(2 * n - 1)]

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, abstol=1e-9, reltol=1e-9, feastol=1e-9, verbose=False)
    except Exception as e:
        # In case the solver crashes we simply give up on this LP step.
        return None

    if g.value is None:
        return None
    return np.asarray(g.value).flatten()


def _mix_improve(f: np.ndarray, g: np.ndarray, M: float) -> tuple[np.ndarray, float]:
    """
    Find a step size t∈(0,1] such that the mixture
        f_new = (1‑t)·f + t·g
    yields a strictly smaller convolution maximum.
    Returns (f_new, M_new) if successful, otherwise (f, M).
    """
    # try a geometric sequence of step sizes, start relatively aggressive
    for t in (0.5, 0.25, 0.125, 0.0625, 0.03125, 0.015625):
        f_new = (1 - t) * f + t * g
        M_new = _conv_max(f_new)
        if M_new + 1e-15 < M:               # strict improvement
            return f_new, M_new
    return f, M   # no improvement found


def _subgradient_optimize(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple exponentiated‑gradient sub‑gradient method using the exact max.
    Stops when time runs out.
    Returns (best_a, best_val) with best_val = evaluate_sequence(best_a).
    """
    n = a0.shape[0]
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    # pre‑allocate FFT length for fast convolutions
    fft_len = _next_pow2(2 * n - 1)

    while time.time() < deadline:
        # exact convolution via FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set (indices where max is attained, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-13)
        if not active.any():
            # no active constraints – nothing to do
            break

        # gradient of the exact objective w.r.t. a:
        # grad = 2 * (rev(active) * a)  where * denotes convolution
        # we can compute via cross‑correlation
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # cross‑correlation (w ★ a)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, fft_len)
        fr = np.fft.rfft(rev_a, fft_len)
        corr = np.fft.irfft(fw * fr, fft_len)[: 2 * n - 1]
        # extract the middle n entries (corresponding to i‑range)
        start = n - 1
        grad = 2.0 * corr[start : start + n]

        # exponentiated‑gradient update
        it += 1
        eta = 0.8 / np.sqrt(it)  # decreasing step size

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

    return best_a, best_val


def _local_random_search(a0: np.ndarray, deadline: float, max_moves: int = 50_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that repeatedly moves a random fraction
    of mass from one index to another.
    """
    n = a0.shape[0]
    a = a0.copy()
    best_a = a.copy()
    best_val = evaluate_sequence(a.tolist())

    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.25)   # up‑to 25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = evaluate_sequence(a2.tolist())
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        moves += 1

    return best_a, best_val


def _initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modestly diverse pool of starting vectors."""
    pool = []
    # warm start (if available)
    if warm is not None:
        pool.append(warm.copy())
        # a few cyclic shifts (helps to break symmetry)
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # edge‑exponential families
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0):
        left = np.exp(-gamma * np.arange(n))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        pool.append(shape)

    # random Dirichlet with several concentration values
    for alpha in (0.1, 0.3, 0.7, 1.5, 3.0):
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    return pool


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine runs inside the given time budget (≈1000 s) and
    always returns the best sequence found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ---------------------------------------------------------
    # problem size – we deliberately keep n modest (2 k) to allow
    # many LP solves while still being large enough to beat the
    # previous 4 k best.
    n = 2048
    # ---------------------------------------------------------

    # ---------------------------------------------------------
    # warm start from the previous global solution, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    # ---------------------------------------------------------

    pool = _initial_pool(n, warm)

    best_seq = None
    best_val = float("inf")

    # ---------------------------------------------------------
    # Main search loop – iterate over the pool, running the
    # LP‑based Frank‑Wolfe descent, followed by a short
    # sub‑gradient fine‑tuning and a cheap random hill‑climber.
    # ---------------------------------------------------------
    for cid, init in enumerate(pool):
        if time.time() >= deadline:
            break

        f = np.maximum(init, 0.0)
        if f.sum() < 0.01:
            f = np.random.dirichlet(np.ones(n))
        f = f / f.sum()

        # -----------------------------------------------------------------
        # Phase A – LP‑based Frank‑Wolfe style improvement.
        # We limit the number of inner iterations so that we stay
        # inside the global deadline.
        # -----------------------------------------------------------------
        while time.time() < deadline - 5.0:
            M = _conv_max(f)
            g = _lp_step(f, M)
            if g is None or g.sum() < 1e-12:
                break
            # try to move towards g – enforce monotone decrease of M
            f_new, M_new = _mix_improve(f, g, M)
            if M_new >= M - 1e-15:
                # could not make progress, stop the LP phase for this seed
                break
            f = f_new
            val = evaluate_sequence(f.tolist())
            if val < best_val:
                best_val, best_seq = val, f.copy()
                print(f"[LP cand={cid}] new best {val:.6f}")

        # -----------------------------------------------------------------
        # Phase B – cheap exponentiated‑gradient sub‑gradient optimisation.
        # -----------------------------------------------------------------
        sub_deadline = min(deadline - 1.0, time.time() + 30.0)
        f_sub, val_sub = _subgradient_optimize(f, sub_deadline)
        if val_sub < best_val:
            best_val, best_seq = val_sub, f_sub.copy()
            print(f"[Subgrad cand={cid}] improved to {val_sub:.6f}")

        # -----------------------------------------------------------------
        # Phase C – random local moves (tiny hill‑climber)
        # -----------------------------------------------------------------
        loc_deadline = min(deadline - 0.5, time.time() + 20.0)
        f_loc, val_loc = _local_random_search(best_seq if best_seq is not None else f, loc_deadline)
        if val_loc < best_val:
            best_val, best_seq = val_loc, f_loc.copy()
            print(f"[Local cand={cid}] improved to {val_loc:.6f}")

    # ---------------------------------------------------------
    # Fallback – guarantee a non‑empty return.
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of three stages:

1. **Diverse seeds** – a small pool of start vectors (uniform, edge‑exponential,
   random Dirichlet draws and the previous solution if it exists).

            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.25)   # up‑to 25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = evaluate_sequence(a2.tolist())
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        moves += 1

    return best_a, best_val


def _initial_pool(n, warm):
    """Create a modestly diverse pool of start vectors."""
    pool = []

    # warm start (if we have a previous solution)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # edge‑exponential families (mass at the ends)
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0):
        left = np.exp(-gamma * np.arange(n))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        pool.append(shape)

    # random Dirichlet draws with various concentrations
    for alpha in (0.1, 0.3, 0.7, 1.5, 3.0):
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    return pool


def propose_candidate(seed=0, budget_s=1000):
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget (≈1000 s) and always returns
    the best sequence (as a Python list) found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # -------------------------------------------------------------
    # problem size – a moderate length gives a good trade‑off
    # between expressive power and the ability to solve many LPs.
    # -------------------------------------------------------------
    n = 2048

    # -------------------------------------------------------------
    # warm start from the previous global solution, if any
    # -------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    pool = _initial_pool(n, warm)

    best_seq = None
    best_val = float("inf")

    # -------------------------------------------------------------
    # Main search loop over the pool of seeds
    # -------------------------------------------------------------
    for cid, init in enumerate(pool):
        if time.time() >= deadline:
            break

        f = np.maximum(init, 0.0)
        if f.sum() < 0.01:
            f = np.random.dirichlet(np.ones(n))
        f = f / f.sum()

        # -----------------------------------------------------------------
        # Phase A – LP‑based Frank‑Wolfe descent
        # -----------------------------------------------------------------
        while time.time() < deadline - 5.0:
            M = _conv_max(f)
            g = _lp_step(f, M)
            if g is None or g.sum() < 1e-12:
                break
            f_new, M_new = _mix_improve(f, g, M)
            if M_new >= M - 1e-15:          # no improvement
                break
            f = f_new
            val = evaluate_sequence(f.tolist())
            if val < best_val:
                best_val, best_seq = val, f.copy()
                print(f"[LP cand={cid}] new best {val:.6f}")

        # -----------------------------------------------------------------
        # Phase B – exponentiated‑gradient sub‑gradient refinement
        # -----------------------------------------------------------------
        sub_deadline = min(deadline - 1.0, time.time() + 30.0)
        f_sub, val_sub = _subgradient_optimize(f, sub_deadline)
        if val_sub < best_val:
            best_val, best_seq = val_sub, f_sub.copy()
            print(f"[Subgrad cand={cid}] improved to {val_sub:.6f}")

        # -----------------------------------------------------------------
        # Phase C – cheap random hill‑climber
        # -----------------------------------------------------------------
        loc_deadline = min(deadline - 0.5, time.time() + 20.0)
        f_loc, val_loc = _local_random_search(best_seq if best_seq is not None else f,
                                              loc_deadline)
        if val_loc < best_val:
            best_val, best_seq = val_loc, f_loc.copy()
            print(f"[Local cand={cid}] improved to {val_loc:.6f}")

    # -------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    # -------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
