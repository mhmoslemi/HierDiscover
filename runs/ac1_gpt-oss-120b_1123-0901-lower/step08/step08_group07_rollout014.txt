"""
Propose a step‑function with a lower evaluation score.

Algorithm Overview
------------------
1. **Warm‑start** – use the global `height_sequence_1` if present;
   otherwise start from a diverse pool of analytic shapes (Dirichlet,
   anti‑pair, power‑law, U‑shape, Gaussian edges, triangle).

2. **Coarse optimisation (n=4096)** – evaluate all warm‑starts,
   keep the three best candidates and improve each of them with
   Adam on a smooth‑max surrogate followed by an exact‑max
   exponentiated‑gradient (sub‑gradient) phase.

3. **Upsample** – linearly interpolate the current best vector to a
   finer grid (n=8192) and repeat the Adam / sub‑gradient refinement
   with a much larger time budget.

4. **Linear‑programming boost** – given the current vector `f`,
   solve a tiny LP that maximises `∑ g` under the constraints
   `f * g ≤ max(f * f)` only on the most critical convolution
   entries.  Mix the solution back into `f` with a small convex
   coefficient (t≈0.15).  Repeat a few times.

5. **Local stochastic hill‑climbing** – finish with a simulated‑annealing
   style random walk (swap / mass transfer moves).

All steps respect the security checks of ``evaluate_sequence``; the
returned list contains non‑negative floats and is clipped to the
allowable range before returning.

The implementation is deliberately time‑aware: every optimisation
routine stops as soon as the global time budget (default 1000 s) is
exhausted.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (used internally) ------------------------------------
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(a*a)/(sum(a)**2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

def _warm_start(n: int):
    """Return the normalised global warm‑start (if any)."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑opposite‑smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape

    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))

    return shape

    vals /= vals.sum()

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.12,
    lam_start: float = 1.0,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best exact solution found before the deadline.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for best‑tracking only)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature (lam) schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active indices (where conv is maximal)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update and re‑normalisation
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).
    Temperature decays linearly from 0.015 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.015
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.45)   # up to 45 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


def _lp_improve(
    f: np.ndarray,
    max_conv: float,
    top_k: int,
    conv_fft_len: int,
) -> np.ndarray:
    """
    One LP step: maximise Σ g  subject to  f∗g ≤ max_conv on the
    `top_k` most critical convolution positions.
    Returns a normalised vector g (sum(g)=1).  If LP fails, returns f.
    """
    n = f.shape[0]

    # compute full convolution once (needed for the active set)
    conv_f = _full_convolution(f, conv_fft_len)

    # select the indices with the largest values (most binding constraints)
    if top_k >= conv_f.size:
        act_idx = np.arange(conv_f.size)
    else:
        act_idx = np.argpartition(conv_f, -top_k)[-top_k:]

    # build constraint matrix A (|act_idx| × n)
    rows = []
    for k in act_idx:
        # row for (f * g)[k] ≤ max_conv
        row = np.zeros(n, dtype=np.float64)
        j_low = max(0, k - (n - 1))
        j_high = min(k, n - 1)
        j_range = np.arange(j_low, j_high + 1, dtype=np.int64)
        row[j_range] = f[k - j_range]
        rows.append(row)

    A = np.vstack(rows)
    b = np.full(A.shape[0], max_conv, dtype=np.float64)

    # LP: maximise  Σ g  ⇒ minimise  -Σ g
    c = -np.ones(n, dtype=np.float64)

    # Upper bound of 1.0 for each variable keeps the problem bounded
    bounds = [(0.0, 1.0) for _ in range(n)]

    # solve
    try:
        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    except Exception:       # fall back in pathological cases
        return f

    if not res.success:
        return f

    g = np.maximum(res.x, 0.0)
    if g.sum() < 1e-12:
        g = np.random.dirichlet(np.ones(n))
    g /= g.sum()
    return g

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine respects the given time budget and returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – tiny coarse grid to get a good launch point
    n0 = 4096
    fft_len0 = _next_pow2(2 * n0 - 1)
    corr_len0 = _next_pow2(3 * n0 - 2)

    # Build diverse warm‑starts
    candidates = []

    warm = _warm_start(n0)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n0 // 4, n0 // 2, 3 * n0 // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n0))

    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n0, alpha))

    for alpha in (0.1, 0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n0, alpha))

    for p in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_powerlaw_candidate(n0, p))
        candidates.append(_ushape_candidate(n0, p))

    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n0, gamma))

    candidates.append(_triangle_candidate(n0))

    # random permutations of an anti‑pair starter
    base_ap = _anti_pair_candidate(n0, alpha=0.7)
    for _ in range(3):
        perm = np.random.permutation(n0)
        candidates.append(base_ap[perm])

    # ------------------------------------------------------------------
    # Quick evaluation of the pool
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n0))
            s = cand.sum()
        cand /= s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 8 == 0:
            print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 3
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam + sub‑gradient on the top candidates (coarse grid)
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.35 / top_k  # 35 % of time for this phase

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # Adam
        adam_dead = time.time() + per_candidate_budget * 0.5
        a_adam, val_adam = _adam_optimize(
            cand, n0, fft_len0, corr_len0, adam_dead,
            max_iters=250_000,
            lr=0.12,
            lam_start=1.0,
            lam_factor=1.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # Sub‑gradient
        sub_dead = time.time() + per_candidate_budget * 0.5
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n0, fft_len0, corr_len0, sub_dead,
            max_iters=300_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Upsample to a finer grid (if time permits)
    if time.time() < deadline - 30.0:
        n_big = 8192
        fft_len_big = _next_pow2(2 * n_big - 1)
        corr_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n0, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        best_seq = a_big
        n = n_big
        fft_len = fft_len_big
        corr_len = corr_len_big
        print(f"[Upsample] now working with n={n}")
    else:
        n = n0
        fft_len = fft_len0
        corr_len = corr_len0

    # ------------------------------------------------------------------
    # Phase 3 – LP‑boosted refinement (few quick LP steps)
    if time.time() < deadline - 20.0:
        for lp_iter in range(3):
            conv_f = _full_convolution(best_seq, fft_len)
            M = conv_f.max()
            g_lp = _lp_improve(best_seq, M, top_k=20, conv_fft_len=fft_len)
            # convex mix
            t = 0.15
            best_seq = (1 - t) * best_seq + t * g_lp
            best_seq = np.maximum(best_seq, 0.0)
            best_seq /= best_seq.sum()
            cur_val = _objective(best_seq)
            if cur_val < best_val:
                best_val = cur_val
                print(f"[LP iter {lp_iter}] improved to {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Long Adam + sub‑gradient on the upsampled grid
    if time.time() < deadline - 10.0:
        # allocate the bulk of the remaining budget to Adam + sub‑grad
        adam_dead = time.time() + (deadline - time.time()) * 0.55
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            corr_len,
            adam_dead,
            max_iters=500_000,
            lr=0.12,
            lam_start=1.0,
            lam_factor=1.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Final Adam] improved to {val_adam:.6f}")

        sub_dead = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            fft_len,
            corr_len,
            sub_dead,
            max_iters=500_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Final Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Local stochastic hill‑climbing (last few seconds)
    if time.time() < deadline - 2.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq, n, fft_len, local_dead, max_moves=250_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
