def _adam_optimize(a0,...):
    eps=1e-12
    logits = np.log(a0+eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9
    beta2=0.999
    step_size=0.1 (maybe adapt)
    t=0
    best_a = a0.copy()
    best_val = _objective(a0)
    while time.time()<deadline:
        t+=1
        a = np.exp(logits)
        a = a / a.sum()
        # compute active set
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.zeros_like(conv)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)  # subgradient w.r.t logits
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        logits = logits - step_size * m_hat/(np.sqrt(v_hat)+1e-8)
        # check best
        a_curr = np.exp(logits)
        a_curr = a_curr / a_curr.sum()
        val = _objective(a_curr)
        if val < best_val:
            best_val = val
            best_a = a_curr.copy()
        # occasional printing
        if t%10000==0:
            print(f"[Adam it={t}] best {best_val}")
    return best_a, best_val

"""
Algorithm for minimizing the convolution peak of a non‑negative step function.

The search proceeds in three stages:

1. **Random warm‑starts** – generate many Dirichlet samples and apply the
   “anti‑pair” ordering (largest opposite smallest).  The best of these
   candidates seeds the later stages.

2. **Permutation annealing** – keep the multiset of heights fixed but
   repeatedly swap or reverse blocks of the ordering.  A simple simulated‑
   annealing schedule accepts uphill moves early on and cools to a greedy
   hill‑climber.

3. **Adam‑style simplex optimisation** – work with logits (softmax
   parametrisation) and use the exact sub‑gradient of the max‑convolution
   (computed via FFT cross‑correlation).  The Adam update respects the
   simplex constraint by the softmax chain rule.

All phases respect a global time budget (default 1000 s).  The final
sequence is clipped to the range required by ``evaluate_sequence`` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    n = a.shape[0]
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Simplex helper (used only for sanity checks)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling (optional global variable)
def _warm_start(n: int):
    """Return the provided global ``height_sequence_1`` (if any) resized to n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Arrangement helpers
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = vals.shape[0]
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _alternating_extremes_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Interleave the largest and smallest values:
    largest, smallest, second‑largest, second‑smallest, …
    """
    n = vals.shape[0]
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    i = 0
    while left <= right:
        a[i] = s[left]
        i += 1
        left += 1
        if left <= right:
            a[i] = s[right]
            i += 1
            right -= 1
    return a


# ----------------------------------------------------------------------
# Candidate generators
def _dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet sample with concentration ``alpha`` (scalar)."""
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    vals = _dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float) -> np.ndarray:
    vals = _dirichlet_candidate(n, alpha)
    return _alternating_extremes_arrangement(vals)

# Permutation annealing (keeps multiset of values fixed)
def _anneal_permutation(start_seq: np.ndarray,
                        deadline: float,
                        temperature_start: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on permutations of ``start_seq``.
    Swaps and random block reversals are proposed.
    Returns the best sequence encountered.
    """
    n = start_seq.shape[0]
    conv_len = _next_pow2(2 * n - 1)

    cur = start_seq.copy()
    cur_val = _objective(cur)

    best = cur.copy()
    best_val = cur_val

    start_time = time.time()
    it = 0
    while time.time() < deadline:
        # temperature linearly decays to zero
        elapsed = time.time() - start_time
        frac = elapsed / (deadline - start_time)
        T = temperature_start * max(0.0, 1.0 - frac)

        # propose move
        if np.random.rand() < 0.5:
            # simple swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            new = cur.copy()
            new[i], new[j] = new[j], new[i]
        else:
            # block reversal
            i, j = sorted(np.random.randint(n, size=2))
            if i == j:
                continue
            new = cur.copy()
            new[i:j + 1] = new[i:j + 1][::-1]

        new_val = _objective(new)

        # acceptance test
        if new_val < cur_val:
            accept = True
        else:
            accept = (T > 0.0) and (np.random.rand() < np.exp(-(new_val - cur_val) / T))

        if accept:
            cur = new
            cur_val = new_val
            if new_val < best_val:
                best = new.copy()
                best_val = new_val

        if it % 5000 == 0:
            print(f"[Anneal] it={it:06d}  best={best_val:.6f}")

        it += 1

    return best, best_val

# Adam‑style optimisation on the simplex (log‑softmax parametrisation)
def _adam_optimize(start_seq: np.ndarray,
                   deadline: float,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   step_size: float = 0.12,
                   eps: float = 1e-12) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the exact max‑convolution objective.
    The variables are the logits of a softmax; the softmax guarantees
    non‑negativity and unit sum.
    """
    n = start_seq.shape[0]
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # initialise logits from the current probabilities
    logits = np.log(start_seq + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    best_seq = start_seq.copy()
    best_val = _objective(best_seq)

    t = 0
    while time.time() < deadline:
        t += 1
        # re‑construct a from logits
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()

        # compute active set of the convolution max
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # chain rule through the softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        logits = logits - step_size * m_hat / (np.sqrt(v_hat) + 1e-8)

        # evaluate
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a_cur = exp_l / exp_l.sum()
        val = _objective(a_cur)

        if val < best_val:
            best_val = val
            best_seq = a_cur.copy()
            if t % 5000 == 0:
                print(f"[Adam] it={t:06d}  new best={best_val:.6f}")

    return best_seq, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size
    n = 4096
    conv_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 0) initialise best using warm‑start if available
    best_seq = None
    best_val = float("inf")
    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    # --------------------------------------------------------------
    # 1) Massive random warm‑starts (Dirichlet + anti‑pair / alternating)
    n_samples = 5000
    print(f"Generating {n_samples} random Dirichlet + anti‑pair candidates …")
    for i in range(n_samples):
        if time.time() >= deadline - 300:      # keep ~5 min for later phases
            break
        # concentration α sampled log‑uniformly
        log_alpha = np.random.uniform(-2.0, 1.5)
        alpha = 10.0 ** log_alpha

        # half of the candidates use anti‑pair, half alternating
        if np.random.rand() < 0.5:
            cand = _anti_pair_candidate(n, alpha)
        else:
            cand = _alternating_candidate(n, alpha)

        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random pool] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2) Permutation annealing on the multiset of heights
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    anneal_deadline = min(deadline - 150.0, time.time() + 200.0)
    if time.time() < anneal_deadline:
        print("[Annealing] starting permutation annealing …")
        seq_anneal, val_anneal = _anneal_permutation(best_seq, anneal_deadline)
        if val_anneal < best_val:
            best_val, best_seq = val_anneal, seq_anneal.copy()
            print(f"[Anneal] improved to {val_anneal:.6f}")

    # --------------------------------------------------------------
    # 3) Adam‑style simplex optimisation (log‑softmax)
    adam_deadline = min(deadline - 20.0, time.time() + 300.0)
    if time.time() < adam_deadline:
        print("[Adam] starting Adam optimisation …")
        seq_adam, val_adam = _adam_optimize(best_seq, adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Final safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
