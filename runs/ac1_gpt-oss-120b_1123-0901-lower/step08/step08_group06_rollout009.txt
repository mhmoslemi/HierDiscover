def _beta_candidate(n, alpha, beta):
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    # compute density; handle alpha,beta small using log to avoid underflow.
    # Use np.exp((alpha-1)*np.log(t) + (beta-1)*np.log(1-t))
    log_pdf = (alpha - 1) * np.log(t) + (beta - 1) * np.log(1 - t)
    pdf = np.exp(log_pdf)
    # Replace any nan with zero (due to t=0 or 1)
    pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
    if pdf.sum() == 0.0:
        pdf = np.full(n, 1.0/n)
    else:
        pdf /= pdf.sum()
    return pdf

def _smooth_surrogate_and_grad(logits, n, conv_fft_len, corr_fft_len, lam):
    # Compute a from logits via softmax
    max_logit = np.max(logits)
    exp_l = np.exp(logits - max_logit)
    a = exp_l / exp_l.sum()
    # convolution
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # smooth max
    z = np.exp(lam * (conv - max_c))
    sum_z = z.sum()
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)
    # soft weights
    w = z / sum_z
    # gradient w.r.t a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return surrogate, grad_log
```

def _adam_optimize(logits, n, conv_fft_len, corr_fft_len, deadline,
                   lam_schedule=(1.0, 5.0, 15.0, 40.0, 120.0, 300.0),
                   max_iter_stage=5000, lr=0.1):
    best_log = logits.copy()
    best_val = float('inf')
    best_a = None
    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t_iter = 0
        while time.time() < deadline and t_iter < max_iter_stage:
            t_iter += 1
            surrogate, grad_log = _smooth_surrogate_and_grad(logits, n, conv_fft_len, corr_fft_len, lam)
            # Adam step
            m = beta1 * m + (1 - beta1) * grad_log
            v = beta2 * v + (1 - beta2) * (grad_log * grad_log)
            m_hat = m / (1 - beta1**t_iter)
            v_hat = v / (1 - beta2**t_iter)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
            # evaluate exact objective
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_log = logits.copy()
                best_a = a.copy()
                print(f"[Adam λ={lam}] new best {val:.6f}")
        # after finishing this λ, set logits to best_log (maybe add jitter)
        logits = best_log.copy()
        # optional jitter: small Gaussian noise to logits to escape plateau
        if time.time() < deadline:
            jitter = np.random.randn(n) * 0.01
            logits += jitter
    if best_a is None:
        # fallback: from logits
        a = np.exp(logits - np.max(logits))
        best_a = a / a.sum()
        best_val = _objective(best_a)
    return best_a, best_val

"""
Propose a non‑negative step function that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in several phases, all respecting a global 1000 s budget.

1. **Warm‑starts & diverse random candidates**  
   – Use any previously supplied `height_sequence_1`.  
   – Hand‑crafted starters (uniform, triangular, Gaussian‑edge).  
   – Hundreds of Dirichlet draws with an “anti‑pair” ordering.  
   – Beta‑distributed shapes, an arcsine‑shaped distribution and low‑frequency
     sinusoidal perturbations of the uniform vector.  
   The best among all candidates is kept.

2. **Smooth‑max surrogate optimisation**  
   Replace the hard max by a temperature‑controlled log‑sum‑exp and run
   Adam on the logits (soft‑max parametrisation).  The temperature λ is
   increased gradually (1 → … → 300) so the surrogate approaches the true
   max.

3. **Exact‑max sub‑gradient refinement** (multiplicative / exponentiated‑gradient).

4. **Local stochastic hill‑climbing** (swap / fractional mass transfer
   with a simple annealing acceptance rule).

5. **Optional up‑sampling** to 8192 points, followed by another round of
   Adam + sub‑gradient on the finer grid.

All stages stop as soon as the global deadline is reached; the best
sequence found up to that moment is returned (clipped to the limits of
``evaluate_sequence``).
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence after clamping)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by additive sub‑gradient (optional)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)                     # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]                     # front gets biggest
    a[-half:] = s[-half:][::-1]             # back gets smallest (reversed)
    if n % 2:
        a[half] = s[half]                    # middle stays
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _arcsine_candidate(n: int) -> np.ndarray:
    """Discrete arcsine pdf ∝ 1/√(t(1‑t)), t∈(0,1)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    shape = 1.0 / np.sqrt(t * (1.0 - t))
    shape /= shape.sum()
    return shape


def _beta_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta(α,β) density sampled on mid‑points of the grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    # work in log‑space to avoid under‑flow for α,β < 1
    log_pdf = (alpha - 1.0) * np.log(t) + (beta - 1.0) * np.log(1.0 - t)
    pdf = np.exp(log_pdf)
    pdf = np.nan_to_num(pdf, nan=0.0, posinf=0.0, neginf=0.0)
    if pdf.sum() == 0.0:
        pdf = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        pdf /= pdf.sum()
    return pdf


def _low_freq_perturb_candidate(
    n: int, num_modes: int = 3, amp_range: float = 0.5
) -> np.ndarray:
    """
    Start from uniform and add a few low‑frequency sine waves.
    Amplitudes are drawn uniformly from [-amp_range,amp_range]/num_modes.
    """
    base = np.full(n, 1.0 / n, dtype=np.float64)
    i = np.arange(n, dtype=np.float64)
    shape = base.copy()
    for _ in range(num_modes):
        k = np.random.randint(1, n // 2)                # frequency
        phase = np.random.rand() * 2.0 * np.pi
        amp = np.random.uniform(-amp_range, amp_range) / num_modes
        shape += amp * np.sin(2.0 * np.pi * k * i / n + phase)
    shape = np.maximum(shape, 0.0)
    if shape.sum() < 1e-12:
        shape = base
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise to sum = 1 (objective is scale‑invariant)
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate and its gradient (log‑sum‑exp)
def _smooth_surrogate_and_grad(
    logits: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    lam: float,
):
    """
    Compute the smooth‑max surrogate
        s(a) = max_c + (1/λ)·log Σ exp(λ·(conv‑max_c))
    and its gradient w.r.t. the logits of the soft‑max parametrisation.
    """
    max_log = np.max(logits)
    exp_l = np.exp(logits - max_log)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp around the current maximum
    z = np.exp(lam * (conv - max_c))
    sum_z = z.sum()
    # surrogate value (unused here but returned for completeness)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                                 # soft weights, sum = 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)                 # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(
    logits: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lam_schedule=(1.0, 5.0, 15.0, 40.0, 120.0, 300.0),
    max_iter_stage: int = 5000,
    lr: float = 0.1,
):
    """Adam on the smooth‑max surrogate with an increasing temperature schedule."""
    best_log = logits.copy()
    best_val = float("inf")
    best_a = None

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        t_iter = 0

        while time.time() < deadline and t_iter < max_iter_stage:
            t_iter += 1
            _, grad_log = _smooth_surrogate_and_grad(
                logits, n, conv_fft_len, corr_fft_len, lam
            )
            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log * grad_log)
            m_hat = m / (1.0 - beta1 ** t_iter)
            v_hat = v / (1.0 - beta2 ** t_iter)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            # exact objective for monitoring
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_log = logits.copy()
                best_a = a.copy()
                print(f"[Adam λ={lam:.1f}] new best {val:.6f}")

        # after each temperature, re‑initialise logits around the best found
        logits = best_log.copy()
        if time.time() < deadline:
            logits += np.random.randn(n) * 0.01  # tiny jitter to escape plateaus

    if best_a is None:
        a = np.exp(logits - np.max(logits))
        best_a = a / a.sum()
        best_val = _objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Multiplicative (exponentiated‑gradient) sub‑gradient optimiser
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.7,
):
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # occasional re‑initialisation
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:  # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:  # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)  # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

            )
            if val_big_sub < best_val:
                best_val, best_seq = val_big_sub, seq_big_sub.copy()
                print(f"[Big Subgrad] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 9) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
