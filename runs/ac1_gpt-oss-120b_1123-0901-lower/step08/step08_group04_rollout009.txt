"""
Propose a low‑value step function by a three‑stage pipeline:

1. **Diverse warm‑starts** – generate many short candidates (uniform,
   Dirichlet, anti‑pair, edge‑focused) and keep the best according to the
   exact objective (2·n·max(a*a)/(sum(a)²)).

2. **Continuous refinement** – run Adam on a smooth‑max (log‑sum‑exp)
   surrogate and then an exact‑max sub‑gradient (exponentiated‑gradient)
   phase.  Both phases use FFT‑based convolutions and keep the vector
   non‑negative and normalised to sum = 1.

3. **Fine‑grained simulated‑annealing** – an incremental‑update SA that
   transfers a random fraction of mass between two positions.  The
   convolution is kept updated in O(n) via slice operations; a full
   FFT recompute is performed every few thousand steps to correct drift.
   Acceptance follows a linear temperature schedule.

The routine works with length 8192 (≈ 8 k) – a good trade‑off between
solution quality and time budget.  The final sequence is clipped to
[0, 1000] and renormalised to sum = 1 before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks) ------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)

# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """If a previous solution (height_sequence_1) exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Place large values at the ends and small values in the centre.
    This pairs large with small when the indices are added (the
    convolution “antidiagonals”).
    """
    n = len(vals)
    v = np.sort(vals)[::-1]                     # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    toggle = True
    for val in v:
        if toggle:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
        toggle = not toggle
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape

# Optimisation primitives -----------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameterisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # soft‑max → simplex

        # exact max for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        obj = 2.0 * n * max_c                     # denominator = 1 (sum=1)
        if obj < best_val:
            best_val = obj
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a (using correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # centring to stay on the simplex
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient (exponentiated‑gradient) phase."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        obj = 2.0 * n * max_c
        if obj < best_val:
            best_val = obj
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {obj:.6f}")

        # active set of the max entries (ties)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental SA utilities ----------------------------------------------

def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Incrementally update the full linear convolution a*a after adding `delta`
    to a[idx] (delta may be negative).  Operates in O(n) using NumPy slices.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (position 2*idx)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.03,
    max_frac: float = 0.3,
    recompute_every: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing with an exact convolution
    recompute every ``recompute_every`` steps to purge accumulated
    floating‑point drift.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    # exact convolution at the start
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_obj = 2.0 * n * conv.max()
    cur_obj = best_obj

    start = time.time()
    it = 0
    while time.time() < time_limit:
        # pick donor i (must have some mass) and recipient j (j!=i)
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move incrementally
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j,  delta)

        new_obj = 2.0 * n * conv.max()
        diff = new_obj - cur_obj

        # linear temperature schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_a = a.copy()
        else:
            # revert (note: the delta signs are swapped)
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i,  delta)

        it += 1
        # periodic full recompute to avoid drift
        if it % recompute_every == 0:
            conv = _full_convolution(a, fft_len)

        if it % 100_000 == 0:
            print(f"[SA it={it:06d}] best objective {best_obj:.6f}")

    return best_a, best_obj

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step function with minimal 2·n·max(a*a)/(sum(a)²).

    The function respects the given time budget, uses a mix of random
    warm‑starts, gradient‑based optimisation and simulated‑annealing, and
    finally returns a non‑negative list (values clipped to [0,1000] and
    normalised to sum = 1) that can be fed directly to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size – 8192 gives a better theoretical lower bound
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a pool of initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few cyclic shifts of the warm start – occasionally helpful
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))
    for alpha in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
    for alpha in (0.2, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # --------------------------------------------------------------
    # 2) Pick the best initial candidate (cheap exact evaluation)
    best_seq = None
    best_obj = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:               # guard against degenerate start
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        obj = _objective(cand)
        print(f"[Init cand {idx:02d}] objective {obj:.6f}")
        if obj < best_obj:
            best_obj = obj
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # Allocate ~30 % of the budget to these phases
    cont_deadline = deadline - 30.0
    if time.time() < cont_deadline:
        # ---- Adam phase
        adam_deadline = min(cont_deadline - 20.0, time.time() + 180.0)
        a_adam, obj_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if obj_adam < best_obj:
            best_obj = obj_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {obj_adam:.6f}")

        # ---- Sub‑gradient phase
        sub_deadline = min(cont_deadline - 5.0, time.time() + 250.0)
        a_sub, obj_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
            eta0=0.7,
        )
        if obj_sub < best_obj:
            best_obj = obj_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {obj_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long incremental SA (use the remaining budget)
    if time.time() < deadline - 5.0:
        sa_seq, obj_sa = _incremental_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.04,
            max_frac=0.25,
            recompute_every=5000,
        )
        if obj_sa < best_obj:
            best_obj = obj_sa
            best_seq = sa_seq.copy()
            print(f"[SA] improved to {obj_sa:.6f}")

    # --------------------------------------------------------------
    # 5) Final sanity checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:                 # extremely unlikely, but guard against it
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        total = 1.0
    best_seq /= total                # enforce sum ≈ 1 (scale‑invariant)

    final_obj = _objective(best_seq)
    print(f"Final objective {final_obj:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
