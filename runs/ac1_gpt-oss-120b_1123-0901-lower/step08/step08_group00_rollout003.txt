"""
stepfunction_optimizer_v2.py

A multi‑phase optimizer for the objective

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

The algorithm combines several complementary ideas that improve on the
previous version:

1. **Rich initial pool** – in addition to the old Dirichlet / spike /
   exponential families we add
   * centred Gaussian shapes (different σ),
   * symmetric Beta‑type shapes (different concentration α),
   * a simple triangular shape and an “interleaved’’ permutation of every
     candidate (large entries are spread evenly).

2. **Adam on a log‑sum‑exp (smooth‑max) surrogate** – short runs on many
   candidates quickly prune the pool.

3. **Exact‑max exponentiated‑gradient sub‑gradient phase** – long runs on the
   current best sequence drive the true max down.

4. **Simulated‑annealing with two move types** – mass transfer (as before)
   **and** occasional swaps of two coordinates.  Swaps allow us to explore
   the space of permutations without altering the marginal distribution.

5. **Permutation hill‑climbing** – after the annealing phase we perform a
   dedicated local search that only swaps entries, accepting only improvements
   (with a tiny probability for uphill moves).  This often reduces the
   maximum convolution a little further.

6. **Final sanity checks** – the best vector is normalised, clipped to the
   allowed range [0,1000] and returned as a plain Python list.

The whole routine respects a user‑supplied time budget (default 1000 s) and
always returns the best sequence it found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value used for bookkeeping (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  The parameters were
    tuned for stability on long runs.
    Returns the best vector seen (normalised to sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase λ)
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random jitter near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with exponentiated‑gradient updates.
    Operates on the probability simplex (a normalized to sum=1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Two move types:
        * mass transfer (as before)
        * swap of two coordinates (probability = swap_prob)
    The temperature decays linearly.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # accept uphill move with probability exp(-Δ/T)
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation hill‑climbing: repeatedly swap two entries, keep the swap
    only if it does not increase the objective (with a tiny probabilistic
    acceptance of uphill moves to escape plateaus).
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance (1 % chance) to get out of flat spots
            if np.random.rand() < 0.01:
                val = val2
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val

# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (augmented with new families)."""
    cand = []

    # Warm start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with varied concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Edge‑polynomial families (currently with negative exponents)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian (peaks at both ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # ---------- NEW families -------------------------------------------------
    # Central Gaussian (single peak)
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            cand.append(shape)

    # Symmetric Beta‑type shapes (α = β > 1, peaked in the centre)
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    # Simple triangular shape (linear rise to middle, then fall)
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Interleaved permutation of every candidate (helps to spread large mass)
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns (equally spaced or random)."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size (fixed – 4096 points gives a good trade‑off)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build diversified pool of starter vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – rapid exploration of many candidates (≈25 % of budget)
    explore_deadline = time.time() + (deadline - time.time()) * 0.25
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)

        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # quick Adam refinement (bounded by a small sub‑deadline)
        sub_dead = min(explore_deadline, time.time() + 4.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improvement → {val_adam:.6f}")

        # a few sub‑gradient steps for polishing (if still time)
        if time.time() < explore_deadline:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, explore_deadline,
                max_iters=20_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Sub‑grad improvement → {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep sub‑gradient refinement of the incumbent (≈45 % budget)
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Long sub‑gradient optimisation started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] New best after sub‑grad → {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – simulated‑annealing with mass‑transfer & swaps (≈20 % budget)
    if best_seq is not None and time.time() < deadline - 15:
        print("[Anneal] Starting simulated‑annealing moves")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=70_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] Improved best → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – permutation‑only hill climbing (≈5 % budget)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Permute] Starting permutation hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=30_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] New best after permutation → {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
