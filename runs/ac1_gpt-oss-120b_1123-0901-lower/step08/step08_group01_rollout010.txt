"""
Simple Frank‑Wolfe style optimizer for the max‑convolution problem.
We start from a wide collection of parametric step‑function shapes,
pick the best, then repeatedly solve a linear program

    max   Σ b_j
    s.t.  (f * b)_k ≤ M  ∀k,   b ≥ 0

where M = max(f*f).  The LP gives a direction that does not increase the
current max‑convolution.  By mixing the current vector f with the LP
solution we obtain a strictly decreasing sequence of M values.
When the time budget is almost exhausted a cheap random‑move hill‑climber
polishes the final vector.  The whole routine works for any length;
we use n=512 (or the length of a supplied warm‑start) so that the LP
remains tiny and many iterations fit into the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog

def _next_pow2(x: int) -> int:
    """Return the smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """Build the matrix A where (A @ b)[k] = (f * b)[k]."""
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # each column i contributes f to rows i .. i+n-1
    for i in range(n):
        A[i:i + n, i] = f
    return A

def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  max Σ b_j   s.t.   A @ b ≤ M,  b ≥ 0,
    where A encodes the convolution (f * b).
    Returns the optimal b (may be all‑zeros if the solver fails).
    """
    # Build constraints
    A = _conv_matrix(f)
    rows = A.shape[0]
    b_ub = np.full(rows, M + 1e-12, dtype=np.float64)   # tiny slack for numeric safety
    c = -np.ones(f.shape[0], dtype=np.float64)         # minimise –Σb ⇒ maximise Σb

    # HiGHS interior‑point method works well for dense problems of this size
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback – zero vector (feasible for any M)
        return np.zeros_like(f)
    return res.x

def _objective_from_conv(max_conv: float, n: int) -> float:
    """Exact evaluation of the objective for a sequence with unit sum."""
    return 2.0 * n * max_conv

def _evaluate_seq(a: np.ndarray, conv_fft_len: int) -> float:
    """Return the exact objective (sum is assumed to be positive)."""
    conv = _full_convolution(a, conv_fft_len)
    return _objective_from_conv(conv.max(), a.shape[0])

def _generate_initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Return a list of diverse, cheap‑to‑evaluate candidates."""
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Simple sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    cand.append(s / s.sum())

    # Edge‑exponential with many gamma values (log‑space)
    for gamma in np.logspace(-2, 2, num=30):          # 0.01 … 100
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Edge power‑law (both sides)
    for alpha in np.linspace(-5.0, 5.0, num=30):
        idx = np.arange(n, dtype=np.float64) + 1.0
        shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
        else:
            shape[:] = 1.0 / n
        cand.append(shape)

    # Beta‑pdf families (a,b in (0.01, 10) log‑space)
    for a in np.logspace(-2, 1, num=15):
        for b in np.logspace(-2, 1, num=15):
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                cand.append(shape)

    # Random Dirichlet draws with various concentration parameters
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        cand.append(rng.dirichlet(np.full(n, alpha)))

    # Sparse spikes (k fixed values)
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = rng.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand

def _local_random_hill_climb(a0: np.ndarray, conv_fft_len: int,
                             deadline: float, max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index
    to another; keep the move if the objective improves.
    """
    a = a0.copy()
    best_val = _evaluate_seq(a, conv_fft_len)
    best_a = a.copy()
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _evaluate_seq(a2, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            # occasional log
            if it % 5000 == 0:
                print(f"[Hill‑climb {it:06d}] improved to {best_val:.6f}")
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The method consists of a fast parametric sweep, a Frank–Wolfe style
    loop that solves a small linear program in each iteration, and a
    final cheap random‑move hill‑climber.  The routine always returns the
    best sequence discovered before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    # keep a small safety margin for the final conversion to Python list
    deadline = start + budget_s - 0.5

    # ------------------------------------------------------------------
    # Choose the working dimensionality.
    # If a warm start is provided we honour its length (truncated/padded)
    # otherwise we work with n = 512 – small enough for the LP.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # keep at most 512 entries – longer vectors are simply cropped
        n = min(warm.shape[0], 512)
        if warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        else:
            warm = warm[:n]
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None
        n = 512

    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep
    pool = _generate_initial_candidates(n, rng)
    if warm is not None:
        pool.append(warm)

    best_a = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        # Normalise (protect against zero‑sum)
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _evaluate_seq(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Initial {idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe‑type refinement via exact LPs
    # We iterate while time permits; the mixing coefficient t is reduced
    # gradually so that the algorithm remains monotone.
    # A simple back‑tracking line‑search guarantees that the max‑convolution
    # never increases.
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    f = best_a.copy()
    val_f = _evaluate_seq(f, conv_fft_len)
    best_val = val_f
    best_seq = f.copy()

    iteration = 0
    while time.time() < deadline - 5.0:      # keep a few seconds for final hill‑climb
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()
        # Solve the LP that finds the direction g
        b = _solve_lp_max_sum(f, M)
        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        g = b / sum_b

        # Mix f and g – decreasing step size as the algorithm proceeds
        t = 0.3 / (iteration + 1)   # starts at 0.3, then 0.15, 0.1, …
        # Ensure monotonicity by back‑tracking if needed
        improved = False
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            # renormalise (numerical drift)
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                break
            f_new = f_new / s
            new_val = _evaluate_seq(f_new, conv_fft_len)
            if new_val < val_f:
                f = f_new
                val_f = new_val
                improved = True
                break
            t *= 0.5
        if not improved:
            # unable to improve – break to avoid endless looping
            break

        if val_f < best_val:
            best_val = val_f
            best_seq = f.copy()
            print(f"[FW iter {iteration}] new best {best_val:.6f}")

        iteration += 1
        # safety stop after many iterations
        if iteration >= 2000:
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap random local moves for the final seconds
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_hill_climb(best_seq, conv_fft_len,
                                                  deadline, max_moves=100_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Final hill‑climb] improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to the evaluator’s accepted domain (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Frank‑Wolfe style optimizer for the max‑convolution problem.

1. **Broad parametric sweep** – a few hundred simple families (edge‑exponential,
   power‑law, Beta‑pdf, uniform, sinusoid, random Dirichlet, sparse spikes) are
   generated and the best of them (according to the exact objective) is kept.
2. **Exact linear‑programming refinement** – given the current vector `f` we
   compute `M = max(f*f)`.  The LP

        max   Σ b_j
        s.t.  (f * b)_k ≤ M  for all k,   b_j ≥ 0

   is solved (SciPy ‘highs’).  The solution `b` is normalised to a probability
   vector `g`.  A short back‑tracking line‑search mixes `f←(1‑t)f + t g` with a
   decreasing step size `t`.  By construction `max(f*f)` never increases,
   therefore the objective `2·n·max(f*f)` monotonically decreases.  Several
   hundred such iterations fit comfortably into the 1000 s budget.
3. **Final cheap hill‑climber** – random mass transfers between two indices are
   performed while the objective improves; this polishes the last few digits.

The algorithm works for any length; we use `n = 512` (or the length of a supplied
warm‑start) so the LP stays tiny and many iterations are possible.  The returned
list is guaranteed to be non‑negative and accepted by `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build matrix A (size (2n‑1)×n) such that (A @ b)[k] = (f * b)[k]
    for any vector b of length n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


def _solve_lp_max_sum(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max Σ b_j   s.t.   (f * b)_k ≤ M (∀k) , b ≥ 0.
    Returns the optimal b (all zeros if the solver fails).
    """
    A = _conv_matrix(f)                     # (2n‑1)×n
    rows = A.shape[0]
    b_ub = np.full(rows, M + 1e-12, dtype=np.float64)   # tiny slack for numerics
    c = -np.ones(f.shape[0], dtype=np.float64)           # minimise –Σb ⇒ maximise Σb

    # HiGHS works well for dense small‑size LPs
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return np.zeros_like(f)
    return res.x


def _objective_from_conv(max_conv: float, n: int) -> float:
    """Exact objective for a unit‑sum vector."""
    return 2.0 * n * max_conv


def _evaluate_seq(a: np.ndarray, conv_fft_len: int) -> float:
    """Return the exact objective for vector a (sum(a) > 0)."""
    conv = _full_convolution(a, conv_fft_len)
    return _objective_from_conv(conv.max(), a.shape[0])


def _generate_initial_candidates(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of cheap, parametric candidates."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n, dtype=np.float64))

    # simple half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    cand.append(s / s.sum())

    # edge‑exponential (log‑spaced gamma)
    for gamma in np.logspace(-2, 2, num=30):          # 0.01 … 100
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # edge power‑law
    for alpha in np.linspace(-5.0, 5.0, num=30):
        idx = np.arange(n, dtype=np.float64) + 1.0
        shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
        else:
            shape[:] = 1.0 / n
        cand.append(shape)

    # beta‑pdf family (log‑spaced a,b)
    for a in np.logspace(-2, 1, num=15):
        for b in np.logspace(-2, 1, num=15):
            x = (np.arange(n, dtype=np.float64) + 0.5) / n
            shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                shape /= s
                cand.append(shape)

    # random Dirichlet draws
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        cand.append(rng.dirichlet(np.full(n, alpha)))

    # sparse spikes (k non‑zero entries)
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = rng.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
        cand.append(a)

    return cand


def _local_random_hill_climb(a0: np.ndarray, conv_fft_len: int,
                             deadline: float, max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index to another,
    keeping the move if the exact objective improves.
    """
    a = a0.copy()
    best_val = _evaluate_seq(a, conv_fft_len)
    best_a = a.copy()
    n = len(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _evaluate_seq(a2, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Hill‑climb {it:06d}] improved to {best_val:.6f}")
        it += 1

    return best_a, best_val


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin for final conversion

    # ------------------------------------------------------------------
    # Choose working dimension (use warm start if supplied, otherwise n=512)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        n = min(warm.shape[0], 512)
        if warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        else:
            warm = warm[:n]
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None
        n = 512

    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep
    pool = _generate_initial_candidates(n, rng)
    if warm is not None:
        pool.append(warm)

    best_a = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _evaluate_seq(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Initial {idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Frank‑Wolfe refinement via exact LPs
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    f = best_a.copy()
    val_f = _evaluate_seq(f, conv_fft_len)
    best_val = val_f
    best_seq = f.copy()

    iteration = 0
    while time.time() < deadline - 5.0:              # reserve a few seconds for final step
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()

        # LP to obtain direction g
        b = _solve_lp_max_sum(f, M)
        sum_b = b.sum()
        if sum_b < 1e-12:
            break
        g = b / sum_b

        # Decreasing mixing coefficient; back‑track if the move does not improve
        t = 0.3 / (iteration + 1)   # 0.3, 0.15, 0.10, …
        improved = False
        while t > 1e-4:
            f_new = (1.0 - t) * f + t * g
            f_new = np.maximum(f_new, 0.0)
            s = f_new.sum()
            if s < 1e-12:
                break
            f_new = f_new / s
            new_val = _evaluate_seq(f_new, conv_fft_len)
            if new_val < val_f:
                f = f_new
                val_f = new_val
                improved = True
                break
            t *= 0.5

        if not improved:
            # No monotone improvement possible – exit the loop
            break

        if val_f < best_val:
            best_val = val_f
            best_seq = f.copy()
            print(f"[FW iter {iteration}] new best {best_val:.6f}")

        iteration += 1
        if iteration >= 500:          # safety cap
            break

    # ------------------------------------------------------------------
    # Phase 3 – cheap random hill‑climber for the very last seconds
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_hill_climb(best_seq, conv_fft_len,
                                                  deadline, max_moves=100_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Final hill‑climb] improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final safety: clip to allowed interval and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
