"""
Propose a step‐function with a lower evaluation value.

The algorithm builds many random warm‑starts, picks the best and then
refines it via three stages:

1. Adam on a smooth‑max (log‑sum‑exp) surrogate – quick rough
   optimisation.
2. Exact‑max sub‑gradient (exponentiated‑gradient) – drives the
   maximum convolution down directly.
3. Targeted simulated‑annealing – each move removes mass from an
   index that contributes most to the *current* maximal convolution
   index and places it at a low‑mass index.  The move is accepted only
   if the global maximum does not increase (or with a tiny
   temperature‑controlled probability).

All operations use FFT‑based convolutions for speed and respect the
global time budget (≈1000 s).  The returned vector is non‑negative,
clamped to the allowed range and normalised to sum ≈ 1, ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Exact objective (no safety checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
#  Incremental convolution utilities (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float) -> None:
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    # contribution of the cross‑terms (ordered pairs)
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (i=j)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


# ----------------------------------------------------------------------
#  Candidate generators
def _warm_start(n: int):
    """Adapt a previously found solution (if any) to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Sort descending and place big values on the left, small on the right."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """1,2,…,⌈n/2⌉,…,2,1 – then normalised."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
#  Adam optimiser (smooth‑max surrogate)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.8f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional tiny jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Exact‑max sub‑gradient (exponentiated‑gradient)
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.8f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

#  Targeted Simulated‑Annealing (focuses on the current maximal index)
def _targeted_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.02,
    max_frac: float = 0.5,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        # index (lag) where the convolution attains its maximum
        max_k = int(conv.argmax())

        # contributions of each position to this lag
        idx = np.arange(n)
        j = max_k - idx
        mask = (j >= 0) & (j < n)
        contrib = np.zeros_like(a)
        contrib[mask] = a[idx[mask]] * a[j[mask]]

        # source index: the one with the largest contribution
        if contrib.max() > 0:
            i_src = int(contrib.argmax())
        else:
            i_src = np.random.randint(n)

        # target index: a location with very little mass (helps keep sum)
        # pick from the lowest 10 % quantile; if none exist pick random
        low_thr = np.percentile(a, 5)
        low_candidates = np.where(a <= low_thr)[0]
        if low_candidates.size == 0:
            i_tgt = np.random.randint(n)
        else:
            i_tgt = int(np.random.choice(low_candidates))

        if i_src == i_tgt:
            i_tgt = (i_src + np.random.randint(1, n)) % n

        # amount of mass to move – a random fraction of the source mass
        max_delta = a[i_src] * max_frac
        if max_delta <= 1e-15:
            # nothing to move – try another iteration
            continue
        delta = max_delta * np.random.rand()

        # apply the transfer
        _apply_delta(a, conv, i_src, -delta)
        _apply_delta(a, conv, i_tgt, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert move
            _apply_delta(a, conv, i_tgt, -delta)
            _apply_delta(a, conv, i_src, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[TSA it={it:06d}] best conv_max {best_val:.8f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The routine consists of:
      1) many random warm‑starts,
      2) a short Adam phase,
      3) a longer exact‑max sub‑gradient phase,
      4) a targeted simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # problem size – 4096 is a good compromise of quality / speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a large pool of diverse candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotations of the warm start – sometimes useful
        for shift in (n // 8, n // 4, 3 * n // 8):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))
    # many random Dirichlet draws with varying concentration parameters
    for _ in range(200):
        alpha = 10 ** np.random.uniform(-1.5, 0.5)   # α ∈ [0.03, 3.16]
        cand = _random_dirichlet_candidate(n, alpha)
        # occasionally apply the anti‑pair trick
        if np.random.rand() < 0.3:
            cand = _anti_pair_arrangement(cand)
        candidates.append(cand)

    # a few handcrafted shapes
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # --------------------------------------------------------------
    # 2) Pick the best starter (cheap exact evaluation)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:03d}] value {val:.8f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Continuous optimisation – Adam + exact‑max sub‑gradient
    # Reserve ~30 s for the final targeted SA
    sa_deadline = deadline - 30.0
    if time.time() < sa_deadline:
        # ---- Adam phase (smooth‑max)
        adam_deadline = min(sa_deadline - 20.0, time.time() + 150.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=2_000_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.8f}")

        # ---- Sub‑gradient refinement (exact max)
        sub_deadline = min(sa_deadline - 5.0, time.time() + 300.0)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=2_000_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.8f}")

    # --------------------------------------------------------------
    # 4) Targeted simulated‑annealing refinement
    if time.time() < deadline - 5.0:
        sa_seq, sa_val = _targeted_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.03,
            max_frac=0.5,
        )
        if sa_val < best_val:
            best_val = sa_val
            best_seq = sa_seq.copy()
            print(f"[Targeted SA] improved to {sa_val:.8f}")

    # --------------------------------------------------------------
    # 5) Final sanitisation and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 0.01:               # safety fallback – should never happen
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        total = best_seq.sum()
    best_seq /= total               # enforce sum ≈ 1 (scale‑invariant)

    print(f"Final conv_max {best_val:.8f}  (length {len(best_seq)})")
    return best_seq.tolist()
