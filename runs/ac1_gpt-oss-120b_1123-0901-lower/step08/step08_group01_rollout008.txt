def _continuation_l_bfgs(a0, n, conv_fft_len, corr_fft_len, deadline):
    # Convert to logits
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float('inf')
    # schedule of alphas
    alphas = [10.0, 100.0, 1000.0, 5000.0, 20000.0]
    for alpha in alphas:
        if time.time() > deadline - 3:
            break
        # define objective for scipy
        def func_and_grad(x):
            # softmax
            mx = x.max()
            exp_x = np.exp(x - mx)
            a = exp_x / exp_x.sum()
            # conv
            fa = np.fft.rfft(a, conv_fft_len)
            conv = np.fft.irfft(fa * fa, conv_fft_len)[:2*n-1]
            # compute smooth max
            max_c = conv.max()
            v = conv - max_c
            # log-sum-exp scaled
            # to avoid overflow, compute using v <= 0
            # logsumexp = max_c + (1/alpha) * np.log(np.exp(alpha * v).sum())
            exp_alpha_v = np.exp(alpha * v)
            sum_exp = exp_alpha_v.sum()
            smooth_max = max_c + (1.0/alpha) * np.log(sum_exp)
            obj = 2.0 * n * smooth_max
            # weights
            w = exp_alpha_v / sum_exp
            # gradient wrt a: grad_a = 2 * crosscorr(w, a)
            rev_a = a[::-1]
            fw = np.fft.rfft(w, corr_fft_len)
            fr = np.fft.rfft(rev_a, corr_fft_len)
            corr = np.fft.irfft(fw * fr, corr_fft_len)
            start = n-1
            grad_a = 2.0 * corr[start:start+n]
            # gradient wrt logits
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)
            return obj, grad_x
        # closure that passes logits as parameter
        def f(x):
            val, _ = func_and_grad(x)
            return val
        def g(x):
            _, grad = func_and_grad(x)
            return grad
        # set maxiter depending on remaining time
        max_iter = 200
        # time aware: reduce if deadline near
        remaining = deadline - time.time()
        if remaining < 30:
            max_iter = 50
        res = scipy.optimize.minimize(f, logits, jac=g, method='L-BFGS-B',
                                      options={'maxiter': max_iter, 'disp': False})
        logits = res.x
        # compute exact objective
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS alpha={alpha}] new best {exact_val:.6f}")
    # return best a
    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val

"""
Search for a non‑negative step‑function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of four stages:

1. **Diverse initialisation** – a large pool of deterministic families
   (edge‑exponential, edge‑power‑law, β‑pdf, double‑Gaussian, sinusoid,
   raised‑cosine, semicircle) together with random Dirichlet and sparse‑spike
   vectors.  The best few candidates are kept.

2. **Long Adam optimisation** – a smooth‑max surrogate (log‑sum‑exp) is
   minimised on the simplex using Adam (logits → softmax).  This stage
   receives the bulk of the time budget.

3. **Continuation L‑BFGS** – the surrogate is refined with a sequence of
   increasing temperature parameters α (10 → 10 000).  For each α an
   L‑BFGS (exact gradient, no constraints) is run on the logits; the best
   exact objective found is retained.

4. **Cheap local hill‑climber** – random mass‑moves between indices are
   accepted only when they improve the true objective.

The routine respects the 1000 s deadline, prints progress and always
returns the best sequence found (as a plain Python list).  All helper
functions are defined at top‑level; no closures or lambda expressions are
used.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate families
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape

# Optimisers
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # inverse temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber moving a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# L‑BFGS continuation on the smooth‑max surrogate
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Return (objective, gradient) for a given temperature α."""
    # soft‑max (logits → simplex)
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    # smooth‑max = (1/α)·log‑sum‑exp
    max_c = conv.max()
    v = conv - max_c                     # ≤ 0
    exp_alpha_v = np.exp(alpha * v)      # safe, max = 1
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    # weights for the gradient (softmax over conv)
    w = exp_alpha_v / sum_exp

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on a sequence of increasing temperatures α.
    Returns the best exact sequence and its true objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    # temperature schedule – start soft, end hard
    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        # SciPy minimiser expects a callable returning (f, grad) when jac=True
        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len, corr_fft_len, alpha)

        # allow a modest number of L‑BFGS steps (each step is expensive)
        remaining = deadline - time.time()
        max_iter = 150 if remaining > 30 else 50

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        # evaluate the exact objective
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    # final best on the simplex
    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- warm start handling --------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ---- build a diverse pool of starters -------------------------
    pool = []

    # warm start and its cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    # edge exponential – log‑spaced gammas
    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))

    # edge power‑law – wide range of exponents
    for alpha in np.linspace(-8, -0.2, 20):
        pool.append(_edge_power_law(n, alpha))

    # Beta PDFs
    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0), (5.0, 0.2),
        (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # Additional smooth shapes
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # Sparse random spikes
    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    # --------------------------------------------------------------
    # Quick scan: evaluate all candidates, keep the best one
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-8:
            continue
        a /= s
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        if time.time() > deadline - 60:   # leave ~1 min for deep phases
            break

    # --------------------------------------------------------------
    # Phase 2 – deep Adam optimisation on the current best
    if best_seq is not None and time.time() < deadline - 40:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam final] new best {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – continuation L‑BFGS refinement
    if best_seq is not None and time.time() < deadline - 15:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap random hill‑climber for the final seconds
    if best_seq is not None and time.time() < deadline - 3:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never trigger)
    if best_seq is None:
        best_seq = _uniform(n)

    # Ensure compliance with evaluator (clip to allowed range)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
