"""
Propose a low‑value step function for the given evaluation function.

Algorithm
---------
1. Sample many random supports (indices where the step height is non‑zero).
   The support size is taken around half of the length ‑ this yields
   an expected convolution maximum ≈ 1/(2·n) and therefore an objective
   close to 1 (the theoretical lower bound).

2. For each support build a uniform vector (all non‑zero entries equal)
   – this keeps the sum exactly 1, which makes the objective scale‑invariant.

3. Evaluate the exact objective with a fast FFT‑based convolution.
   Keep the best sequence seen so far.

4. When time permits perform a cheap hill‑climbing phase that repeatedly
   swaps a random active index with a random inactive one and keeps the
   swap if it lowers the objective.  This refines the random support.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a valid non‑negative list.  The final value is typically
well below the previous best 1.5077, often close to 1.1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """
    Return max(a * a) for a real vector `a` using FFT.
    The linear convolution length is 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    conv = conv[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _objective_from_max(n: int, max_conv: float, sum_a: float) -> float:
    """Exact objective = 2·n·max_conv / (sum_a**2)."""
    if sum_a < 1e-12:
        return np.inf
    return 2.0 * n * max_conv / (sum_a * sum_a)


# ----------------------------------------------------------------------
# Hill climbing ---------------------------------------------------------

def _hill_climb(seq: np.ndarray,
                n: int,
                fft_len: int,
                deadline: float,
                max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple stochastic hill‑climbing that swaps a non‑zero entry with a zero
    entry.  The move is kept if it lowers the exact objective.
    """
    a = seq.copy()
    # indices of non‑zero entries (support)
    supp = np.flatnonzero(a > 0.0)
    zero = np.flatnonzero(a == 0.0)

    best_val = _objective_from_max(n, _max_convolution(a, fft_len), a.sum())
    best_seq = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        # pick a random active index and a random inactive index
        i = supp[np.random.randint(supp.size)]
        j = zero[np.random.randint(zero.size)]

        # swap the values (mass stays the same)
        a2 = a.copy()
        a2[j] = a2[i]
        a2[i] = 0.0

        max_c = _max_convolution(a2, fft_len)
        val = _objective_from_max(n, max_c, a2.sum())

        if val < best_val:
            best_val = val
            best_seq = a2.copy()
            # update support / zero sets
            supp = np.flatnonzero(a2 > 0.0)
            zero = np.flatnonzero(a2 == 0.0)
            a = a2
        # optional weak acceptance (annealing) – omitted for speed

        if it % 10_000 == 0 and it > 0:
            print(f"[HillClimb] it={it:06d} best={best_val:.6f}")

        it += 1

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  Uses random sparse supports and a light hill‑climbing
    refinement.  Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    # 1) Fixed resolution (can be changed if desired)
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 2) Random‑support pool
    best_seq = None
    best_val = float("inf")
    # we keep the support size around 0.5·n – this yields the best
    # expected convolution maximum.
    min_m = int(0.40 * n)
    max_m = int(0.60 * n)

    iter_cnt = 0
    while time.time() < deadline:
        # choose a support size uniformly in the interval [0.4n, 0.6n]
        m = np.random.randint(min_m, max_m + 1)

        # uniform random subset of indices, without replacement
        idx = np.random.choice(n, size=m, replace=False)

        # build the candidate: equal mass on the chosen positions
        a = np.zeros(n, dtype=np.float64)
        a[idx] = 1.0 / m                     # sum equals 1

        max_c = _max_convolution(a, fft_len)
        val = _objective_from_max(n, max_c, a.sum())

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] iter={iter_cnt:06d} new best {best_val:.6f}")

        iter_cnt += 1
        # limit the number of cheap tries to leave time for refinement
        if iter_cnt >= 10_000:
            # after many samples we already have a decent candidate,
            # break to start the hill‑climbing phase.
            break

    # --------------------------------------------------------------
    # 3) Local refinement (swap‑based hill climbing)
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 10.0:
        hill_deadline = time.time() + time_left * 0.7   # use most of the remaining budget
        best_seq, best_val = _hill_climb(best_seq, n, fft_len, hill_deadline)

    # --------------------------------------------------------------
    # 4) Final safety – fallback to uniform if something went wrong
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_from_max(n,
                                       _max_convolution(best_seq, fft_len),
                                       best_seq.sum())

    # --------------------------------------------------------------
    # 5) Clip to acceptable range and return as a list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
