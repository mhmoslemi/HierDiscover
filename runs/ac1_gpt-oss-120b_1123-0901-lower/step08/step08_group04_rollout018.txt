def _targeted_sa(a0, time_limit, start_temp=0.02, max_frac=0.3):
    a = a0.copy()
    n = len(a)
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)
    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val
    start = time.time()
    while time.time() < time_limit:
        # compute index of max conv:
        k_opt = int(np.argmax(conv))
        # compute contributions to that conv: We need to compute for each i value of a_i * a_{k_opt - i}, limited to within bounds.
        # Let's compute reversed index array:
        # produce vector r = np.zeros_like(a)
        # for i in range(n): j = k_opt - i; if 0 <= j < n: contribution = a[i] * a[j]; keep
        # To avoid loops, compute via broadcasting? Use vectorized: indices = np.arange(n); j = k_opt - indices; mask = (j>=0) & (j<n); contrib = np.where(mask, a* a[j], 0.0)
        # compute product array
        idx = np.arange(n)
        j = k_opt - idx
        mask = (j >= 0) & (j < n)
        prod = np.where(mask, a[idx] * a[j], 0.0)
        # Choose i among indices where prod > 0 weighted by prod (i.e., more contribution)
        if mask.sum() == 0:
            # fallback: random i
            i = np.random.randint(n)
        else:
            probs = prod / prod.sum()
            i = np.random.choice(np.arange(n)[mask], p=probs[mask])  # product distribution
        # Choose random j target not equal i
        j_target = np.random.randint(n-1)
        if j_target >= i:
            j_target += 1
        # Transfer fraction of mass from i to j_target
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue
        # apply delta with _apply_delta
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j_target, delta)
        new_val = conv.max()
        # compute temperature:
        elapsed = time.time() - start
        total = time_limit - start  # but start is timestamp, we need overall time allowed.
        # Actually compute remaining = time_limit - time.time()
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))  # linear
        # accept if new_val <= cur_val else with probability exp(-(new_val-cur_val)/temp)
        diff = new_val - cur_val
        if diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j_target, -delta)
            _apply_delta(a, conv, i, delta)
        # optionally, every 50000 iterations, print status
    return best_a, best_val

"""
Improved step‑function search.

The pipeline is:

1. Build a diverse pool of warm‑starts (uniform, Dirichlet, anti‑pair,
   triangular, Gaussian‑edge, cosine‑edge …).  The best of them (by the
   exact objective) is kept.

2. Continuous refinement:
   – Adam on a smooth‑max (log‑sum‑exp) surrogate.
   – Exponentiated‑gradient sub‑gradient iteration.
   Both phases run as long as the time budget allows.

3. Two‑stage simulated‑annealing:
   – A long “random‑move” SA (identical to the previous implementation)
   – A short “targeted” SA that repeatedly reduces the contribution to the
     currently worst convolution entry by moving mass away from the most
     responsible index.

4. Final normalisation and return.

The code respects the 1000 s budget, prints progress and stores the best
solution in the global variable ``height_sequence_1`` for warm‑starting
future runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum a = 1) -------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
def _warm_start(n: int):
    """Adapt a previously saved solution (`height_sequence_1`) if it exists."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _cosine_edges_candidate(n: int) -> np.ndarray:
    """Two‑lobe cosine – peaks at both ends, zero in the middle."""
    i = np.arange(n, dtype=np.float64)
    # 1+cos(2π i/(n‑1)) is 2 at the ends, 0 in the middle.
    shape = 1.0 + np.cos(2.0 * np.pi * i / (n - 1))
    shape[shape < 0] = 0.0
    shape /= shape.sum()
    return shape

# Optimisation primitives -----------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of constraints (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental convolution updates (used by SA) -------------------------
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.03,
    max_frac: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Classic random‑move simulated annealing.  Moves are transfers of a
    random fraction of the mass from one position to another.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    total = time_limit - start
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / total)

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 50000 == 0:
            print(f"[RndSA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


def _targeted_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.01,
    max_frac: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Fine‑grained SA that always attacks the currently worst convolution
    entry.  One mass transfer is performed from the index that contributes
    most to the maximal entry to a random destination.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    total = time_limit - start
    while time.time() < time_limit:
        # index where convolution attains its maximum
        k_opt = int(np.argmax(conv))

        # find the i which gives the largest product a[i] * a[k_opt-i]
        lo = max(0, k_opt - (n - 1))
        hi = min(k_opt, n - 1)
        idx = np.arange(lo, hi + 1)
        j = k_opt - idx
        prod = a[idx] * a[j]
        i = idx[np.argmax(prod)]

        # random destination (different from i)
        j_dest = np.random.randint(n - 1)
        if j_dest >= i:
            j_dest += 1

        # move a small random fraction of the mass at i
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j_dest, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / total)

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j_dest, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 20000 == 0:
            print(f"[TgtSA it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The routine mixes diverse warm‑starts, Adam + sub‑gradient
    continuous optimisation, a long random‑move SA and a short
    targeted SA for fine‑tuning.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – 4096 provides a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a rich pool of initial candidates
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # a few rotations of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))

    for alpha in (0.2, 0.5, 0.8, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    for alpha in (0.2, 0.7, 1.5):
        candidates.append(_anti_pair_candidate(n, alpha))

    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    candidates.append(_triangle_candidate(n))
    candidates.append(_cosine_edges_candidate(n))

    # --------------------------------------------------------------
    # 2) Pick the best starter (cheap exact evaluation)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # keep a few seconds for the final SA phases
    sa_start_deadline = deadline - 60.0   # reserve at least 60 s for SA
    if time.time() < sa_start_deadline:
        # ---- Adam phase (smooth‑max)
        adam_deadline = min(sa_start_deadline - 30.0,
                            time.time() + 180.0)   # at most ≈180 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # ---- Sub‑gradient refinement
        sub_deadline = min(sa_start_deadline - 5.0,
                           time.time() + 300.0)    # at most ≈300 s
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=500_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long random‑move SA + short targeted SA
    remaining = deadline - time.time()
    if remaining > 10.0:
        # Allocate most of the remaining budget to the random SA,
        # keep roughly 30 s for the focused SA.
        targeted_time = min(30.0, remaining * 0.2)
        random_sa_time = remaining - targeted_time - 5.0
        if random_sa_time > 5.0:
            print(f"[SA] random phase for {random_sa_time:.1f}s")
            sa_seq, sa_val = _incremental_sa(
                best_seq,
                time_limit=time.time() + random_sa_time,
                start_temp=0.04,
                max_frac=0.45,
            )
            if sa_val < best_val:
                best_val = sa_val
                best_seq = sa_seq.copy()
                print(f"[RndSA] improved to {sa_val:.6f}")

        # Targeted fine‑tuning
        if targeted_time > 5.0:
            print(f"[SA] targeted phase for {targeted_time:.1f}s")
            targ_seq, targ_val = _targeted_sa(
                best_seq,
                time_limit=time.time() + targeted_time,
                start_temp=0.015,
                max_frac=0.25,
            )
            if targ_val < best_val:
                best_val = targ_val
                best_seq = targ_seq.copy()
                print(f"[TgtSA] improved to {targ_val:.6f}")

    # --------------------------------------------------------------
    # 5) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    best_seq /= best_seq.sum()                     # ensure sum ≈ 1 (scale‑invariant)

    # store globally for possible warm‑starts in later runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
