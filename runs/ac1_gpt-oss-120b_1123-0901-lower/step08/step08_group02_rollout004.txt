        temp = max(0.0, (frac) * 0.001) # small maybe

        j = random index != i

                if np.random.rand() < np.exp(-(val2 - cur_val) / temp):

def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, lam=100.0, max_iter=200):
    from scipy.optimize import minimize
    a0 = np.maximum(a0, 0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0/n)
    a0 = a0 / a0.sum()
    logits0 = np.log(a0 + 1e-12)
    # define objective function and gradient
    def func(logits):
        a = np.exp(logits - np.max(logits))
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        smoothmax, w = _smoothmax_weights(conv, lam)
        obj = 2.0 * n * smoothmax
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        grad_log = 2.0 * n * grad_log
        return obj, grad_log
    res = minimize(lambda logits: func(logits), logits0, method='L-BFGS-B', jac=True,
                   options={'maxiter': max_iter, 'disp': False})
    # compute final a
    logits_opt = res.x
    a_opt = np.exp(logits_opt - np.max(logits_opt))
    a_opt = a_opt / a_opt.sum()
    # compute final objective
    conv_opt = _full_convolution(a_opt, conv_fft_len)
    val_opt = 2.0 * n * conv_opt.max()
    return a_opt, val_opt

"""
Propose a step‑function (non‑negative heights) that minimizes
    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm combines a diverse set of seeds, a smooth‑max Adam
optimiser, a high‑precision L‑BFGS‑B refinement (using the same
smooth‑max surrogate), an exact‑max sub‑gradient phase and a cheap
local hill‑climber.  A simple hierarchical seed (optimisation on a
smaller grid and up‑sampling) is also added.  All stages respect the
global time budget (default 1000 s) and the best feasible sequence is
always returned.
"""

import time
import numpy as np
import random

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = logits.max()
    e = np.exp(logits - m)
    return e / e.sum()


def _smoothmax_weights(conv: np.ndarray, lam: float):
    """
    Smooth max using log‑sum‑exp with temperature lam.
    Returns (smooth_max, weight_vector) where
        smooth_max = (1/lam)·log Σ exp(lam·conv_i)
        weight_vector = softmax(lam·conv)
    """
    cmax = conv.max()
    # shift for numerical stability
    shifted = np.exp(lam * (conv - cmax))
    w_sum = shifted.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
        smooth = cmax
    else:
        w = shifted / w_sum
        smooth = cmax + np.log(w_sum) / lam
    return smooth, w

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Generate a modest but diverse pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # edge‑exponential families
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # a few Beta‑family shapes
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
                 (3.0, 3.0), (0.2, 0.2), (5.0, 1.0), (1.0, 5.0)]:
        x = np.linspace(0.0, 1.0, n, dtype=np.float64)
        shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # hierarchical small‑grid seed (size n/4, then up‑sample)
    n_small = n // 4
    a_small = np.random.dirichlet(np.ones(n_small))
    a_large = np.repeat(a_small, 4)
    a_large = a_large[:n]
    a_large /= a_large.sum()
    cand.append(a_large)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with equally spaced spikes."""
    out = []
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    return out

def _adam_optimize_smooth(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          lam: float = 12.0,
                          lr: float = 0.05) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best distribution (sum≈1) encountered and its exact value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max projection
        a = _softmax(logits)

        # convolution and smooth‑max
        conv = _full_convolution(a, conv_fft_len)
        smooth, w = _smoothmax_weights(conv, lam)

        # exact objective (max‑based) used for bookkeeping
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam] it={it:06d}  val={val:.6f}")

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # small random shake near the end
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    deadline: float,
                    lam: float = 80.0,
                    max_iter: int = 250) -> tuple[np.ndarray, float]:
    """
    High‑precision L‑BFGS‑B optimisation of the smooth‑max surrogate.
    Returns the refined distribution and its exact (max‑based) value.
    """
    try:
        from scipy.optimize import minimize
    except Exception as exc:
        print("SciPy not available – skipping L‑BFGS‑B phase.")
        return a0, 2.0 * n * _full_convolution(a0, conv_fft_len).max()

    a0 = np.maximum(a0, 0.0)
    if a0.sum() == 0:
        a0 = np.full(n, 1.0 / n)
    a0 = a0 / a0.sum()
    logits0 = np.log(a0 + 1e-12)

    def func(logits, n, lam, conv_fft_len, corr_fft_len):
        """Return (smooth objective, gradient w.r.t. logits)."""
        a = _softmax(logits)
        conv = _full_convolution(a, conv_fft_len)
        smooth, w = _smoothmax_weights(conv, lam)
        obj = 2.0 * n * smooth

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        grad_log = 2.0 * n * grad_log
        return obj, grad_log

    # minimise – we pass arguments via the `args` tuple
    res = minimize(func,
                   logits0,
                   args=(n, lam, conv_fft_len, corr_fft_len),
                   method='L-BFGS-B',
                   jac=True,
                   options={'maxiter': max_iter, 'disp': False})

    logits_opt = res.x
    a_opt = _softmax(logits_opt)
    conv_opt = _full_convolution(a_opt, conv_fft_len)
    val_opt = 2.0 * n * conv_opt.max()
    return a_opt, val_opt


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation (exponentiated‑gradient).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad] it={it:06d}  val={val:.6f}")

        # active set (indices where max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from one
    index to another, keep the move only if the exact objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.25)  # move up to 25 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 15_000 == 0:
                print(f"[Local] it={it:06d}  improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)      # = 8192
    corr_fft_len = _next_pow2(3 * n - 2)      # = 16384

    # --------------------------------------------------------------
    # optional warm‑start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # build a rather large pool of diverse seeds
    pool = (_initial_candidates(n, warm) +
            _spike_candidates(n) +
            [np.random.dirichlet(np.ones(n)) for _ in range(5)])

    # keep a few best seeds after a cheap evaluation
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand = cand / cand.sum()

        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if time.time() >= deadline:
            break
    print(f"[Init] best seed value = {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – moderate‑budget Adam runs on all seeds (≈15 % of budget)
    per_candidate_budget = (deadline - time.time()) * 0.15 / max(1, len(pool))
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 10.0, time.time() + per_candidate_budget)
        a_adam, val_adam = _adam_optimize_smooth(
            a0, n, conv_fft_len, corr_fft_len, local_deadline,
            max_iters=120_000, lam=12.0, lr=0.05)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Stage1‑Adam] cand {cid} improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – intensive refinement of the current best (≈55 % of budget)
    while time.time() < deadline - 30.0:
        # ---- Adam with a slightly larger temperature ----
        a_adam, val_adam = _adam_optimize_smooth(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000, lam=20.0, lr=0.04)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Stage2‑Adam] new best {val_adam:.6f}")

        # ---- L‑BFGS‑B phases with increasing sharpness ----
        for lam in (30.0, 80.0, 200.0):
            if time.time() >= deadline - 30.0:
                break
            a_lb, val_lb = _lbfgs_optimize(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, lam=lam, max_iter=200)
            if val_lb < best_val:
                best_val, best_seq = val_lb, a_lb.copy()
                print(f"[Stage2‑LBFGS] lam={lam:.0f} new best {val_lb:.6f}")

        # ---- sub‑gradient exact‑max phase ----
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=150_000, eta0=0.80)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Stage2‑Sub] new best {val_sub:.6f}")

        # ---- cheap random local search ----
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=60_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Stage2‑Local] improved best to {val_loc:.6f}")

        # occasional random perturbation of logits to escape plateaus
        logits = np.log(best_seq + 1e-12)
        logits += np.random.randn(n) * 0.07
        a_pert = _softmax(logits)
        a_pert_adam, val_pert = _adam_optimize_smooth(
            a_pert, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=80_000, lam=25.0, lr=0.045)
        if val_pert < best_val:
            best_val, best_seq = val_pert, a_pert_adam.copy()
            print(f"[Perturb] new best {val_pert:.6f}")

    # --------------------------------------------------------------
    # Fallback – make sure something is returned
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # expose for possible warm‑starts of a next call
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
