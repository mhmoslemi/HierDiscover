"""
Propose a step‑function (non‑negative vector on the simplex) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)²

using a multi‑stage search:

1.  Build a rich pool of analytically‑defined shapes (Beta‐densities,
    edge‑power families, exponentials, spikes and random Dirichlet draws).
2.  Evaluate all candidates cheaply (FFT‑based max‑convolution) and keep the
    best few.
3.  Refine the survivors with two complementary continuous optimisers:
        • Adam on a smooth‑max surrogate (log‑sum‑exp) with a slowly
          increasing temperature.
        • Exact‑max sub‑gradient (exponentiated‑gradient) updates.
4.  Perform a cheap hill‑climber that moves a random fraction of the mass
    between two positions.
5.  Symmetrise the final vector (a ← (a+a[::-1])/2) – the objective is
    invariant under reversal and symmetry often reduces the max‑convolution.

The routine respects the global time budget (default 1000 s) and always
returns the best feasible sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value 2·n·max(a∗a)/(sum(a)²)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # start on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                                 # a bit more conservative
    lam = 0.01                                # initial temperature (soft)
    lam_factor = 1.025                        # anneal slowly

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule for the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (slow increase)
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation (exponentiated‑gradient mirror descent).
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-13)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Extremely cheap hill‑climber: move a random fraction of the mass from i → j
    and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.25)   # up to 25 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 15000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (∑a)²

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (adapt to remaining time – larger n gives more freedom)
    # For a 1000 s budget we keep n=4096 (the size used for previous best).
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)      # 8192
    corr_fft_len = _next_pow2(3 * n - 2)      # 12288

    # ------------------------------------------------------------------
    # 1) build a diverse pool of analytic/init candidates
    cand_pool = []

    # Warm‑start from a previously stored solution, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            cand_pool.append(w)

    # Uniform
    cand_pool.append(np.full(n, 1.0 / n))

    # Positive half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand_pool.append(s)

    # Edge‑exponential families (left‑right symmetry)
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand_pool.append(shape)

    # Edge‑power families (product^(−p))
    idx = np.arange(n, dtype=np.float64)
    prod = idx * (n - 1 - idx) + 1.0          # avoid zero
    for p in np.linspace(0.2, 2.5, 12):
        shape = prod ** (-p)
        shape /= shape.sum()
        cand_pool.append(shape)

    # Symmetric beta densities: a = b = α   (α ∈ (0.2,1.2))
    x = (np.arange(n) + 0.5) / n
    for a in np.linspace(0.2, 1.2, 12):
        # pdf ∝ x^{a-1} (1-x)^{a-1}
        shape = (x ** (a - 1.0)) * ((1.0 - x) ** (a - 1.0))
        shape /= shape.sum()
        cand_pool.append(shape)

    # Asymmetric beta densities (random α,β)
    for _ in range(15):
        a_beta = np.random.rand() * 0.9 + 0.1
        b_beta = np.random.rand() * 0.9 + 0.1
        shape = (x ** (a_beta - 1.0)) * ((1.0 - x) ** (b_beta - 1.0))
        shape /= shape.sum()
        cand_pool.append(shape)

    # Random Dirichlet (various concentrations)
    for _ in range(30):
        alpha = np.random.rand() * 4.0 + 0.1   # α ∈ [0.1,4.1]
        cand_pool.append(np.random.dirichlet(np.full(n, alpha)))

    # Sparse spike patterns (equally spaced spikes, various K)
    for K in (32, 64, 128, 256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, K)).astype(int)
        a[pos] = 1.0 / K
        cand_pool.append(a)

    # ------------------------------------------------------------------
    # 2) cheap evaluation of the whole pool, keep the top few
    best_vals = []
    for vec in cand_pool:
        if time.time() > deadline:
            break
        a = np.maximum(vec, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a /= s
        val = _objective(a, n, conv_fft_len)
        best_vals.append((val, a))

    # sort by objective, keep the 7 best for deep refinement
    best_vals.sort(key=lambda x: x[0])
    survivors = [a for (_, a) in best_vals[:7]]
    best_val = best_vals[0][0] if best_vals else float("inf")
    best_seq = best_vals[0][1].copy() if best_vals else np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # 3) Deep refinement of survivors (alternating Adam / sub‑gradient)
    while time.time() < deadline - 15.0:
        # pick the current best as start point
        start = best_seq.copy()

        # run a short Adam phase
        a_adam, val_adam = _adam_optimize(
            start, n, conv_fft_len, corr_fft_len, deadline, max_iters=150_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Refine] Adam improved to {val_adam:.6f}")

        # run a sub‑gradient phase
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Refine] Subgrad improved to {val_sub:.6f}")

        # occasional random perturbation of the logits to escape plateaus
        logits = np.log(best_seq + 1e-12)
        logits += np.random.randn(n) * 0.06
        pert = np.exp(logits - np.max(logits))
        pert /= pert.sum()
        a_tmp, val_tmp = _adam_optimize(
            pert, n, conv_fft_len, corr_fft_len, deadline, max_iters=80_000
        )
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, a_tmp.copy()
            print(f"[Refine] Perturb Adam improved to {val_tmp:.6f}")

        # stop early if the remaining time is tiny
        if time.time() > deadline - 8.0:
            break

    # ------------------------------------------------------------------
    # 4) Final cheap random local search
    a_loc, val_loc = _local_random_search(
        best_seq, n, conv_fft_len, deadline, max_moves=60_000
    )
    if val_loc < best_val:
        best_val, best_seq = val_loc, a_loc.copy()
        print(f"[Local] final improvement to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Symmetrise (the objective is invariant under reversal)
    best_seq = (best_seq + best_seq[::-1]) * 0.5
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
