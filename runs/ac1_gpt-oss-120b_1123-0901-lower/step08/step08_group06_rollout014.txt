def _genetic_algorithm(base_seq, n, conv_fft_len, corr_fft_len, deadline):
    # Initialize population
    pop_size = 100
    elite_size = 10
    # generate random individuals
    pop = []
    for i in range(pop_size):
        # random Dirichlet with random alpha
        log_alpha = np.random.uniform(-2, 2) # adjust
        alpha = 10**log_alpha
        a = np.random.dirichlet(alpha * np.ones(n))
        # optionally apply random ordering: choose ordering type randomly
        order_type = np.random.choice(['none','anti_pair','spread','bitrev'])
        if order_type == 'anti_pair':
            a = _anti_pair_arrangement(a)
        elif order_type == 'spread':
            step = np.random.randint(1, n, dtype=int)
            if step % 2 == 0:
                step += 1
            a = _spread_arrangement(a, step)
        elif order_type == 'bitrev':
            perm = _bitrev_permutation(n)
            a = a[perm]
        # else none
        pop.append(a)

    # include base_seq also
    if base_seq is not None:
        pop[0] = np.array(base_seq, dtype=np.float64)

    # Evaluate all
    fitness = np.array([_objective(a) for a in pop])
    best_idx = fitness.argmin()
    best_a = pop[best_idx].copy()
    best_val = fitness[best_idx]

    it = 0
    while time.time() < deadline and it < 2000: # number of generations maybe
        # select elites
        elite_idx = fitness.argsort()[:elite_size]
        elites = [pop[i] for i in elite_idx]

        # create new population
        new_pop = elites.copy()
        # generate children until pop_size
        while len(new_pop) < pop_size:
            # pick two parents randomly from elites
            p1, p2 = elites[np.random.randint(elite_size)], elites[np.random.randint(elite_size)]
            w = np.random.rand()
            child = w * p1 + (1 - w) * p2
            # add small Dirichlet noise
            epsilon = 0.02
            if epsilon > 0:
                noise = np.random.dirichlet(np.ones(n))
                child = (1 - epsilon) * child + epsilon * noise
            # mutation: swap or random perturbation
            # randomly choose to either swap two entries or perturb a randomly chosen index.
            if np.random.rand() < 0.3:
                i, j = np.random.randint(n, size=2)
                child[i], child[j] = child[j], child[i]
            else:
                # Gaussian perturbation: add small Gaussian noise and clip
                idx = np.random.randint(n)
                child[idx] += np.random.normal(scale=0.01)
                child[idx] = max(child[idx], 0.0)
            child = child / child.sum()
            new_pop.append(child)

        # evaluate new population
        fitness = np.array([_objective(a) for a in new_pop])
        # update best
        cur_best_idx = fitness.argmin()
        cur_best_val = fitness[cur_best_idx]
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_a = new_pop[cur_best_idx].copy()
            print(f"[GA gen={it}] new best {best_val:.6f}")

        # prepare for next iteration
        pop = new_pop
        it += 1

    return best_a, best_val

def _alternating_extremes_arrangement(vals):
    n = len(vals)
    a = np.empty(n)
    left = 0
    right = n-1
    for i, val in enumerate(vals):
        if i % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
    return a

"""
propose_candidate
-----------------

A time‑bounded, multi‑stage optimiser for the non‑negative step‑function
objective

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm works on a modest size (n=2048) to explore the search space
quickly and then refines the best candidate on a finer grid (n=8192).

Stages
~~~~~~
1. Warm‑start (global variable ``height_sequence_1``) and a few handcrafted
   seeds (uniform, triangular, Gaussian‑edge).
2. Massive random Dirichlet sampling – each sample is sorted descending
   and examined under several spreading permutations:
   * anti‑pair (largest ↔ smallest),
   * random “step” permutation (a coprime odd step),
   * alternating‑half interleaving,
   * identity (sorted order).
3. A small genetic algorithm (population≈80, elite≈10) that mixes elite
   individuals, adds Dirichlet noise and occasional swaps.
4. Smooth‑max optimisation with L‑BFGS (log‑sum‑exp surrogate) and a
   slowly increasing temperature λ.
5. Projected additive sub‑gradient descent (Euclidean projection onto the
   simplex) for fine tuning.
6. Upsampling to 8192 points (linear interpolation) and a final L‑BFGS +
   sub‑gradient pass.

All heavy work respects the global 1000 s budget; the function returns the
best sequence found (list of floats) before the deadline.
"""

import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Exact F(a) = 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm start handling
def _warm_start(target_n: int):
    """Return a truncated / padded version of the global `height_sequence_1`."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > target_n:
        warm = warm[:target_n]
    elif warm.shape[0] < target_n:
        warm = np.pad(warm, (0, target_n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()

# Basic deterministic seeds
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)  # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape

# Ordering / permutation helpers
def _anti_pair_arrangement_from_sorted(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest (anti‑pair) ordering."""
    n = vals.shape[0]
    half = n // 2
    a = np.empty(n, dtype=np.float64)
    a[:half] = vals[:half]
    a[-half:] = vals[-half:][::-1]
    if n % 2:
        a[half] = vals[half]
    return a


def _interleaved_arrangement_random(vals: np.ndarray) -> np.ndarray:
    """Spread sorted values using a random odd step (coprime to n)."""
    n = vals.shape[0]
    step = np.random.randint(1, n, dtype=int)
    if step % 2 == 0:
        step += 1
        if step >= n:
            step = 1
    perm = (np.arange(n) * step) % n
    a = np.empty(n, dtype=np.float64)
    a[perm] = vals
    return a


def _alternating_half_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Interleave largest values alternately into the first and second half.
    Assumes n is even (true for our powers of two).
    """
    n = vals.shape[0]
    half = n // 2
    a = np.empty(n, dtype=np.float64)
    left = 0
    right = half
    for i, v in enumerate(vals):
        if i % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right += 1
    return a

# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()                     # softmax → a on simplex

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # stable log‑sum‑exp around max_c
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, Σw = 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0),
                    max_iter_stage: int = 300) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # allocate a modest number of iterations respecting remaining time
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 5)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        # retrieve a from logits
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val

# Additive projected sub‑gradient optimisation
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 400_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """Projected gradient on the exact max‑convolution objective."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it}] new best {val:.6f}")

    return best_a, best_val

# Simple genetic algorithm (population based)
def _genetic_algorithm(base_seq: np.ndarray, n: int,
                       conv_fft_len: int, corr_fft_len: int,
                       deadline: float,
                       pop_size: int = 80,
                       elite_size: int = 10,
                       generations: int = 30) -> tuple[np.ndarray, float]:
    """Small GA that mixes elite individuals and adds Dirichlet noise."""
    # initialise population
    pop = []
    for _ in range(pop_size):
        # random concentration
        log_alpha = np.random.uniform(-2.0, 2.0)
        alpha = 10.0 ** log_alpha
        a = np.random.dirichlet(alpha * np.ones(n))
        # pick a random ordering strategy
        ordering = np.random.choice([
            _anti_pair_arrangement_from_sorted,
            _interleaved_arrangement_random,
            _alternating_half_arrangement,
            lambda x: x  # identity (sorted descending)
        ])
        # use sorted descending values for deterministic ordering
        a_sorted = np.sort(a)[::-1]
        a = ordering(a_sorted)
        pop.append(a)

    # replace the first individual with the supplied base sequence (if any)
    if base_seq is not None:
        pop[0] = base_seq.copy()

    fitness = np.array([_objective(a) for a in pop])
    best_idx = fitness.argmin()
    best_a, best_val = pop[best_idx].copy(), fitness[best_idx]
    print(f"[GA init] best {best_val:.6f}")

    for gen in range(generations):
        if time.time() >= deadline:
            break

        # select elites
        elite_idx = fitness.argsort()[:elite_size]
        elites = [pop[i] for i in elite_idx]

        # create new population
        new_pop = elites.copy()
        while len(new_pop) < pop_size:
            # blend two elite parents
            p1, p2 = elites[np.random.randint(elite_size)], elites[np.random.randint(elite_size)]
            w = np.random.rand()
            child = w * p1 + (1 - w) * p2

            # inject small Dirichlet noise
            epsilon = 0.02
            if epsilon > 0:
                noise = np.random.dirichlet(np.ones(n))
                child = (1 - epsilon) * child + epsilon * noise

            # occasional mutation: swap two entries or perturb one entry
            if np.random.rand() < 0.3:
                i, j = np.random.randint(n, size=2)
                child[i], child[j] = child[j], child[i]
            else:
                idx = np.random.randint(n)
                child[idx] += np.random.normal(scale=0.01)
                child[idx] = max(child[idx], 0.0)

            child = child / child.sum()
            new_pop.append(child)

        pop = new_pop
        fitness = np.array([_objective(a) for a in pop])

        cur_best_idx = fitness.argmin()
        cur_best_val = fitness[cur_best_idx]
        if cur_best_val < best_val:
            best_val, best_a = cur_best_val, pop[cur_best_idx].copy()
            print(f"[GA gen={gen}] new best {best_val:.6f}")

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Small problem size for cheap exploration
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Warm start (if provided by the harness)
    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    # --------------------------------------------------------------
    # Simple deterministic starters
    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # --------------------------------------------------------------
    # Massive random pool with several ordering strategies
    n_samples = 8000
    ordering_funcs = [
        _anti_pair_arrangement_from_sorted,
        _interleaved_arrangement_random,
        _alternating_half_arrangement,
        lambda x: x,  # identity (sorted)
    ]

    print(f"Generating {n_samples} random Dirichlet candidates …")
    for idx in range(n_samples):
        if time.time() >= deadline:
            break
        # random concentration α in a wide log‑range
        log_alpha = np.random.uniform(-2.0, 1.5)   # α∈[0.01,≈31.6]
        alpha = 10.0 ** log_alpha
        vals = np.random.dirichlet(alpha * np.ones(n))
        sorted_vals = np.sort(vals)[::-1]         # descending

        for ordering in ordering_funcs:
            cand = ordering(sorted_vals.copy())
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Random pool] new best {val:.6f} at sample {idx}")

    # --------------------------------------------------------------
    # Genetic algorithm refinement (if time permits)
    if time.time() < deadline - 200.0:
        seq_ga, val_ga = _genetic_algorithm(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            deadline - 150.0,
                                            pop_size=80, elite_size=10,
                                            generations=30)
        if val_ga < best_val:
            best_val, best_seq = val_ga, seq_ga.copy()
            print(f"[GA] improved to {val_ga:.6f}")

    # --------------------------------------------------------------
    # L‑BFGS smooth‑max refinement
    if time.time() < deadline - 80.0:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         deadline - 60.0,
                                         lam_schedule=(5, 15, 40, 120, 300),
                                         max_iter_stage=300)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # Additive projected sub‑gradient fine‑tuning
    if time.time() < deadline - 30.0:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          deadline - 10.0,
                                                          max_iters=400_000,
                                                          eta0=0.5)
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # Upsample to a finer grid (8192) and final refinement
    if time.time() < deadline - 5.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best word
        x_small = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_small, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # quick L‑BFGS on the big grid
        seq_big, val_big = _lbfgs_optimize(a_big, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline,
                                           lam_schedule=(5, 20, 80, 200, 500),
                                           max_iter_stage=200)
        if val_big < best_val:
            best_val, best_seq = val_big, seq_big.copy()
            print(f"[Big L‑BFGS] improved to {val_big:.6f}")

        # final projected sub‑gradient on the big grid (if time left)
        if time.time() < deadline - 2.0:
            seq_big2, val_big2 = _additive_subgradient_optimize(best_seq, n_big,
                                                                conv_fft_len_big,
                                                                corr_fft_len_big,
                                                                deadline,
                                                                max_iters=300_000,
                                                                eta0=0.5)
            if val_big2 < best_val:
                best_val, best_seq = val_big2, seq_big2.copy()
                print(f"[Big AddSubgrad] improved to {val_big2:.6f}")

    # --------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
