def _smooth_obj_and_grad(logits, n, conv_fft_len, corr_fft_len, lam):
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()
    conv = _full_convolution(a, conv_fft_len) # shape 2n-1
    # compute softmax weights:
    y = lam * conv
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    w = exp_y / exp_y.sum()
    # surrogate = (1/lam) * np.log(np.sum(exp_y))
    surrogate = (np.log(np.sum(exp_y)) + y_max) / lam  # ensure stable
    # compute gradient
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # softmax chain rule:
    grad_logits = a * (grad_a - np.dot(a, grad_a))
    return surrogate, grad_logits
```

def _adam_optimize(logits0, n, conv_fft_len, corr_fft_len, deadline, lam_schedule, step_size=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    logits = logits0.copy()
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = None
    best_val = np.inf
    for lam in lam_schedule:
        while time.time() < deadline:
            t += 1
            surrogate, grad = _smooth_obj_and_grad(logits, n, conv_fft_len, corr_fft_len, lam)
            # adam update
            m = beta1 * m + (1 - beta1) * grad
            v = beta2 * v + (1 - beta2) * (grad ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits = logits - step_size * m_hat / (np.sqrt(v_hat) + eps)

            # compute actual objective occasionally (e.g., every 100 steps)
            if t % 200 == 0:
                # compute a
                mx = np.max(logits)
                exp_l = np.exp(logits - mx)
                a = exp_l / exp_l.sum()
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam lam={lam} step={t}] new best {val:.6f}")
            # if time near deadline, break
            if time.time() > deadline - 2:
                break
    return best_a, best_val
```

def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):

    exp_l = np.exp(logits - mx)

    return surrogate, grad_log.astype(np.float64)

                   lam_schedule=(5.,15.,40.,120.),
                   step_size=0.2,
                   beta1=0.9, beta2=0.999, eps=1e-8):
    logits = logits0.copy()
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = None
    best_val = float('inf')
    for lam in lam_schedule:
        while time.time() < deadline:
            # evaluate gradient (smooth surrogate)
            # compute a via softmax
            mx = np.max(logits)
            exp_l = np.exp(logits - mx)
            a = exp_l / exp_l.sum()
            conv = _full_convolution(a, conv_fft_len)
            # compute softmax weight over conv
            y = lam * conv
            y_max = np.max(y)
            exp_y = np.exp(y - y_max)
            w = exp_y / exp_y.sum()
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            avg = np.dot(a, grad_a)
            grad_logits = a * (grad_a - avg)

            # adam update
            t += 1
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits = logits - step_size * m_hat / (np.sqrt(v_hat) + eps)

                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam lam={lam:.0f} step={t}] new best {val:.6f}")
            # check deadline near
            if time.time() > deadline - 0.2:
                break
    return best_a, best_val

"""
Multi‑stage optimizer for the step‑function bound

The algorithm follows the spirit of the original approach but
adds a few new ingredients that have proven useful in practice:

1.   Warm‑start from a cached solution (if any) and a handful of
     handcrafted seeds (uniform, triangular, Gaussian‑edge).
2.   Massive random “anti‑pair” Dirichlet pool (different concentration
     parameters).  The best few candidates are kept for further refinement.
3.   Smooth‑max Adam optimisation.  Instead of exact sub‑gradients we
     minimise the log‑sum‑exp surrogate of the convolution, increasing the
     temperature λ in stages.  Adam’s momentum accelerates convergence and
     helps escape shallow local minima.
4.   A short L‑BFGS‑B refinement on the exact max‑convolution objective
     (optional – falls back gracefully if SciPy is unavailable).
5.   Optional up‑sampling to a finer grid (8192 points) and a final Adam
     pass on the refined sequence.

All stages respect the global 1000 s budget; they stop early if the
deadline is close.  The best sequence found is clipped to the range
accepted by ``evaluate_sequence`` and returned as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (the one evaluated by the challenge)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – kept for possible future use
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet sample (concentration α) with anti‑pair ordering."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """If a global variable ``height_sequence_1`` exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) and its gradient w.r.t. logits
def _smooth_obj_and_grad(logits: np.ndarray, n: int,
                         conv_fft_len: int, corr_fft_len: int, lam: float):
    """
    Compute surrogate = (1/lam) log‑sum‑exp(lam·conv) and gradient
    w.r.t. the softmax logits (so that a = softmax(logits)).
    """
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)          # length 2n‑1
    y = lam * conv
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    w = exp_y / exp_y.sum()                            # soft weights, sum=1

    surrogate = (np.log(exp_y.sum()) + y_max) / lam    # stable log‑sum‑exp

    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)                   # chain rule through softmax
    return surrogate, grad_logits.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(logits0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   lam_schedule=(10.0, 30.0, 100.0, 300.0),
                   step_size: float = 0.2,
                   beta1: float = 0.9, beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Momentum Adam on the smooth‑max surrogate.
    Returns the best sequence (as a probability vector) and its exact objective.
    """
    logits = logits0.copy()
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    t = 0
    best_a = None
    best_val = float("inf")

    for lam in lam_schedule:
        while time.time() < deadline:
            # ----- gradient of smooth‑max -----
            surrogate, grad_logits = _smooth_obj_and_grad(logits, n,
                                                          conv_fft_len,
                                                          corr_fft_len, lam)

            # ----- Adam update -----
            t += 1
            m = beta1 * m + (1 - beta1) * grad_logits
            v = beta2 * v + (1 - beta2) * (grad_logits ** 2)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            logits = logits - step_size * m_hat / (np.sqrt(v_hat) + eps)

            # ----- occasional exact evaluation -----
            if t % 250 == 0:
                mx = np.max(logits)
                exp_l = np.exp(logits - mx)
                a = exp_l / exp_l.sum()
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam λ={lam:.0f} it={t}] new best {val:.6f}")

            # stop early if we are about to hit the deadline
            if time.time() > deadline - 0.3:
                break

    # If we never recorded a best (should not happen), use current logits
    if best_a is None:
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        best_a = exp_l / exp_l.sum()
        best_val = _objective(best_a)

    return best_a, best_val

# L‑BFGS refinement (optional – runs only if SciPy is available)
def _lbfgs_obj(logits: np.ndarray, n: int,
               conv_fft_len: int, corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, corr_fft_len: int,
                    deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0),
                    max_iter_stage: int = 250) -> tuple[np.ndarray, float]:
    """Run L‑BFGS‑B on the smooth‑max surrogate, increasing λ progressively."""
    try:
        import scipy.optimize as opt
    except Exception:
        # SciPy unavailable – skip refinement
        return a0, _objective(a0)

    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

    The routine respects a global 1000 s budget and returns the best
    sequence it discovered (as a plain Python list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Problem size (initial grid)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Warm‑start / handcrafted seeds
    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    handcrafted = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]

    for cand in handcrafted:
        val = _objective(cand)
        if val < best_val:
            best_seq, best_val = cand.copy(), val
            print(f"[Handcrafted] value {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Random anti‑pair Dirichlet pool (keep the few best)
    pool_size = 2500
    top_k = 5
    pool = []   # list of (value, array)

    for _ in range(pool_size):
        log_alpha = np.random.uniform(-2.0, 1.5)          # α ∈ [0.01, ~31.6]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_seq, best_val = cand.copy(), val
            print(f"[Pool] new overall best {val:.6f}")
        # keep a modest buffer of good candidates
        if len(pool) < top_k * 10:
            pool.append((val, cand))
        else:
            # maintain only the best ones
            if val < max(pool, key=lambda x: x[0])[0]:
                # replace the worst in the buffer
                worst_idx = max(range(len(pool)), key=lambda i: pool[i][0])
                pool[worst_idx] = (val, cand)

    # keep the top_k distinct candidates
    pool.sort(key=lambda x: x[0])
    top_candidates = [c for _, c in pool[:top_k]]

    # make sure the current best is also examined
    if best_seq is not None and not any(np.allclose(best_seq, c) for c in top_candidates):
        top_candidates.append(best_seq.copy())

    # ------------------------------------------------------------------
    # 3) Adam optimisation on each promising seed
    lam_schedule = (10.0, 30.0, 100.0, 300.0)
    remaining = deadline - time.time()
    if remaining > 60 and top_candidates:
        per_candidate_time = remaining / len(top_candidates)
        for idx, seed_seq in enumerate(top_candidates):
            if time.time() >= deadline:
                break
            cand_deadline = min(deadline, time.time() + per_candidate_time)
            logits0 = np.log(seed_seq + 1e-12)
            seq_opt, val_opt = _adam_optimize(logits0, n,
                                              conv_fft_len, corr_fft_len,
                                              cand_deadline,
                                              lam_schedule=lam_schedule,
                                              step_size=0.2)
            if seq_opt is not None and val_opt < best_val:
                best_seq, best_val = seq_opt.copy(), val_opt
                print(f"[Adam seed {idx}] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional L‑BFGS refinement (uses SciPy if available)
    if time.time() < deadline - 30 and best_seq is not None:
        lbfgs_deadline = min(deadline, time.time() + 180.0)
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         lbfgs_deadline,
                                         lam_schedule=(5.0, 15.0, 40.0, 120.0),
                                         max_iter_stage=300)
        if val_lb < best_val:
            best_seq, best_val = seq_lb.copy(), val_lb
            print(f"[L‑BFGS final] improved to {val_lb:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid
    if time.time() < deadline - 20:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline, time.time() + 150.0)
        logits0_big = np.log(a_big + 1e-12)
        seq_big, val_big = _adam_optimize(logits0_big, n_big,
                                          conv_fft_len_big, corr_fft_len_big,
                                          big_deadline,
                                          lam_schedule=(30.0, 100.0, 300.0),
                                          step_size=0.15)
        if val_big < best_val:
            best_seq, best_val = seq_big.copy(), val_big
            print(f"[Big grid Adam] improved to {val_big:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
