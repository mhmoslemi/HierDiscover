```python
"""
Proposes a non‑negative step function minimizing
    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search combines:
* a cheap random‑sampling stage exploring many parametric families,
* a long Adam optimisation that works directly on the exact objective
  (gradient of the max‑convolution plus the denominator term),
* a few short noisy‑restart Adam runs,
* a final cheap local hill‑climb.

All phases respect the given time budget; the best sequence seen so far
is always returned as a Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize_exact(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the exact objective.
    The variables are kept on the probability simplex via a soft‑max
    (logits) parametrisation.
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                          # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ---- simplex projection (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective -----------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[AdamExact it={it:06d}] new best {val:.6f}")

        # ---- gradient of the exact objective --------
        # active set – indices attaining the max (with a tiny tolerance)
        active = conv >= max_c - 1e-14
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ∇ max_c (exact)
        grad_max_c = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # full gradient of F = 2·n·max_c / (sum a)^2  (sum a = 1)
        grad_F = 2.0 * n * grad_max_c - 4.0 * n * max_c * a

        # ---- chain rule through the soft‑max ----------
        avg = np.dot(a, grad_F)
        grad_log = a * (grad_F - avg)

        # optional stochastic perturbation
        if noise_sigma > 0 and (it % 500 == 0):
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ---- Adam update on logits -------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # occasional shake when we are close to the deadline
        if time.time() > deadline - 30 and (it % 400 == 0):
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 70_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

"""
Proposes a non‑negative step function minimizing

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search consists of four stages that respect a 1000 s budget:

1. **Cheap random sampling** – explores many parametric families
   (Dirichlet, exponentials, power‑laws, cosine‑shapes, spike‑sets,
   two‑Gaussian mixtures).  The best candidate is kept.

2. **Long Adam optimisation on the *exact* objective** – the gradient is
   the exact sub‑gradient of the max‑convolution plus the denominator term.
   Variables are kept on the probability simplex via a soft‑max (logits)
   parametrisation.  Noise is injected periodically to escape plateaux.

3. **Few noisy‑restart Adam runs** – start from perturbed versions of the
   current best and run a shorter Adam optimisation.

4. **Very cheap hill‑climbing** – move a random fraction of mass from one
   coordinate to another and keep the move if the objective improves.

The best sequence seen at any time is always returned, guaranteeing a
valid answer even if the time budget is exhausted early.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
def _adam_optimize_exact(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the *exact* objective.
    The iterate lives on the probability simplex (soft‑max on logits).
    Returns the best vector encountered and its exact objective value.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                 # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ----- project to simplex (soft‑max) ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective ------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[AdamExact it={it:06d}] new best {val:.6f}")

        # ----- gradient of the exact objective ---------
        active = conv >= max_c - 1e-14
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ∇ max_c (exact)
        grad_max_c = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # full gradient of F = 2·n·max_c / (sum a)^2 (sum a = 1)
        grad_F = 2.0 * n * grad_max_c - 4.0 * n * max_c * a

        # ----- chain rule through soft‑max -------------
        avg = np.dot(a, grad_F)
        grad_log = a * (grad_F - avg)

        # optional stochastic perturbation
        if noise_sigma > 0 and (it % 500 == 0):
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ----- Adam update on logits ------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # occasional shake when we are close to the deadline
        if time.time() > deadline - 30 and (it % 400 == 0):
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val

            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val
