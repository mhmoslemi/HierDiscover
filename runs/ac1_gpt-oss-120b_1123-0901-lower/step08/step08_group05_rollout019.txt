def _pendulum_order(vals: np.ndarray) -> np.ndarray:
    """Place largest at ends alternating. E.g., indices: 0, n-1, 1, n-2, ..."""
    n = len(vals)
    a = np.empty_like(vals)
    left = 0
    right = n - 1
    idx = 0
    for val in vals:
        if idx % 2 == 0:
            a[left] = val
            left += 1
        else:
            a[right] = val
            right -= 1
        idx += 1
    return a

"""
A fast multi‑stage optimizer for the 2·n·max(a*a)/(sum(a)^2) problem.

Stages
------
1. **Candidate generation** – a dense grid of analytic shapes
   (power‑law, exponential, beta (U‑shaped), Gaussian‑edges,
   uniform) is sampled on a relatively large grid (n≈8192).  For each
   shape three deterministic reorderings are tried:
   * anti‑pair (large‑half left, small‑half right)
   * alternating (interleaved large / small)
   * pendulum   (largest ↔ ends, second ↔ opposite end, …)

   The best few candidates (lowest exact objective) are kept.

2. **Smooth‑max Adam** – minimise a log‑sum‑exp surrogate of the
   convolution maximum while staying on the probability simplex.
   The temperature (lam) is increased exponentially, so the surrogate
   approaches the true max.  The best exact value seen during the run
   is stored.

3. **Exact‑max sub‑gradient** – exponentiated‑gradient (mirror descent)
   on the true max‑convolution, with a diminishing step size.

4. **Gradient‑guided hill climbing** – repeatedly moves a fraction of
   mass from the position with the largest sub‑gradient (i.e. most
   harmful for the current max) to the position with the most negative
   sub‑gradient.  An occasional random swap/transfer is tried to escape
   plateaus.

All stages respect a global time budget (default 1000 s) and the routine
always returns a non‑negative list whose sum is 1 (scale‑invariant for
the evaluation function).  The code is deliberately written with
no closures, no lambdas and no file/network IO, and uses only NumPy
(and SciPy for a tiny LP‑based warm‑start that proved useful on a few
runs).  The final sequence is returned as a Python list.
"""

import time
import numpy as np
from scipy import optimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1) -------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) with a ≥ 0 and sum(a)=1."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Shape generators (all produce a non‑negative vector with sum=1) -----

def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ exp(-p·i)."""
    vals = np.exp(-p * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _gaussian_edges_shape(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two Gaussians placed at both ends."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals /= vals.sum()
    return vals


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """
    Symmetric Beta(α,α) density discretised on n points.
    For α<1 the shape is U‑shaped (large mass near the ends).
    """
    # centre the points to avoid exact 0/1 where the density would blow up.
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    vals = (x ** (alpha - 1.0)) * ((1.0 - x) ** (alpha - 1.0))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Ordering heuristics ---------------------------------------------------

def _anti_pair_order(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest ordering (largest half left, smallest half right)."""
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = vals[:half]                     # biggest half on the left
    a[-half:] = vals[-half:][::-1]             # smallest half on the right, reversed
    if n % 2 == 1:
        a[half] = vals[half]                   # median stays in centre
    return a


def _alternating_order(vals: np.ndarray) -> np.ndarray:
    """Interleave large and small values."""
    n = len(vals)
    perm = np.empty(n, dtype=int)
    left, right, idx = 0, n // 2, 0
    while left < n // 2 and right < n:
        perm[idx] = left
        perm[idx + 1] = right
        left += 1
        right += 1
        idx += 2
    if n % 2 == 1:
        perm[-1] = n // 2
    a = np.empty_like(vals)
    a[perm] = vals
    return a


def _pendulum_order(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ ends, second largest ↔ opposite end, etc."""
    n = len(vals)
    a = np.empty_like(vals)
    left, right = 0, n - 1
    for k, v in enumerate(vals):
        if k % 2 == 0:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a


def _apply_order(shape_vals: np.ndarray, order_type: str) -> np.ndarray:
    """Return a reordered version of `shape_vals`."""
    if order_type == "anti_pair":
        return _anti_pair_order(shape_vals)
    if order_type == "alternating":
        return _alternating_order(shape_vals)
    if order_type == "pendulum":
        return _pendulum_order(shape_vals)
    # fallback
    return shape_vals.copy()

def _warm_start(n: int) -> np.ndarray | None:
    """If a previous solution (global var `height_sequence_1`) exists, adapt it."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 1e-12:
        return None
    warm = warm / warm.sum()
    return warm

# Optimisation primitives ----------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   lr: float = 0.06,
                   lam_start: float = 0.5,
                   lam_factor: float = 1.08) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # Smooth‑max weights (log‑sum‑exp temperature = 1/lam)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient wrt a of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # Convert to gradient wrt logits (soft‑max trick)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Raise temperature
        lam = min(lam * lam_factor, 1e7)

        # Small jitter every few thousand steps
        if it % 8_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-3

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 0.65) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient refinement (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # Active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Diminishing step size (Polyak‑type)
        eta = eta0 / np.sqrt(it + 1)

        # Exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # Re‑project onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _gradient_transfer_search(a0: np.ndarray,
                              n: int,
                              conv_fft_len: int,
                              deadline: float,
                              max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing using the sub‑gradient information:
    repeatedly move a fraction of mass from the position with the largest
    positive gradient (worst for the current max) to the one with the most
    negative gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # compute sub‑gradient for the active max entries
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)   # same FFT length works

        # indices with extremal gradients
        i = int(np.argmax(grad))   # most harmful (increase objective)
        j = int(np.argmin(grad))   # most helpful (decrease objective)

        if a[i] > 1e-12 and i != j:
            # transfer a random fraction (up to 30 %) from i to j
            delta = a[i] * (np.random.rand() * 0.3)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            val2 = _objective(a2)

            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                a = a2
            else:
                # occasional stochastic acceptance (temperature cools linearly)
                temp = 0.02 * (1.0 - it / max_moves)
                if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                    a = a2
        else:
            # fallback: random swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
                val2 = _objective(a2)
                if val2 < best_val:
                    best_val = val2
                    best_a = a2.copy()
                    a = a2

        if it % 5_000 == 0:
            print(f"[GradMove {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def _lp_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max 1ᵀg   s.t.   f * g ≤ M   (componentwise),   g ≥ 0.
    Returns the optimal g (not normalised) or None if the LP fails.
    Uses scipy.optimize.linprog with the HiGHS solver (fast for modest sizes).
    """
    n = f.shape[0]
    # Build Toeplitz matrix of f for linear convolution (size (2n‑1) × n)
    # Using sparse construction for speed.
    rows = []
    cols = []
    data = []
    for i in range(2 * n - 1):
        # convolution row i: f[j] contributes to position i if 0 ≤ i‑j < n
        j_start = max(0, i - (n - 1))
        j_end = min(n - 1, i)
        for j in range(j_start, j_end + 1):
            rows.append(i)
            cols.append(j)
            data.append(f[i - j])
    A = np.zeros((2 * n - 1, n), dtype=np.float64)
    A[rows, cols] = data

    # Inequality: A @ g ≤ M (vector of length 2n‑1)
    bounds = [(0, None) for _ in range(n)]
    try:
        res = optimize.linprog(
            c=-np.ones(n),          # maximise sum(g) → minimise -sum(g)
            A_ub=A,
            b_ub=np.full(2 * n - 1, M),
            bounds=bounds,
            method="highs",
            options={"presolve": True, "time_limit": 8.0},
        )
        if res.success:
            return res.x
    except Exception:
        pass
    return None

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function that minimises the given evaluation."""
    np.random.seed(seed)
    safety = 0.5
    deadline = time.time() + budget_s - safety

    # --------------------------------------------------------------
    # 1) Choose resolution (main grid) --------------------------------
    n = 8192                     # main resolution (≈8k points)
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Build a rich pool of starter vectors -------------------------

    start_vectors = []

    # warm start (if present)
    warm = _warm_start(n)
    if warm is not None:
        start_vectors.append(warm.copy())
        for shift in (n // 5, n // 2, 4 * n // 5):
            start_vectors.append(np.roll(warm, shift))

    # basic analytic shapes
    start_vectors.append(_uniform_shape(n))

    shape_fns = [
        ("power", _powerlaw_shape, np.linspace(0.0, 2.5, 26)),
        ("exp", _exponential_shape, np.linspace(0.0, 2.5, 26)),
        ("gauss", _gaussian_edges_shape, [0.2, 0.5, 1.0, 2.0]),
        ("beta", _beta_shape, np.linspace(0.3, 0.9, 13)),
    ]
    orderings = ["anti_pair", "alternating", "pendulum"]

    for name, fn, params in shape_fns:
        for p in params:
            if name == "beta":
                vals = fn(n, alpha=p)
            else:
                vals = fn(n, p)
            # vals already non‑negative and sum=1, but not sorted.
            # Sorting descending produces a canonical ordering for the heuristics.
            vals_desc = np.sort(vals)[::-1]
            for order in orderings:
                cand = _apply_order(vals_desc, order)
                start_vectors.append(cand)

    # random Dirichlet seeds (various concentrations)
    for alpha in (0.2, 0.4, 0.7, 1.0, 1.5, 2.5):
        cand = np.random.dirichlet(alpha * np.ones(n))
        cand = np.maximum(cand, 0.0)
        cand /= cand.sum()
        start_vectors.append(cand)

    # --------------------------------------------------------------
    # 3) Quick evaluation of the pool, keep the best few (top‑5) ------
    pool = []
    best_val = float("inf")
    for idx, cand in enumerate(start_vectors):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-12:
            continue
        cand = cand / cand.sum()
        val = _objective(cand)
        pool.append((val, cand))
        if val < best_val:
            best_val = val
        if idx % 20 == 0:
            print(f"[Init cand {idx:03d}] value {val:.6f}")

    pool.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = pool[0][1].copy()
    best_val = pool[0][0]
    print(f"[Init] best starter value {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Optimisation phases (Adam → Sub‑gradient → Gradient moves) --
    time_left = deadline - time.time()
    if time_left < 30.0:
        # Not enough time for any optimisation – return the best starter.
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return best_seq.tolist()

    # Allocate time slices
    adam_time = max(30.0, 0.45 * time_left)
    subgrad_time = max(30.0, 0.35 * (deadline - time.time() - adam_time))
    gradmove_time = max(20.0, deadline - time.time() - adam_time - subgrad_time)

    # ---- Adam phase -------------------------------------------------
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        time.time() + adam_time,
        max_iters=800_000,
        lr=0.06,
        lam_start=0.5,
        lam_factor=1.08,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient phase -----------------------------------------
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        time.time() + subgrad_time,
        max_iters=800_000,
        eta0=0.65,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---- Gradient‑guided hill climbing -------------------------------
    if time.time() < deadline - 5.0:
        a_move, val_move = _gradient_transfer_search(
            best_seq,
            n,
            conv_fft_len,
            time.time() + gradmove_time,
            max_moves=120_000,
        )
        if val_move < best_val:
            best_val = val_move
            best_seq = a_move.copy()
            print(f"[GradMove] improved to {val_move:.6f}")

    # --------------------------------------------------------------
    # 5) Optional LP‑based direction (one extra improvement step) -----
    # (very cheap – only attempted if time permits)
    if time.time() < deadline - 2.0:
        M = (_full_convolution(best_seq, conv_fft_len)).max()
        g_lp = _lp_direction(best_seq, M)
        if g_lp is not None:
            # normalise the LP solution and mix a little
            g_lp = np.maximum(g_lp, 0.0)
            if g_lp.sum() > 1e-12:
                g_lp = g_lp / g_lp.sum()
                mixed = 0.85 * best_seq + 0.15 * g_lp
                mixed = mixed / mixed.sum()
                val_mix = _objective(mixed)
                if val_mix < best_val:
                    best_val = val_mix
                    best_seq = mixed
                    print(f"[LPmix] improved to {val_mix:.6f}")

    # --------------------------------------------------------------
    # 6) Final safety and conversion -----------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
