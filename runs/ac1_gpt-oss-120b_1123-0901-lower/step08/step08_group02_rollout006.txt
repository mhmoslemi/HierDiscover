"""
Propose a step‑function sequence that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The optimiser works in three stages within the allotted time budget:

1. **Broad random search** – many cheap Dirichlet / uniform draws are
   evaluated to obtain a decent starting point.

2. **Smooth‑max Adam** – the non‑differentiable max is replaced by a
   log‑sum‑exp surrogate with a temperature that is increased slowly.
   The parameters are the logits of a soft‑max (so the iterate stays on
   the probability simplex).  Adam with decaying learning‑rate is used.

3. **Exact‑max sub‑gradient** – after the surrogate has become sharp,
   a pure sub‑gradient (exponentiated‑gradient) phase refines the
   solution to high accuracy.  The step size decays as η₀/√t.

All three phases respect the global time limit (default 1000 s) and the
best feasible sequence seen so far is always returned.  The implementation
uses FFT‑based convolution for speed and works for any problem size; a
moderate size `n = 1024` is used so that many optimisation steps fit
comfortably into the time budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (always sums to 1)."""
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000,
                   ) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (sum≈1) and its exact objective.
    """
    eps = 1e-12
    logits = np.log(a0 + eps)          # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                        # a little larger than the reference
    lam = 0.005                       # start temperature (soft)
    lam_inc = 1.015                   # slow annealing
    lam_max = 1e8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- simplex via soft‑max ----
        a = _softmax(logits)

        # ---- exact objective (for tracking) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 → denominator =1
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 25_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        # w ∝ exp(lam * (conv - max_c))  →  w sums to 1
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- mirror‑descent (soft‑max) chain rule ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_inc, lam_max)

        # ---- gentle learning‑rate decay (very slow) ----
        if it % 150_000 == 0 and it > 0:
            lr *= 0.94

        # ---- occasional random shake near the deadline ----
        if (it % 4_000 == 0) and (time.time() > deadline - 30):
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 1_500_000,
                          eta0: float = 0.9,
                          ) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ---- gradient of the exact max ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- decreasing step size η_t = η₀ / sqrt(t+1) ----
        eta = eta0 / np.sqrt(it + 1.0)

        # ---- exponentiated‑gradient update (mirror descent on simplex) ----
        a = a * np.exp(-eta * grad_a)

        # ---- renormalise ----
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 120_000,
                         ) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from i→j.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 20 % of the source mass
        delta = a[i] * (np.random.rand() * 0.20)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
            if it % 20_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (smaller n gives many more optimisation steps)
    n = 1024
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start from the previous global solution if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # initial best (uniform if no warm start)
    if warm is not None:
        best_seq = warm.copy()
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    conv = _full_convolution(best_seq, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    print(f"[Init] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – cheap random sampling (≈10 % of the budget)
    phase1_deadline = time.time() + 0.10 * (deadline - time.time())
    while time.time() < phase1_deadline:
        # Dirichlet with random concentration
        alpha = np.random.rand() * 4.0 + 0.2      # α∈[0.2,4.2]
        cand = np.random.dirichlet(np.full(n, alpha))
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam on smooth‑max (≈45 % of budget)
    phase2_deadline = time.time() + 0.45 * (deadline - time.time())
    while time.time() < phase2_deadline:
        # initialise from current best but add a small random perturbation
        pert = best_seq + (np.random.rand(n) - 0.5) * 0.02
        pert = np.maximum(pert, 0.0)
        if pert.sum() < 1e-12:
            pert = np.random.dirichlet(np.ones(n))
        pert = pert / pert.sum()

        a_adam, val_adam = _adam_optimize(
            pert,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline=phase2_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – long exact‑max sub‑gradient phase (≈35 % of budget)
    phase3_deadline = deadline - 5.0   # keep a couple of seconds for final tidy‑up
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline=phase3_deadline,
        max_iters=1_200_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, a_sub.copy()
        print(f"[Subgrad] improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap random‑move hill‑climber (use any remaining time)
    if time.time() < deadline:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – clip to the evaluator’s accepted range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
