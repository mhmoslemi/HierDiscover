"""
Propose a step function with a low evaluation value.

The search consists of three stages:

1.  Warm‑starts – a small, diverse pool of candidate vectors (uniform,
    Dirichlet draws, anti‑pair, edge‑focused, triangular, alternating)
    is built; the one with the smallest exact objective is kept.

2.  Continuous refinement – a short Adam phase on a smooth‑max (log‑sum‑exp)
    surrogate followed by an exact‑max exponentiated‑gradient phase.
    Both phases use FFT‑based convolutions and respect the remaining budget.

3.  Long simulated‑annealing – the best vector is further improved by an
    incremental O(n) convolution update.  Moves transfer a random fraction
    of mass from one index to another; the temperature and the allowed
    transfer fraction decay with the remaining time, allowing both exploration
    and fine‑grained tuning.

Multiple short “re‑starts’’ are performed while time permits; the best
overall sequence is returned (scaled to the conventional ‑‑ sum = sqrt(2·n)
normalisation so that the final evaluation equals max(conv)).
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (sum of a is 1) ---------------------------------------

def _objective(a: np.ndarray) -> float:
    """
    Return 2·n·max(conv) for a vector whose entries sum to 1.
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """If a previous solution exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place the biggest half on the left, smallest half on the right."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    v = np.random.dirichlet(alpha * np.ones(n))
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return _anti_pair_arrangement(v)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular (peak in the centre)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _alternating_candidate(n: int) -> np.ndarray:
    """Large values on even indices, small on odd indices (alternating)."""
    vals = np.random.dirichlet(np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    order = np.argsort(vals)[::-1]
    out = np.empty_like(vals)
    # put the biggest half on even positions, the rest on odds
    evens = np.arange(0, n, 2)
    odds = np.arange(1, n, 2)
    half = (len(order) + 1) // 2
    out[evens] = vals[order[:len(evens)]]
    out[odds] = vals[order[len(evens):]]
    return out


# ----------------------------------------------------------------------
# Optimisation primitives -----------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.07,
                   lam_start: float = 1.0,
                   lam_factor: float = 1.07) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    The parameter vector is kept on the simplex via a soft‑max.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)           # initialise in the log‑space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → ensure non‑negativity and sum = 1
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (used only for bookkeeping)
        cur_val = _objective(a)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if it % 10_000 == 0:
                print(f"[Adam it={it}] new best {best_val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        w = np.exp(lam * (conv - M))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)       # chain rule through soft‑max

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase λ to approach the true max
        lam = min(lam * lam_factor, 1e7)

        # small jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient updates).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        cur_val = 2.0 * n * M
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it}] new best {best_val:.6f}")

        active = np.isclose(conv, M, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Increment a[idx] by `delta` (delta may be negative) and update
    the full linear convolution a*a in O(n) using NumPy slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx:idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (appears at index 2*idx)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(a0: np.ndarray,
                    time_limit: float,
                    start_temp: float = 0.025,
                    max_frac_start: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing with an O(n) incremental convolution update.
    The allowed transfer fraction shrinks together with the temperature,
    enabling both global moves and fine‑grained polishing.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        # pick source and destination indices
        i = np.random.randint(n)
        if a[i] < 1e-16:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # temperature‑dependent maximum transfer fraction
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))
        max_frac = max_frac_start * (remaining / (time_limit - start))

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move (incremental update)
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 200_000 == 0:
            print(f"[SA it={it}] best conv_max {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    evaluate_sequence(a) within the allocated time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # problem size – 4096 is a good compromise between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a pool of diverse starters
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # deterministic starters
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))
    candidates.append(_gaussian_edges_candidate(n, gamma=0.5))

    # stochastic starters – a few draws of each type
    for _ in range(2):
        candidates.append(_random_dirichlet_candidate(n, alpha=0.8))
        candidates.append(_random_dirichlet_candidate(n, alpha=1.5))
        candidates.append(_anti_pair_candidate(n, alpha=0.8))
        candidates.append(_anti_pair_candidate(n, alpha=1.5))
        candidates.append(_alternating_candidate(n))

    # ------------------------------------------------------------------
    # 2) Pick the best starter (exact objective)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] objective {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) Main optimisation loop.  We repeatedly allocate a short
    #    budget to Adam → sub‑gradient → SA while we still have time.
    while time.time() < deadline - 30.0:                     # keep 30 s for final polishing
        # how much time shall we spend on this round?
        round_deadline = min(deadline - 5.0, time.time() + 30.0)

        # ------- Adam (smooth‑max) ------------------------------------
        adam_deadline = min(round_deadline - 5.0, time.time() + 10.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Round Adam] improved to {best_val:.6f}")

        # ------- Sub‑gradient (exact max) ------------------------------
        sub_deadline = min(round_deadline - 2.0, time.time() + 8.0)
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=200_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Round Subgrad] improved to {best_val:.6f}")

        # ------- Simulated‑annealing (long) ----------------------------
        sa_time = deadline - time.time() - 5.0
        if sa_time > 0:
            a_sa, conv_max = _incremental_sa(
                best_seq,
                time_limit=time.time() + sa_time,
                start_temp=0.03,
                max_frac_start=0.35,
            )
            val_sa = 2.0 * n * conv_max
            if val_sa < best_val:
                best_val = val_sa
                best_seq = a_sa.copy()
                print(f"[Round SA] improved to {best_val:.6f}")

        # if we made a solid improvement break early – otherwise continue
        # hoping another random start will escape the current basin.
        if time.time() > deadline - 30.0:
            break

    # ------------------------------------------------------------------
    # 4) Final fine‑tuning: a last short SA with a very low temperature
    #    and tiny transfer fractions to polish the last digits.
    remaining = deadline - time.time() - 2.0
    if remaining > 5.0:
        a_final, conv_max = _incremental_sa(
            best_seq,
            time_limit=time.time() + remaining,
            start_temp=0.015,
            max_frac_start=0.12,
        )
        val_final = 2.0 * n * conv_max
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final SA] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Rescale to the conventional normalisation (sum = sqrt(2·n)).
    #    This does not change the evaluation value because the objective
    #    is homogeneous of degree 0.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    target_sum = math.sqrt(2.0 * n)
    best_seq = best_seq / best_seq.sum() * target_sum

    print(f"Final objective {best_val:.6f} (conv_max ≈ {best_val/(2.0*n):.6f}) length {len(best_seq)}")
    return best_seq.tolist()
