def _refine_by_active_pair(a, n, fft_len, deadline):
    a = a.copy()
    best_a = a.copy()
    best_val = _objective(a)
    while time.time() < deadline:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        idx_max = conv.argmax()
        # compute contributions: for i in range(n):
        # j = idx_max - i
        # if 0 <= j < n: contribution = a[i] * a[j]
        # else 0.
        # We'll compute vector contributions
        i_range = np.arange(n)
        j_range = idx_max - i_range
        mask = (j_range >=0) & (j_range < n)
        # contributions = a[i_range[mask]] * a[j_range[mask]]
        # Choose one random pair from those with contribution > threshold.
        contributions = a[i_range[mask]] * a[j_range[mask]]
        if contributions.size == 0:
            break
        # spectral weighting: choose pair weighted by contribution magnitude
        probs = contributions / contributions.sum()
        pair_idx = np.random.choice(contributions.size, p=probs)
        i = i_range[mask][pair_idx]
        j = j_range[mask][pair_idx]
        if a[i] < 1e-12 or a[j] < 1e-12:
            continue
        # transfer portion
        delta = min(a[i], a[j]) * (np.random.rand() * 0.4)  # up to 40% of smaller
        # We'll transfer delta from both i and j (reducing each) and add equally to two random other positions
        k = np.random.randint(n)
        l = np.random.randint(n)
        # ensure k,l not i,j maybe
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] -= delta
        a_new[k] += delta/2
        a_new[l] += delta/2
        # Clip to >=0
        a_new = np.maximum(a_new, 0)
        # Renormalize sum to 1.
        s = a_new.sum()
        if s > 0:
            a_new /= s
        val_new = _objective(a_new)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            # optionally continue
        else:
            # maybe accept some probability
            if np.random.rand() < 0.1:
                a = a_new
        # continue
    return best_a, best_val

"""Step‑function optimiser.

The optimiser builds a diverse pool of warm‑starts (uniform, Dirichlet,
anti‑pair, power‑law, U‑shape, Gaussian‑edge, sinusoid, etc.).  Each
candidate is first coarse‑optimised with a smooth‑max Adam optimiser,
then refined with an exact‑max exponentiated‑gradient (sub‑gradient) loop.
The best candidate is further polished by a focussed local search that
moves mass away from the active convolution peak.  If time permits the
solution is up‑sampled to 8192 points and refined once more.

All heavy work respects the 1000 s wall‑clock limit; the routine returns
a plain Python list of non‑negative floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (fast, no safety checks) -----------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)^2) – assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helpers ----------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm‑start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Deterministic / random candidate generators ---------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑opposite‑smallest ordering (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape: 1,2,…,⌈n/2⌉,…,2,1."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    """Valley in the centre, larger at the two ends."""
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _sinusoid_candidate(n: int) -> np.ndarray:
    """Half‑wave sinusoid, zero at the boundaries."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * i / (n - 1))
    shape = np.maximum(shape, 0.0)
    return shape / shape.sum()


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.04,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam it={it:06d}] new best {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaux
        if it % 7_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient) phase.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where conv is maximal)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size – slower decay than 1/√t gives steadier progress
        eta = eta0 / (np.sqrt(it + 1) + 1e-8)

        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _active_pair_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Local hill‑climbing focused on the active convolution peak.
    In each move a pair (i,j) that contributes to the current maximal
    entry is weakened and the freed mass is redistributed elsewhere.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        idx = int(conv.argmax())

        # contributions of each pair (i,j) with i+j == idx
        i_arr = np.arange(n, dtype=np.int32)
        j_arr = idx - i_arr
        mask = (j_arr >= 0) & (j_arr < n)

        if not mask.any():
            break

        i_sel = i_arr[mask]
        j_sel = j_arr[mask]
        contrib = a[i_sel] * a[j_sel]

        # pick a pair proportional to its contribution
        prob = contrib / contrib.sum()
        pair_idx = np.random.choice(contrib.size, p=prob)
        i = i_sel[pair_idx]
        j = j_sel[pair_idx]

        # move a fraction of the smaller side to two random other slots
        if a[i] < 1e-14 or a[j] < 1e-14:
            it += 1
            continue

        delta = min(a[i], a[j]) * (np.random.rand() * 0.35)   # up to 35 % of the smaller entry
        if delta <= 0:
            it += 1
            continue

        a2 = a.copy()
        a2[i] -= delta
        a2[j] -= delta

        # pick two destination indices (could be equal)
        dst1 = np.random.randint(n)
        dst2 = np.random.randint(n)
        a2[dst1] += delta * 0.5
        a2[dst2] += delta * 0.5

        # keep feasibility
        a2 = np.maximum(a2, 0.0)
        a2 /= a2.sum()

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # occasional uphill moves (simulated‑annealing style)
            temp = 0.02 * max(0.0, 1.0 - it / max_moves)
            if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Candidate builder ------------------------------------------------------
def _build_candidates(n: int) -> list[np.ndarray]:
    """Create a diverse pool of normalised candidates."""
    cand = []

    # warm start (if present)
    warm = _warm_start(n)
    if warm is not None:
        cand.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # simple deterministic shapes
    cand.append(_uniform_candidate(n))
    cand.append(_triangle_candidate(n))
    cand.append(_sinusoid_candidate(n))

    # Dirichlet with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0):
        cand.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair Dirichlet samples
    for alpha in (0.2, 0.5, 1.0):
        cand.append(_anti_pair_candidate(n, alpha))

    # power‑law & U‑shape families
    for p in (0.2, 0.5, 1.0, 2.0):
        cand.append(_powerlaw_candidate(n, p))
        cand.append(_ushape_candidate(n, p))

    # Gaussian edges
    for gamma in (0.2, 0.5, 1.0):
        cand.append(_gaussian_edges_candidate(n, gamma))

    # random permutations of a strong anti‑pair starter
    base_ap = _anti_pair_candidate(n, alpha=0.7)
    for _ in range(4):
        perm = np.random.permutation(n)
        cand.append(base_ap[perm])

    return cand

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The function obeys the time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1️⃣  Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2️⃣  Build and prune the candidate pool
    candidates = _build_candidates(n)

    evaluated = []
    print("[Init] evaluating warm‑starts …")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:              # fallback – should not happen often
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 10 == 0:
            print(f"  candidate {idx:02d} → {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq, best_val = evaluated[0][1].copy(), evaluated[0][0]
    print(f"[Init] best initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Coarse optimisation of the top candidates (Adam)
    remaining = deadline - time.time()
    per_candidate = remaining * 0.25 / top_k          # 25 % of the remaining budget

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        adam_dead = time.time() + per_candidate
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=500_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam #{cid}] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Long exact‑max sub‑gradient refinement on the current best
    sub_dead = deadline - 3.0           # leave a few seconds for the final steps
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_dead,
        max_iters=1_000_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val, best_seq = val_sub, a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Focused local search around the active convolution peak
    if time.time() < deadline - 2.0:
        loc_dead = deadline - 1.0
        a_loc, val_loc = _active_pair_local_search(
            best_seq,
            n,
            conv_fft_len,
            loc_dead,
            max_moves=500_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Optional upscale to a finer grid (8192) -------------------------------------------------
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # a quick Adam pass on the larger grid
        big_adam_dead = min(deadline - 5.0, time.time() + 150.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=300_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # final sub‑gradient fine‑tune
        big_sub_dead = deadline
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_dead,
            max_iters=600_000,
            eta0=0.8,
        )

            print(f"[Upscaled] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
