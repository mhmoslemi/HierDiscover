"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm blends several ideas:

1. **Warm‑starts** – use any previously stored `height_sequence_1`,
   a few handcrafted seeds and a large pool of random Dirichlet samples
   (with “anti‑pair” ordering) to obtain a good initial shape.

2. **Adam optimisation on a smooth‑max surrogate** – the hard max is
   replaced by a log‑sum‑exp with an increasing temperature λ.
   The variables are logits of a soft‑max (guaranteeing non‑negativity
   and unit sum).  Adam’s adaptive steps quickly drive the surrogate
   towards a lower exact max, and we keep the best iterate (checked
   against the true objective).

3. **Exact sub‑gradient refinement** – a multiplicative exponentiated‑gradient
   update that directly follows a sub‑gradient of the real max‑convolution.

4. **Optional up‑sampling** – if time permits the best vector is linearly
   interpolated to a double‑length grid (n=8192) and the Adam phase is
   repeated, allowing the larger space to lower the bound further.

All heavy work respects the global 1000 s budget; intermediate results are
printed for monitoring.  The returned list is clipped to the limits accepted
by `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) = n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (the one we finally minimise)
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)^2); a must be non‑negative, sum>0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simple helpers
def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    mx = np.max(logits)
    e = np.exp(logits - mx)
    return e / e.sum()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """If a previous solution `height_sequence_1` exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two symmetric exponential bells placed at the borders."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


# ----------------------------------------------------------------------
# Adam optimisation of the smooth‑max surrogate
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   lam_schedule=(5.0, 15.0, 40.0, 120.0, 400.0, 1200.0),
                   max_iter_per_lam: int = 2000) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best vector found (in probability simplex) and its true objective.
    """
    a0 = np.maximum(a0, 0.0)
    a0 = a0 / a0.sum()
    logits = np.log(a0 + 1e-12)

    best_logits = logits.copy()
    best_val = _objective(_softmax(logits))

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0

    for lam in lam_schedule:
        # step size inversely proportional to sqrt(lam) – larger λ needs smaller steps
        lr = 0.05 / np.sqrt(lam)

        # allocate iterations respecting the remaining time budget
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        # very rough estimate: ~2 ms per iteration on a 4096‑vector
        est_iters = int(remaining / 0.0025)
        max_iters = min(max_iter_per_lam, max(1, est_iters))

        for _ in range(max_iters):
            if time.time() >= deadline:
                break

            a = _softmax(logits)                     # current probability vector
            conv = _full_convolution(a, conv_fft_len)   # a*a
            max_c = conv.max()

            # soft‑weights approximating the arg‑max
            z = np.exp(lam * (conv - max_c))
            w = z / z.sum()

            # gradient w.r.t. a (exact derivative of the smooth surrogate)
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # chain rule to logits
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

            # occasional exact evaluation
            if (t % 200) == 0:
                val = _objective(_softmax(logits))
                if val < best_val:
                    best_val = val
                    best_logits = logits.copy()
                    print(f"[Adam λ={lam:.0f} step={t}] new best {val:.6f}")

    best_seq = _softmax(best_logits)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Multiplicative (exponentiated‑gradient) exact‑max sub‑gradient optimisation
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with multiplicative updates."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # occasional re‑initialisation
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Warm‑starts & cheap handcrafted seeds
    best_seq = None
    best_val = float("inf")

    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    # hand‑crafted seeds (very cheap)
    seeds = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for c in seeds:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Seed] value {val:.6f}")

    # --------------------------------------------------------------
    # 2) Massive random anti‑pair Dirichlet pool (time‑boxed)
    remaining = deadline - time.time()
    pool_time = min(150.0, remaining * 0.2)   # up to 20 % of the remaining budget
    pool_deadline = time.time() + pool_time
    pool_iter = 0
    while time.time() < pool_deadline:
        # concentration α drawn from a wide log‑uniform range
        log_alpha = np.random.uniform(-3.0, 2.0)      # α ∈ [1e‑3, 1e2]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random pool] it={pool_iter} new best {val:.6f}")
        pool_iter += 1

    # --------------------------------------------------------------
    # 3) Adam optimisation on the smooth‑max surrogate
    # allocate roughly half of the remaining budget to this phase
    adam_deadline = min(deadline - 120.0, time.time() + 250.0)
    if time.time() < adam_deadline:
        seq_adam, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Exact‑max sub‑gradient polishing
    sub_deadline = min(deadline - 40.0, time.time() + 200.0)
    if time.time() < sub_deadline:
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 max_iters=600_000,
                                                 eta0=0.7)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[Sub‑gradient] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() < deadline - 15.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline - 5.0, time.time() + 150.0)
        seq_big, val_big = _adam_optimize(a_big, n_big,
                                          conv_fft_len_big, corr_fft_len_big,
                                          big_deadline)
        if val_big < best_val:
            best_val, best_seq = val_big, seq_big.copy()
            print(f"[Big‑grid Adam] improved to {val_big:.6f}")

    # --------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
