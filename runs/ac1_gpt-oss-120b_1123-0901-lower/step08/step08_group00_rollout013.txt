"""
Step‑function optimizer (LP + sub‑gradient)
------------------------------------------------
The evaluator is homogeneous: scaling the whole vector does not change the
score, therefore we keep the L1‑norm equal to 1 (simplex).  The algorithm
consists of two stages.

1. **Linear‑programming (cut‑plane) smoothing**
   For the current a we compute its self‑convolution c = a*a.  Let M = max(c).
   The active shifts are the indices where c is almost equal to M.  For each
   active shift k the inequality   (a ⊛ b)[k] ≤ M   is linear in the new
   vector b, so we can maximise 1ᵀb under all those linear constraints.
   The optimum puts most mass on positions where a is small – this flattens
   the autocorrelation.  We repeatedly replace a by a convex combination with
   the LP‑solution b (step size ≈ 0.2) while the time budget permits.

2. **Exact‑max sub‑gradient refinement**
   After the LP stage we switch to the exact‑max sub‑gradient method used in
   the reference code (exponentiated‑gradient on the probability simplex).
   This quickly drives the remaining tiny peaks down.

Both phases respect the 1000 s budget and always keep track of the best
vector seen.  The final vector is returned as a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two not smaller than x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)².
    a is assumed to be non‑negative.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _lp_smoothing(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iter: int = 2500,
    start_t: float = 0.25,
    min_t: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Cut‑plane / LP phase.
    Returns the best vector encountered and its objective.
    """
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    t = start_t                     # convex‑combination step size
    # number of active constraints – a modest value keeps the LP cheap
    K_target = 30

    it = 0
    while time.time() < deadline and it < max_iter:
        # current convolution and bound
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        cur_val = 2.0 * n * M / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[LP it={it:04d}] new best {cur_val:.6f}")

        # ---- pick active shifts (largest entries of conv) ----
        # we use the top K_target indices; if fewer distinct values, take all.
        if K_target >= conv.size:
            active = np.arange(conv.size)
        else:
            # indices of the K_target largest convolutions
            active = np.argpartition(conv, -K_target)[-K_target:]

        K = active.size
        # ---- build constraint matrix A (K × n) ----
        A = np.zeros((K, n), dtype=np.float64)
        for row, k in enumerate(active):
            # j runs over all positions of b that appear in (a * b)[k]
            j_start = max(0, k - (n - 1))
            j_end = min(n - 1, k)
            j = np.arange(j_start, j_end + 1, dtype=np.int64)
            i = k - j
            A[row, j] = a[i]

        # ---- linear programme: maximise sum(b)  <=> minimise -1ᵀb ----
        c = -np.ones(n, dtype=np.float64)          # minimise –sum(b)
        bounds = [(0.0, None)] * n                 # b ≥ 0
        b_ub = np.full(K, M, dtype=np.float64)    # (a * b)[k] ≤ M  for active k

        res = linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # If the LP fails (numerical trouble) we stop the LP phase.
            print(f"[LP it={it:04d}] linprog failed – exiting LP phase")
            break

        b = res.x
        sb = b.sum()
        if sb <= 0.0:
            break
        b /= sb                                   # keep sum(b)=1 (simplex)

        # ---- convex combination a ← (1‑t)·a + t·b ----
        a_new = (1.0 - t) * a + t * b

        # keep the simplex normalisation (numerical drift)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        # evaluate new max‑convolution; if it gets worse we shrink t
        conv_new = _full_convolution(a_new, conv_fft_len)
        M_new = conv_new.max()
        val_new = 2.0 * n * M_new / (a_new.sum() ** 2)

        # simple backtracking on t
        backtrack = 0
        while val_new > cur_val and backtrack < 4:
            t *= 0.5
            a_new = (1.0 - t) * a + t * b
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            conv_new = _full_convolution(a_new, conv_fft_len)
            M_new = conv_new.max()
            val_new = 2.0 * n * M_new / (a_new.sum() ** 2)
            backtrack += 1

        # adapt step size gradually (slow decay)
        t = max(min_t, t * 0.985)

        a = a_new
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient).
    Returns the best vector and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # renormalise to the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    A fast LP‑smoothing stage is followed by an exact‑max sub‑gradient
    refinement.  The routine respects the given time budget and always returns
    the best vector (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # problem size – we keep the original length (4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Initial vector: either the warm start or a random Dirichlet draw
    if warm is not None:
        a = warm.copy()
    else:
        a = np.random.dirichlet(np.ones(n))
    a = np.maximum(a, 0.0)
    a /= a.sum()

    best_seq = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase I – LP‑based smoothing (≈ 30 % of the budget)
    lp_deadline = time.time() + (deadline - time.time()) * 0.30
    a_lp, val_lp = _lp_smoothing(a, n, conv_fft_len, lp_deadline)
    if val_lp < best_val:
        best_val = val_lp
        best_seq = a_lp.copy()
        print(f"[Phase I] LP best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exact‑max sub‑gradient refinement (remaining budget)
    if time.time() < deadline - 5:
        print("[Phase II] Sub‑gradient refinement")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Phase II] improved best to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final safety – ensure a valid non‑empty return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
