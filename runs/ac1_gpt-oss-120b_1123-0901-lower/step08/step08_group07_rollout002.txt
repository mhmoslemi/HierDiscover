"""
Propose a low‑value step function.

Algorithm
---------
* **Stage 0 – Warm‑start** – use the previous best (if present) and a few
  simple deterministic candidates (uniform, Dirichlet) as seeds.

* **Stage 1 – Random family search (small size)** – sample many
  analytical families (power‑law, U‑shape, Gaussian‑edges,
  sine‑valley, beta‑type, edge‑block) with random parameters on a
  reduced grid (n≈2048).  Keep the top‑k (k≈12) vectors according to the
  exact objective 2·n·max(a∗a)/(∑a)².

* **Stage 2 – Upscale & continuous refinement** – linearly interpolate
  each top candidate to the full size (n=4096) and improve it with two
  gradient‑based phases:
      – Adam on a smooth‑max (log‑sum‑exp) surrogate,
      – exact‑max sub‑gradient (exponentiated‑gradient) refinement.
  Each candidate receives a share of the remaining time.

* **Stage 3 – Stochastic hill‑climbing** – a simulated‑annealing‑style
  local search (swaps and mass transfers) on the current best vector.

* **Stage 4 – (optional) fine up‑sampling** – if time permits, upscale
  to 8192 points and repeat a short Adam/sub‑gradient pass.

All operations respect the security checks of `evaluate_sequence`.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no security checks) ---------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helper -----------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Deterministic / random generators -------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _powerlaw_sequence(n: int, p: float, offset: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i + offset) ** (-p) + (n - 1 - i + offset) ** (-p)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _ushape_sequence(n: int, p: float, c: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + c) ** (-p)
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _gaussian_edges_sequence(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _sine_valley_sequence(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.abs(np.cos(np.pi * i / (n - 1))) ** p
    vals = np.maximum(vals, 0.0)
    return vals / vals.sum()


def _beta_shape_sequence(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discretised Beta(α,β) on {1,…,n}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    vals = (i ** (alpha - 1)) * ((n + 1 - i) ** (beta - 1))
    vals = np.maximum(vals, 0.0)
    total = vals.sum()
    if total < 1e-12:
        return _uniform_candidate(n)
    return vals / total


def _edge_block_sequence(n: int, edge_frac: float, mult: float) -> np.ndarray:
    """
    Two‑level block: a high‑weight edge block (size edge_frac·n on each side)
    and a uniform interior.  `mult` is the ratio (edge weight)/(interior weight).
    """
    edge_frac = np.clip(edge_frac, 0.0, 0.5)
    k = int(round(edge_frac * n / 2.0))
    interior_len = n - 2 * k
    if interior_len <= 0:
        interior_len = 1
        k = (n - interior_len) // 2
    # Let interior weight = w_i, edge weight = mult * w_i
    denom = 2 * k * mult + interior_len
    w_i = 1.0 / denom
    w_e = mult * w_i
    a = np.empty(n, dtype=np.float64)
    a[:k] = w_e
    a[k : n - k] = w_i
    a[n - k :] = w_e
    return a

# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if (it % 5000) == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (softmax over convolution entries)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (see derivation in original code)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if (it % 5000) == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if (it % 10000) == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).
    Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if (it % 5000) == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def _sample_powerlaw(n: int) -> np.ndarray:
    p = 10 ** np.random.uniform(-0.8, 0.6)   # ≈ [0.16, 4]

def _sample_ushape(n: int) -> np.ndarray:

    c = 10 ** np.random.uniform(-2, 1)      # ≈ [0.01, 10]
    return _ushape_sequence(n, p, c)


def _sample_gaussian_edges(n: int) -> np.ndarray:
    gamma = 10 ** np.random.uniform(-3, 0)   # ≈ [0.001, 1]
    return _gaussian_edges_sequence(n, gamma)


def _sample_sine_valley(n: int) -> np.ndarray:
    p = 10 ** np.random.uniform(-0.8, 0.6)   # ≈ [0.16, 4]
    return _sine_valley_sequence(n, p)


def _sample_beta_shape(n: int) -> np.ndarray:
    # To prefer edge‑heavy shapes we bias α,β towards <1.
    alpha = 10 ** np.random.uniform(-1, 0)   # ≈ [0.1, 1]
    beta  = 10 ** np.random.uniform(-1, 0)   # ≈ [0.1, 1]
    return _beta_shape_sequence(n, alpha, beta)


def _sample_edge_block(n: int) -> np.ndarray:
    edge_frac = np.random.uniform(0.01, 0.25)   # size of each edge block
    mult = np.random.uniform(1.0, 12.0)        # how much heavier edges are
    return _edge_block_sequence(n, edge_frac, mult)


def _random_dirichlet(n: int) -> np.ndarray:

    return _random_dirichlet_candidate(n, alpha)


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Parameters
    n_big = 4096
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Stage 0 – Warm‑starts & deterministic seeds
    candidates_small = []

    # Warm start (previous best) – down‑sample to n_small
    warm = _warm_start(n_small)
    if warm is not None:
        candidates_small.append(warm.copy())

    # Uniform
    candidates_small.append(_uniform_candidate(n_small))

    # A few Dirichlet baselines
    for a in (0.2, 0.5, 1.0, 2.0):
        candidates_small.append(_random_dirichlet_candidate(n_small, a))

    # ------------------------------------------------------------------
    # Stage 1 – Random family search on reduced size
    family_samplers = [
        _sample_powerlaw,
        _sample_ushape,
        _sample_gaussian_edges,
        _sample_sine_valley,
        _sample_beta_shape,
        _sample_edge_block,
        _random_dirichlet,
    ]

    top_k = 12                         # keep this many across the whole search
    top_heap = []                      # min‑heap of (‑value, array) so that heap[0] is worst

    # helper to push a candidate onto the heap
    def _push_candidate(val, arr):
        if len(top_heap) < top_k:
            heapq.heappush(top_heap, (-val, arr))
        else:
            if -val > top_heap[0][0]:   # better than current worst
                heapq.heapreplace(top_heap, (-val, arr))

    # evaluate the deterministic seeds first
    for cand in candidates_small:
        val = _objective(cand)
        _push_candidate(val, cand)
        print(f"[Init deterministic] val={val:.6f}")

    # allocate about 30 % of total budget to this exploratory phase
    explore_deadline = time.time() + 0.30 * (deadline - time.time())
    total_trials = 0
    while time.time() < explore_deadline:
        sampler = np.random.choice(family_samplers)
        cand = sampler(n_small)
        # guarantee non‑negativity and normalise (samplers already do this)
        if cand.sum() < 1e-12:
            continue
        val = _objective(cand)
        _push_candidate(val, cand)
        total_trials += 1
        if (total_trials % 2000) == 0:
            worst_val = -top_heap[0][0]
            print(f"[Explore] tried {total_trials} trials, current worst of top-{top_k}: {worst_val:.6f}")

    print(f"[Explore] finished with {total_trials} trials; extracting top candidates.")

    # Extract the top candidates (as a list sorted by increasing value)
    top_candidates_small = [arr for _, arr in sorted(top_heap, key=lambda x: -x[0])]

    # ------------------------------------------------------------------
    # Stage 2 – Upscale & continuous refinement
    remaining_time = deadline - time.time()
    if remaining_time < 10.0:
        # Not enough time left – just return the best small candidate upscaled.
        best_small = top_candidates_small[0]
        best_up = np.interp(
            np.linspace(0.0, 1.0, n_big, endpoint=False),
            np.linspace(0.0, 1.0, n_small, endpoint=False),
            best_small,
        )
        best_up = np.maximum(best_up, 0.0)
        best_up /= best_up.sum()
        print("Time exhausted after exploration; returning upscaled best.")
        return best_up.tolist()

    # Give each top candidate an equal share of about half the remaining budget
    per_candidate_budget = 0.5 * remaining_time / len(top_candidates_small)

    best_seq = None
    best_val = np.inf

    for idx, cand_small in enumerate(top_candidates_small):
        # Upscale to full size
        x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        cand_big = np.interp(x_big, x_small, cand_small)
        cand_big = np.maximum(cand_big, 0.0)
        cand_big /= cand_big.sum()

        # ----- Adam phase -----
        adam_deadline = time.time() + per_candidate_budget * 0.55
        a_adam, val_adam = _adam_optimize(
            cand_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_deadline,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {idx} Adam] improved to {val_adam:.6f}")

        # ----- Sub‑gradient phase -----
        sub_deadline = time.time() + per_candidate_budget * 0.45
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            max_iters=250_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {idx} Subgrad] improved to {val_sub:.6f}")

        # Safety break if time is almost gone
        if time.time() > deadline - 15.0:
            print("Approaching deadline, breaking candidate loop.")
            break

    # ------------------------------------------------------------------
    # Stage 3 – Stochastic local hill‑climbing on the current best
    if best_seq is None:
        best_seq = _uniform_candidate(n_big)

    if time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            local_deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – Optional finer up‑sampling (8192) if enough time
    if time.time() < deadline - 30.0:
        n_fine = 8192
        conv_fft_len_fine = _next_pow2(2 * n_fine - 1)
        corr_fft_len_fine = _next_pow2(3 * n_fine - 2)

        # linear interpolation to the finer grid
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        x_fine = np.linspace(0.0, 1.0, n_fine, endpoint=False)
        a_fine = np.interp(x_fine, x_big, best_seq)

        a_fine = np.maximum(a_fine, 0.0)
        a_fine /= a_fine.sum()

        # brief Adam on fine grid
        fine_deadline = min(deadline - 5.0, time.time() + 120.0)
        a_fine_adam, val_fine_adam = _adam_optimize(
            a_fine,
            n_fine,
            conv_fft_len_fine,
            corr_fft_len_fine,
            fine_deadline,
            max_iters=200_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # sub‑gradient refinement on fine grid
        if time.time() < deadline - 5.0:
            a_fine_sub, val_fine_sub = _subgradient_optimize(
                a_fine_adam,
                n_fine,
                conv_fft_len_fine,
                corr_fft_len_fine,
                deadline,
                max_iters=200_000,
                eta0=0.6,
            )
            if val_fine_sub < best_val:
                best_val = val_fine_sub
                best_seq = a_fine_sub.copy()
                print(f"[Fine] improved to {val_fine_sub:.6f}")

    # ------------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
