def solve_lp_max_sum(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    b = np.full(2*n-1, M)

    warm_arr = np.asarray(warm, dtype=np.float64)
    if warm_arr.size > n:

    warm_arr = np.maximum(warm_arr, 0)
    if warm_arr.sum() < 0.01:

mixing f←(1−t)f+t·g gradually reduces the bound.
Finally a local random search refines the best solution.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)−1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(a))**2."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)

def _warm_start(n: int):
    """Extract warm start from global `height_sequence_1` if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s

def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)

def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a

def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)

def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _build_convolution_constraints(f: np.ndarray, M: float):
    """Return (A_ub, b_ub) for constraints (f * g)_i ≤ M."""
    n = f.shape[0]
    rows = []
    cols = []
    data = []
    # For each output index i (0..2n-2) we collect contributions f_j * g_{i-j}
    for i in range(2 * n - 1):
        # j in [max(0, i-(n-1)), min(i, n-1)]
        j_min = max(0, i - (n - 1))
        j_max = min(i, n - 1)
        # iterate over j and fill coefficient for g_{i-j}
        for j in range(j_min, j_max + 1):
            k = i - j  # index of g
            rows.append(i)
            cols.append(k)
            data.append(f[j])
    A_ub = csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    return A_ub, b_ub

def _lp_balancing_step(f: np.ndarray, t: float, deadline: float):
    """
    Perform one LP‑driven balancing step starting from `f`.
    Returns (f_new, val_new, success_flag).
    """
    n = f.shape[0]
    # current bound
    M = _full_convolution(f, _next_pow2(2 * n - 1)).max()

    # build LP constraints
    A_ub, b_ub = _build_convolution_constraints(f, M)

    # maximize sum(g)  ⇔ minimize -sum(g)
    c = -np.ones(n, dtype=np.float64)

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        return f, _objective(f), False

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0.0:
        return f, _objective(f), False

    # normalize g to have unit L1 norm (same as f)
    g_norm = g / sum_g

    # convex mixing
    f_new = (1.0 - t) * f + t * g_norm
    f_new = np.clip(f_new, 0.0, None)
    if f_new.sum() == 0.0:
        f_new = np.full_like(f_new, 1.0 / n)
    else:
        f_new /= f_new.sum()
    val_new = _objective(f_new)
    return f_new, val_new, True

def _lp_iterative_optimize(f0: np.ndarray, deadline: float):
    """Iteratively apply LP balancing until time runs out."""
    n = f0.shape[0]
    f = f0.copy()
    best_f = f.copy()
    best_val = _objective(f)
    iteration = 0
    # schedule for mixing weight t (decreasing)
    t_initial = 0.4
    t_min = 0.02
    decay = 0.95
    while time.time() < deadline:
        t = max(t_initial * (decay ** iteration), t_min)
        f_new, val_new, ok = _lp_balancing_step(f, t, deadline)
        if not ok:
            # if LP fails, break to avoid endless loop
            break
        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[LP it={iteration:04d}] improved value {best_val:.6f}")
        f = f_new
        iteration += 1
        # safety stop after many iterations
        if iteration >= 200:
            break
    return best_f, best_val

def _local_random_search(a0: np.ndarray, deadline: float, max_moves: int = 200_000):
    """Simple simulated‑annealing hill‑climbing."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    n = a.shape[0]
    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.5:
            # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:
            # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-16:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.5)  # up to 50%
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # Metropolis acceptance
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2
        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step function with a low evaluation value.
    The method builds diverse warm‑starts, runs an LP‑driven balancing
    phase on the best few, and finishes with a short stochastic local
    search.  It respects the 1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Choose problem size (trade‑off speed vs resolution)
    n = 1024
    # --------------------------------------------------------------
    # 1) Build candidate pool
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # shifted versions to diversify
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))

    for alpha in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
        candidates.append(_anti_pair_candidate(n, alpha))

    for p in (0.3, 0.7, 1.5):
        candidates.append(_powerlaw_candidate(n, p))

    # --------------------------------------------------------------
    # 2) Quick evaluation, keep the top few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = _random_dirichlet_candidate(n, 1.0)
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 5 == 0:
            print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 3
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) LP‑driven balancing on the top candidates
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.6 / top_k  # reserve some time for final search
    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        cand_dead = time.time() + per_candidate_budget
        f_opt, val_opt = _lp_iterative_optimize(cand, cand_dead)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = f_opt.copy()
            print(f"[Cand {cid} LP] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 4) Local stochastic refinement
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(best_seq, local_dead)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5) Safeguard – ensure a valid return
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # ensure positivity for evaluate_sequence (already non‑negative)
    if best_seq.sum() < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

        f_opt, val_opt = _lp_iterative_optimize(cand, cand_dead)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = f_opt.copy()
            print(f"[Cand {cid} LP] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic refinement
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(best_seq, local_dead)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Safeguard – guarantee a valid return
    if best_seq is None or len(best_seq) == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:
        best_seq = np.full_like(best_seq, 1.0 / n)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
