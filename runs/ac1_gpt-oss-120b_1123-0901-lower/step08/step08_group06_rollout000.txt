"""
Multi‑stage optimizer for the step‑function problem.

1.  Generates a diverse pool of initial shapes:
    – Uniform / triangle / Gaussian‑edge baselines.
    – Dirichlet samples with three different placements:
        * anti‑pair (largest opposite smallest, as in previous code),
        * alternating‑extremes (largest → leftmost, 2nd‑largest → rightmost, …),
        * plain random (no reordering).
    – Parametric “tail” families a_i ∝ (i+1)^(-γ)+(n-i)^(-γ) and
      distance‑decay a_i ∝ (min(i,n‑1‑i)+1)^(-δ) with several γ/δ values.
    The best of these is kept.

2.  Refines the best shape with an L‑BFGS optimizer on a smooth‑max
    surrogate (log‑sum‑exp of the convolution).  The temperature λ is
    increased in stages (5 → 20 → 80 → 200 → 500).

3.  Performs a long additive projected sub‑gradient descent (simplex
    projection) with a diminishing step‑size.  This phase usually yields
    the final improvement.

4.  Optional local stochastic hill‑climbing (swap / mass‑transfer moves)
    with an annealing schedule, used only if time permits.

All stages respect the global time budget (default 1000 s).  The
returned list is clipped to the limits accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (max‑convolution bound)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Arrangement helpers
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _alternating_extremes_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternating between the leftmost and rightmost
    free positions: largest → leftmost, 2nd largest → rightmost, 3rd → next
    left, etc.
    """
    n = len(vals)
    s = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    toggle = True
    for v in s:
        if toggle:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
        toggle = not toggle
    return a


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_extremes_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _alternating_extremes_arrangement(vals)


def _tail_distribution_candidate(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-γ) + (n-i)^(-γ) (U‑shaped tails)."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + 1) ** (-gamma) + (n - i) ** (-gamma)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _distance_decay_candidate(n: int, delta: float) -> np.ndarray:
    """a_i ∝ (min(i,n‑1‑i)+1)^(-δ)."""
    i = np.arange(n, dtype=np.float64)
    dist = np.minimum(i, n - 1 - i) + 1.0
    shape = dist ** (-delta)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 20.0, 80.0, 200.0, 500.0),
                    max_iter_stage: int = 300) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    import scipy.optimize as opt

    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # bound iterations by remaining time (10 it/s approx)
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 12)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.0f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient optimisation (exact max)
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 2_000_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:07d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 600_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                    # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:07d}] best {best_val:.6f}")

        it += 1
    return best_a, best_val

# Warm‑start handling
def _warm_start(n: int):
    """Return a stored sequence from the harness if present."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    best_seq = None
    best_val = float("inf")

        best_seq, best_val = warm.copy(), cand_val
        print(f"[Warm start] value {cand_val:.6f}")

    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),

    for c in starters:
