def _simulated_annealing(a0, n, conv_fft_len, deadline, max_moves=200000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    val = best_val

    # temperature schedule
    T = 0.1
    decay = 0.9999995 # adjust
    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * np.random.rand() * 0.5   # move up to 50% mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        # acceptance
        if val2 < val or np.random.rand() < np.exp(-(val2-val)/max(T,1e-12)):
            a = a2
            val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[SA it={it}] new best {val2:.6f}")
        # update temperature
        T *= decay
        it += 1
    return best_a, best_val
```

def _anti_pair_sequence(orig):
    sorted_vals = np.sort(orig)[::-1]   # descending
    n = sorted_vals.shape[0]
    a = np.empty_like(sorted_vals)
    left = 0

    for idx, v in enumerate(sorted_vals):

            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
    return a

"""
Propose a non‑negative step function (vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search consists of several stages that respect a 1000 s budget:

1.  Warm start from the previously best sequence (if available).
2.  A large random‑sampling phase, now enriched with
    * heavy‑tailed weight samples (log‑normal / gamma)
    * an “anti‑pairing’’ ordering that tries to keep large values
      far from each other (large ↔ small placement).
3.  Long Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate.
4.  Exponentiated‑gradient sub‑gradient descent on the exact max.
5.  Simulated‑annealing (SA) that moves a random fraction of mass
    between two positions, accepting uphill moves with a decaying
    temperature.
6.  Very cheap local hill‑climb as a final polish.

All phases constantly keep the best vector seen so far and always
return a feasible (non‑negative, normalised) sequence before the
deadline.  The implementation uses FFT‑based convolutions for speed
and works for any length (default 4096).  The hyper‑parameters have
been tuned to give a good chance of beating the previous bound
≈1.5077 within the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _anti_pair_sequence(v: np.ndarray) -> np.ndarray:
    """
    Order the entries so that large values are paired with small ones.
    Largest goes to the leftmost free slot, second largest to the rightmost,
    third to the next left slot, etc.
    """
    sorted_vals = np.sort(v)[::-1]          # descending
    n = sorted_vals.shape[0]
    out = np.empty_like(sorted_vals)
    left, right = 0, n - 1
    for idx, val in enumerate(sorted_vals):
        if idx % 2 == 0:
            out[left] = val
            left += 1
        else:
            out[right] = val
            right -= 1
    return out

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.08,
    lam_factor: float = 1.015,
    noise_sigma: float = 0.025,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        # soft‑max projection (onto the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for book‑keeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp with temperature 1/lam)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # injected noise for exploration
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e6)

        # learning‑rate decay every 50 k steps
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake near the deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climbing: move a random fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * np.random.rand() * 0.3   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1
    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simple Metropolis‑type SA on the exact objective."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    # temperature schedule
    T = 0.01
    decay = 0.999995   # slowly decays; ~150k steps → T ≈ 0.2

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * np.random.rand() * 0.5   # up to 50 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        # Metropolis acceptance
        if val2 < val or np.random.rand() < np.exp(-(val2 - val) / max(T, 1e-12)):
            a = a2
            val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[SA it={it:06d}] new best {val2:.6f}")

        T *= decay
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (larger n gives a tighter objective, but also more work)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from the previous global if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # ---------- Fast random generation phase (≈30 s) ------------------
    best_seq = None
    best_val = float("inf")

    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # baseline
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # Dirichlet (full support)
    for _ in range(12):
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # Simple families (already used before)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # Half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # Spike candidates with anti‑pair ordering
    for k in (32, 64, 128, 256, 512):
        for _ in range(8):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            a = _anti_pair_sequence(a)       # scramble to anti‑pair large/small
            a /= a.sum()
            try_candidate(a)

    # Heavy‑tailed families (log‑normal / gamma) + anti‑pair
    for dist in ("lognorm", "gamma"):
        for _ in range(20):
            if dist == "lognorm":
                v = np.random.lognormal(mean=0.0, sigma=1.0, size=n)
            else:
                v = np.random.gamma(shape=0.5, scale=1.0, size=n)
            a = _anti_pair_sequence(v)
            a /= a.sum()
            try_candidate(a)

    # Two‑Gaussian mixtures
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(12):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        try_candidate(a)

    # ------------------------------------------------------------------
    # -------------------------------------------------------- Adam phase
    remaining = deadline - time.time()
    if remaining <= 5:
        # hardly any time left – just return the best we have
        final_seq = best_seq if best_seq is not None else np.full(n, 1.0 / n)
        final_seq = np.clip(final_seq, 0.0, 1000.0)
        print(f"Final best value {best_val:.6f} (length {len(final_seq)})")
        return final_seq.tolist()

    # allocate roughly 45 % of the remaining budget to Adam
    adam_deadline = time.time() + 0.45 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=250_000,
        lr=0.07,
        lam_start=0.08,
        lam_factor=1.015,
        noise_sigma=0.03,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # --------------------------------------------------- Noisy‑restart Adam
    for restart in range(3):
        if time.time() >= deadline - 30:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.35, size=n)   # stronger perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()
        r_deadline = min(deadline - 5, time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=150_000,
            lr=0.06,
            lam_start=0.08,
            lam_factor=1.02,
            noise_sigma=0.025,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # --------------------------------------------------- Sub‑gradient refinement
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # --------------------------------------------------- Simulated annealing phase
    if time.time() < deadline - 15:
        sa_deadline = deadline - 10
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            sa_deadline,
            max_moves=180_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # --------------------------------------------------- Cheap local search
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid non‑empty return (fallback)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
