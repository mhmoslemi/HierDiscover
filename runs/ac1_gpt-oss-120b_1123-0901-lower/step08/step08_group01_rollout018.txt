"""
Propose a high‑quality step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

where a∗a is the linear convolution of a with itself.
The optimisation proceeds in three stages:

1. **Diverse warm‑starts** – a large pool of deterministic families
   (exponential, power‑law, beta, Gaussian, sinusoid, uniform) together
   with many random Dirichlet draws and sparse‑spike vectors.
   The best few are kept for deep refinement.

2. **Deep refinement** – each selected start is optimised with a
   sequence of
   * Adam on a smooth‑max surrogate (log‑sum‑exp) with a slowly increasing
     temperature,
   * Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution,
   * Euclidean projected‑gradient descent (with optional Nesterov momentum),
   * A cheap random hill‑climber for the final seconds.
   All phases respect the global deadline and always keep the best vector
   seen so far.

3. **Finalisation** – the globally best vector is clipped to the evaluator’s
   domain and returned as a plain Python list.

The implementation relies on FFT‑based convolution for speed and contains
no closures – every helper is a top‑level function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate factories (deterministic families)
def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑pdf on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _center_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the centre."""
    centre = (n - 1) / 2.0
    d = np.abs(np.arange(n, dtype=np.float64) - centre)
    shape = np.exp(-gamma * d)
    shape /= shape.sum()
    return shape


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k equal spikes at random locations."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum==1) observed and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02                 # inverse temperature (small ⇒ smooth)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max (probability simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – tighten progressively
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 100000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimiser on the exact max.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent (optionally with Nesterov momentum).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        # Nesterov‑style momentum
        momentum = 0.9 * momentum - eta * grad_a
        a_next = a + momentum
        a = _project_onto_simplex(a_next)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)  # move up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Deep optimisation wrapper (no closures)
def deep_optimize(
    a_init: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Run the full stack of optimisers on `a_init` while respecting the deadline.
    Returns the best vector encountered and its exact objective value.
    """
    cur_a = a_init.copy()
    cur_val = float("inf")

    # Stage 1 – Adam (smooth‑max)
    a_adam, val_adam = _adam_optimize(
        cur_a,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=1_000_000,
        lr_start=0.05,
    )
    if val_adam < cur_val:
        cur_a, cur_val = a_adam, val_adam

    # Stage 2 – Sub‑gradient (exponentiated‑gradient)
    a_sub, val_sub = _subgradient_optimize(
        cur_a,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=1_000_000,
        eta0=0.9,
    )
    if val_sub < cur_val:
        cur_a, cur_val = a_sub, val_sub

    # Stage 3 – Projected‑gradient (with momentum)
    a_pgd, val_pgd = _pgd_optimize(
        cur_a,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=800_000,
        eta0=0.5,
    )
    if val_pgd < cur_val:
        cur_a, cur_val = a_pgd, val_pgd

    # Stage 4 – Cheap random local moves for the last seconds
    a_loc, val_loc = _local_random_search(
        cur_a,
        n,
        conv_fft_len,
        deadline,
        max_moves=150_000,
    )
    if val_loc < cur_val:
        cur_a, cur_val = a_loc, val_loc

    return cur_a, cur_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size – a longer vector gives the optimisation more freedom
    n = 4096  # power of two, works well with the FFT utilities
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a pool of diverse starts
    pool = []

    # deterministic families
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        pool.append(_edge_exponential(n, gamma))
    for alpha in (-3.0, -2.5, -2.0, -1.5, -1.0):
        pool.append(_edge_power_law(n, alpha))
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))
    for sigma in (0.03, 0.06, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    for gamma in (0.05, 0.1, 0.2, 0.5):
        pool.append(_center_exponential(n, gamma))

    if warm is not None:
        # include warm start and a few cyclic shifts
        pool.append(warm)
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Random Dirichlet draws with a wide range of concentration parameters
    for alpha in (0.01, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0, 10.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    # Sparse random spikes
    for k in (8, 16, 32, 64, 128, 256):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick evaluation of the pool – keep the best few
    evaluated = []
    for idx, cand in enumerate(pool):
        val = evaluate_sequence(cand.tolist())
        evaluated.append((val, cand))
        if idx % 20 == 0:
            print(f"[Pool] candidate {idx} => {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    top_candidates = evaluated[:top_k]

    best_seq = top_candidates[0][1].copy()
    best_val = top_candidates[0][0]
    print(f"Initial best from pool: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Deep optimisation of the selected top candidates
    for rank, (val, cand) in enumerate(top_candidates, start=1):
        if time.time() >= deadline - 5:
            break
        print(f"--- Deep optimisation of candidate #{rank} (initial {val:.6f}) ---")
        opt_seq, opt_val = deep_optimize(cand, n, conv_fft_len, corr_fft_len, deadline)
        if opt_val < best_val:
            best_val = opt_val
            best_seq = opt_seq.copy()
            print(f"*** New global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
