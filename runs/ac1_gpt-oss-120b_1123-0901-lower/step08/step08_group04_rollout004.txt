import time, math, numpy as np
try:
    from scipy.optimize import linprog
    from scipy.sparse import diags, csr_matrix
    SCIPY_AVAILABLE = True
except Exception:
    SCIPY_AVAILABLE = False

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*a.shape[0]-1]

def _objective(a):
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2*a.shape[0]-1))
    return float(2 * a.shape[0] * conv.max() / (s*s))

def _max_convolution(a):
    conv = _full_convolution(a, _next_pow2(2*a.shape[0]-1))
    return conv.max(), conv

def _lp_step(f, M, sqrt2n):
    """
    Solve LP: max sum(b) s.t. b>=0, conv(f,b) <= M.
    Return scaled g with sum sqrt2n.
    """
    n = f.shape[0]
    # Build sparse convolution matrix A where (A b)_k = sum_i f_i b_{k-i}
    # Diagonal offsets i from 0..n-1 have entries f_i
    data = [np.full(n, f[i]) for i in range(n)]
    offsets = np.arange(n)
    A = diags(data, offsets, shape=(2*n-1, n), format='csr')
    # RHS
    ub = np.full(2*n-1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)   # maximize sum(b) => minimize -sum(b)
    bounds = [(0, None)] * n

    # Solve
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return None
        return None
    b = res.x
    S = b.sum()
    if S < 1e-12:
        return None
    # Scale to have sum sqrt2n
    g = b * (sqrt2n / S)
    return g

def _apply_delta(a, conv, idx, delta):
    """
    Increment a[idx] by delta and update convolution a*a.
    """
    # conv length is 2*len(a)-1
    n = a.shape[0]
    # Update conv at positions idx...idx+n-1
    conv[idx: idx + n] += 2.0 * delta * a
    # self-product term at 2*idx
    conv[2*idx] += delta * delta
    a[idx] += delta

def _incremental_sa(a0, time_limit, start_temp=0.02, max_frac=0.4):
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)
    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val
    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            continue
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # transfer a fraction of a[i] to a[j]
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0:
            continue
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)
        new_val = conv.max()
        diff = new_val - cur_val
        # linear cooling
        elapsed = time.time() - start
        remain = max(0.0, time_limit - time.time())
        temp = start_temp * (remain / (time_limit - start))
        if diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp)):
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert move
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)
        it += 1
        if it % 200000 == 0:
            print(f"[SA it={it:07d}] best {best_val:.6f}")
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with low
    ``2·n·max(a*a)/(sum a)^2``.
    The method builds diverse warm‑starts, refines them by an
    alternating linear‑program (maximising the total mass under a
    convolution bound) and finishes with a fast incremental
    simulated‑annealing local search.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    # Problem size – 4096 proved a good trade‑off.
    n = 4096
    sqrt2n = math.sqrt(2.0 * n)

    # ---- 1. warm‑starts -------------------------------------------------
    def _warm_start(n):
        warm = globals().get("height_sequence_1")
        if warm is None:
            return None
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() < 0.01:
            return None
        warm = warm / warm.sum() * sqrt2n
        return warm

    # candidate generators (non‑exhaustive)
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n//4, n//2, 3*n//4):
            candidates.append(np.roll(warm, shift))

    # simple patterns
    candidates.append(np.full(n, sqrt2n / n, dtype=np.float64))      # uniform on simplex sqrt2n
    # Dirichlet samples
    for alpha in (0.3, 0.7, 1.0, 2.0):
        d = np.random.dirichlet(alpha * np.ones(n))
        candidates.append(d * sqrt2n)
    # anti‑pair arrangements
    for alpha in (0.3, 0.7, 1.0):
        d = np.random.dirichlet(alpha * np.ones(n))
        # apply anti‑pair arrangement
        sorted_vals = np.sort(d)[::-1]
        a = np.empty_like(d)
        half = n // 2
        a[:half] = sorted_vals[:half]
        a[-half:] = sorted_vals[-half:][::-1]
        if n % 2 == 1:
            a[half] = sorted_vals[half]
        candidates.append(a * sqrt2n)
    # edge‑Gaussian
    for gamma in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
        shape /= shape.sum()
        candidates.append(shape * sqrt2n)
    # triangle shape
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x+1, n-x)
    tri /= tri.sum()
    candidates.append(tri * sqrt2n)

    # ---- 2. select best start -----------------------------------------
    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:

        # already normalised to sqrt2n

        conv = _full_convolution(cand, _next_pow2(2*n-1))
        val = 2.0 * n * conv.max() / (cand.sum()**2)   # = conv.max()*2*n/(2n)=conv.max()?
        # Actually with sum sqrt2n, denom = 2n => val = conv.max()

        print(f"[Init cand {idx:02d}] obj {val_exact:.6f}")
        if val_exact < best_val:
            best_val = val_exact
            best_seq = cand.copy()

    # ----- 3. LP refinement ---------------------------------------------

    it_lp = 0
    while time.time() < lp_deadline and SCIPY_AVAILABLE:

        # Solve LP to obtain g
        g = _lp_step(best_seq, M, sqrt2n)

    if time.time() < deadline - 5.0:

        final_max = sa_max
        best_seq = sa_seq
        print(f"[SA finished] best conv max {final_max:.6f}")

    total = best_seq.sum()
    if total < 1e-12:

    else:

"""
Propose a step‑function with a very low value of
    F(a)=2·n·max(a∗a)/(∑a)² .

The search consists of three phases

1. **Warm‑starts** – a small, diverse pool of candidates (uniform,
   Dirichlet, anti‑pair, edge‑Gaussian, triangular) is built and the
   best one (according to the exact objective) is kept.

2. **Linear‑programming refinement** – following the classical
   “alternating LP” scheme:  for the current vector f (normalised to
   √(2·n)) we compute M=‖f∗f‖∞, then solve a LP that maximises the
   total mass of a vector g under the linear constraints
   (f∗g)≤M.  After scaling g to the same ℓ₁‑norm as f we replace
   f←(1‑t)f+t·g with a small mixing factor t.  The maximum of the
   self‑convolution never increases, so the objective improves
   monotonically.  Only a handful of LP iterations are performed
   (each takes ≲0.5 s for n=4096).

3. **Fast simulated‑annealing (incremental)** – the final vector is
   fine‑tuned by a long SA run that updates the full convolution in
   O(n) per move using a simple mass‑transfer move.  The temperature
   decays linearly and millions of moves are executed in the remaining
   time budget.

All vectors are kept non‑negative; the final result is re‑scaled to
∑a=1 and clipped to the limits required by ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# optional – scipy is used only for the LP refinement
try:
    from scipy.optimize import linprog          # type: ignore
    from scipy.sparse import diags, csr_matrix  # type: ignore
    _SCIPY = True
except Exception:  # pragma: no cover
    _SCIPY = False


# ----------------------------------------------------------------------
# FFT utilities (convolution, objective)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(a*a)/(sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return float(2.0 * a.shape[0] * conv.max() / (s * s))


def _max_convolution(a: np.ndarray):
    """Return (max, full convolution) of a with itself."""
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return conv.max(), conv


# ----------------------------------------------------------------------
# Linear‑programming step (alternating LP scheme)
def _lp_step(f: np.ndarray, M: float, sqrt2n: float):
    """
    Solve  max ∑ b_j   s.t.   b ≥ 0 ,   (f∗b) ≤ M .
    Return g = (sqrt(2·n)/∑b)·b  (same ℓ₁‑norm as f).
    If the LP fails, ``None`` is returned.
    """
    n = f.shape[0]

    # Build the sparse convolution matrix A where (A·b)_k = (f∗b)_k .
    # Each diagonal offset i (0 ≤ i < n) contains the constant f[i].
    data = [np.full(n, f[i], dtype=np.float64) for i in range(n)]
    offsets = np.arange(n, dtype=int)
    A = diags(data, offsets, shape=(2 * n - 1, n), format="csr")

    # Upper bounds – all entries of f∗b must not exceed M.
    ub = np.full(2 * n - 1, M, dtype=np.float64)

    # Linear programme: minimise –∑b  (i.e. maximise ∑b)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(
        c,
        A_ub=A,
        b_ub=ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s < 1e-15:
        return None

    # Scale to the common ℓ₁‑norm √(2·n)
    g = b * (sqrt2n / s)
    return g


# ----------------------------------------------------------------------
# Incremental convolution update (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    convolution a*a in O(n) using the linear structure of the convolution.
    """
    n = a.shape[0]
    # contributions that involve a[idx]
    conv[idx : idx + n] += 2.0 * delta * a
    # the self‑product term for the index 2·idx
    conv[2 * idx] += delta * delta
    a[idx] += delta


def _incremental_sa(a0: np.ndarray, time_limit: float,
                    start_temp: float = 0.02, max_frac: float = 0.4):
    """
    Simulated‑annealing local search with O(n) incremental convolution
    updates.  Mass is transferred from a random source index to a random
    destination index.  The temperature decays linearly.
    Returns (best_vector, best_conv_max).
    """
    a = a0.copy()
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            continue
        # pick a destination different from i
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear cooling schedule
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert the move
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 200_000 == 0:
            print(f"[SA it={it:07d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a list of non‑negative heights that (hopefully) improves the
    current best upper bound.  The routine combines diverse warm‑starts,
    a few iterations of the alternating LP refinement and a long
    incremental simulated‑annealing finish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 gave a good trade‑off in earlier experiments
    n = 4096
    sqrt2n = math.sqrt(2.0 * n)

    # ------------------------------------------------------------------
    # 1) build a pool of warm‑starts
    candidates = []

    # possible warm‑start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() > 0.01:
            w = w / w.sum() * sqrt2n
            candidates.append(w.copy())
            # a few rotations – sometimes beneficial
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(w, shift))

    # uniform on the simplex √(2·n)
    candidates.append(np.full(n, sqrt2n / n, dtype=np.float64))

    # Dirichlet samples
    for a in (0.3, 0.7, 1.0, 2.0):
        d = np.random.dirichlet(a * np.ones(n))
        candidates.append(d * sqrt2n)

    # anti‑pair (large opposite small)
    for a in (0.3, 0.7, 1.0):
        d = np.random.dirichlet(a * np.ones(n))
        sorted_vals = np.sort(d)[::-1]                # descending
        ap = np.empty_like(d)
        half = n // 2
        ap[:half] = sorted_vals[:half]
        ap[-half:] = sorted_vals[-half:][::-1]
        if n % 2:
            ap[half] = sorted_vals[half]
        candidates.append(ap * sqrt2n)

    # edge‑focused Gaussian
    for gamma in (0.2, 0.5, 1.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape * sqrt2n)

    # simple triangle (symmetric, larger at edges)
    x = np.arange(n, dtype=np.float64)
    tri = np.minimum(x + 1, n - x)
    tri /= tri.sum()
    candidates.append(tri * sqrt2n)

    # ------------------------------------------------------------------
    # 2) pick the best starter (exact evaluation)
    best_seq = None
    best_val = np.inf
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n)) * sqrt2n
        cand = cand / cand.sum() * sqrt2n
        val = _objective(cand)          # exact objective for the current norm
        print(f"[Init cand {idx:02d}] obj {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # ------------------------------------------------------------------
    # 3) alternating LP refinement (monotone descent)
    # keep a few seconds for the final SA
    lp_deadline = deadline - 30.0
    lp_iter = 0
    while time.time() < lp_deadline and _SCIPY:
        # current maximum of the self‑convolution
        M, _ = _max_convolution(best_seq)

        # solve the LP, obtain a new vector g
        g = _lp_step(best_seq, M, sqrt2n)
        if g is None:
            break

        # convex combination – small mixing factor keeps the descent monotone
        t = 0.07
        mixed = (1.0 - t) * best_seq + t * g
        new_M, _ = _max_convolution(mixed)

        if new_M < M - 1e-12:          # accept only if the bound improves
            best_seq = mixed
            best_val = 2.0 * n * new_M / (sqrt2n ** 2)  # = new_M (since sum=√(2n))
            lp_iter += 1
            if lp_iter % 5 == 0:
                print(f"[LP iter {lp_iter}] conv‑max {new_M:.6f}  obj {best_val:.6f}")
        else:
            # no improvement – stop the LP phase
            break

    # ------------------------------------------------------------------
    # 4) long incremental SA for fine‑tuning
    if time.time() < deadline - 5.0:
        sa_seq, sa_max = _incremental_sa(
            best_seq,
            time_limit=deadline - 5.0,
            start_temp=0.03,
            max_frac=0.4,
        )
        best_seq = sa_seq
        best_val = 2.0 * n * sa_max / (sqrt2n ** 2)  # = sa_max
        print(f"[SA] final conv‑max {sa_max:.6f}")

    # ------------------------------------------------------------------
    # 5) final safety checks and scaling to ∑a=1
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / total                     # now ∑a = 1

    final_obj = _objective(best_seq)
    print(f"Final objective {final_obj:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
