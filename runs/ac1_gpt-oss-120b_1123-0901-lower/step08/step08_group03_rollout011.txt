def upscale(a, n_new):
    factor = n_new // len(a)
    if n_new % len(a) == 0:
        a_up = np.repeat(a, factor)
    else:
        # use np.interp
        x_old = np.linspace(0, 1, len(a))
        x_new = np.linspace(0, 1, n_new)
        a_up = np.interp(x_new, x_old, a)
    return a_up / a_up.sum()

"""
propose_candidate
-----------------

Multi‑phase stochastic search for a non‑negative step function that
minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine runs completely inside the given time budget (default 1000 s)
and always returns the best sequence found so far.

Algorithm outline
-----------------
1. **Diverse random seeding** – generate a large pool of candidates from
   several analytically‑motivated families (Dirichlet, Beta, exponential,
   sinusoid, Gaussian mixtures, sparse spikes).  The 50‑best are kept.

2. **Convex blending** – repeatedly form random convex combinations of the
   stored elite candidates (with optional tiny Gaussian noise).  New blends
   that improve the objective replace the worst elite entry.

3. **Large‑scale smooth‑max optimisation** – run a long Adam optimisation
   on the smooth‑max surrogate (log‑sum‑exp of the convolution) starting
   from the current best candidate.  The temperature (λ) is increased very
   slowly so that the optimiser can escape shallow basins.

4. **Exact max sub‑gradient phase** – an exponentiated‑gradient (mirror
   descent on the simplex) iteration that directly follows the true
   max‑convolution objective.

5. **Cheap local hill‑climb** – random pairwise mass transfers that keep
   improvements; useful for polishing the final solution.

All heavy loops check the global deadline, so the routine adapts automatically
to the available time.  The final vector is clipped to the limits imposed by
`evaluate_sequence` (non‑negative, ≤1000) and returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr: float = 0.05,
    lam_start: float = 0.05,
    lam_factor: float = 1.008,
    noise_sigma: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(np.clip(a0, eps, None))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # Soft‑max to enforce simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp with temperature 1/lam)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate (see paper)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Optional stochastic perturbation
        if noise_sigma > 0.0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # Temperature schedule (slow increase)
        lam = min(lam * lam_factor, 1e6)

        # Learning‑rate decay every 100 k steps
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        # Small random shake near the deadline
        if time.time() > deadline - 30 and it % 2_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # Active set where conv attains its maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Decreasing step size η_t = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # Exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # Renormalise, protecting against under‑/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:07d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

    # baseline (warm start or uniform)
    if warm is not None:
        _maybe_store(eval_obj(warm), warm)
    else:
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        _maybe_store(eval_obj(uniform), uniform)

    # --------------------------------------------------------------
    # 1️⃣  Wide random exploration
    # --------------------------------------------------------------
    # Dirichlet draws with various concentration values
    for conc in (0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(10):
            a = np.random.dirichlet(np.full(n, conc))
            _maybe_store(eval_obj(a), a)

    # Beta‑family shapes (symmetric and asymmetric)
    for a_param in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        for b_param in (0.2, 0.5, 1.0, 2.0, 3.0):
            x = np.linspace(0.0, 1.0, n)
            a = (np.power(x + 1e-12, a_param - 1) *
                 np.power(1 - x + 1e-12, b_param - 1))
            a /= a.sum()
            _maybe_store(eval_obj(a), a)

    # Exponential “double‑sided” families
    for gamma in (0.05, 0.1, 0.2, 0.4, 0.8, 1.6):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        _maybe_store(eval_obj(a), a)

    # Half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    _maybe_store(eval_obj(a), a)

    # Random two‑Gaussian mixtures
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(20):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        _maybe_store(eval_obj(a), a)

    # Sparse spike patterns
    for k in (64, 128, 256, 512):
        for _ in range(8):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            _maybe_store(eval_obj(a), a)

    # --------------------------------------------------------------
    # 2️⃣  Convex blending of elite candidates
    # --------------------------------------------------------------
    blend_deadline = deadline - 120.0   # keep ~2 min for later refinements
    while time.time() < blend_deadline:
        k = np.random.randint(2, 7)          # blend 2‑6 elites
        chosen = np.random.choice(len(elite), size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))
        a_blend = np.zeros(n, dtype=np.float64)
        for w, idx in zip(weights, chosen):
            a_blend += w * elite[idx][1]
        # tiny random jitter helps to break symmetry
        if np.random.rand() < 0.15:
            a_blend += np.random.normal(scale=1e-4, size=n)
        a_blend = np.clip(a_blend, 0.0, None)
        if a_blend.sum() < 1e-12:
            continue
        a_blend /= a_blend.sum()
        _maybe_store(eval_obj(a_blend), a_blend)

    # --------------------------------------------------------------
    # Ensure we have a starting point for the next phases
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation on smooth‑max surrogate
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60.0:
        adam_deadline = time.time() + 0.60 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=1_200_000,   # large upper bound; loop respects deadline
            lr=0.05,
            lam_start=0.05,
            lam_factor=1.008,
            noise_sigma=0.02,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam main] improved best to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Noisy‑restart Adam (few short runs)
    # --------------------------------------------------------------
    for restart in range(3):
        if time.time() >= deadline - 30.0:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.25, size=n)   # moderate perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()

        r_deadline = min(deadline - 5.0, time.time() + 0.10 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=200_000,
            lr=0.045,
            lam_start=0.05,
            lam_factor=1.01,
            noise_sigma=0.015,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Exact‑max sub‑gradient refinement
    # --------------------------------------------------------------
    if time.time() < deadline - 20.0:
        sg_deadline = deadline - 15.0
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=300_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final cheap local hill‑climb
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # Final safety checks and return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
