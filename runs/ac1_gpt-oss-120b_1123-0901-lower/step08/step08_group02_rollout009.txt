def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)

    fft_len = 1 << ((2*n-2).bit_length())   # nearest power of two >= 2n-1
    # functions conv_fft, cross_corr defined earlier

    # define objective function
    def objective(a):
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        s = a.sum()
        if s < 0.01:
            return float('inf')
        return 2.0 * n * max_c / (s * s)

    # Candidate generation
    candidates = []

    # Add some deterministic candidates from previous code if available
    warm = globals().get('height_sequence_1')
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n-w.shape[0]))
        w = np.maximum(w, 0.0)
        w = w / w.sum()
        candidates.append(w)

    # Uniform
    candidates.append(np.full(n, 1.0/n))

    # Half wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0)
    s = s / s.sum()
    candidates.append(s)

    # Edge exponential (left)
    for gamma in [0.1,0.2,0.5,1.0,2.0]:
        left = np.exp(-gamma*np.arange(n))
        shape = left / left.sum()
        candidates.append(shape)

    # Edge exponential right
        right = left[::-1]
        shape = right / right.sum()
        candidates.append(shape)

    # Edge polynomial (power -p)
    prod = np.arange(1,n+1)*(n - np.arange(1,n+1) +1)
    for alpha in np.linspace(-2.0,-0.2,9):
        shape = np.where(prod>0, prod**alpha, 0.0)
        if shape.sum()>0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # Double Gaussian (multiple sigma)
    i = np.arange(n, dtype=np.float64)
    for sigma in [0.05, 0.1, 0.2, 0.4]:
        g = np.exp(-0.5 * ((i)/(sigma*n))**2) + np.exp(-0.5 * ((i-(n-1))/(sigma*n))**2)
        g = g / g.sum()
        candidates.append(g)

    # Random Dirichlet (various concentration)
    for alpha in [0.2,0.5,1.0,2.0,5.0]:
        candidates.append(np.random.dirichlet(np.full(n, alpha)))

    # Random beta shapes
    for _ in range(50):
        a = 10**(-np.random.rand()*2)  # a in [0.01,1]
        b = 10**(-np.random.rand()*2)
        x = (np.arange(n)+0.5)/n
        shape = np.power(x, a-1) * np.power(1-x, b-1)
        # avoid overflow / NaN by replacing non-finite
        shape = np.where(np.isfinite(shape) & (shape>0), shape, 0.0)
        if shape.sum()>0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # Random mixtures of two shapes
    for _ in range(30):
        # choose two random candidates from earlier pool (except warm)
        s1 = candidates[np.random.randint(len(candidates))]
        s2 = candidates[np.random.randint(len(candidates))]
        w = np.random.rand()
        mix = w*s1 + (1-w)*s2
        mix = mix / mix.sum()
        candidates.append(mix)

    # Evaluate all candidates quickly

"""
Propose a step‑function sequence that (hopefully) beats the previous best
value ≈1.5077.  The algorithm works in three stages:

1. **Diverse warm‑starts** – a large pool of candidate shapes
   (uniform, half‑wave sinusoid, left/right exponentials,
   power‑law, double‑Gaussian, random Dirichlet, Beta PDFs,
   random mixtures, asymmetric left/right blends) is generated.
   Each candidate is normalised to sum‑one and its exact objective
   `2·n·max(a*a)/(sum(a)^2)` is computed with a fast FFT convolution.
   The best few (default 5) are kept.

2. **Refinement** – each elite candidate is refined by two gradient‑based
   phases:

   * **Adam on a smooth‑max surrogate** – we minimise
     `log‑sum‑exp( λ·conv(a) )` with increasing temperature λ.
     The gradient is obtained via a cross‑correlation computed in the
     frequency domain.  The iterates live on the probability simplex
     using a soft‑max parametrisation (logits).  The schedule for λ
     grows slowly (factor 1.02 per iteration) so that the optimiser can
     first move freely and then focus on the true maximum.

   * **Exact sub‑gradient (exponentiated‑gradient)** – after the Adam
     phase we switch to the true maximum and perform mirror‑descent
     updates on the simplex, decreasing the step‑size as η₀/√t.

   The best vector seen across all phases is kept.

3. **Cheap local moves** – a final hill‑climbing stage repeatedly transfers
   a random fraction of mass from one index to another and accepts the
   move only if the objective improves.  This can shave off the last
   few thousandths.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a feasible non‑negative list.  It uses only NumPy and the
standard library, works on a single thread and makes no filesystem or
network accesses.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],  i = 0…n‑1
    where n = len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the competition objective.
    a must be non‑negative; it is assumed to be already normalised."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000,
                   lr: float = 0.06,
                   lam0: float = 0.1,
                   lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns (best_vector, best_value) where best_value is the true
    objective (not the smooth approximation).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)        # parametrisation on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam0

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection (maintains sum‑to‑one)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val, best_a = val, a.copy()
            # occasional progress report
            if it % 25_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule for the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly increase the focus on the true max
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake close to the deadline
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # optional decay of learning rate
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 400_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation (exponentiated‑gradient on the simplex).
    Returns (best_vector, best_value).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 25_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, regenerate if numerical trouble
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from i→j.
    Keep the move only if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 25 % of the source mass
        delta = a[i] * (np.random.rand() * 0.25)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the time budget and always returns the best
    sequence found so far (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise a pool of diverse candidates
    pool = []

    # warm start if supplied by the environment
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w.copy())
            # also try a reversed version (introduces asymmetry)
            pool.append(w[::-1].copy())

    # deterministic families
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))                     # uniform
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)                         # half‑wave sinusoid
    s = np.maximum(s, 0.0)
    pool.append(s / s.sum())

    # left‑only exponential families (asymmetric)
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        left = left / left.sum()
        pool.append(left.copy())               # pure left
        # blend left/right with a random weight
        wgt = np.random.rand()
        blended = wgt * left + (1 - wgt) * left[::-1]
        pool.append(blended / blended.sum())

    # power‑law / edge‑polynomial (prod**α) with finer grid
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.1, 20):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # double‑Gaussian (mass at both ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        g = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
            np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        g = g / g.sum()
        pool.append(g)

    # random Dirichlet draws with random concentration
    for _ in range(30):
        alpha = np.random.rand() * 5.0 + 0.1  # α ∈ [0.1,5.1]
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # random Beta‑pdf shapes (asymmetric by construction)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for _ in range(100):
        a = 10 ** (np.random.uniform(-2, 0))   # a ∈ [0.01,1]
        b = 10 ** (np.random.uniform(-2, 0))   # b ∈ [0.01,1]
        shape = np.power(x, a - 1.0) * np.power(1 - x, b - 1.0)
        shape = np.where(np.isfinite(shape) & (shape > 0), shape, 0.0)
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum()
        pool.append(shape)

    # random mixtures of two already built shapes
    for _ in range(80):
        s1 = pool[np.random.randint(len(pool))]
        s2 = pool[np.random.randint(len(pool))]
        w = np.random.rand()
        mix = w * s1 + (1 - w) * s2
        mix = mix / mix.sum()
        pool.append(mix)

    # ------------------------------------------------------------------
    # quick evaluation of all candidates, keep the top few
    top_k = 6
    best_candidates = []   # list of (value, vector)
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        # protect against pathological zero‑sum candidates
        if cand.sum() < 1e-12:
            continue
        cand = cand / cand.sum()
        val = _objective(cand, n, conv_fft_len)
        if len(best_candidates) < top_k:
            best_candidates.append((val, cand.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, cand.copy())
                best_candidates.sort(key=lambda x: x[0])

    # keep only the vectors (discard the values)
    elite = [vec for _, vec in best_candidates]
    print(f"Initial elite size {len(elite)} – best value {best_candidates[0][0]:.6f}")

    # ------------------------------------------------------------------
    # refinement of elite candidates
    best_seq = elite[0].copy()
    best_val = best_candidates[0][0]

    # allocate roughly 60 % of the remaining budget to deep refinement
    while time.time() < deadline - 30.0:
        # pick the current best as a starting point
        a0 = best_seq.copy()

        # ---- Adam phase ------------------------------------------------
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=250_000,
            lr=0.05, lam0=0.1, lam_factor=1.02
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Refine] Adam new best {val_adam:.6f}")

        # ---- Sub‑gradient phase ---------------------------------------
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000, eta0=0.85
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Refine] Subgrad new best {val_sub:.6f}")

        # ---- Random perturbation of logits (escape plateaus) ---------
        logits = np.log(best_seq + 1e-12)
        logits += np.random.randn(n) * 0.09
        pert = np.exp(logits - np.max(logits))
        pert = pert / pert.sum()
        a_tmp, val_tmp = _adam_optimize(
            pert, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=100_000, lr=0.04, lam0=0.2, lam_factor=1.03
        )
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, a_tmp.copy()
            print(f"[Perturb] Adam new best {val_tmp:.6f}")

        # ---- cheap local random hill‑climber -------------------------
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=60_000
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce evaluator’s range limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose for possible warm‑starts in a subsequent run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
