"""
Step‑function optimizer for the objective

    F(a) = 2·n·max(a∗a) / (∑a)²

The new version expands on the previous “Adam → sub‑gradient → anneal”
pipeline by:
* adding a few extra, analytically‑motivated initial shapes,
* using a *cyclic* refinement loop (sub‑gradient → annealing → sub‑gradient)
  that runs repeatedly while time permits,
* employing a slightly more aggressive step‑size schedule for the exponentiated‑gradient
  phase,
* allocating a larger share of the time‑budget to the fine‑grained refinement,
* finally up‑sampling to 8192 points (if time is left) and repeating the whole
  cycle on the finer grid.

All helper functions are defined at the top‑level and the routine always
returns the best sequence found before the deadline (the grader will
call ``evaluate_sequence`` itself).  The implementation is deliberately
deterministic apart from the random draws that seed the search.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the exact sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero; return uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (∑≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)              # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                               # slightly smaller for stability
    lam = 0.12                               # initial temperature
    lam_factor = 1.018

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (slowly tighten)
        lam = min(lam * lam_factor, 1e8)

        # occasional jitter close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size ηₜ = η₀ / (t+1)^α with α≈0.6 (a little slower decay).
    Returns best vector and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    alpha = 0.58  # exponent for step‑size decay

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / ((it + 1) ** alpha)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Moves consist of transferring a random fraction of mass from i→j.
    The temperature decays log‑linearly.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    # schedule: T_start tuned to current objective magnitude
    T_start = 0.12 * np.sqrt(val)          # heuristic scaling
    T_end = 5e-5
    it = 0

    while time.time() < deadline and it < max_moves:
        # linear decay on a log‑scale improves early exploration
        fraction = it / max_moves
        T = T_end + (T_start - T_end) * (1 - fraction) ** 1.1

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # random transfer amount (up to 45 % of source mass)
        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        # accept if better, otherwise with Metropolis probability
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create an eclectic pool of start vectors."""
    cand = []

    # warm start (if any) and a few cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with different concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # left‑right exponential families
    for gamma in (0.2, 0.5, 0.9, 1.5):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # monotone ramp (linear increase) – tends to decorrelate when reversed
    ramp = np.linspace(1, 0, n, dtype=np.float64)
    ramp = np.maximum(ramp, 0.0)
    ramp /= ramp.sum()
    cand.append(ramp)

    # quadratic ramp (convex shape)
    quad = (np.arange(n, dtype=np.float64) + 1) ** 0.5
    quad = quad[::-1]                     # descending
    quad /= quad.sum()
    cand.append(quad)

    # double‑sided cosine‑bump (broader than the pure sine)
    x = np.linspace(0, np.pi, n, dtype=np.float64)
    bump = (1 - np.cos(2 * x)) / 2       # smooth bump, non‑negative
    bump = np.maximum(bump, 0.0)
    bump /= bump.sum()
    cand.append(bump)

    # half‑wave sinusoid (already used but kept)
    half_sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    half_sin = np.maximum(half_sin, 0.0)
    half_sin /= half_sin.sum()
    cand.append(half_sin)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse patterns with a limited number of equally‑spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(4):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

        elif warm.shape[0] < n:

        a0 = a0 / a0.sum()

        a_adam, val_adam = _adam_optimize(

        if val_adam < best_val:

            best_seq = a_adam.copy()

    # ------------------------------------------------------------------
    # Phase B – intensive cyclic refinement on the primary grid.

    if best_seq is not None:

    while time.time() < deadline - 4.0:
