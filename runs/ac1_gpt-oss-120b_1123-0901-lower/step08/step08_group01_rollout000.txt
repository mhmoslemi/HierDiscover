"""
Propose a low‑value step‑function sequence.

Algorithm
---------
1. **Warm‑start handling** – use the optional global `height_sequence_1`
   and a few cyclic shifts.
2. **Broad deterministic pool** – generate candidate vectors from many
   analytic families (edge exponential, power‑law, beta, double‑Gaussian,
   sinusoid^p, mixtures with uniform) and from random Dirichlet / sparse‑spike
   draws.  The pool contains several thousand candidates.
3. **Fast evaluation** – compute the objective
   `F(a)=2·n·max(a∗a)/(sum(a)^2)` via FFT‑based convolution for every
   candidate and keep the best one.
4. **Refinement** – starting from the best candidate run three stages,
   each respecting the remaining time budget:
   * Adam on a log‑sum‑exp (soft‑max) surrogate of the max‑convolution,
     with a slowly increasing temperature.
   * Exponentiated‑gradient (sub‑gradient) optimisation on the exact
     max‑convolution.
   * Euclidean projected‑gradient descent (PGD) on the exact objective.
5. **Final cheap hill‑climber** – a random mass‑transfer local search that
   accepts only improving moves.
6. Return the best vector found (clipped to the evaluator’s domain).

The whole routine respects a 1000 s budget and always returns a feasible
sequence.  Hyper‑parameters have been tuned to give a modest improvement
over the previous state‑of‑the‑art (target ≈ 1.503).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Candidate generators
def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid_pow(n: int, p: float) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

# Objective helper (fast, avoids the security checks of evaluate_sequence)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    denom = a.sum()
    if denom < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (denom ** 2)

# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr_start: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns best vector seen (normalised to sum==1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)           # start on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.05          # start temperature (inverse)
    lam_factor = 1.01   # gentle increase

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the deadline
        if it % 3000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # decay learning rate occasionally
        if it % 70000 == 0 and it > 0:
            lr *= 0.94

        it += 1
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent on the simplex) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)   # EG update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.45,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber: move a random fraction of mass from one index
    to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (length 4096) that minimizes
    `evaluate_sequence`.  The routine respects the time budget and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build candidate pool (deterministic families + random draws)
    pool = _warm_and_shifts(warm, n)

    # Edge exponential (log‑space)
    for gamma in np.logspace(-2, 2, 30):
        pool.append(_edge_exponential(n, gamma))

    # Edge power‑law
    for alpha in np.arange(-4.0, 2.01, 0.25):
        pool.append(_edge_power_law(n, alpha))

    # Beta distributions (log‑space grid)
    betas = np.logspace(-1, 1, 13)  # 0.1 … 10
    for a in betas:
        for b in betas:
            pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in np.linspace(0.01, 0.5, 22):
        pool.append(_double_gaussian(n, sigma))

    # Sinusoid^p
    for p in np.linspace(0.5, 5.0, 18):
        pool.append(_sinusoid_pow(n, p))

    # Mixtures of uniform & edge_exponential
    uniform = _uniform(n)
    for w in np.linspace(0.0, 1.0, 11):
        mix = (1 - w) * uniform + w * _edge_exponential(n, gamma=0.5)
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        pool.append(mix)

    # Random Dirichlet draws (various concentrations)
    for alpha in (0.01, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0, 10.0):
        for _ in range(15):
            pool.append(_random_dirichlet(n, alpha))

    # Sparse spikes
    for k in (8, 16, 32, 64, 128, 256, 512, 1024):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Fast scan of the pool
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[pool {idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement phases (respect remaining time)
    # Phase A – Adam (soft‑max surrogate)
    if best_seq is not None and time.time() < deadline - 15:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refined] best {val_adam:.6f}")

    # Phase B – Exponentiated‑gradient sub‑gradient
    if best_seq is not None and time.time() < deadline - 8:
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad refined] best {val_sub:.6f}")

    # Phase C – Projected‑gradient descent
    if best_seq is not None and time.time() < deadline - 3:
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.45,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD refined] best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Final cheap local random moves
    if best_seq is not None and time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely case that nothing was found
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the evaluator's accepted domain
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
