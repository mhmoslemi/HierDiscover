def _lp_mix_optimize(a0, n, conv_fft_len, deadline, max_iters=5, t_vals=(0.2,0.5,0.8)):
    a = a0.copy()
    best_val = _objective(a)
    best_seq = a.copy()
    # precompute row and col indices
    row_idx = np.concatenate([np.arange(i, i + n) for i in range(n)])
    col_idx = np.concatenate([np.full(n, i, dtype=int) for i in range(n)])
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        M = _full_convolution(a, conv_fft_len).max()
        # Build constraint matrix
        data = np.repeat(a, n)
        A = scipy.sparse.csr_matrix((data, (row_idx, col_idx)), shape=(2*n-1, n))
        b = np.full(2*n-1, M, dtype=float)
        c = -np.ones(n, dtype=float)
        # solve LP with time limit small
        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n,
                                    method='highs-ds',

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The search pipeline (≈1000 s budget) combines:

1. Warm‑start / handcrafted starters.
2. Massive random Dirichlet / “anti‑pair’’ candidates.
3. Spike‑type candidates (few equally‑weighted non‑zero entries).
4. Smooth‑max surrogate optimisation with L‑BFGS (increasing λ).
5. Exact‑max sub‑gradient descent (multiplicative & projected‑gradient).
6. Linear‑programming refinement: maximise total mass of a
   partner g under the current convolution bound, then mix.
7. Local stochastic hill‑climbing (simulated annealing).

All heavy work respects the global time limit.  The best sequence found
is returned as a plain list of floats.
"""

import time
import numpy as np
import scipy.optimize as opt
import scipy.sparse as sp

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (fast)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Candidate generators ----------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _spike_candidate(n: int, k: int) -> np.ndarray:
    """k equally weighted non‑zero entries placed at random positions."""
    idx = np.random.choice(n, k, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[idx] = 1.0 / k
    return a

# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp (soft‑max surrogate)
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)    # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0, 800.0),
                    max_iter_stage: int = 300) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val

# Multiplicative (exponentiated‑gradient) sub‑gradient optimisation
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Additive projected sub‑gradient optimisation
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                    # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1
    return best_a, best_val

# Linear‑programming refinement (mixing step)
def _lp_mix_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                     deadline: float,
                     max_iters: int = 8,
                     t_vals: tuple[float, ...] = (0.2, 0.5, 0.8)) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve:
        maximise Σ g_j   s.t.  a * g ≤ M   (entry‑wise)
    and mix the found partner into a.
    """
    a = a0.copy()
    best_val = _objective(a)
    best_seq = a.copy()

    # Fixed index pattern for the Toeplitz‐type convolution matrix
    row_idx = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    col_idx = np.repeat(np.arange(n), n)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # current bound M
        M = _full_convolution(a, conv_fft_len).max()

        # Build sparse matrix for constraints a * g ≤ M
        data = np.repeat(a, n)
        A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))

        b = np.full(2 * n - 1, M, dtype=np.float64)
        c = -np.ones(n, dtype=np.float64)          # maximise Σ g  ⇔ minimise −Σ g
        bounds = [(0, None)] * n

        remaining = deadline - time.time()
        if remaining < 1.0:
            break
        lp_time = min(30.0, remaining)

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                          method='highs',
                          options={'presolve': True, 'time_limit': lp_time})

        if not res.success:
            print("[LP] linprog failed or timed out")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 < 1e-12:
            break

        # Rescale to keep total mass equal to that of a (≈1)
        scale = 1.0 / sum_g0
        g = g0 * scale

        improved = False
        for t in t_vals:
            cand = (1.0 - t) * a + t * g
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                a = cand
                improved = True
                print(f"[LP it={it} t={t:.2f}] new best {val:.6f}")
                break

        # If none of the preset t gave improvement, try a very small step
        if not improved:
            t = 0.05
            cand = (1.0 - t) * a + t * g
            val = _objective(cand)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                a = cand
                print(f"[LP it={it} small t] new best {val:.6f}")
                improved = True

        if not improved:
            # No improvement – stop early
            break

    return best_seq, best_val

# Main entry point --------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 2048                         # a comfortable size for LP + FFT work
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Warm‑start / handcrafted starters
    best_seq = None
    best_val = float("inf")

    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
        _anti_pair_candidate(n, 1.0),
    ]
    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # --------------------------------------------------------------
    # 2) Massive random anti‑pair Dirichlet pool
    n_samples = 2500
    print(f"Generating {n_samples} random anti‑pair candidates …")
    for _ in range(n_samples):
        log_alpha = np.random.uniform(-2.0, 1.5)   # α ∈ [0.01, ≈31.6]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random pool] new best {val:.6f}")

    # --------------------------------------------------------------
    # 3) Spike‑type candidates (few equal masses)
    for k in (4, 8, 16, 32, 64, 128):
        for _ in range(150):
            cand = _spike_candidate(n, k)
            val = _objective(cand)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Spike k={k}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 4) Smooth‑max L‑BFGS refinement
    if time.time() < deadline - 200:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n, conv_fft_len,
                                         corr_fft_len, deadline - 5,
                                         lam_schedule=(5, 15, 40, 120, 300, 800),
                                         max_iter_stage=300)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 5) Exact‑max sub‑gradient phases
    # (a) multiplicative (exponentiated‑gradient)
    if time.time() < deadline - 150:
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 deadline - 5,
                                                 max_iters=1_200_000,
                                                 eta0=0.7)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[MulSubgrad] improved to {val_sub:.6f}")

    # (b) additive projected sub‑gradient
    if time.time() < deadline - 80:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          deadline - 5,
                                                          max_iters=800_000,
                                                          eta0=0.5)
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # 6) Linear‑programming mixing refinement
    if time.time() < deadline - 30:
        seq_lp, val_lp = _lp_mix_optimize(best_seq, n, conv_fft_len,
                                          deadline - 5,
                                          max_iters=8,
                                          t_vals=(0.2, 0.5, 0.8))
        if val_lp < best_val:
            best_val, best_seq = val_lp, seq_lp.copy()
            print(f"[LP mix] improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # 7) Final local stochastic hill‑climbing
    if time.time() < deadline - 10:
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len,
                                                deadline - 5,
                                                max_moves=200_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # --------------------------------------------------------------
    # 8) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
