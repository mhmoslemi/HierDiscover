import time
import numpy as np

def _next_pow2(x: int) -> int:
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=300_000,
                   lr=0.06,
                   lam0=0.02,
                   lam_factor=1.07):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    lam = lam0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # compute smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # small random shake near deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle decay
        if it % 100000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                          max_iters=300_000,
                          eta0=0.8):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a, best_val = a.copy(), float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 25000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _local_random_search(a0, n, conv_fft_len, deadline,
                         max_moves=80_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.25)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
            if it % 10000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def _upscale(seq_small, factor):
    # repeat each entry factor times, then renormalize to sum 1
    seq_big = np.repeat(seq_small, factor)
    total = seq_big.sum()
    if total < 1e-12:
        seq_big = np.full_like(seq_big, 1.0 / seq_big.size)
    else:
        seq_big = seq_big / total
    return seq_big

def _initial_candidates_general(n, warm):
    cand = []
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))
    # random Dirichlets
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))
    # edge-exponential
    for gamma in (0.1,0.2,0.5,1.0,2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)
    # edge-polynomial
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)
    # double-Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05,0.1,0.2,0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)
    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)
    # uniform
    cand.append(np.full(n, 1.0 / n))
    # spikes
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)
    return cand

    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2*len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.06,
    lam0: float = 0.02,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    lam = lam0

    while time.time() < deadline and it < max_iters:
        # simplex projection via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake when we are close to the deadline
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # very gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 25_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of the mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move up to 25 % of the source mass
        delta = a[i] * (np.random.rand() * 0.25)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates_general(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Diverse pool of start vectors for a given size."""
    cand = []

    # warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet draws with several concentrations
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # edge‑exponential families (mass near the borders)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double‑Gaussian bumps at both ends
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform distribution
    cand.append(np.full(n, 1.0 / n))

    # sparse spike patterns (equally spaced)
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        cand.append(a)

    return cand


def _upscale(seq_small: np.ndarray, factor: int) -> np.ndarray:
    """Repeat each entry `factor` times and renormalise to sum 1."""
    seq_big = np.repeat(seq_small, factor)
    total = seq_big.sum()
    if total < 1e-12:
        seq_big = np.full_like(seq_big, 1.0 / seq_big.size)
    else:
        seq_big = seq_big / total
    return seq_big

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine works on a reduced problem size for most of the budget,
    then up‑scales the best result to the full size and performs a short
    final polish.  The best feasible sequence found so far is always kept
    and finally returned as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem sizes
    n_big = 4096
    factor = 4                     # 4096 / 4 = 1024
    n_small = n_big // factor

    # ------------------------------------------------------------------
    # warm start (if a previous solution exists)
    warm_global = globals().get("height_sequence_1")
    warm_small = None
    if warm_global is not None:
        warm = np.asarray(warm_global, dtype=np.float64)
        if warm.shape[0] > n_small:
            warm_small = warm[:n_small]
        else:
            warm_small = np.pad(warm, (0, n_small - warm.shape[0]))
        warm_small = np.maximum(warm_small, 0.0)
        if warm_small.sum() >= 0.01:
            warm_small = warm_small / warm_small.sum()
        else:
            warm_small = None

    # ------------------------------------------------------------------
    # build a modestly diverse pool for the small problem
    pool = _initial_candidates_general(n_small, warm_small)

    # FFT lengths for the small problem
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    best_small_seq = None
    best_small_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – intensive optimisation on the reduced size
    small_phase_deadline = time.time() + (deadline - time.time()) * 0.60

    # approximate budget per candidate (used only for a simple early‑stop)
    per_cand_budget = (small_phase_deadline - time.time()) * 0.25 / max(1, len(pool))

    for cid, cand in enumerate(pool):
        if time.time() >= small_phase_deadline:
            break

        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n_small))
        a0 = a0 / a0.sum()

        # --- Adam optimisation (smooth‑max) ---------------------------------
        local_deadline = min(small_phase_deadline - 5.0,
                             time.time() + per_cand_budget)
        a_adam, val_adam = _adam_optimize(
            a0,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            local_deadline,
            max_iters=250_000,
            lr=0.07,
            lam0=0.01,
            lam_factor=1.06,
        )
        if val_adam < best_small_val:
            best_small_val, best_small_seq = val_adam, a_adam.copy()
            print(f"[small cand {cid}] Adam improves to {val_adam:.6f}")

        # --- cheap sub‑gradient refinement -----------------------------------
        sub_deadline = min(small_phase_deadline - 5.0,
                           time.time() + max(0.5, per_cand_budget * 0.20))
        a_sub, val_sub = _subgradient_optimize(
            best_small_seq,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            sub_deadline,
            max_iters=30_000,
            eta0=0.9,
        )
        if val_sub < best_small_val:
            best_small_val, best_small_seq = val_sub, a_sub.copy()
            print(f"[small cand {cid}] Subgrad improves to {val_sub:.6f}")

        # --- very cheap local random search ----------------------------------
        a_loc, val_loc = _local_random_search(
            best_small_seq,
            n_small,
            conv_fft_len_small,
            sub_deadline,
            max_moves=30_000,
        )
        if val_loc < best_small_val:
            best_small_val, best_small_seq = val_loc, a_loc.copy()
            print(f"[small cand {cid}] Local improves to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing was discovered (should never happen)
    if best_small_seq is None:
        best_small_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)

    # ------------------------------------------------------------------
    # Upscale to the full size and polish there
    best_big_seq = _upscale(best_small_seq, factor)

    # FFT lengths for the full problem
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    best_big_val = float("inf")

    # Phase 2 – final refinement on the full size (uses the remaining budget)
    while time.time() < deadline - 10.0:
        # ---- short Adam run -------------------------------------------------
        sub_deadline = min(deadline - 5.0, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_big_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            max_iters=100_000,
            lr=0.05,
            lam0=0.02,
            lam_factor=1.08,
        )
        if val_adam < best_big_val:
            best_big_val, best_big_seq = val_adam, a_adam.copy()
            print(f"[big] Adam refined best {val_adam:.6f}")

        # ---- exact‑max sub‑gradient -----------------------------------------
        a_sub, val_sub = _subgradient_optimize(
            best_big_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
            eta0=0.8,
        )
        if val_sub < best_big_val:
            best_big_val, best_big_seq = val_sub, a_sub.copy()
            print(f"[big] Subgrad refined best {val_sub:.6f}")

        # ---- cheap local random moves ----------------------------------------
        a_loc, val_loc = _local_random_search(
            best_big_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            max_moves=40_000,
        )
        if val_loc < best_big_val:
            best_big_val, best_big_seq = val_loc, a_loc.copy()
            print(f"[big] Local improved best {val_loc:.6f}")

        # ---- random perturbation + short Adam -------------------------------
        logits = np.log(best_big_seq + 1e-12)
        logits += np.random.randn(n_big) * 0.07
        pert = np.exp(logits - np.max(logits))
        pert = pert / pert.sum()
        a_pert, val_pert = _adam_optimize(
            pert,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=60_000,
            lr=0.04,
            lam0=0.015,
            lam_factor=1.07,
        )
        if val_pert < best_big_val:
            best_big_val, best_big_seq = val_pert, a_pert.copy()
            print(f"[big] Perturb Adam best {val_pert:.6f}")

        # stop early if we are within a few seconds of the deadline
        if time.time() > deadline - 15.0:
            break

    # ------------------------------------------------------------------
    # Final safety net – ensure a valid non‑empty return
    if best_big_seq is None:
        best_big_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_big_seq = np.clip(best_big_seq, 0.0, 1000.0)

    # Final logging (the evaluator may already have been called inside the loops)
    try:
        final_val = evaluate_sequence(best_big_seq.tolist())
    except Exception:
        final_val = best_big_val
    print(f"Final best value {final_val:.6f} (length {len(best_big_seq)})")
    return best_big_seq.tolist()
