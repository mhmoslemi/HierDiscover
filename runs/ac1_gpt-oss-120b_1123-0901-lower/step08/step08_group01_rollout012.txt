```
def adam_optimize(a0, n, conv_len, corr_len, deadline, max_iters=200_000, lr_start=0.04):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = lr_start
    lam = 0.01
    lam_factor = 1.02
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # smooth max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e8)
        if it % 50000 == 0 and it > 0:
            lr *= 0.9
        it += 1
    return best_a, best_val
```

"""
Hybrid multi‑start gradient‑descent + simulated‑annealing search.

The algorithm builds a diverse pool of simplex vectors (uniform,
edge‑exponential, power‑law, beta, double‑Gaussian, sinusoid, Dirichlet
draws and random mixtures).  The best few seeds are refined by a
sequence of Adam (log‑sum‑exp surrogate), exponentiated‑gradient
sub‑gradient and Euclidean projected‑gradient phases.  Afterwards a
perturb‑and‑re‑optimise loop and a short simulated‑annealing hill‑climber
use the remaining time budget.  The returned vector is non‑negative,
normalised (scale‑invariant) and clipped to the range accepted by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# ---------- INITIAL CANDIDATE GENERATORS --------------------------------
def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (i+1)^α + (n-i)^α."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# ---------- OBJECTIVE ----------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact evaluation of F(a) = 2·n·max(a*a) / (sum a)^2."""
    conv = _full_convolution(a, conv_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# ---------- OPTIMISERS --------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best simplex vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # initial logits for soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # inverse temperature for the soft‑max
    lam_factor = 1.02   # growth of temperature

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- simplex from logits -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective (track best) -----
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # ----- chain rule through the soft‑max (mirror‑descent form) -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        # ----- occasional learning‑rate decay -----
        if it % 50000 == 0 and it > 0:
            lr *= 0.93

        # ----- tiny random shake if we are close to the deadline -----
        if (it % 5000 == 0) and (time.time() > deadline - 30):
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # decreasing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set‑based sub‑gradient (same as before)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # step‑size (decreasing)
        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# ---------- LOCAL SEARCH / PERTURBATION ---------------------------------
def _perturb_and_refine(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    sigma: float,
) -> tuple[np.ndarray, float]:
    """
    Apply multiplicative Gaussian noise to a0, project to simplex,
    then perform a short Adam refinement (few thousand iterations).
    """
    a = a0 * (1.0 + sigma * np.random.randn(n))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12 or np.isnan(total):
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / total

    # run a short Adam optimisation (fixed budget ~2 s or 15000 iters)
    sub_deadline = min(deadline, time.time() + min(2.0, (deadline - time.time()) * 0.2))
    a_opt, val_opt = _adam_optimize(
        a,
        n,
        conv_len,
        corr_len,
        sub_deadline,
        max_iters=30_000,
        lr_start=0.06,
    )
    return a_opt, val_opt


def _annealed_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_move_frac: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing‑style cheap hill climbing.
    """
    a = a0.copy()
    cur_val = _objective(a, n, conv_len)
    best_a = a.copy()
    best_val = cur_val

    # linear temperature schedule
    T_start = 5e-4
    T_end = 1e-8
    total_steps = int((deadline - time.time()) * 500)  # roughly 500 moves per second
    if total_steps <= 0:
        return best_a, best_val

    for step in range(total_steps):
        if time.time() >= deadline:
            break
        T = T_start * (1 - step / total_steps) + T_end * (step / total_steps)

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * max_move_frac)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_len)

        # Metropolis acceptance
        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / max(T, 1e-12)):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if step % 5000 == 0:
                    print(f"[Anneal step={step}] new best {val2:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial vectors
    pool = _warm_and_shifts(warm, n)

    # deterministic families
    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))
    for alpha in (-5.0, -3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0), (0.3, 0.7), (0.7, 0.3)]:
        pool.append(_beta_shape(n, a, b))
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # random Dirichlet draws
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        for _ in range(30):
            pool.append(_random_dirichlet(n, alpha, rng))

    # random mixtures of two existing candidates (helps create hybrid shapes)
    pool_len_before = len(pool)
    for _ in range(200):
        i1 = rng.integers(pool_len_before)
        i2 = rng.integers(pool_len_before)
        lam = rng.random()
        mix = lam * pool[i1] + (1 - lam) * pool[i2]
        pool.append(mix)

    # ------------------------------------------------------------------
    # Phase 1 – quick evaluation of the pool, keep the best few
    top_k = 30
    best_seeds = []  # list of (value, vector)
    phase1_deadline = min(deadline - 10.0, time.time() + 0.20 * (deadline - time.time()))
    for idx, cand in enumerate(pool):
        if time.time() >= phase1_deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_len)

        if len(best_seeds) < top_k:
            best_seeds.append((val, a.copy()))
            best_seeds.sort(key=lambda x: x[0])
        else:
            if val < best_seeds[-1][0]:
                best_seeds[-1] = (val, a.copy())
                best_seeds.sort(key=lambda x: x[0])

        if idx % 1000 == 0:
            print(f"[Initial scan] processed {idx} candidates, current best {best_seeds[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep refinement of the top candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # allocate roughly half of the remaining budget to this phase
    refinement_deadline = deadline - 8.0
    for rank, (seed_val, seed_vec) in enumerate(best_seeds):
        if time.time() >= refinement_deadline:
            break
        # Adam phase (smooth‑max)
        sub_deadline = min(refinement_deadline, time.time() + 0.12 * (deadline - time.time()))
        a_adam, val_adam = _adam_optimize(
            seed_vec,
            n,
            conv_len,
            corr_len,
            sub_deadline,
            max_iters=180_000,
            lr_start=0.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine #{rank}] Adam improved to {val_adam:.6f}")

        # Sub‑gradient (exponentiated‑gradient) phase
        if time.time() < refinement_deadline - 4:
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_len,

"""
Hybrid multi‑start gradient‑descent + simulated‑annealing search.

The algorithm builds a diverse pool of simplex vectors (uniform,
edge‑exponential, power‑law, beta, double‑Gaussian, sinusoid, Dirichlet
draws and random mixtures).  The best few seeds are refined by a
sequence of Adam (log‑sum‑exp surrogate), exponentiated‑gradient
sub‑gradient and Euclidean projected‑gradient phases.  Afterwards a
perturb‑and‑re‑optimise loop and a short simulated‑annealing hill‑climber
use the remaining time budget.  The returned vector is non‑negative,
normalised (scale‑invariant) and clipped to the range accepted by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# ---------- INITIAL CANDIDATE GENERATORS --------------------------------
def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _uniform(n: int) -> np.ndarray:
    """Uniform distribution on the simplex."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# ---------- OBJECTIVE ----------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact evaluation of F(a) = 2·n·max(a*a) / (sum a)^2."""
    conv = _full_convolution(a, conv_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (sum_a ** 2))


# ----------------------------------------------------------------------
# ---------- OPTIMISERS --------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best simplex vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02                        # temperature (inverse)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.93

        if (it % 5000 == 0) and (time.time() > deadline - 30):
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient phase on the exact max.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# ---------- LOCAL SEARCH / PERTURBATION ---------------------------------
def _perturb_and_refine(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    sigma: float,
) -> tuple[np.ndarray, float]:
    """
    Apply multiplicative Gaussian noise, project to the simplex,
    then perform a short Adam refinement.
    """
    a = a0 * (1.0 + sigma * np.random.randn(n))
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total < 1e-12 or np.isnan(total):
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / total

    sub_deadline = min(deadline, time.time() + min(2.0, (deadline - time.time()) * 0.2))
    a_opt, val_opt = _adam_optimize(
        a,
        n,
        conv_len,
        corr_len,
        sub_deadline,
        max_iters=30_000,
        lr_start=0.06,
    )
    return a_opt, val_opt


def _annealed_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_move_frac: float = 0.2,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing‑style cheap hill climbing.
    """
    a = a0.copy()
    cur_val = _objective(a, n, conv_len)
    best_a = a.copy()
    best_val = cur_val

    T_start = 5e-4
    T_end = 1e-8
    total_steps = int((deadline - time.time()) * 500)
    if total_steps <= 0:
        return best_a, best_val

    for step in range(total_steps):
        if time.time() >= deadline:
            break
        T = T_start * (1 - step / total_steps) + T_end * (step / total_steps)

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * max_move_frac)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_len)

        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / max(T, 1e-12)):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if step % 5000 == 0:
                    print(f"[Anneal step={step}] new best {val2:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """

    rng = np.random.default_rng(seed)
    start_time = time.time()

            warm = warm[:n]

        warm = np.maximum(warm, 0.0)
