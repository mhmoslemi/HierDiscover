def propose_candidate(seed: int = 0, budget_s: float = 1000):
    """
    Generate a non‑negative step function minimizing 2·n·max(a*a)/(sum a)^2.
    Combines a dense hyper‑parameter sweep of simple shapes, a short
    exponentiated‑gradient refinement and a long targeted simulated‑annealing
    phase that updates the convolution incrementally.
    """
    import numpy as np, time, math, random
    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5

    # problem size
    n = 4096
    fft_len = 1 << ((2*n-2).bit_length())
    
    # utility conv
    def full_conv(a):
        # using FFT
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)
        return conv[: 2*n-1]

    def max_conv_val(a):
        conv = full_conv(a)
        return conv.max()
    
    # candidate pool
    candidates = []
    # warm-start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        arr = np.array(warm, dtype=np.float64)
        if arr.shape[0] > n: arr = arr[:n]
        elif arr.shape[0] < n: arr = np.pad(arr, (0, n-arr.shape[0]))
        arr = np.clip(arr, 0.0, None)
        s = arr.sum()
        if s >= 0.01:
            arr = arr / s
            candidates.append(arr)
    # uniform
    candidates.append(np.full(n, 1.0/n))
    # triangular
    candidates.append(_triangle_candidate(n))
    # Gaussian edges sweep
    for gamma in np.logspace(-2, 0, 15): # 0.01 to 1
        candidates.append(_gaussian_edges_candidate(n, gamma))
    # Power law edges sweep
    for alpha in np.linspace(0.5, 5.0, 15):
        i = np.arange(n, dtype=np.float64)
        shape = (i+1) ** (-alpha) + (n-i) ** (-alpha)
        shape /= shape.sum()
        candidates.append(shape)
    # Dirichlet random + anti-pair and alternating arrangement
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            vals = np.random.dirichlet(alpha * np.ones(n))
            # anti-pair arrangement
            candidates.append(_anti_pair_arrangement(vals))
            # alternating arrangement
            candidates.append(_alternating_arrangement(vals))
    # random Dirichlet plain
    for alpha in (0.3, 0.7, 1.0, 2.0):
        for _ in range(3):
            candidates.append(np.random.dirichlet(alpha*np.ones(n)))
    
    # evaluate all candidates (exact objective)
    best_seq = None
    best_val = math.inf
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            total = cand.sum()
        cand = cand / total
        t = max_conv_val(cand)  # max conv
        val = 2.0 * n * t
        # keep best
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if time.time() > deadline - 30:
            break  # keep some time for refinement
        # optional print
        print(f"[Init {idx}] val={val:.6f}")
    
    # Phase 2: sub‑gradient improvement (exponentiated gradient)
    # limit to remaining time (maybe 200 sec)
    sub_deadline = min(deadline - 20, time.time() + 200)
    if time.time() < sub_deadline:
        # We'll reuse the subgradient code but adapt iteration count to time
        a = best_seq.copy()
        n = a.shape[0]
        fft_len = _next_pow2(2*n-1)
        corr_fft_len = _next_pow2(3*n-2)
        best_a = a.copy()
        best_obj = best_val
        it = 0
        while time.time() < sub_deadline:
            conv = _full_convolution(a, fft_len)
            max_c = conv.max()
            obj = 2.0 * n * max_c
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[Subgrad it={it}] improved to {obj:.6f}")
            # locate active offsets where conv approx max
            active = np.isclose(conv, max_c, atol=1e-12)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
                grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
                # exponentiated gradient update
                eta = 0.7 / math.sqrt(it + 1)
                a = a * np.exp(-eta * grad)
                a_sum = a.sum()
                if a_sum < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    a_sum = a.sum()
                a = a / a_sum
            else:
                # no active? rarely happen
                a = np.random.dirichlet(np.ones(n))
            it += 1
        # after loop, update best_seq
        if best_obj < best_val:
            best_val = best_obj
            best_seq = best_a.copy()
            print("[Subgrad] final best", best_val)
    
    # Phase 3: refined SA with proper objective scaling
    # define SA function inside propose_candidate (top-level) maybe.
    # Use time limit
    if time.time() < deadline - 5:
        # define functions (maybe as locals)
        a = best_seq.copy()
        n = a.shape[0]
        conv = full_conv(a)  # length 2n-1
        cur_val = 2.0 * n * conv.max()
        best_val_sa = cur_val
        best_a_sa = a.copy()
        start_sa = time.time()
        total_sa_time = deadline - time.time() - 5
        # schedule parameters
        start_temp = 0.02
        min_temp = 0.001
        max_frac = 0.4
        # main loop
        it = 0
        while time.time() < deadline - 5:
            # choose source index with >0 mass
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            # choose target index far from i (optional)
            # use uniform else ensure not equal
            j = np.random.randint(n-1)
            if j >= i:
                j += 1
            # fraction
            frac = np.random.rand() * max_frac
            delta = a[i] * frac
            if delta <= 0.0:

            _apply_delta(a, conv, i, -delta)
            _apply_delta(a, conv, j, delta)
            new_max = conv.max()
            new_val = 2.0 * n * new_max
            # linear temperature schedule
            elapsed = time.time() - start_sa
            remaining = max(0.0, total_sa_time - elapsed)
            temp = start_temp * (remaining / total_sa_time)
            temp = max(temp, min_temp)
            accept = False
            if new_val <= cur_val:
                accept = True
            else:
                # Metropolis acceptance
                if np.random.rand() < np.exp(-(new_val - cur_val) / temp):
                    accept = True
            if accept:
                cur_val = new_val
                if new_val < best_val_sa:
                    best_val_sa = new_val
                    best_a_sa = a.copy()
                # adapt max_frac: slowly decay as we improve
                if it % 10000 == 0 and max_frac > 0.02:
                    max_frac *= 0.98
            else:
                # revert move
                _apply_delta(a, conv, j, -delta)
                _apply_delta(a, conv, i, delta)
            it += 1
            if it % 50000 == 0:
                print(f"[SA it={it}] best obj={best_val_sa:.6f} cur obj={cur_val:.6f}")
        # after SA
        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print("[SA] improved final to", best_val)

    # final scaling and clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq /= total
    print(f"Final best objective {best_val:.6f} (t ≈ {best_val/(2*n):.8f})")
    return best_seq.tolist()

"""
A three‑phase optimizer for the objective

    F(a) = 2·n·max(a*a) / (sum a)^2 ,

where *a* is a non‑negative step‑function (list of floats) and
max(a*a) is the maximum entry of the linear convolution a*a.
The algorithm is:

1.  **Rich warm‑start pool** – many simple parametric shapes
    (uniform, triangular, Gaussian edges with a dense sweep of the
    scale parameter, power‑law edges, anti‑pair and alternating
    arrangements of Dirichlet samples, plus the previous best if
    available) are generated and the one with the smallest exact
    objective is kept.

2.  **Exponential‑gradient refinement** – a short sub‑gradient
    descent on the true L∞‑objective (no smoothing) using the
    exact convolution.  This quickly drives the solution into a
    basin of attraction of a good local minimum.

3.  **Targeted simulated‑annealing** – an incremental O(n) update
    of the full convolution allows millions of tiny moves.
    Moves transfer a random fraction of the mass from an index that
    contributes heavily to the current peak to a random other index.
    A linear temperature schedule together with a slowly decaying
    move‑size (max fraction) yields a monotone descent in the
    exact objective.  The best sequence found is returned.

The whole search respects the supplied time budget (default 1000 s)
and guarantees a non‑negative, normalised output that can be fed
directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT / convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Incremental update of the convolution (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the
    full linear convolution a*a in O(n) via slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (the centre of the slice)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


# ----------------------------------------------------------------------
# Simple parametric candidates
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – larger at the ends, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two mirrored Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Large values on the left half, small values on the right half."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _alternating_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place the sorted values alternately at the two ends (largest ↔ left, second ↔ right)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    lo, hi = 0, n - 1
    p = 0
    while lo <= hi:
        a[lo] = sorted_vals[p]
        p += 1
        if lo == hi:
            break
        a[hi] = sorted_vals[p]
        p += 1
        lo += 1
        hi -= 1
    return a


def _power_law_edges_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Edges dominate via a power‑law decay."""
    i = np.arange(n, dtype=np.float64)
    shape = (i + 1) ** (-alpha) + (n - i) ** (-alpha)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a low‑value step function within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    fft_len = _next_pow2(2 * n - 1)          # for full convolutions
    corr_fft_len = _next_pow2(3 * n - 2)     # for cross‑correlations

    # ------------------------------------------------------------------
    # helper to evaluate the *exact* objective (no safety checks)
    def _objective_val(a: np.ndarray) -> float:
        conv = _full_convolution(a, fft_len)
        t = conv.max()                # max of the convolution
        return 2.0 * n * t            # exactly the value returned by evaluate_sequence

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of warm‑starts
    candidates = []

    # previous best (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        s = w.sum()
        if s >= 0.01:
            candidates.append(w / s)

    # simple parametric families
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))

    # Gaussian edges – sweep a wide range of scales
    for gamma in np.logspace(-2, 0, 20):   # 0.01 … 1.0 (log‑spaced)
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # Power‑law edges – sweep exponents
    for alpha in np.linspace(0.5, 5.0, 20):
        candidates.append(_power_law_edges_candidate(n, alpha))

    # Dirichlet based arrangements (anti‑pair and alternating)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        for _ in range(3):
            vals = np.random.dirichlet(alpha * np.ones(n))
            candidates.append(_anti_pair_arrangement(vals))
            candidates.append(_alternating_arrangement(vals))

    # plain Dirichlet draws (no re‑ordering)
    for alpha in (0.3, 0.7, 1.0, 2.0):
        for _ in range(3):
            candidates.append(np.random.dirichlet(alpha * np.ones(n)))

    # ------------------------------------------------------------------
    # 2) Pick the best starter (exact evaluation)
    best_seq = None
    best_val = math.inf
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        total = cand.sum()
        if total < 0.01:                     # safeguard against degenerate draws
            cand = np.random.dirichlet(np.ones(n))
            total = cand.sum()
        cand = cand / total
        val = _objective_val(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if time.time() > deadline - 30:      # keep ~30 s for refinement
            break
        print(f"[Init #{idx:03d}] obj={val:.6f}")

    # ------------------------------------------------------------------
    # 3) Short sub‑gradient / exponentiated‑gradient refinement
    sub_deadline = min(deadline - 20, time.time() + 200)   # ≤200 s
    if time.time() < sub_deadline:
        a = best_seq.copy()
        best_a = a.copy()
        best_obj = best_val
        it = 0
        while time.time() < sub_deadline:
            conv = _full_convolution(a, fft_len)
            max_c = conv.max()
            obj = 2.0 * n * max_c
            if obj < best_obj:
                best_obj = obj
                best_a = a.copy()
                print(f"[Subgrad it={it:06d}] new best {obj:.6f}")

            # active set (indices where conv equals the current max)
            active = np.isclose(conv, max_c, atol=1e-12)
            if not active.any():
                # should not happen, but fall back to a random restart
                a = np.random.dirichlet(np.ones(n))
                continue

            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # exponentiated‑gradient update (step size decays as 1/√t)
            eta = 0.7 / math.sqrt(it + 1)
            a = a * np.exp(-eta * grad)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s
            it += 1

        if best_obj < best_val:
            best_val = best_obj
            best_seq = best_a.copy()
            print("[Subgrad] final best =", best_val)

    # ------------------------------------------------------------------
    # 4) Long targeted simulated‑annealing search
    if time.time() < deadline - 5:
        a = best_seq.copy()
        conv = _full_convolution(a, fft_len)                # current full convolution
        cur_obj = 2.0 * n * conv.max()
        best_obj_sa = cur_obj
        best_a_sa = a.copy()

        start_sa = time.time()
        total_sa = deadline - time.time() - 5               # time left for SA
        start_temp = 0.02
        min_temp = 0.001
        max_frac = 0.4

        it = 0
        while time.time() < deadline - 5:
            # choose a source index with some mass
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue

            # choose a destination index (uniform, but never equal to i)
            j = np.random.randint(n - 1)
            if j >= i:
                j += 1

            # transfer a random fraction of the source mass
            frac = np.random.rand() * max_frac
            delta = a[i] * frac
            if delta <= 0.0:
                continue

            # apply the move incrementally
            _apply_delta(a, conv, i, -delta)
            _apply_delta(a, conv, j, delta)

            new_obj = 2.0 * n * conv.max()
            # linear temperature schedule (remaining fraction of SA time)
            elapsed = time.time() - start_sa
            remaining = max(0.0, total_sa - elapsed)
            temp = start_temp * (remaining / total_sa)
            temp = max(temp, min_temp)

            accept = False
            if new_obj <= cur_obj:
                accept = True
            else:
                if np.random.rand() < math.exp(-(new_obj - cur_obj) / temp):
                    accept = True

            if accept:
                cur_obj = new_obj
                if new_obj < best_obj_sa:
                    best_obj_sa = new_obj
                    best_a_sa = a.copy()
                # decay the move size slowly as we progress
                if it % 10000 == 0 and max_frac > 0.02:
                    max_frac *= 0.98
            else:
                # revert the move
                _apply_delta(a, conv, j, -delta)
                _apply_delta(a, conv, i, delta)

            it += 1
            if it % 50000 == 0:
                print(f"[SA it={it:06d}] best obj={best_obj_sa:.6f}  cur obj={cur_obj:.6f}")

        if best_obj_sa < best_val:
            best_val = best_obj_sa
            best_seq = best_a_sa.copy()
            print("[SA] improved final objective =", best_val)

    # ------------------------------------------------------------------
    # 5) Final safety checks and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / total

    print(f"Final best objective {best_val:.6f}  (t ≈ {best_val/(2*n):.9f})")
    return best_seq.tolist()
