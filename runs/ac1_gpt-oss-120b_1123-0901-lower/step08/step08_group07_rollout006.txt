    a·exp(∑c_k cos(2πk·i/(n‑1)))).
  The best few (by the exact objective) are kept.

* **Two‑phase continuous optimisation** – for each retained candidate
  1) an Adam optimiser on a smooth‑max surrogate (log‑sum‑exp) and
  2) an exact‑max sub‑gradient phase (exponentiated‑gradient).  
  Both phases respect the simplex constraints (non‑negative, sum=1).

* **Intensive stochastic hill‑climbing** – a simulated‑annealing style
  local search that swaps entries or transfers a random fraction of mass.
  The temperature decays linearly to zero.

* **Time‑aware budgeting** – the total 1000 s budget is split dynamically
  among the stages; the code never exceeds the deadline.

The whole pipeline runs in < 1000 s on a single core and returns the
best sequence found (as a plain Python list of floats).  All numbers are
clipped to the interval [0, 1000] to satisfy the safety checks of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without safety checks) -------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helper ------------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Deterministic / random generators --------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p).  p>0 concentrates mass on the first entries."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    """
    Valley in the centre, larger at the two ends:
    a_i ∝ ( i·(n‑1‑i) + 1 )^{-p}.
    """
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _random_fft_candidate(n: int) -> np.ndarray:
    """
    Generate a random vector by sampling a random complex spectrum,
    enforcing Hermitian symmetry (so the inverse FFT is real),
    clipping negatives, and normalising.
    """
    # length needed for a clean linear convolution later
    fft_len = _next_pow2(2 * n - 1)

    # Random complex spectrum with Hermitian symmetry
    half = fft_len // 2 + 1
    rand_spec = np.random.randn(half) + 1j * np.random.randn(half)
    rand_spec[0] = np.random.rand()  # DC must be real
    if fft_len % 2 == 0:
        rand_spec[-1] = np.random.rand()  # Nyquist term must be real

    full_spec = np.concatenate([rand_spec, np.conj(rand_spec[-2:0:-1])])
    a = np.fft.irfft(full_spec, fft_len)[:n]
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()
    return a


def _cosine_param_candidate(n: int, K: int = 6) -> np.ndarray:
    """
    Low‑frequency cosine parametrisation:
        a_i ∝ exp( Σ_{k=0}^K c_k·cos(2πk·i/(n‑1)) )
    The exponential guarantees positivity.
    The coefficients c_k are drawn from a small normal distribution.
    """
    i = np.arange(n, dtype=np.float64)
    coeffs = np.random.randn(K + 1) * 0.2  # fairly small amplitudes
    # cos(0·i) = 1, so coeffs[0] simply shifts the overall level
    raw = np.zeros_like(i)
    for k, c in enumerate(coeffs):
        raw += c * np.cos(2.0 * np.pi * k * i / (n - 1))
    a = np.exp(raw)                     # positivity
    a = a / a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.02,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    ``lam`` controls the temperature; it is increased slowly so that the
    optimisation first sees a smooth landscape and only at the end focuses
    on the true peak.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # unconstrained parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex (ensures sum=1, non‑negativity)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate with respect to a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)       # chain rule for soft‑max

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape shallow plateaux
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    The update is a multiplicative step ``a ← a * exp(-η·grad)`` followed by
    renormalisation onto the simplex.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or according to a linearly decreasing temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.012
    it = 0
    while time.time() < deadline and it < max_moves:
        # move type
        if np.random.rand() < 0.5:          # swap two positions
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer a fraction of mass
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.45)   # up to 45 % of i’s mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # Metropolis acceptance
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The search proceeds through diverse warm‑starts, two‑phase continuous
    optimisation (Adam + sub‑gradient) and a final stochastic hill‑climb.
    The function respects the allotted budget and returns the best sequence
    discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # ------------------------------------------------------------------
    # Basic parameters
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a huge and diverse candidate pool
    candidates = []

    # Warm start (if any)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        # few shifted copies (they are equivalent for the objective but give
        # the optimiser different starting points)
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Deterministic shapes
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))
    for p in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_powerlaw_candidate(n, p))
        candidates.append(_ushape_candidate(n, p))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # Dirichlet draws (various concentrations)
    for alpha in (0.03, 0.07, 0.1, 0.2, 0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # Anti‑pair Dirichlet draws
    for alpha in (0.03, 0.1, 0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # Random Fourier candidates
    for _ in range(10):
        candidates.append(_random_fft_candidate(n))

    # Cosine‑parametrised candidates (low‑frequency)
    for _ in range(15):
        candidates.append(_cosine_param_candidate(n, K=6))

    # ------------------------------------------------------------------
    # 2) Quick evaluation of all candidates, keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:               # degenerate – replace by a fresh Dirichlet draw
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 20 == 0:
            print(f"[Init cand {idx:03d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = min(5, len(evaluated))
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuous optimisation on the top candidates
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_seq.tolist()

    # allocate roughly half of the remaining budget to continuous refinement
    per_candidate_budget = remaining * 0.5 / top_k

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # ----- Adam phase (smooth‑max) -----
        adam_deadline = time.time() + per_candidate_budget * 0.6
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr=0.07,
            lam_start=0.02,
            lam_factor=1.02,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # ----- Sub‑gradient phase -----
        sub_deadline = time.time() + per_candidate_budget * 0.4
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

        # If we run out of time, quit early
        if time.time() >= deadline - 5.0:
            break

    # ------------------------------------------------------------------
    # 4) Stochastic hill‑climbing (local refinements)
    if time.time() < deadline - 5.0:
        local_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Final safeguard – guarantee a valid return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
