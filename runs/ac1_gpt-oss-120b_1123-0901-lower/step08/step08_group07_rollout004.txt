```python
def _alternating_arrangement(vals):
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    left = 0

    toggle = True
    idx = 0
    while idx < n:
        if toggle:
            a[left] = sorted_vals[idx]
            left += 1
        else:
            a[right] = sorted_vals[idx]
            right -= 1
        toggle = not toggle
        idx += 1
    return a
```

assistantfinal**Solution Overview**

The goal is to minimise  

\[
F(a)=\frac{2\,n\; \max(a\star a)}{(\sum_i a_i)^2},
\qquad a_i\ge 0,
\]

where `a★a` is the linear self‑convolution of the step‑function.  
`F(a)` is invariant under a global scaling of `a`, therefore the optimisation can be carried out on the simplex (∑a=1) and the final scaling does not matter for the score.

The previous implementation already finds a very good value (≈ 1.5077).  
To push the bound further we combine three ideas:

1. **Richer initial pool** – besides the old analytical families we add an *alternating* ordering (largest, smallest, 2‑nd largest, …) which spreads the large coefficients as uniformly as possible over all pairwise distances.

2. **Longer exact‑max sub‑gradient phase** – the exact‑max sub‑gradient (exponentiated‑gradient) converges to a local minimum of the true objective.  
   We devote the majority of the time budget to this phase and run it from several diverse starts.

3. **Fast stochastic refinement** – after the sub‑gradient runs a lightweight simulated‑annealing that moves small amounts of mass between a few coordinates.  
   This can escape shallow basins that the deterministic sub‑gradient cannot leave.

All heavy lifting (convolution, gradient) is performed by FFT, which makes the inner loop fast enough to explore many restarts within the 1000 s limit.

The function `propose_candidate` respects the security checks of `evaluate_sequence`, returns a plain Python list of non‑negative floats and prints progress information.

```python
"""
Propose a step‑function with a low value of
    2·n·max(a∗a)/(sum(a)²).

Algorithm (≈ 1000 s):
  1. Build a diverse warm‑start pool:
     - uniform,
     - Dirichlet samples (various concentrations),
     - power‑law, u‑shape, Gaussian‑edge, triangle,
     - anti‑pair and *alternating* arrangements of the above.
  2. Evaluate all candidates cheaply (exact objective) and keep the
     best `top_k` shapes.
  3. For each of the best shapes run a long exact‑max sub‑gradient
     optimisation (exponentiated‑gradient) – this directly minimises the
     true objective.
  4. With the remaining time run a fast stochastic hill‑climb that
     swaps entries or transfers a fraction of mass, accepting worsening
     moves with a slowly cooling temperature.
  5. Return the best sequence found (clipped to the allowed range).

The heavy inner loop uses FFT‑based convolution and cross‑correlation.
"""

import time
import numpy as np

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (without the safety‑check wrapper) --------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals /= vals.sum()
    return vals


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑half left, smallest‑half right (previous approach)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _alternating_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Largest, smallest, 2nd‑largest, 2nd‑smallest, ….
    This spreads big coefficients as uniformly as possible across the
    index distance spectrum.
    """
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    toggle = True
    for v in sorted_vals:
        if toggle:
            a[left] = v
            left += 1
        else:
            a[right] = v
            right -= 1
        toggle = not toggle
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _alternating_arrangement(vals)


# ----------------------------------------------------------------------
# Optimisation primitives -----------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient) on the true objective.
    Works directly on the simplex (no explicit projection needed – we
    renormalise after each multiplicative update).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size (standard sub‑gradient schedule)
        eta = eta0 / np.sqrt(it + 1)

        # multiplicative update – guarantees non‑negativity
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Lightweight simulated‑annealing hill‑climbing.
    Two move types:
      * swap two entries,
      * transfer a random fraction (≤40 %) of mass from i to j.
    Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                               # transfer
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)  # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    """
    np.random.seed(seed)
    # a small safety margin so that the runner does not timeout
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a large, diverse pool of warm‑starts
    candidates = []

    # warm start from the previous run (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            candidates.append(warm / warm.sum())

    # uniform
    candidates.append(_uniform_candidate(n))

    # Dirichlet (different concentrations)
    for alpha in (0.05, 0.1, 0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # power‑law shapes
    for p in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_powerlaw_candidate(n, p))

    # U‑shape (valley in the centre)
    for p in (0.5, 1.0, 2.0):
        candidates.append(_ushape_candidate(n, p))

    # Gaussian edges
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # triangle
    candidates.append(_triangle_candidate(n))

    # anti‑pair and alternating arrangements of a Dirichlet base
    for alpha in (0.1, 0.3, 0.7, 1.0):
        base = np.random.dirichlet(alpha * np.ones(n))
        candidates.append(_anti_pair_arrangement(base / base.sum()))
        candidates.append(_alternating_arrangement(base / base.sum()))

    # a few random permutations of an anti‑pair starter (helps escape sym.)
    base_ap = _alternating_candidate(n, alpha=0.7)
    for _ in range(3):
        perm = np.random.permutation(n)
        cand = base_ap[perm]
        candidates.append(cand)

    # --------------------------------------------------------------
    # 2) Quick exact evaluation, keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 10 == 0:
            print(f"[Init cand {idx:03d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Exact‑max sub‑gradient optimisation on the top candidates
    # Allocate about 70 % of the remaining budget to these deterministic runs
    remaining = deadline - time.time()
    per_candidate_budget = remaining * 0.7 / top_k

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        sub_dead = time.time() + per_candidate_budget
        print(f"[Subgrad start cand {cid}] budget {per_candidate_budget:.1f}s")
        a_sub, val_sub = _subgradient_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=500_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid}] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Stochastic hill‑climbing with the remaining time
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        print("[Local search] starting")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 5) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
