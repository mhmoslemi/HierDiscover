"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that has a low
value of

    F(a) = 2·n·max(a*a) / (sum(a)^2)

where “a*a” is the linear convolution of a with itself.  The algorithm works
in three stages under the (≈1000 s) time budget:

1. **Diverse warm‑starts** – a large pool of deterministic families
   (edge‑exponential, beta, double‑Gaussian, sinusoid, uniform) together with
   many Dirichlet draws.  Each candidate is normalised to sum = 1 and the
   best few are retained.

2. **Deep optimisation** – the current best vector is refined by
   *Adam* on a smooth‑max surrogate (log‑sum‑exp) followed by an
   exponentiated‑gradient sub‑gradient phase and a projected‑gradient phase.
   The updates are performed on the simplex (soft‑max / Euclidean projection)
   and after every iteration the vector is symmetrised
   ``a = (a + a[::-1]) / 2`` because the problem is symmetric.

3. **Directed local search** – a cheap but focused hill‑climber that moves
   a fraction of mass from indices that contribute most to the current
   maximal convolution (large gradient) to indices that contribute least
   (small gradient).  The move is accepted only if the exact objective
   improves; otherwise the step size is shrunk.

The routine always returns the best sequence seen before the deadline.
All numbers are clipped to the evaluator’s admissible range
``[0, 1000]``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Families of initial vectors -----------------------------------------
def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    pool = []

    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
                 (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    for k in (32, 64, 128, 256, 512, 1024):
        pool.append(_sparse_spikes(n, k, rng))

    return pool

# Exact objective (faster than evaluate_sequence for internal use) -----
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)

# Optimisers -----------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr_start: float = 0.02,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # soft‑max temperature (inverse)
    lam_factor = 1.01   # slower increase -> smoother optimisation

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a point on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # enforce symmetry (problem is symmetric)
        a = (a + a[::-1]) * 0.5

        # exact objective
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirror‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        # occasional tiny random shake near deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)          # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # maintain symmetry
        a = (a + a[::-1]) * 0.5

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:

    best_a = a.copy()

            best_a = a.copy()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a -= eta * grad_a
        a = _project_onto_simplex(a)

        # symmetry
        a = (a + a[::-1]) * 0.5

def _directed_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    move_frac_start: float = 0.2,
    move_frac_min: float = 1e-5,
    top_k: int = 5,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    move_frac = move_frac_start

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[Local it={it:06d}] improved to {val:.6f}")

        # gradient w.r.t. the active (max) indices
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        src_idx = np.argsort(-grad)[:top_k]          # biggest contributors
        dst_idx = np.argsort(grad)[:top_k]           # smallest contributors

                a2[dst] += delta
                # enforce symmetry after the move
                a2 = (a2 + a2[::-1]) * 0.5

                val2 = _objective(a2, n, conv_fft_len)
                if val2 < val:
                    a = a2
                    improved = True
                    break
            if improved:
                break

        if not improved:
            move_frac = max(move_frac * 0.7, move_frac_min)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point required by the evaluation harness.
    Returns a list of non‑negative heights (Python floats).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size (feel free to change; 4096 works well with the existing
    # helper functions and gives a good trade‑off between speed and quality)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a diverse pool of starts (deterministic families + random)
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # Keep the K best after a cheap evaluation
    top_k = 5
    best_candidates = []  # list of (value, vector)

    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if len(best_candidates) < top_k:
            best_candidates.append((val, a))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, a)
                best_candidates.sort(key=lambda x: x[0])

    best_val = best_candidates[0][0]
    best_seq = best_candidates[0][1].copy()
    print(f"[Init] best of pool value {best_val:.6f}")

    # --------------------------------------------------------------
    # Allocate time for the three optimisation phases
    time_left = lambda: deadline - time.time()
    if time_left() < 5.0:
        # not enough time – give back what we have
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        print(f"Final (early) best {best_val:.6f}")
        return best_seq.tolist()

    # Phase‑2a: Adam on smooth‑max
    adam_deadline = time.time() + 0.45 * time_left()
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=600_000,
        lr_start=0.02,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Phase Adam] new best {val_adam:.6f}")

    # Phase‑2b: Exponentiated‑gradient sub‑gradient
    sub_deadline = time.time() + 0.35 * time_left()
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_deadline,
        max_iters=800_000,
        eta0=0.8,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Phase Subgrad] new best {val_sub:.6f}")

    # Phase‑2c: Projected‑gradient descent (different dynamics)
    pgd_deadline = time.time() + 0.15 * time_left()
    a_pgd, val_pgd = _pgd_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        pgd_deadline,
        max_iters=300_000,
        eta0=0.4,
    )
    if val_pgd < best_val:
        best_val = val_pgd
        best_seq = a_pgd.copy()
        print(f"[Phase PGD] new best {val_pgd:.6f}")

    # --------------------------------------------------------------
    # Phase‑3: Directed local hill‑climber for the last seconds
    if time.time() < deadline - 2.0:
        local_deadline = deadline
        a_loc, val_loc = _directed_local_search(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            move_frac_start=0.2,
            top_k=5,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase Local] improved best {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform (guaranteed feasible) if everything failed
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted domain
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
