"""
Propose a low‑value step‑function.

Algorithm
---------

1. **Small‑scale exploration** (n≈1024)  
   – generate many simple parametric shapes (exponential edges,
     power‑law edges, double‑Gaussian, half‑wave sinusoid).  
   – pick the one with the lowest surrogate objective
     `2·n·max(conv)/sum^2`.

2. **Linear‑programming boost**  
   For the current distribution `a` compute `M = max(a*a)`.  
   Solve a linear program  

        maximise   1ᵀ g
        subject to  a * g ≤ M        (element‑wise, linear in g)
                    g ≥ 0

   (here `*` is linear convolution with zero padding).  The LP
   pushes mass into directions that keep the convolution under the
   current bound.  Replace `a` by a convex combination with the
   normalised LP solution and repeat a few times.

3. **Exact‑max sub‑gradient refinement**  
   Using the exponentiated‑gradient (mirror‑descent on the simplex)
   with the true max‑convolution gradient.  Run as many iterations
   as time permits.

4. **Upsampling** – linearly interpolate the refined `a` to the target
   length (default 4096) and renormalise.

5. **Final refinement** – another round of sub‑gradient steps
   followed by a short simulated‑annealing local search.

All phases respect the global time budget (default 1000 s).  The
routine always returns the best sequence seen (as a Python list);
its elements are clipped to `[0, 1000]` as required by
`evaluate_sequence`.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize

# --------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient on the simplex."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)          # mirror‑descent on simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing style local moves."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a, a, val = val2, a2.copy(), a2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2

        it += 1

    return best_a, best_val

def _build_conv_matrix(a: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1)×n matrix C such that C @ g = a * g (linear conv).
    Sparse representation with O(n²) non‑zeros (manageable for n≤1024).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []

    # column k contributes a shifted copy of a on rows k … k+n‑1
    for k in range(n):
        rows.append(np.arange(k, k + n, dtype=np.int32))
        cols.append(np.full(n, k, dtype=np.int32))
        data.append(a)

    rows = np.concatenate(rows)
    cols = np.concatenate(cols)
    data = np.concatenate(data)

    return sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))


def _lp_improve(a: np.ndarray, deadline: float, max_iters: int = 5, t: float = 0.5):
    """Perform a few LP‑based improvement steps (see description)."""
    n = a.shape[0]
    conv_fft_len = _next_pow2(2 * n - 1)

    for it in range(max_iters):
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # Build constraint matrix for current a
        C = _build_conv_matrix(a)

        # Linear program: maximise sum(g)  <=> minimise -1ᵀg
        c = -np.ones(n)
        A_ub = C
        b_ub = np.full(C.shape[0], M)

        # time guard – give at most the remaining seconds to the solver
        remaining = max(0.0, deadline - time.time())
        if remaining < 2.0:
            break
        options = {"presolve": True}
        try:
            res = scipy.optimize.linprog(
                c,
                A_ub=A_ub,
                b_ub=b_ub,
                bounds=(0, None),
                method="highs",
                options=options,
            )
        except Exception:  # pylint: disable=broad-except
            res = None

        if res is None or not res.success:
            break

        g = res.x
        if g.sum() < 1e-12:
            break

        g_norm = g / g.sum()
        a_new = (1 - t) * a + t * g_norm  # convex combination (remains on simplex)
        conv_new = _full_convolution(a_new, conv_fft_len)
        if conv_new.max() < M - 1e-12:
            print(f"[LP it={it}] improved max‑conv from {M:.9f} to {conv_new.max():.9f}")
            a = a_new
        else:
            break
    return a

def _initial_small_candidates(n: int) -> list[np.ndarray]:
    """Create a modest pool of simple parametric shapes (n ≈ 1024)."""
    cand = []

    # exponential edges
    for gamma in np.logspace(-2, 0.5, 10):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # power‑law edges (negative exponent)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-4.0, -0.2, 12):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # double Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.15, 0.3):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # half‑wave sinusoid (positive part)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # uniform (baseline)
    cand.append(np.full(n, 1.0 / n))

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Phase I – small‑scale exploration (n≈1024)
    n_small = 1024
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    pool = _initial_small_candidates(n_small)

    best_a = None
    best_metric = float("inf")  # metric = n * max(conv)  (lower is better)

    for a0 in pool:
        if time.time() >= deadline:
            break
        a = a0 / a0.sum()
        conv = _full_convolution(a, conv_fft_len_small)
        metric = n_small * conv.max()
        if metric < best_metric:
            best_metric = metric
            best_a = a.copy()
            print(f"[Explore] candidate metric {metric:.9f}")

    # ------------------------------------------------------------------
    # Phase II – LP boost (a few cheap LP solves)
    if best_a is not None and time.time() < deadline - 5:
        best_a = _lp_improve(best_a, deadline, max_iters=5, t=0.5)

    # ------------------------------------------------------------------
    # Phase III – long sub‑gradient refinement on the small grid
    if best_a is not None and time.time() < deadline - 10:
        print("[Refine‑small] start sub‑gradient")
        best_a, _ = _subgradient_optimize(
            best_a,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            deadline,
            max_iters=300_000,
            eta0=0.8,
        )

    # ------------------------------------------------------------------
    # Phase IV – upsample to target size (default 4096)
    n_large = 4096
    if best_a is not None and time.time() + 30 < deadline:
        print("[Upsample] interpolate to larger grid")
        x_small = np.linspace(0, 1, n_small, endpoint=False)
        x_large = np.linspace(0, 1, n_large, endpoint=False)
        a_large = np.interp(x_large, x_small, best_a)
        a_large = np.maximum(a_large, 0.0)
        if a_large.sum() < 1e-12:
            a_large = np.full(n_large, 1.0 / n_large)
        else:
            a_large /= a_large.sum()
        best_a = a_large
    else:
        # fall back to a plain up‑sample of a uniform vector if something went wrong
        if best_a is None:
            best_a = np.full(n_large, 1.0 / n_large)

    # ------------------------------------------------------------------
    # Phase V – final sub‑gradient refinement on the full size
    conv_fft_len_large = _next_pow2(2 * n_large - 1)
    corr_fft_len_large = _next_pow2(3 * n_large - 2)

    if time.time() < deadline - 8:
        print("[Refine‑large] sub‑gradient on full size")
        best_a, _ = _subgradient_optimize(
            best_a,
            n_large,
            conv_fft_len_large,
            corr_fft_len_large,
            deadline,
            max_iters=240_000,
            eta0=0.85,
        )

    # ------------------------------------------------------------------
    # Phase VI – simulated‑annealing fine‑tuning
    if time.time() < deadline - 2:
        print("[Anneal] final simulated‑annealing")
        best_a, _ = _anneal_optimize(
            best_a, n_large, conv_fft_len_large, deadline, max_moves=70_000
        )

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best_a is None:
        best_a = np.full(4096, 1.0 / 4096)

    best_a = np.clip(best_a, 0.0, 1000.0)

    # Optional: report the final objective using the provided evaluator
    try:
        from __main__ import evaluate_sequence  # the harness injects it
        final_val = evaluate_sequence(best_a.tolist())
        print(f"Final objective (evaluate_sequence) = {final_val:.6f}")
    except Exception:  # pragma: no cover
        pass

    return best_a.tolist()
