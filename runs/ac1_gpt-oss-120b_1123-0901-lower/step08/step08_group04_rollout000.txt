"""
Propose a low‑value step function.

Algorithm (multi‑scale):
1.  Quick continuous optimisation on a small problem (n≈2048) using
    Adam on a smooth‑max surrogate and a sub‑gradient phase.
2.  Upsample the obtained vector to a larger size (n≈8192) by linear
    interpolation – this preserves the shape while providing many more
    degrees of freedom.
3.  Long simulated‑annealing (SA) on the large vector.  The SA uses an
    O(n) incremental update of the full linear convolution, constantly
    transferring a random fraction of mass between two positions.
    The temperature decays linearly and the maximal transferred fraction
    is reduced as the run proceeds, allowing a fast exploration phase
    followed by a fine‑grained polishing phase.
4.  The best vector found (normalized to sum = 1) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _warm_start(n: int):
    """If a previous solution exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives ----------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # soft‑max → simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c               # denominator is 1 (sum=1)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Incremental convolution update (used by the long SA) ------------------

def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


def _incremental_sa(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.02,
    max_frac: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill climbing that uses the incremental
    convolution update.  Moves are transfers of a random fraction of
    mass from one position to another.  The maximal transferred
    fraction linearly decays, allowing coarse moves early and fine
    tuning later.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    while time.time() < time_limit:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # transfer a random fraction of the mass at i to j
        # the allowed fraction shrinks as the run progresses
        elapsed = time.time() - start
        frac_allowed = max_frac * (1.0 - elapsed / (time_limit - start))
        frac_allowed = max(frac_allowed, 0.02)   # never go below ~2 %
        frac = np.random.rand() * frac_allowed
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # apply move
        _apply_delta(a, conv, i, -delta)
        _apply_delta(a, conv, j, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        # linear temperature schedule (hot -> cold)
        remaining = max(0.0, time_limit - time.time())
        temp = start_temp * (remaining / (time_limit - start))

        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert
            _apply_delta(a, conv, j, -delta)
            _apply_delta(a, conv, i, delta)

        it += 1
        if it % 200_000 == 0:
            print(f"[SA it={it:,}] best maxconv {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    The routine works in two stages:
      * a cheap continuous optimisation on a modest size (n≈2048);
      * an up‑sample to a larger size (n≈8192) followed by a long
        simulated‑annealing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – small problem (fast Adam + sub‑gradient)
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    # build a diverse pool of warm‑starts for the small size
    small_cands = []

    warm = _warm_start(n_small)
    if warm is not None:
        small_cands.append(warm.copy())

    small_cands.append(_uniform_candidate(n_small))
    small_cands.append(_triangle_candidate(n_small))
    small_cands.append(_gaussian_edges_candidate(n_small, 0.2))
    small_cands.append(_gaussian_edges_candidate(n_small, 0.5))

    for alpha in (0.3, 0.7, 1.0, 2.0):
        small_cands.append(_random_dirichlet_candidate(n_small, alpha))
    for alpha in (0.3, 0.7, 1.0):
        small_cands.append(_anti_pair_candidate(n_small, alpha))

    # pick the best starter (exact evaluation)
    best_small = None
    best_small_val = float("inf")
    for idx, cand in enumerate(small_cands):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n_small))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init small {idx:02d}] value {val:.6f}")
        if val < best_small_val:
            best_small_val = val
            best_small = cand.copy()

    # ---- Adam phase ---------------------------------------------------
    if time.time() < deadline - 30.0:
        adam_deadline = min(deadline - 20.0, time.time() + 180.0)
        a_adam, val_adam = _adam_optimize(
            best_small,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            adam_deadline,
            max_iters=400_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_small_val:
            best_small_val = val_adam
            best_small = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient phase -------------------------------------------
    if time.time() < deadline - 25.0:
        sub_deadline = min(deadline - 15.0, time.time() + 250.0)
        a_sub, val_sub = _subgradient_optimize(
            best_small,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            sub_deadline,
            max_iters=400_000,
            eta0=0.7,
        )
        if val_sub < best_small_val:
            best_small_val = val_sub
            best_small = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – upsample to a larger size and run long SA
    n_big = 8192
    # linear interpolation to obtain an initial guess for the big problem
    x_small = np.linspace(0.0, 1.0, n_small)
    x_big = np.linspace(0.0, 1.0, n_big)
    init_big = np.interp(x_big, x_small, best_small)
    init_big = np.maximum(init_big, 0.0)
    init_big /= init_big.sum()

    conv_fft_len_big = _next_pow2(2 * n_big - 1)

    # Run SA for essentially the whole remaining budget
    if time.time() < deadline - 5.0:
        sa_time_limit = deadline - 5.0
        print(f"[SA] starting on n={n_big} for up to {sa_time_limit - time.time():.1f}s")
        a_sa, maxconv_sa = _incremental_sa(
            init_big,
            time_limit=sa_time_limit,
            start_temp=0.04,
            max_frac=0.45,
        )
        # compute the true objective for the SA result
        final_val = 2.0 * n_big * maxconv_sa
        print(f"[SA] finished – maxconv {maxconv_sa:.6f}, objective {final_val:.6f}")
        best_seq = a_sa.copy()
        best_val = final_val
    else:
        best_seq = init_big.copy()
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    # Final sanity checks
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.shape[0])
    else:
        best_seq = best_seq / total

    print(f"Returning sequence of length {len(best_seq)} with objective {best_val:.6f}")
    return best_seq.tolist()
