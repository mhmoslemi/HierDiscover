def _smooth_max_obj_grad(x, n, lam, conv_fft_len, corr_fft_len):
    a = _softmax(x)
    conv = _full_convolution(a, conv_fft_len)
    # smooth max
    # compute Z = sum(exp(lam * conv)) maybe with shift for stability
    # but conv values small ~ <= maybe 0.001; lam large 5 -> still small -> compute safely:
    # Use max_conv = conv.max() for numerical stability
    max_conv = conv.max()
    exp_term = np.exp(lam * (conv - max_conv))
    Z = exp_term.sum()
    smooth = max_conv + (np.log(Z) / lam)
    # gradient w.r.t a:
    w = exp_term / Z  # normalized weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # gradient w.r.t x: a * (grad_a - np.dot(a, grad_a))
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    # overall objective factor: 2 * n * smooth
    obj = 2.0 * n * smooth
    grad = 2.0 * n * grad_log
    return obj, grad

"""
Propose a low‑value step function for
    F(a) = 2·n·max(a∗a) / (sum(a)²)

The search consists of three stages (time‑aware):

1. **Broad random sampling of low‑dimensional parametric families**  
   (edge‑exponential, power‑law, beta‑pdf, double‑Gaussian, sinusoid,
   uniform, Dirichlet, sparse spikes) on several lengths (256‑4096).
   Every candidate is normalised to sum = 1 and evaluated quickly with an
   FFT‑based convolution.  The best shape so far is kept.

2. **Refinement with a smooth‑max surrogate using L‑BFGS‑B**.  
   The surrogate is
        Sλ(a) = (1/λ)·log Σ exp(λ·(a∗a))
   with λ slowly increased (fixed to 5.0 here).  We optimise the
   soft‑max parameters `x` (a = softmax(x)) on the simplex with the
   exact gradient (cross‑correlation).  The optimisation stops when the
   global deadline is reached.

3. **Cheap final hill‑climber** – repeatedly move a random fraction of the
   mass from one index to another and keep the move if the true objective
   improves.

All steps respect the 1000 s budget and always return the best sequence
found.  The code works with any existing warm‑start stored in the global
variable ``height_sequence_1``.
"""

import time

# FFT helpers (identical to the reference implementation)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (returns a vector that sums to 1)."""
    mx = x.max()
    e = np.exp(x - mx)
    return e / e.sum()

# 1️⃣  Parametric families (all return a normalised non‑negative vector)
def _gen_edge_exponential(n: int, rng: np.random.Generator) -> np.ndarray:
    # γ drawn log‑uniform in [1e‑2, 1e1]
    gamma = 10 ** rng.uniform(-2, 1)
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _gen_edge_power_law(n: int, rng: np.random.Generator) -> np.ndarray:
    # α drawn uniformly in [‑3.5, ‑0.2]
    alpha = rng.uniform(-3.5, -0.2)
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _gen_beta_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.uniform(0.2, 5.0)
    b = rng.uniform(0.2, 5.0)
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _gen_double_gaussian(n: int, rng: np.random.Generator) -> np.ndarray:
    sigma = 10 ** rng.uniform(-2, -0.1)   # σ∈[0.01,0.79]
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _gen_sinusoid(n: int, rng: np.random.Generator) -> np.ndarray:  # noqa: ARG001
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _gen_uniform(n: int, rng: np.random.Generator) -> np.ndarray:  # noqa: ARG001
    return np.full(n, 1.0 / n, dtype=np.float64)


def _gen_dirichlet(n: int, rng: np.random.Generator) -> np.ndarray:
    # concentration drawn from a small set
    alpha = rng.choice([0.1, 0.3, 0.6, 1.0, 2.0, 5.0])
    return rng.dirichlet(np.full(n, alpha))


def _gen_sparse_spikes(n: int, rng: np.random.Generator) -> np.ndarray:
    # number of spikes from {32, 64, …, 1024}
    k = rng.choice([32, 64, 128, 256, 512, 1024])
    a = np.zeros(n, dtype=np.float64)
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# 2️⃣  Fast exact objective (no clipping, sum is assumed >0)
def _eval_f(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a∗a) / (sum(a)²).  a is assumed non‑negative."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# 3️⃣  Smooth‑max surrogate and gradient (used by L‑BFGS‑B)
def _smooth_max_obj_grad(
    x: np.ndarray,
    n: int,
    lam: float,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Return (objective, gradient) for the smooth‑max surrogate:
        obj = 2·n·Sλ(a)   with   Sλ(a)= (1/λ)·log Σ exp(λ·(a∗a))
    where a = softmax(x) lives on the simplex.
    The gradient is taken w.r.t. the unconstrained logits x.
    """
    a = _softmax(x)                     # a lies on the simplex
    conv = _full_convolution(a, conv_fft_len)

    # stable soft‑max over the convolution entries
    m = conv.max()
    exp_term = np.exp(lam * (conv - m))
    Z = exp_term.sum()
    smooth = m + (np.log(Z) / lam)       # this equals Sλ(a)

    # weight vector for the smooth‑max (normalised)
    w = exp_term / Z

    # gradient wrt a  (cross‑correlation, see derivation)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the simplex (soft‑max) → gradient wrt logits
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)          # mirror‑descent form

    obj = 2.0 * n * smooth
    grad = 2.0 * n * grad_x
    return obj, grad


# ----------------------------------------------------------------------
# 4️⃣  L‑BFGS‑B wrapper respecting the global deadline
def _lbfgs_refine(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iter: int = 500,
    lam: float = 5.0,
) -> np.ndarray:
    """Run L‑BFGS‑B on the smooth‑max surrogate, aborting on timeout."""
    if scipy is None:  # pragma: no cover
        return a0

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    eps = 1e-12
    x0 = np.log(a0 + eps)  # initialise logits

    class TimeOut(RuntimeError):
        pass

    def _callback(xk):
        if time.time() > deadline:
            raise TimeOut()

    try:
        res = scipy.optimize.minimize(
            fun=_smooth_max_obj_grad,
            x0=x0,
            args=(n, lam, conv_fft_len, corr_fft_len),
            method="L-BFGS-B",
            jac=True,
            callback=_callback,
            options={"maxiter": max_iter, "disp": False},
        )
        best_x = res.x
    except TimeOut:
        # optimisation cut short – fall back to current point
        best_x = x0

    a_opt = _softmax(best_x)
    # guard against numerical drift
    a_opt = np.maximum(a_opt, 0.0)
    a_opt /= a_opt.sum()
    return a_opt


# ----------------------------------------------------------------------
# 5️⃣  Very cheap random local moves (final polishing)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> np.ndarray:
    """
    Simple hill‑climber: transfer a random fraction of mass from one index
    to another; keep the move if the true objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # (a) possible lengths – we explore a few to let the optimiser pick the
    # best scale without being stuck at a single resolution.
    lengths = [256, 512, 1024, 2048, 4096]

    # ------------------------------------------------------------------
    # (b) warm start handling (global variable `height_sequence_1` may exist)
    warm = globals().get("height_sequence_1")
    warm_arr = None
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        warm_arr = np.maximum(warm_arr, 0.0)

    # ------------------------------------------------------------------
    # (c) helper to evaluate a candidate quickly (no clipping)
    def _fast_eval(vec: np.ndarray, n_: int) -> float:
        fft_len = _next_pow2(2 * n_ - 1)
        return _eval_f(vec, n_, fft_len)

    best_seq = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Phase 1 – random exploration of parametric families
    phase1_deadline = min(deadline - 10.0, start_time + 0.30 * budget_s)

    # list of generator functions (no lambdas, per the rules)
    generators = [
        _gen_edge_exponential,
        _gen_edge_power_law,
        _gen_beta_shape,
        _gen_double_gaussian,
        _gen_sinusoid,
        _gen_uniform,
        _gen_dirichlet,
        _gen_sparse_spikes,
    ]

    while time.time() < phase1_deadline:
        # pick a length – give a slight preference to larger n (more expressive)
        n = int(rng.choice(lengths, p=np.array([0.1, 0.15, 0.25, 0.25, 0.25]))
                )
        # pick a generator
        gen = rng.choice(generators)
        a = gen(n, rng)                     # each generator returns a normalised vector

        # occasionally inject a (truncated/padded) warm start
        if warm_arr is not None and rng.random() < 0.05:
            w = warm_arr
            if w.size > n:
                w = w[:n]
            elif w.size < n:
                w = np.pad(w, (0, n - w.size))
            a = np.maximum(w, 0.0)
            a = a / a.sum()

        # safety – ensure a is not degenerate
        if a.sum() < 0.01:
            a = rng.dirichlet(np.ones(n))

        val = _fast_eval(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_n = n
            print(f"[Random] new best {val:.6f} (n={n}, gen={gen.__name__})")

    # ------------------------------------------------------------------
    # Phase 2 – L‑BFGS‑B refinement of the current best (if any)
    if best_seq is not None and time.time() < deadline - 8.0:
        # keep the best length for the refined optimisation
        n = best_n
        print(f"Refining best candidate from phase 1 (n={n})")
        a_refined = _lbfgs_refine(best_seq, n, deadline - 3.0, max_iter=800)
        val_refined = _fast_eval(a_refined, n)
        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[L‑BFGS] improved to {val_refined:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap local random moves (few seconds)
    if best_seq is not None and time.time() < deadline - 1.0:
        n = best_seq.shape[0]
        conv_fft_len = _next_pow2(2 * n - 1)
        a_final = _local_random_search(best_seq, n, conv_fft_len, deadline)
        val_final = _fast_eval(a_final, n)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Local] final improvement to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback to uniform if everything failed (should never happen)
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _fast_eval(best_seq, n)

    # Final safety clip (matches the evaluator’s post‑processing)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {best_seq.shape[0]})")
    return best_seq.tolist()
