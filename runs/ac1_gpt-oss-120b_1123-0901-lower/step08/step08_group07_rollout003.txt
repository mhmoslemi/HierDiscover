"""
Propose a low‑value non‑negative step function.

Algorithm outline (≈1000 s budget):
1. Choose a relatively large grid (n = 8192) – larger n gives
   more degrees of freedom while the objective 2·n·max(conv) stays
   scale‑invariant.
2. Build a diverse pool of warm‑starts:
   – previous best (if any) and its cyclic shifts,
   – uniform,
   – Dirichlet samples (various concentrations),
   – anti‑pair (big ↔ small) ordering,
   – two new deterministic placements:
        • “alternating’’ (large/small interleaved),
        • “stride’’ (large values spaced by a coprime stride).
   – power‑law, U‑shape, Gaussian‑edge and triangular shapes.
   All candidates are normalised to sum‑1.
3. Evaluate the pool with the exact objective and keep the top‑k
   (k = 3) candidates.
4. For each top candidate run two continuous refinements:
   – Adam on a smooth‑max surrogate (log‑sum‑exp) with an
     exponentially increasing temperature (λ).  The surrogate
     pushes the convolution towards flatness.
   – Exact‑max exponentiated‑gradient (sub‑gradient) phase,
     which directly minimises the true maximum.
   Time is split evenly among the top‑k candidates.
5. A long stochastic hill‑climbing phase on the current best
   sequence (≈300 k moves).  Moves are either swaps or mass
   transfers; acceptance follows a linear‑cooling simulated‑annealing
   schedule, allowing occasional uphill steps to escape local minima.
6. If time permits, a brief up‑sampling to 16384 points followed
   by one more Adam+sub‑gradient pass (very cheap because the
   objective hardly improves on larger grids).
7. Return the best sequence found (clipped to the allowed [0,1000] range).

All heavy work (convolutions, gradients) is performed via FFT
for O(n log n) speed.  The code respects the security checks of
``evaluate_sequence`` – the returned list contains only non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------

def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)

def _warm_start(n: int):
    """Return a normalised warm‑start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair) ordering."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _alternating_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """
    Interleaved large/small pattern: fill even indices with the largest
    half of the Dirichlet sample, odd indices with the smallest half.
    """
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    evens = np.arange(0, n, 2)
    odds = np.arange(1, n, 2)
    a[evens] = vals[: evens.size]
    a[odds] = vals[evens.size : evens.size + odds.size]
    # If n is odd the last element (odd index) already received a value.
    return a


def _stride_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """
    Place sorted values on positions spaced by a coprime stride
    (n//2 + 1).  This spreads large entries uniformly over the grid.
    """
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.sort(vals)[::-1]  # descending
    stride = n // 2 + 1
    a = np.empty(n, dtype=np.float64)
    pos = 0
    for v in vals:
        a[pos] = v
        pos = (pos + stride) % n
    return a


def _triangle_candidate(n: int) -> np.ndarray:
    """Tent shape (large at the edges, small in the middle)."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape = shape / shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape = shape / shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p).  p>0 concentrates mass on the first entries."""
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    """
    Valley in the centre, larger at the two ends:
    a_i ∝ ( i·(n‑1‑i) + 1 )^{-p}.
    """
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (sum‑normalised) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # map logits → simplex via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if (it % 2000) == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp temperature)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if (it % 4000) == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    # final renormalisation
    best_a = best_a / best_a.sum()
    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient) phase.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if (it % 5000) == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active mask (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (projected onto simplex)
        a = a * np.exp(-eta * grad_a)
        a_sum = a.sum()
        if a_sum < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        it += 1

    best_a = best_a / best_a.sum()
    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing.
    Moves: swap two entries or transfer a fraction of mass.
    Temperature linearly decays from 0.02 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.5:            # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                 # transfer
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.5)   # up to 50 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if (it % 5000) == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    best_a = best_a / best_a.sum()
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(sum(a)²).

    The routine follows the stages described in the module docstring.
    """
    np.random.seed(seed)
    # safety margin for the outer framework
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Grid size – we aim for 8192 points (more degrees of freedom).
    # If time is already very low we fall back to 4096.
    n = 8192
    if (deadline - time.time()) < 200.0:      # not enough time → downgrade
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of warm‑starts
    candidates = []

    # warm start (global variable)
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # uniform
    candidates.append(_uniform_candidate(n))

    # Dirichlet with a wide range of concentrations
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # anti‑pair Dirichlet
    for alpha in (0.1, 0.3, 0.5, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # new deterministic placements
    for alpha in (0.5, 1.0):
        candidates.append(_alternating_candidate(n, alpha))
        candidates.append(_stride_candidate(n, alpha))

    # deterministic shapes
    for p in (0.2, 0.5, 1.0, 2.0):
        candidates.append(_powerlaw_candidate(n, p))
        candidates.append(_ushape_candidate(n, p))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # random permutations of a strong anti‑pair starter (more diversification)
    base_ap = _anti_pair_candidate(n, alpha=0.7)
    for _ in range(3):
        perm = np.random.permutation(n)
        candidates.append(base_ap[perm])

    # ------------------------------------------------------------------
    # 2) Quick evaluation – keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 10 == 0:
            print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 3
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuous optimisation on the top candidates
    remaining = deadline - time.time()
    per_candidate_budget = max(0.0, remaining * 0.5 / top_k)

    for cid, (_, cand) in enumerate(evaluated[:top_k]):
        # ---- Adam phase ----
        adam_dead = time.time() + per_candidate_budget * 0.5
        a_adam, val_adam = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=500_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

        # ---- Sub‑gradient phase ----
        sub_dead = time.time() + per_candidate_budget * 0.5
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=500_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing (large budget)
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (if time left)
    if time.time() < deadline - 30.0 and n == 8192:
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on larger grid
        big_adam_dead = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.07,
        )
        # sub‑gradient refinement
        big_sub_dead = deadline
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_dead,
            max_iters=250_000,
            eta0=0.6,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
