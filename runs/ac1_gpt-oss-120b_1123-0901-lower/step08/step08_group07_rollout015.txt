"""
Propose a low‑value step function.

Algorithm
---------
1. **Warm‑start + diverse population** – build a small population of
   probability vectors (length n).  The previous best sequence is used
   if available, otherwise random Dirichlet samples with various
   concentration parameters are generated.

2. **Differential Evolution (DE)** – a classic global optimiser that
   works directly on the non‑convex max‑convolution objective.
   For each individual we create a mutant vector by the DE/rand/1 scheme,
   crossover it with the parent, enforce non‑negativity and renormalise.
   If the trial improves the objective (lower max‑convolution) it replaces
   the parent.  The best individual is kept globally.

3. **Local fine‑tuning** – the best DE individual is refined with the
   Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate and then with an
   exact‑max exponentiated‑gradient (sub‑gradient) phase, both from the
   original code base.  These phases use only the current best vector,
   allowing many more iterations in the remaining time.

4. **Safety & return** – clip values to the limits imposed by
   ``evaluate_sequence`` and return the best sequence (as a plain list).

The whole routine respects the 1000 s budget, repeats DE cycles while
time permits, and finishes with a high‑precision local optimisation.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no security checks) ----------------------------------
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) / (sum(a)^2); assumes a ≥ 0."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helper -----------------------------------------------------
def _warm_start(n: int):
    """Return a normalised warm start from the global ``height_sequence_1``."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Random candidate generators --------------------------------------------
def _random_dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet(α,…,α) vector and return it normalised."""
    # gamma sampling is faster and more stable than np.random.dirichlet for very small α
    g = np.random.gamma(alpha, 1.0, size=n)
    if g.sum() == 0.0:
        g = np.ones(n)
    g = g / g.sum()
    return g.astype(np.float64)


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact value for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # Temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # Occasional jitter to avoid stagnation
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Decaying step size
        eta = eta0 / np.sqrt(it + 1)

        # Multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad_a)

        # Renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found as a Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 1.0   # 1 s safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Initialise a modest population for differential evolution
    popsize = 20
    population = []
    scores = []

    # Warm‑start (if available) is inserted as the first individual
    warm = _warm_start(n)
    if warm is not None:
        population.append(warm.copy())
        scores.append(_objective(warm))

    # Fill the rest with Dirichlet samples of varying concentration
    while len(population) < popsize:
        alpha = 10 ** np.random.uniform(-2, 1)       # α ∈ [0.01, 10]
        cand = _random_dirichlet_candidate(n, alpha)
        population.append(cand)
        scores.append(_objective(cand))

    # Keep track of the global best
    best_idx = int(np.argmin(scores))
    best_seq = population[best_idx].copy()
    best_val = scores[best_idx]
    print(f"[Init] best initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Differential Evolution loop (global search)
    F = 0.8          # mutation factor
    CR = 0.9         # crossover probability

    # Reserve a small chunk for the final fine‑tuning
    final_refine_time = 30.0

    while time.time() < deadline - final_refine_time:
        for i in range(popsize):
            # Pick three distinct individuals ≠ i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)

            # DE/rand/1 mutation
            mutant = population[r1] + F * (population[r2] - population[r3])

            # Binomial crossover
            cross_mask = np.random.rand(n) < CR
            # Ensure at least one dimension is taken from mutant
            if not np.any(cross_mask):
                cross_mask[np.random.randint(n)] = True
            trial = np.where(cross_mask, mutant, population[i])

            # Enforce feasibility: non‑negative and normalised
            trial = np.maximum(trial, 0.0)
            s = trial.sum()
            if s < 1e-12:            # degenerate case – re‑initialise
                trial = _random_dirichlet_candidate(n, 1.0)
                s = trial.sum()
            trial = trial / s

            # Evaluate
            val = _objective(trial)

            # Greedy selection
            if val < scores[i]:
                population[i] = trial
                scores[i] = val
                if val < best_val:
                    best_val = val
                    best_seq = trial.copy()
                    print(f"[DE] new best {best_val:.6f} (t={time.time():.1f})")
        # optional: re‑inject fresh random individuals if stagnating
        # (not strictly necessary for the budget we have)

    # ------------------------------------------------------------------
    # 3) Local fine‑tuning of the best individual (Adam → sub‑gradient)
    remaining = deadline - time.time()
    if remaining > 2.0:
        # split remaining time roughly half/half
        half = remaining * 0.5
        adam_dead = time.time() + half
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_dead,
            max_iters=300_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Final Adam] improved to {best_val:.6f}")

        sub_dead = deadline - 1.0   # leave a tiny margin for returning
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=300_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Final Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Safety clipping and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
