def _objective_and_grad(y, n, conv_fft_len, corr_fft_len):
    # compute a
    a = _softmax(y)
    # conv
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # gradient wrt a (subgradient)
    active = np.isclose(conv, max_c, atol=1e-15)
    w = np.zeros_like(conv)
    if np.any(active):
        w[active] = 1.0 / np.sum(active)
    else:
        w[active] = 0.0
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # gradient wrt logits
    avg = np.dot(a, grad_a)
    grad_y = a * (grad_a - avg)
    # objective for optimizer (max_c, not scaled)
    return max_c, grad_y

"""
Non‑negative step‑function optimizer.

The algorithm builds a diverse pool of starter vectors (uniform,
Dirichlet with many concentrations, edge‑exponential, Gaussian,
sinusoid, sparse spikes).  Each candidate is first evaluated with the
exact objective and the best few are refined with an *annealed
L‑BFGS* optimizer.  The surrogate objective is the smooth‑max
(log‑sum‑exp) of the convolution, which provides a differentiable
approximation of the true `max`.  L‑BFGS is run several times while
the temperature (λ) is increased, allowing the optimisation to first
smooth the landscape and then focus on the true peak.  The final
solution is returned as a Python list; if the time budget expires a
fallback uniform sequence is supplied.

The code respects the 1000‑second budget, uses only top‑level
functions (no closures) and works with the provided `evaluate_sequence`
function.
"""

from __future__ import annotations
import time
import numpy as np
from scipy.optimize import minimize
from scipy.special import logsumexp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max (produces a probability vector)."""
    mx = np.max(y)
    exp_y = np.exp(y - mx)
    return exp_y / exp_y.sum()


# ----------------------------------------------------------------------
def _smooth_max_obj(
    y: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Smooth‑max (log‑sum‑exp) surrogate and its gradient in the
    log‑probability (logits) space.

    Returns
        obj  –  λ‑scaled smooth maximum of the convolution,
        grad –  gradient w.r.t. y (logits).
    """
    a = _softmax(y)                         # a ≥ 0 , Σa = 1
    conv = _full_convolution(a, conv_fft_len)

    scaled = lam * conv
    m = np.max(scaled)
    exp_scaled = np.exp(scaled - m)
    w = exp_scaled / exp_scaled.sum()       # soft‑max weights, Σw = 1

    # smooth‑max value (exactly log‑sum‑exp(λ·conv)/λ)
    obj = np.dot(w, conv)

    # gradient of the surrogate w.r.t. a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max (mirror descent)
    avg = np.dot(a, grad_a)
    grad_y = a * (grad_a - avg)

    return obj, grad_y


def _annealed_lbfgs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters_per_stage: int = 150,
) -> tuple[np.ndarray, float]:
    """
    Annealed L‑BFGS optimisation.
    Starts from a0 (already normalised Σa = 1) and runs a short L‑BFGS
    optimisation for a sequence of increasing λ‑values, each time using
    the previous solution as a warm start.

    Returns the best vector found (still normalised) and its exact
    objective value according to `evaluate_sequence`.
    """
    eps = 1e-12
    y = np.log(a0 + eps)                     # initialise logits

    # λ values – start smooth and increase to a hard max
    for lam in (5.0, 10.0, 20.0, 50.0, 100.0):
        if time.time() >= deadline:
            break

        res = minimize(
            _smooth_max_obj,
            y,
            args=(lam, n, conv_fft_len, corr_fft_len),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iters_per_stage, "disp": False},
        )
        y = res.x

    a_opt = _softmax(y)
    # exact objective (the evaluator is already imported)
    val = float(2 * n * np.max(_full_convolution(a_opt, conv_fft_len)) / (a_opt.sum() ** 2))
    return a_opt, val


# ----------------------------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from one
    index to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.25)   # up‑to 25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects a total wall‑clock budget (default 1000 s) and
    always returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size ---------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ----------------------------------------------------------------------
    # build a modest but diverse pool of start vectors
    pool = []

    # warm start and a few shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with several concentrations
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # edge‑exponential families
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        pool.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # double‑Gaussian (different widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        pool.append(shape)

    # positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    pool.append(shape)

    # uniform
    pool.append(np.full(n, 1.0 / n))

    # sparse equally‑spaced spikes
    for k in (32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # extra random Dirichlet draws with random concentration
    for _ in range(8):
        alpha = np.random.rand() * 4.0 + 0.1   # α∈[0.1,4.1]
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # ----------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ----------------------------------------------------------------------
    # Phase 1 – quick evaluation and selection of a handful of promising seeds
    top_candidates: list[tuple[float, np.ndarray]] = []
    for idx_c, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a0 = np.maximum(cand, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # exact objective (cheap relative to optimisation)
        val = float(2 * n * np.max(_full_convolution(a0, conv_fft_len)) / (a0.sum() ** 2))

        if val < best_val:
            best_val, best_seq = val, a0.copy()
            print(f"[Init cand {idx_c}] new best {val:.6f}")

        top_candidates.append((val, a0.copy()))

    # keep only the ten best starters
    top_candidates.sort(key=lambda x: x[0])
    top_candidates = top_candidates[:10]

    # ----------------------------------------------------------------------
    # Phase 2 – annealed L‑BFGS refinement of the elite candidates
    for rank, (init_val, a_start) in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        local_deadline = min(deadline - 5.0, time.time() + (deadline - time.time()) * 0.4)
        a_opt, val_opt = _annealed_lbfgs_optimize(
            a_start, n, conv_fft_len, corr_fft_len, local_deadline
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[L‑BFGS #{rank}] improved best to {val_opt:.6f}")

    # ----------------------------------------------------------------------
    # Phase 3 – cheap random hill‑climbing to polish the best found vector
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_loc, val_loc = _local_random_search(
        best_seq, n, conv_fft_len, deadline, max_moves=120_000
    )
    if val_loc < best_val:
        best_val, best_seq = val_loc, a_loc.copy()
        print(f"[Final local] improved best to {val_loc:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – ensure a valid, non‑empty return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
