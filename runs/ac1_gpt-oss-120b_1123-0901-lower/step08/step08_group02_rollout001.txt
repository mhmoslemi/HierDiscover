"""
propose_candidate – a multi‑phase optimizer for the step‑function
objective  F(a)=2·n·max(a*a)/(sum(a)²).

The search proceeds as follows:

1. **Diverse initial pool** – a mixture of symmetric, left‑only and
   right‑only exponential / power‑law families, double‑Gaussian bumps,
   Dirichlet draws with random concentration and sparse “spike”
   patterns.  Every vector is projected onto the simplex (sum=1).

2. **Elite selection** – the best `K` candidates (lowest objective) are
   kept for further processing.

3. **Convex‑combination refinement** – all pairs of elite vectors are
   line‑searched over the mixing weight  t∈[0,1] (grid of 21 points).
   Mixing can lower the L∞‑norm of the convolution because the
   convolution is quadratic and the maximum is a convex functional.
   New promising mixtures replace the worst elite members.

4. **Gradient‑based polishing** – each elite member is improved by
   *Adam* on the smooth‑max surrogate (log‑sum‑exp) followed by a
   sub‑gradient (exponentiated‑gradient) phase that directly targets the
   true max.  Both phases respect the simplex through a soft‑max
   parametrisation.

5. **Local stochastic hill‑climbing** – from the current global best a
   cheap random‑move local search (mass transfer between two positions)
   is performed, accepting improving moves.  Occasionally an
   uphill move is accepted according to a simple simulated‑annealing
   schedule to escape shallow minima.

All phases are time‑boxed using the global `deadline`.  The routine
always returns the best feasible sequence found before the 1000 s
budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _obj_value(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/(sum(a)²)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    # protect against tiny sum – should never happen for valid candidates
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _smooth_max(a: np.ndarray, n: int, fft_len: int, lam: float):
    """Smooth approximation of max using log‑sum‑exp with temperature lam."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    # avoid overflow
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    smooth = max_c + np.log(w_sum) / lam
    return smooth, conv, w


def _adam_optimize(a0: np.ndarray, n: int, fft_len: int,
                   deadline: float, max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns the best vector
    encountered (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.02            # start temperature (soft)
    lam_factor = 1.06     # slower annealing than the previous implementation

    best_a = a0.copy()
    best_val = _obj_value(a0, n, fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        val = _obj_value(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate
        smooth, conv, w = _smooth_max(a, n, fft_len, lam)

        # gradient of the surrogate w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 5e6)

        # occasional tiny random shake near deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle decay of learning‑rate every 100k iters
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, fft_len: int,
                         deadline: float, max_iters: int = 250_000,
                         eta0: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _obj_value(a, n, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray, n: int, fft_len: int,
                         deadline: float, max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of the mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    best_val = _obj_value(a, n, fft_len)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.25)  # up to 25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _obj_value(a2, n, fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1
    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (including asymmetric families)."""
    cand = []

    # warm start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # random Dirichlet with several concentration parameters
    for alpha in (0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # left‑only exponential families
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        left /= left.sum()
        cand.append(left)

    # right‑only exponential families (mirror of left)
    for gamma in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        right = np.exp(-gamma * np.arange(n, dtype=np.float64))[::-1]
        right /= right.sum()
        cand.append(right)

    # power‑law decreasing (left) and its mirror (right)
    idx = np.arange(1, n + 1, dtype=np.float64)
    for alpha in (0.3, 0.5, 0.8, 1.2, 1.6):
        pl = idx ** (-alpha)
        pl /= pl.sum()
        cand.append(pl)
        cand.append(pl[::-1])

    # double‑Gaussian bumps (potentially asymmetric)
    i = np.arange(n, dtype=np.float64)
    for sigma1, sigma2, shift in ((0.04, 0.08, 0.0),
                                 (0.06, 0.03, 0.3),
                                 (0.07, 0.07, 0.6)):
        mu1 = shift * (n - 1)
        mu2 = (1 - shift) * (n - 1)
        g1 = np.exp(-0.5 * ((i - mu1) / (sigma1 * n)) ** 2)
        g2 = np.exp(-0.5 * ((i - mu2) / (sigma2 * n)) ** 2)
        mix = g1 + g2
        mix /= mix.sum()
        cand.append(mix)

    # half‑wave sinusoid (asymmetry by clipping the second half)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with a few equally spaced spikes."""
    out = []
    for k in (8, 16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²).

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Phase 0 – build a pool of diverse seeds
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # add a few extra Dirichlet draws with random concentration
    for _ in range(6):
        alpha = np.random.rand() * 4.0 + 0.1        # α ∈ [0.1,4.1]
        pool.append(np.random.dirichlet(np.full(n, alpha)))

    # ------------------------------------------------------------------
    # quick evaluation of the pool, keep top K elite members
    K = 12                     # size of elite set
    elite = []                 # (value, vector) tuples

    for idx, vec in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(vec, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _obj_value(a, n, fft_len)
        elite.append((val, a))
        if idx % 20 == 0:
            print(f"[Init] candidate {idx:04d} val={val:.6f}")

    elite.sort(key=lambda x: x[0])
    elite = elite[:K]
    best_val, best_seq = elite[0]

    print(f"[Init] best of pool = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – convex‑combination mixing of elites
    mix_grid = np.linspace(0.0, 1.0, 21)      # coarse grid for t

    while time.time() < deadline - 60.0:
        improved = False
        # try all unordered pairs (including a pair where both are the same)
        for i in range(len(elite)):
            for j in range(i, len(elite)):
                a_i = elite[i][1]
                a_j = elite[j][1]
                for t in mix_grid:
                    a_mix = (1 - t) * a_i + t * a_j
                    a_mix = np.maximum(a_mix, 0.0)
                    if a_mix.sum() < 1e-12:
                        continue
                    a_mix /= a_mix.sum()
                    val = _obj_value(a_mix, n, fft_len)
                    if val < best_val - 1e-8:
                        best_val, best_seq = val, a_mix.copy()
                        improved = True
                        print(f"[Mix] new best {val:.6f} (pair {i},{j} t={t:.2f})")
        if not improved:
            break
        # after mixing, refresh elite list (keep best K)
        elite.append((best_val, best_seq.copy()))
        elite.sort(key=lambda x: x[0])
        elite = elite[:K]

    # ------------------------------------------------------------------
    # Phase 2 – deep gradient polishing of elite members
    # allocate remaining time roughly equally among elites
    remaining = deadline - time.time()
    per_elite = remaining / max(1, len(elite))

    for idx, (val0, a0) in enumerate(elite):
        if time.time() >= deadline - 10.0:
            break
        sub_deadline = min(deadline - 5.0, time.time() + per_elite * 0.6)
        a_sub, val_sub = _subgradient_optimize(a0, n, fft_len,
                                                sub_deadline,
                                                max_iters=200_000,
                                                eta0=0.9)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Elite {idx}] Subgrad improved to {val_sub:.6f}")

        # short Adam fine‑tuning
        adam_deadline = min(deadline - 5.0, time.time() + per_elite * 0.3)
        a_adam, val_adam = _adam_optimize(a_sub, n, fft_len,
                                          adam_deadline,
                                          max_iters=120_000)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Elite {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climber from the current best
    if time.time() < deadline - 5.0:
        lc_deadline = deadline - 2.0
        a_loc, val_loc = _local_random_search(best_seq, n, fft_len,
                                              lc_deadline,
                                              max_moves=100_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] final improvement to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
