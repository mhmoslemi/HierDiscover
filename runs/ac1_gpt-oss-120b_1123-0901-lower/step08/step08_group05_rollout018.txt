"""
Propose a step‑function (non‑negative list) that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)²)

where a∗a is the linear convolution of a with itself.
The routine builds a diverse pool of seeds (analytic shapes,
warm‑start, random Dirichlet), selects the best, then performs a
long continuous optimisation (Adam on a smooth‑max surrogate,
exact‑max sub‑gradient) followed by an aggressive stochastic
hill‑climbing phase.  If time permits the solution is interpolated
to a finer grid (8192 points) and refined again.  All phases respect
the overall budget (default 1000 s) and the best sequence seen is
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) (the gradient w.r.t. a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (assumes a≥0, sum(a)=1)
def _objective(a: np.ndarray) -> float:
    """2·n·max(a*a) for a sum‑normalised vector."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Shape generators (all return sum‑normalised vectors)
def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ exp(-p·i)."""
    vals = np.exp(-p * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _gaussian_edges_shape(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two Gaussians placed at both ends."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals /= vals.sum()
    return vals


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Ordering heuristics
def _anti_pair_order(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest ordering (expects descending input)."""
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = vals[:half]                 # biggest half left‑hand side
    a[-half:] = vals[-half:][::-1]         # smallest half right‑hand side reversed
    if n % 2 == 1:
        a[half] = vals[half]
    return a


def _alternating_order(vals: np.ndarray) -> np.ndarray:
    """Interleave large / small values (expects descending input)."""
    n = len(vals)
    perm = np.empty(n, dtype=int)
    left, right, idx = 0, n // 2, 0
    while left < n // 2 and right < n:
        perm[idx] = left
        perm[idx + 1] = right
        left += 1
        right += 1
        idx += 2
    if n % 2 == 1:
        perm[-1] = n // 2
    a = np.empty_like(vals)
    a[perm] = vals
    return a


def _apply_order(shape_vals: np.ndarray, order_type: str) -> np.ndarray:
    """Re‑order `shape_vals` (which must be sorted descending)."""
    if order_type == "anti_pair":
        return _anti_pair_order(shape_vals)
    if order_type == "alternating":
        return _alternating_order(shape_vals)
    # identity fallback
    return shape_vals.copy()

# Warm‑start handling
def _warm_start(n: int) -> np.ndarray | None:
    """If a global `height_sequence_1` exists, adapt it to size n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 1e-12:
        return None
    return warm / s

# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
    lr: float = 0.09,
    lam_start: float = 1.0,
    lam_factor: float = 1.03,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate of max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex projection via softmax
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (tracking only)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp surrogate)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)          # chain rule through softmax

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # raise temperature slowly
        lam = min(lam * lam_factor, 1e7)

        # small jitter every few thousand steps to escape plateaus
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
    eta0: float = 0.75,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient refinement using an exponentiated‑gradient update."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {best_val:.6f}")

        # sub‑gradient of the active max entry (or a uniform mix if tie)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (guarantees positivity)
        a = a * np.exp(-eta * grad_a)

        # projection onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing (swap & mass‑transfer moves).
    Accepts downhill moves always; uphill moves with a linearly cooling temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.03
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        if np.random.rand() < 0.55:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.5)   # up to 50 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # annealing acceptance
            temp = max(0.0, start_temp * (1.0 - (it / max_moves)))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 10_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²).  The routine respects the time budget
    and returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5     # small safety cushion

    # ------------------------------------------------------------------
    # 1) Choose resolution (keep 4096 for safety – 8192 can be used later)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 2) Build a rich pool of starter vectors
    start_vectors = []

    # warm start (if a previous solution exists)
    warm = _warm_start(n)
    if warm is not None:
        start_vectors.append(warm.copy())

    # analytic families (dense sweep of parameters)
    shape_fns = [
        ("power", _powerlaw_shape, np.linspace(0.0, 2.5, 21)),
        ("exp", _exponential_shape, np.linspace(0.0, 2.5, 21)),
        ("gauss", _gaussian_edges_shape, [0.2, 0.5, 1.0, 2.0]),
    ]
    orderings = ["anti_pair", "alternating", "identity"]
    for name, fn, params in shape_fns:
        for p in params:
            if name == "gauss":
                vals = fn(n, gamma=p)
            else:
                vals = fn(n, p)
            # ensure descending for anti‑pair / alternating heuristics
            vals_desc = np.sort(vals)[::-1]
            for order in orderings:
                if order == "identity":
                    cand = vals_desc.copy()
                else:
                    cand = _apply_order(vals_desc, order)
                start_vectors.append(cand)

    # uniform baseline
    start_vectors.append(_uniform_shape(n))

    # a handful of random Dirichlet draws with various concentrations
    for alpha in (0.1, 0.3, 0.5, 0.8, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(alpha * np.ones(n))
        cand = np.sort(cand)[::-1]          # sort descending for ordering tricks
        for order in ["identity", "anti_pair", "alternating"]:
            if order == "identity":
                start_vectors.append(cand.copy())
            else:
                start_vectors.append(_apply_order(cand, order))

    # ------------------------------------------------------------------
    # 3) Quick evaluation of the pool, keep the best starter
    best_seq = None
    best_val = float("inf")
    print("[Init] evaluating seed pool...")
    for idx, cand in enumerate(start_vectors):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:          # safeguard against numerical accidents
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand /= s               # normalise to sum = 1
        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
        if idx % 50 == 0:
            print(f"[Init] candidate {idx:03d} val {val:.6f} (best {best_val:.6f})")
    print(f"[Init] best starter value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Continuous optimisation phases (Adam → Sub‑gradient)
    time_left = deadline - time.time()
    # allocate ~45 % to Adam, ~30 % to sub‑gradient, remainder to local search/upscale
    adam_deadline = time.time() + max(30.0, 0.45 * time_left)
    subgrad_deadline = time.time() + max(30.0, 0.30 * (deadline - adam_deadline))

    # ---- Adam (smooth‑max) -------------------------------------------------
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=1_200_000,
        lr=0.09,
        lam_start=1.0,
        lam_factor=1.03,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient refinement --------------------------------------------
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=1_200_000,
        eta0=0.75,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Stochastic local search (swap / transfer moves)
    time_left = deadline - time.time()
    if time_left > 20.0:
        local_deadline = time.time() + max(30.0, 0.20 * time_left)
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=500_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional upscale to a finer grid (8192 points) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big_sum = a_big.sum()
        if a_big_sum < 1e-12:
            a_big = np.full(n_big, 1.0 / n_big, dtype=np.float64)
        else:
            a_big /= a_big_sum

        # brief but deeper Adam on the larger grid
        remaining = deadline - time.time()
        adam_big_deadline = time.time() + min(150.0, 0.5 * remaining)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_big_deadline,
            max_iters=800_000,
            lr=0.09,
            lam_start=1.0,
            lam_factor=1.03,
        )
        if val_big_adam < best_val:
            best_val = val_big_adam
            best_seq = a_big_adam.copy()
            print(f"[Upscale‑Adam] improved to {val_big_adam:.6f}")

        # sub‑gradient refinement on the larger grid
        remaining = deadline - time.time()
        sub_big_deadline = time.time() + min(150.0, remaining)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=800_000,
            eta0=0.75,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Upscale‑Sub] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 7) Final safety and conversion
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's accepted range (should already be satisfied)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```

This implementation expands the original search space (more shape parameters, random Dirichlet seeds, extra ordering tricks), dramatically lengthens the Adam and sub‑gradient phases (up to ~1 M iterations each), and strengthens the stochastic hill‑climbing move set.  
If the overall budget allows, the solution is finally interpolated to 8192 points and refined again on the finer grid, which often yields a further reduction of the evaluation score.  The function always returns a non‑negative list that respects the evaluator’s constraints.
