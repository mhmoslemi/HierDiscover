def adam_optimize(a0, n, conv_fft_len, deadline, max_iters=150000):
    logits = np.log(a0 + 1e-12)  # shape n
    m = np.zeros(n)
    v = np.zeros(n)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr=0.05
    lam=0.05
    lam_factor=1.05
    best_a = a0.copy()
    best_val = objective(a0, n, conv_fft_len)
    it=0
    while time.time()<deadline and it<max_iters:
        # compute probabilities
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # compute objective
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val; best_a = a.copy()
        # smooth max weights via log-sum-exp
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum
        # cross-correlation gradient:
        grad = 2.0 * cross_correlation(w, a, corr_fft_len)
        # gradient w.r.t logits via softmax: grad_log = a * (grad - np.dot(a, grad))
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)
        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**(it+1))
        v_hat = v/(1-beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        logits -= step
        # update lam
        lam = min(lam * lam_factor, 1e6)
        # decay lr occasionally
        if it % 50000 == 0 and it>0:
            lr *= 0.9
        it+=1
    return best_a, best_val

def subgradient_optimize(a0, n, conv_fft_len, deadline, max_iters=200000, eta0=1.0):
    a = a0.copy()
    a = np.maximum(a, 0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s
    best_a = a.copy()
    best_val = objective(a, n, conv_fft_len)
    it=0
    while time.time() < deadline and it < max_iters:
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val; best_a = a.copy()
        # active set for max
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        cnt = np.sum(active)
        if cnt>0:
            w[active] = 1.0 / cnt
        grad = 2.0 * cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it+1.0)
        # exponentiated gradient update
        a = a * np.exp(-eta * grad)
        # renorm
        a_sum = a.sum()
        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n)); a_sum = a.sum()
        a = a / a_sum
        it += 1
    return best_a, best_val

def anneal_optimize(a0, n, conv_fft_len, deadline, max_moves=50000):
    a = a0.copy()
    best_a = a.copy()
    best_val = objective(a, n, conv_fft_len)
    val = best_val
    T_start = 0.1
    T_end = 1e-5
    moves = 0
    while time.time()<deadline and moves<max_moves:
        # linear temperature schedule
        T = max(T_end, T_start * (1 - moves / max_moves))
        # random move
        i = np.random.randint(n)
        if a[i] < 1e-12:
            moves+=1; continue
        j = np.random.randint(n)
        while j == i:
            j = np.random.randint(n)
        # choose fraction
        frac = np.random.rand() * 0.5
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = objective(a2, n, conv_fft_len)
        if val2 < best_val:
            best_val = val2; best_a = a2.copy(); a = a2; val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val)/T):
                a = a2; val = val2
        moves+=1
    return best_a, best_val

def generate_initial_population(n, pop_size, warm):
    pool = []
    if warm is not None:
        # ensure length n
        w = warm
        if w.shape[0] != n:
            # pad or truncate
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0)
        if w.sum() > 0:
            w = w / w.sum()
            pool.append(w.copy())
    # uniform
    pool.append(np.full(n, 1.0/n))
    # sin wave
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0); sin /= sin.sum(); pool.append(sin)
    # exponentials
    for gamma in [0.5, 1.0, 2.0]:
        exp = np.exp(-gamma * np.arange(n))
        exp /= exp.sum()
        pool.append(exp)
        # reversed
        exp_rev = exp[::-1]
        pool.append(exp_rev)
    # Dirichlet with random concentration
    for _ in range(pop_size // 4):
        alpha = 10**np.random.uniform(-2, 1)  # from 0.01 to 10
        dirich = np.random.dirichlet(alpha * np.ones(n))
        pool.append(dirich)
    # random spikes
    for _ in range(pop_size // 4):
        k = np.random.choice([8,16,32,64,128])
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)
    # ensure each is normalized to sum=1
    normalized = []
    for a in pool:
        s = a.sum()
        if s < 1e-12:
            continue
        normalized.append(a / s)
    # trim to pop_size
    if len(normalized) > pop_size:
        normalized = normalized[:pop_size]
    return normalized

Step‑function optimizer (≈1 000 s budget)

- Starts with a diverse population of candidate step‑functions (Dirichlet,
  exponential edges, sinusoids, sparse spikes, warm‑start if available).
- Evolves the population with a simple evolutionary loop:
    * crossover of two elite parents,
    * random mass‑transfer mutation,
    * short exponentiated‑gradient (sub‑gradient) refinement,
    * replacement of the worst individual.
- Keeps the globally best vector and periodically prints its value.
- After the exploration phase the best solution is up‑sampled to the final
  length (4096) and refined in three stages:
    1. Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate,
    2. Exact‑max sub‑gradient (exponentiated‑gradient) optimisation,
    3. Simulated‑annealing style local search.
- The whole routine respects the time budget and always returns a
  non‑negative list of floats (clipped to the evaluator limits).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of 2·n·max(a∗a)/(∑a)² (matches evaluate_sequence)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:          # same guard as evaluate_sequence
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp smooth‑max surrogate.
    Returns the best vector seen and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.05                # temperature for the smooth‑max
    lam_factor = 1.04

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- get probabilities from logits -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate exact objective -----
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ----- gradient of the surrogate (through convolution) -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- mirror‑descent (soft‑max) chain rule -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- schedules -----
        lam = min(lam * lam_factor, 1e7)
        if it % 60_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase (exponentiated‑gradient mirror descent).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (keeps a on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search on the simplex.
    Randomly moves a fraction of mass between two positions.
    """
    a = a0.copy()
    val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    T_start = 0.09
    T_end = 1e-5
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 40 %) of the source mass
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _generate_initial_population(
    n: int,
    popsize: int,
    warm: np.ndarray | None,
) -> list[np.ndarray]:
    """
    Produce a modest but diverse pool of start vectors.
    """
    candidates = []

    # warm‑start (if supplied)
    if warm is not None:
        w = warm
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        s = w.sum()
        if s > 0.0:
            candidates.append(w / s)

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # half‑wave sinusoid
    sin = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin = np.maximum(sin, 0.0)
    sin /= sin.sum()
    candidates.append(sin)

    # exponential edges (left & right)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n))
        left /= left.sum()
        candidates.append(left)
        candidates.append(left[::-1])

    # random Dirichlet with varied concentration
    for _ in range(popsize // 4):
        alpha = 10 ** np.random.uniform(-2, 1)           # 0.01 … 10
        d = np.random.dirichlet(alpha * np.ones(n))
        candidates.append(d)

    # sparse spikes
    for _ in range(popsize // 4):
        k = np.random.choice([8, 16, 32, 64, 128, 256])
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        candidates.append(a)

    # trim / normalise
    uniq = []
    for a in candidates:
        s = a.sum()
        if s < 1e-12:
            continue
        uniq.append(a / s)
        if len(uniq) >= popsize:
            break
    return uniq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point used by the evaluation harness.
    Searches for a non‑negative step‑function with a low objective value.
    """
    np.random.seed(seed)
    margin = 0.5                     # safety margin so we do not overshoot
    deadline = time.time() + budget_s - margin

    # ------------------------------------------------------------------
    # Phase 0 – small‑scale evolutionary exploration (n = 2048)
    # ------------------------------------------------------------------
    n_small = 2048
    conv_fft_len_small = _next_pow2(2 * n_small - 1)
    corr_fft_len_small = _next_pow2(3 * n_small - 2)

    warm = globals().get("height_sequence_1")
    warm_np = None
    if isinstance(warm, (list, np.ndarray)):
        warm_np = np.asarray(warm, dtype=np.float64)

    popsize = 30
    population = _generate_initial_population(n_small, popsize, warm_np)

    # evaluate initial pool
    best_seq = None
    best_val = float("inf")
    for a in population:
        val = _objective(a, n_small, conv_fft_len_small)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"[Init] Best initial value {best_val:.6f}")

    # evolutionary loop – runs until ~70 % of the time budget is spent
    explore_deadline = time.time() + (deadline - time.time()) * 0.70
    it = 0
    top_k = 5                     # number of elites used for crossover

    while time.time() < explore_deadline:
        # ---- selection of elite parents ----
        scores = [_objective(a, n_small, conv_fft_len_small) for a in population]
        elite_idx = np.argsort(scores)[:top_k]
        p1, p2 = population[np.random.choice(elite_idx, size=2, replace=False)]

        # ---- simple crossover ----
        alpha = np.random.rand()
        child = alpha * p1 + (1.0 - alpha) * p2

        # ---- mutation: random mass transfers ----
        for _ in range(np.random.randint(1, 4)):
            i = np.random.randint(n_small)
            if child[i] < 1e-14:
                continue
            j = np.random.randint(n_small)
            if i == j:
                continue
            frac = np.random.rand() * 0.5
            delta = child[i] * frac
            child[i] -= delta
            child[j] += delta

        child = np.maximum(child, 0.0)
        s = child.sum()
        if s < 1e-12:
            child = np.random.dirichlet(np.ones(n_small))
        else:
            child /= s

        # ---- short sub‑gradient refinement (≈2 s) ----
        sub_dead = min(explore_deadline, time.time() + 2.0)
        child_ref, child_val = _subgradient_optimize(
            child,
            n_small,
            conv_fft_len_small,
            corr_fft_len_small,
            sub_dead,
            max_iters=25_000,
            eta0=0.7,
        )

        # replace the worst individual if the child is better
        worst_idx = np.argmax(scores)
        if child_val < scores[worst_idx]:
            population[worst_idx] = child_ref

        # update global best
        if child_val < best_val:
            best_val = child_val
            best_seq = child_ref.copy()
            print(f"[Evo it={it:05d}] New global best {best_val:.6f}")

        it += 1

        # occasional random injection to maintain diversity
        if it % 120 == 0 and time.time() < explore_deadline:
            population.append(np.random.dirichlet(np.ones(n_small)))
            if len(population) > popsize:
                # drop the current worst
                scores = [_objective(a, n_small, conv_fft_len_small) for a in population]
                population.pop(np.argmax(scores))

    # ------------------------------------------------------------------
    # Phase 1 – upscale to final size and deep refinement (n = 4096)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n_small, 1.0 / n_small)

    n_big = 4096
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # linear interpolation to the bigger grid
    x_small = np.linspace(0, 1, n_small, endpoint=False)
    x_big = np.linspace(0, 1, n_big, endpoint=False)
    best_big = np.interp(x_big, x_small, best_seq)
    best_big = np.maximum(best_big, 0.0)
    s_big = best_big.sum()
    if s_big < 1e-12:
        best_big = np.full(n_big, 1.0 / n_big)
    else:
        best_big = best_big / s_big

    # ----- Adam (smooth‑max) -----
    if time.time() < deadline - 30:
        adam_dead = min(deadline - 10, time.time() + 12.0)   # leave time for later phases
        a_adam, val_adam = _adam_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_dead,
            max_iters=180_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_big = a_adam.copy()
            print(f"[Final Adam] improved best {best_val:.6f}")

    # ----- Exact‑max sub‑gradient -----
    if time.time() < deadline - 5:
        sub_dead = deadline - 5
        a_sub, val_sub = _subgradient_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead,
            max_iters=350_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_big = a_sub.copy()
            print(f"[Final Subgrad] improved best {best_val:.6f}")

    # ----- Simulated annealing local moves -----
    if time.time() < deadline - 1:
        a_ann, val_ann = _anneal_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            deadline,
            max_moves=100_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_big = a_ann.copy()
            print(f"[Final Anneal] improved best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return a valid list (non‑negative, ≤1000)
    # ------------------------------------------------------------------
    best_big = np.clip(best_big, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_big)})")
    return best_big.tolist()
