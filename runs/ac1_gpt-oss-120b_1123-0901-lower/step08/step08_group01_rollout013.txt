"""
Hybrid random‑start + long Adam + PGD + cheap hill‑climber.

* Phase 1 – generate a large, diverse set of candidates (Dirichlet,
  edge‑exponential / power‑law, beta, double‑Gaussian, sparse spikes,
  classic windows – Hann, Hamming, Blackman, Kaiser).  Each candidate is
  evaluated by the exact objective (FFT‑based convolution) and the best
  one is kept.

* Phase 2 – deep optimisation of the current best vector with a
  soft‑max (log‑sum‑exp) surrogate and Adam.  The temperature (lam) starts
  very low and is increased slowly so that the optimiser first smoothes
  the objective and then focuses on the true max‑convolution.  The loop
  runs until the global deadline, keeping the best exact value seen.

* Phase 3 – Euclidean projected‑gradient descent on the true max‑convolution
  (sub‑gradient at the active set).  A decreasing stepsize and projection
  onto the simplex keep the iterate feasible.

* Phase 4 – a cheap random‑move local search for the last seconds.

The routine always returns a feasible non‑negative list; it respects the
budget (default 1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a∗a)/(sum(a)²).  a is assumed non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_c / (s ** 2))

# ------------------- FAMILY GENERATORS ---------------------------------
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    w /= w.sum()
    return w


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    w /= w.sum()
    return w


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    w = np.kaiser(n, beta)
    w = w.astype(np.float64)
    w /= w.sum()
    return w


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

# ------------------- OPTIMISERS ---------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 2_000_000,
                   lr_start: float = 0.06) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns the best
    vector (sum≈1) found and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # start on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.001           # start with almost flat soft‑max
    lam_factor = 1.03
    lam_max = 1e7

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- simplex from logits (soft‑max) ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (soft‑max over convolution) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain‑rule through soft‑max (mirror‑descent form) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_max)

        # tiny random perturbation near the end
        if it % 5000 == 0 and time.time() > deadline - 30.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  corr_fft_len: int,
                  deadline: float,
                  max_iters: int = 500_000,
                  eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active‑set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a

        # Euclidean projection onto simplex
        # ---- simple O(n) projection ----
        u = np.sort(a)[::-1]
        cssv = np.cumsum(u) - 1
        rho = np.where(u - cssv / (np.arange(1, n + 1)) > 0)[0]
        if rho.size == 0:
            theta = 0.0
        else:
            rho = rho[-1]
            theta = cssv[rho] / (rho + 1)
        a = np.maximum(a - theta, 0.0)

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber that moves a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    rng = np.random.default_rng()

    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (rng.random() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises   2·n·max(a∗a) / (sum(a)²).

    The routine respects the time budget and always returns the best
    sequence discovered.  It combines a wide random initialisation with a
    long Adam optimisation, a PGD refinement and a final cheap hill‑climber.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # -------- problem dimensions ------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -------- warm start handling -----------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # -------- Phase 1 – diverse quick candidates --------------------
    best_seq = None
    best_val = float("inf")
    phase1_deadline = min(deadline - 30.0, time.time() + 0.20 * (deadline - time.time()))

    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "dirichlet", "sparse", "hann", "hamming", "blackman", "kaiser"
    ]

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = np.exp(rng.uniform(np.log(0.01), np.log(20.0)))
            cand = _edge_exponential(n, gamma)
        elif fam == "edge_pow":
            alpha = rng.uniform(-3.5, 1.0)
            cand = _edge_power_law(n, alpha)
        elif fam == "beta":
            a = rng.uniform(0.2, 10.0)
            b = rng.uniform(0.2, 10.0)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.6)
            cand = _double_gaussian(n, sigma)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.05, 5.0)
            cand = rng.dirichlet(np.full(n, alpha))
        elif fam == "sparse":
            k = rng.choice([16, 32, 64, 128, 256, 512, 1024, 2048])
            cand = _sparse_spikes(n, k, rng)
        elif fam == "hann":
            cand = _hann_window(n)
        elif fam == "hamming":
            cand = _hamming_window(n)
        elif fam == "blackman":
            cand = _blackman_window(n)
        elif fam == "kaiser":
            beta = rng.uniform(0.0, 12.0)
            cand = _kaiser_window(n, beta)
        else:
            cand = _uniform(n)

        # incorporate warm start with a small probability
        if warm is not None and rng.random() < 0.05:
            cand = 0.5 * cand + 0.5 * warm

        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-8:
            cand = np.full(n, 1.0 / n)
        cand = cand / cand.sum()

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase1] new best {val:.6f}")

    # -------- Phase 2 – long Adam optimisation ----------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # Give Adam the bulk of the remaining time
    adam_deadline = min(deadline - 15.0, time.time() + 0.55 * (deadline - time.time()))
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=5_000_000,
        lr_start=0.06,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam‑global] best {val_adam:.6f}")

    # -------- Phase 3 – PGD refinement -------------------------------
    if time.time() < deadline - 10.0:
        pgd_deadline = deadline - 10.0
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=400_000,
            eta0=0.4,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD‑refine] new best {val_pgd:.6f}")

    # -------- Phase 4 – cheap local moves (last seconds) ----------
    if time.time() < deadline - 1.0:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local‑final] improved best {val_loc:.6f}")

    # -----------------------------------------------------------------
    # Final safety: clip to evaluator's allowed domain
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
