"""
propose_candidate
=================

A multi‑restart, multi‑stage optimizer for the non‑negative step‑function
problem.  The goal is to minimise

    F(a) = 2·n·max_k (a∗a)[k]          with   a_i ≥ 0,  Σ a_i = 1

(the denominator of the original evaluator becomes 1 after fixing the
ℓ₁‑norm).

The algorithm works in four phases:

1. **Warm‑start & diverse initialisation** – reuse a previously found
   solution (``height_sequence_1``) if it exists, add a few analytic
   shapes and many random Dirichlet draws.

2. **Exploratory Adam runs** – for each start run a few hundred Adam
   iterations on the smooth‑max surrogate (log‑sum‑exp).  The best few
   candidates are kept.

3. **Long deterministic refinement** – on the selected candidates run
   a long Adam phase (with a slowly increasing temperature) followed by
   an exact‑max sub‑gradient phase and a directed local‑move phase that
   transfers mass from indices with a large positive sub‑gradient to
   indices with a large negative sub‑gradient.

4. **Optional up‑sampling** – if time permits the best solution is
   interpolated to a finer grid (8192 points) and the refinement phases
   are repeated.

All steps respect the global time budget (``budget_s``) and the final
sequence is clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (assumes sum(a)=1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) where a≥0 and sum(a)=1."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Shape generators (all produce a non‑negative vector with sum=1)
def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ exp(-p·i)."""
    vals = np.exp(-p * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _gaussian_edges_shape(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two Gaussians placed at both ends."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals /= vals.sum()
    return vals


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Ordering heuristics (apply to a *descending* vector)
def _anti_pair_order(vals: np.ndarray) -> np.ndarray:
    """Largest ↔ smallest ordering."""
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = vals[:half]                     # biggest half on the left
    a[-half:] = vals[-half:][::-1]             # smallest half on the right, reversed
    if n % 2 == 1:
        a[half] = vals[half]                   # median stays in centre
    return a


def _alternating_order(vals: np.ndarray) -> np.ndarray:
    """Interleave large and small values."""
    n = len(vals)
    perm = np.empty(n, dtype=int)
    left = 0
    right = n // 2
    idx = 0
    while left < n // 2 and right < n:
        perm[idx] = left
        perm[idx + 1] = right
        left += 1
        right += 1
        idx += 2
    if n % 2 == 1:
        perm[-1] = n // 2
    a = np.empty_like(vals)
    a[perm] = vals
    return a


def _apply_order(shape_vals: np.ndarray, order_type: str) -> np.ndarray:
    """Return a reordered version of `shape_vals`."""
    # shape_vals is descending by construction
    if order_type == "anti_pair":
        return _anti_pair_order(shape_vals)
    elif order_type == "alternating":
        return _alternating_order(shape_vals)
    else:
        return shape_vals.copy()


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int) -> np.ndarray | None:
    """If a previous solution (global var `height_sequence_1`) exists, adapt it."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 1e-12:
        return None
    warm = warm / warm.sum()
    return warm


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.1,
    lam_factor: float = 1.03,
    jitter: bool = True,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best exact objective encountered.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # build simplex point from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (track)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # center the gradient (project onto simplex tangent)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape plateaus
        if jitter and it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient refinement using an exponentiated‑gradient update.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # sub‑gradient for active max entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _directed_local_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Gradient‑guided local moves.
    At each step the sub‑gradient of the current max entry is computed;
    mass is transferred from an index with large positive gradient to one
    with large negative gradient.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] new best {best_val:.6f}")

        # sub‑gradient for the max entry:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, _next_pow2(3 * n - 2))

        # indices with strongest positive / negative gradient
        donor = np.argmax(grad)            # moving mass away from this index reduces max
        receiver = np.argmin(grad)         # moving mass towards this index reduces max

        if donor == receiver or a[donor] < 1e-14:
            # fall back to a random tiny transfer
            i, j = np.random.randint(n, size=2)
            donor, receiver = i, j

        # transfer a fraction (up to 30 %) of donor's mass
        delta = a[donor] * (np.random.rand() * 0.3)
        a[donor] -= delta
        a[receiver] += delta

        # re‑project onto simplex (should stay on simplex automatically,
        # but numerical drift may appear)
        s = a.sum()
        if s <= 0:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    corr_fft_len = _next_pow2(3 * n - 2)

    if warm is not None:

            start_vectors.append(np.roll(warm, shift))

    start_vectors.append(_uniform_shape(n))

    shape_fns = [

        ("exp",   _exponential_shape, [0.0, 0.05, 0.1, 0.3, 0.6, 1.2, 2.0]),
        ("gauss", _gaussian_edges_shape, [0.2, 0.5, 1.0, 2.0]),
    ]
    orderings = ["anti_pair", "alternating"]

    for name, fn, params in shape_fns:
        for p in params:
            vals = fn(n, p) if name != "gauss" else fn(n, gamma=p)
            # vals are descending for power / exp / gauss
            for order in orderings:
                cand = _apply_order(vals, order)
                start_vectors.append(cand)

    # random Dirichlet seeds with a range of concentration parameters
    for alpha in (0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        cand = np.random.dirichlet(alpha * np.ones(n))
        start_vectors.append(cand)

    # ------------------------------------------------------------------
    # 3) Quick evaluation of the pool; keep the best few (up to 5)
    best_seq = None
    best_val = float("inf")
    top_candidates = []   # (value, vector)
    for idx, cand in enumerate(start_vectors):
        cand = np.maximum(cand, 0.0)
        cand = cand / cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")

        # maintain a small heap of the best candidates
        if len(top_candidates) < 5:
            top_candidates.append((val, cand.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif val < top_candidates[-1][0]:
            top_candidates[-1] = (val, cand.copy())
            top_candidates.sort(key=lambda x: x[0])

        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    print(f"[Init] best starter value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Short exploratory Adam runs on the top candidates
    #    (each gets a few thousand iterations, time‑boxed)
    time_left = deadline - time.time()
    explor_budget = min(120.0, 0.15 * time_left)  # 15 % of remaining budget
    per_cand_deadline = time.time() + explor_budget / len(top_candidates)

    for val, cand in top_candidates:
        if time.time() >= deadline:
            break
        cat_deadline = min(per_cand_deadline, deadline - 0.5)
        a_opt, val_opt = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            cat_deadline,
            max_iters=200_000,
            lr=0.06,
            lam_start=0.1,
            lam_factor=1.04,
            jitter=True,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Explor] improved to {best_val:.6f}")

    # ------------------------------------------------------------------

    # allocate most of the remaining time to Adam → Sub‑gradient → Local
    time_rem = deadline - time.time()
    if time_rem <= 30.0:

        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return best_seq.tolist()

        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=500_000,
        lr=0.07,
        lam_start=0.05,
        lam_factor=1.02,
        jitter=True,
    )
    if val_adam < best_val:

        print(f"[Adam long] improved to {best_val:.6f}")

    # ---- Exact sub‑gradient refinement
    subgrad_deadline = time.time() + max(30.0, 0.30 * (deadline - time.time()))
    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=600_000,
        eta0=1.5,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad long] improved to {best_val:.6f}")

    # ---- Gradient‑guided local search
    loc_deadline = deadline - 5.0  # keep a few seconds safety margin
    a_loc, val_loc = _directed_local_search(
        best_seq,
        n,
        conv_fft_len,
        loc_deadline,
        max_moves=200_000,
    )
    if val_loc < best_val:
        best_val = val_loc
        best_seq = a_loc.copy()
        print(f"[Local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6) Optional up‑scaling to a finer grid if time permits
    time_left = deadline - time.time()
    if time_left > 60.0:
        # upscale to 8192 points (double resolution)
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        print("Upscaling to", n_big, "points and refining...")

        # short Adam on the larger grid
        up_adam_deadline = time.time() + min(90.0, 0.4 * time_left)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            up_adam_deadline,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.05,
            lam_factor=1.02,
            jitter=True,
        )

        # sub‑gradient on large grid
        up_sub_deadline = time.time() + min(90.0, deadline - time.time() - 5.0)
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            up_sub_deadline,
            max_iters=300_000,
            eta0=1.5,
        )

        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Upscaled] improved to {best_val:.6f}")

        # final tiny local refinement on the big grid
        up_loc_deadline = deadline - 2.0
        a_big_loc, val_big_loc = _directed_local_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            up_loc_deadline,
            max_moves=120_000,
        )
        if val_big_loc < best_val:
            best_val = val_big_loc
            best_seq = a_big_loc.copy()
            print(f"[Upscaled local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7) Final safety, clipping and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # store for possible future warm‑starts
    globals()["height_sequence_1"] = best_seq.copy()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
