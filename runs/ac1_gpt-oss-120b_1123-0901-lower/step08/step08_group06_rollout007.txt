```
import heapq

def _balanced_arrangement(vals):
    n = len(vals)
    a = np.zeros(n)
    # priority queue of intervals: store (-length, start, end)
    heap = [(-n, 0, n-1)]
    heapq.heapify(heap)
    for v in vals:
        # pop largest interval
        length, start, end = heapq.heappop(heap)
        length = -length
        mid = (start + end) // 2
        a[mid] = v
        # left interval
        if mid - 1 >= start:
            heapq.heappush(heap, (-(mid - start), start, mid - 1))
        # right interval
        if end >= mid + 1:
            heapq.heappush(heap, (-(end - mid), mid + 1, end))
    return a
```

def _local_weighted_search(a0, n, conv_fft_len, deadline, max_moves=200000):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    # compute initial gradient (subgradient)
    grad = np.zeros_like(a)
    for it in range(max_moves):
        if time.time() >= deadline:
            break
        # compute convolution and active set
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0/active.sum()
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)
        # bias selection of i based on abs(grad)
        prob = np.abs(grad)
        prob_sum = prob.sum()
        if prob_sum > 0:
            prob = prob / prob_sum
        else:
            prob = np.full(n, 1.0/n)
        i = np.random.choice(n, p=prob)
        # try to move some mass from i to a random other index j
        if a[i] <= 1e-15:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        delta = a[i] * (np.random.rand() * 0.5)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # Metropolis accept w/ small temperature
            # temperature schedule: exponential decay
            temp = max(0.01 * (1 - it/max_moves), 1e-4)
            if np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2
        if it % 5000 == 0:
            print(f"[LocalWeighted it={it}] best {best_val:.6f}")
    return best_a, best_val

"""
Propose a non‑negative step‑function (list of heights) that minimises

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The search consists of five stages:

1.  Warm‑start & massive random sampling.
    *  Sample Dirichlet vectors with a wide range of concentration.
    *  For each sample generate several deterministic orderings:
        - anti‑pair (large opposite small)
        - balanced (mid‑point of the largest free interval)
        - modular (i·step mod n, step = n//2+1, co‑prime to n)
        - bucket‑interleaved (interleaving of several buckets)
    *  Keep the best of all candidates.

2.  Smooth‑max L‑BFGS refinement.
    *  Optimise a log‑sum‑exp surrogate of the max‑convolution.
    *  λ (temperature) is increased progressively so the surrogate
      approaches the true max.

3.  Additive projected sub‑gradient descent (very many iterations,
    step size η_t = η₀/√t).  The iterate is projected onto the simplex
    after every step.

4.  Gradient‑biased stochastic hill‑climbing.
    *  At each iteration a gradient of the active (max‑) convolution
      is computed.
    *  Indices with large absolute gradient are preferentially chosen
      for a small mass transfer move; occasional uphill moves are
      accepted via a simple annealing schedule.

5.  Optional up‑sampling to a finer grid (8192 points) followed by a
    quick L‑BFGS + sub‑gradient fine‑tune.

The whole pipeline respects the global time budget (default 1000 s)
and always returns a valid non‑negative list.
"""

import time
import heapq
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, we keep sum = 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling (global variable `height_sequence_1` may be provided)
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise to sum = 1 (objective is scale‑invariant)
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Candidate generators
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]  # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _balanced_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Greedy balanced placement: repeatedly put the next value into the
    centre of the currently longest free interval.
    """
    n = len(vals)
    a = np.zeros(n, dtype=np.float64)
    # heap of intervals, stored as (-length, start, end)
    heap = [(-n, 0, n - 1)]
    heapq.heapify(heap)
    for v in vals:
        # take the longest interval
        length, start, end = heapq.heappop(heap)
        length = -length
        mid = (start + end) // 2
        a[mid] = v
        # left sub‑interval
        if mid - 1 >= start:
            heapq.heappush(heap, (-(mid - start), start, mid - 1))
        # right sub‑interval
        if end >= mid + 1:
            heapq.heappush(heap, (-(end - mid), mid + 1, end))
    return a


def _modular_arrangement(vals: np.ndarray, step: int) -> np.ndarray:
    """Place values at positions (i·step) mod n."""
    n = len(vals)
    a = np.zeros(n, dtype=np.float64)
    for i, v in enumerate(vals):
        pos = (i * step) % n
        a[pos] = v
    return a


def _bucket_interleaved(vals: np.ndarray, buckets: int = 8) -> np.ndarray:
    """
    Interleave values by buckets:
    take indices 0, b, 2b,…; then 1, 1 + b, 1 + 2b,… etc.
    """
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    perm = []
    for b in range(buckets):
        perm.extend(range(b, n, buckets))
    # perm length is exactly n
    for idx, pos in enumerate(perm[:n]):
        a[pos] = vals[idx]
    return a


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)  # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Smooth‑max (log‑sum‑exp) surrogate for L‑BFGS
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Surrogate objective and gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # log‑sum‑exp over (lam·(conv - max_c))
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                       # soft weights, sum = 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)       # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0),
                    max_iter_stage: int = 250) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smooth‑max surrogate, progressively increasing λ.
    Returns the best sequence and its exact objective value.
    """
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # limit iterations by remaining time
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Additive projected sub‑gradient optimisation
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact max‑convolution objective.
    Uses Euclidean projection onto the simplex after each step.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # sub‑gradient of max
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Gradient‑biased stochastic hill‑climbing
def _local_weighted_search(a0: np.ndarray, n: int,
                           conv_fft_len: int, deadline: float,
                           max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Stochastic local search where the choice of a source index i is
    biased by the magnitude of the sub‑gradient of the active convolution.
    Occasional uphill moves are accepted with an annealing temperature.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # recompute gradient only every 200 steps (saves time)
        if it % 200 == 0:
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, conv_fft_len)

            # sampling probabilities proportional to |grad|
            prob = np.abs(grad)
            total = prob.sum()
            if total > 0:
                prob = prob / total
            else:
                prob = np.full(n, 1.0 / n)

        # pick source index with bias
        i = np.random.choice(n, p=prob)
        if a[i] < 1e-15:
            it += 1
            continue
        # pick destination uniformly
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        # transfer a fraction of mass from i to j
        delta = a[i] * (np.random.rand() * 0.7)    # up to 70 % move
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # annealed acceptance of uphill move
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[LocalWeighted it={it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Warm‑start & massive random sampling
    best_seq = None
    best_val = float("inf")

    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    # a few cheap handcrafted starters
    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_seq, best_val = c.copy(), val
            print(f"[Starter] value {val:.6f}")

    # random Dirichlet pool with several deterministic orderings
    pool_deadline = deadline - 200.0  # reserve time for later phases
    step_mod = n // 2 + 1            # co‑prime step for modular arrangement
    while time.time() < pool_deadline:
        # choose a concentration α from a wide log‑uniform range
        log_alpha = np.random.uniform(-3.0, 2.0)   # α ≈ 0.001 … 100
        alpha = 10.0 ** log_alpha
        vals = np.random.dirichlet(alpha * np.ones(n))
        vals.sort()
        vals = vals[::-1]  # descending

        # generate several orderings
        candidates = [
            _anti_pair_arrangement(vals),
            _balanced_arrangement(vals),
            _modular_arrangement(vals, step_mod),
            _bucket_interleaved(vals, buckets=8),
        ]

        for cand in candidates:
            val = _objective(cand)
            if val < best_val:
                best_seq, best_val = cand.copy(), val
                print(f"[Random pool] new best {val:.6f}")

        # limit the pool size a bit to keep runtime in check
        if time.time() > pool_deadline:
            break

    # ------------------------------------------------------------------
    # 2) Smooth‑max L‑BFGS refinement
    if time.time() < deadline - 120.0:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         deadline,
                                         lam_schedule=(5, 15, 45, 150),
                                         max_iter_stage=300)
        if val_lb < best_val:
            best_seq, best_val = seq_lb.copy(), val_lb
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # ------------------------------------------------------------------
    # 3) Additive projected sub‑gradient optimisation
    if time.time() < deadline - 60.0:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          deadline,
                                                          max_iters=800_000,
                                                          eta0=0.6)
        if val_add < best_val:
            best_seq, best_val = seq_add.copy(), val_add
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # ------------------------------------------------------------------
    # 4) Gradient‑biased stochastic local search
    if time.time() < deadline - 20.0:
        seq_loc, val_loc = _local_weighted_search(best_seq, n,
                                                  conv_fft_len,
                                                  deadline,
                                                  max_moves=250_000)
        if val_loc < best_val:
            best_seq, best_val = seq_loc.copy(), val_loc
            print(f"[LocalWeighted] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192 points)
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best sequence
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_big, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # a quick L‑BFGS on the big grid
        seq_big, val_big = _lbfgs_optimize(a_big, n_big,
                                           conv_fft_len_big,
                                           corr_fft_len_big,
                                           deadline,
                                           lam_schedule=(5, 20, 80, 200),
                                           max_iter_stage=150)
        if val_big < best_val:
            best_seq, best_val = seq_big.copy(), val_big
            print(f"[Big‑grid L‑BFGS] improved to {val_big:.6f}")

        # final few sub‑gradient steps on the big grid
        if time.time() < deadline:
            seq_big2, val_big2 = _additive_subgradient_optimize(best_seq, n_big,
                                                               conv_fft_len_big,
                                                               corr_fft_len_big,
                                                               deadline,
                                                               max_iters=400_000,
                                                               eta0=0.5)
            if val_big2 < best_val:
                best_seq, best_val = seq_big2.copy(), val_big2
                print(f"[Big‑grid Subgrad] improved to {val_big2:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
