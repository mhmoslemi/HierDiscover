def _cross_correlation(w, a, fft_len):

    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000, lr_start=0.04):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.03
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    target_sum = np.sqrt(2.0 * n)
    while time.time() < deadline and it < max_iters:
        # compute softmax probabilities a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # normalization
        a *= target_sum / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            # optionally print
        # compute smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # compute gradient of logit (mirror)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        lam = min(lam * lam_factor, 1e7)
        if it % 50000 == 0 and it > 0:
            lr *= 0.95
        it += 1
    return best_a, best_val
```

        w = np.zeros_like(conv)

        eta = eta0 / (np.sqrt(it + 1) + 1e-8)

            a = np.random.dirichlet(np.ones(n)) * target_sum
        else:
            a = a * (target_sum / a_sum)
        it += 1
    return best_a, best_val

def _lp_refine(a, n, conv_fft_len, deadline, max_refine_iter=10, t=0.2):
    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _full_convolution(a, conv_fft_len).max()
    for it in range(max_refine_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(best_a, conv_fft_len)
        M = conv.max()
        active = np.where(conv >= M - 1e-12)[0]  # tolerance
        if active.size == 0:
            break
        # Build A_ub
        m = active.size
        A = np.zeros((m, n), dtype=np.float64)
        for row_idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # Fill columns lo..hi
            # columns index j correspond to a_{k-j} where j in [lo, hi]
            # Set A[row_idx, lo:hi+1] = best_a[k - np.arange(lo, hi+1)]
            idxs = np.arange(lo, hi+1)
            A[row_idx, lo:hi+1] = best_a[k - idxs]
        # Solve LP: minimize -sum g_i => objective c = -1 vector
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b = np.full(m, M, dtype=np.float64)
        # Use scipy.optimize.linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        g = g0 * (target_sum / sum_g0)
        # Mix
        best_a = (1 - t) * best_a + t * g
        # Re-normalize exact
        best_a = np.clip(best_a, 0.0, None)
        # recompute value
        new_val = _full_convolution(best_a, conv_fft_len).max()
        # if not improving, break
        if new_val >= best_val - 1e-12:
            # maybe break or continue with smaller t
            # For safety break
            break
        best_val = new_val
    return best_a, best_val

        from scipy.optimize import linprog
    except ImportError:

    np.random.seed(seed)

"""
Propose a low‑value step‑function (non‑negative heights).

The search builds on the previous optimizer but adds several new ideas:

* A dedicated LP‑refinement phase (active‑set linear programming) that
  mimics the classic “alternating LP” method from the literature.  The LP
  maximises the total mass of a new candidate g subject to the current
  convolution bound f∗f ≤ M, then mixes g back into f.  This step provably
  never increases the objective and often pushes it noticeably lower.

* Longer exponentiated‑gradient (sub‑gradient) and projected‑gradient
  phases with more conservative step‑size schedules, giving the optimiser
  extra time to fine‑tune the distribution.

* A final cheap random‑move hill‑climber for the very last seconds.

All phases respect the global 1000 s deadline and always keep the best
vector seen so far.  The result is returned as a plain Python list.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Helper utilities – FFT‑based convolutions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ---------------------------------------------------------------
# Optimisers ----------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # start on simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # temperature
    lam_factor = 1.02   # raise temperature slowly

    target_sum = np.sqrt(2.0 * n)
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- soft‑max to obtain a on the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        a *= target_sum / a.sum()          # keep exact ℓ1‑norm

        # exact objective = max convolution (since sum² = 2n)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {max_c:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient phase (mirror descent on simplex)."""
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum() * target_sum

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {max_c:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step‑size – slower decay for a finer final search
        eta = eta0 / (np.sqrt(it + 1) + 1e-12)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the ℓ1‑sphere
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n)) * target_sum
        else:
            a = a * (target_sum / s)

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Projected‑gradient descent on the exact max‑convolution."""
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum() * target_sum

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {max_c:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / (np.sqrt(it + 1) + 1e-12)
        a = a - eta * grad_a

        # Euclidean projection onto the simplex and then scale to target_sum
        a = _project_onto_simplex(a / a.sum()) * target_sum

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climber for the final seconds."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # Move a random fraction (up to 30 %) of mass from i to j
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ---------------------------------------------------------------
# LP refinement – active‑set linear programming --------------------
def _lp_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_refine_iter: int = 20,
    t: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Solve a tiny LP: maximise sum(g)  s.t.  a★g ≤ M  (only active rows).
    Then mix g back into a using a convex combination with weight t.
    Repeats while it improves the max‑convolution.
    """
    try:
        from scipy.optimize import linprog
    except Exception:      # SciPy not available – skip this phase
        return a, _full_convolution(a, conv_fft_len).max()

    target_sum = np.sqrt(2.0 * n)
    best_a = a.copy()
    best_val = _full_convolution(best_a, conv_fft_len).max()

    for it in range(max_refine_iter):
        if time.time() >= deadline:
            break
        conv = _full_convolution(best_a, conv_fft_len)
        M = conv.max()
        # indices where the current maximum is attained (tiny tolerance)
        active = np.where(conv >= M - 1e-12)[0]
        if active.size == 0:
            break

        # Build a sparse (active × n) constraint matrix:  a_i * g_{k-i}
        rows = []
        cols = []
        data = []
        for row_idx, k in enumerate(active):
            lo = max(0, k - (n - 1))
            hi = min(n - 1, k)
            # columns lo … hi contribute a[k‑col]
            cols_range = np.arange(lo, hi + 1, dtype=np.int64)
            rows.extend([row_idx] * cols_range.size)
            cols.extend(cols_range.tolist())
            data.extend(best_a[k - cols_range].tolist())

        A_ub = np.zeros((len(active), n), dtype=np.float64)
        A_ub[rows, cols] = data   # dense matrix is fine – active set is tiny

        c = -np.ones(n, dtype=np.float64)          # maximise sum(g)  ⇔ minimise -∑g
        bounds = [(0.0, None)] * n
        b_ub = np.full(len(active), M, dtype=np.float64)

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
        if not res.success:
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break
        # Rescale to the required ℓ1‑norm
        g = g0 * (target_sum / sum_g0)

        # Convex combination
        best_a = (1.0 - t) * best_a + t * g
        best_a = np.clip(best_a, 0.0, None)

        # Evaluate new objective
        new_val = _full_convolution(best_a, conv_fft_len).max()
        if new_val >= best_val - 1e-12:
            # no progress – stop early
            break
        best_val = new_val
        print(f"[LP refine {it}] improved to {new_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """

    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Problem size – 4096 works well with the existing warm‑start variable.
    n = 4096
    target_sum = np.sqrt(2.0 * n)
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -----------------------------------------------------------
    # Warm start handling (optional)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to required length
            x_old = np.linspace(0.0, 1.0, warm.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            warm = np.interp(x_new, x_old, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum() * target_sum
        else:
            warm = None
    else:
        warm = None

    # -----------------------------------------------------------
    # Initial candidate – either warm start or uniform distribution
    if warm is not None:
        a0 = warm.copy()
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64) * target_sum

    best_a = a0.copy()
    best_val = _full_convolution(best_a, conv_fft_len).max()
    print(f"[Init] value {best_val:.6f}")

    # -----------------------------------------------------------
    # Phase 1 – Adam (smooth‑max)

    # -----------------------------------------------------------
