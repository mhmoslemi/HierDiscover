"""
Step‑function optimizer for the objective

    F(a) = 2·n·max(a∗a) / (∑a)²

The core idea is to start from a distribution whose self‑convolution is already
almost flat (the discrete *arcsine* family) and then iteratively flatten the
convolution further by minimizing the ℓ₂‑error to the ideal constant
convolution.  The error gradient can be computed by a single cross‑correlation,
so every iteration costs only two FFTs.  A simple projected gradient descent on
the probability simplex (with an adaptive step size) drives the maximum of the
convolution down close to the theoretical lower bound 2·n/(2·n‑1)≈1.

The routine respects the given time budget and always returns a valid sequence.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex (sum=1, v≥0).
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _arcsine_candidate(n: int, p: float) -> np.ndarray:
    """
    Produce a discrete arcsine‑type shape:
        a_i ∝ ( (i+½)*(n‑i‑½) )^{-p}
    The parameter p controls the strength of the edge singularity
    (p≈0.5 reproduces the classic arcsine law).
    """
    idx = np.arange(n, dtype=np.float64) + 0.5
    prod = idx * (n - idx + 1.0)          # (i+½)*(n‑i‑½)  with +1 to stay safe
    shape = prod ** (-p)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _flatten_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 200_000,
                     lr0: float = 0.2) -> tuple[np.ndarray, float]:
    """
    Gradient descent on the L2‑error between a∗a and the uniform target.
    The target constant is 1/(2·n‑1) (the average of the convolution when
    ∑a = 1).  After each gradient step we project back onto the simplex.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    target = 1.0 / (2 * n - 1)          # ideal constant convolution value
    best_a = a.copy()
    best_val = float("inf")
    lr = lr0

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # current convolution and objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # sum(a) == 1 → denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Flatten it={it:06d}] new best {val:.6f}")

        # gradient of the L2 error (conv - target)
        diff = conv - target
        grad = 4.0 * _cross_correlation(diff, a, corr_fft_len)

        # gradient step with simple step‑size adaptation
        a_new = a - lr * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)

        # evaluate to decide whether to increase or decrease the step size
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new > val:               # no improvement → shrink step
            lr *= 0.5
        else:                           # improvement → allow modest growth
            lr = min(lr * 1.02, 1.0)

        a = a_new

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function uses an arcsine‑based initial shape and a fast gradient‑based
    flattening routine; it adapts its effort to the supplied time budget and
    always returns a valid sequence.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – keep it moderate for fast FFTs
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build a small diversified pool of starters (arcsine shapes + a few baselines)
    starts = [
        _arcsine_candidate(n, p) for p in (0.3, 0.4, 0.5, 0.6, 0.8, 1.0)
    ]
    starts.append(np.full(n, 1.0 / n))                     # uniform
    starts.append(np.random.dirichlet(np.ones(n)))        # random Dirichlet

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Give each starter a short time slice, keep the best one.
    per_start = max(5.0, (deadline - time.time()) * 0.15 / max(1, len(starts)))
    for a0 in starts:
        if time.time() > deadline:
            break
        sub_dead = min(deadline, time.time() + per_start)
        a_opt, val_opt = _flatten_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead,
            max_iters=120_000,
            lr0=0.2,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Exploration] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Deep refinement of the current best (most of the remaining time)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Refine] long flattening phase")
        a_ref, val_ref = _flatten_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            lr0=0.1,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] new best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # One final check using the provided evaluator (optional, just for logging)
    try:
        from __main__ import evaluate_sequence  # the environment provides it
    except Exception:
        evaluate_sequence = None
    if evaluate_sequence is not None:
        val_check = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated value {val_check:.6f}")
    else:
        print(f"Final best internal value {best_val:.6f}")

    return best_seq.tolist()
