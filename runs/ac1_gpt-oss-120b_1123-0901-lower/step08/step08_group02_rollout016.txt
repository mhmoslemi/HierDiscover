Improving moves are always accepted; worsening moves are accepted with
probability exp(−Δ/Τ).  The temperature decays slowly, allowing the walk
to escape plateaus.  After the annealing phase a few iterations of the
exact‑max sub‑gradient (exponentiated‑gradient) method are run for a
final polishing.  The best sequence found within the time budget is
returned as a plain Python list (all entries are non‑negative and ≤1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient step.
    Returns a vector of length len(a) with
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 25_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _apply_move(
    a: np.ndarray,
    conv: np.ndarray,
    i: int,
    j: int,
    delta: float,
    n: int,
) -> tuple[np.ndarray, np.ndarray]:
    """
    Perform the elementary move: transfer `delta` mass from index i to j.
    Returns the updated probability vector and the exactly updated convolution.
    The update uses the closed‑form rank‑1 change described in the analysis.
    """
    L = 2 * n - 1                      # length of the linear convolution
    a_new = a.copy()
    a_new[i] -= delta
    a_new[j] += delta

    conv_new = conv.copy()

    # contribution of the decrease at i
    start_i = i
    len_i = min(n, L - start_i)
    conv_new[start_i:start_i + len_i] -= 2.0 * delta * a[:len_i]

    # contribution of the increase at j
    start_j = j
    len_j = min(n, L - start_j)
    conv_new[start_j:start_j + len_j] += 2.0 * delta * a[:len_j]

    # quadratic delta^2 corrections
    conv_new[2 * i] += delta * delta
    conv_new[2 * j] += delta * delta
    conv_new[i + j] -= 2.0 * delta * delta

    return a_new, conv_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size (same as the reference implementation)
    n = 4096
    L = 2 * n - 1
    conv_fft_len = _next_pow2(L)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # initialise from warm start if available, otherwise random Dirichlet
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.shape[0] > n:
            a = a[:n]
        elif a.shape[0] < n:
            a = np.pad(a, (0, n - a.shape[0]))
        a = np.clip(a, 0.0, 1000.0)
        if a.sum() < 0.01:
            a = None
        else:
            a = a / a.sum()
    else:
        a = None

    if a is None:
        a = np.random.dirichlet(np.ones(n))

    # current best
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    cur_val = best_val
    cur_a = a.copy()
    cur_conv = conv.copy()
    print(f"[Init] value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Simulated annealing with exact convolution updates
    temp = 0.02                     # initial temperature
    decay = 0.99995                 # exponential cooling
    min_temp = 1e-8
    move_counter = 0
    recompute_every = 5000         # periodic full FFT recompute for safety

    while time.time() < deadline - 5.0:
        # pick a source index with non‑zero mass
        i = np.random.randint(n)
        if cur_a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # step size scales with temperature and available mass at i
        max_move = cur_a[i] * min(0.5, temp * 10.0)
        if max_move <= 0:
            continue
        delta = np.random.rand() * max_move

        a_new, conv_new = _apply_move(cur_a, cur_conv, i, j, delta, n)

        max_c = conv_new.max()
        val_new = 2.0 * n * max_c / (a_new.sum() ** 2)   # sum stays = 1

        # Metropolis acceptance
        if val_new < cur_val or np.random.rand() < np.exp(-(val_new - cur_val) / max(temp, 1e-12)):
            cur_a, cur_conv, cur_val = a_new, conv_new, val_new
            if val_new < best_val:
                best_val, best_a = val_new, a_new.copy()
                print(f"[SA] new best {best_val:.6f} after {move_counter} moves")
        move_counter += 1

        # temperature schedule
        temp = max(min_temp, temp * decay)

        # occasional full recomputation to avoid drift
        if move_counter % recompute_every == 0:
            cur_conv = _full_convolution(cur_a, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – final exponentiated‑gradient (exact‑max sub‑gradient) polish
    if time.time() < deadline:
        a_sub, val_sub = _subgradient_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Final subgrad] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # safety fallback (should never trigger)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to what the evaluator accepts
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
