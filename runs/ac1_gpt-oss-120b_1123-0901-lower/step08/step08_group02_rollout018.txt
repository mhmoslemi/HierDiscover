"""
Propose a better step‑function sequence.

The algorithm proceeds in three stages.

1. **Diverse deterministic starts** – in addition to the warm start
   (if any) we generate a large pool of hand‑crafted shapes:
   * pure left‑hand exponential decay,
   * pure power‑law decay (a_i ∝ (i+1)^{-p}),
   * a product of the two (exp·power) which gives an extra degree of
     freedom,
   * the original pool from the reference implementation (Dirichlet,
     spikes, sinusoid, etc.).
   All candidates are normalised to sum 1.

2. **Fast screening** – each candidate is evaluated with the exact
   objective (using an FFT based convolution).  The best few (the best
   one and the five best) are kept for refinement.

3. **Long‑run refinement** – the surviving candidates are polished with
   two complementary first‑order methods:
   * Adam on a smooth‑max (log‑sum‑exp) surrogate, with an aggressive
     temperature schedule,
   * exponentiated‑gradient (mirror descent on the simplex) using the
     exact sub‑gradient of the max‑convolution.
   The phase lasts until the time budget is exhausted, always keeping
   the best feasible sequence seen so far.

The whole routine respects the supplied time budget (default 1000 s)
and never returns an empty list – a uniform distribution is used as a
fallback.  The final sequence is clipped to the range required by
``evaluate_sequence`` (non‑negative, ≤ 1000).
"""

import time
import numpy as np

# ------------------------------------------------------------
# FFT based convolution utilities (same as the reference code)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ------------------------------------------------------------
# Objective (exact) – avoids the overhead of the provided evaluate_sequence
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a∗a)/(sum(a)²) .  a is assumed non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    # protect against degenerate case (should never happen for our candidates)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ------------------------------------------------------------
# Optimisers -------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector found and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                # a comfortable step size for the simplex
    lam = 0.02               # start temperature (soft)
    lam_factor = 1.08        # faster annealing towards hard max

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule for the soft‑max parametrisation
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – approach the true max quickly
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny jitter when we are close to the deadline
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # very gentle learning‑rate decay
        if it % 150_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices achieving the max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)          # diminishing stepsize
        a = a * np.exp(-eta * grad_a)        # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: shift a random fraction of the mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.25)   # up to 25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Candidate factories
def _exponential_left(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i)  (pure left‑hand exponential)."""
    a = np.exp(-gamma * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _powerlaw_left(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}  (pure left‑hand power law)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _exp_power_mix(n: int, gamma: float, p: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i)·(i+1)^{-p}."""
    exp_part = np.exp(-gamma * np.arange(n, dtype=np.float64))
    pow_part = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
    a = exp_part * pow_part
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a

            # 4) Very cheap local hill‑climber
            a_loc, val_loc = _local_random_search(
                best_seq, n, conv_fft_len, deadline, max_moves=60_000
            )
            if val_loc < best_val:
                best_val, best_seq = val_loc, a_loc.copy()
                print(f"[Local {base_idx}] improved best to {val_loc:.6f}")

        # break early if we are very close to the deadline
        if time.time() > deadline - 15.0:
            break

    # --------------------------------------------------------
    # fallback – guarantee a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
