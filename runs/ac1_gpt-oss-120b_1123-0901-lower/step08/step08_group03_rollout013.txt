def project_simplex(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _project_to_simplex(v):
    """Project vector v onto the probability simplex (non-negative sum to 1)."""
    # ensure v is numpy array
    v = np.asarray(v, dtype=np.float64)
    # shift to make all values <= something? Using algorithm:
    # Based on https://arxiv.org/pdf/1309.1541.pdf (Duchi et al.)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries <= 0 -> return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm is a blend of

* quick systematic shape sweeps (power‑law, exponential, sinusoid),
* massive cheap random spikes / Dirichlet draws,
* long Adam optimisation on a smooth‑max (log‑sum‑exp) surrogate,
* an exponentiated‑gradient sub‑gradient phase,
* a lightweight differential‑evolution exploration,
* a final cheap hill‑climb.

All phases respect the 1000 s budget; the best sequence found so far is
always kept and returned.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a length‑len(a) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex {x ≥ 0, Σx = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive: return the uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.07,
                   lam_start: float = 0.08,
                   lam_factor: float = 1.015,
                   noise_sigma: float = 0.025) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(np.clip(a0, eps, None))
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _eval_objective(best_a, n, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        # --- soft‑max projection to simplex ---
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # --- exact objective (for bookkeeping) ---
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # --- smooth‑max weights (log‑sum‑exp) ---
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --- gradient of surrogate ---
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional stochastic noise
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # --- Adam update on logits ---
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake near deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 300_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.clip(a0, 0.0, None)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)

    it = 0
    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _differential_evolution_optimize(a0: np.ndarray, n: int,
                                    conv_fft_len: int, deadline: float,
                                    pop_size: int = 15,
                                    F: float = 0.5,
                                    CR: float = 0.9,
                                    max_gens: int = None) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex. Returns the best individual found.
    """
    # initialise population (first member = a0)
    population = [a0.copy()]
    while len(population) < pop_size:
        cand = np.random.dirichlet(np.ones(n))
        population.append(cand)

    pop = np.stack(population, axis=0)
    obj = np.array([_eval_objective(pop[i], n, conv_fft_len) for i in range(pop_size)])

    best_idx = int(np.argmin(obj))
    best_a = pop[best_idx].copy()
    best_val = float(obj[best_idx])

    gen = 0
    while time.time() < deadline:
        for i in range(pop_size):
            idxs = list(range(pop_size))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            donor = pop[r1] + F * (pop[r2] - pop[r3])

            # binomial crossover
            mask = np.random.rand(n) < CR
            if not np.any(mask):
                mask[np.random.randint(n)] = True
            trial = np.where(mask, donor, pop[i])

            # project onto simplex (ensures non‑negativity & Σ=1)
            trial = _project_to_simplex(trial)

            val = _eval_objective(trial, n, conv_fft_len)
            if val < obj[i]:
                pop[i] = trial
                obj[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    print(f"[DE gen={gen:04d} ind={i}] new best {val:.6f}")

        gen += 1
        if max_gens is not None and gen >= max_gens:
            break
        # occasional parameter jitter
        if gen % 10 == 0:
            F = np.clip(F + np.random.normal(scale=0.05), 0.1, 0.9)
            CR = np.clip(CR + np.random.normal(scale=0.05), 0.1, 0.95)

    return best_a, best_val


def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    best_val = _eval_objective(a, n, conv_fft_len)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _eval_objective(a2, n, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # problem size (feel free to adjust – larger n gives a better asymptotic constant)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
    else:
        warm = None

    # --------------------------------------------------------------
    # initial best (warm or uniform)
    if warm is not None:
        best_seq = warm.copy()
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _eval_objective(best_seq, n, conv_fft_len)
    print(f"[Init] best {best_val:.6f}")

    # --------------------------------------------------------------
    # 0) systematic shape sweep (power‑law / exponential / sinusoid)
    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Shape] new best {val:.6f}")

    # power‑law a_i ∝ (i+1)^‑p
    for p in (0.2, 0.5, 1.0, 1.5, 2.0):
        a = (np.arange(1, n + 1, dtype=np.float64) ** -p)
        a = a / a.sum()
        try_candidate(a)

    # exponential decay a_i ∝ exp(-γ i)
    for gamma in (0.001, 0.003, 0.01, 0.03, 0.1):
        a = np.exp(-gamma * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        try_candidate(a)

    # half‑wave sinusoid (already used in the reference code)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a = a / a.sum()
    try_candidate(a)

    # --------------------------------------------------------------
    # 1) fast random phase (Dirichlet, spikes, Gaussian mixtures)
    def random_phase():
        nonlocal best_seq, best_val
        # Dirichlet full support
        for _ in range(12):
            a = np.random.dirichlet(np.ones(n))
            try_candidate(a)

        # spike candidates
        for k in (32, 64, 128, 256, 512):
            for _ in range(6):
                idx = np.random.choice(n, size=k, replace=False)
                weights = np.random.dirichlet(np.ones(k))
                a = np.zeros(n, dtype=np.float64)
                a[idx] = weights
                try_candidate(a)

        # two‑Gaussian mixtures
        xs = np.arange(n, dtype=np.float64) / n
        for _ in range(12):
            mu1, mu2 = np.random.rand(2)
            sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
            g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
            g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
            a = g1 + g2
            a /= a.sum()
            try_candidate(a)

    random_phase()

    # --------------------------------------------------------------
    # 2) Adam refinement (allocate ≈ 55 % of remaining budget)
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.55 * remaining
    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=300_000,
        lr=0.07,
        lam_start=0.05,
        lam_factor=1.012,
        noise_sigma=0.02,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Noisy‑restart Adam (a few short runs)
    for restart in range(3):
        if time.time() >= deadline - 30:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.4, size=n)
        a0 = np.exp(logits - logits.max())
        a0 /= a0.sum()

        r_deadline = min(deadline - 5, time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=180_000,
            lr=0.06,
            lam_start=0.05,
            lam_factor=1.01,
            noise_sigma=0.018,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # --------------------------------------------------------------
    # 4) Sub‑gradient refinement (≈ 20 % of remaining budget)
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Differential‑evolution exploration (≈ 15 % of remaining budget)
    if time.time() < deadline - 10:
        de_deadline = deadline - 8
        a_de, val_de = _differential_evolution_optimize(
            best_seq,
            n,
            conv_fft_len,
            de_deadline,
            pop_size=12,
            F=0.5,
            CR=0.9,
        )
        if val_de < best_val:
            best_val = val_de
            best_seq = a_de.copy()
            print(f"[DE] improved best {val_de:.6f}")

    # --------------------------------------------------------------
    # 6) Cheap local hill‑climb (use any remaining seconds)
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=100_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # final safety – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
