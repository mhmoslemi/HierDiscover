"""
Propose a step‑function that minimizes
    F(a)=2·n·max(a∗a)/(sum(a)²)

The algorithm combines:
  * a rich family of parametric initial shapes (exponential edges,
    power‑law edges, double Gaussians, half‑sine, uniform and spikes);
  * dense parameter sweeps (≈300 candidates) evaluated with an FFT‑based
    convolution (O(n log n));
  * a fast simulated‑annealing hill‑climber that moves a random fraction
    of mass between two positions and accepts worsening moves according
    to a temperature schedule;
  * a final short exponentiated‑gradient (sub‑gradient) refinement.

All work is bounded by the supplied time‑budget (default 1000 s) and the
best feasible sequence found so far is always returned.
"""

import time
import numpy as np
from scipy.linalg import toeplitz

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size ηₜ = η₀ / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)          # exponentiated‑gradient
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with the smallest bound."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # helper ----------------------------------------------------------------
    def obj(a: np.ndarray) -> float:
        """Exact objective (no clipping)."""
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ----------------------------------------------------------------------
    # warm start (if supplied)
    best_seq = None
    best_val = float("inf")
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            v = obj(w)
            best_seq, best_val = w.copy(), v
            print(f"[Warm] initial value {v:.6f}")

    # ----------------------------------------------------------------------
    # 1) generate a large, diverse cloud of parametric candidates ----------
    candidates = []

    # uniform
    candidates.append(np.full(n, 1.0 / n))

    # half‑sine (positive half wave)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)

    candidates.append(s)

    for gamma in np.logspace(-2, 0.7, 30):            # 0.01 … ≈5

        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        candidates.append(shape)

    # edge‑polynomial (power‑law) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-5.0, -0.5, 30):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # double‑Gaussian centred at both boundaries
    i = np.arange(n, dtype=np.float64)
    for sigma in np.linspace(0.02, 0.4, 20):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        candidates.append(shape)

    # beta‑like families a_i ∝ (i+1)^{α‑1}·(n‑i)^{β‑1}
    for α in (0.2, 0.5, 0.8, 1.2, 1.5):
        for β in (0.2, 0.5, 0.8, 1.2, 1.5):
            shape = (idx ** (α - 1)) * ((n - idx + 1) ** (β - 1))
            shape = np.maximum(shape, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

    # random Dirichlet draws with random concentration
    for _ in range(200):
        α = np.random.rand() * 4.0 + 0.1          # α ∈ [0.1,4.1]
        candidates.append(np.random.dirichlet(np.full(n, α)))

    # sparse equally‑spaced spikes
    for k in (8, 16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        candidates.append(a)

    # ----------------------------------------------------------------------
    # 2) evaluate the cloud and keep the currently best candidate
    for idx, cand in enumerate(candidates):
        if time.time() >= deadline:
            break
        v = obj(cand)
        if v < best_val:
            best_val, best_seq = v, cand.copy()
            print(f"[Init] candidate {idx:03d} improves to {v:.6f}")

    # ----------------------------------------------------------------------
    # 3) simulated‑annealing hill‑climber around the best solution
    def simulated_hillclimb(a_start: np.ndarray, iters: int = 300_000) -> tuple:
        """Move a random fraction of mass; accept according to temperature."""
        a_cur = a_start.copy()
        cur_val = obj(a_cur)
        best_local = a_cur.copy()
        best_local_val = cur_val

        # temperature schedule – start fairly hot, cool exponentially
        T0 = 0.01
        decay = 0.99997

        for it in range(iters):
            if time.time() >= deadline:
                break
            # pick source and destination
            i = np.random.randint(n)
            if a_cur[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            # propose moving a random fraction (up to 30 %) of the source mass
            delta = a_cur[i] * (np.random.rand() * 0.30)
            a_new = a_cur.copy()
            a_new[i] -= delta
            a_new[j] += delta

            new_val = obj(a_new)

            # Metropolis acceptance
            T = T0 * (decay ** it)
            if new_val < cur_val or np.random.rand() < np.exp(-(new_val - cur_val) / max(T, 1e-12)):
                a_cur, cur_val = a_new, new_val
                if new_val < best_local_val:
                    best_local_val, best_local = new_val, a_new.copy()
                    # occasional progress report
                    if it % 20000 == 0:
                        print(f"[Hillclimb it={it:06d}] new local best {new_val:.6f}")

        return best_local, best_local_val

    # a few independent hill‑climb runs (with random restarts)
    for round_id in range(5):
        if time.time() >= deadline:
            break
        # perturb current best a bit before each run to diversify the search
        pert_logits = np.log(best_seq + 1e-12) + 0.2 * np.random.randn(n)
        a0 = np.exp(pert_logits - np.max(pert_logits))
        a0 /= a0.sum()
        a_hc, v_hc = simulated_hillclimb(a0, iters=120_000)
        if v_hc < best_val:
            best_val, best_seq = v_hc, a_hc.copy()
            print(f"[Round {round_id}] hill‑climb improves to {v_hc:.6f}")

    # ----------------------------------------------------------------------
    # 4) a short exact‑max sub‑gradient polishing phase (exponentiated‑gradient)
    if best_seq is not None:
        a_sub, v_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=150_000,
            eta0=0.8,
        )
        if v_sub < best_val:
            best_val, best_seq = v_sub, a_sub.copy()
            print(f"[Subgrad] final polishing improves to {v_sub:.6f}")

    # ----------------------------------------------------------------------
    # fallback – guarantee a non‑empty, valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
