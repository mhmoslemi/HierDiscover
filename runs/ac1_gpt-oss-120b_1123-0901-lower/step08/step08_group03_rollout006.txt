def l_bfgs_optimize(logits, lam, conv_fft_len, corr_fft_len, maxiter, deadline):
    def func_and_grad(logits):
        a = softmax(logits)
        conv = full_convolution(a, conv_fft_len)
        max_c = conv.max()
        exp_term = np.exp(lam * (conv - max_c))
        sum_exp = exp_term.sum()
        # smooth max
        smooth_val = max_c + np.log(sum_exp) / lam
        # compute weights normalized
        w = exp_term / sum_exp
        grad_a = 2.0 * cross_correlation(w, a, corr_fft_len)
        # chain rule
        avg = np.dot(a, grad_a)  # expectation of grad under a
        grad_log = a * (grad_a - avg)
        return smooth_val, grad_log.astype(np.float64)
    # Use scipy.optimize.minimize
    res = scipy.optimize.minimize(
        func_and_grad,
        logits,
        method='L-BFGS-B',
        jac=True,
        options={'maxiter': maxiter, 'disp': False}
    )
    return res.x
```

def continuation_optimize(a0, lam_schedule, deadline):

    eps=1e-12
    logits = np.log(np.clip(a0, eps, None))
    best_a = a0.copy()
    best_val = _eval_objective(a0, n, conv_fft_len)

            break
        # Estimate remaining time
        remaining = deadline - time.time()
        max_iters = int(min(500, max(50, remaining * 10)))  # approx 10 iter per second
        # Call L-BFGS
        try:
            logits = _l_bfgs_optimize(logits, lam, max_iters, deadline)
        except Exception as e:
            # If early stopping due to time, just break
            break
        a = softmax(logits)
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L-BFGS lam={lam:.2f}] new best {val:.6f}")

"""
Propose a step‑function (non‑negative vector) with a low evaluation score.

Algorithm
---------
1. Generate a large set of cheap random candidates (Dirichlet,
   exponential, sinusoid, Gaussian mixtures, sparse spikes).
2. Keep the best candidate (exact objective).
3. Refine it by a *continuation* of smooth‑max minimisation:
   – optimise the log‑softmax parameters (logits) with L‑BFGS‑B on a
     smooth approximation of max(a∗a) (log‑sum‑exp) for a schedule of
     increasing temperatures (λ).
   – after each λ we evaluate the exact objective and keep the best.
   – a few noisy restarts are performed.
4. Perform an exact‑max exponentiated‑gradient (sub‑gradient) phase
   to further reduce the true max‑convolution.
5. A brief simulated‑annealing hill‑climber explores non‑local moves.
6. Finish with a cheap random‑move local search.
7. Return the best vector found (clipped to the evaluator’s allowed range).

The whole routine respects a 1000 s budget and always returns a valid
sequence.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    mx = x.max()
    ex = np.exp(x - mx)
    return ex / ex.sum()


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
def _smooth_max_and_grad(logits: np.ndarray, lam: float,
                         conv_fft_len: int, corr_fft_len: int):
    """
    Returns (smooth_max, grad_logits) for a given temperature λ.
    The smooth_max is the log‑sum‑exp / λ (numerically stable).
    """
    a = _softmax(logits)

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()

    # stable softmax over the λ‑scaled convolution values
    exp_term = np.exp(lam * (conv - max_c))
    sum_exp = exp_term.sum()
    smooth_val = max_c + np.log(sum_exp) / lam

    w = exp_term / sum_exp                     # normalised weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max (mirror‑descent on simplex)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    return smooth_val, grad_log


def _l_bfgs_optimize(logits: np.ndarray, lam: float,
                     max_iter: int, conv_fft_len: int, corr_fft_len: int) -> np.ndarray:
    """
    One L‑BFGS‑B optimisation loop for a fixed λ.
    Returns the updated logits.
    """
    def fun(x):
        val, g = _smooth_max_and_grad(x, lam, conv_fft_len, corr_fft_len)
        return val, g

    # scipy's minimize accepts a function returning (f, grad) when jac=True
    res = scipy.optimize.minimize(fun,
                                  logits,
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxiter': max_iter, 'disp': False})
    return res.x


def _continuation_optimize(a0: np.ndarray,
                           lam_schedule,
                           deadline: float,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int):
    """
    Continuation method: start from a0, run L‑BFGS for each λ in
    lam_schedule, keep the best vector (exact objective).
    """
    eps = 1e-12
    logits = np.log(np.clip(a0, eps, None))

    best_a = a0.copy()
    best_val = _eval_objective(a0, n, conv_fft_len)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        # allocate a modest number of L‑BFGS iterations;
        # the exact number is tuned to the remaining time.
        remaining = deadline - time.time()
        max_iter = max(20, int(min(300, remaining * 8)))  # ~8 it/s on typical hardware

        logits = _l_bfgs_optimize(logits, lam, max_iter, conv_fft_len, corr_fft_len)

        a = _softmax(logits)
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.2f}] new best {val:.6f}")

    return best_a, best_val

# Exponentiated‑gradient sub‑gradient (exact max) refinement
def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 250_000,
                          eta0: float = 0.85):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)      # EG update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Simulated annealing (non‑local hill‑climb)
def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_iters: int = 200_000):
    a = a0.copy()
    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)

    # initial temperature proportional to current objective
    T0 = best_val * 0.5
    it = 0

    while it < max_iters and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of the source mass to the destination
        delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % of a[i]
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _eval_objective(a2, n, conv_fft_len)
        Δ = val2 - best_val
        # linear cooling schedule
        T = T0 * max(0.0, 1.0 - it / max_iters)
        accept = False
        if Δ < 0:
            accept = True
        else:
            if T > 0 and np.exp(-Δ / T) > np.random.rand():
                accept = True
        if accept:
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[SA it={it:06d}] new best {val2:.6f}")
        it += 1

    return best_a, best_val

# Cheap random local search (single‑mass moves)
def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 80_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % move
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # --------------------------------------------------- cheap random phase
    best_seq = None
    best_val = float("inf")

    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # deterministic baselines
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # Dirichlet (full support)
    for _ in range(12):
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # Exponential tails (both sides)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # Half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # Sparse spike supports
    for k in (32, 64, 128, 256, 512):
        for _ in range(6):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            try_candidate(a)

    # Two‑Gaussian mixtures
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(12):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.15 + 0.02
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        try_candidate(a)

    # ------------------------------------------------------------------
    # --------------------------------------------------- continuation L‑BFGS
    if time.time() < deadline - 30:
        lam_sched = [0.05, 0.2, 1.0, 5.0, 20.0, 100.0]
        a_cont, val_cont = _continuation_optimize(best_seq, lam_sched,
                                                  deadline, n,
                                                  conv_fft_len, corr_fft_len)
        if val_cont < best_val:
            best_val = val_cont
            best_seq = a_cont.copy()
            print(f"[Continuation] improved best {val_cont:.6f}")

    # --------------------------------------------------- noisy restarts of continuation
    for restart in range(3):
        if time.time() >= deadline - 25:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        # add noticeable Gaussian noise to the logits
        logits += np.random.normal(scale=0.5, size=n)
        a_restart, val_restart = _continuation_optimize(_softmax(logits),
                                                        lam_sched,
                                                        deadline, n,
                                                        conv_fft_len, corr_fft_len)
        if val_restart < best_val:
            best_val = val_restart
            best_seq = a_restart.copy()
            print(f"[Restart {restart}] new best {val_restart:.6f}")

    # --------------------------------------------------- sub‑gradient refinement
    if time.time() < deadline - 15:
        a_sub, val_sub = _subgradient_optimize(best_seq,
                                                n,
                                                conv_fft_len,
                                                corr_fft_len,
                                                deadline,
                                                max_iters=250_000,
                                                eta0=0.85)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # --------------------------------------------------- simulated annealing
    if time.time() < deadline - 10:
        a_sa, val_sa = _simulated_annealing(best_seq,
                                            n,
                                            conv_fft_len,
                                            deadline,
                                            max_iters=180_000)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved best {val_sa:.6f}")

    # --------------------------------------------------- cheap local search
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(best_seq,
                                               n,
                                               conv_fft_len,
                                               deadline,
                                               max_moves=100_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # final sanity – ensure a valid non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
