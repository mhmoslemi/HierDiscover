"""
Propose a step‑function (non‑negative list) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)²)

The search consists of three stages:

1️⃣  Warm‑start: use any stored sequence, a few handcrafted shapes,
    and a large pool of random Dirichlet samples.  For each sample
    we also try three orderings (original, anti‑pair, interleaved)
    and keep the best.

2️⃣  Smooth‑max Adam optimisation:  a soft‑max surrogate of the max
    convolution (log‑sum‑exp) is minimised with Adam on the logits
    of a soft‑max parametrisation (guaranteeing the simplex).  The
    temperature λ is increased progressively so that the surrogate
    converges to the true max.

3️⃣  Light local stochastic refinement (swap / mass‑transfer moves).

All heavy work respects the global 1000 s budget.  The final vector
is clipped to the range accepted by ``evaluate_sequence`` and returned
as a plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence but faster)
def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(a*a) / (sum(a)²).  a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – used by additive updates if needed
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate constructions ------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place the largest half at the left, the smallest half at the right."""
    n = len(vals)
    s = np.sort(vals)[::-1]            # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]                # biggest values left‑to‑right
    a[-half:] = s[-half:][::-1]        # smallest values right‑to‑left
    if n % 2:
        a[half] = s[half]
    return a


def _interleaved_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Alternate large and small values:
    positions 0,2,4,… get the largest values,
    positions 1,3,5,… get the smallest values (reversed).
    """
    n = len(vals)
    s = np.sort(vals)[::-1]            # descending
    a = np.empty(n, dtype=np.float64)
    # number of slots for the “large” side
    n_large = (n + 1) // 2
    a[0::2] = s[:n_large]              # fill even positions with biggest
    a[1::2] = s[n_large:][::-1]        # fill odd positions with smallest (reversed)
    return a


# ----------------------------------------------------------------------
# Random candidate generator (Dirichlet or power‑law) -----------------
def _random_candidate(n: int) -> np.ndarray:
    """
    Return a Dirichlet or power‑law sample, then randomly choose an
    arrangement (original / anti‑pair / interleaved).
    """
    if np.random.rand() < 0.7:                      # Dirichlet most of the time
        log_alpha = np.random.uniform(-2.0, 1.5)     # α ∈ [0.01, ≈31.6]
        alpha = 10.0 ** log_alpha
        cand = np.random.dirichlet(alpha * np.ones(n))
    else:                                           # heavy‑tailed power law
        beta = np.random.uniform(0.5, 3.0)
        r = np.random.rand(n)
        cand = r ** (-beta)
        cand /= cand.sum()

    # random ordering
    arr_type = np.random.choice(['orig', 'anti', 'inter'])
    if arr_type == 'anti':
        cand = _anti_pair_arrangement(cand)
    elif arr_type == 'inter':
        cand = _interleaved_arrangement(cand)
    # else keep original order
    return cand.astype(np.float64)


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0),
                   iters_per_stage: int = 4000) -> tuple[np.ndarray, float]:
    """
    Adam on the log‑sum‑exp surrogate of max(conv).  The vector is
    represented by soft‑max logits to keep it on the simplex.
    """
    # initialise
    logits = np.log(a0 + 1e-16)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_a = a0.copy()
    best_val = _objective(best_a)

    for lam in lam_schedule:
        # a modest per‑stage iteration budget, but we also respect the global deadline
        for _ in range(iters_per_stage):
            if time.time() >= deadline:
                break

            # --- forward pass -------------------------------------------------
            # soft‑max to obtain a probability vector
            mx_log = np.max(logits)
            exp_logits = np.exp(logits - mx_log)
            a = exp_logits / exp_logits.sum()          # shape (n,)

            # convolution
            conv = _full_convolution(a, conv_fft_len)  # length 2n‑1

            # smooth‑max weights (soft‑max over conv)
            mx_c = np.max(conv)
            exp_lam = np.exp(lam * (conv - mx_c))      # stable
            w = exp_lam / exp_lam.sum()                # length 2n‑1, sums to 1

            # gradient of the surrogate w.r.t. a
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # chain rule through soft‑max (logits → a)
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)              # shape (n,)

            # --- Adam update --------------------------------------------------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # per‑coordinate step size (standard Adam schedule)
            step = 0.5 / (np.sqrt(v_hat) + eps)
            logits = logits - step * m_hat

            # occasional exact evaluation
            if t % 250 == 0:
                val = _objective(a)
                if val < best_val:
                    best_val = val
                    best_a = a.copy()
                    print(f"[Adam λ={lam:.1f} it={t}] new best {val:.6f}")

        if time.time() >= deadline:
            break

    return best_a, best_val

# Light local stochastic refinement (swap / mass transfer)
def _local_search(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing with an annealing acceptance rule.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    start_temp = 0.015

    for it in range(max_moves):
        if time.time() >= deadline:
            break

        # random move: swap or fractional transfer
        if np.random.rand() < 0.5:                     # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                           # transfer
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # annealing acceptance
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously stored sequence (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            val = _objective(warm)
            best_seq, best_val = warm.copy(), val
            print(f"[Warm start] value {val:.6f}")

    # --------------------------------------------------------------
    # Hand‑crafted starters
    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for cand in starters:
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Starter] value {val:.6f}")

    # --------------------------------------------------------------
    # Massive random warm‑starts (anti‑pair / interleaved / original)
    n_samples = 3000
    print(f"Generating {n_samples} random candidates …")
    for _ in range(n_samples):
        if time.time() >= deadline - 250:   # reserve time for optimisation phases
            break
        cand = _random_candidate(n)
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random pool] new best {val:.6f}")

    # --------------------------------------------------------------
    # Adam optimisation on smooth‑max surrogate
    # allocate the majority of the remaining budget to this stage
    adam_deadline = deadline - 120.0
    if time.time() < adam_deadline:
        seq_adam, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            adam_deadline)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Light local stochastic hill‑climbing (if time permits)
    local_deadline = deadline - 20.0
    if best_seq is not None and time.time() < local_deadline:
        seq_loc, val_loc = _local_search(best_seq, n,
                                         conv_fft_len,
                                         local_deadline)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final report
    final_val = _objective(best_seq)
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
