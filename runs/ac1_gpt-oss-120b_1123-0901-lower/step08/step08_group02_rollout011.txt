"""
Propose a step‑function (non‑negative vector) that (approximately) minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

using a multi‑stage search:

1.  Build a rich set of low‑dimensional parametric families
    (U‑shapes, power‑law, exponentials, Beta‑pdfs, random Dirichlet,
    sparse spikes).  Each candidate is normalised to sum = 1 and its
    exact objective is evaluated by an FFT based convolution.

2.  Keep the best candidate and improve it by a **sub‑gradient
    exponentiated‑gradient** phase (the exact max of the convolution is
    used to build the sub‑gradient).  The step size decays like 1/√t.

3.  Refinement with a **smooth‑max Adam** optimiser.  The smooth‑max
    is the log‑sum‑exp of the convolution with an annealed temperature
    that finally approximates the true ℓ∞‑norm.

4.  Finally a cheap **simulated‑annealing / hill‑climbing** that moves
    a random fraction of mass from one index to another and accepts
    occasional worsening moves.  This explores directions that the
    gradient methods may miss.

All phases respect a global time budget (default 1000 s) and the best
feasible sequence found so far is always returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, identical to evaluate_sequence for non‑clipped vectors)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a∗a) / (sum(a)²).  Assumes a ≥ 0."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Candidate factories ----------------------------------------------------
def _candidate_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _candidate_u_shape(n: int, power: float) -> np.ndarray:
    """U‑shape: |i-centre|^power (larger at the edges)."""
    idx = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = np.abs(idx - centre) ** power + 1e-12
    shape /= shape.sum()
    return shape


def _candidate_power_negative(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^alpha , with α<0 → edge‑heavy."""
    i = np.arange(1, n + 1, dtype=np.float64)
    prod = i * (n - i + 1.0)
    shape = prod ** alpha
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _candidate_exponential(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(gamma·i) + exp(gamma·(n‑1‑i))."""
    idx = np.arange(n, dtype=np.float64)
    left = np.exp(gamma * idx)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _candidate_beta(n: int, a: float, b: float) -> np.ndarray:
    """U‑shape from a Beta pdf (a,b<1)."""
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-12
    # safe version of x^(a-1)*(1‑x)^(b-1)
    shape = np.power(x + eps, a - 1.0) * np.power(1.0 - x + eps, b - 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _candidate_dirichlet(n: int, concentration: float) -> np.ndarray:
    """Random Dirichlet draw."""
    return np.random.dirichlet(np.full(n, concentration))


def _candidate_spikes(n: int, k: int) -> np.ndarray:
    """k equally spaced spikes (each gets 1/k of the total mass)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        positions = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[positions] = 1.0 / k
    return a


def _mix_two(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Convex mix w·a + (1‑w)·b."""
    mixed = w * a + (1.0 - w) * b
    mixed /= mixed.sum()
    return mixed


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.8,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set for the exact max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)  # 1/√t decay
        a = a * np.exp(-eta * grad)    # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best point (sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)   # initialise on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.02                 # initial temperature (soft)
    lam_factor = 1.08          # geometric annealing of temperature

    best_a = a0.copy()
    best_val = _objective(a0, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (exact)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing / hill‑climbing: move a random
    fraction of mass from i → j, accept worsening moves with
    probability exp(-Δ/temp).  Temperature decays geometrically.
    """
    a = a0.copy()
    cur_val = _objective(a, conv_fft_len)

    best_a = a.copy()
    best_val = cur_val

    it = 0
    # initialise temperature as a fraction of the current objective
    temp = max(0.05 * cur_val, 1e-4)
    cooling = 0.99995

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 20 %) of the source mass
        frac = np.random.rand() * 0.20
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, conv_fft_len)

        improve = val2 < cur_val
        if improve or (np.random.rand() < np.exp(-(val2 - cur_val) / max(temp, 1e-12))):
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()

        # anneal temperature
        temp = max(temp * cooling, 1e-9)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous solution is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a large and diverse pool of cheap candidates
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # add a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    pool.append(_candidate_uniform(n))

    # parametric families ------------------------------------------------
    for _ in range(50):                     # U‑shapes (|i−c|^p)
        p = np.random.uniform(0.5, 4.0)
        pool.append(_candidate_u_shape(n, p))

    for _ in range(50):                     # power‑law with negative exponent
        alpha = np.random.uniform(-5.0, -0.2)
        pool.append(_candidate_power_negative(n, alpha))

    for _ in range(50):                     # double‑exponential (edge‑heavy)
        gamma = np.random.uniform(0.01, 4.0)
        pool.append(_candidate_exponential(n, gamma))

    for _ in range(30):                     # Beta‑pdf U‑shapes
        a = np.random.uniform(0.1, 0.9)
        b = np.random.uniform(0.1, 0.9)
        pool.append(_candidate_beta(n, a, b))

    for _ in range(30):                     # random Dirichlet
        conc = np.random.rand() * 4.0 + 0.1   # α ∈ [0.1,4.1]
        pool.append(_candidate_dirichlet(n, conc))

    for _ in range(30):                     # sparse spike patterns
        k = np.random.choice([8, 16, 32, 64, 128, 256])
        pool.append(_candidate_spikes(n, k))

    # random convex mixes of two already generated candidates
    for _ in range(30):
        a, b = np.random.choice(pool, 2, replace=False)
        w = np.random.rand()
        pool.append(_mix_two(a, b, w))

    # ------------------------------------------------------------------
    # cheap evaluation of the whole pool
    best_seq = None
    best_val = float("inf")

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        seq = cand.astype(np.float64)
        seq = np.maximum(seq, 0.0)
        if seq.sum() < 0.01:                # avoid degenerate cases
            seq = np.random.dirichlet(np.ones(n))
        seq /= seq.sum()
        val = _objective(seq, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, seq.copy()
            print(f"[Init {idx:03d}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – sub‑gradient refinement (≈40 % of remaining budget)
    while time.time() < deadline - 30.0:
        # sub‑gradient phase
        seq_sg, val_sg = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            eta0=0.9,
            max_iters=400_000,
        )
        if val_sg < best_val:
            best_val, best_seq = val_sg, seq_sg.copy()
            print(f"[SubGrad] improved to {val_sg:.6f}")

        # Adam fine‑tuning
        seq_ad, val_ad = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
        )
        if val_ad < best_val:
            best_val, best_seq = val_ad, seq_ad.copy()
            print(f"[Adam] improved to {val_ad:.6f}")

        # Simulated‑annealing local moves (quick hill‑climb)
        seq_an, val_an = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_an < best_val:
            best_val, best_seq = val_an, seq_an.copy()
            print(f"[Anneal] improved to {val_an:.6f}")

        # break if we are very close to the deadline
        if time.time() > deadline - 10.0:
            break

    # ------------------------------------------------------------------
    # Fallback – ensure a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
