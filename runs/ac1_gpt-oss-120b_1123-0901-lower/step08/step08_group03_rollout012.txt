#!/usr/bin/env python
"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The routine combines several orthogonal ideas:

* Very diverse random initialisations (spikes, Dirichlet, mixtures,
  exponentials, triangular shapes, etc.).
* A long Adam run on the smooth‑max surrogate (log‑sum‑exp) with a
  slow temperature schedule – this gives a solid ‘global’ improvement.
* A custom LP‑refinement step:  given the current vector *f* and its
  maximum autocorrelation *M*, we solve

        max   Σ g_j
        s.t.  (f∗g)_k ≤ M   for all k,
              g_j ≥ 0

  The optimum pushes mass into positions that interfere least with *f*.
  A convex combination ``f←(1‑α)f+α g`` reduces the worst convolution.
* Exact‑max sub‑gradient (exponentiated‑gradient) fine‑tuning.
* Cheap random hill‑climbing on the exact objective for the final push.

All phases respect the global 1000 s budget and the best sequence found
so far is always returned.  The implementation works for any length;
the default length is 4096 (the same as the reference solution) but the
code can be tweaked to larger n if desired.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# optional LP imports – if unavailable the LP phase is skipped
try:
    import scipy.sparse as sp
    import scipy.optimize as opt
    _has_scipy = True
except Exception:                     # pragma: no cover
    _has_scipy = False

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) via FFT."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]
    """
    rev_a = a[::-1]
    fw = rfft(w, fft_len)
    fr = rfft(rev_a, fft_len)
    corr = irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

# Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.08,
    lam_factor: float = 1.015,
    noise_sigma: float = 0.025,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    a0 = np.clip(a0, eps, None)                     # avoid log(0)
    logits = np.log(a0)                             # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # soft‑max projection (simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (cheap enough)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # add a tiny stochastic perturbation
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.9

        # occasional shake near deadline
        if it % 4_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val

# Exact‑max sub‑gradient (exponentiated‑gradient) refinement
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] best {val:.6f}")

        # active‑set weights (uniform on the ties)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:                               # should never happen
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against under‑/overflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Very cheap random hill‑climb on the exact objective
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Linear‑programming refinement (requires SciPy)
def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_steps: int = 6,
    alpha: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly solve  max Σ g_j   s.t.  (a∗g)_k ≤ M, g≥0.
    Update a ← (1‑α)·a + α·ĝ, where ĝ is the scaled solution
    (scaled so that Σ ĝ = Σ a).  Returns the best vector seen.
    """
    if not _has_scipy:                     # pragma: no cover
        print("[LP] SciPy not available – skipping LP refinement.")
        return a0.copy(), _eval_objective(a0, n, conv_fft_len)

    a = a0.copy()
    best_a = a.copy()
    best_val = _eval_objective(a, n, conv_fft_len)
    it = 0

    # pre‑allocate the constraint matrix shape
    rows = 2 * n - 1
    cols = n

    while it < max_steps and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # Build the Toeplitz‐like convolution matrix A (shape rows×cols)
        #   (A·g)_k = Σ_i a_i·g_{k-i}
        # each diagonal d (=k‑j) carries the constant a[d] (for d≥0)
        diags = [np.full(n, a[d]) for d in range(n)]
        A = sp.diags(diags, offsets=np.arange(n), shape=(rows, cols), format="csr")

        # RHS of the constraints
        b = np.full(rows, M, dtype=np.float64)

        # Linear programme: maximise Σ g_j  ⇔ minimise -Σ g_j
        c = -np.ones(cols, dtype=np.float64)

        # bounds g_j ≥ 0
        bounds = [(0, None) for _ in range(cols)]

        # Solve (using HiGHS – the default for recent SciPy)
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print(f"[LP] iteration {it} failed ({res.message}) – abort.")
            break

        g = np.maximum(res.x, 0.0)          # numerical safety
        sum_g = g.sum()
        if sum_g < 1e-12:                    # degenerate solution – stop
            break

        # Scale g so that it has the same total mass as a
        scale = a.sum() / sum_g
        g_scaled = g * scale

        # Convex update
        a_new = (1 - alpha) * a + alpha * g_scaled
        # (the sum stays unchanged up to round‑off)
        a_new = np.maximum(a_new, 0.0)
        a_new /= a_new.sum()

        val_new = _eval_objective(a_new, n, conv_fft_len)
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[LP it={it}] improved to {val_new:.6f}")

        # accept the update if it did not worsen the objective,
        # otherwise keep the old a (helps when LP is noisy)
        if val_new <= best_val * 1.001:
            a = a_new

        it += 1

    return best_a, best_val

    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # helper to evaluate a candidate with the exact objective
    def try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # ------------------------------------------------------------------
    # random‑spike / distribution phase
    best_seq = None
    best_val = float("inf")

    # baseline
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # 1. full‑support Dirichlet draws
    for _ in range(30):
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # 2. exponential tails (different decay rates)
    for gamma in (0.15, 0.25, 0.4, 0.6, 0.9, 1.4, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # 3. simple triangle / hat
    m = n // 2
    tri = np.concatenate([np.linspace(0, 1, m, endpoint=False),
                          np.linspace(1, 0, m + 1)])
    a = np.maximum(tri, 0)
    a /= a.sum()
    try_candidate(a)

    # 4. half‑period sinusoid (already present in reference)
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # 5. spike‑sets with various supports
    for k in (32, 64, 128, 256, 512, 1024):
        for _ in range(12):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            try_candidate(a)

    # 6. mixture of two Gaussians (more draws)
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(20):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.12 + 0.03
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        try_candidate(a)

    # 7. random clustered spikes (few blocks)
    for _ in range(10):
        blocks = np.random.randint(2, 6)       # 2‑5 clusters
        a = np.zeros(n, dtype=np.float64)
        for _ in range(blocks):
            centre = np.random.randint(n)
            width = np.random.randint(5, 30)
            left = max(0, centre - width // 2)
            right = min(n, centre + width // 2)
            a[left:right] = np.random.rand(right - left) + 0.5
        a /= a.sum()
        try_candidate(a)

    # ------------------------------------------------------------------
    # Adam‑based smooth‑max optimisation
    remaining = deadline - time.time()
    if remaining < 30:                         # too little time left
        best_seq = np.full(n, 1.0 / n) if best_seq is None else best_seq
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return best_seq.tolist()

    # allocate ~55 % of the remaining budget to the long Adam run
    adam_deadline = time.time() + 0.55 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=500_000,
        lr=0.07,
        lam_start=0.08,
        lam_factor=1.012,
        noise_sigma=0.025,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # LP‑refinement phase (if SciPy is available)
    if _has_scipy and time.time() < deadline - 30:
        lp_deadline = time.time() + 0.20 * (deadline - time.time())
        a_lp, val_lp = _lp_refine(
            best_seq,
            n,
            conv_fft_len,
            lp_deadline,
            max_steps=8,
            alpha=0.12,
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] improved best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Noisy‑restart Adam (short bursts)
    for restart in range(5):
        if time.time() >= deadline - 25:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.35, size=n)   # stronger perturbation
        a0 = np.exp(logits - np.max(logits))
        a0 /= a0.sum()

        r_deadline = min(deadline - 5,
                         time.time() + 0.07 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=200_000,
            lr=0.06,
            lam_start=0.08,
            lam_factor=1.015,
            noise_sigma=0.02,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (exact‑max)
    if time.time() < deadline - 20:
        sg_deadline = deadline - 15
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sg_deadline,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Final cheap random hill‑climb
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
