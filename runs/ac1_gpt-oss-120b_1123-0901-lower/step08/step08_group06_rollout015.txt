def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, lam=100.0, max_iters=300000,
                   lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    # Use logits representation: x = log(a+eps)
    x = np.log(a0 + 1e-12)
    m = np.zeros_like(x)
    v = np.zeros_like(x)
    t = 0
    best_a = a0
    best_val = _objective(a0)
    while time.time() < deadline and t < max_iters:
        t += 1
        surrogate, grad_log = _lbfgs_obj(x, n, conv_fft_len, corr_fft_len, lam)  # gradient of smooth max
        # Actually _lbfgs_obj returns surrogate and grad_log given logits x.
        # So we can reuse _lbfgs_obj.
        # Now Adam update on x
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        # recompute a
        mx = np.max(x)
        exp_l = np.exp(x - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
    return best_a, best_val

"""
Propose a step‑function (non‑negative list) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)²)

Algorithm overview (≈1000 s budget)

1. **Initial portfolio** – a small set of handcrafted families
   (uniform, linear‑decrease, exponential, power‑law, Hann,
   Gaussian‑edges, triangle, anti‑pair Dirichlet).  The best of these
   seeds the later phases.

2. **Heavy random warm‑start** – thousands of Dirichlet samples with
   assorted concentrations, each optionally sorted (monotone) or
   anti‑paired.  The best survivor is kept.

3. **Smooth‑max L‑BFGS refinement** – minimise a log‑sum‑exp
   surrogate of the convolution peak.  The temperature λ is raised
   through a schedule (2 → 300) and a generous iteration budget per
   stage.

4. **Adam optimisation of the same surrogate** – a first‑order
   method with momentum often escapes shallow local minima that
   L‑BFGS can miss.

5. **Exact‑max sub‑gradient phases** – two flavours are tried:
   multiplicative (exponentiated‑gradient) and additive (Euclidean
   projection onto the simplex).  Both run for many iterations with
   a decreasing step‑size.

6. **Local stochastic annealing** – random swaps, mass transfers,
   and segment reversals with a simple temperature schedule.

7. **Optional up‑sampling** – if time permits, linearly interpolate
   the current best vector to double the resolution (8192 points) and
   repeat steps 3–5 on the finer grid.

All phases respect the global deadline; the routine finally returns
the best sequence found (clipped to the range accepted by
``evaluate_sequence``).  The implementation uses only NumPy, SciPy
(and optionally CVXPy for the tiny‑scale experiments) and runs on a
single CPU.
"""

import time
import numpy as np
import scipy.optimize as _opt

# ----------------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum‑normalised to 1)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) with a already normalised (sum(a)=1)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Warm‑start helper (global variable from the harness)
def _warm_start(n: int):
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()

# Candidate factories
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_decreasing_candidate(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)  (largest at the leftmost index)."""
    i = np.arange(n, dtype=np.float64)
    shape = (n - i)
    shape /= shape.sum()
    return shape


def _exponential_candidate(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ exp(-alpha·i) (monotonically decreasing)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-alpha * i)
    shape /= shape.sum()
    return shape


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = i ** (-p)
    shape /= shape.sum()
    return shape


def _hann_candidate(n: int) -> np.ndarray:
    """Classic Hann window (sin²)."""
    i = np.arange(n, dtype=np.float64)
    shape = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Dirichlet draw + anti‑pair ordering (largest opposite smallest)."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals /= vals.sum()
    s = np.sort(vals)[::-1]                     # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by LBFGS and Adam)
def _smoothmax_obj(logits: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, lam: float):
    """Log‑sum‑exp surrogate of max(conv) and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp on the shifted values
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                       # soft‑weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)       # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the surrogate
def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(2.0, 5.0, 10.0, 30.0, 100.0, 300.0),
                    max_iter_stage: int = 500) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # allocate a tiny safety margin for later phases
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 12)))  # generous
        res = _opt.minimize(_smoothmax_obj, logits,
                            args=(n, conv_fft_len, corr_fft_len, lam),
                            method='L-BFGS-B', jac=True,
                            options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] improved to {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   lam: float = 200.0,
                   max_iters: int = 300_000,
                   lr: float = 0.05,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (logits space)."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    # initialise logits
    logits = np.log(a + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        surrogate, grad_log = _smoothmax_obj(logits, n,
                                             conv_fft_len, corr_fft_len, lam)
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # re‑normalise to a probability vector
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it}] improved to {val:.6f}")
    return best_a, best_val

# Exact‑max sub‑gradient: multiplicative (exponentiated‑gradient)
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 0.7) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12:                        # occasional re‑initialisation
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:06d}] improved to {val:.6f}")
        it += 1
    return best_a, best_val

# Exact‑max sub‑gradient: additive (projected Euclidean)
def _additive_subgradient_optimize(a0: np.ndarray, n: int,
                                  conv_fft_len: int, corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 600_000,
                                  eta0: float = 0.5) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # sub‑gradient
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[AddSubgrad it={it:06d}] improved to {val:.6f}")
        it += 1
    return best_a, best_val

# Local stochastic annealing (swap / transfer / segment‑reverse)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    for it in range(max_moves):
        if time.time() >= deadline:
            break
        move_type = np.random.rand()
        if move_type < 0.4:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        elif move_type < 0.8:                   # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
        else:                                   # reverse a random segment
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                a2 = a.copy()
            else:
                lo, hi = sorted((i, j))
                a2 = a.copy()
                a2[lo:hi + 1] = a2[lo:hi + 1][::-1]

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
        else:
            # simple annealing acceptance
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size (base resolution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1) Warm start / handcrafted seeds
    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    handcrafted = [
        _uniform_candidate(n),
        _linear_decreasing_candidate(n),
        _exponential_candidate(n, alpha=0.001),   # almost uniform
        _exponential_candidate(n, alpha=0.02),
        _exponential_candidate(n, alpha=0.1),
        _powerlaw_candidate(n, p=0.5),
        _powerlaw_candidate(n, p=1.0),
        _powerlaw_candidate(n, p=2.0),
        _hann_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
        _triangle_candidate(n),
    ]

    for cand in handcrafted:
        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Handcrafted] new best {val:.6f}")

    # --------------------------------------------------------------
    # 2) Massive random warm‑starts (Dirichlet + optional sorting)
    n_samples = 4000
    print(f"Generating {n_samples} random Dirichlet candidates …")
    while time.time() < deadline - 200 and n_samples > 0:
        # pick concentration
        log_alpha = np.random.uniform(-2.5, 1.5)   # α ∈ [~0.003, ~31]
        alpha = 10.0 ** log_alpha
        # sample and optionally sort to obtain monotone shapes
        vals = np.random.dirichlet(alpha * np.ones(n))
        if np.random.rand() < 0.5:
            # monotone decreasing (simple sort)
            vals = np.sort(vals)[::-1]
        else:
            # anti‑pair arrangement (as used before)
            vals = _anti_pair_candidate(n, alpha=alpha)
        vals /= vals.sum()
        val = _objective(vals)
        if val < best_val:
            best_val, best_seq = val, vals.copy()
            print(f"[Random pool] new best {val:.6f}")
        n_samples -= 1

    # --------------------------------------------------------------
    # 3) L‑BFGS smooth‑max refinement
    lbfgs_deadline = min(deadline - 100.0, time.time() + 180.0)
    if time.time() < lbfgs_deadline:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n, conv_fft_len,
                                         corr_fft_len, lbfgs_deadline)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 4) Adam optimisation of the same surrogate
    adam_deadline = min(deadline - 70.0, time.time() + 180.0)
    if time.time() < adam_deadline:
        seq_adam, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            adam_deadline,
                                            lam=200.0, max_iters=300_000)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 5) Exact‑max sub‑gradient phases
    sub_deadline = min(deadline - 50.0, time.time() + 250.0)
    if time.time() < sub_deadline:
        seq_mul, val_mul = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 max_iters=1_200_000,
                                                 eta0=0.7)
        if val_mul < best_val:
            best_val, best_seq = val_mul, seq_mul.copy()
            print(f"[MulSubgrad] improved to {val_mul:.6f}")

    add_deadline = min(deadline - 30.0, time.time() + 180.0)
    if time.time() < add_deadline:
        seq_add, val_add = _additive_subgradient_optimize(best_seq, n,
                                                          conv_fft_len,
                                                          corr_fft_len,
                                                          add_deadline,
                                                          max_iters=800_000,
                                                          eta0=0.5)
        if val_add < best_val:
            best_val, best_seq = val_add, seq_add.copy()
            print(f"[AddSubgrad] improved to {val_add:.6f}")

    # --------------------------------------------------------------
    # 6) Local stochastic annealing
    if best_seq is not None and time.time() < deadline - 20.0:
        loc_deadline = deadline - 5.0
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len,
                                                loc_deadline,
                                                max_moves=400_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 7) Optional up‑sampling to a finer grid (8192 points)
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of current best
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline - 2.0, time.time() + 180.0)
        seq_big_lb, val_big_lb = _lbfgs_optimize(a_big, n_big,
                                                 conv_fft_len_big,
                                                 corr_fft_len_big,
                                                 big_deadline,
                                                 lam_schedule=(5, 20, 80, 200),
                                                 max_iter_stage=120)
        if val_big_lb < best_val:
            best_val, best_seq = val_big_lb, seq_big_lb.copy()
            print(f"[Big L‑BFGS] improved to {val_big_lb:.6f}")

        final_deadline = min(deadline, time.time() + 120.0)
        seq_big_sub, val_big_sub = _subgradient_optimize(best_seq, n_big,
                                                         conv_fft_len_big,
                                                         corr_fft_len_big,
                                                         final_deadline,
                                                         max_iters=600_000,
                                                         eta0=0.6)
        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, seq_big_sub.copy()
            print(f"[Big Subgrad] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 8) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
