def _adam_subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                               max_iters=500000, lr0=0.3, beta1=0.9, beta2=0.999,
                               eps=1e-8):
    a = _project_onto_simplex(np.maximum(a0, 0.0))
    best_a = a.copy()
    best_val = _objective(a)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    while time.time() < deadline and t < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        t += 1
        # Adam moments
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        lr = lr0 / np.sqrt(t)  # optional decay
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project onto simplex to keep nonnegative and sum=1
        a = _project_onto_simplex(a_new)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if t % 2000 == 0:
                print(f"[AdamSubgrad it={t}] new best {val:.6f}")
    return best_a, best_val

"""
Improved step‑function search.

The algorithm consists of several time‑controlled stages:

1. Warm‑start & cheap handcrafted candidates.
2. Massive random Dirichlet sampling.  For each sample we try:
   – anti‑pair ordering,
   – a spaced permutation (a “step‑permutation” with a random coprime step).
   The best of these cheap evaluations seeds the optimisation.
3. Smooth‑max L‑BFGS refinement with a longer λ‑schedule (up to λ=1200),
   which makes the surrogate approach the true max more closely.
4. Adam‑style projected sub‑gradient optimisation on the exact max‑convex
   (gradient obtained from the active set of the convolution).
5. Local stochastic hill‑climbing (swap / fractional transfer) with a
   slowly‑decaying temperature.
All stages respect the global deadline (default 1000 s).  The final
vector is clipped to the limits accepted by ``evaluate_sequence``.
"""

import time
import math
import numpy as np
from scipy import optimize as _opt

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2) for a non‑negative vector."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Simple handcrafted starters
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Anti‑pair ordering
def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place largest opposite smallest (anti‑pair ordering)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                    # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    return _anti_pair_arrangement(vals)


# ----------------------------------------------------------------------
# Step‑permutation arrangement (evenly spaced large values)
def _step_permutation_candidate(vals: np.ndarray, step: int) -> np.ndarray:
    """
    Sort vals descending and place them on a cyclic lattice with step size `step`
    (step must be coprime to n so that the permutation is a full cycle).
    """
    n = len(vals)
    s = np.sort(vals)[::-1]
    perm = (np.arange(n) * step) % n
    a = np.empty_like(s)
    a[perm] = s
    return a


# ----------------------------------------------------------------------
# Warm‑start handling
def _warm_start(n: int):
    """Return a previously stored sequence (global `height_sequence_1`) if usable."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()


# ----------------------------------------------------------------------
# L‑BFGS optimisation of smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and its gradient w.r.t. logits."""
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp with temperature lam
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                       # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)       # chain rule through softmax
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 400.0, 1200.0),
                    max_iter_stage: int = 200) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively.
    """
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 8)))
        res = _opt.minimize(_lbfgs_obj, logits,
                            args=(n, conv_fft_len, corr_fft_len, lam),
                            method='L-BFGS-B', jac=True,
                            options={'maxiter': iters, 'disp': False})
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[L‑BFGS λ={lam:.0f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Adam‑style projected sub‑gradient optimisation (exact max)
def _adam_subgradient_optimize(a0: np.ndarray, n: int,
                               conv_fft_len: int, corr_fft_len: int,
                               deadline: float,
                               max_iters: int = 800_000,
                               lr0: float = 0.5,
                               beta1: float = 0.9,
                               beta2: float = 0.999,
                               eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """
    Adam‑style optimizer on the exact max‑convolution objective.
    The iterate stays on the probability simplex.
    """
    a = _project_onto_simplex(np.maximum(a0, 0.0))
    best_a = a.copy()
    best_val = _objective(a)

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0

    while time.time() < deadline and t < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        t += 1
        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        lr = lr0 / np.sqrt(t)          # simple decay on top of Adam’s schedule
        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a = _project_onto_simplex(a - step)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if t % 2000 == 0:
                print(f"[AdamSubgrad it={t}] new best {val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """Simple annealing on swaps / fractional mass transfers."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # decide move type
        if np.random.rand() < 0.5:               # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                                    # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # annealing acceptance
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
          F(a) = 2·n·max(a∗a) / (sum(a)²)
    within the given time budget (seconds).  Returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – we stick to 4096 points (fast FFT, good trade‑off)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 0) Warm start / cheap handcrafted candidates
    warm = _warm_start(n)
    if warm is not None:
        val = _objective(warm)
        best_seq, best_val = warm.copy(), val
        print(f"[Warm start] value {val:.6f}")

    starters = [
        _uniform_candidate(n),
        _triangle_candidate(n),
        _gaussian_edges_candidate(n, 0.3),
        _gaussian_edges_candidate(n, 0.7),
        _gaussian_edges_candidate(n, 1.2),
    ]
    for c in starters:
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # --------------------------------------------------------------
    # 1) Massive random warm‑starts (Dirichlet + two orderings)
    max_random = 3000                     # will be cut short if time runs out
    for idx in range(max_random):
        if time.time() >= deadline - 30:
            break
        # sample a concentration α on a wide log‑scale
        log_alpha = np.random.uniform(-2.5, 1.5)   # α ∈ [~0.003, ~31.6]
        alpha = 10.0 ** log_alpha
        vals = np.random.dirichlet(alpha * np.ones(n))

        # (a) anti‑pair ordering
        cand_ap = _anti_pair_arrangement(vals)
        val_ap = _objective(cand_ap)
        if val_ap < best_val:
            best_val, best_seq = val_ap, cand_ap.copy()
            print(f"[Random anti‑pair {idx}] new best {val_ap:.6f}")

        # (b) step‑permutation with a random coprime step
        step = np.random.randint(1, n)
        if math.gcd(step, n) != 1:
            continue
        cand_sp = _step_permutation_candidate(vals, step)
        val_sp = _objective(cand_sp)
        if val_sp < best_val:
            best_val, best_seq = val_sp, cand_sp.copy()
            print(f"[Random step‑perm {idx}] new best {val_sp:.6f}")

    # --------------------------------------------------------------
    # 2) L‑BFGS smooth‑max refinement
    if best_seq is not None and time.time() < deadline - 60:
        lbfgs_deadline = min(deadline - 20, time.time() + 180)
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         lbfgs_deadline,
                                         lam_schedule=(5, 15, 40, 120, 400, 1200),
                                         max_iter_stage=250)
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 3) Adam projected sub‑gradient optimisation (exact max)
    if best_seq is not None and time.time() < deadline - 30:
        sub_deadline = min(deadline - 10, time.time() + 200)
        seq_adam, val_adam = _adam_subgradient_optimize(best_seq, n,
                                                        conv_fft_len, corr_fft_len,
                                                        sub_deadline,
                                                        max_iters=800_000,
                                                        lr0=0.5)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[AdamSubgrad] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Local stochastic annealing
    if best_seq is not None and time.time() < deadline - 10:
        local_deadline = deadline - 5
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len,
                                                local_deadline,
                                                max_moves=300_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # Clip to the limits required by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose the final vector for a possible next run
    globals()["height_sequence_2"] = best_seq.tolist()
    return best_seq.tolist()
