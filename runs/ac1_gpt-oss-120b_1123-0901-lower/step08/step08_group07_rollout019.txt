"""
Propose a step‑function sequence with a low evaluation value.

Strategy
--------
1. **Diverse warm‑starts** – build many candidates from analytical families
   (uniform, Dirichlet, power‑law, U‑shape, Gaussian edges, exponential,
   spiky‑uniform, anti‑pair permutations, the previous best, etc.).
   All candidates are normalised to have L1‑norm = 1 and evaluated with a
   fast FFT‑based objective.

2. **Select the most promising** – keep the top‑k candidates (default 5).

3. **Continuous optimisation** – run an Adam optimiser on a smooth‑max
   surrogate followed by an exact sub‑gradient optimiser (exponentiated‑gradient)
   on each selected candidate.  Time is split among the candidates.

4. **Intensive local search** – a simulated‑annealing hill‑climber swaps or
   transfers mass between positions and accepts deteriorations with a decaying
   temperature.  This is run on the best sequence for the remaining time.

5. **Optional up‑sampling** – if time permits the solution is linearly
   interpolated to 8192 points and refined once more.

All operations respect the validation performed inside ``evaluate_sequence``.
The algorithm is deterministic apart from the random seeds (set by ``seed``)
and is designed to work comfortably within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks) ------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)^2).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Warm‑start helper -----------------------------------------------------
def _warm_start(n: int):
    """Return the global previous best (if present) resized to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
# Deterministic / random generators ------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest‑half left, smallest‑half right (anti‑pair)."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]
    a[-half:] = sorted_vals[-half:][::-1]
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = _random_dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(vals)


def _powerlaw_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    vals = i ** (-p)
    vals /= vals.sum()
    return vals


def _ushape_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = (i * (n - 1 - i) + 1.0) ** (-p)
    vals /= vals.sum()
    return vals


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _exponential_candidate(n: int, rate: float) -> np.ndarray:
    """a_i ∝ exp(-rate * i/(n-1)).  rate>0."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-rate * i / (n - 1))
    vals /= vals.sum()
    return vals


def _spike_candidate(n: int, m: int = 16, seed: int = 0) -> np.ndarray:
    """m equal spikes placed uniformly at random."""
    rng = np.random.RandomState(seed)
    positions = rng.choice(n, size=m, replace=False)
    a = np.zeros(n, dtype=np.float64)
    a[positions] = 1.0 / m
    return a


# ----------------------------------------------------------------------
# Optimisation primitives ------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c / (a.sum() ** 2)

        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a (exact for the surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # temperature schedule → true max
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient optimisation for the exact max objective
    (exponentiated‑gradient updates on the simplex).
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < 300_000:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decaying step size
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing hill‑climbing:
    - swap two entries OR transfer a fraction of mass.
    - accept if improvement or with probability exp(-Δ/Temp).
    Temperature decays linearly from 0.01 to 0.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:          # swap two entries
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:                               # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # up to 40 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5_000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    F(a) = 2·n·max(a∗a)/(sum(a)²) within the given time budget.
    Returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Build a large diverse pool of candidates
    candidates = []

    # warm start (if present) and some rolled variants
    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Simple deterministic shapes
    candidates.append(_uniform_candidate(n))
    candidates.append(_triangle_candidate(n))

    # Dirichlet samples (various concentrations)
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))

    # Anti‑pair Dirichlet samples (these are usually strong)
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        candidates.append(_anti_pair_candidate(n, alpha))

    # Power‑law and U‑shape families
    for p in (0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0):
        candidates.append(_powerlaw_candidate(n, p))
        candidates.append(_ushape_candidate(n, p))

    # Gaussian edges with different widths
    for gamma in (0.05, 0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))

    # Exponential decay (and its reverse)
    for rate in (0.5, 1.0, 2.0, 4.0, 6.0, 8.0):
        candidates.append(_exponential_candidate(n, rate))
        candidates.append(_exponential_candidate(n, rate)[::-1])

    # Spiky uniform (few equal spikes)
    for m in (8, 12, 16, 24):
        candidates.append(_spike_candidate(n, m, seed=seed + m))

    # Random permutations of a strong anti‑pair starter
    base_ap = _anti_pair_candidate(n, alpha=0.2)
    for _ in range(3):
        perm = np.random.permutation(n)
        candidates.append(base_ap[perm])

    # ------------------------------------------------------------------
    # 2) Quick evaluation and keep the best few
    evaluated = []
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand)
        evaluated.append((val, cand))
        if idx % 10 == 0:
            print(f"[Init cand {idx:02d}] value {val:.6f}")

    evaluated.sort(key=lambda x: x[0])
    top_k = 5
    best_seq = evaluated[0][1].copy()
    best_val = evaluated[0][0]
    print(f"Initial best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuous optimisation on the top candidates
    remaining = deadline - time.time()
    if remaining < 30.0:
        # not enough time for further refinement
        pass
    else:
        per_candidate_deadline = time.time() + remaining * 0.5 / top_k

        for cid, (_, cand) in enumerate(evaluated[:top_k]):
            # Adam phase
            adam_dead = time.time() + remaining * 0.25 / top_k
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_dead,
                max_iters=300_000,
                lr=0.07,
                lam_start=0.5,
                lam_factor=1.05,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Cand {cid} Adam] improved to {val_adam:.6f}")

            # Sub‑gradient phase
            sub_dead = time.time() + remaining * 0.25 / top_k
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead,
                eta0=0.6,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Cand {cid} Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic hill‑climbing on the current best
    if time.time() < deadline - 5.0:
        local_dead = deadline - 2.0
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_dead,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (8192) if time permits
    if time.time() < deadline - 30.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        big_adam_dead = min(deadline - 5.0, time.time() + 180.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_adam_dead,
            max_iters=250_000,
            lr=0.07,
            lam_start=0.5,
            lam_factor=1.05,
        )
        # sub‑gradient refinement on the large grid
        big_sub_dead = deadline
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_sub_dead,
            eta0=0.6,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Big] improved to {val_big_sub:.6f}")

    # ------------------------------------------------------------------
    # 6) Final safety – guarantee a valid return
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # ensure the sum is not too tiny (evaluate_sequence would reject otherwise)
    if best_seq.sum() < 0.01:
        best_seq = best_seq + 0.01 / best_seq.size

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
