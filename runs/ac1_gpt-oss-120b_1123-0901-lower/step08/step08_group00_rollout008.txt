from scipy.optimize import linprog
import scipy.sparse as sp

def solve_lp(f, M):
    n = len(f)
    rows = []
    cols = []
    data = []
    # for each j, for rows k = j .. j+n-1
    for j in range(n):
        # rows index range
        row_start = j
        row_end = j + n # exclusive
        # indices k from row_start to row_end-1
        rows.extend(range(row_start, row_end))
        cols.extend([j] * n)
        data.extend([f[j]] * n)
    A = sp.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    # Maximize sum b_j -> minimize -sum b_j
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        return None
```

"""
step‑function optimizer (≈1000 s budget)

This version focuses on a *large* grid (default 8192 points) and
allocates the majority of the time budget to a long exponentiated‑gradient
sub‑gradient phase, followed by a fine‑grained Adam (log‑sum‑exp) phase
and a high‑quality simulated‑annealing local search.
A warm start (the global variable ``height_sequence_1``) is up‑sampled
to the large grid; if none is available a uniform distribution is used.
All updates keep the vector on the probability simplex (sum = 1, ≥0).

The pipeline:

1. up‑sample warm start → a₀
2. **Phase A** – exponentiated‑gradient sub‑gradient (≈ 0.6 · budget)
3. **Phase B** – Adam on a smooth‑max surrogate (≈ 0.25 · budget)
4. **Phase C** – simulated‑annealing with mass‑transfer moves
   (≈ 0.15 · budget)

Final sequence is clipped to the evaluator limits (≥0, ≤1000) and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    ``w`` has length 2·len(a)-1 (the convolution length).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact value of the evaluation function for a (sum already =1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * a.shape[0] * conv.max() / (a.sum() ** 2)


def _expgrad_subgradient(
    a0: np.ndarray,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Keeps the vector on the simplex via multiplicative updates.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones_like(a))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    n = a.shape[0]

    while it < max_iters and time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, with tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        # renormalise – stay on the simplex, protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    The vector is parametrised by logits → soft‑max → simplex.
    Returns the best vector encountered.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # start on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                         # slightly larger initial learning rate
    lam = 0.01                         # start temperature (very smooth)
    lam_factor = 1.03                  # gradual sharpening

    best_a = a0.copy()
    best_val = _objective(a0, conv_fft_len)

    it = 0
    n = a0.shape[0]

    while it < max_iters and time.time() < deadline:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (same as for exact max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam moment updates
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase smoothing → concentrate)
        lam = min(lam * lam_factor, 1e7)

        # gentle learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Randomly transfers a fraction of mass from one index to another.
    Accepts uphill moves with probability exp(‑Δ/T).
    """
    a = a0.copy()
    n = a.shape[0]

    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 2e-4
    it = 0

    while it < max_moves and time.time() < deadline:
        # linear cooling
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 45 %) of the source mass
        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # accept uphill move with probability exp(-Δ/T)
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

    """
    Search for a non‑negative step‑function (probability distribution)

    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # 1.  Choose a starting vector.
    # Use the provided warm start if it exists; otherwise fall back
    # to the uniform distribution.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # keep only positive mass, normalise to a probability vector
        warm = np.maximum(warm, 0.0)
        if warm.sum() < 0.01:
            warm = None
        else:
            warm = warm / warm.sum()
    if warm is None:
        # uniform on a modest size (will be up‑sampled)
        warm = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # --------------------------------------------------------------
    # 2.  Upsample to a larger grid (more degrees of freedom).
    # Larger n gives the optimiser room to flatten the convolution.
    n_large = 8192
    x_old = np.linspace(0.0, 1.0, warm.shape[0], endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_large, endpoint=False)
    a = np.interp(x_new, x_old, warm)
    a = np.maximum(a, 0.0)
    a = a / a.sum()

    conv_fft_len = _next_pow2(2 * n_large - 1)
    corr_fft_len = _next_pow2(3 * n_large - 2)

    # --------------------------------------------------------------
    # 3.  Phase A – long exponentiated‑gradient sub‑gradient descent.
    # Allocate roughly 60 % of the remaining budget.
    phase_a_dead = time.time() + (deadline - time.time()) * 0.60
    a, val = _expgrad_subgradient(
        a,
        conv_fft_len,
        corr_fft_len,
        phase_a_dead,
        max_iters=800_000,
        eta0=1.0,
    )
    best_a = a
    best_val = val
    print(f"[Phase A] best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 4.  Phase B – Adam on a smooth‑max surrogate (more precise refinement).
    # Use about 30 % of the remaining time.
    if time.time() < deadline - 5:
        phase_b_dead = time.time() + (deadline - time.time()) * 0.30
        a, val = _adam_optimize(
            best_a,
            conv_fft_len,
            corr_fft_len,
            phase_b_dead,
            max_iters=300_000,
        )
        if val < best_val:
            best_val = val
            best_a = a
        print(f"[Phase B] best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 5.  Phase C – simulated‑annealing local optimisation.
    # Remaining time (≈ 10 %).
    if time.time() < deadline - 2:
        a, val = _anneal_optimize(
            best_a,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val < best_val:
            best_val = val
            best_a = a
        print(f"[Phase C] best value {best_val:.6f}")

    # --------------------------------------------------------------
    # 6.  Final sanity – clip to evaluator limits and return.
    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"FINAL best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
