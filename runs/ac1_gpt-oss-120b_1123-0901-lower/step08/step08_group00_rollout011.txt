"""
Propose a low‐value step function for the objective

    F(a) = 2·n·max(a∗a) / (∑a)² .

The method consists of three stages:

1. **Diversified seeding** – generate a modest number of candidate
   distributions (uniform, Dirichlet draws, exponential edges,
   double‑Gaussian blobs, sinusoid, and sparse spike patterns).  The
   best of those is kept as a warm start.

2. **Smooth‑max optimisation** – minimise a log‑sum‑exp surrogate for
   max(a∗a) with Adam on the soft‑max parametrisation of the simplex.
   The temperature λ is increased very slowly, keeping the landscape
   smooth for many iterations.

3. **Exact‑max sub‑gradient + targeted annealing** – continue the
   optimisation with exponentiated‑gradient updates that use the true
   active set of the max‑convolution, then perform many stochastic
   mass‑transfer moves that specifically reduce the current peak.
   The best vector found at any time is returned.

All work respects the supplied time budget; the routine always returns
a valid non‑negative sequence (the zero‑vector is never produced because
the sum is re‑normalised to 1).  No file I/O or external libraries are
used beyond NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns an array of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto {x | x≥0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Compute the objective 2·n·max(conv) (the denominator Σa = 1).
    a is assumed to be non‑negative and sum≈1.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (a.sum() ** 2)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a log‑sum‑exp surrogate for max(a∗a).
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                         # a bit larger (more exploration)
    lam = 0.05                        # initial temperature
    lam_factor = 1.005                # increase very slowly

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # Soft‑max -> probability vector on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (slowly approach the true max)
        lam = min(lam * lam_factor, 1e8)

        # occasional tiny random shake when close to the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with exponentiated‑gradient updates.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against loss of mass
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Randomly transfers a fraction of mass; uphill moves are accepted
    with a temperature‑dependent probability.  The move set also prefers
    indices that currently contribute heavily to the peak.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        # Identify the current peak index and its contributors
        conv = _full_convolution(a, conv_fft_len)
        k_max = int(conv.argmax())
        # contribution of each coordinate to the peak (single‑sided)
        contrib = np.zeros(n)
        low = max(0, k_max - (n - 1))
        high = min(n - 1, k_max)
        i_vals = np.arange(low, high + 1)
        contrib[i_vals] = a[i_vals] * a[k_max - i_vals]

        # Choose a source index proportionally to its contribution
        if contrib.sum() == 0:
            src = np.random.randint(n)
        else:
            src = np.random.choice(n, p=contrib / contrib.sum())

        # Choose a destination that currently carries little mass
        dst_candidates = np.where(a < a.mean())[0]
        if dst_candidates.size == 0:
            dst = np.random.randint(n)
        else:
            dst = np.random.choice(dst_candidates)

        if src == dst or a[src] < 1e-14:
            it += 1
            continue

        # Transfer a random fraction (up to 30 %) of the source mass
        frac = np.random.rand() * 0.3
        delta = a[src] * frac

        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta

        # evaluate new vector
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        # accept if improvement or according to Metropolis rule
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    # warm start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with a spread of concentration parameters
    for alpha in (0.02, 0.1, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # Exponential decay from left / right
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge shape
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian blobs
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.5):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Sparse equally spaced spikes
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Random sparse spikes with Dirichlet masses
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            cand.append(a)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    safety = 0.5                      # keep a small safety margin
    deadline = time.time() + max(0.0, budget_s - safety)

    # problem dimension – we keep it fixed (changing n scarcely helps)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous solution is stored in the global)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Phase I – generate a pool of diverse starts and keep the best one
    pool = _initial_candidates(n, warm)
    best_seq = None
    best_val = float("inf")

    print("[Phase I] evaluating initial candidates")
    while pool and time.time() < deadline:
        a0 = pool.pop(0)
        if a0.sum() < 1e-6:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        val = _objective(a0, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a0.copy()
            print(f"[Phase I] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – smooth‑max (Adam) optimisation of the current best
    if best_seq is not None and time.time() < deadline - 10:
        remain = deadline - time.time()
        adam_deadline = time.time() + min(0.35 * remain, 30.0)  # at most 30 s
        print("[Phase II] Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase II] Adam improved best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – exact‑max sub‑gradient (exponentiated‑gradient)
    if best_seq is not None and time.time() < deadline - 8:
        remain = deadline - time.time()
        sub_deadline = time.time() + min(0.45 * remain, 50.0)   # allocate most of the rest
        print("[Phase III] Sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=1_200_000, eta0=0.9
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase III] Sub‑gradient improved best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – targeted simulated‑annealing moves
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase IV] Annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] annealing improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty, valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
