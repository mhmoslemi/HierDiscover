def _guided_simulated_annealing(a0, final_deadline, start_temp=0.02, max_frac_start=0.4):
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2*n-1))
    best_a = a.copy()

    start = time.time()
    # precompute cumulative sum for source selection weighted by a
    total_iter = 0
    while time.time() < final_deadline:
        # compute temperature linearly decreasing (or exponential)
        elapsed = time.time() - start
        total_time = final_deadline - start
        temp = start_temp * max(0.0, 1.0 - elapsed / total_time)  # linear decrease
        # dynamic max_frac maybe decreasing as time passes

"""Step‑function optimisation for minimise  F(a)=2·n·max(a∗a)/(∑a)².

The optimiser works in three stages:

1. **Diverse warm‑starts** – builds a small pool of candidates
   (uniform, Dirichlet, anti‑pair, triangular, edge‑Gaussian, and a
   possible previous solution) and picks the best by the exact objective.

2. **Continuous refinement** – a smooth‑max Adam phase followed by an
   exact‑max exponentiated‑gradient sub‑gradient phase.  Both use FFT‑based
   convolutions and respect the time budget.

3. **Guided simulated annealing** – a fast O(n) incremental SA that
   transfers a fraction of mass from a source index (chosen with a bias
   proportional to its current mass) to a random target.  The step size
   and temperature decay linearly, and the move acceptance follows the
   standard Metropolis rule.  This phase dominates the remaining budget
   and typically performs millions of cheap moves.

If time permits after the SA on the initial length, the solution is
upsampled (linear interpolation) to a larger length (×2) and a short
second SA is performed, allowing a finer discretisation of the same
continuous shape.

All vectors are kept non‑negative and renormalised to sum≈1 (the
objective is scale‑invariant).  The final sequence is returned as a
plain list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a). Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no safety checks)
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²).  Assumes a is non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Candidate generators -------------------------------------------------
def _warm_start(n: int):
    """Load the previous solution (global variable) and adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() >= 0.01:
        warm /= warm.sum()
        return warm
    return None


def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Arrange a sorted vector so that large values are opposite small ones."""
    n = len(vals)
    sorted_vals = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = sorted_vals[:half]             # biggest half on the left
    a[-half:] = sorted_vals[-half:][::-1]     # smallest half on the right (reversed)
    if n % 2 == 1:
        a[half] = sorted_vals[half]
    return a


def _anti_pair_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    vals = np.random.dirichlet(alpha * np.ones(n))
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return _anti_pair_arrangement(vals)


def _triangle_candidate(n: int) -> np.ndarray:
    """Triangular shape – large at the edges, small in the centre."""
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                 # initialise in the interior of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # softmax → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (cheap, because sum(a)=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val_exact = 2.0 * n * max_c
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase smoothness parameter
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to avoid stagnation
        if it % 5_000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Incremental convolution utilities (used by SA)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Incrementally update the full linear convolution a*a when a[idx] changes by delta.
    The operation is O(n) via slice arithmetic.
    """
    # contribution of the self‑product term (position 2*idx)
    conv[2 * idx] += delta * delta
    # contribution with every other entry
    conv[idx : idx + a.shape[0]] += 2.0 * delta * a
    a[idx] += delta


def _guided_simulated_annealing(
    a0: np.ndarray,
    time_limit: float,
    start_temp: float = 0.03,
    max_frac_start: float = 0.4,
    min_frac: float = 0.001,
) -> tuple[np.ndarray, float]:
    """
    Fast SA that moves a random fraction of mass from a source index
    (chosen proportionally to its current mass) to a random target.
    The step‑size fraction and temperature decay linearly with elapsed time.
    """
    a = a0.copy()
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    conv = _full_convolution(a, fft_len)

    best_a = a.copy()
    best_val = conv.max()
    cur_val = best_val

    start = time.time()
    it = 0
    # pre‑compute an array of indices to speed up target selection
    all_idx = np.arange(n)

    while time.time() < time_limit:
        elapsed = time.time() - start
        remaining = max(0.0, time_limit - time.time())
        # Linear schedules
        frac = max_frac_start * (1.0 - elapsed / (time_limit - start)) + min_frac
        temp = start_temp * (remaining / (time_limit - start))

        # source index – bias proportional to current mass
        src = np.random.choice(n, p=a)          # np.choice is fast (C implementation)
        if a[src] < 1e-15:
            continue

        # pick a target different from src uniformly
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # amount to move
        delta = a[src] * np.random.rand() * frac
        if delta <= 0.0:
            continue

        # apply move (incremental update)
        _apply_delta(a, conv, src, -delta)
        _apply_delta(a, conv, tgt, delta)

        new_val = conv.max()
        diff = new_val - cur_val

        accept = False
        if diff <= 0.0:
            accept = True
        else:
            if temp > 0.0 and np.random.rand() < np.exp(-diff / temp):
                accept = True

        if accept:
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
        else:
            # revert move
            _apply_delta(a, conv, tgt, -delta)
            _apply_delta(a, conv, src, delta)

        it += 1
        if it % 200_000 == 0:
            print(f"[Guided SA it={it:07d}] best conv‑max {best_val:.9f}")

        # occasional tiny random perturbation to escape plateaus
        if it % 1_000_000 == 0:
            noise = np.random.normal(scale=1e-4, size=n)
            a += noise
            a = np.clip(a, 0.0, None)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s
            conv = _full_convolution(a, fft_len)
            cur_val = conv.max()
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
            print(f"[Guided SA restart it={it}] cur {cur_val:.9f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising
    F(a)=2·n·max(a∗a)/(∑a)² within the given time budget.
    The routine mixes diverse warm‑starts, Adam + sub‑gradient descent,
    and a long guided simulated‑annealing phase.  If time permits the
    final solution is up‑sampled to double the length and refined again.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------
    # Problem size – start with a medium resolution that balances quality / speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Build a pool of initial candidates (very cheap)
    candidates = []

    warm = _warm_start(n)
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    candidates.append(_uniform_candidate(n))
    for alpha in (0.3, 0.7, 1.0, 2.0):
        candidates.append(_random_dirichlet_candidate(n, alpha))
    for alpha in (0.3, 0.7, 1.0):
        candidates.append(_anti_pair_candidate(n, alpha))
    for gamma in (0.2, 0.5, 1.0):
        candidates.append(_gaussian_edges_candidate(n, gamma))
    candidates.append(_triangle_candidate(n))

    # --------------------------------------------------------------
    # 2) Pick the best starter (exact evaluation)
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand /= cand.sum()
        val = _objective(cand)
        print(f"[Init cand {idx:02d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()

    # --------------------------------------------------------------
    # 3) Continuous optimisation (Adam + sub‑gradient)
    # Allocate approximate time slices
    remaining = deadline - time.time()
    adam_time = min(0.3 * budget_s, remaining * 0.35)   # ~30 % of total budget
    sub_time  = min(0.2 * budget_s, remaining * 0.25)   # ~20 % of total budget

    # ---- Adam phase
    if time.time() + adam_time < deadline:
        adam_deadline = time.time() + adam_time
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=500_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient phase
    if time.time() + sub_time < deadline:
        sub_deadline = time.time() + sub_time
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=500_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Long guided simulated‑annealing local search
    if time.time() < deadline - 5.0:
        sa_deadline = deadline - 5.0
        a_sa, conv_max = _guided_simulated_annealing(
            best_seq,
            time_limit=sa_deadline,
            start_temp=0.03,
            max_frac_start=0.4,
            min_frac=0.001,
        )
        val_sa = 2.0 * n * conv_max          # exact objective (sum is 1)
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # --------------------------------------------------------------
    # 5) Optional up‑sampling to a finer grid (if we still have time)
    remaining = deadline - time.time()
    if remaining > 30.0:                     # at least half a minute left
        n2 = n * 2
        print(f"[Upsample] Refining on length {n2}")
        # linear interpolation, then rescale to keep total mass = 1
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n2, endpoint=False)
        up_seq = np.interp(x_new, x_old, best_seq)
        up_seq = np.maximum(up_seq, 0.0)
        if up_seq.sum() < 0.01:
            up_seq = np.random.dirichlet(np.ones(n2))
        up_seq /= up_seq.sum()

        conv_fft_len2 = _next_pow2(2 * n2 - 1)

        # a short SA on the finer grid
        a_fine, conv_max_fine = _guided_simulated_annealing(
            up_seq,
            time_limit=deadline - 2.0,
            start_temp=0.02,
            max_frac_start=0.2,
            min_frac=5e-4,
        )
        val_fine = 2.0 * n2 * conv_max_fine
        if val_fine < best_val:
            best_val = val_fine
            best_seq = a_fine.copy()
            n = n2
            print(f"[Fine SA] improved to {val_fine:.6f}")

    # --------------------------------------------------------------
    # 6) Final safety checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    total = best_seq.sum()
    if total > 0:
        best_seq /= total
    else:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # expose the result for possible warm‑starts of future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
