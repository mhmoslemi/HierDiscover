"""
Propose a step‑function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

using a multi‑restart gradient / sub‑gradient scheme.

The algorithm works in two resolution levels:

1. **Small resolution (n≈1024)** – many cheap initialisations are generated
   (uniform, power‑law, exponential, Gaussian‑edge, Dirichlet samples and
   random permutations).  Each start is refined with a short Adam run on the
   smooth‑max surrogate and a short exact‑max sub‑gradient run.  The best
   small‑scale vector is kept.

2. **Large resolution (n=4096)** – the best small vector is interpolated to the
   final size (or the existing warm‑start is used).  On this vector we run a
   longer Adam phase, a longer sub‑gradient phase, a simulated‑annealing style
   local search and finally a fine‑tuned Adam run with a very high temperature
   (λ) to squeeze out the remaining slack.

All optimisation respects the global time budget (default 1000 s).  The
returned sequence is clipped to the evaluator’s admissible range
[0, 1000] and summed to 1 (the objective is scale‑invariant).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a of length 2·len(a)-1 via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) (assumes sum(a)=1)."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Shape generators (produce vectors with sum = 1)
def _uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_shape(n: int, p: float) -> np.ndarray:
    vals = np.exp(-p * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _gaussian_edges_shape(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals /= vals.sum()
    return vals

    warm = globals().get("height_sequence_1")
    if warm is None:

    warm = np.maximum(warm, 0.0)

    return warm

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,

    Adam on the smooth‑max (log‑sum‑exp) surrogate of max(a*a).
    Returns the best exact objective encountered.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # parametrise a via softmax
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

            best_a = a.copy()

        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # move to log‑space (mirror‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter
        if it % 5000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val

# Exact‑max sub‑gradient (exponentiated‑gradient) refinement
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient refinement using an exponentiated‑gradient update.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of max indices → uniform sub‑gradient
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto simplex
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Stochastic local search (swap / fractional transfer)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill climbing with swaps and mass transfers.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.01
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:
            # swap two coordinates
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:
            # fractional transfer i → j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)  # up to 40 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # simple annealing acceptance
            temp = start_temp * (1.0 - (it / max_moves))
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local move {it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Candidate generators for the *small* resolution
def _generate_small_candidates(n: int) -> list[np.ndarray]:
    """Return a list of diverse starter vectors (all sum to 1)."""
    cand = []

    # warm start, if any
    warm = _warm_start_n(n)
    if warm is not None:
        cand.append(warm.copy())

    # deterministic families
    cand.append(_uniform_shape(n))

    for p in np.linspace(0.05, 2.0, 10):
        cand.append(_powerlaw_shape(n, float(p)))
        cand.append(_exponential_shape(n, float(p)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        cand.append(_gaussian_edges_shape(n, gamma))

    # Dirichlet samples (different concentrations)
    for alpha in (0.3, 0.7, 1.0, 2.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # random permutations of the first few deterministic vectors
    for base in cand[:6]:
        perm = base.copy()
        np.random.shuffle(perm)
        cand.append(perm)

    # normalise (just in case)
    out = []
    for v in cand:
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            v = np.full_like(v, 1.0 / n)
        else:
            v = v / s
        out.append(v)
    return out

# Small‑resolution optimisation (quick Adam + sub‑gradient)
def _optimize_small(a0: np.ndarray, n: int, deadline: float) -> tuple[np.ndarray, float]:
    """Run a short Adam + sub‑gradient chain on the small problem."""
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # quick Adam (≈80 k iters or time‑limited)
    a_adam, val_adam = _adam_optimize(
        a0,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=80_000,
        lr=0.07,
        lam_start=1.0,
        lam_factor=1.07,
    )

    # quick sub‑gradient refinement
    a_sub, val_sub = _subgradient_optimize(
        a_adam,
        n,
        conv_fft_len,
        corr_fft_len,
        deadline,
        max_iters=80_000,
        eta0=0.7,
    )

    if val_adam <= val_sub:
        return a_adam, val_adam
    else:
        return a_sub, val_sub

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine respects the supplied time budget
    and returns the best sequence found (as a Python list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    # --------------------------------------------------------------
    # 1)  Small‑scale exploration (n≈1024) – many cheap restarts.
    n_small = 1024
    small_candidates = _generate_small_candidates(n_small)

    best_small = None
    best_small_val = float("inf")

    for idx, cand in enumerate(small_candidates):
        # give each start a tiny slice of the remaining budget
        if time.time() + 5.0 > deadline:
            break
        # allocate at most 30 % of the remaining time for this start,
        # but never less than 10 s to let the optimiser move.
        remaining = deadline - time.time()
        per_start = min(30.0, max(10.0, 0.2 * remaining))
        start_deadline = time.time() + per_start

        a_opt, val_opt = _optimize_small(cand, n_small, start_deadline)

        print(f"[Small cand {idx:02d}] value {val_opt:.6f}")

        if val_opt < best_small_val:
            best_small_val = val_opt
            best_small = a_opt.copy()
            print(f"   --> new small‑scale best {best_small_val:.6f}")

    # --------------------------------------------------------------
    # 2) Upscale the best small vector to the final resolution (n=4096)
    n_big = 4096
    if best_small is None:
        best_big = _uniform_shape(n_big)
    else:
        x_small = np.linspace(0.0, 1.0, n_small, endpoint=False)
        x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
        best_big = np.interp(x_big, x_small, best_small)
        best_big = np.maximum(best_big, 0.0)
        best_big /= best_big.sum()

    # also consider the global warm‑start, if it beats the interpolated one
    warm_big = _warm_start_n(n_big)
    if warm_big is not None:
        warm_val = _objective(warm_big)
        interp_val = _objective(best_big)
        if warm_val < interp_val:
            best_big = warm_big.copy()
            print(f"[Warm] using warm‑start with value {warm_val:.6f}")

    best_val = _objective(best_big)
    print(f"[Big init] objective {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Full‑scale optimisation on n=4096
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # ---- Adam phase (≈40 % of remaining budget) --------------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        adam_deadline = time.time() + max(30.0, 0.4 * time_left)
        a_adam, val_adam = _adam_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_deadline,
            max_iters=300_000,
            lr=0.07,
            lam_start=1.0,
            lam_factor=1.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_big = a_adam.copy()
            print(f"[Big Adam] improved to {val_adam:.6f}")

    # ---- Sub‑gradient phase (≈30 % of remaining budget) -------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        sub_deadline = time.time() + max(30.0, 0.3 * time_left)
        a_sub, val_sub = _subgradient_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_deadline,
            max_iters=300_000,
            eta0=0.7,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_big = a_sub.copy()
            print(f"[Big Subgrad] improved to {val_sub:.6f}")

    # ---- Local stochastic hill‑climbing (≈20 % of remaining budget) --
    time_left = deadline - time.time()
    if time_left > 15.0:
        local_deadline = time.time() + max(20.0, 0.2 * time_left)
        a_loc, val_loc = _local_random_search(
            best_big,
            n_big,
            conv_fft_len_big,
            local_deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_big = a_loc.copy()
            print(f"[Big Local] improved to {val_loc:.6f}")

    # ---- Final fine‑tuned Adam with a very high λ (≈remaining time) ----
    time_left = deadline - time.time()
    if time_left > 5.0:
        fine_deadline = time.time() + time_left
        a_fine, val_fine = _adam_optimize(
            best_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            fine_deadline,
            max_iters=200_000,
            lr=0.03,
            lam_start=10.0,
            lam_factor=1.02,
        )
        if val_fine < best_val:
            best_val = val_fine
            best_big = a_fine.copy()
            print(f"[Big Fine‑Adam] improved to {val_fine:.6f}")

    # --------------------------------------------------------------
    # 4) Final safety clipping and return
    best_big = np.clip(best_big, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_big)})")
    return best_big.tolist()
