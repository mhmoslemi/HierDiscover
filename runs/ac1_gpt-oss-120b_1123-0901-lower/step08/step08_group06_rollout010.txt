def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n -1)
    corr_fft_len = _next_pow2(3*n -2)

    best_val = float('inf')
    best_seq = None

    if warm is not None:
        val = _objective(warm)
        if val < best_val:
            best_val, best_seq = val, warm.copy()
            print(f"[Warm start] value {val:.6f}")

    # Handcrafted starters
    for c_func in [_uniform_candidate, _triangle_candidate,
                   lambda n: _gaussian_edges_candidate(n, 0.3),
                   lambda n: _gaussian_edges_candidate(n, 0.7),
                   lambda n: _gaussian_edges_candidate(n, 1.2)]:
        c = c_func(n)
        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] {c_func.__name__} value {val:.6f}")

"""
Propose a non‑negative step‑function that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

The search proceeds in several stages inside the 1000 s budget:

1. **Warm‑start** – use a previously found sequence if provided.
2. **Targeted random sampling** – Dirichlet draws with several
   deterministic rearrangements (identity, anti‑pair, interleaved) and
   a few low‑dimensional parametric families (Beta, powered‑sine,
   cosine‑power, centred Gaussian).  The best of these is kept.
3. **Smooth‑max L‑BFGS refinement** – optimises a log‑sum‑exp
   surrogate of the max‑convolution, gradually increasing the
   temperature (λ) to approach the true objective.
4. **Exact‑max multiplicative sub‑gradient (mirror descent)** with a
   long diminishing step‑size schedule.
5. **Local stochastic hill‑climbing** – random swaps / mass transfers
   with a simple annealing rule.
6. **Optional up‑sampling** – interpolate the current best vector to
   8192 points and repeat steps 3‑4 on the finer grid.

All heavy work respects the global deadline; the routine always returns
a valid (clipped) list even if the time runs out early.
"""
import time
import numpy as np
import scipy.optimize as opt

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact) ----------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(a*a) / (sum(a)²); a must be non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) ----------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Warm‑start handling ----------------------------------------------------
def _warm_start(n: int):
    """If a global sequence `height_sequence_1` exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    # normalise – objective is scale‑invariant, any normalisation works
    return warm / warm.sum()


# ----------------------------------------------------------------------
# Simple deterministic arrangements ---------------------------------------
def _identity_arrangement(vals: np.ndarray) -> np.ndarray:
    """Leave values as they are."""
    return vals.copy()


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest half first, smallest half reversed (existing anti‑pair)."""
    n = len(vals)
    s = np.sort(vals)[::-1]                     # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _interleaved_arrangement(vals: np.ndarray) -> np.ndarray:
    """
    Place the sorted values alternating from the ends towards the centre.
    This spreads large entries as far apart as possible.
    """
    n = len(vals)
    s = np.sort(vals)[::-1]                     # descending
    a = np.empty(n, dtype=np.float64)
    left, right = 0, n - 1
    idx = 0
    while left <= right:
        a[left] = s[idx]
        idx += 1
        left += 1
        if left <= right:
            a[right] = s[idx]
            idx += 1
            right -= 1
    return a


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangle_candidate(n: int) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    shape = np.minimum(x + 1, n - x)          # 1,2,…,⌈n/2⌉,…,2,1
    shape /= shape.sum()
    return shape


def _gaussian_edges_candidate(n: int, gamma: float = 0.5) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    shape /= shape.sum()
    return shape


def _gaussian_central_candidate(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    shape /= shape.sum()
    return shape


def _beta_candidate(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = (i ** (alpha - 1.0)) * ((n + 1 - i) ** (beta - 1.0))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _sin_power_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** p
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _cosine_power_candidate(n: int, p: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate -------------------------
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """
    Smooth‑max (log‑sum‑exp) surrogate and its gradient w.r.t. logits.
    """
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # log‑sum‑exp stabilised
    z = np.exp(lam * (conv - max_c))
    sum_z = np.sum(z)
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    w = z / sum_z                      # soft‑weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)     # chain‑rule through softmax

    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 300.0),
                    max_iter_stage: int = 300) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smooth‑max surrogate, successively increasing λ.
    Returns the best vector found and its exact objective value.
    """
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 10)))
        res = opt.minimize(_lbfgs_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': iters, 'disp': False})
        logits = res.x
        # recompute a from logits
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Multiplicative (exponentiated‑gradient) sub‑gradient optimisation -----
def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 2_000_000,
                          eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex with a diminishing step size.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)        # multiplicative update
        s = a.sum()
        if s < 1e-12:                         # re‑initialise if numerically collapsed
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[MulSubgrad it={it:07d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Local stochastic hill‑climbing (simulated annealing) -----------------
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 600_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        if np.random.rand() < 0.5:                 # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                      # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a = a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:07d}] best {best_val:.6f}")
        it += 1
    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function within the given time budget.  Returns the best sequence found.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_val = float("inf")

    # 0) Warm start (if supplied)

    for func in starters:

    ]
    for _ in range(fam_iters):

            break
        name, maker = families[np.random.randint(len(families))]
        cand0 = maker()
        # interleaved arrangement tends to spread mass best for these shapes
        cand = _interleaved_arrangement(cand0)

            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192 points)

                                                         final_deadline,

        if val_big_sub < best_val:
            best_val, best_seq = val_big_sub, seq_big_sub.copy()
            print(f"[Big grid] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # Safeguard – guarantee a valid return

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
