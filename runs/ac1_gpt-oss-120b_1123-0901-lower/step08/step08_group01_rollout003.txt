def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np, scipy.optimize as opt
    start = time.time()
    deadline = start + budget_s - 0.5
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # Determine n target; try larger n if time budget > some threshold
    # Use n = 8192 if we think beneficial; else fallback to 4096.
    # Let's attempt n=8192 if budget_s > 600 sec else 4096.
    if budget_s > 600:
        n = 8192
    else:
        n = 4096

    conv_len = _next_pow2(2*n-1)
    # Precompute uniform shape and its FFT
    uniform = shape_uniform(n)
    fft_uniform = np.fft.rfft(uniform, conv_len)

    # Base shapes will be generated on the fly for parameters.

    # Objective function for DE
    def de_obj(x):
        # x: shape (8,)
        # weight logits
        w_logits = x[:4]
        gamma_log = x[4]
        sigma_log = x[5]
        a_log = x[6]
        b_log = x[7]

        # weights via softmax
        maxlog = np.max(w_logits)
        wraw = np.exp(w_logits - maxlog)
        w = wraw / wraw.sum()

        # parameters
        gamma = np.exp(gamma_log)
        sigma = np.exp(sigma_log)
        a = np.exp(a_log)
        b = np.exp(b_log)

        # generate param shapes and FFTs
        edge = shape_edge_exponential(n, gamma)
        fft_edge = np.fft.rfft(edge, conv_len)

        gauss = shape_double_gaussian(n, sigma)
        fft_gauss = np.fft.rfft(gauss, conv_len)

        beta = shape_beta(n, a, b)
        fft_beta = np.fft.rfft(beta, conv_len)

        mix_fft = w[0]*fft_uniform + w[1]*fft_edge + w[2]*fft_gauss + w[3]*fft_beta

        conv = np.fft.irfft(mix_fft * mix_fft, conv_len)[: 2*n - 1]
        max_c = conv.max()
        # sum of mixture = sum_i w_i*1 =1
        val = 2.0 * n * max_c  # denominator sum^2 =1
        return val

    # set bounds
    weight_bounds = [(-10,10)]*4
    gamma_bounds = (np.log(0.01), np.log(20.0))
    sigma_bounds = (np.log(0.01), np.log(0.5))
    a_bounds = (np.log(0.1), np.log(10.0))
    b_bounds = (np.log(0.1), np.log(10.0))
    bounds = weight_bounds + [gamma_bounds, sigma_bounds, a_bounds, b_bounds]

    # DE callback for deadline
    def de_callback(xk, convergence):
        # return True to stop
        return time.time() > deadline - 5.0  # stop slightly before final

    # Run DE with limited time
    # Determine max_iter maybe 200, popsize 10
    options = {'maxiter': 200, 'popsize': 12, 'disp': False, 'tol': 1e-5}
    # Use differential_evolution
    try:
        result = opt.differential_evolution(de_obj, bounds, seed=seed, updating='deferred', workers=1,
                                            callback=de_callback, polish=False, **options)
        best_weights = result.x
    except Exception as e:
        print("DE failed:", e)
        best_weights = None

    # Acquire best mixture
    if best_weights is not None:
        # Evaluate best_weights and obtain mixture shape
        w_logits = best_weights[:4]
        gamma = np.exp(best_weights[4])
        sigma = np.exp(best_weights[5])
        a = np.exp(best_weights[6])
        b = np.exp(best_weights[7])
        # compute weights softmax
        maxlog = np.max(w_logits)
        wraw = np.exp(w_logits - maxlog)
        w = wraw / wraw.sum()
        # compute shapes
        edge = shape_edge_exponential(n, gamma)
        gauss = shape_double_gaussian(n, sigma)
        beta = shape_beta(n, a, b)
        # mixture
        a0 = w[0]*uniform + w[1]*edge + w[2]*gauss + w[3]*beta
        # ensure sum approx 1, normalize just in case
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 = a0 / a0.sum()
    else:

        a0 = uniform.copy()

    # Evaluate initial best value
    best_seq = a0.copy()
    conv_len = conv_len # already defined
    best_val = eval_obj(best_seq, n, conv_len)
    print(f"[Init] DE best objective {best_val:.6f}")

    # Phase 2: refined subgradient optimization

    remaining = deadline - time.time()
    if remaining > 20.0:
        # Allocate time, compute max_iters based on estimated iter per sec

"""
Propose a low‑value step‑function by combining a few analytically‑motivated
families (uniform, edge‑exponential, double‑Gaussian, Beta‑pdf) and
optimising the mixture weights and the family parameters with a fast
differential‑evolution (DE) surrogate.  The DE result is refined by a
large‑scale exponentiated‑gradient (sub‑gradient) optimiser and a final
cheap random‑move hill‑climber.  All phases respect the global deadline,
use FFT‑based convolutions and stay strictly within the evaluator’s
non‑negative, bounded domain.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape generators (each returns a probability vector of length n)
def _shape_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _shape_edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _shape_double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _shape_beta(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


# ----------------------------------------------------------------------
# Optimisers (same as in the previous baseline, but trimmed)
def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Euclidean projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climber: move a random fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)  # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine obeys the time budget and always returns the best sequence
    (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # --------------------------------------------------------------
    # Choose a problem size.
    # If we have plenty of time we go to a larger n, which gives a better
    # asymptotic bound (the limit seems to be ~1.5).  Otherwise we stay at 4096.
    if budget_s > 650:
        n = 8192
    else:
        n = 4096

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # pre‑compute the uniform shape and its FFT (the uniform family needs no parameters)
    uniform = _shape_uniform(n)
    fft_uniform = np.fft.rfft(uniform, conv_fft_len)

    # ------------------------------------------------------------------
    # Differential‑Evolution on a low‑dimensional mixture model
    # ------------------------------------------------------------------
    def de_obj(x):
        """
        x has 8 components:
            0‑3 : logits for the four families (uniform, edge‑exp, gauss, beta)
            4   : log(gamma)   for edge‑exponential
            5   : log(sigma)   for double‑Gaussian
            6   : log(a)       for Beta
            7   : log(b)       for Beta
        The function returns the exact objective value (no scaling needed,
        because each family is already normalised to sum=1).
        """
        # logits → softmax weights
        w_logits = x[:4]
        maxlog = np.max(w_logits)
        w_raw = np.exp(w_logits - maxlog)
        w = w_raw / w_raw.sum()

        # family parameters (positive)
        gamma = np.exp(x[4])
        sigma = np.exp(x[5])
        a = np.exp(x[6])
        b = np.exp(x[7])

        # generate the three parametric families (uniform is pre‑computed)
        edge = _shape_edge_exponential(n, gamma)
        gauss = _shape_double_gaussian(n, sigma)
        beta = _shape_beta(n, a, b)

        # FFT of the three families
        fft_edge = np.fft.rfft(edge, conv_fft_len)
        fft_gauss = np.fft.rfft(gauss, conv_fft_len)
        fft_beta = np.fft.rfft(beta, conv_fft_len)

        # convex combination in the Fourier domain (linear because FFT is linear)
        mix_fft = (
            w[0] * fft_uniform
            + w[1] * fft_edge
            + w[2] * fft_gauss
            + w[3] * fft_beta
        )

        conv = np.fft.irfft(mix_fft * mix_fft, conv_fft_len)[: 2 * n - 1]
        max_c = conv.max()
        # sum of the mixture = 1, therefore denominator = 1
        val = 2.0 * n * max_c
        return val

    # Bounds for the 8 parameters
    weight_bounds = [(-10.0, 10.0)] * 4
    gamma_bounds = (np.log(0.01), np.log(20.0))
    sigma_bounds = (np.log(0.01), np.log(0.5))
    a_bounds = (np.log(0.1), np.log(10.0))
    b_bounds = (np.log(0.1), np.log(10.0))
    bounds = weight_bounds + [gamma_bounds, sigma_bounds, a_bounds, b_bounds]

    # callback to stop DE a few seconds before the overall deadline
    def de_callback(xk, convergence):
        return time.time() > deadline - 5.0

    # Run DE with a modest budget (fast enough to leave time for deep refinements)
    options = {
        "maxiter": 200,
        "popsize": 12,
        "tol": 1e-6,
        "disp": False,
        "polish": False,
        "updating": "deferred",
    }
    try:
        result = differential_evolution(
            de_obj,
            bounds,
            seed=seed,
            callback=de_callback,
            **options,
        )
        best_x = result.x
    except Exception as e:
        print("Differential evolution failed:", e)
        best_x = None

    # --------------------------------------------------------------
    # Build the actual mixture vector from the DE solution (or fall back)
    if best_x is not None:
        w_logits = best_x[:4]
        maxlog = np.max(w_logits)
        w_raw = np.exp(w_logits - maxlog)
        w = w_raw / w_raw.sum()

        gamma = np.exp(best_x[4])
        sigma = np.exp(best_x[5])
        a_par = np.exp(best_x[6])
        b_par = np.exp(best_x[7])

        edge = _shape_edge_exponential(n, gamma)
        gauss = _shape_double_gaussian(n, sigma)
        beta = _shape_beta(n, a_par, b_par)

        a0 = (
            w[0] * uniform
            + w[1] * edge
            + w[2] * gauss
            + w[3] * beta
        )
        a0 = np.maximum(a0, 0.0)
        if a0.sum() > 0:
            a0 /= a0.sum()
    else:
        # Very simple fallback – uniform distribution
        a0 = uniform.copy()

    # Initial objective (for logging)
    best_seq = a0.copy()
    best_val = de_obj(np.concatenate([np.log(w + 1e-12), np.log([gamma, sigma, a_par, b_par])]))
    print(f"[Init] DE mixture objective {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – deep sub‑gradient refinement (exponentiated‑gradient)
    if time.time() < deadline - 20.0:
        # allocate remaining budget proportionally to expected iteration cost
        remaining = deadline - time.time()
        max_iters = int(min(300_000, max(10_000, remaining * 800)))  # ~800 iters/sec estimate
        a_refined, val_refined = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=max_iters,
            eta0=0.8,
        )
        if val_refined < best_val:
            best_val = val_refined
            best_seq = a_refined.copy()
            print(f"[Subgrad refine] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – projected‑gradient descent for a different dynamics
    if time.time() < deadline - 10.0:
        a_refined2, val_refined2 = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
            eta0=0.4,
        )
        if val_refined2 < best_val:
            best_val = val_refined2
            best_seq = a_refined2.copy()
            print(f"[PGD refine] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – final cheap random local moves (hill‑climber)
    if time.time() < deadline - 2.0:
        # use a move budget scaled to the remaining seconds
        moves_budget = int(min(200_000, max(10_000, (deadline - time.time()) * 500)))
        a_local, val_local = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=moves_budget,
        )
        if val_local < best_val:
            best_val = val_local
            best_seq = a_local.copy()
            print(f"[Local improve] new best {best_val:.6f}")

    # --------------------------------------------------------------
    # Clip to evaluator's accepted domain and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
