"""
Propose a non‑negative step function (list of floats) minimizing

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm follows a multi‑phase strategy:

1. **Broad random sampling** – Dirichlet draws, spikes, mixtures,
   exponential / sinusoidal shapes.  The best candidate is kept.
2. **Long Adam run** on the smooth‑max (log‑sum‑exp) surrogate.
   A very slow temperature schedule (lam_factor≈1.003) is used so that
   the surrogate stays close to the true max.
3. **Noisy‑restart Adam** – a few short Adam runs from perturbed copies
   of the current best.
4. **Linear‑program refinement** – for the current vector `f` compute
   `M = max(f*f)`.  Solve the LP

        max   Σ b_j
        s.t.  f * b ≤ M   (linear convolution)
              b_j ≥ 0

   (implemented with CVXPY’s `conv` which treats the convolution as a
   linear operator).  The LP gives a vector `b` with larger ℓ₁‑norm while
   respecting the same bound `M`.  Rescale `b` to the original ℓ₁‑norm,
   then move the current iterate towards it.  Repeat while time permits.
5. **Final short Adam + cheap hill‑climb** – polish the solution.

All phases respect the overall time budget (default 1000 s) and
the best sequence found so far is always returned.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the Adam gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _eval_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation of the objective (no security checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.04,
    lam_start: float = 0.02,
    lam_factor: float = 1.003,
    noise_sigma: float = 0.015,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (ℓ₁‑normalised) and its exact objective.
    """
    eps = 1e-15
    a0 = np.clip(a0, eps, None)
    logits = np.log(a0)                     # initialise on the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while it < max_iters and time.time() < deadline:
        # ----- simplex (soft‑max) projection ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # optional noise injection
        if noise_sigma > 0:
            grad_log += np.random.normal(scale=noise_sigma, size=grad_log.shape)

        # ----- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule ----------
        lam = min(lam * lam_factor, 1e6)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.9

        # final‑minute shaking
        if it % 5_000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a random fraction of mass from one
    coordinate to another, keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while it < max_moves and time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % move
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val


def _lp_max_sum(f: np.ndarray, M: float, deadline: float) -> np.ndarray | None:
    """
    Solve the LP:
        max   Σ b_j
        s.t.  conv(f, b) ≤ M   (element‑wise)
              b ≥ 0
    Returns the optimal b (as a dense ndarray) or None on failure.
    """
    n = f.shape[0]
    b = cp.Variable(n, nonneg=True)
    # linear convolution of constant f with variable b
    conv_fb = cp.conv(f, b)          # length 2·n‑1
    constraints = [conv_fb <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)
    # Solve with ECOS (fast for moderate‑size LPs)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                    max_iters=5000, abstol=1e-7, reltol=1e-7,
                    feastol=1e-7, verbose=False)
    except Exception as e:
        print("[LP] solver exception:", e)
        return None
    if b.value is None:
        return None
    b_arr = np.maximum(np.asarray(b.value, dtype=np.float64), 0.0)
    # Ensure we finish before the global deadline
    if time.time() > deadline:
        return None
    return b_arr

        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # -------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    def try_candidate(a):
        nonlocal best_seq, best_val
        val = _eval_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random] new best {val:.6f}")

    # ---- baseline candidates -------------------------------------------------
    if warm is not None:
        try_candidate(warm)
    else:
        try_candidate(np.full(n, 1.0 / n))

    # Dirichlet (full support)
    for _ in range(36):
        a = np.random.dirichlet(np.ones(n))
        try_candidate(a)

    # Exponential tails (left/right)
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a /= a.sum()
        try_candidate(a)

    # Half‑wave sinusoid
    a = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    try_candidate(a)

    # Spike candidates – many random supports
    for k in (32, 64, 128, 256, 512, 1024):
        for _ in range(12):
            idx = np.random.choice(n, size=k, replace=False)
            weights = np.random.dirichlet(np.ones(k))
            a = np.zeros(n, dtype=np.float64)
            a[idx] = weights
            try_candidate(a)

    # Two‑Gaussian mixtures
    xs = np.arange(n, dtype=np.float64) / n
    for _ in range(20):
        mu1, mu2 = np.random.rand(2)
        sigma1, sigma2 = np.random.rand(2) * 0.12 + 0.03
        g1 = np.exp(-0.5 * ((xs - mu1) / sigma1) ** 2)
        g2 = np.exp(-0.5 * ((xs - mu2) / sigma2) ** 2)
        a = g1 + g2
        a /= a.sum()
        try_candidate(a)

    # -------------------------------------------------------------
    # Long Adam optimisation (≈60 % of remaining budget)
    remaining = deadline - time.time()
    adam_deadline = time.time() + 0.60 * remaining
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=400_000,
        lr=0.04,
        lam_start=0.02,
        lam_factor=1.003,
        noise_sigma=0.015,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam main] improved best {val_adam:.6f}")

    # -------------------------------------------------------------
    # Noisy‑restart Adam (few short runs)
    for restart in range(4):
        if time.time() >= deadline - 30:
            break
        eps = 1e-12
        logits = np.log(np.clip(best_seq, eps, None))
        logits += np.random.normal(scale=0.35, size=n)   # bigger perturbation
        a0 = np.exp(logits - logits.max())
        a0 /= a0.sum()
        r_deadline = min(deadline - 5, time.time() + 0.12 * (deadline - time.time()))
        a_tmp, val_tmp = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            r_deadline,
            max_iters=200_000,
            lr=0.035,
            lam_start=0.02,
            lam_factor=1.002,
            noise_sigma=0.012,
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Adam restart {restart}] new best {val_tmp:.6f}")

    # -------------------------------------------------------------
    # Linear‑program refinement (if time permits)
    lp_iters = 0
    while time.time() < deadline - 30 and lp_iters < 8:
        conv = _full_convolution(best_seq, conv_fft_len)
        M = conv.max()
        sum_f = best_seq.sum()

        b_opt = _lp_max_sum(best_seq, M, deadline)
        if b_opt is None:
            break
        sum_b = b_opt.sum()
        if sum_b <= 0:
            break

        # rescale b to keep ℓ₁‑norm unchanged
        g = b_opt * (sum_f / sum_b)

        # try both a full step and a convex combination
        candidates = []
        candidates.append(g)                              # full step
        candidates.append((0.6 * best_seq + 0.4 * g))     # blending

        improved = False
        for cand in candidates:
            cand = np.clip(cand, 0.0, None)
            cand /= cand.sum()               # normalise ℓ₁ to avoid drift
            val = _eval_objective(cand, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                improved = True
                print(f"[LP it={lp_iters}] new best {val:.6f}")
                break
        if not improved:
            # if neither candidate improves, reduce step size and retry
            # (just move a little towards g)
            best_seq = 0.9 * best_seq + 0.1 * g
        lp_iters += 1

    # -------------------------------------------------------------
    # Short Adam polish (≈10 % of remaining time)
    if time.time() < deadline - 15:
        polish_deadline = time.time() + 0.10 * (deadline - time.time())
        a_polish, val_polish = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            polish_deadline,
            max_iters=150_000,
            lr=0.03,
            lam_start=0.015,
            lam_factor=1.002,
            noise_sigma=0.01,
        )
        if val_polish < best_val:
            best_val = val_polish
            best_seq = a_polish.copy()
            print(f"[Adam polish] improved best {val_polish:.6f}")

    # -------------------------------------------------------------
    # Final cheap hill‑climb
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved best {val_loc:.6f}")

    # -------------------------------------------------------------
    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
