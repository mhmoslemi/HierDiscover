### Implementation details for project_simplex

We'll implement standard algorithm: given v, compute u = sort(v)[::-1]; compute cumulative sum cs = np.cumsum(u) - 1; find rho = max{j | u_j - cs_j/(j+1) > 0}, compute theta = cs[rho]/(rho+1); output w = np.maximum(v - theta, 0).

We must ensure sum close to 1.

Implementation:

```
def project_onto_simplex(v):
    """Projects a vector onto the probability simplex {x >=0, sum=1}."""
    if np.sum(v) == 1 and np.all(v >= 0):
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All <=0: return uniform?
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""A hybrid optimizer for the step‑function bound.

The algorithm combines:
  * a diverse set of handcrafted seed sequences;
  * a fast simulated‑annealing on a low‑dimensional block
    representation (piecewise‑constant);
  * a projected‑gradient descent (sub‑gradient of the exact max)
    that works directly on the full‑resolution vector.

All steps respect the global time budget and keep the best feasible
sequence seen so far.  The final sequence is returned as a plain Python
list, ready for ``evaluate_sequence``.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    conv = _full_convolution(a, conv_fft_len)
    max_b = conv.max()
    return float(2.0 * n * max_b / (s * s))

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _replicate_block(block: np.ndarray, n: int, B: int) -> np.ndarray:
    """Expand a block vector (length B) to length n by repetition."""
    rep = n // B
    remainder = n % B
    a = np.repeat(block, rep)
    if remainder:
        a = np.concatenate([a, block[:remainder]])
    return a

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest pool of diverse seed vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # add a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws with random concentration
    for _ in range(6):
        alpha = np.random.rand() * 4.0 + 0.1   # α ∈ [0.1,4.1]
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # double‑exponential (mass near the edges)
    i = np.arange(n, dtype=np.float64)
    for gamma in (0.05, 0.1, 0.2, 0.5):
        left = np.exp(-gamma * i)
        right = left[::-1]
        shape = left + right
        shape = shape / shape.sum()
        cand.append(shape)

    # arcsine‑like distribution (peaked at the ends, diverging like 1/√x)
    idx = np.arange(n, dtype=np.float64) + 0.5
    den = np.sqrt(idx * (n + 0.5 - idx))
    arcsine = 1.0 / np.maximum(den, 1e-12)
    arcsine = arcsine / arcsine.sum()
    cand.append(arcsine)

    # half‑wave sinusoid
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid = sinusoid / sinusoid.sum()
    cand.append(sinusoid)

    return cand

def _block_annealing(start_block: np.ndarray,
                     n: int,
                     B: int,
                     conv_fft_len: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on a piecewise‑constant representation.
    The block vector lives on the simplex; we repeatedly perturb it,
    evaluate the full‑resolution objective and keep improvements
    (occasionally accepting worse moves to escape local minima).
    """
    cur_block = start_block.copy()
    cur_full = _replicate_block(cur_block, n, B)
    cur_val = _objective(cur_full, n, conv_fft_len)

    best_block = cur_block.copy()
    best_val = cur_val

    start = time.time()
    init_sigma = 0.2
    min_sigma = 0.001
    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / max(1e-12, (deadline - start))
        sigma = init_sigma * (1 - progress) + min_sigma

        proposal = cur_block + sigma * np.random.randn(B)
        proposal = np.maximum(proposal, 0.0)
        if proposal.sum() == 0:
            continue
        proposal = proposal / proposal.sum()

        full_prop = _replicate_block(proposal, n, B)
        val = _objective(full_prop, n, conv_fft_len)

        # Metropolis acceptance; temperature shrinks with progress
        temperature = 0.05 * (1.0 - progress) + 1e-8
        accept = False
        if val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(val - cur_val) / temperature):
                accept = True

        if accept:
            cur_block = proposal
            cur_val = val
            if val < best_val:
                best_val = val
                best_block = proposal.copy()
                print(f"[BlockAnneal] new best {val:.6f}")

    return best_block, best_val

def _projected_gradient_optimize(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected (additive) gradient descent on the exact max‑convolution.
    Uses the sub‑gradient of max_i (a*a)_i and projects onto the simplex
    after each update.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # current convolution and active set
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        # build weight vector for the active constraints
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # sub‑gradient of the max term
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size (decreasing)
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        val = _objective(a_new, n, conv_fft_len)

        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")

        a = a_new
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Hybrid search for a non‑negative step‑function that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """

    deadline = time.time() + budget_s - 0.5   # small safety margin

    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    cand.append(sinusoid)

    return cand


# ----------------------------------------------------------------------
def _block_annealing(start_block: np.ndarray,
                     n: int,
                     B: int,
                     conv_fft_len: int,
                     deadline: float) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing on a piecewise‑constant (block) representation.
    The block vector lives on the simplex; we perturb it with Gaussian
    noise, rebuild the full‑resolution sequence, evaluate the exact
    objective and keep improvements (occasionally accepting worse moves
    to escape local minima).
    """
    cur_block = start_block.copy()
    cur_full = _replicate_block(cur_block, n, B)
    cur_val = _objective(cur_full, n, conv_fft_len)

    best_block = cur_block.copy()
    best_val = cur_val

    start = time.time()
    init_sigma = 0.20
    min_sigma = 0.001

    while time.time() < deadline:
        elapsed = time.time() - start
        progress = elapsed / max(1e-12, (deadline - start))
        sigma = init_sigma * (1 - progress) + min_sigma

        proposal = cur_block + sigma * np.random.randn(B)
        proposal = np.maximum(proposal, 0.0)
        if proposal.sum() == 0:
            continue
        proposal = proposal / proposal.sum()

        full_prop = _replicate_block(proposal, n, B)
        val = _objective(full_prop, n, conv_fft_len)

        temperature = 0.05 * (1.0 - progress) + 1e-8
        accept = False
        if val < cur_val:
            accept = True
        else:
            if np.random.rand() < np.exp(-(val - cur_val) / temperature):
                accept = True

        if accept:
            cur_block = proposal
            cur_val = val
            if val < best_val:
                best_val = val
                best_block = proposal.copy()
                print(f"[BlockAnneal] new best {val:.6f}")

    return best_block, best_val


# ----------------------------------------------------------------------
def _projected_gradient_optimize(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 300_000,
                                 eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Additive projected gradient descent on the exact max‑convolution.
    Each iteration evaluates the current convolution, builds the
    sub‑gradient of the active constraints and takes a decreasing step.
    After the update the point is projected onto the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)

        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")

        a = a_new
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Hybrid search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The routine respects the given time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]), mode='constant')
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Phase 0 – evaluate a modest set of handcrafted seeds
    candidates = _initial_candidates(n, warm)
    best_seq = None
    best_val = float("inf")

    for idx, cand in enumerate(candidates):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand = cand / cand.sum()
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Init {idx}] new best {val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – block‑wise simulated annealing (≈30 % of remaining budget)
    remaining = deadline - time.time()
    block_budget = remaining * 0.35
    block_deadline = time.time() + block_budget

    B = 256                         # number of blocks (must divide n)
    # initialise block representation from the best full vector
    start_block = np.zeros(B, dtype=np.float64)
    rep = n // B
    for b in range(B):
        start_block[b] = best_seq[b * rep:(b + 1) * rep].sum()
    if start_block.sum() == 0:
        start_block = np.full(B, 1.0 / B)
    else:
        start_block = start_block / start_block.sum()

    best_block, block_val = _block_annealing(start_block, n, B,
                                            conv_fft_len, block_deadline)

    best_full_from_block = _replicate_block(best_block, n, B)
    best_full_from_block = best_full_from_block / best_full_from_block.sum()
    val = _objective(best_full_from_block, n, conv_fft_len)
    if val < best_val:
        best_val, best_seq = val, best_full_from_block.copy()
        print(f"[Block→Full] new best {val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – fine‑grained projected‑gradient refinement (rest of budget)
    a_refined, val_refined = _projected_gradient_optimize(
        best_seq, n, conv_fft_len, corr_fft_len, deadline,
        max_iters=500_000, eta0=0.9
    )
    if val_refined < best_val:
        best_val, best_seq = val_refined, a_refined.copy()
        print(f"[Refine] new best {val_refined:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
