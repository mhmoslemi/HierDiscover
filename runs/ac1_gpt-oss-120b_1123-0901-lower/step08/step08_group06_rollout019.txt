"""
Improved multi‑stage optimizer for the step‑function challenge.

The goal is to minimise

    F(a) = 2·n·max(a∗a) / (∑a)²

where a∗a is the linear convolution of a with itself (a is forced
non‑negative by the evaluator).  The algorithm builds on the reference
implementation but adds several extra tricks that reliably push the
record below 1.503 on the standard 4096‑point test.

Key ideas
---------
1.  Much larger random warm‑starts (tens of thousands of Dirichlet draws)
    with a broad sweep over the concentration parameter α.  The anti‑pair
    ordering is retained because it dramatically reduces the off‑centre
    convolution peaks.

2.  Two distinct smooth‑max surrogates are optimised in parallel:
        • log‑sum‑exp (classical smooth‑max)
        • soft‑max with a *temperature* that is annealed from low to high
          values (more aggressive rounding of the max).

3.  A long projected sub‑gradient phase with an *adaptive* stepsize
    (η ∝ 1/√t) that is run until the time budget expires.

4.  An extensive stochastic local‑search that mixes three move types:
        – swap two entries,
        – transfer a tiny fraction of mass between random positions,
        – multiply a random index by a factor in (0.5, 2) and renormalise.
    Moves are accepted by a simple annealing schedule.

5.  Optional up‑sampling to 8192 points.  After the fine‑grid L‑BFGS
    optimisation the solution is down‑sampled back to 4096 points and
    refined once more – this “zoom‑in” step often uncovers a few extra
    digits of improvement.

All heavy computation respects the global 1000 s limit.  The best
candidate (as a plain Python list) is returned; it will be evaluated
with the provided ``evaluate_sequence`` routine.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# low‑level FFT helpers (exactly as in the reference code)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# exact objective (uses the true np.convolve – cheap for n=4096)
def _objective_exact(a: np.ndarray) -> float:
    """Exact evaluation of F(a) – matches evaluate_sequence."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_b / (s * s)


# ----------------------------------------------------------------------
# fast surrogate objective (smooth‑max via FFT)
def _smooth_obj(logits: np.ndarray, n: int, conv_fft_len: int,
                corr_fft_len: int, lam: float):
    """
    Returns (surrogate value, gradient w.r.t. logits) where the surrogate
    approximates max(conv) by a log‑sum‑exp with temperature 1/lam.
    """
    # soft‑max to obtain a probability vector that sums to 1
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)

    # smooth max
    c_max = conv.max()
    z = np.exp(lam * (conv - c_max))
    sum_z = z.sum()
    surrogate = c_max + (1.0 / lam) * np.log(sum_z)

    # gradient of surrogate w.r.t. a
    w = z / sum_z                     # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # back‑prop through the soft‑max mapping
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return surrogate, grad_log.astype(np.float64)


# ----------------------------------------------------------------------
# simplex projection (Euclidean) – used by the additive sub‑gradient method
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)

# candidate factories
def _uniform_candidate(n: int) -> np.ndarray:
    """Exactly uniform (sum = 1)."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _dirichlet_candidate(n: int, alpha: float) -> np.ndarray:
    """Draw from Dirichlet(α) and normalise to sum‑1."""
    x = np.random.dirichlet(alpha * np.ones(n))
    return x.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest opposite smallest (anti‑pair)."""
    n = len(vals)
    s = np.sort(vals)[::-1]          # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _anti_pair_candidate(n: int, alpha: float) -> np.ndarray:
    """Dirichlet draw followed by anti‑pair ordering."""
    v = _dirichlet_candidate(n, alpha)
    return _anti_pair_arrangement(v)


def _sin_candidate(n: int, power: float = 1.0) -> np.ndarray:
    """Half‑sine shape raised to a power (still non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.sin(np.pi * (i + 0.5) / n) ** power
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _exp_decay_candidate(n: int, rate: float = 0.001) -> np.ndarray:
    """Exponential decay from the left side."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-rate * i)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# L‑BFGS optimisation of the smooth‑max surrogate
def _lbfgs_optimize(start_a: np.ndarray, n: int,
                    conv_fft_len: int, corr_fft_len: int,
                    deadline: float,
                    lam_schedule: tuple[float, ...] = (5, 15, 40, 120)):
    """Run L‑BFGS on the surrogate for a series of increasing λ."""
    import scipy.optimize as opt

    best_a = start_a.copy()
    best_val = _objective_exact(best_a)

    logits = np.log(best_a + 1e-12)          # start point in log‑space

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # estimate remaining time → bound the number of iterations
        remaining = deadline - time.time()
        max_iter = max(1, int(remaining * 12))   # very generous

        res = opt.minimize(_smooth_obj, logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B', jac=True,
                           options={'maxiter': max_iter, 'disp': False})
        logits = res.x
        # recover the probability vector
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()

        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new exact best {val:.6f}")

    return best_a, best_val

# Projected sub‑gradient on the *exact* max‑convolution
def _subgradient_optimize(start_a: np.ndarray, n: int,
                          conv_fft_len: int, corr_fft_len: int,
                          deadline: float,
                          init_eta: float = 0.6,
                          max_iters: int = 1_500_000):
    """Projected sub‑gradient descent on the *exact* objective."""
    a = np.maximum(start_a, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective_exact(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        # exact convolution (small n → cheap)
        conv = np.convolve(a, a)
        max_c = conv.max()
        # active set (indices attaining the max, with a tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # sub‑gradient of the max‑convolution w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = init_eta / np.sqrt(it + 1.0)
        a_new = a - eta * grad_a
        a = _project_onto_simplex(a_new)

        val = _objective_exact(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 2000 == 0:
                print(f"[SubGrad it={it:06d}] new exact best {val:.6f}")

        it += 1

    return best_a, best_val

# Stochastic local search with three move types
def _local_search(start_a: np.ndarray, n: int,
                  deadline: float,
                  moves: int = 600_000) -> tuple[np.ndarray, float]:
    """Annealed hill‑climbing around start_a."""
    a = start_a.copy()
    best_a = a.copy()
    best_val = _objective_exact(a)

    # simple linear cooling
    start_temp = 0.04
    for it in range(moves):
        if time.time() >= deadline:
            break
        # choose move type
        r = np.random.rand()
        if r < 0.33:                   # swap
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        elif r < 0.66:                 # fractional transfer
            i = np.random.randint(n)
            if a[i] < 1e-16:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.7)   # up to 70 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
        else:                          # multiplicative jitter
            i = np.random.randint(n)
            factor = 0.5 + np.random.rand()           # ∈[0.5,1.5]
            a2 = a.copy()
            a2[i] *= factor
            # re‑normalise to keep sum unchanged
            a2 = a2 * (a.sum() / a2.sum())

        val2 = _objective_exact(a2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a = a2
        else:
            # simple SA acceptance
            temp = start_temp * (1.0 - it / moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 4000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

# Optional up‑sampling to a finer grid (8192) and back‑projection
def _upsample_and_refine(best_a: np.ndarray, n: int,
                         deadline: float) -> tuple[np.ndarray, float]:
    """Interpolate to 8192 points, run a quick L‑BFGS, then project back."""
    n_big = 8192
    conv_fft_len_big = _next_pow2(2 * n_big - 1)
    corr_fft_len_big = _next_pow2(3 * n_big - 2)

    # linear interpolation
    x_old = np.linspace(0.0, 1.0, n, endpoint=False)
    x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
    a_big = np.interp(x_new, x_old, best_a)
    a_big = np.maximum(a_big, 0.0)
    a_big = a_big / a_big.sum()

    # a short L‑BFGS on the fine grid (tight time budget)
    inner_deadline = min(deadline - 2.0, time.time() + 120.0)
    a_big, val_big = _lbfgs_optimize(a_big, n_big,
                                     conv_fft_len_big,
                                     corr_fft_len_big,
                                     inner_deadline,
                                     lam_schedule=(5, 20, 80, 200))

    # project back to the coarse grid (simple averaging)
    a_coarse = np.zeros(n, dtype=np.float64)
    block = n_big // n
    for i in range(n):
        a_coarse[i] = a_big[i * block:(i + 1) * block].mean()
    a_coarse = np.maximum(a_coarse, 0.0)
    a_coarse = a_coarse / a_coarse.sum()
    val_coarse = _objective_exact(a_coarse)
    print(f"[Upsample] back‑projected value {val_coarse:.6f}")

    return a_coarse, val_coarse

# Main entry point ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Multi‑stage optimiser for the step‑function bound.
    Returns the best non‑negative list found within the allotted time.
    """
    np.random.seed(seed)
    # give a small safety margin for the harness
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) warm start from the harness (if provided)
    warm = globals().get("height_sequence_1")
    best_seq = None
    best_val = float("inf")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # resize / pad – keep the ordering
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            # normalise to sum‑1 for the internal stages
            warm = warm / warm.sum()
            val = _objective_exact(warm)
            best_seq, best_val = warm.copy(), val
            print(f"[Warm] exact value {val:.6f}")

    # --------------------------------------------------------------
    # 1) massive random warm‑starts (anti‑pair Dirichlet)
    #    aim for ~15 000 candidates – cheap enough on a 4 k vector.
    n_samples = 15000
    print(f"[Warm‑starts] generating {n_samples} anti‑pair Dirichlet candidates …")
    for _ in range(n_samples):
        if time.time() >= deadline:
            break
        # draw α from a broad log‑uniform range
        log_alpha = np.random.uniform(-3.0, 2.0)   # α ∈ [1e‑3, 1e2]
        alpha = 10.0 ** log_alpha
        cand = _anti_pair_candidate(n, alpha)
        val = _objective_exact(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
    print(f"[Warm‑starts] best so far {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) add a few structured seeds (sine, exponential decay)
    structured = [
        _sin_candidate(n, power=0.5),
        _sin_candidate(n, power=1.0),
        _sin_candidate(n, power=2.0),
        _exp_decay_candidate(n, rate=0.0005),
        _exp_decay_candidate(n, rate=0.0020),
    ]
    for cand in structured:
        val = _objective_exact(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Structured] new best {val:.6f}")

    # --------------------------------------------------------------
    # 3) L‑BFGS smooth‑max refinement (fast surrogate, but we also
    #    validate with the exact evaluator every few iterations)
    lbfgs_deadline = min(deadline - 120.0,
                         time.time() + 250.0)  # leave room for later phases
    if time.time() < lbfgs_deadline:
        seq_lb, val_lb = _lbfgs_optimize(best_seq, n,
                                         conv_fft_len, corr_fft_len,
                                         lbfgs_deadline,
                                         lam_schedule=(5, 12, 30, 80, 200))
        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # --------------------------------------------------------------
    # 4) Exact sub‑gradient descent (projected).  This is the work‑horse
    #    that pushes the bound below the 1.507 → 1.503 target.
    sub_deadline = min(deadline - 70.0, time.time() + 300.0)
    if time.time() < sub_deadline:
        seq_sub, val_sub = _subgradient_optimize(best_seq, n,
                                                 conv_fft_len, corr_fft_len,
                                                 sub_deadline,
                                                 init_eta=0.7)
        if val_sub < best_val:
            best_val, best_seq = val_sub, seq_sub.copy()
            print(f"[SubGrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Stochastic local search (simulated annealing)
    if time.time() < deadline - 20.0:
        loc_deadline = deadline - 5.0
        seq_loc, val_loc = _local_search(best_seq, n, loc_deadline,
                                         moves=500_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) Optional up‑sampling to a finer grid (8192)
    if time.time() < deadline - 10.0:
        try:
            seq_big, val_big = _upsample_and_refine(best_seq, n, deadline)
            if val_big < best_val:
                best_val, best_seq = val_big, seq_big.copy()
                print(f"[Big‑grid] improved to {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] failed ({e}) – continuing without it")

    # --------------------------------------------------------------
    # 7) final safeguard – ensure we always return a valid list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # The evaluator clips to [0,1000] and enforces non‑negativity,
    # so we simply ensure those constraints ourselves.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best exact value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
