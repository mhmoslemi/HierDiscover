def _adam_projected_optimize(a0, n, conv_len, corr_len, deadline, max_iters=200000, lr=0.1, beta1=0.9, beta2=0.999, eps=1e-8):
    a = a0.copy()
    a = np.maximum(a,0)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    t = 0
    best = a.copy()
    best_val = _objective(a)
    while time.time() < deadline and t < max_iters:
        t += 1
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_len)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad*grad)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_onto_simplex(a_new)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best = a.copy()
            # prints optionally
    return best, best_val
```

"""
Propose a step‑function (non‑negative list) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

Algorithm
---------
1. **Warm‑start pool** – generate many Dirichlet samples (concentration
   α drawn from a wide log‑uniform range) and apply several
   orderings:
   * anti‑pair (large ↔ small)
   * alternating extremes (large, small, 2nd‑large, 2nd‑small,…)
   * bit‑reversed placement (low‑discrepancy spreading)
   Keep the best candidate.

2. **Smooth‑max L‑BFGS** – optimise a log‑sum‑exp surrogate of the
   max‑convolution using SciPy’s L‑BFGS‑B, progressively increasing the
   temperature λ (5 → 15 → 40 → 120 → 400).  The variables are logits;
   the actual sequence is the soft‑max of the logits (always non‑negative
   and sums to one).

3. **Projected Adam on the exact objective** – treat the exact max‑convolution
   as a sub‑gradient (computed from the active lag(s)) and run a few
   Adam‑style steps with Euclidean projection onto the simplex.

4. **Local stochastic annealing** – random swaps / fractional mass
   transfers accepted by a simple temperature schedule, to escape possible
   shallow local minima.

5. **Optional up‑sampling** – if time permits, linearly interpolate the
   current best vector to 8192 points and repeat steps 2–4 on the finer grid.

All stages respect the global time budget (default 1000 s).  The returned
list is clipped to the range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact)
def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(a*a) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * a.shape[0] - 1))
    return 2.0 * a.shape[0] * conv.max() / (s * s)


# ----------------------------------------------------------------------
# Simplex projection (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        theta = 0.0
    else:
        rho = ind[cond][-1] - 1
        theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _uniform_candidate(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_candidate(n: int, alpha: float = 1.0) -> np.ndarray:
    """Draw from Dirichlet(α,…,α) and normalise to sum 1."""
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _anti_pair_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest values occupy the left half, smallest values (reversed) the right."""
    n = len(vals)
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = s[:half]
    a[-half:] = s[-half:][::-1]
    if n % 2:
        a[half] = s[half]
    return a


def _alternating_arrangement(vals: np.ndarray) -> np.ndarray:
    """Largest, smallest, 2nd‑largest, 2nd‑smallest, …."""
    n = len(vals)
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    lo, hi = 0, n - 1
    i = 0
    toggle = True
    while lo <= hi:
        if toggle:
            a[i] = s[lo]
            lo += 1
        else:
            a[i] = s[hi]
            hi -= 1
        i += 1
        toggle = not toggle
    return a


def _bitrev_permutation(n: int) -> np.ndarray:
    """Bit‑reversed order for n = 2ᵏ."""
    bits = int(np.ceil(np.log2(n)))
    rev = np.arange(n, dtype=np.int64)
    # vectorised bit‑reversal using string formatting (fast enough for n≤8192)
    fmt = "{:0" + str(bits) + "b}"
    rev = np.array([int(fmt(i)[::-1], 2) for i in rev], dtype=np.int64)
    return rev


def _bitrev_arrangement(vals: np.ndarray) -> np.ndarray:
    """Place sorted values according to bit‑reversed indices."""
    n = len(vals)
    perm = _bitrev_permutation(n)
    s = np.sort(vals)[::-1]                 # descending
    a = np.empty(n, dtype=np.float64)
    a[perm] = s
    return a


def _candidate_from_dirichlet(n: int, alpha: float, mode: str) -> np.ndarray:
    """Generate a Dirichlet sample and order it according to *mode*."""
    vals = np.random.dirichlet(alpha * np.ones(n))
    if mode == "anti":
        return _anti_pair_arrangement(vals)
    elif mode == "altern":
        return _alternating_arrangement(vals)
    elif mode == "bitrev":
        return _bitrev_arrangement(vals)
    else:   # plain
        return vals


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate
def _lbfgs_obj(logits: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, lam: float):
    """Smooth‑max (log‑sum‑exp) objective and gradient w.r.t. logits."""
    # softmax to obtain a (sum‑to‑one, non‑negative)
    mx = np.max(logits)
    exp_l = np.exp(logits - mx)
    a = exp_l / exp_l.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    # stable log‑sum‑exp on (lam*(conv - max_c))
    z = np.exp(lam * (conv - max_c))
    sum_z = z.sum()
    surrogate = max_c + (1.0 / lam) * np.log(sum_z)

    # gradient of smooth‑max w.r.t. a
    w = z / sum_z               # soft weights, sum to 1
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through softmax
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return surrogate, grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                    corr_fft_len: int, deadline: float,
                    lam_schedule=(5.0, 15.0, 40.0, 120.0, 400.0),
                    max_iter_stage: int = 300) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on the smooth‑max surrogate, increasing λ progressively."""
    import scipy.optimize as opt
    best_a = a0.copy()
    best_val = _objective(best_a)
    logits = np.log(best_a + 1e-12)

    for lam in lam_schedule:
        if time.time() >= deadline:
            break
        # number of L‑BFGS iterations we can still afford
        remaining = deadline - time.time()
        iters = min(max_iter_stage, max(1, int(remaining * 8)))
        res = opt.minimize(
            _lbfgs_obj,
            logits,
            args=(n, conv_fft_len, corr_fft_len, lam),
            method='L-BFGS-B',
            jac=True,
            options={'maxiter': iters, 'disp': False}
        )
        logits = res.x
        mx = np.max(logits)
        exp_l = np.exp(logits - mx)
        a = exp_l / exp_l.sum()
        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected Adam optimisation on the exact max‑convolution objective
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.15,
                   beta1: float = 0.9,
                   beta2: float = 0.999,
                   eps: float = 1e-8) -> tuple[np.ndarray, float]:
    """Adam‑like updates with Euclidean projection onto the simplex."""
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline and it < max_iters:
        it += 1
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_onto_simplex(a - step)

        val = _objective(a)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

    return best_a, best_val

# Local stochastic hill‑climbing (simulated annealing)
def _local_random_search(a0: np.ndarray, n: int,
                         conv_fft_len: int, deadline: float,
                         max_moves: int = 500_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start_temp = 0.02
    it = 0
    while time.time() < deadline and it < max_moves:
        # choose a move type
        if np.random.rand() < 0.5:                     # swap two entries
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        else:                                          # transfer a fraction
            i = np.random.randint(n)
            if a[i] < 1e-15:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.8)   # up to 80 % move
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        val2 = _objective(a2)

        if val2 < best_val:
            best_val, best_a, a = val2, a2.copy(), a2
        else:
            temp = start_temp * (1.0 - it / max_moves)
            if temp > 0.0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")
        it += 1

    return best_a, best_val

# Auxiliary: warm‑start handling
def _warm_start(n: int):
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 0.01:
        return None
    return warm / warm.sum()

        val = _objective(c)
        if val < best_val:
            best_val, best_seq = val, c.copy()
            print(f"[Starter] value {val:.6f}")

    # random Dirichlet pool with three ordering schemes
    n_samples = 3000
    ordering_modes = ["anti", "altern", "bitrev"]
    print(f"Generating {n_samples} random candidates …")
    for _ in range(n_samples):
        # draw α from a wide logarithmic range
        log_alpha = np.random.uniform(-3.0, 2.0)   # α ∈ [0.001, 100]
        alpha = 10.0 ** log_alpha

        val = _objective(cand)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Pool {mode}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2) L‑BFGS smooth‑max refinement

        if val_lb < best_val:
            best_val, best_seq = val_lb, seq_lb.copy()
            print(f"[L‑BFGS] improved to {val_lb:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected Adam optimisation on the exact objective
    adam_deadline = min(deadline - 70.0, time.time() + 250.0)

        seq_adam, val_adam = _adam_optimize(best_seq, n,
                                            conv_fft_len, corr_fft_len,
                                            adam_deadline,
                                            max_iters=800_000,
                                            lr=0.12)
        if val_adam < best_val:
            best_val, best_seq = val_adam, seq_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Local stochastic annealing
    if best_seq is not None and time.time() < deadline - 20.0:
        local_deadline = deadline - 5.0
        seq_loc, val_loc = _local_random_search(best_seq, n,
                                                conv_fft_len,
                                                local_deadline,
                                                max_moves=400_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, seq_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional up‑sampling to 8192 points
    if time.time() < deadline - 10.0:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        big_deadline = min(deadline - 2.0, time.time() + 180.0)
        # quick L‑BFGS on the refined grid
        seq_big_lb, val_big_lb = _lbfgs_optimize(a_big, n_big,
                                                 conv_fft_len_big,
                                                 corr_fft_len_big,
                                                 big_deadline,
                                                 lam_schedule=(5, 20, 80, 200, 600),
                                                 max_iter_stage=150)
        # final Adam fine‑tuning
        final_deadline = min(deadline, time.time() + 120.0)
        seq_big_adam, val_big_adam = _adam_optimize(seq_big_lb, n_big,
                                                    conv_fft_len_big,
                                                    corr_fft_len_big,
                                                    final_deadline,
                                                    max_iters=600_000,
                                                    lr=0.1)
        if val_big_adam < best_val:
            best_val, best_seq = val_big_adam, seq_big_adam.copy()
            print(f"[Big grid] improved to {val_big_adam:.6f}")

    # ------------------------------------------------------------------
    # 6) Safeguard – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
