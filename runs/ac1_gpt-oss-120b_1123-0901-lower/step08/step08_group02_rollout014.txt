def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    
    # choose n optionally dynamic? Use 4096 as before.
    n = 4096
    conv_fft_len = _next_pow2(2*n -1)
    corr_fft_len = _next_pow2(3*n -2)

    # warm start:
    warm = globals().get('height_sequence_1')
    if warm is None or len(warm)==0:
        warm = None
    else:
        warm = np.asarray(warm, dtype=np.float64)
        warm = warm[:n] if warm.shape[0] > n else np.pad(warm, (0, n-warm.shape[0]), mode='constant')
        warm = np.maximum(warm, 0.0)
        if warm.sum() < 1e-12:
            warm = None
        else:
            warm = warm / warm.sum()
    # Create initial pool: warm if exists, plus random Dirichlet draws (maybe 8), plus edge-exponential/double-gaussian etc.

    # generate candidates:
    pool = []
    if warm is not None:
        pool.append(warm)
        # Add some shifts:
        for shift in [n//4, n//2, 3*n//4]:
            pool.append(np.roll(warm, shift))
    # add random Dirichlet:
    for _ in range(6):
        alpha = np.random.rand()*4 + 0.1
        pool.append(np.random.dirichlet(np.full(n, alpha)))
    # add edge-exponential:
    for gamma in [0.12,0.25,0.5,1.0,2.0]:
        left = np.exp(-gamma*np.arange(n))
        right = left[::-1]
        shape = left+right
        shape/=shape.sum()
        pool.append(shape)
    # add edge-polynomial:
    idx=np.arange(1,n+1, dtype=np.float64)
    prod=idx*(n-idx+1)
    for a in [-1.5,-1.0,-0.75,-0.5]:
        shape = np.where(prod>0, prod**a,0.0)
        if shape.sum()>0:
            shape/=shape.sum()
            pool.append(shape)
    # add double-gaussian
    i = np.arange(n)
    for sigma in [0.04,0.08,0.12,0.2]:
        shape = np.exp(-0.5*(i/(sigma*n))**2) + np.exp(-0.5*((i-(n-1))/(sigma*n))**2)
        shape/=shape.sum()
        pool.append(shape)

    # best tracker
    best_seq = None
    best_val = np.inf

    # Function to evaluate objective directly:
    def obj_val(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max() / (a.sum()**2)

    # Process each candidate:
    per_candidate_budget = (deadline - time.time()) * 0.15 / max(1,len(pool)) # allocate moderate portion
    for idx,cand in enumerate(pool):
        if time.time() > deadline - 5: break
        a = cand.copy()
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        # Run L-BFGS on logits
        logits = np.log(a + 1e-12)  # shape
        # define function f(logits)
        def fun_log(l):
            # stable softmax
            mx = np.max(l)
            exp_l = np.exp(l - mx)
            a_ = exp_l / exp_l.sum()
            conv = _full_convolution(a_, conv_fft_len)
            max_c = conv.max()
            # smooth surrogate maybe lam large? for now true max.
            f = 2.0 * n * max_c  # since sum=1
            # gradient w.r.t a: g_a = 2 * (w ★ a) where w is indicator on max conv(s)
            # compute w as uniform over active (approx)
            tol = 1e-12
            w = (conv >= max_c - tol).astype(np.float64)
            if w.sum()>0:
                w /= w.sum()
            else:
                w = np.ones_like(conv) / conv.size
            grad_a = 2.0 * _cross_correlation(w, a_, corr_fft_len)  # gradient of max conv
            # chain rule: gradient wrt logits:
            avg = np.dot(a_, grad_a)
            grad_log = a_ * (grad_a - avg)
            # return f and grad_log (length n)
            return f, grad_log
        # Use scipy minimize
        try:
            import scipy.optimize as opt
            res = opt.minimize(fun_log, logits, method='L-BFGS-B', jac=True,
                               options={'maxiter': 800, 'ftol':1e-12, 'gtol':1e-12})
            a_opt = np.exp(res.x - np.max(res.x))
            a_opt = a_opt / a_opt.sum()
            val_opt = obj_val(a_opt)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[L-BFGS cand {idx}] best {val_opt:.6f}")
        except Exception as e:
            print(f"L-BFGS exception: {e}")
        # If still time, run few subgradient iterations
        if time.time() >= deadline - 15: break

    # Phase deeper: refine best in a loop with subgradient and local search.
    # Use a while loop until deadline:
    while time.time() < deadline - 2:
        # Determine step size schedule based on iteration count
        # Use exponentiated gradient with adaptive step
        a = best_seq.copy()
        if a is None:

        a = a / a.sum()
        # subgradient loop
        it_sub = 0
        # dynamic step size: decreasing sqrt
        max_sub_iters = 200_000
        start = time.time()
        while time.time() < deadline - 5 and it_sub < max_sub_iters:
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum()**2)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Subgrad refine] it {it_sub} val {val:.6f}")
            # active set
            active = np.isclose(conv, max_c, atol=1e-14)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
            eta = 0.5 / np.sqrt(it_sub + 1)  # step size
            a = a * np.exp(-eta * grad_a)
            # project to simplex (should stay on simplex due to scaling, sum still 1? Exponential update preserves positivity but sum changes)
            s = a.sum()
            if s <= 0 or np.isnan(s):
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a = a / s
            # occasionally random move to avoid stagnation
            if it_sub % 5000 == 0 and np.random.rand() < 0.1:
                # random Gaussian perturbation on logits
                logits = np.log(a + 1e-12)
                logits += np.random.randn(n) * 0.05
                a = np.exp(logits - np.max(logits))
                a = a / a.sum()
            it_sub += 1

        # cheap random local search
        # perform few thousand moves
        a_local = best_seq.copy()
        conv_local = _full_convolution(a_local, conv_fft_len)
        best_val_local = 2.0 * n * conv_local.max() / (a_local.sum()**2)
        moves = 0
        max_moves = 100_000
        while time.time() < deadline - 1 and moves < max_moves:

            prob_src_sum = prob_src.sum()
            if prob_src_sum <= 0:
                src = np.random.randint(n)
            else:
                prob_src = prob_src / prob_src_sum
                src = np.searchsorted(np.cumsum(prob_src), np.random.rand())
            # target: prefer lower contribution
            prob_tgt = a_local * (grad_a.max() - grad_a + 1e-12)  # invert
            prob_tgt = prob_tgt / prob_tgt.sum()
            tgt = np.searchsorted(np.cumsum(prob_tgt), np.random.rand())
            if src == tgt: continue
            delta = a_local[src] * (np.random.rand() * 0.2)  # move up to 20% of src mass
            a2 = a_local.copy()
            a2[src] -= delta
            a2[tgt] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max() / (a2.sum()**2)
            if val2 < best_val_local:
                best_val_local = val2
                a_local = a2
                # immediately update global best if improved
                if val2 < best_val:
                    best_val = val2
                    best_seq = a2.copy()
                    print(f"[Local move] new best {val2:.6f}")
            moves += 1

    # final clip and return
    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} length {len(best_seq)}")
    return best_seq.tolist()

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The algorithm builds a diverse initial pool (warm start, Dirichlet draws,
edge‑exponential / polynomial shapes and Gaussian edges).  For each
candidate we run a short L‑BFGS optimisation on a smoothed
log‑sum‑exp surrogate of the max‑convolution.  The best candidate is
then refined for the remaining budget by a long exponentiated‑gradient
sub‑gradient descent (mirror descent on the simplex) and a cheap
gradient‑guided random‑move hill‑climber.  All updates keep the vector
on the probability simplex, so the objective simplifies to
`2·n·max(a∗a)`.  The routine respects the global time budget and
always returns the best feasible sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a): (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)].
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.ndim != 1:
        raise ValueError("_simplex_projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) because Σa is kept equal to 1."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


def _subgrad_max(a: np.ndarray, n: int,
                 conv_fft_len: int, corr_fft_len: int,
                 atol: float = 1e-14):
    """Sub‑gradient of the exact max‑convolution (returns gradient wrt a)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    active = np.isclose(conv, max_c, atol=atol)
    if active.any():
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
    else:
        w = np.zeros_like(conv)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂ max_c / ∂ a
    grad_obj = 2.0 * n * grad_a                           # ∂ F / ∂ a (Σa = 1)
    return grad_obj, conv, max_c


def _smooth_max_grad(a: np.ndarray, lam: float, n: int,
                     conv_fft_len: int, corr_fft_len: int):
    """
    Gradient of the log‑sum‑exp (smooth‑max) surrogate.
    Returns gradient w.r.t. a and the surrogate value.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    # smoothed value = Σ w_i * conv_i
    smooth_val = np.dot(w, conv)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂ smooth_val / ∂ a
    grad_obj = 2.0 * n * grad_a                            # ∂ obj / ∂ a
    return grad_obj, smooth_val

# L‑BFGS optimisation on logits (softmax parametrisation)
def _lbfgs_obj(logits: np.ndarray, n: int,
               conv_fft_len: int, corr_fft_len: int,
               lam: float):
    """Return (objective, gradient) for L‑BFGS (smooth max)."""
    # soft‑max
    mx = np.max(logits)
    exp_logits = np.exp(logits - mx)
    a = exp_logits / exp_logits.sum()

    # smooth‑max and its gradient
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum
    smooth_val = np.dot(w, conv)                     # = soft‑max value
    f = 2.0 * n * smooth_val

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂ smooth_val / ∂ a
    grad_a = 2.0 * n * grad_a                               # ∂ f / ∂ a

    # chain rule to logits
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return f, grad_log


def _lbfgs_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, corr_fft_len: int,
                    max_iter: int, deadline: float,
                    lam: float = 50.0):
    """
    Short L‑BFGS run on the smooth‑max surrogate.
    Returns the best a encountered and its exact objective.
    """
    import scipy.optimize as opt

    logits0 = np.log(a0 + 1e-12)
    # time‑aware max_iter: don’t exceed deadline
    max_iter = min(max_iter, int(1e6))
    start = time.time()
    # SciPy will stop earlier if max_iter reached or convergence achieved.
    res = opt.minimize(_lbfgs_obj, logits0,
                       args=(n, conv_fft_len, corr_fft_len, lam),
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12, 'gtol': 1e-12})
    # Convert back to probability vector
    a_opt = np.exp(res.x - np.max(res.x))
    a_opt = a_opt / a_opt.sum()
    val_opt = _objective(a_opt, n, conv_fft_len)
    return a_opt, val_opt

def _exp_grad_step(a: np.ndarray, grad_obj: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror descent) update on the simplex."""
    a_new = a * np.exp(-eta * grad_obj / (2.0 * len(a)))  # divide by 2n? we will renorm
    # The factor 2n is absorbed in step size; we keep eta reasonable.
    s = a_new.sum()
    if s <= 0 or np.isnan(s):
        # fallback to uniform vector
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new = a_new / s
    return a_new


def _subgradient_refine(a_start: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float,
                        max_iters: int = 500_000,
                        eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient sub‑gradient descent.
    Returns the best vector and its exact objective.
    """
    a = a_start.copy()
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline - 0.5 and it < max_iters:
        grad_obj, conv, max_c = _subgrad_max(a, n, conv_fft_len, corr_fft_len)
        # step size decreasing like η₀ / √t
        eta = eta0 / np.sqrt(it + 1)
        a = _exp_grad_step(a, grad_obj, eta)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad it={it:06d}] new best {val:.6f}")
        # occasional random logit shake to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.1:
            logits = np.log(a + 1e-12)
            logits += np.random.randn(n) * 0.04
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
        it += 1

    return best_a, best_val


def _local_random_moves(a_start: np.ndarray, n: int,
                        conv_fft_len: int, deadline: float,
                        max_moves: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from one index
    to another, keep the move if it improves the exact objective.
    """
    a = a_start.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline - 0.2 and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.2)   # move ≤ 20 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val

"""
Hybrid optimiser for the step‑function max‑convolution problem.

The algorithm builds a diverse initial pool (warm start, Dirichlet draws,
edge‑exponential / polynomial shapes and Gaussian edges).  For each
candidate a short L‑BFGS optimisation on a smoothed log‑sum‑exp surrogate
of the max‑convolution is performed.  The best candidate is then refined
by a long exponentiated‑gradient sub‑gradient descent (mirror descent on
the probability simplex) and a cheap random‑move hill‑climber.  All
updates keep the vector on the simplex, so the objective simplifies to
`2·n·max(a∗a)`.  The routine respects the global time budget and always
returns the best feasible sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a length‑len(a) vector:
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   for i = 0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Projection of a vector onto the probability simplex {x≥0, Σx=1}."""
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(v.shape[0]) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / v.shape[0])
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a∗a).
    The vector `a` is assumed to sum to 1 (probability simplex).
    """
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _subgrad_max(a: np.ndarray, n: int,
                 conv_fft_len: int, corr_fft_len: int,
                 atol: float = 1e-14):
    """
    Sub‑gradient of `max(a∗a)` (returns ∂max/∂a, the convolution,
    and the maximal value).
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    active = np.isclose(conv, max_c, atol=atol)
    w = np.zeros_like(conv)
    if active.any():
        w[active] = 1.0 / active.sum()
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂max/∂a
    return grad_a, conv, max_c


# ----------------------------------------------------------------------
# L‑BFGS optimisation of a smooth‑max surrogate (log‑sum‑exp)
def _lbfgs_obj(logits: np.ndarray, n: int,
               conv_fft_len: int, corr_fft_len: int,
               lam: float):
    """Objective and gradient for L‑BFGS (smooth max)."""
    mx = np.max(logits)
    exp_logits = np.exp(logits - mx)
    a = exp_logits / exp_logits.sum()

    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w /= w_sum

    smooth_val = np.dot(w, conv)          # smoothed max
    f = 2.0 * n * smooth_val               # exact objective using the surrogate

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂smooth_val/∂a
    grad_a = 2.0 * n * grad_a                               # ∂f/∂a

    # chain rule to logits (soft‑max)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return f, grad_log


def _lbfgs_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, corr_fft_len: int,
                    max_iter: int, deadline: float,
                    lam: float = 80.0):
    """
    Run a short L‑BFGS optimisation on the smooth‑max surrogate.
    Returns the (exact‑objective) best vector encountered.
    """
    import scipy.optimize as opt

    logits0 = np.log(a0 + 1e-12)
    # Stop early if the global deadline is reached
    max_iter = int(min(max_iter, max(1, (deadline - time.time()) / 0.001)))
    res = opt.minimize(_lbfgs_obj, logits0,
                       args=(n, conv_fft_len, corr_fft_len, lam),
                       method='L-BFGS-B', jac=True,
                       options={'maxiter': max_iter,
                                'ftol': 1e-12, 'gtol': 1e-12})
    a_opt = np.exp(res.x - np.max(res.x))
    a_opt = a_opt / a_opt.sum()
    val_opt = _objective(a_opt, n, conv_fft_len)
    return a_opt, val_opt

def _exp_gradient_step(a: np.ndarray, grad_a: np.ndarray, eta: float) -> np.ndarray:
    """Exponentiated‑gradient (mirror‑descent) update on the simplex."""
    a_new = a * np.exp(-eta * grad_a)
    s = a_new.sum()
    if s <= 0 or np.isnan(s):
        # fall back to uniform distribution if something goes wrong
        return np.full_like(a, 1.0 / a.shape[0])
    return a_new / s


def _subgradient_refine(a_start: np.ndarray, n: int,
                        conv_fft_len: int, corr_fft_len: int,
                        deadline: float,
                        max_iters: int = 500_000,
                        eta0: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Long exponentiated‑gradient sub‑gradient descent (mirror descent).
    Returns the best vector seen and its exact objective.
    """
    a = a_start.copy()
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline - 0.3 and it < max_iters:
        grad_a, conv, max_c = _subgrad_max(a, n, conv_fft_len, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)   # classic 1/√t schedule
        a = _exp_gradient_step(a, grad_a, eta)
        val = 2.0 * n * max_c   # exact objective (sum(a)=1)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # occasional random shake to escape plateaus
        if it % 5000 == 0 and np.random.rand() < 0.12:
            logits = np.log(a + 1e-12)
            logits += np.random.randn(n) * 0.04
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()

        it += 1
    return best_a, best_val


def _local_random_moves(a_start: np.ndarray, n: int,
                        conv_fft_len: int, deadline: float,
                        max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of the mass from one index
    to another and keep the move if the objective improves.
    """
    a = a_start.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline - 0.2 and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.25)   # move ≤25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1
    return best_a, best_val

        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        local_deadline = min(deadline - 2.0, time.time() + per_candidate_budget)

        try:
            a_opt, val_opt = _lbfgs_optimize(a0, n,
                                             conv_fft_len, corr_fft_len,
                                             max_lbfgs_iters, local_deadline,
                                             lam=120.0)
        except Exception as exc:   # SciPy may be unavailable on some platforms
            print(f"[L‑BFGS error] {exc}")
            a_opt = a0
            val_opt = _objective(a_opt, n, conv_fft_len)

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[cand {cid}] L‑BFGS new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient sub‑gradient refinement (≈65 % budget)
    while time.time() < deadline - 30.0:
        # 1) sub‑gradient descent
        a_sub, val_sub = _subgradient_refine(best_seq, n,
                                             conv_fft_len, corr_fft_len,
                                             deadline,
                                             max_iters=300_000,
                                             eta0=0.85)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Refine] Sub‑grad new best {val_sub:.6f}")

        # 2) occasional L‑BFGS polishing with a higher temperature
        if np.random.rand() < 0.35:
            a_pol, val_pol = _lbfgs_optimize(best_seq, n,
                                             conv_fft_len, corr_fft_len,
                                             max_iter=200,
                                             deadline=deadline,
                                             lam=180.0)
            if val_pol < best_val:
                best_val, best_seq = val_pol, a_pol.copy()
                print(f"[Polish] L‑BFGS new best {val_pol:.6f}")

        # 3) cheap random local hill‑climbing
        a_loc, val_loc = _local_random_moves(best_seq, n,
                                             conv_fft_len, deadline,
                                             max_moves=60_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – ensure a valid, non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
