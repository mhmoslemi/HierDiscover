    warm = warm / warm.sum()

"""
Propose a step‑function that minimizes
    2·n·max(a*a)/(sum(a)^2)

The algorithm works in three stages:

1. Warm‑start – reuse the best known sequence (`height_sequence_1`) if it
   exists, otherwise fall back to a uniform distribution.

2. Quick random exploration – a few seconds of random Dirichlet candidates
   are evaluated to possibly improve the start.

3. Guided hill‑climbing with incremental convolution updates:
   * The current maximum entry of the linear convolution is identified.
   * One index that contributes the largest product to this active lag is
     selected as a source; mass is transferred to a random target index.
   * Moves that do not increase the maximum are always accepted.
     Occasionally (temperature‑controlled) uphill moves are allowed to
     escape plateaus.
   * The step size (fraction of the source mass) decays linearly with the
     remaining budget, and the temperature decays as well.
   * The best sequence seen so far is stored and returned.

All operations respect the non‑negativity and sum‑to‑one constraints.
The routine runs for the whole `budget_s` (default 1000 s) and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear convolution a*a (length 2·len(a)‑1) via FFT.
    The result is truncated to the exact length.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Incremental convolution update (used by the hill‑climbing)
def _apply_delta(a: np.ndarray, conv: np.ndarray, idx: int, delta: float):
    """
    Add `delta` to a[idx] (delta may be negative) and update the full
    linear convolution a*a in O(n) using slice operations.
    """
    old = a[idx]
    # update all entries that involve a[idx]
    conv[idx: idx + a.shape[0]] += 2.0 * delta * a
    # self‑product term (appears once at position 2*idx)
    conv[2 * idx] += delta * delta
    a[idx] = old + delta


# ----------------------------------------------------------------------
# Warm‑start helper
def _warm_start(n: int):
    """
    Return a size‑n vector taken from the global `height_sequence_1`
    (if present) and normalised to sum = 1.
    """
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s < 0.01:
        return None
    return warm / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function minimising the evaluation
    function within the supplied time budget.
    Returns the best sequence found (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # safety margin
    total_budget = deadline - start_time

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Warm start
    a = _warm_start(n)
    if a is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # 2) Very brief random search (≈2 s) for a potentially better start
    best_a = a.copy()
    conv = _full_convolution(best_a, fft_len)
    best_max = conv.max()
    cur_max = best_max

    short_deadline = min(deadline - 5.0, start_time + 2.0)
    while time.time() < short_deadline:
        cand = np.random.dirichlet(np.ones(n))
        conv_c = _full_convolution(cand, fft_len)
        max_c = conv_c.max()
        if max_c < best_max:
            best_max = max_c
            best_a = cand.copy()
            a = cand.copy()
            conv = conv_c
            cur_max = max_c
            print(f"[Init random] new best max {best_max:.6e}")

    # --------------------------------------------------------------
    # 3) Guided hill‑climbing with incremental updates
    # Parameters for the schedule
    start_temp = 0.02           # initial temperature for uphill moves
    max_frac_start = 0.4        # initial maximum fraction of mass to move
    min_frac = 1e-5             # lower bound for the fraction
    no_improve = 0
    it = 0

    # Helper to recompute the full convolution from scratch (reduces drift)
    def recompute_full():
        return _full_convolution(a, fft_len)

    while time.time() < deadline:
        # ------------------------------------------------------------------
        # Schedule: fraction and temperature decrease linearly with remaining time
        remaining = max(0.0, deadline - time.time())
        frac = max_frac_start * (remaining / total_budget)
        frac = max(frac, min_frac)
        temp = start_temp * (remaining / total_budget)

        # ------------------------------------------------------------------
        # Identify the active lag(s) where the convolution attains its maximum
        cur_max = conv.max()
        active = np.where(conv >= cur_max - 1e-15)[0]
        # Randomly pick one of the active lags
        lag = int(np.random.choice(active))

        # ------------------------------------------------------------------
        # Find the index i which contributes the largest product a[i] * a[lag-i]
        lo = max(0, lag - (n - 1))
        hi = min(n - 1, lag)
        idx_range = np.arange(lo, hi + 1, dtype=np.int64)
        partner = lag - idx_range
        prod = a[idx_range] * a[partner]
        src = int(idx_range[np.argmax(prod)])

        # Random target (different from source)
        tgt = np.random.randint(n - 1)
        if tgt >= src:
            tgt += 1

        # Amount to move
        delta = a[src] * frac
        if delta <= 0.0:
            continue

        # ------------------------------------------------------------------
        # Apply the move
        _apply_delta(a, conv, src, -delta)
        _apply_delta(a, conv, tgt, delta)

        new_max = conv.max()
        diff = new_max - cur_max

        # Acceptance rule (down‑hill always, occasional uphill)
        accept = diff <= 0.0 or (temp > 0.0 and np.random.rand() < np.exp(-diff / temp))
        if accept:
            cur_max = new_max
            if new_max < best_max - 1e-12:
                best_max = new_max
                best_a = a.copy()
                print(f"[Iter {it:07d}] new best max {best_max:.6e}")
            no_improve = 0
        else:
            # revert the move
            _apply_delta(a, conv, tgt, -delta)
            _apply_delta(a, conv, src, delta)
            no_improve += 1

        it += 1

        # ------------------------------------------------------------------
        # Periodic maintenance
        if it % 50000 == 0:
            # Re‑normalise to avoid drift and recompute the convolution exactly
            s = best_a.sum()
            if s > 0:
                best_a /= s
                a = best_a.copy()
                conv = recompute_full()
                cur_max = conv.max()
                best_max = cur_max
                print(f"[Maintenance] iteration {it}, recomputed max {best_max:.6e}")

        # ------------------------------------------------------------------
        # If we have been stuck for a long time, try a random shake / restart
        if no_improve > 200_000:
            # 50 % chance to shake the current solution, otherwise restart from best
            if np.random.rand() < 0.5:
                i_shake = np.random.randint(n)
                j_shake = np.random.randint(n - 1)
                if j_shake >= i_shake:
                    j_shake += 1
                delta_shake = a[i_shake] * max_frac_start
                if delta_shake > 0:
                    _apply_delta(a, conv, i_shake, -delta_shake)
                    _apply_delta(a, conv, j_shake, delta_shake)
                    cur_max = conv.max()
                print(f"[Shake] performed mass transfer {delta_shake:.2e}")
            else:
                # restart from the globally best known sequence
                a = best_a.copy()
                conv = recompute_full()
                cur_max = conv.max()
                print(f"[Restart] reverted to best known max {best_max:.6e}")
            no_improve = 0

    # --------------------------------------------------------------
    # 4) Final safety checks and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    s = best_a.sum()
    if s < 1e-12:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_a /= s

    print(f"Final best max {best_max:.6e} (objective ≈ {2 * n * best_max:.6f})")
    return best_a.tolist()
