def solve_g(f, M, S):
    n = len(f)
    # Build A matrix
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    # Actually toeplitz first row should be [f[0], 0,0, ...,0] of length n
    # So row = np.concatenate([f[:1], np.zeros(n-1)])
    # Then A = toeplitz(col, row)[:2*n-1, :]
    from scipy.linalg import toeplitz
    A = toeplitz(col, row)[:2*n-1, :]
    # Upper bound vector b
    b = np.full(2*n-1, M)
    # Solve max sum(g) -> min -sum(g)
    c = -np.ones(n)
    bounds = [(0, None)]*n
    # Use high method
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        return None
    g0 = res.x
    # Scale to target sum S
    sum_g0 = g0.sum()
    if sum_g0 <= 0:
        return None
    scaling = S / sum_g0
    g = g0 * scaling
    return g

import time, numpy as np, scipy.optimize as opt, scipy.linalg as la

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len=None):
    n = len(a)
    if fft_len is None:
        fft_len = _next_pow2(2*n-1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[:2*n-1]

def _objective(a, n):
    # compute max convolution
    max_c = _full_convolution(a).max()
    # sum = sum(a)
    s = a.sum()
    return 2.0 * n * max_c / (s * s)

def propose_candidate(seed=0, budget_s=1000):
    import numpy as np, time, scipy.optimize, scipy.linalg

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # size
    n = 2048  # moderate size
    target_sum = np.sqrt(2.0 * n)   # normalization used by original algorithm, any value works

    # initialise f
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] >= n:
            f = w[:n].copy()
        else:
            f = np.pad(w, (0, n - w.shape[0]), constant_values=0.0)
        f = np.maximum(f, 0.0)
        if f.sum() < 0.01:
            f = None
        else:
            f = f / f.sum() * target_sum
    if warm is None or f is None:
        # default initial shape: decreasing linear
        idx = np.arange(n, 0, -1, dtype=np.float64)   # n, n-1, ...,1
        f = idx / idx.sum() * target_sum

    best_f = f.copy()
    best_val = _objective(f, n)
    print(f"[Init] objective {best_val:.6f}")

    iteration = 0
    while time.time() < deadline - 1.0 and iteration < 40:
        # compute max convolution
        max_c = _full_convolution(f).max()
        M = max_c

        # build convolution matrix for constraints: f * g <= M
        # Toeplitz matrix
        col = np.concatenate([f, np.zeros(n-1, dtype=np.float64)])
        row = np.concatenate([f[:1], np.zeros(n-1, dtype=np.float64)])
        # use scipy.linalg.toeplitz to create dense matrix
        A = scipy.linalg.toeplitz(col, row)   # shape (2n-1, n)

        # upper bounds
        b = np.full(2 * n - 1, M, dtype=np.float64)

        # linear program: maximize sum(g) => minimize -sum(g)
        c = -np.ones(n, dtype=np.float64)

        bounds = [(0, None)] * n

        res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP] iteration {iteration} failed: {res.message}")
            break
        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            print(f"[LP] iteration {iteration} got zero sum")
            break
        # rescale to target sum
        g = g0 * (target_sum / sum_g0)

        # mixing step
        t = 0.3   # mixing coefficient
        f_new = (1 - t) * f + t * g

        # evaluate new f
        val_new = _objective(f_new, n)
        if val_new < best_val:
            best_val = val_new
            best_f = f_new.copy()
            print(f"[Iter {iteration}] improved objective {val_new:.6f}")

        # update f
        f = f_new
        iteration += 1

    # optional cheap local random search around best_f (few moves)
    # define local move function
    def _local_random_move(a, max_moves=20000):
        a_cur = a.copy()
        n_len = a_cur.shape[0]
        conv_cur = _full_convolution(a_cur).max()
        best_val_local = _objective(a_cur, n_len)
        best_a_local = a_cur.copy()
        for move in range(max_moves):
            if time.time() > deadline:
                break
            i = np.random.randint(n_len)
            if a_cur[i] < 1e-12:
                continue
            j = np.random.randint(n_len)
            if i == j:
                continue
            delta = a_cur[i] * (np.random.rand() * 0.3)   # move up to 30% of mass
            a2 = a_cur.copy()
            a2[i] -= delta
            a2[j] += delta
            val2 = _objective(a2, n_len)
            if val2 < best_val_local:
                best_val_local = val2
                best_a_local = a2.copy()
                a_cur = a2
        return best_a_local, best_val_local

    # perform local refinement if time permits
    if time.time() < deadline - 1.0:
        a_ref, val_ref = _local_random_move(best_f, max_moves=50000)
        if val_ref < best_val:
            best_val = val_ref
            best_f = a_ref
            print(f"[Local] refined objective {val_ref:.6f}")

    # final clipping
    best_f = np.clip(best_f, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_f)})")

"""
Propose a step‑function using an iterative linear‑programming (LP) refinement.
Each iteration solves

    max   Σ g_i
    s.t.  (f * g)[k] ≤ M   for all k,
          g_i ≥ 0,

where M = max(f * f).  The solution g is rescaled to the same ℓ₁‑norm as f,
mixed with f, and the process repeats.  The LP spreads mass to indices where f
contributes little to the convolution, thereby flattening the autocorrelation
and lowering the objective 2·n·max(a*a)/sum(a)².  A cheap random‑move hill‑climber
finishes the search.  The routine respects the time budget and always returns
a feasible sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _local_random_search(
    a0: np.ndarray, n: int, conv_fft_len: int, deadline: float, max_moves: int = 60_000
) -> tuple[np.ndarray, float]:
    """Simple hill‑climber: move a random fraction of mass from i→j."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.25)      # move ≤25 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The algorithm performs a few LP‑based refinement steps starting from a
    decreasing linear shape (or a warm start if available), followed by a cheap
    random‑move hill‑climber.  It always returns the best sequence found before
    the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin
    # --------------------------------------------------------------
    # problem size
    n = 2048                               # moderate size – fits comfortably in memory
    fft_len = _next_pow2(2 * n - 1)         # length for FFT based convolutions
    target_sum = np.sqrt(2.0 * n)          # convenient normalisation (scale‑invariant)

    # --------------------------------------------------------------
    # initialise f (warm start if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        f = np.asarray(warm, dtype=np.float64)
        if f.shape[0] > n:
            f = f[:n]
        elif f.shape[0] < n:
            f = np.pad(f, (0, n - f.shape[0]), constant_values=0.0)
        f = np.maximum(f, 0.0)
        if f.sum() >= 0.01:
            f = f / f.sum() * target_sum
        else:
            f = None
    if warm is None or f is None:
        # simple decreasing linear start (all entries positive)
        idx = np.arange(n, 0, -1, dtype=np.float64)          # n, n‑1, …, 1
        f = idx / idx.sum() * target_sum

    best_f = f.copy()
    # current objective (exact, no rounding)
    best_val = 2.0 * n * _full_convolution(f, fft_len).max() / (target_sum ** 2)
    print(f"[Init] objective {best_val:.6f}")

    # --------------------------------------------------------------
    # LP‑based refinement loop
    # --------------------------------------------------------------
    iteration = 0
    max_iters = 30                     # enough for the 1000 s budget
    while time.time() < deadline - 1.0 and iteration < max_iters:
        # current worst convolution value
        conv_f = _full_convolution(f, fft_len)
        M = conv_f.max()

        # build the (2n‑1) × n convolution matrix A  (A·g = f∗g)
        # first column = [f, 0,…,0] length 2n‑1, first row = [f0, 0,…,0] length n
        col = np.concatenate([f, np.zeros(n - 1, dtype=np.float64)])
        row = np.concatenate([f[:1], np.zeros(n - 1, dtype=np.float64)])
        # dense Toeplitz matrix (size ≈ 8 M entries → ~64 MiB)
        A = np.linalg.toeplitz(col, row)

        # linear programme: maximise Σ g_i  ↔ minimise –Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        try:
            import scipy.optimize as opt
            res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                              method='highs', options={'presolve': True})
        except Exception as e:
            print(f"[LP] iteration {iteration} failed to import/solve: {e}")
            break

        if not res.success:
            print(f"[LP] iteration {iteration} infeasible: {res.message}")
            break

        g0 = res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0.0:
            print(f"[LP] iteration {iteration} produced zero‑sum g")
            break

        # rescale g to the same ℓ₁‑norm as f (target_sum)
        g = g0 * (target_sum / sum_g0)

        # convex combination – the mixing weight can be tuned; 0.3 works well
        t = 0.30
        f = (1.0 - t) * f + t * g

        # evaluate new objective
        conv_f = _full_convolution(f, fft_len)
        val = 2.0 * n * conv_f.max() / (target_sum ** 2)

        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[Iter {iteration:02d}] new best {val:.6f}")

        iteration += 1

    # --------------------------------------------------------------
    # Final cheap random‑move hill‑climber (if time permits)
    # --------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_ref, val_ref = _local_random_search(best_f, n, fft_len, deadline,
                                              max_moves=40_000)
        if val_ref < best_val:
            best_val = val_ref
            best_f = a_ref
            print(f"[Local] refined objective {val_ref:.6f}")

    # --------------------------------------------------------------
    # Clip to the evaluator’s accepted range and return
    # --------------------------------------------------------------
    best_f = np.clip(best_f, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_f)})")
    return best_f.tolist()
