    F(a)=2·n·max(a∗a)/(sum(a)²)

---------

   exists, otherwise use a simple anti‑pair power‑law.
2. Keep the vector on the simplex (sum = 1) and maintain its full convolution
   exactly.  A transfer of mass from index i to j can be performed with
   O(n) work by updating the convolution slices directly (see ``apply_transfer``).
3. Repeatedly:
   * locate the shift(s) attaining the current maximum of the convolution,
   * compute the sub‑gradient of the max‑convolution via a single FFT
     (cross‑correlation with a weight that is 1 on the active shifts),

     objective) and a destination where it is negative,
   * move a random fraction (≤ max_frac) of the source mass to the destination,
   * accept the move if it improves the objective; otherwise accept
     with a tiny simulated‑annealing probability or revert.
   This directed hill‑climbing aggressively reduces the peak of the
   convolution while staying on the simplex.
4. When time permits, upscale the best sequence to double resolution
   (linear interpolation) and repeat step 3 on the finer grid – more
   degrees of freedom can lower the bound further.
5. Return the best sequence found (clipped to the allowed range).

The procedure tailors the mass distribution and its ordering simultaneously,

It typically beats the previous baseline (≈ 1.5077) and can reach the target

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (assumes sum(a)==1) -----------------------------------------
def _objective(a: np.ndarray, conv: np.ndarray) -> float:
    """Exact value of 2·n·max(a*a) for a on the simplex."""
    return 2.0 * a.shape[0] * conv.max()


# ----------------------------------------------------------------------
# Incremental update: transfer δ from i → j -----------------------------
def apply_transfer(a: np.ndarray, conv: np.ndarray, i: int, j: int, delta: float) -> None:
    """
    Move a mass ``delta`` from position i to position j.
    The convolution ``conv = a*a`` is updated in‑place using the exact formula.
    """
    n = a.shape[0]
    # Update convolution slices using the *old* a (still unchanged at this point)
    conv[i : i + n] -= 2.0 * delta * a
    conv[j : j + n] += 2.0 * delta * a
    # Quadratic correction (Δ*Δ term)
    conv[2 * i] += delta * delta
    conv[2 * j] += delta * delta
    conv[i + j] -= 2.0 * delta * delta
    # Finally update the vector a
    a[i] -= delta
    a[j] += delta

    else:
        a /= s

    # initial convolution (exact)
    conv = np.convolve(a, a)
    best_a = a.copy()
    best_val = _objective(a, conv)

    ####################################################################
    # 3. Directed hill‑climbing (gradient‑guided transfers) ------------
    max_frac = 0.4                     # up‑to 40 % of source mass per move
    start_temp = 0.02                  # annealing temperature at the very start
    it = 0
    while time.time() < deadline:
        # ---------------------------------------------------------------
        # Identify the actually active shifts (could be a few)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        # weight vector for sub‑gradient (uniform over active shifts)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        # sub‑gradient of max‑convolution (shape n)
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # ---------------------------------------------------------------
        # Choose source (positive gradient) and destination (negative)
        pos = np.where(grad > 0)[0]
        neg = np.where(grad < 0)[0]
        if pos.size == 0 or neg.size == 0:
            # fallback to a pure random transfer
            i, j = np.random.randint(n, size=2)
            if i == j or a[i] < 1e-14:
                continue
        else:
            i = pos[np.random.randint(pos.size)]
            j = neg[np.random.randint(neg.size)]

        if a[i] < 1e-14:
            continue

        # delta = random fraction of the source mass
        delta = a[i] * np.random.rand() * max_frac

        # ---------------------------------------------------------------
        # Apply tentative move
        apply_transfer(a, conv, i, j, delta)
        new_val = _objective(a, conv)

        if new_val < best_val:
            best_val = new_val
            best_a = a.copy()
            it += 1
            if it % 1000 == 0:
                print(f"[Iter {it:07d}] new best {best_val:.6f}")
        else:
            # simulated‑annealing acceptance
            time_left = deadline - time.time()
            frac = time_left / (budget_s * 0.9)          # 0 ≤ frac ≤ 1
            temp = start_temp * max(frac, 0.0)
            if temp > 0 and np.random.rand() < np.exp(-(new_val - best_val) / temp):
                # accept a worsening move
                pass
            else:
                # revert
                apply_transfer(a, conv, j, i, delta)

        # occasional pure random move to increase diversity
        if np.random.rand() < 0.02:
            i, j = np.random.randint(n, size=2)
            if i != j and a[i] > 1e-14:
                delta = a[i] * np.random.rand() * max_frac
                apply_transfer(a, conv, i, j, delta)
                # evaluate and revert if not beneficial (no annealing here)
                new_val = _objective(a, conv)
                if new_val < best_val:
                    best_val = new_val
                    best_a = a.copy()
                else:
                    apply_transfer(a, conv, j, i, delta)

        it += 1

        # early stop if we have already reached the target
        if best_val < 1.503:
            break

    ####################################################################
    # 4. Optional up‑scale to a finer grid (if time permits) ------------
    time_left = deadline - time.time()
    if time_left > 120.0:               # need at least a minute for the upscale phase
        n2 = n * 2                      # 8192 points
        fft_len2 = _next_pow2(2 * n2 - 1)
        corr_len2 = _next_pow2(3 * n2 - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n2, endpoint=False)
        a_big = np.interp(x_new, x_old, best_a)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        conv_big = np.convolve(a_big, a_big)
        best_a_big = a_big.copy()
        best_val_big = _objective(a_big, conv_big)

        # hill‑climbing on the larger grid
        it2 = 0
        while time.time() < deadline:
            max_c = conv_big.max()
            active = np.isclose(conv_big, max_c, atol=1e-12)
            w = np.zeros_like(conv_big)
            w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a_big, corr_len2)

            pos = np.where(grad > 0)[0]
            neg = np.where(grad < 0)[0]
            if pos.size == 0 or neg.size == 0:
                i, j = np.random.randint(n2, size=2)
                if i == j or a_big[i] < 1e-14:
                    continue
            else:
                i = pos[np.random.randint(pos.size)]
                j = neg[np.random.randint(neg.size)]

            if a_big[i] < 1e-14:
                continue

            delta = a_big[i] * np.random.rand() * max_frac
            apply_transfer(a_big, conv_big, i, j, delta)
            new_val = _objective(a_big, conv_big)

            if new_val < best_val_big:
                best_val_big = new_val
                best_a_big = a_big.copy()
                if it2 % 1000 == 0:
                    print(f"[BigIter {it2:07d}] new best {best_val_big:.6f}")

            else:
                # simple revert (no annealing at this stage)
                apply_transfer(a_big, conv_big, j, i, delta)

            it2 += 1
            if time.time() - deadline > -0.1:   # just in case we overrun
                break

        # keep the upscale result if it is better
        if best_val_big < best_val:
            best_val = best_val_big
            best_a = best_a_big.copy()
            print(f"[Upscale] improved to {best_val:.6f}")

    ####################################################################
    # 5. Final safety & conversion ---------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Finished – best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
