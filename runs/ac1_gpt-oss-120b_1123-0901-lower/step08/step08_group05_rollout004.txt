"""
Propose a step‑function that minimises
    F(a)=2·n·max(a∗a)/(∑a)²
using a multi‑stage search:

1. Warm‑start & a rich pool of analytic candidates (power‑law, exponential,
   Gaussian edges, linear, sinusoidal) with several orderings (anti‑pair,
   alternating, random).  The best of these is kept.

2. Adam on a smooth‑max (log‑sum‑exp) surrogate – fast reduction of the
   convolution peak.

3. Exact‑max sub‑gradient (exponentiated‑gradient) – guarantees descent of the
   true objective.

4. Stochastic hill‑climbing (swap / mass‑transfer / random perturbation)
   on the simplex.

5. Optional up‑scaling to a finer grid (8192) and a short repeat of steps 2‑4.

All phases respect the overall time budget (default 1000 s).  The routine
always returns a non‑negative list; the evaluator normalises the sum, so the
vector is returned with sum≈1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (assumes sum may be arbitrary)
def _objective(a: np.ndarray, conv_len: int | None = None) -> float:
    """Return 2·n·max(conv) where conv = a∗a and a is normalised to sum=1."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s
    n = a.shape[0]
    if conv_len is None:
        conv_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, conv_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Shape generators (all return a normalised non‑negative vector)
def _powerlaw_shape(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    vals = idx ** (-p)
    vals /= vals.sum()
    return vals


def _exponential_shape(n: int, lam: float) -> np.ndarray:
    """a_i ∝ exp(-lam·i)."""
    vals = np.exp(-lam * np.arange(n, dtype=np.float64))
    vals /= vals.sum()
    return vals


def _gaussian_edges_shape(n: int, gamma: float = 0.5) -> np.ndarray:
    """Two Gaussians placed at both ends."""
    i = np.arange(n, dtype=np.float64)
    vals = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
    vals /= vals.sum()
    return vals


def _linear_decreasing_shape(n: int) -> np.ndarray:
    """a_i ∝ (n‑i)."""
    vals = np.arange(n, 0, -1, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _linear_increasing_shape(n: int) -> np.ndarray:
    """a_i ∝ i+1."""
    vals = np.arange(1, n + 1, dtype=np.float64)
    vals /= vals.sum()
    return vals


def _sinusoidal_shape(n: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/n) (positive half‑wave)."""
    i = np.arange(n, dtype=np.float64) + 0.5
    vals = np.sin(np.pi * i / n)
    vals = np.maximum(vals, 0.0)
    vals /= vals.sum()
    return vals


def _uniform_shape(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Ordering heuristics
def _anti_pair_order(vals: np.ndarray) -> np.ndarray:
    """
    Largest ↔ smallest ordering.
    Input should already be sorted descending.
    """
    n = len(vals)
    a = np.empty(n, dtype=np.float64)
    half = n // 2
    a[:half] = vals[:half]                 # biggest half on the left
    a[-half:] = vals[-half:][::-1]         # smallest half on the right (reversed)
    if n % 2:
        a[half] = vals[half]               # middle element stays
    return a


def _alternating_order(vals: np.ndarray) -> np.ndarray:
    """
    Interleave large and small values: 0, n/2, 1, n/2+1, …
    Input should already be sorted descending.
    """
    n = len(vals)
    perm = np.empty(n, dtype=int)
    left, right, pos = 0, n // 2, 0
    while left < n // 2 and right < n:
        perm[pos] = left
        perm[pos + 1] = right
        left += 1
        right += 1
        pos += 2
    if n % 2:
        perm[-1] = n // 2
    a = np.empty_like(vals)
    a[perm] = vals
    return a


def _random_order(vals: np.ndarray) -> np.ndarray:
    """Random permutation (keeps the multiset unchanged)."""
    perm = np.random.permutation(len(vals))
    return vals[perm]


def _apply_order(vals: np.ndarray, order_type: str) -> np.ndarray:
    """Return a reordered version of `vals` according to `order_type`."""
    if order_type == "anti_pair":
        return _anti_pair_order(vals)
    if order_type == "alternating":
        return _alternating_order(vals)
    if order_type == "random":
        return _random_order(vals)
    # default: identity
    return vals.copy()


# ----------------------------------------------------------------------
# Warm start handling
def _warm_start(n: int) -> np.ndarray | None:
    """If a previous solution (`height_sequence_1`) exists, adapt it to length n."""
    warm = globals().get("height_sequence_1")
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    if warm.sum() < 1e-12:
        return None
    warm = warm / warm.sum()
    return warm


# ----------------------------------------------------------------------
# Optimisation primitives
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr: float = 0.09,
    lam_start: float = 1.0,
    lam_factor: float = 1.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate of max(convolution)."""
    eps = 1e-15
    logits = np.log(a0 + eps)  # work in log‑space (simplex via soft‑max)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate exact objective (for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best exact {best_val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # move to gradient in log‑space (mirrored simplex)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # increase temperature (approach true max)
        lam = min(lam * lam_factor, 1e6)

        # occasional jitter to escape flat regions
        if it % 4000 == 0 and (deadline - time.time()) > 5.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient refinement via an exponentiated‑gradient update.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # build sub‑gradient for the active maximal entries
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (keeps positivity)
        a = a * np.exp(-eta * grad_a)

        # re‑project onto simplex
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style hill‑climbing:
    - swap two positions,
    - transfer a fraction of mass,
    - random small perturbations (Gaussian).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_moves:
        # choose move type
        move_type = np.random.rand()
        if move_type < 0.4:
            # swap two positions
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                continue
        elif move_type < 0.7:
            # transfer a fraction from i to j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.35)  # up to 35 % of source mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
        else:
            # small Gaussian perturbation on a few entries
            idxs = np.random.choice(n, size=min(5, n), replace=False)
            perturb = np.random.randn(len(idxs)) * 0.01
            a2 = a.copy()
            a2[idxs] += perturb
            a2 = np.maximum(a2, 0.0)
            a2 = _project_simplex(a2)

        # compute objective
        val2 = _objective(a2, conv_fft_len)

        # accept if better, otherwise probabilistic acceptance
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # simple annealing schedule (linear cooling)
            temp = 0.015 * (1.0 - it / max_moves)
            if temp > 0 and np.random.rand() < np.exp(-(val2 - best_val) / temp):
                a = a2

        if it % 5000 == 0:
            print(f"[Local it={it:06d}] best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a low‑value step function within the given time budget."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # 1) Base resolution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 2) Build a diverse pool of starter vectors
    start_vectors: list[np.ndarray] = []

    # Warm start (if present)
    warm = _warm_start(n)
    if warm is not None:
        start_vectors.append(warm.copy())
        # rotated copies (sometimes help)
        for shift in (n // 4, n // 2, 3 * n // 4):
            start_vectors.append(np.roll(warm, shift))

    # Simple uniform
    start_vectors.append(_uniform_shape(n))

    # Analytic shapes + orderings
    shape_fns = [
        ("power", _powerlaw_shape, np.linspace(0.0, 6.0, 25)),   # fine grid
        ("exp", _exponential_shape, np.linspace(0.0, 6.0, 25)),
        ("gauss", _gaussian_edges_shape, np.linspace(0.2, 3.0, 12)),
        ("lin_dec", _linear_decreasing_shape, [0]),          # single param
        ("lin_inc", _linear_increasing_shape, [0]),
        ("sin", _sinusoidal_shape, [0]),
    ]
    orderings = ["anti_pair", "alternating", "random"]

    for name, fn, params in shape_fns:
        for p in params:
            if name in ("lin_dec", "lin_inc", "sin"):
                vals = fn(n)                     # no param
            else:
                vals = fn(n, p)
            # ensure descending for orderings that assume it
            vals_desc = np.sort(vals)[::-1]
            for order in orderings:
                cand = _apply_order(vals_desc, order)
                # small random jitter to escape exact symmetry
                if order == "random":
                    # already random; keep as is
                    pass
                start_vectors.append(cand)

    # Random Dirichlet samples (different concentration)
    alphas = [0.2, 0.5, 0.8, 1.0, 2.0, 5.0]
    for alpha in alphas:
        for _ in range(3):
            cand = np.random.dirichlet(alpha * np.ones(n))
            start_vectors.append(cand)

    # ------------------------------------------------------------------
    # 3) Quick evaluation of the pool – keep the best starter
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(start_vectors):
        # sanity – ensure non‑negative and normalised
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
            s = cand.sum()
        cand = cand / s
        val = _objective(cand, conv_fft_len)
        print(f"[Init cand {idx:03d}] value {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
    print(f"[Init] best starter value {best_val:.6f}")

    # --------------------------------------------------------------
    # 4) Continuous optimisation phases (Adam → Sub‑gradient)
    time_left = deadline - time.time()
    if time_left <= 0:
        # very late – return the best we have
        return best_seq.tolist()

    # Allocate roughly 40 % for Adam, 30 % for sub‑gradient, rest for local search
    adam_alloc = min(0.4 * time_left, 250.0)          # cap to keep room for later phases
    adam_deadline = time.time() + adam_alloc

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=500_000,
        lr=0.09,
        lam_start=1.0,
        lam_factor=1.07,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # Sub‑gradient refinement
    time_left = deadline - time.time()
    if time_left <= 0:
        return best_seq.tolist()
    subgrad_alloc = min(0.3 * time_left, 250.0)
    subgrad_deadline = time.time() + subgrad_alloc

    a_sub, val_sub = _subgradient_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=500_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Stochastic local search (swap / transfer / perturb)
    time_left = deadline - time.time()
    if time_left > 15.0:
        # allocate the bulk of the remaining time to local moves
        local_deadline = time.time() + max(20.0, 0.5 * time_left)
        max_moves = 400_000
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=max_moves,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # 6) Optional upscale to a finer grid (if time permits)
    time_left = deadline - time.time()
    if time_left > 60.0:
        # upscale to 8192 points (double resolution)
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation of the current best solution
        x_old = np.linspace(0.0, 1.0, n, endpoint=False)
        x_new = np.linspace(0.0, 1.0, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)

        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # brief Adam on the larger grid
        rem = deadline - time.time()
        adam_big_alloc = min(0.4 * rem, 120.0)
        adam_big_deadline = time.time() + adam_big_alloc
        a_big_adam, val_big_adam = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            adam_big_deadline,
            max_iters=300_000,
            lr=0.09,
            lam_start=1.0,
            lam_factor=1.07,
        )

        # sub‑gradient refinement on the large grid
        rem = deadline - time.time()
        sub_big_alloc = min(0.4 * rem, 120.0)
        sub_big_deadline = time.time() + sub_big_alloc
        a_big_sub, val_big_sub = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_big_deadline,
            max_iters=300_000,
            eta0=0.9,
        )
        if val_big_sub < best_val:
            best_val = val_big_sub
            best_seq = a_big_sub.copy()
            print(f"[Upscaled] improved to {val_big_sub:.6f}")

    # --------------------------------------------------------------
    # 7) Final safety & conversion
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range the evaluator accepts
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
