    # adam parameters

    v = np.zeros(n)

    lr = 0.05
    best_val = np.inf
    best_a = None
    t = 0
    # smoothing schedule: start with alpha=1, increase gradually up to maybe 30
    alpha = 1.0
    alpha_inc = 0.02  # per iteration maybe
    while time.time() < deadline and t < max_iters:
        a = softmax(x)  # sum=1
        # compute conv via FFT for efficiency
        conv = full_convolution(a)
        max_c = conv.max()
        # compute smooth weights

        exp_term = np.exp(alpha * (conv - max_c))
        p = exp_term / exp_term.sum()
        # gradient of smoothed max approximation: g = 2 * correlate(p, a, mode='valid')
        grad_a = 2.0 * np.correlate(p, a, mode='valid')
        # compute objective val for true objective
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # backprop through softmax: grad_x = a * (grad_a - a @ grad_a)
        avg = np.dot(a, grad_a)

        t += 1
        # Adam update
        m = beta1 * m + (1-beta1) * grad_x
        v = beta2 * v + (1-beta2) * (grad_x * grad_x)
        m_hat = m / (1 - beta1**t)

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

using a *smooth‑max* Adam optimiser followed by a (1+1)‑ES fine‑tuner.
The smooth‑max replaces the non‑differentiable “max” by a softmax
over the full self‑convolution, which yields a well‑behaved gradient.
The optimiser works on the simplex (softmax parametrisation) so the
constraint Σa = 1 is always satisfied.  Several restarts, a few
different initialisations and a final ES polishing are performed
within the allocated time budget (default 1000 s).  The best
sequence found (as plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps logits to a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returned array is real.
    """
    n = a.shape[0]
    # length of the FFT (next power of two for speed)
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by the external evaluate_sequence.
    The vector a is assumed to be non‑negative.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    # max of the linear convolution a*a
    max_conv = full_convolution(a).max()
    return 2.0 * a.shape[0] * max_conv / (s * s)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def run_smooth_adam(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a *smooth* approximation of max(a*a).
    The vector a lives on the simplex (softmax of logits).
    Returns the best a seen and its exact objective value.
    """
    # initialise logits from the starting probability vector
    x = np.log(init_a + 1e-15)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    # base learning rate – empirically works for all tested n
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    # smoothing coefficient (soft‑max temperature).  Starts small,
    # then slowly grows so that the approximation becomes tighter.
    alpha = 1.0
    alpha_step = 0.02  # increase per iteration, capped later

    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        # probability vector on the simplex
        a = softmax(x)                     # a ≥ 0, Σa = 1

        # exact convolution and the *true* max (needed for the objective)
        conv = full_convolution(a)
        max_c = conv.max()

        # ----- smooth‑max gradient -------------------------------------------------
        # p_k  =  exp(alpha * (conv_k - max_c))  /  Σ_j exp(alpha * (conv_j - max_c))
        # (subtract max_c for numerical stability)
        exp_term = np.exp(alpha * (conv - max_c))
        p = exp_term / exp_term.sum()            # probability over 2n‑1 indices

        # gradient of Σ_k p_k * conv_k  w.r.t. a  is  2 * correlate(p, a, "valid")
        grad_a = 2.0 * np.correlate(p, a, mode="valid")

        # exact objective value for tracking
        val = 2.0 * n * max_c                     # Σa = 1 → denominator = 1

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:06d}] new best: {val:.6f}")

        # back‑propagation through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits in a safe range
        x = np.clip(x, -50.0, 50.0)

        # gradually tighten the smooth‑max
        if alpha < 50.0:
            alpha = min(50.0, alpha + alpha_step)

        # occasional jitter to escape plateaus
        if t % 5000 == 0:
            x += (np.random.rand(n) - 0.5) * 1e-2

        # early‑stop if no improvement for a while
        if time.time() - last_improve > 5.0:
            # small random shake and continue searching
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    # fallback (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) Evolution Strategy on the simplex.
    Gaussian perturbations are projected back onto the simplex.
    Returns the best vector discovered before *deadline*.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.3)           # slightly increase step size
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        # reset when stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

    best_val = np.inf

    # try several problem sizes – larger n typically gives more freedom
    lengths = [8192, 4096, 2048, 1024]

    # ------------------------------------------------------------------
    # Helper to obtain a sensible initial candidate for a given n
    # ------------------------------------------------------------------
    def make_initial_candidates(n):
        cand = []

        # warm‑start from the globally best sequence if it already has length n
        if best_seq is not None and len(best_seq) == n:
            cand.append(np.array(best_seq, dtype=np.float64))

        # historic construction, if available in the global namespace
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 1e-6:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            cand.append(hs)

        # a simple decreasing (triangular) shape
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        cand.append(dec)

        # uniform distribution (the baseline)
        cand.append(np.full(n, 1.0 / n, dtype=np.float64))

        return cand

    # ------------------------------------------------------------------
    # Iterate over the selected lengths
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----------------------------------------------------------------
        # 1) Smooth‑max Adam phase (several restarts)
        # ----------------------------------------------------------------
        init_cands = make_initial_candidates(n)
        restarts = max(4, len(init_cands))
        for r in range(restarts):
            if time.time() > deadline:
                break

            if r < len(init_cands):
                a0 = init_cands[r]
            else:
                # random Dirichlet – concentration 0.5 favours sparser starts
                a0 = np.random.dirichlet(np.full(n, 0.5))

            # allocate a modest time slice for this Adam run
            remaining = deadline - time.time()
            slice_time = min(remaining * 0.18, 35.0)   # at most ~35 s per run
            sub_deadline = time.time() + slice_time

            a_opt, val_opt = run_smooth_adam(n, a0, sub_deadline)

            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ----------------------------------------------------------------
        # 2) (1+1)‑ES fine‑tuning – only if we have a candidate of this size
        # ----------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            es_slice = max(0.0, deadline - time.time())
            a_es, val_es = run_es_opt(n, best_seq, time.time() + es_slice)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n}] ES refined best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
