"""
Search for a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm consists of three layers:

1. **Random‑restart soft‑max Adam**  
   The sequence is represented as a soft‑max of logits `x`.  
   A smooth approximation of the max (log‑sum‑exp) is optimised with
   Adam.  The temperature τ is annealed aggressively, driving the
   surrogate ever closer to the true max.

2. **Targeted max‑convolution refinement**  
   After Adam stops, we identify the index where the self‑convolution
   attains its maximum and take a (projected) gradient step that
   reduces this peak.  The step uses a multiplicative update
   (exponential map) and a projection onto the simplex
   `{a ≥ 0, sum(a)=√(2 n)}`.

3. **Global orchestration**  
   * Many independent restarts with lengths 800‑5000.  
   * Warm‑starts from the best sequence seen so far (including the
     optional global `height_sequence_1`).  
   * Occasionally up‑sample the current best to a larger length.  
   * The whole process respects the 1000 s budget and continuously
     reports improvements.

The returned object is a plain Python list of floats.

import time

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    # next power of two for efficient FFT
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives caused by numerical error
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the simplex {x >= 0, sum(x) = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex with total mass s (>0)."""
    if s <= 0:
        raise ValueError("Simplex sum must be positive")
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero; return zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

    it = 0

    while time.time() < deadline:
        # --- 1. Map logits to the simplex via soft‑max -----------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # improve numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # actual sequence (non‑negative, correct sum)

        # --- 2. Compute convolution and current (smoothed) max -------
        conv = full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # --- 3. Smooth‑max weights (log‑sum‑exp) ----------------------
        tau = max(
            init_tau_factor * max_val * (0.95 ** it),
            min_tau_factor * max_val,
        )
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # --- 4. Gradient w.r.t. a (cross‑correlation) ----------------
        w_rev = w[::-1]
        conv2 = full_convolution_fft(a, w_rev)         # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)               # alignment for gradient
        grad_a = 2.0 * conv2[idx]

        # --- 5. Chain rule: gradient w.r.t. logits x -----------------
        dotp = np.dot(p, grad_a)                       # <p, grad_a>
        grad_x = target_sum * p * (grad_a - dotp)

        # --- 6. Adam update -------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny stochastic bump to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate (helps final convergence)
        alpha *= 0.9995

    # Final evaluation with the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Targeted refinement that directly attacks the maximal convolution entry
# ----------------------------------------------------------------------
def refine_max_convolution(
    seq: np.ndarray,
    target_sum: float,
    max_steps: int,
    step_size: float,
    deadline: float,
) -> np.ndarray:
    """
    Perform a small number of gradient‑like steps that reduce the
    largest entry of the self‑convolution.  The update uses a multiplicative
    rule (exponential map) followed by a projection onto the simplex.
    """
    a = seq.copy()
    n = a.shape[0]

    for step in range(max_steps):
        if time.time() >= deadline:
            break

        conv = full_convolution_fft(a, a)
        max_val = conv.max()
        k_max = int(np.argmax(conv))

        # gradient of conv[k_max] wrt a: grad_i = 2*a[k_max-i] if index in range
        i = np.arange(n)
        j = k_max - i
        mask = (j >= 0) & (j < n)
        grad = np.zeros_like(a)
        grad[mask] = 2.0 * a[j[mask]]

        # multiplicative (exponential) update
        a_new = a * np.exp(-step_size * grad)

        # project back onto the simplex (preserving sum = target_sum)
        a_new = project_onto_simplex(a_new, target_sum)

        # evaluate improvement
        conv_new = full_convolution_fft(a_new, a_new)
        if conv_new.max() < max_val - 1e-12:  # accept only strict improvement
            a = a_new
            # shrink step size gradually
            step_size *= 0.97
        else:
            # no improvement – reduce step size and continue
            step_size *= 0.90

    return a


# ----------------------------------------------------------------------
# Random up‑sampling / interpolation of an existing sequence
# ----------------------------------------------------------------------
def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly interpolate seq to length new_n."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    # old positions (0 … old_n‑1)
    old_pos = np.arange(old_n, dtype=float)
    # new positions spaced uniformly over the same interval
    new_pos = np.linspace(0, old_n - 1, new_n)
    interpolated = np.interp(new_pos, old_pos, seq)
    return interpolated


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the provided
    evaluation function.  The routine respects the time budget and
    returns the best sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Warm‑start from a globally supplied construction, if available.
    # ------------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global = {score:.6f}")
    except Exception:
        pass  # ignore missing global seed

    # ------------------------------------------------------------------
    # Main restart loop – allocate many short runs early, then longer ones.
    # ------------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 2.0:
            break  # insufficient time for a meaningful run

        # Choose length n (larger lengths later in the budget)
        if remaining > 600:
            n = np.random.randint(1200, 4500 + 1)
        elif remaining > 300:
            n = np.random.randint(900, 4000 + 1)
        elif remaining > 120:
            n = np.random.randint(600, 3500 + 1)
        else:
            n = np.random.randint(400, 2500 + 1)

        target_sum = math.sqrt(2.0 * n)   # Σ a_i = √(2 n)

        # -------------------------- initialise ---------------------------
        # Option 1: up‑sample current best (if reasonably sized)
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                # up‑sample by linear interpolation
                init = upsample_sequence(best_seq, n)
        else:
            # Random power‑law shape (the classic baseline)
            p_exp = np.random.uniform(0.2, 0.9)              # exponent
            idx = np.arange(1, n + 1, dtype=float)
            init = idx ** (-p_exp)

            # optional exponential taper
            if np.random.rand() < 0.4:
                tau_exp = np.random.uniform(0.4, 2.5) * n
                init *= np.exp(-idx / tau_exp)

            # optional smoothing by a short moving average
            if np.random.rand() < 0.25:
                klen = np.random.randint(3, 9)
                kernel = np.ones(klen) / klen
                init = np.convolve(init, kernel, mode='same')

            # Tiny jitter to avoid perfect symmetry
            init += np.random.rand(n) * 1e-6

        # Enforce positivity for the log‑it conversion
        init = np.maximum(init, 1e-12)

        # Scale to the required sum √(2 n) and switch to logits
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)

        # --------------------------------------------------------------
        # Run the Adam optimiser on this restart
        # --------------------------------------------------------------
        # Allocate a slice of the remaining budget (max 12 s per run)
        slice_time = min(12.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = adam_softmax_optimize(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.08,
            beta1=0.9,
            beta2=0.999,
        )

        # --------------------------------------------------------------
        # Targeted refinement of the maximal convolution entry
        # --------------------------------------------------------------
        refine_deadline = min(deadline, time.time() + 2.0)  # at most 2 s
        seq_refined = refine_max_convolution(
            seq_local,
            target_sum,
            max_steps=200,
            step_size=0.07,
            deadline=refine_deadline,
        )
        # Evaluate the refined sequence
        refined_score = evaluate_sequence(seq_refined.tolist())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = seq_refined.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:          # target bound reached – stop early
                break

        # ----------------------------------------------------------------
        # Occasionally, perform a cheap random pair‑swap mutation on the
        # global best to explore its neighbourhood.
        # ----------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.12:
            # small mass transfer between two random indices
            i, j = np.random.choice(best_seq.size, 2, replace=False)
            eps = 0.001 * target_sum
            if best_seq[i] > eps:
                candidate = best_seq.copy()
                candidate[i] -= eps
                candidate[j] += eps
                cand_score = evaluate_sequence(candidate.tolist())
                if cand_score < best_score:
                    best_score = cand_score
                    best_seq = candidate
                    print(f"[{time.time():.2f}] GLOBAL swap improve = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
