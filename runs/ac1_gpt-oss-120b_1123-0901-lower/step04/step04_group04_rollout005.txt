"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm – two‑phase optimisation
----------------------------------
* Phase 1 – smooth optimisation (L‑BFGS).  The objective 2·n·max(p∗p) is
  non‑smooth.  We replace the max by a smooth log‑sum‑exp (temperature τ)
  and minimise the smooth surrogate Lτ(p) = τ·log Σ exp((p∗p)/τ) using
  L‑BFGS on the unconstrained variables y with p_i = softmax(y_i).
  A modest cooling schedule (τ←0.7·τ) progressively tightens the
  approximation to the true max.  After each L‑BFGS stage the current
  probability vector is symmetrised (p←½(p+p[::-1])) – the optimal
  solution is known to be symmetric – and the corresponding y‑vector is
  updated.

* Phase 2 – sub‑gradient refinement.  Once τ is very small the smooth
  problem is essentially the original max‑problem.  We therefore switch
  to a true sub‑gradient method:  locate the index k where (p∗p) is
  maximal, the sub‑gradient w.r.t. p is 2·shift(p,k).  The gradient for
  y is obtained via the softmax Jacobian and a simple diminishing‑step
  rule is applied.  After each iteration we again enforce symmetry and
  keep the best sequence seen (exact objective computed by a fast FFT).

The code works with any length (default 2100, or the length of the global
``height_sequence_1`` if available).  It uses only NumPy and SciPy
(if present); a pure‑NumPy fallback is provided.  The whole search is
driven by a wall‑clock deadline (default 1000 s) and returns the best
sequence found before the timeout.
"""

def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Compute a stable soft‑max over the convolution values.
    Returns (weights, smooth max value).
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int):
    """
    Full convolution of p with itself using real FFT.
    Returns an array of length 2·len(p)‑1.
    """
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}.
    Implemented via FFT for speed.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep only the valid part (length = len(w)+len(a)-1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # rev = conv_rev[idx_start - i]  with idx_start = 2·len(a)‑2
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth objective Lτ and its gradient w.r.t. the log‑probability
    parameters y (where p = softmax(y)).
    """
    # soft‑max to obtain a probability vector
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stable exponentiation
    p = exp_y / exp_y.sum()

    # full self‑convolution
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p: 2·rev (weighted reverse convolution)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # transform to gradient w.r.t. y using the softmax Jacobian
    dot = np.dot(p, grad_p)                 # scalar = Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y


def _symmetrize(p: np.ndarray) -> np.ndarray:
    """Force symmetry p_i = p_{n-1-i}."""
    return 0.5 * (p + p[::-1])


def _run_l_bfgs(y0: np.ndarray,
                init_tau: float,

                deadline: float,

                best_score: float,
                eps_log: float = 1e-12) -> (np.ndarray, float):

    Repeatedly minimise the smooth surrogate using L‑BFGS while decreasing τ.
    Returns the best sequence (as a probability vector) and its exact score.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.7
    min_tau = 1e-12

    # pre‑compute factor 2·n which appears in the exact score
    two_n = 2.0 * n

    while time.time() < deadline and tau > min_tau:
        # ---- L‑BFGS on the current temperature ---------------------------------
        if not _SCIPY_AVAILABLE:          # fallback to a few manual gradient steps
            # simple gradient descent (10 steps) if SciPy missing
            for _ in range(10):
                L, g = _objective_and_grad(y, tau, Nfft)
                y -= 0.1 * g
        else:
            def fun(y_vec):
                L, g = _objective_and_grad(y_vec, tau, Nfft)
                return L, g

            opts = {'maxfun': 500, 'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
            y = res.x

        # ----- retrieve the probability vector, enforce symmetry ---------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        p = _symmetrize(p)                     # symmetry projection
        # re‑parametrise y from the symmetrised p (keeps softmax unchanged)
        y = np.log(p + eps_log)

        # ----- exact objective (no scaling needed) -----------------------------
        conv = _conv_full(p, Nfft)
        cur_score = two_n * conv.max()        # = evaluate_sequence for sum=1

        if cur_score < best_score:
            best_score = cur_score
            best_seq = p.copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature schedule & occasional jitter -----------------------
        tau *= tau_decay
        if np.random.rand() < 0.15:
            y += np.random.randn(n) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _subgradient_refine(p: np.ndarray,
                        y: np.ndarray,
                        n: int,
                        Nfft: int,
                        deadline: float,
                        best_seq: np.ndarray,
                        best_score: float,
                        eps_log: float = 1e-12) -> (np.ndarray, float):
    """
    Refine a solution using a true sub‑gradient method for the max‑convolution.
    """
    two_n = 2.0 * n
    t = 0
    step0 = 0.5   # initial step size (tuned empirically)

    while time.time() < deadline:
        # compute convolution and locate a maximal index
        conv = _conv_full(p, Nfft)
        max_val = conv.max()
        idx = np.argmax(conv)

        # construct rev = shift(p, idx)
        rev = np.zeros_like(p)
        lo = max(0, idx - (n - 1))
        hi = min(n - 1, idx)
        # i runs from lo..hi -> rev[i] = p[idx-i]
        if lo <= hi:
            rev[lo:hi + 1] = p[idx - np.arange(lo, hi + 1)]

        grad_p = 2.0 * rev

        # gradient w.r.t. y (softmax Jacobian)
        dot = np.dot(p, grad_p)
        grad_y = p * (grad_p - dot)

        # diminishing step size
        t += 1
        step = step0 / math.sqrt(t)

        y = y - step * grad_y

        # back to probability vector, enforce symmetry
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        p = _symmetrize(p)
        y = np.log(p + eps_log)   # re‑initialise y from the symmetrised p

        # occasional exact evaluation
        if t % 200 == 0:
            cur_score = two_n * max_val
            if cur_score < best_score:
                best_score = cur_score
                best_seq = p.copy()
                print(f"[subgrad improve] iter {t} score {best_score:.6f}")

        if time.time() >= deadline:
            break

    return best_seq, best_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.

    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # a small safety margin to guarantee we finish before the external timeout
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Determine the length n (use the reference construction if available)
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
        n = base.shape[0]
    except Exception:
        n = 2100                     # fall back to the length used previously
        base = np.ones(n, dtype=float)

    # we work directly with a probability vector (Σp = 1)
    eps_log = 1e-12

    # initialise best record with the provided construction (normalised)
    base_sum = base.sum()
    if base_sum == 0:
        base = np.ones(n, dtype=float)
        base_sum = n
    # normalise to a probability vector
    p0 = base / base_sum
    y0 = np.log(p0 + eps_log)

    best_seq = p0.copy()
    # exact objective for the initial point (sum=1 ⇒ 2·n·max)
    conv0 = _conv_full(p0, 1 << ((2 * n - 1).bit_length()))
    best_score = 2.0 * n * conv0.max()
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # generate a diverse set of starting points
    starts = []

    # 1) the known (possibly good) solution
    starts.append(y0)

    # 2) uniform distribution (log‑softmax of zeros)
    starts.append(np.zeros(n, dtype=float))

    # 3) a few random Dirichlet draws (via exponential variables)
    for _ in range(3):
        rand_exp = np.random.exponential(scale=1.0, size=n)
        starts.append(np.log(rand_exp + eps_log))

    # ------------------------------------------------------------------
    # main optimisation loop – each start runs L‑BFGS + sub‑gradient refinement
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # estimate a reasonable initial temperature
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv_start = _conv_full(p_start, Nfft)
        init_tau = max(conv_start.max() * 5.0, 1e-12)   # a bit larger than before

        # ---- Phase 1: smooth optimisation --------------------------------
        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            n=n,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            eps_log=eps_log,
        )

        # ---- Phase 2: sub‑gradient refinement ----------------------------
        # reconstruct y from the best probability vector found so far
        y_sym = np.log(best_seq + eps_log)
        best_seq, best_score = _subgradient_refine(
            p=best_seq,
            y=y_sym,
            n=n,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            eps_log=eps_log,
        )

    # ------------------------------------------------------------------
    # Final sanity check using the official evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
        print("[final check] improved by official evaluator")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # The evaluator tolerates any positive scaling; we keep the probability
    # vector (sum≈1) which respects all constraints.
    return best_seq.tolist()
