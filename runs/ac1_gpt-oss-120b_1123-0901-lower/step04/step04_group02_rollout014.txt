"""
Hybrid optimisation for the a*a‑max problem.

The algorithm works on the probability simplex (∑a_i = 1) and
tries to minimise

        F(a) = 2·n·max(conv(a,a))

where ``conv`` is the linear convolution.
It combines three ideas:

1. **Smooth‑max Adam** – a softmax approximation of the max
   (temperature λ) is gradually cooled while Adam updates
   move the point on the simplex.

2. **Exact‑max sub‑gradient steps** – after the temperature has
   become large the true arg‑max lag is used and a sub‑gradient
   of that single lag is applied (this yields a sharper direction
   when we are already close to a good region).

3. **Random local refinement** – simple mass‑transfer moves
   (pick i≠j, move a tiny δ from i to j) are accepted if they
   improve the exact objective.  This stage cleans up the
   remaining noise.

The routine respects the global time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but fall back to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_obj(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv(a,a)) (assuming ∑a_i = 1)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


def _smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Softmax weights proportional to exp(lam·conv)."""
    mx = conv.max()
    w = np.exp(lam * (conv - mx))
    s = w.sum()
    if s == 0.0:
        return np.full_like(w, 1.0 / w.shape[0])
    return w / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Hybrid optimiser for the step‑function problem.
    Returns a plain Python list that (when passed to evaluate_sequence)
    yields a low score.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # ------------------------------------------------------------------
    # Problem sizes we will explore (larger n gives more freedom).
    # The very first size is small so that we can find a decent prototype
    # quickly; later sizes are fine‑tuned.
    lengths = [1024, 2048, 4096]

    best_seq = None          # numpy array, sum exactly 1
    best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)

        # ------------------------------------------------------------------
        # Number of independent runs for this length.
        restarts = 6
        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------- initialise a -------------------
            if r == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm‑start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                a = cand / s if s > 0 else np.random.dirichlet(np.ones(n))
                print("[restart] start from historic sequence")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random Dirichlet start")

            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.12          # a little more conservative than previous code

            # smooth‑max temperature schedule
            lam = 0.2          # start very smooth
            lam_factor = 1.03  # increase slowly
            lam_max = 1e4

            t = 0
            max_iters = 30000
            last_improve = time.time()
            current_val = _exact_obj(a, n)

            # ------------------------------------------------------------------
            # Main optimisation loop (Adam with smooth‑max → exact‑max)
            while time.time() < deadline and t < max_iters:
                # ---- convolution ----
                conv = _full_convolution(a, fft_len_conv)

                # ---- choose gradient direction ----
                if lam < lam_max / 10:                     # smooth phase
                    w = _smooth_weights(conv, lam)
                    grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
                else:                                      # exact max phase
                    k_max = int(conv.argmax())
                    # gradient of conv[k_max] (sub‑gradient of the max)
                    #  g_i = 2 * a_{k_max-i} for i != k_max/2, else = a_i
                    # This can be obtained by a cross‑correlation with a one‑hot.
                    e = np.zeros_like(conv)
                    e[k_max] = 1.0
                    grad = 2.0 * _cross_correlation(e, a, fft_len_corr)

                # ---- Adam update (before projection) ----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ---- projection onto the simplex ----
                a = _project_simplex(a_new)

                # ---- temperature schedule ----
                lam = min(lam * lam_factor, lam_max)

                # ---- monitoring ----
                if t % 250 == 0 or time.time() - last_improve > 5.0:
                    val = _exact_obj(a, n)
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[n={n} r={r} iter={t:05d}] new best = {val:.6f}")
                        if best_val < 1.503:
                            print("Target reached – terminating early.")
                            deadline = time.time()
                            break
                    # occasional shaking to avoid stagnation
                    if time.time() - last_improve > 6.0:
                        noise = (np.random.rand(n) - 0.5) * 2e-2
                        a = _project_simplex(a + noise)
                        last_improve = time.time()
                        print(f"[n={n} r={r} iter={t}] shake")

            # end of one restart
        # end of restarts for this length

        # ------------------------------------------------------------------
        # Local random hill‑climbing refinement for the current best
        # (only if we still have time)
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            a = best_seq.copy()
            print("\n--- Local random refinement (pairwise transfers) ---")
            refine_deadline = deadline
            while time.time() < refine_deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a tiny fraction of mass from i to j
                delta = 0.004 * a[i] * np.random.rand()
                if delta <= 0:
                    continue
                a_prop = a.copy()
                a_prop[i] -= delta
                a_prop[j] += delta
                a_prop = _project_simplex(a_prop)
                val = _exact_obj(a_prop, n)
                if val < best_val:
                    best_val = val
                    best_seq = a_prop.copy()
                    a = a_prop
                    print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback to uniform, value = {best_val:.6f}")

    # Return as plain Python list.
    return best_seq.tolist()
