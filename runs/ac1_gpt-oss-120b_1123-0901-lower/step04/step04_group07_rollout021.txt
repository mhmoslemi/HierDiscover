def propose_candidate(seed=12345, budget_s=1000, **kwargs):
    import time, math, numpy as np, random, sys
    from scipy.optimize import linprog

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0

    # Choose length n
    n = 2000 # can adjust based on time

    target_sum = math.sqrt(2.0 * n)

    # Initialize a: either from height_sequence_1 interpolated, or random
    try:
        base = np.array(height_sequence_1, dtype=float)
    except Exception:
        base = None

    if base is not None and base.shape[0] != n:
        # interpolate
        old_x = np.linspace(0,1,base.shape[0])
        new_x = np.linspace(0,1,n)
        a = np.interp(new_x, old_x, base)
        a = np.maximum(a, 0)
    else:
        # random Dirichlet
        rng = np.random.default_rng()
        a = rng.gamma(1.0, 1.0, size=n)
    # Normalize to sum target_sum
    a = a * (target_sum / a.sum())

    best_a = a.copy()
    best_score = 2 * n * np.convolve(best_a, best_a).max() / (target_sum**2)  # should equal max conv after norm

    # gradient descent
    max_iters = 300000
    eps_active = 1e-12
    lr0 = 0.2
    for it in range(1, max_iters+1):
        if time.time() >= deadline:
            break
        # fft convolution
        n_fft = 1 << ((2*n-1).bit_length())
        A_fft = np.fft.rfft(a, n=n_fft)
        conv = np.fft.irfft(A_fft * A_fft, n=n_fft)[:2*n-1]
        M = conv.max()
        # active set
        mask = conv >= M - eps_active
        # compute weighted rev (average)
        # Determine active indices array
        idxs = np.nonzero(mask)[0]
        w = np.zeros_like(conv)
        if len(idxs) > 0:
            w[idxs] = 1.0 / len(idxs)
        # weighted reverse convolution via FFT
        # rev = convolution of a reversed with w (like earlier code)
        W_rev_fft = np.fft.rfft(w[::-1], n=n_fft)
        rev = np.fft.irfft(W_rev_fft * A_fft, n=n_fft)
        # rev length = len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2, we need slice corresponding to i=0..n-1 as earlier
        start_idx = 2*n - 2
        rev_slice = rev[start_idx - np.arange(n)]
        grad = 2.0 * rev_slice  # subgradient of max conv
        # gradient descent step
        lr = lr0 / math.sqrt(it)
        a = a - lr * grad
        # projection onto nonnegative simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)
        # evaluate score
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_a = a.copy()
            print(f"[iter {it}] new best {best_score:.6f}")

        # jitter occasional
        if it % 5000 == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

"""
Hybrid search for a low‑peak step function.

The objective returned by ``evaluate_sequence`` is
    f(a) = 2·n·max(a∗a) / (sum(a)²)

For a vector normalised to sum = √(2n) the objective equals
    max_convolution = max(a∗a).

The algorithm consists of two phases:

1️⃣  Projected sub‑gradient descent (using the exact sub‑gradient of the
    max‑convolution).  The gradient is obtained by a weighted reverse
    convolution; the step size decays as 1/√it.  After each step the
    vector is projected onto the non‑negative simplex of the required ℓ₁
    norm.  Occasional tiny Gaussian jitter helps escape plateaus.

2️⃣  A few refinement steps based on a linear programme:
    given the current vector `f` with peak `M = max(f∗f)`,
    solve   max 1ᵀb   s.t.   b ≥ 0 and   f∗b ≤ M   (elementwise).
    The solution `g` spreads the mass as much as possible without
    raising the peak.  After scaling `g` to the correct sum we mix it
    with `f` (f ← (1‑t)·f + t·g) and repeat.

Both phases are cheap (FFT for the gradient, a single dense LP for the
refinement) and fit easily into the 1000 s budget.  The code keeps the
best vector seen so far and finally returns it as a plain Python list.
"""
import time
import math
import numpy as np
from scipy.optimize import linprog


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k‑i}
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    The result has length len(a).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep only the part that is needed (len(w) + n - 1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # slice that corresponds to rev_i for i = 0 .. n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max(a∗a) using FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _project_onto_simplex(v: np.ndarray, target_sum: float) -> np.ndarray:
    """
    Euclidean projection of a vector onto the non‑negative simplex
    {x ≥ 0, Σx = target_sum}.  The simple scaling used in the reference
    implementation is sufficient for our purpose.
    """
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0.0:
        v = v * (target_sum / s)
    else:
        # all entries were zero – fill uniformly
        v = np.full_like(v, target_sum / v.shape[0])
    return v
