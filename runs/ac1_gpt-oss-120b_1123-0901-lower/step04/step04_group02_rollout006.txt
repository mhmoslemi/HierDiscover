"""
propose_candidate
-----------------

Search for a non‑negative step function (probability vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

where “*” denotes the linear convolution.  The objective is invariant to
overall scaling, therefore the search is carried out on the probability
simplex (∑a = 1, a ≥ 0).

The algorithm consists of three stages

1. **Broad random sampling** on several sequence lengths.  Dirichlet
   samples give a cheap way to explore the space; the best few
   candidates are kept.

2. **Projected Adam optimisation with a sub‑gradient of the exact
   ℓ∞‑norm**.  Instead of the smooth‑max surrogate used previously we
   take the (average) sub‑gradient of the true maximum convolution
   entry.  A small tolerance around the maximum is used to keep a few
   near‑max entries active, which yields a stable descent direction.
   The iterates are projected onto the simplex after each Adam step.

3. **Fine‑grained local refinement** by transferring a tiny amount of
   mass between two coordinates (hill‑climbing) and by occasional
   random “shakes’’ when progress stalls.

The search respects a global time budget (default 1000 s).  Several
sequence lengths up to 16384 are examined – larger lengths provide more
degrees of freedom, allowing the maximal convolution value to be
reduced below the target 1.503.  The best sequence found (as a plain
Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures, no lambdas)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _exact_objective(a: np.ndarray) -> float:
    """
    Exact evaluation of 2·n·max(a*a) for a probability vector.
    (sum(a) == 1)
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()

    Returns the best sequence as a plain Python list.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None      # numpy array, always lies on the simplex
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Random sampling on a range of lengths
    # ------------------------------------------------------------------
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # a rough budget for the sampling phase (≈ 20 % of total time)
    sampling_deadline = time.time() + (budget_s * 0.18)

    for n in lengths:
        if time.time() > sampling_deadline:
            break
        # number of Dirichlet draws – more for smaller n
        draws = max(30, int(300_000 / n))
        print(f"[sampling] length={n}, draws={draws}")

        for _ in range(draws):
            if time.time() > sampling_deadline:
                break
            a = np.random.dirichlet(np.ones(n))
            val = _exact_objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"  new global best (sample) = {val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Projected Adam optimisation (sub‑gradient of exact max)
    # ------------------------------------------------------------------
    # we keep the best length found so far; later we will also try
    # enlargements of that solution.
    if best_seq is None:
        # very unlikely, but fall back to uniform of modest size
        n0 = 1024
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_val = _exact_objective(best_seq)

    # move up to larger lengths using linear interpolation;
    # this lets us reuse a good shape at higher resolution.
    target_lengths = [best_seq.shape[0], 4096, 8192, 16384]
    target_lengths = [ln for ln in target_lengths if ln >= best_seq.shape[0]]

    for n in target_lengths:
        if time.time() > deadline:
            break
        if n != best_seq.shape[0]:
            # upscale the current best by linear interpolation
            x_old = np.linspace(0, 1, best_seq.shape[0])
            x_new = np.linspace(0, 1, n)
            a = np.interp(x_new, x_old, best_seq)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            print(f"[upscale] {best_seq.shape[0]} → {n}")
        else:
            a = best_seq.copy()

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.25                     # base learning rate

        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)

        # optimisation loop
        max_iters = 25000
        t = 0
        last_improve = time.time()
        # tolerance for the “max‑set” – 0.1 % of the current maximum
        rel_tol = 1e-3

        while time.time() < deadline and t < max_iters:
            # ---------- convolution ----------
            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            # ---------- sub‑gradient (indicator of near‑max) ----------
            thresh = max_c - max(rel_tol * max_c, 1e-12)
            mask = (conv >= thresh).astype(np.float64)
            if mask.sum() == 0:
                mask[conv.argmax()] = 1.0
            w = mask / mask.sum()

            # ---------- gradient ----------
            grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

            # ---------- Adam update ----------
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a = _project_simplex(a_new)

            # ---------- evaluate ----------
            val = 2.0 * n * conv.max()        # same as _exact_objective(a) but reuse conv
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[n={n} iter={t:05d}] new best = {val:.6f}")

                # terminate early if we already beat a comfortable margin
                if best_val < 1.503:
                    print("Target reached – exiting optimisation early.")
                    deadline = time.time()   # forces outer loops to stop
                    break

            # ---------- stagnation handling ----------
            if time.time() - last_improve > 6.0:
                # add a tiny uniform noise and re‑project
                noise = (np.random.rand(n) - 0.5) * 1e-2
                a = _project_simplex(a + noise)
                last_improve = time.time()
                print(f"[n={n} iter={t}] shake (stagnation)")

        # finished this length – keep the best that was found
        print(f"[summary] length={n}, best value so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  Final local random refinement (mass‑transfer hill‑climbing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # transfer a small fraction of the mass from i to j
            delta = 0.005 * a[i] * np.random.rand()   # up to 0.5 % of a[i]
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            # re‑project to be safe (maintains positivity & sum=1)
            a_prop = _project_simplex(a_prop)
            val = 2.0 * n * _full_convolution(a_prop, fft_len_conv).max()
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Convert to plain Python list as required by the external evaluator
    print(f"\n=== Finished: best value = {best_val:.6f}, length = {best_seq.shape[0]} ===")
    return best_seq.tolist()
