def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5
    best_seq = None
    best_val = np.inf
    lengths = [256,512,1024,2048,4096,8192,16384]
    for n in lengths:
        if time.time() > deadline:
            break
        # compute FFT lengths
        conv_fft = 1 << int(np.ceil(np.log2(2*n-1)))
        corr_fft = 1 << int(np.ceil(np.log2(3*n)))  # safe
        # generate candidate list
        candidates = []
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq))
        elif best_seq is not None:
            # upsample previous best
            prev = np.array(best_seq)
            old_n = prev.shape[0]
            # linear interpolation
            x_old = np.linspace(0,1,old_n)
            x_new = np.linspace(0,1,n)
            up = np.interp(x_new, x_old, prev)
            up = np.maximum(up,0)
            if up.sum() < 1e-12:
                up = np.random.dirichlet(np.ones(n))
            else:
                up = up / up.sum()
            candidates.append(up)
        # add random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))
        # add edge-concentration shapes
        idx = np.arange(n) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in (-1.5, -1.0, -0.5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            shape /= shape.sum()
            candidates.append(shape)
        # add Gaussian shapes
        i = np.arange(n, dtype=np.float64)
        sigma = n/8.0
        shape = np.exp(-(i-n/2)**2/(2*sigma**2))
        shape /= shape.sum()
        candidates.append(shape)

        for cid, a0 in enumerate(candidates):
            # Stop if deadline
            if time.time() > deadline: break
            # ensure probabilities
            a = a0.copy()
            if a.sum() <= 0:
                a = np.random.dirichlet(np.ones(n))
            a = a / a.sum()
            # Adam scaling
            a_opt = adam_optimize(a, n, conv_fft, corr_fft, deadline, best_val)
            # subgradient
            a_opt = subgradient_optimize(a_opt, n, conv_fft, corr_fft, deadline, best_val)
            # maybe random shake final
            a_opt = random_shake_optimize(a_opt, n, conv_fft, deadline, best_val)
            # compute exact val
            val = exact_F(a_opt, n, conv_fft)
            if val < best_val:
                best_val = val
                best_seq = a_opt.copy()
                print(f'New best at n={n} cand={cid} val={val:.6f}')
    if best_seq is None:
        best_seq = np.full(1024, 1/1024)
    return best_seq.tolist()

def adam_optimize(a0, n, fft_len_conv, fft_len_corr, deadline, current_best):
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02
    max_iters = int(20000 * 1024 / n) + 3000
    last_improve = time.time()
    best_local_val = np.inf
    best_local_a = a.copy()
    for t in range(1, max_iters+1):
        if time.time() > deadline - 0.5:
            break
        # softmax
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()
        # conv
        conv = _full_convolution(a, fft_len_conv)
        # smooth max weights
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # backprop through softmax.
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* Work on a hierarchy of lengths: 256 → 512 → … → 16384.
  The best solution from a smaller length is up‑sampled (linear
  interpolation) and used as a warm‑start for the next length.
* For each length we generate a modest set of diverse seeds:
  – the up‑sampled best,
  – the historic ``height_sequence_1`` (if present),
  – a random Dirichlet draw,
  – a few deterministic shapes (edge‑concentrated power‑law,
    Gaussian‑ish window).
* Every seed is refined by three optimisation phases that work
  directly on the exact objective:
  1. Adam on a smooth‑max surrogate of the convolution (temperature
     λ is raised geometrically).  The parameters are tuned per‑size
     and the phase stops after a few seconds without improvement.
  2. An exponentiated‑gradient sub‑gradient method on the true
     max‑convolution (active shifts receive uniform weight).
  3. A short “high‑λ Adam” run that acts as a fine‑grained
     polishing step.
* After each exact evaluation the global best sequence is updated.
* The whole search respects the supplied time budget (default 1000 s)
  and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]                     # reverse once
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]        # length n


def _exact_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of the objective F(a) for a given length."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    # sum(a) is always positive (handled by callers)
    return float(2.0 * n * max_c / (a.sum() ** 2))

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    current_best: float,
) -> np.ndarray:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best sequence (probability vector) found during this phase.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    # iteration budget grows when n is small, caps for large n
    max_iters = int(25000 * 1024 / n) + 3000
    max_iters = min(max_iters, 8000)          # safety cap
    best_local = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline - 0.5:
            break

        # ---------- simplex via soft‑max ----------
        mx = logits.max()
        exp_logits = np.exp(logits - mx)
        a = exp_logits / exp_logits.sum()

        # ---------- convolution ----------
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # ---------- smooth‑max weights ----------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---------- gradient of the surrogate ----------
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # ---------- back‑prop through soft‑max ----------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---------- Adam update ----------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---------- temperature schedule ----------
        lam = min(lam * lam_factor, 1e6)

        # ---------- exact evaluation ----------
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_local = a.copy()
            last_improve = time.time()
            # keep global best in outer loop (outside this function)

        # ---------- stagnation handling ----------
        if time.time() - last_improve > 4.0:
            # small random shake to escape flat regions
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_local

def _subgradient_exponentiated(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
) -> np.ndarray:
    """
    Sub‑gradient descent on the true max‑convolution using an
    exponentiated‑gradient (multiplicative) update.
    """
    a = a0.copy()
    eta = 0.6
    best_local = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    max_iters = 5000 if n <= 4096 else 2500
    for it in range(max_iters):
        if time.time() > deadline - 0.5:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_local = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 4.0:
            break

    return best_local

def _high_lambda_fine_tune(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,

    """
    a = a0.copy()

    beta1, beta2, eps = 0.9, 0.999, 1e-8

def _upscale_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a larger size."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.random.dirichlet(np.ones(new_n))
    else:
        up = up / s
    return up

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Length schedule – start small, gradually increase.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for all lengths.
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # cross‑correlation needs at most conv_len + n - 1 ≤ 3·n
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    for n in lengths:
        if time.time() > deadline:

        print(f"\n=== Optimising length n={n} ===")
        conv_fft = fft_len_conv[n]
        corr_fft = fft_len_corr[n]

        candidates = []

        # warm start from the best of the previous size (up‑scaled)
        if best_seq is not None:
            cand = _upscale_sequence(np.asarray(best_seq, dtype=np.float64), n)
            candidates.append(cand)

        # historic solution ``height_sequence_1`` (if present)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=np.float64)
            if h.shape[0] != n:

            h = np.maximum(h, 0.0)
            s = h.sum()
            if s < 1e-12:
                h = np.random.dirichlet(np.ones(n))
            else:

            candidates.append(h)

        # pure random Dirichlet start
        candidates.append(np.random.dirichlet(np.ones(n)))

        # deterministic families – edge‑concentrated power‑law

        # Gaussian‑ish window centred in the middle

        gauss = gauss / gauss.sum()
        candidates.append(gauss)
