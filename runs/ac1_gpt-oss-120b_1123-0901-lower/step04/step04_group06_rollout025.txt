Now, we need to implement projection on simplex.

Standard algorithm from Duchi et al. "Efficient Projections onto the l1-Ball..." Modified for simplex: Given vector v, we want nonnegative w s.t sum w = S. To project onto simplex, subtract offset lambda so that w = max(v - lambda, 0) and sum w = S. Equivalent to sorting v descending and finding rho.

Implement:

def project_onto_simplex(v, s):
    """Project onto simplex {x >= 0, sum x = s}."""
    n = len(v)
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # find rho = max i such that u_i - (cssv_i / (i+1)) > 0
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if len(rho) == 0:
        # all zero?
        theta = cssv[-1] / n
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Edge cases: if sum(v) == s and all non-negative, returns same.

def _optimize_fw(n, a0, target_sum, run_deadline, max_iter=80, t0=0.5, t_decay=0.9):
    a = a0.copy()
    best_seq = a.copy()
    best_score = np.max(_full_convolution_fft(a, a))
    iteration = 0
    t = t0
    while time.time() < run_deadline and iteration < max_iter:
        conv = _full_convolution_fft(a, a)
        M = conv.max()
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} new local best = {M:.6f}")
        # Build constraint matrix
        # idx matrix for a indexing
        rows = np.arange(2*n-1)[:, None] - np.arange(n)
        A_ub = np.where((rows >= 0) & (rows < n), a[rows], 0.0)   # shape (2n-1, n)
        # Solve LP
        c = -np.ones(n)  # minimize -sum(b)
        b_ub = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        # optionally limit time per linprog by using time left
        # but high-level not available; we trust it'll be fast.

        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs', options={'presolve': True})
        if not res.success:
            # fallback: break
            break
        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break
        g = target_sum * b0 / sum_b0
        # convex combo
        a = (1 - t) * a + t * g
        # reproject onto simplex due to numeric errors
        a_sum = a.sum()
        if a_sum != 0:
            a *= (target_sum / a_sum)
        # decay step size
        t *= t_decay
        iteration += 1
    # final evaluation
    final_score = np.max(_full_convolution_fft(best_seq, best_seq))
    # using evaluator for safety
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

"""
Propose a non‑negative height sequence that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a Frank‑Wolfe style optimisation that iteratively
solves a very small linear programme:

    maximise   Σ b_j
    subject to 0 ≤ b_j
                (a * b)_k ≤ M   for all k,
    where M = max(conv(a, a))

The LP adds as much total “mass’’ as possible without increasing the
current worst convolution value M.  After solving, the LP solution b is
renormalised to have the required total sum √(2 n) and blended with the
current a (a ← (1‑t) a + t g).  Repeating this step strictly reduces M
and drives the objective towards the theoretical optimum.

To avoid getting stuck in a bad local shape we perform many
independent restarts with random power‑law initialisations, occasionally
seeding from the best sequence discovered so far.  Each restart receives a
time slice (≈ 8‑12 s) and the overall run stops as soon as the bound falls
below the target 1.503.  The final result is returned as a plain Python
list.

All helper functions are top‑level; the code uses only numpy, scipy and
standard library facilities and respects the 1000 s budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.linalg import toeplitz


# ----------------------------------------------------------------------
# Full linear convolution via FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # tiny negative values can appear because of round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Projection onto the simplex {z ≥ 0, Σ z = s}
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of total mass s."""
    if s < 0:
        raise ValueError("Simplex mass must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    # Find the smallest rho such that u[rho] - cssv[rho]/(rho+1) ≤ 0
    rho_arr = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        theta = cssv[-1] / n
    else:
        rho = rho_arr[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    """
    Frank‑Wolfe optimiser for a fixed length ``n``.
    Returns the best sequence found and its evaluation value.
    """
    a = a0.copy()
    best_seq = a.copy()
    # because we keep the sum exactly target_sum, the true score equals max(conv)
    best_score = np.max(_full_convolution_fft(a, a))
    t = t_init
    iteration = 0

    while time.time() < run_deadline and iteration < max_iter:
        # --------------------------------------------------------------
        # 1. current worst convolution value M
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # keep the best ever seen in this run
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f} (iter={iteration})"
            )

        # --------------------------------------------------------------
        # 2. build the (a * b) ≤ M constraints: a * b is a convolution,
        #    which can be written as a dense Toeplitz matrix.
        # --------------------------------------------------------------
        # first column of the Toeplitz matrix: a followed by zeros
        col = np.concatenate([a, np.zeros(n - 1, dtype=a.dtype)])
        # first row: a[0] followed by zeros
        row = np.concatenate([[a[0]], np.zeros(n - 1, dtype=a.dtype)])
        A_ub = toeplitz(col, row)  # shape (2n‑1, n)

        # --------------------------------------------------------------
        # 3. linear programme: maximise Σ b  <=> minimise –Σ b
        # --------------------------------------------------------------
        c = -np.ones(n, dtype=a.dtype)          # minimise –total mass
        b_ub = np.full(2 * n - 1, M, dtype=a.dtype)
        bounds = [(0.0, None)] * n

        # solve with HiGHS (fast interior‑point solver)
        res = linprog(
            c,
            A_ub=A_ub,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if not res.success:
            # if the LP fails we abort the inner loop – keep the current a
            print(f"[{time.time():.2f}]   LP failed (iter={iteration}), exiting FW loop")
            break

        b0 = res.x
        sum_b0 = b0.sum()
        if sum_b0 <= 0:
            break

        # --------------------------------------------------------------
        # 4. rescale to the required total mass and convex‑combine
        # --------------------------------------------------------------
        g = target_sum * b0 / sum_b0
        a = (1.0 - t) * a + t * g

        # tiny numerical drift – re‑project onto the simplex
        a = _project_onto_simplex(a, target_sum)

        # --------------------------------------------------------------
        # 5. schedule the convex‑combination weight
        # --------------------------------------------------------------
        t *= t_decay
        iteration += 1

    # final evaluation using the provided evaluator (guards against any clipping)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length n (≈ 500 – 2000) and initialise a
      power‑law shape a_i ∝ (i+1)^{‑p} (p∈[0.2,0.8]), optionally tapered
      by an exponential factor.
    * Run the Frank‑Wolfe optimiser (linear‑programming sub‑step) for a
      short time slice (≈ 8–12 s).  Each iteration reduces the worst
      self‑convolution value M.
    * Keep the global best across many independent restarts,
      occasionally warm‑starting from the current best.
    * Stop early if we beat the target bound 1.503.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start global seed = {best_score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Choose a problem size.  Early on we explore larger n,
        # later we concentrate on cheaper runs.
        if remaining > 400:
            n = np.random.randint(1200, 2000 + 1)
        elif remaining > 150:
            n = np.random.randint( 800, 1500 + 1)
        else:
            n = np.random.randint( 500, 1000 + 1)

        target_sum = np.sqrt(2.0 * n)          # Σ a_i = √(2 n)

        # ------------------- initial shape ----------------------------
        p_exp = np.random.uniform(0.2, 0.8)                 # power‑law exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # optional exponential taper
        if np.random.rand() < 0.35:
            tau_exp = np.random.uniform(0.5, 3.0) * n
            init *= np.exp(-idx / tau_exp)

        # ensure positivity before the log‑it conversion
        init = np.maximum(init, 1e-12)
        init = init / init.sum() * target_sum

        # occasional warm‑start from the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])
            init = np.maximum(init, 1e-12)
            init = init / init.sum() * target_sum

        # --------------------------------------------------------------
        # Allocate a time slice for this run (≈ 8‑12 s, never exceed deadline)
        # --------------------------------------------------------------
        slice_time = min(12.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # --------------------------------------------------------------
        # Run the Frank‑Wolfe optimiser
        # --------------------------------------------------------------
        seq_local, score_local = _optimize_fw(
            n,
            init,
            target_sum,
            run_deadline,
            max_iter=80,
            t_init=0.3,
            t_decay=0.93,
        )

        # --------------------------------------------------------------
        # Update the global best if improvement was found
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:               # target exceeded – stop early
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
