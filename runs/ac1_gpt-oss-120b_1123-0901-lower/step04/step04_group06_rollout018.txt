def project_onto_simplex(v, S):
    # follows algorithm from Duchi et al.
    if np.sum(v) == S and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - S
    ind = np.arange(len(v)) + 1
    # find rho where u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The algorithm is a hybrid of three ideas

1. **Soft‑max Adam optimisation** – the coefficients are parametrised as a
   soft‑max of unconstrained logits; a smooth approximation of the max
   (log‑sum‑exp) provides a differentiable surrogate.  Adam with a
   slowly decaying learning‑rate and temperature explores many random
   initialisations (different lengths, power‑law / exponential shapes,
   random Dirichlet vectors).  This stage finds good “continuous” minima.

2. **Projected sub‑gradient descent** – starting from the best candidate,
   the exact sub‑gradient of the (non‑smooth) maximum convolution peak is
   computed (it is simply a reversed slice of the current sequence).  A
   step in the opposite direction is taken and the result is projected
   back onto the simplex (∑ aᵢ = √(2 n), aᵢ ≥ 0) with the O(n log n)
   simplex‑projection algorithm of Duchi et al.

3. **Simple hill‑climbing** – a final stochastic local‑search that repeatedly
   moves a tiny amount of mass from a random index i to a random index j,
   accepting the move if the peak decreases (or with a small
   temperature‑controlled probability).  This refines the result without
   any gradient machinery.

All three stages respect the 1000 s budget, share a global best answer
and fall back to a uniform construction if nothing is found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT (used by all stages)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full (linear) convolution of real vectors x and y."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # tiny negative values appear because of rounding – clip them
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. 2008
# ----------------------------------------------------------------------
def _project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex {w ≥ 0, Σw = s}."""
    if s < 0:
        raise ValueError("simplex radius must be non‑negative")
    n = v.shape[0]
    # make a copy to avoid mutating the input
    u = np.sort(v)[::-1]                # descending sort
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)           # 1‑based indices
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would be projected to zero – return the zero vector
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1              # zero‑based index of the last positive entry
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 0.001,
    alpha0: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser for the smoothed‑max surrogate.
    Returns the best coefficient vector encountered and its
    **exact** evaluation (via evaluate_sequence).
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_a = None
    best_val = np.inf
    it = 0
    alpha = alpha0

    while time.time() < deadline:
        # ----- map logits to the simplex (a = target_sum * softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)           # stable soft‑max
        p = exp_shift / np.sum(exp_shift)       # probabilities, Σp = 1
        a = target_sum * p

        # ----- current (exact) max of the convolution -----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_a = a.copy()
            print(f"[{time.time():.2f}] n={n}   new local best max={cur_max:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp surrogate) -----
        tau = max(init_tau_factor * cur_max * (0.99 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a  (2 · a ⊛ w_rev) -----
        w_rev = w[::-1]                         # reversed weights
        size2 = 1 << ((2 * n - 2).bit_length())
        Fa = np.fft.rfft(a, n=size2)
        Fw = np.fft.rfft(w_rev, n=size2)
        conv2 = np.fft.irfft(Fa * Fw, n=size2)[: 3 * n - 2]   # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                     # positions that correspond to Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- back‑prop to logits (chain rule) -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny jitter to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    # final exact evaluation via the provided evaluator (identical to max)
    final_score = evaluate_sequence(best_a.tolist())
    return best_a, float(final_score)


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *true* max‑convolution
# ----------------------------------------------------------------------
def _subgradient_descent(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on max(conv(a, a)) with projection onto the simplex.
    Returns the best sequence and its evaluation.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()
    step_size = 0.1 * target_sum / n          # initial step (roughly 1 % of the mean)

    it = 0
    while time.time() < deadline:
        it += 1
        conv = _full_convolution_fft(a, a)
        k = int(conv.argmax())                # index of the current maximum

        # gradient of conv_k w.r.t. a (sub‑gradient of the max)
        grad = np.zeros_like(a)
        # valid range for j such that 0 ≤ k‑j < n
        j_start = max(0, k - (n - 1))
        j_end = min(n - 1, k)
        js = np.arange(j_start, j_end + 1, dtype=int)
        grad[js] = 2.0 * a[k - js]

        # descent step and projection
        a_candidate = a - step_size * grad
        a_candidate = _project_onto_simplex(a_candidate, target_sum)

        val_candidate = _full_convolution_fft(a_candidate, a_candidate).max()

        # accept if improvement, otherwise shrink the step size
        if val_candidate < best_val - 1e-12:
            a = a_candidate
            best_a = a_candidate
            best_val = val_candidate
            # modestly increase step size after a successful move
            step_size *= 1.02
        else:
            step_size *= 0.70  # aggressive reduction on failure

        # occasional random jitter to keep the search alive
        if it % 300 == 0 and np.random.rand() < 0.15:
            jitter = np.random.rand(n) * 1e-4
            a = _project_onto_simplex(a + jitter, target_sum)

        if it % 500 == 0:
            print(
                f"[{time.time():.2f}] subgrad it={it} step={step_size:.3e} best={best_val:.6f}"
            )
        # stop early if we reach a very good bound
        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the exact objective
# ----------------------------------------------------------------------
def _hillclimb(
    a0: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Random local search that transfers a tiny amount of mass between two
    positions while keeping the simplex constraint.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    best_val = _full_convolution_fft(a, a).max()

    # temperature schedule – start relatively hot, cool to near zero
    T0 = best_val * 0.3
    T_end = 1e-4
    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # linear cooling based on elapsed fraction
        elapsed = time.time() - start
        fraction = elapsed / (deadline - start)
        T = max(T_end, T0 * (1.0 - fraction))

        i, j = np.random.randint(0, n, size=2)
        if i == j or a[i] <= 0:
            continue

        # move at most 10 % of the mass currently at i
        delta = a[i] * np.random.rand() * 0.10
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _full_convolution_fft(a_new, a_new).max()

        accept = False
        if new_val < best_val:
            accept = True
        else:
            # Metropolis criterion
            if np.random.rand() < np.exp(-(new_val - best_val) / (T + 1e-12)):
                accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[{time.time():.2f}] hillclimb it={it} new best={best_val:.6f}")

        # occasional full‑re‑normalisation to avoid drift
        if it % 2000 == 0:
            a = _project_onto_simplex(a, target_sum)

        if best_val < 1.503:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Hybrid search for a step‑function with a minimal evaluation score.
    The routine respects the given time budget, uses several optimisation
    strategies and finally returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15    # tiny safety margin

    best_seq = None
    best_score = np.inf

    # --------------------------------------------------------------
    # Warm‑start from a globally available construction, if any.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        n0 = hs_arr.shape[0]
        target0 = np.sqrt(2.0 * n0)
        hs_norm = hs_arr * (target0 / hs_arr.sum())
        score0 = evaluate_sequence(hs_norm.tolist())
        best_seq = hs_norm.copy()
        best_score = float(score0)
        print(f"[{time.time():.2f}] warm‑start seed score={best_score:.6f}")
    except Exception:
        pass

    # --------------------------------------------------------------
    # Phase 1 – many short Adam runs with diverse lengths / initialisations.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate a modest slice (≈ 8‑12 s) for a single Adam optimisation
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = time.time() + slice_time

        # pick a problem size – gradually bias towards cheaper runs as time runs out
        if remaining > 400:
            n = np.random.randint(1800, 3500 + 1)
        elif remaining > 180:
            n = np.random.randint(1300, 2800 + 1)
        else:
            n = np.random.randint( 800, 2000 + 1)

        target_sum = np.sqrt(2.0 * n)

        # ---------- initialise logits ----------
        # 70 % of runs use a random Dirichlet start, the rest a power‑law/exponential blend
        if np.random.rand() < 0.7:
            a0 = np.random.rand(n)
        else:
            p_exp = np.random.uniform(0.20, 0.80)
            idx = np.arange(1, n + 1, dtype=float)
            a0 = idx ** (-p_exp)
            if np.random.rand() < 0.4:
                tau_exp = np.random.uniform(0.5, 2.0) * n
                a0 *= np.exp(-idx / tau_exp)
        a0 = a0 / a0.sum() * target_sum
        x0 = np.log(a0 + 1e-12)                     # logits for the soft‑max

        # optionally inject the current global best (trimmed / padded)
        if best_seq is not None and np.random.rand() < 0.35:
            if best_seq.shape[0] >= n:
                a0 = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.shape[0]) * 1e-5
                a0 = np.concatenate([best_seq, pad])
            a0 = a0 / a0.sum() * target_sum
            x0 = np.log(a0 + 1e-12)

        # ---------- run the Adam optimiser ----------
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=0.001,
            alpha0=0.03,
        )

        # update the global best if we improved
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best={best_score:.6f} (n={n})"
            )
            if best_score < 1.503:                # early exit condition
                break

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent (exact max).
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 8.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        sub_end = time.time() + min(30.0, time_left * 0.5)
        seq_sub, val_sub = _subgradient_descent(best_seq.copy(), target_sum, sub_end)
        if val_sub < best_score:
            best_score = val_sub
            best_seq = seq_sub.copy()
            print(f"[{time.time():.2f}] after sub‑gradient best={best_score:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final stochastic hill‑climb for fine‑tuning.
    # --------------------------------------------------------------
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        hill_end = time.time() + time_left
        seq_hill, val_hill = _hillclimb(best_seq.copy(), target_sum, hill_end)
        if val_hill < best_score:
            best_score = val_hill
            best_seq = seq_hill.copy()
            print(f"[{time.time():.2f}] after hill‑climb best={best_score:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never be used).
    # --------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)
        best_score = evaluate_sequence(best_seq.tolist())
        print(f"[{time.time():.2f}] fallback uniform score={best_score:.6f}")

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
