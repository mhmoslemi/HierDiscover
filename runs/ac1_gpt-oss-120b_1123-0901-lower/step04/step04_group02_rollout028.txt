"""
propose_candidate
-----------------

Construct a step‑function by placing a few *blocks* of equal mass on the line.
Each block has constant height (mass / width).  The start positions of the
blocks are chosen as a Sidon set (all unordered sums are distinct) so that
no two block‑pair convolutions peak at the same index.  For a block of mass
1/k and width w the maximal convolution value is (1/k²)/w, giving the
objective

    2·n·max(conv) = 2·n·(1/k²·1/w) ≈ 2·(2k‑1)/k²,

which for k≥3 is well below the target 1.503.  The routine tries several
values of (n,k) inside the time budget, builds a Sidon layout with a simple
greedy randomised search, evaluates the exact score with the supplied
``evaluate_sequence`` and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _build_sidon_layout(n: int, k: int, w: int, deadline: float) -> list[int] | None:
    """
    Try to place ``k`` blocks of width ``w`` inside a vector of length ``n``.
    The start indices are required to be a Sidon set (all unordered sums are
    distinct) and blocks must not overlap.  Several random greedy attempts are
    performed; on success a list of start positions is returned, otherwise
    ``None``.
    """
    max_attempts = 80                     # enough chances for small k
    for _ in range(max_attempts):
        if time.time() > deadline:
            break
        # candidate start positions – random order each attempt
        candidates = np.arange(0, n - w + 1, dtype=np.int64)
        np.random.shuffle(candidates)

        starts: list[int] = []
        sums: set[int] = set()            # stores all a+b for a,b in starts

        for cand in candidates:
            # ---- block overlap test ----
            overlap = False
            for s in starts:
                if not (cand + w - 1 < s or cand > s + w - 1):
                    overlap = True
                    break
            if overlap:
                continue

            # ---- Sidon sum uniqueness test ----
            conflict = False
            # check sums with already placed starts
            for s in starts:
                if (cand + s) in sums:
                    conflict = True
                    break
            if conflict:
                continue
            # self‑sum must also be new
            if (cand + cand) in sums:
                continue

            # ---- accept this start ----
            for s in starts:
                sums.add(cand + s)
            sums.add(cand + cand)
            starts.append(int(cand))
            if len(starts) == k:
                return starts
    return None


def _make_sequence(n: int, k: int, w: int, starts: list[int]) -> np.ndarray:
    """
    Given length ``n``, number of blocks ``k``, block width ``w`` and the
    start positions, create the full step‑function (probability vector).
    """
    a = np.zeros(n, dtype=np.float64)
    block_mass = 1.0 / k
    height = block_mass / w
    for s in starts:
        a[s:s + w] = height
    # renormalise to guard against rounding errors
    tot = a.sum()
    if tot > 0:
        a /= tot
    return a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list[float]:
    """
    Search for a low‑score sequence using a multi‑block Sidon construction.
    The function respects the global time budget and returns a plain Python
    list suitable for ``evaluate_sequence``.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Try a few different lengths – larger vectors give more flexibility.
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        # Upper bound for a Sidon set inside [0,n) is roughly sqrt(n).
        max_k = int(np.sqrt(n))               # heuristic
        for k in range(max_k, 1, -1):
            if time.time() > deadline:
                break

            # We need w such that the trivial spacing condition (2k‑1)·w ≤ n.
            w = n // (2 * k - 1)
            if w < 1:
                continue

            starts = _build_sidon_layout(n, k, w, deadline)
            if starts is None:
                continue

            seq = _make_sequence(n, k, w, starts)

            # The external checker returns ``float('inf')`` for invalid inputs,
            # but with our construction the sequence is always valid.
            val = evaluate_sequence(seq.tolist())

            print(f"[n={n} k={k} w={w}] value = {val:.6f}")

            if val < best_val:
                best_val = val
                best_seq = seq.copy()
                if best_val < 1.503:          # target already reached
                    print(f"Target reached – best value {best_val:.6f}")
                    deadline = time.time()    # stop further work
                    break

        # early exit if target already hit
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # If nothing succeeded (very unlikely) fall back to a uniform vector.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform sequence – value = {best_val:.6f}")

    # Return plain list as required by the external evaluator.
    return best_seq.tolist()
