"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer works in the log‑it (soft‑max) space so that the
non‑negativity and the simplex constraint Σ a_i = √(2 n) are satisfied
exactly.  The objective is approximated by a *smooth max* (soft‑max)
plus a variance penalty on the convolution.  Adam is used for fast,
adaptive optimisation.  A multi‑restart scheme explores many random
lengths (≈ 500‑3500) and a variety of parametric initial shapes; the best
sequence seen so far is kept as a warm‑start for later runs.

Key ideas
---------
* **Soft‑max parametrisation** – a_i = S·exp(x_i)/∑exp(x) with
  S = √(2 n).  Guarantees a_i ≥ 0 and Σa_i = S.
* **Smooth‑max surrogate** – w_max ∝ exp((b–max(b))/τ) gives a smooth
  approximation of max(b).  τ decays exponentially.
* **Flat‑convolution bias** – the variance of the convolution
  (Var(b) = mean((b–mean(b))²)) is added to the surrogate with a
  time‑decaying weight λ.  This pushes the self‑convolution towards a
  flat profile, which empirically lowers the peak.
* **Gradient** – for a weighted sum L = Σ_k w_k b_k the gradient w.r.t.
  a is 2·(a ⊛ w_rev).  Using the chain rule we obtain the gradient w.r.t.
  the log‑it variables x.
* **Adam update** – adaptive steps with β₁=0.9, β₂=0.999.  Small random
  jitters are injected periodically to escape plateaus.
* **Coarse‑to‑fine multi‑restart** – many short runs with different
  lengths and initial shapes (power‑law, exponential, Gaussian,
  beta‑distribution) are performed within the 1000 s budget.  The
  current global best (or a stored global seed) is used as a warm‑start.

The routine prints progress and finally returns the best sequence found
as a plain Python list.

import time

# --------------------------------------------------------------
# Helper – fast full linear convolution via real FFT
# --------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negatives from numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# --------------------------------------------------------------
# Adam optimiser on the log‑it variables with smooth‑max + variance
# --------------------------------------------------------------
def _optimise_adam_softmax_variance(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    var_lambda_initial: float = 5.0,
    var_lambda_decay: float = 0.995,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it space.
    Returns the best sequence (as a NumPy array) and its exact score
    obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- 1. map logits to the simplex (a = S·softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)   # stable soft‑max
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p               # a lives on the simplex Σa = S

        # ----- 2. convolution and current max -----
        conv = _full_convolution_fft(a, a)
        max_b = conv.max()
        # local best (max_b equals the true objective because Σa = √(2n))
        if max_b < best_score:
            best_score = max_b
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_b:.6f}")

        # ----- 3. smooth‑max weights -----
        tau = max(init_tau_factor * max_b * (0.95 ** iteration),
                  min_tau_factor * max_b)
        scaled = (conv - max_b) / tau
        w_max = np.exp(scaled)
        w_sum = w_max.sum()
        if w_sum == 0.0:
            w_max = np.full_like(w_max, 1.0 / w_max.size)
        else:
            w_max /= w_sum

        # ----- 4. variance weights (encourages flat convolution) -----
        avg_b = conv.mean()
        m_len = conv.shape[0]                     # = 2*n - 1
        w_var = (2.0 / m_len) * (conv - avg_b)    # sum(w_var) == 0

        # time‑decaying variance weight
        var_lambda = max(var_lambda_initial * (var_lambda_decay ** iteration), 0.0)

        # ----- 5. combine weights for gradient -----
        w = w_max + var_lambda * w_var

        # ----- 6. gradient w.r.t. a (weighted sum) -----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                  # pick Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- 7. chain rule to logits x -----
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- 8. Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if iteration % 200 == 0 and np.random.rand() < 0.3:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate (but keep it positive)
        alpha *= 0.9995

    # ----- final evaluation with the provided evaluator -----
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function. The routine runs for at most ``budget_s``
    seconds and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length ``n`` (≈ 500 – 3500) for each restart.
    * Initialise a shape using one of several parametric families
      (power‑law, exponential, Gaussian, beta‑distribution) with a tiny
      jitter.
    * Optimise the shape with the Adam‑based soft‑max + variance‐penalty
      descent defined above.
    * Keep the overall best across many independent restarts; warm‑start
      from the current global best with probability 0.45.
    * Stop early if we beat the target bound (≈ 1.503).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose a problem size. Earlier we explore a broad range,
        # later we concentrate to keep iterations cheap.
        if remaining > 500:
            n = np.random.randint(1500, 3500 + 1)
        elif remaining > 200:
            n = np.random.randint(1000, 2500 + 1)
        else:
            n = np.random.randint(600, 1800 + 1)

        # ------------------- initial shape ----------------------------
        shape_type = np.random.choice(["power", "exp", "gauss", "beta"])
        idx = np.arange(n, dtype=float)

        if shape_type == "power":
            exp_pow = np.random.uniform(0.1, 1.5)
            init = (idx + 1) ** (-exp_pow)
        elif shape_type == "exp":
            lam = np.random.uniform(0.5, 3.0) / n
            init = np.exp(-lam * idx)
        elif shape_type == "gauss":
            mu = np.random.uniform(0, n - 1)
            sigma = np.random.uniform(0.1 * n, 0.3 * n)
            init = np.exp(-0.5 * ((idx - mu) / sigma) ** 2)
        else:  # beta distribution
            a = np.random.uniform(0.5, 2.5)
            b = np.random.uniform(0.5, 2.5)
            t = (idx + 0.5) / n
            init = (t ** (a - 1)) * ((1 - t) ** (b - 1))

        # tiny jitter to break symmetry
        init += np.random.rand(n) * 1e-5

        # Occasionally inject the current global best as a warm‑start
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        # positivity and scaling to the correct L1 norm
        init = np.maximum(init, 1e-12)
        target_sum = np.sqrt(2.0 * n)               # Σ a_i = √(2 n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)                   # logits for soft‑max

        # ------------------------------------------------------------------
        # Allocate a time slice for this run (≈ 8‑12 s, never exceed deadline)
        # ------------------------------------------------------------------
        slice_time = min(12.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # Run the optimiser for this length
        seq_local, score_local = _optimise_adam_softmax_variance(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.05,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
            var_lambda_initial=5.0,
            var_lambda_decay=0.995,
        )

        # Update the global best if improvement was found
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            # early exit if we comfortably beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
