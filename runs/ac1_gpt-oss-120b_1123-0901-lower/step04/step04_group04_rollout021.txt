"""
Propose a step‑function (non‑negative vector) that minimises the
evaluate_sequence objective.

Algorithm
---------
* Work with the probability vector  p = a / Σa  (∑p = 1).  The exact
  objective equals 2·n·max(p∗p) where ∗ is the ordinary (linear) convolution.
* Approximate the max with a smooth log‑sum‑exp function
        Lτ(p) = τ·log Σ exp((p∗p)_k/τ) .
  Its gradient can be written using a weighted reverse convolution, which
  can be evaluated efficiently with FFTs.
* Optimise Lτ with L‑BFGS‑B on the unconstrained log‑probability vector y
  (p_i = softmax(y_i)).  A continuation scheme gradually reduces the
  temperature τ, starting from a value a few times the current smooth max.
* Perform many random restarts (uniform, Dirichlet, linear / quadratic ramps,
  and perturbed versions of the known solution).  For each start run the
  continuation optimiser and keep the best exact solution found.
* Finally apply a cheap stochastic “local‑refinement’’ that perturbs the
  logits y and accepts any improvement (a few hundred cheap Adam‑style
  steps).  This helps escape shallow basins after the deterministic phase.
* The whole search respects the supplied time budget (≈1000 s).  The best
  sequence (as a plain Python list) is returned.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy – used for the L‑BFGS‑B optimiser.
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Helper utilities (all top‑level, no closures)

def _softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max, returns a probability vector that sums to 1."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of convolution values.
    Returns weight vector w (summing to 1) and the smooth max Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full linear convolution p∗p using real FFTs."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute      rev_i = Σ_k w_k · a_{k‑i}
    efficiently via FFTs (indices outside the support are ignored).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _smooth_obj_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Returns (Lτ, ∇_y Lτ) for the given logits y at temperature τ.
    Also returns p (softmax of y) and max(conv(p,p)) for convenience.
    """
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # stable
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient wrt p (using the weighted reverse convolution)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # transform gradient to y (Jacobian of soft‑max)
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y, p, conv.max()


def _run_lbfgs(y0: np.ndarray,
               init_tau: float,
               target_sum: float,
               Nfft: int,
               deadline: float,
               best_seq: np.ndarray,
               best_score: float):
    """
    Run a continuation of L‑BFGS‑B on the smooth max.
    Returns the (possibly improved) best sequence and its exact score,
    together with the final log‑probability vector y.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.85          # slower cooling than the reference code
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS‑B on the current temperature -----------------------
        if not _SCIPY_AVAILABLE:                # pragma: no cover
            # fall back to a few Adam steps (very crude)
            step_size = 0.02
            m = np.zeros_like(y)
            v = np.zeros_like(y)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            for _ in range(200):
                L, g, _, _ = _smooth_obj_and_grad(y, tau, Nfft)
                m = beta1 * m + (1 - beta1) * g
                v = beta2 * v + (1 - beta2) * (g * g)
                m_hat = m / (1 - beta1)
                v_hat = v / (1 - beta2)
                y -= step_size * m_hat / (np.sqrt(v_hat) + eps)
            # go immediately to evaluation
        else:
            def fun(y_vec):
                L, g, _, _ = _smooth_obj_and_grad(y_vec, tau, Nfft)
                return L, g

            opts = {'maxfun': 500, 'maxiter': 500,
                    'ftol': 1e-12, 'gtol': 1e-12}
            res = minimize(fun, y, method='L-BFGS-B', jac=True,
                           options=opts)
            y = res.x

        # ----- exact evaluation -----------------------------------------
        p = _softmax(y)
        max_conv = _conv_full(p, Nfft).max()
        cur_score = (target_sum ** 2) * max_conv   # exact evaluate_sequence result

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- temperature update & occasional jitter --------------------
        tau *= tau_decay
        if np.random.rand() < 0.20:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break

    return best_seq, best_score, y


def _local_refinement(best_seq: np.ndarray,
                      target_sum: float,
                      Nfft: int,
                      deadline: float):
    """
    Simple stochastic refinement in log‑probability space.
    Small Gaussian perturbations are accepted if they improve the exact
    objective.  A few hundred attempts are cheap compared to the main
    optimisation and often yield a tiny additional gain.
    """
    n = best_seq.shape[0]
    p = best_seq / target_sum
    best_score = evaluate_sequence(best_seq.tolist())
    sigma = 0.03
    attempts = 0
    max_attempts = 1200

    while time.time() < deadline and attempts < max_attempts:
        # propose a random perturbation of the logits
        y = np.log(p + 1e-12)
        y_new = y + np.random.randn(n) * sigma
        p_new = _softmax(y_new)
        seq_new = p_new * target_sum
        score = evaluate_sequence(seq_new.tolist())
        if score < best_score:
            best_score = score
            best_seq = seq_new
            p = p_new
            print(f"[local] improved score={best_score:.6f}")
        attempts += 1
        # slowly shrink the step size
        if attempts % 200 == 0:
            sigma *= 0.85
    return best_seq, best_score

    starts.append(np.log(p0 + eps))

    # 3) random Dirichlet – exponential(1) then normalise
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + eps))

    # 4) linear decreasing ramp
    lin = np.linspace(1.0, 0.0, n)
    lin = lin / lin.sum()
    starts.append(np.log(lin + eps))

    # 5) quadratic decreasing ramp
    quad = (np.linspace(1.0, 0.0, n) ** 2)
    quad = quad / quad.sum()
    starts.append(np.log(quad + eps))

    # 6) perturbed uniform
    starts.append(np.random.randn(n) * 0.05)

    # 7‑9) a few extra random seeds
    for _ in range(3):
        starts.append(np.random.randn(n) * 0.1)

    # ------------------------------------------------------------------
    # Main optimisation loop – run the continuation optimiser from each seed.
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # Initial temperature – a couple of times the current smooth max.
        p_start = _softmax(y_start)
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score, _ = _run_lbfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # A cheap stochastic fine‑tuning starting from the best vector found.
    if time.time() < deadline:
        best_seq, best_score = _local_refinement(
            best_seq=best_seq,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
        )

    # Final sanity check.
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
