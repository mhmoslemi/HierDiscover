def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimizing length n={n} ===")
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # Determine number of restarts based on remaining time
        # We'll attempt up to 5 restarts for smaller n, fewer for larger.
        restarts = 5 if n <= 4096 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # initialization
            if best_seq is not None and len(best_seq) == n:
                # warm start
                a0 = best_seq.copy()
                a0 = a0 / a0.sum()
                x = np.log(a0 + 1e-15)
            elif 'height_sequence_1' in globals():
                init = np.array(height_sequence_1, dtype=np.float64)
                if init.shape[0] >= n:
                    init = init[:n]
                else:
                    init = np.pad(init, (0, n - init.shape[0]))
                init = np.maximum(init, 1e-12)
                init = init / init.sum()
                x = np.log(init + 1e-15)
            else:
                init = np.random.dirichlet(np.ones(n))
                x = np.log(init + 1e-15)

            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2
            alpha = 1.0 # smoothness parameter
            alpha_max = 200.0
            decay_alpha_every = 3000

            last_improve = time.time()
            max_iters = 50000  # large, will be limited by deadline
            t_iter = 0

            while time.time() < deadline and t_iter < max_iters:
                # softmax
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # conv
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_conv = conv.max()
                # compute smooth weights
                # compute scaled conv for numerical stability
                # shift by max_conv to avoid overflow
                scaled = conv - max_conv
                w = np.exp(alpha * scaled)
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.zeros_like(conv)
                    w[conv.argmax()] = 1.0
                else:
                    w = w / w_sum

                # gradient wrt a
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # softmax backprop
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t_iter += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t_iter)
                v_hat = v / (1.0 - beta2 ** t_iter)

                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # update learning rate decay
                if t_iter % 2000 == 0:
                    lr *= 0.7

                # update alpha (smoothness) gradually
                if t_iter % decay_alpha_every == 0:
                    alpha = min(alpha * 1.5, alpha_max)

                # evaluate exact objective (hard max)
                val = 2.0 * n * max_conv
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t_iter}] new best {val:.6f}")

                # stagnation shake
                if time.time() - last_improve > 30.0:
                    # add small random perturbation to logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t_iter}] shake (stagnation)")

            # local greedy refinement after each restart
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 2000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    # pick two indices i (source) and j (dest)
                    i, j = np.random.choice(n, 2, replace=False)
                    if a_local[i] <= 1e-12:
                        continue
                    delta = min(a_local[i] * 0.1, a_local[i] * (np.random.rand() * 0.5 + 0.5))
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} local_refine] improved to {val_new:.6f}")

    if best_seq is None:
        # fallback uniform vector length 1024
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0/n_fallback)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a candidate step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)**2)

where * denotes the linear convolution.  The method combines several
ideas:

*   A smooth approximation of the max‑function (softmax over the
    convolution values) gives a dense sub‑gradient which helps the
    optimiser escape flat regions.
*   Adam (exponential‑moving‑average) updates on the logits of a soft‑max
    parametrisation keep the iterates on the probability simplex.
*   A schedule slowly sharpens the soft‑max (parameter ``alpha``) so the
    optimisation first flattens the whole convolution and later focuses
    on the true maximum.
*   Multiple initialisations (uniform, random Dirichlet, raised‑cosine
    window and warm‑starts from previous best) and several restarts
    explore the search space.
*   Light‑weight local refinement swaps a small amount of mass from one
    coordinate to another whenever it improves the exact objective.
*   The whole routine is time‑bounded (default 1000 s); it returns the
    best sequence found so far.

The implementation uses only NumPy and the standard library, works
without any file or network access and respects the required signature
``propose_candidate``.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using an FFT.
    """
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def raised_cosine_window(n: int) -> np.ndarray:
    """A smooth non‑negative window that sums to one."""
    t = np.arange(n)
    w = 0.5 * (1 - np.cos(2 * np.pi * t / (n - 1)))   # Hann window
    w /= w.sum()
    return w


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Run a time‑bounded optimisation and return the best non‑negative
    sequence (as a plain Python list) found within the time budget.
    """
    np.random.seed(seed)
    # keep a small safety margin so the interpreter can exit cleanly
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # best probability vector (sums to 1)
    best_val = float("inf")  # corresponding value of 2·n·max(conv)

    # problem sizes – larger n gives a lower theoretical bound,
    # but optimisation becomes slower.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of restarts – fewer for the biggest instances
        restarts = 5 if n <= 4096 else 3 if n <= 8192 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -----------------------------------------------------------------
            # 1) Initialise logits (unconstrained parameters for the soft‑max)
            # -----------------------------------------------------------------
            init_mode = np.random.choice(["warm", "random", "cosine", "uniform"])
            if init_mode == "warm" and best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
                a0 /= a0.sum()
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] warm start")
            elif init_mode == "cosine":
                a0 = raised_cosine_window(n)
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] cosine window start")
            elif 'height_sequence_1' in globals():
                # try to reuse the globally supplied sequence
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 1e-12)
                cand /= cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={r}] random Dirichlet start")

            # -----------------------------------------------------------------
            # 2) Adam state & hyper‑parameters
            # -----------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2                # initial learning rate for logits
            alpha = 1.0             # smoothness of the soft‑max over conv
            alpha_max = 200.0       # final (hard) max
            alpha_inc_every = 3000  # increase alpha every this many iters

            last_improve = time.time()
            max_iters = 60000       # upper bound – real limit is the deadline
            t_iter = 0

            # -----------------------------------------------------------------
            # 3) Main optimisation loop (Adam on the log‑soft‑max)
            # -----------------------------------------------------------------
            while time.time() < deadline and t_iter < max_iters:
                # a) logits → probabilities (soft‑max, stays on the simplex)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)          # stability trick
                a = exp_x / exp_x.sum()            # a_i ≥ 0, Σ a_i = 1

                # b) exact self‑convolution (linear)
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_conv = conv.max()

                # c) smooth approximation of the max‑operator
                #    w_k ∝ exp(alpha * (conv_k - max_conv))
                #    Guarantees non‑negative weights that sum to one.
                shifted = conv - max_conv
                w = np.exp(alpha * shifted)
                w_sum = w.sum()
                if w_sum == 0.0:
                    # numerically all under‑flow – fall back to hard max
                    w = np.zeros_like(conv)
                    w[conv.argmax()] = 1.0
                else:
                    w /= w_sum

                # d) gradient w.r.t. a (∇_a 2·n·softmax(conv) = 2·n·(w ★ a))
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # e) back‑prop through the soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # f) Adam update on logits
                t_iter += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)
                m_hat = m / (1.0 - beta1 ** t_iter)
                v_hat = v / (1.0 - beta2 ** t_iter)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # g) learning‑rate decay (simple step schedule)
                if t_iter % 2000 == 0:
                    lr *= 0.7

                # h) sharpen the smooth‑max periodically
                if t_iter % alpha_inc_every == 0 and alpha < alpha_max:
                    alpha = min(alpha * 1.5, alpha_max)

                # i) evaluate the *exact* objective (hard max)
                val = 2.0 * n * max_conv            # sum(a) == 1 → denominator omitted
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t_iter:05d}] new best {val:.6f}")

                # j) occasional random shake if we have been idle too long
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t_iter}] shake (stagnation)")

            # -----------------------------------------------------------------
            # 4) Light‑weight local greedy refinement (move tiny packets of mass)
            # -----------------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                local_iters = 3000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    if a_local[i] <= 1e-12:
                        continue
                    # move a random fraction of a_i to a_j
                    delta = min(a_local[i] * 0.1,
                                a_local[i] * (np.random.rand() * 0.5 + 0.5))
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[n={n} local refine] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    # Fallback – should never happen, but keep a valid candidate
    # -------------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
