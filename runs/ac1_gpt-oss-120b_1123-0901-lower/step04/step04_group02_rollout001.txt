def propose_candidate(seed=0, budget_s=1000, **kwargs):

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # define helper functions: fft_len_pow2, conv_fft, grad_convolution, softmax, evaluate_obj

    best_seq = None
    best_score = float('inf')
    best_n = None

    # sizes to explore in increasing order
    sizes = [512, 1024, 2048, 4096, 8192, 16384, 32768]  # up to maybe 32768

    for n in sizes:
        if time.time() > deadline: break

        # Determine number of restarts based on size (more restarts for small n)
        restarts = 6 if n <= 2048 else 3

        # Precompute FFT length for conv
        fft_len = 1 << ((2*n - 1).bit_length())

        for r in range(restarts):
            if time.time() > deadline: break

            # initialization
            if r == 0 and best_seq is not None and best_n == n:
                p = best_seq.copy()
                # ensure p shape n etc.
                init_msg = "warm"
            elif r == 0 and best_seq is not None and best_n < n:
                # upscale previous best
                p = upscale(best_seq, n)
                init_msg = "upscale"
            else:
                # random Dirichlet
                p = np.random.dirichlet(np.ones(n))
                init_msg = "random"

            # Transform to logits
            w = np.log(p + 1e-12)   # avoid log 0

            # Adam parameters
            m = np.zeros_like(w)
            v = np.zeros_like(w)
            beta1=0.9
            beta2=0.999
            eps=1e-8
            lr = 0.5   # maybe bigger for logits
            # temperature for smooth max
            alpha = 0.5
            alpha_factor = 1.15

            # iterate
            iter = 0
            max_iters = 40000  # maybe.
            last_improve = time.time()
            while time.time() < deadline and iter < max_iters:
                iter += 1

                # compute p from logits
                max_w = np.max(w)
                exp_w = np.exp(w - max_w)
                p = exp_w / np.sum(exp_w)

                # conv
                conv = _full_convolution(p, fft_len)

                # smooth max weights
                # compute softmax of (alpha * conv)
                conv_shift = conv - np.max(conv)  # shift for stability
                wgt = np.exp(alpha * conv_shift)
                wgt_sum = np.sum(wgt)
                wgt = wgt / wgt_sum

                # gradient w.r.t p: 2 * cross_correlation(wgt, p)
                grad_p = 2.0 * _cross_correlation(wgt, p, fft_len*2)  # need proper fft length

                # chain rule: d f / d w = diag(p) (grad_p - <p, grad_p>)
                avg = np.dot(p, grad_p)
                grad_w = p * (grad_p - avg)

                # Adam update
                m = beta1 * m + (1 - beta1) * grad_w
                v = beta2 * v + (1 - beta2) * (grad_w ** 2)

                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                w = w - step

                # update temperature
                alpha = min(alpha * alpha_factor, 1e6)

                # compute exact objective for monitoring
                val = 2.0 * n * np.max(conv)

                    best_seq = p.copy()
                    best_n = n
                    last_improve = time.time()
                    print(f"[n={n} start={init_msg} restart={r} iter={iter}] new best: {val:.6f}")

                    if best_score < 1.503:

def _upscale(p_old, new_n):
    old_n = len(p_old)
    if old_n == new_n:
        return p_old.copy()
    # Use linear interpolation
    idx = np.linspace(0, old_n-1, new_n)
    p_new = np.interp(idx, np.arange(old_n), p_old)
    # ensure non-negative and renorm
    if p_new.sum() <= 0:
        p_new = np.full(new_n, 1.0/new_n)
    else:
        p_new = p_new / p_new.sum()
    return p_new

"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (probability vector) that
minimises

    F(p) = 2·n·max(p * p)   with  Σ p_i = 1

where “*” denotes linear convolution.
The algorithm works on a hierarchy of lengths (powers of two) and uses
a log‑softmax (logit) parametrisation of the probabilities.  Gradient
steps are taken by Adam on the logit variables, the gradient of a
smooth‐max surrogate of the max‑convolution is obtained via FFT‑based
cross‑correlation.  A temperature (α) slowly increases so that the
surrogate approaches the true max.  Between gradient phases a
simulated‑annealing‑style local search (random mass swaps) refines the
best solution.  The routine respects a global time budget (default
1000 s) and prints progress.  It returns the best sequence found as a
plain Python list so that ``evaluate_sequence`` can be called on it.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(p: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution p*p (length 2·len(p)-1) using FFT.
    fft_len must be a power of two ≥ 2·len(p)-1.
    """
    fp = np.fft.rfft(p, fft_len)
    conv = np.fft.irfft(fp * fp, fft_len)[: 2 * p.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, p: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Gradient of Σ_l w_l·(p*p)_l w.r.t. p (up to the factor 2).
    Returns a vector of length len(p).
    """
    n = p.shape[0]
    rev_p = p[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_p, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _upscale(p_old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Upscale a probability vector to a larger size by linear interpolation
    and re‑normalise.
    """
    old_n = p_old.shape[0]
    if old_n == new_n:
        return p_old.copy()
    # positions of the new vector in the old index space
    xi = np.linspace(0, old_n - 1, new_n)
    p_new = np.interp(xi, np.arange(old_n), p_old)
    s = p_new.sum()
    if s <= 0:
        p_new = np.full(new_n, 1.0 / new_n)
    else:
        p_new /= s
    return p_new


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative sequence with a low evaluation score.
    The function respects ``budget_s`` seconds and prints progress.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # reserve a safety margin

    best_seq = None            # probability vector (sum == 1)
    best_score = float("inf")  # 2·n·max(conv) since sum=1
    best_n = None

    # ------------------------------------------------------------------
    # Helper to compute the exact objective for a given probability vector.
    def _exact_obj(p: np.ndarray, n: int) -> float:
        conv = _full_convolution(p, _next_pow2(2 * n - 1))
        return 2.0 * n * np.max(conv)

    # ------------------------------------------------------------------
    # Lengths to explore – start small, finish large.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)   # safe for (2n-1)+n convolution

        # Number of restarts – more restarts for smaller dimensions.
        restarts = 6 if n <= 2048 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- Initialise p (probability vector) --------------------
            init_tag = "random"
            if best_seq is not None and best_n == n:
                p = best_seq.copy()
                init_tag = "warm"
            elif best_seq is not None and best_n is not None and best_n < n:
                # Upscale the best solution from a smaller size.
                p = _upscale(best_seq, n)
                init_tag = "upscale"
            elif 'height_sequence_1' in globals():
                # Use the historic construction if it exists.
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 1e-8:
                    p = np.random.dirichlet(np.ones(n))
                else:
                    p = cand / s
                init_tag = "historic"
            else:
                # Pure Dirichlet random start.
                p = np.random.dirichlet(np.ones(n))
                init_tag = "random"

            # Logit parametrisation (w such that p = softmax(w))
            w = np.log(p + 1e-12)

            # Adam state on the logits.
            m = np.zeros_like(w)
            v = np.zeros_like(w)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.4                     # relatively aggressive for logits

            # Smooth‑max temperature (α) – start low, grow.
            alpha = 0.5
            alpha_factor = 1.15

            iter_cnt = 0
            max_iters = 30000            # safe upper bound
            last_improve = time.time()
            patience_seconds = 6.0

            while time.time() < deadline and iter_cnt < max_iters:
                iter_cnt += 1

                # ----- Convert logits to probabilities (softmax) -----
                max_w = np.max(w)
                exp_w = np.exp(w - max_w)
                p = exp_w / exp_w.sum()

                # ----- Convolution p * p -----
                conv = _full_convolution(p, fft_len_conv)

                # ----- Smooth‑max weights (softmax on α·conv) -----
                # shift by max for numerical stability
                shift = conv - np.max(conv)
                wgt = np.exp(alpha * shift)
                wgt_sum = wgt.sum()
                if wgt_sum == 0.0:
                    wgt = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    wgt /= wgt_sum

                # ----- Gradient w.r.t. p (2·cross‑correlation) -----
                grad_p = 2.0 * _cross_correlation(wgt, p, fft_len_corr)

                # ----- Chain rule to obtain gradient w.r.t. logits w -----
                # dF/dw = diag(p)·(grad_p - <p,grad_p>)
                avg = np.dot(p, grad_p)
                grad_w = p * (grad_p - avg)

                # ----- Adam update on logits -----
                m = beta1 * m + (1.0 - beta1) * grad_w
                v = beta2 * v + (1.0 - beta2) * (grad_w * grad_w)
                m_hat = m / (1.0 - beta1 ** iter_cnt)
                v_hat = v / (1.0 - beta2 ** iter_cnt)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                w = w - step

                # ----- Temperature schedule -----
                alpha = min(alpha * alpha_factor, 1e6)

                # ----- Exact objective for monitoring (no scaling needed) -----
                val = 2.0 * n * np.max(conv)

                # ----- Record best -----
                if val < best_score:
                    best_score = val
                    best_seq = p.copy()
                    best_n = n
                    last_improve = time.time()
                    print(f"[n={n} {init_tag} restart={r} iter={iter_cnt:05d}] "
                          f"new best: {val:.6f}")

                    # Early stop if target reached.
                    if best_score < 1.503:
                        print("Target achieved – terminating early.")
                        deadline = time.time()   # force outer loops to quit
                        break

                # ----- Stagnation handling -----
                if time.time() - last_improve > patience_seconds:
                    # Small random perturbation to logits.
                    w += (np.random.rand(n) - 0.5) * 0.02
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={iter_cnt}] shake")

            # End of a restart
        # End of lengths loop

    # ------------------------------------------------------------------
    # Final local refinement (hill‑climbing / simulated annealing)
    # -----------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        p = best_seq.copy()
        n = p.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local refinement (mass swaps) ---")
        # Simple annealing: temperature gradually decays.
        T0 = 0.01
        decay = 0.9999
        T = T0
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # Propose moving a small fraction of mass from i to j.
            delta = 0.005 * p[i] * np.random.rand()
            if delta <= 0:
                continue
            p_prop = p.copy()
            p_prop[i] -= delta
            p_prop[j] += delta
            # Ensure non‑negativity (tiny numerical errors may create tiny negatives)
            p_prop = np.clip(p_prop, 0.0, None)
            p_prop /= p_prop.sum()   # re‑normalise (should be ~1)

            val = _exact_obj(p_prop, n)
            if val < best_score:
                best_score = val
                best_seq = p_prop.copy()
                p = p_prop
                print(f"[refine] improved to {val:.6f}")
                T = T0       # reset temperature on improvement
            else:
                # Metropolis acceptance
                prob = math.exp(-(val - best_score) / max(T, 1e-12))
                if np.random.rand() < prob:
                    p = p_prop
                T *= decay

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_score = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_score:.6f}")

    print(f"\nBest value found: {best_score:.6f} (length = {len(best_seq)})")
    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
