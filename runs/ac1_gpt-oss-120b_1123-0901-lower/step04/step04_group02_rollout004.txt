def exp_gradient_update(a, grad, lr):
    # maintain positivity
    # a >0. Use small epsilon for zeros from rounding
    log_a = np.log(a + 1e-20)
    log_new = log_a - lr*grad
    a_new = np.exp(log_new)
    a_new_sum = a_new.sum()
    if a_new_sum == 0:
        # fallback to uniform
        a_new = np.full_like(a, 1.0 / a.shape[0])
    else:
        a_new /= a_new_sum
    return a_new

"""
Propose a non‑negative step function (sequence) that minimizes

        F(a) = 2·n·max(a*a) / (sum(a)^2)

where “a*a” is the linear convolution of a with itself.
The search is performed on the probability simplex (sum(a)=1).

Algorithm
---------
1. Work on a set of moderate sequence lengths (256 … 2048).  Larger
   lengths give more freedom but the inner linear program becomes
   expensive, so we stop at 2048.
2. For each length we run a few *restarts*.
   * Restart 0: warm‑start from the best sequence of the same length,
     or an up‑sampled version of the globally best sequence.
   * Restart 1: use the historic construction stored in the global
     variable ``height_sequence_1`` (if it exists) resized to the
     current length.
   * Other restarts: random Dirichlet initialisation.
3. **LP‑guided update** – the core iteration (inspired by the paper
   description):
   * Let f be the current sequence (∑f = 1) and M = max(conv(f,f)).
   * Build the convolution matrix A (size (2·n‑1)×n) such that
     (A·g)_k = (f * g)_k.
   * Solve the linear programme

        maximize   Σ_j g_j
        subject to A·g ≤ M·1,   0 ≤ g_j ≤ 1000 .

     (We minimise –Σg_j with ``scipy.optimize.linprog``.)
   * Scale the solution to a probability vector:
        ĝ = g / Σg .
   * Move a small step towards ĝ:
        f ← (1‑t)·f + t·ĝ      (t≈0.4).
   * Repeat the LP‑step a few times – each step provably does not
     increase M.
4. After the LP phase a cheap *local random refinement* (hill‑climbing)
   tweaks a tiny amount of mass between two coordinates.
5. Keep the best sequence over all lengths and restarts.  Abort early
   once the target value 1.503 is reached.
6. If nothing better is found, fall back to a uniform vector.

All operations stay on the simplex, are fully vectorised and respect
the time budget (default 1000 s).  The function returns a plain Python
list of floats, ready for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a probability vector by linear interpolation."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s == 0.0:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up = up / s
    return up

                        deadline = time.time()   # force outer loops to stop
                        break

                # ------------------- build convolution matrix A -------------------
                L = 2 * n - 1
                # A_{k,j} = f_{k-j} if 0≤k-j<n else 0
                A = np.zeros((L, n), dtype=np.float64)
                for j in range(n):
                    A[j:j + n, j] = f      # column j is f shifted by j positions

                # ------------------- solve linear programme --------------------
                # maximise Σ g_j  <=> minimise -Σ g_j
                c = -np.ones(n, dtype=np.float64)

                bounds = [(0.0, 1000.0) for _ in range(n)]
                b_ub = np.full(L, M, dtype=np.float64)   # M·1_{2n‑1}

                # ``highs`` is fast and robust for dense problems of this size
                res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                              method='highs', options={'presolve': True})

                if not res.success:
                    # In the rare case the LP fails (numerical issues) we skip further LP steps.
                    print(f"[n={n} r={r} lp={lp_it}] LP failed: {res.message}")
                    break

                g = res.x
                sum_g = g.sum()
                if sum_g <= 0.0:
                    # Degenerate solution – abort this LP chain.
                    print(f"[n={n} r={r} lp={lp_it}] zero‑sum LP solution")
                    break

                # normalise to a probability vector
                g_hat = g / sum_g

                # convex update towards ĝ
                f = (1.0 - step_size) * f + step_size * g_hat

                # small stochastic perturbation every few LP steps (helps escape plateaus)
                if lp_it % 4 == 0:
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    f = np.maximum(f + noise, 0.0)
                    f /= f.sum()

            # ------------------------------------------------------------------
            # Short random hill‑climbing refinement (very cheap)
            # ------------------------------------------------------------------
            if time.time() < deadline:
                a = f.copy()
                n_iters_refine = 2000
                for _ in range(n_iters_refine):
                    if time.time() >= deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # move a tiny fraction of mass from i to j
                    delta = 0.003 * a[i] * np.random.rand()
                    if delta <= 0.0:
                        continue
                    a_prop = a.copy()
                    a_prop[i] -= delta
                    a_prop[j] += delta
                    if a_prop[i] < 0.0:
                        continue
                    a_prop /= a_prop.sum()
                    convp = _full_convolution(a_prop, fft_len)
                    valp = 2.0 * n * convp.max()
                    if valp < best_val:
                        best_val = valp
                        best_seq = a_prop.copy()
                        a = a_prop
                        print(f"[refine n={n}] improved to {valp:.6f}")
                        if best_val < 1.503:
                            print("Target reached during refinement.")
                            deadline = time.time()
                            break
            # end of one restart
        # end of length loop

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution (guaranteed feasible)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        fft_len_f = _next_pow2(2 * n_fallback - 1)
        best_val = _exact_obj(best_seq, n_fallback, fft_len_f)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Convert to plain list for the external evaluator
    return best_seq.tolist()
