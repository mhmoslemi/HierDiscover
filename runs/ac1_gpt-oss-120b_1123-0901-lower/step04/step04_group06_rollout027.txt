def project_to_simplex(v, z):
    # returns vector w >=0 with sum w = z and minimal Euclidean distance to v.
    # algorithm: sort descending v, compute rho etc.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - z
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def project_to_simplex(v, target_sum):
    # v: array
    # target_sum >0
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    cond = u - cssv / np.arange(1, n+1) > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The search pipeline
-------------------
* **Parameterisation** – the vector a lives on the simplex
  Σ a_i = √(2 n).  We keep this invariant by projecting after each

* **Surrogate objective** – we replace the non‑smooth max by a

* **Gradient** – for a weight vector w the gradient of Lτ is
      ∇_a Lτ = 2·(a ⊛ w_rev) .

  The temperature τ is annealed exponentially, the Adam step‑size

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of two real vectors via FFT (output length len(x)+len(y)-1)."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate harmless negatives caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection of a vector onto the simplex {z ≥ 0, Σz = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Return the Euclidean projection of v onto the simplex with sum s."""
    if s < 0:
        raise ValueError("simplex sum must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it representation (with simplex projection)
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """Run Adam on logits x; returns the best sequence found and its true score."""
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- 1. map logits to the simplex (soft‑max) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numerical stability
        probs = exp_shift / np.sum(exp_shift)  # probability simplex
        a = target_sum * probs                 # a lives on Σ a = target_sum

        # ----- 2. full self‑convolution and current objective -----
        conv = _full_convolution_fft(a, a)       # length = 2n‑1
        max_val = conv.max()                     # = true evaluation because Σ a = √(2 n)
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ----- 3. smooth‑max (soft‑max) weight vector -----
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        # avoid overflow/underflow in exp
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 4. gradient w.r.t. a (2·a⊛w_rev) -----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 2n‑1
        idx = (2 * n - 2) - np.arange(n)                 # pick the part that matches a
        grad_a = 2.0 * conv2[idx]

        # ----- 5. gradient w.r.t. logits (chain rule) -----
        dotp = np.dot(probs, grad_a)
        grad_x = target_sum * probs * (grad_a - dotp)

        # ----- 6. Adam update -----
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny jitter to escape flat plateaus
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 1e-3

        # slowly decay the base learning‑rate
        alpha *= 0.9995

    # ------------------------------------------------------------------
    # 7. Optional hard‑max refinement (sub‑gradient on the true max)
    # ------------------------------------------------------------------
    refinement_deadline = min(run_deadline + 4.0, run_deadline + (run_deadline - time.time()))
    if time.time() < refinement_deadline:
        a = best_seq.copy()
        # keep the same Adam state variables but re‑initialise them for the refinement
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        # use a much smaller learning‑rate for fine‑tuning
        ref_alpha = 0.02
        iter_ref = 0
        while time.time() < refinement_deadline:
            conv = _full_convolution_fft(a, a)
            max_val = conv.max()
            # active set (indices whose value is within 1e‑8 relative tolerance)
            tol = max_val * 1e-8
            active = np.where(conv >= max_val - tol)[0]
            if active.size == 0:
                active = np.array([conv.argmax()], dtype=int)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size

            # gradient of the (hard) max surrogate
            w_rev = w[::-1]
            conv2 = _full_convolution_fft(a, w_rev)
            grad_a = 2.0 * conv2[idx]

            # Adam on the primal variable (no logits this time)
            iter_ref += 1
            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)

            m_hat = m / (1.0 - beta1 ** iter_ref)
            v_hat = v / (1.0 - beta2 ** iter_ref)

            step = ref_alpha * m_hat / (np.sqrt(v_hat) + eps)
            a -= step
            # project back onto the simplex Σ a = target_sum, a ≥ 0
            a = _project_to_simplex(a, target_sum)
            # keep values ≤ 1000 (they are far below this bound in practice)
            np.clip(a, 0.0, 1000.0, out=a)

        # replace the best candidate with the refined one if it improved
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  refined best = {max_val:.6f}")

    # ------------------------------------------------------------------
    # Final evaluation using the supplied evaluator (identical rounding)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.15            # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a globally supplied construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Helper – generate an initial shape given n and a random family
    # --------------------------------------------------------------
    def _initial_shape(n):
        """Return a positive array of length n (not yet scaled)."""
        kind = np.random.choice(["power", "exp", "hann", "gauss"])
        idx = np.arange(n, dtype=float)

        if kind == "power":
            p = np.random.uniform(0.25, 0.85)
            base = (idx + 1) ** (-p)
        elif kind == "exp":
            tau = np.random.uniform(0.3, 1.5) * n
            base = np.exp(-idx / tau)
        elif kind == "hann":
            # Hann window ranging from 0 to 1, shifted up a little to avoid zeros
            base = 0.5 * (1 - np.cos(2 * np.pi * idx / (n - 1)))
            # add a tiny constant to keep strict positivity
            base += 1e-6
        else:  # "gauss"
            mu = np.random.uniform(0.3, 0.7) * n
            sigma = np.random.uniform(0.1, 0.3) * n
            base = np.exp(-0.5 * ((idx - mu) / sigma) ** 2) + 1e-6

        # Optional random jitter
        base += np.random.rand(n) * 1e-5
        # Ensure strictly positive values for the log‑it conversion
        base = np.maximum(base, 1e-12)
        return base

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size – we favour larger n early on.
        if remaining > 400:
            n = np.random.randint(3000, 8000 + 1)
        elif remaining > 200:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(1200, 3500 + 1)

        # ------------------- initial shape ----------------------------
        init = _initial_shape(n)

        # optionally blend with the current global best
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # --------------------------------------------------------------
        # Scale to the required simplex (Σ a = √(2 n))
        # --------------------------------------------------------------
        target_sum = math.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        # Convert to logits for the soft‑max Adam optimiser
        x0 = np.log(init + 1e-12)

        # ------------------------------------------------------------------
        # Allocate a time slice for this run (≈ 15‑20 s, never exceed deadline)
        # ------------------------------------------------------------------
        slice_time = min(20.0, remaining * 0.45)
        run_deadline = min(deadline, time.time() + slice_time)

        # ------------------------------------------------------------------
        # Run the optimiser for the chosen length
        # ------------------------------------------------------------------
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=5e-4,
            alpha=0.07,
        )

        # ------------------------------------------------------------------
        # Keep the globally best sequence
        # ------------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            # early exit if we beat the target comfortably
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to a plain Python list as required by the evaluator.
    return best_seq.tolist()
