"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

where a*a is the linear convolution of a with itself.  The algorithm
performs a multistart, Adam‑based optimisation of a smooth surrogate of
the max‑convolution on increasingly large problem sizes.  For each size
the best solution is kept, then up‑sampled and refined on the next
larger size.  After the gradient phase a simple hill‑climbing local
search finishes the job.  The whole search respects the supplied time
budget (default 1000 s) and returns the best sequence found as a plain
Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using real FFT.
    fft_len must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],  i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k   (the factor 2 is added later).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _interpolate_seq(old: np.ndarray, new_n: int) -> np.ndarray:
    """
    Upsample / downsample an existing sequence to a new length by linear
    interpolation, then re‑normalise to sum = 1.
    """
    old_n = old.shape[0]
    if old_n == new_n:
        return old.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    new = np.interp(x_new, x_old, old)
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / s
    return new


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  Returns the best sequence (plain Python list) found before the
    time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # a little safety margin

    best_seq = None          # numpy array, sum == 1
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # Helper to evaluate the *exact* objective for a probability vector.
    def _exact_obj(a: np.ndarray, n: int) -> float:
        # conv.max() is max(a*a) since sum(a)=1
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Problem sizes – we start small and grow, re‑using the current best.
    lengths = [4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)   # safe bound for correlation

        # Heuristic number of restarts (more for small n, fewer for large)
        restarts = 6 if n <= 8192 else 4 if n <= 16384 else 2

        # iteration budget roughly inversely proportional to problem size
        base_iters = 40000
        max_iters = max(5000, int(base_iters * 4096 / n))

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- initialise a (probability vector) --------------------
            if r == 0 and best_seq is not None and best_n == n:
                a = best_seq.copy()
                print("[restart 0] warm start from global best of same size")
            elif r == 0 and best_seq is not None and best_n is not None and best_n < n:
                a = _interpolate_seq(best_seq, n)
                print("[restart 0] up‑sampled previous best")
            elif 'height_sequence_1' in globals():
                # historic construction, resized to current n
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from height_sequence_1")
            else:
                # Mix of Dirichlet and power‑law starts
                if np.random.rand() < 0.5:
                    a = np.random.dirichlet(np.ones(n))
                    print("[restart] Dirichlet start")
                else:
                    p = np.random.uniform(-2.0, 2.0)   # power‑law exponent
                    idx = np.arange(n) + 1
                    a = np.power(idx, p)
                    a = np.maximum(a, 0.0)
                    a = a / a.sum()
                    print("[restart] power‑law start (p={:.2f})".format(p))

            # Adam state
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.15                     # base learning rate
            lam = 0.5                     # smooth‑max temperature
            lam_factor = 1.1              # geometric increase

            t = 0
            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # ---- convolution and smooth‑max weights ----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # smooth‑max (soft‑max) weights on the convolution vector
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum

                # ---- gradient of the surrogate ----
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ---- Adam update (additive) ----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---- take step, enforce non‑negativity, re‑normalise ----
                a_new = a - step
                a_new = np.maximum(a_new, 0.0)
                s = a_new.sum()
                if s < 1e-12:
                    a_new = np.random.dirichlet(np.ones(n))
                else:
                    a_new /= s
                a = a_new

                # ---- temperature schedule ----
                lam = min(lam * lam_factor, 1e6)

                # ---- exact objective for monitoring (sum(a)==1) ----
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    best_n = n
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t:05d}] new best: {val:.6f}")

                    if best_val < 1.503:
                        print("Target reached – stopping early.")
                        deadline = time.time()  # force outer loops to stop
                        break

                # ---- stagnation handling ----
                if time.time() - last_improve > 6.0:
                    # small random perturbation before projection
                    noise = (np.random.rand(n) - 0.5) * 1e-3
                    a = a + noise
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    if s < 1e-12:
                        a = np.random.dirichlet(np.ones(n))
                    else:
                        a = a / s
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={t}] shake (stagnation)")

            # end of one restart

        # after all restarts for this length
        # (optional) quick sanity: if best_val already very close to 1.0, we could stop
        if best_val < 1.40:   # far beyond any realistic bound – just in case
            break

    # ------------------------------------------------------------------
    # Final simple hill‑climbing refinement (if time permits)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local hill‑climbing refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move a tiny fraction of mass from i to j
            delta = 0.001 * a[i] * np.random.rand()
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            # re‑normalise (should stay on simplex, but protect against rounding)
            a_prop = np.maximum(a_prop, 0.0)
            s = a_prop.sum()
            if s < 1e-12:
                continue
            a_prop /= s
            val = _exact_obj(a_prop, n)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if everything failed return a uniform vector of moderate size
    if best_seq is None:
        fallback_n = 4096
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _exact_obj(best_seq, fallback_n)
        print(f"Fallback uniform sequence of length {fallback_n}, value = {best_val:.6f}")

    # Return plain Python list as required by the external evaluator.
    return best_seq.tolist()
