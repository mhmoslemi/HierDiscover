from scipy.optimize import linprog

def solve_lp_max_sum(a, M):
    n = len(a)
    # Build A_ub
    rows = []
    for k in range(2*n-1):
        row = np.zeros(n)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        # row[i] = a[k-i]
        row[i_min:i_max+1] = a[k - np.arange(i_min,i_max+1)]
        rows.append(row)
    A_ub = np.vstack(rows)
    b_ub = np.full(2*n-1, M)
    bounds = [(0, None)] * n
    # maximize sum(b) -> minimize -sum(b)
    res = linprog(c=-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback return zero
        return np.zeros(n)
    return res.x

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Algorithm
---------
* **Phase 1 – many Adam runs** (same idea as the reference solution).
  The vector is parameterised by a soft‑max so the non‑negativity and the
  L¹‑normalisation Σa = √(2 n) are satisfied automatically.
  A smoothed version of the max‑operator (log‑sum‑exp with a temperature
  τ) is used as a surrogate; Adam updates the logits.  Random restarts
  with different lengths and initial shapes explore the search space.

* **Phase 2 – LP refinement**.
  For a given sequence `a` let `M = max(conv(a, a))`.  The linear program

        maximise   1ᵀ b
        subject to  conv(a, b) ≤ M  (element‑wise)
                    b ≥ 0

  finds a vector `b` that can carry as much mass as possible while still
  respecting the current peak `M`.  After normalising `b` to the required
  sum we take a convex combination `a ← (1‑α) a + α b̂`.  Because
  `conv(a, b̂) ≤ M` the new peak strictly decreases for a suitable step
  size `α`.  Repeating the LP a few times drives the objective down.

Both phases respect the 1000 s total budget, print progress and finally
return the best sequence found (as a plain Python list).  The implementation
uses only NumPy and SciPy (no file or network access) and avoids closures.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical noise can create tiny negative values; clip them away.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimisation of the smoothed‑max surrogate on the log‑it space
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser in the log‑it (soft‑max) parametrisation.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # 1. soft‑max mapping to the simplex (Σa = target_sum)
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # improve numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # a lives on the simplex

        # 2. convolution and current (smooth) max
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # 3. smooth‑max weights (temperature τ)
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4. gradient w.r.t. a  (2·(a ⊛ w_rev))
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # positions that give Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # 5. gradient w.r.t. logits x (chain rule)
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # 6. Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming refinement (Phase 2)
# ----------------------------------------------------------------------
def _build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """Build the dense matrix A (size (2n‑1)×n) s.t. A·b = conv(a,b)."""
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = a
    return A


def _solve_lp_max_sum(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve  maximise  Σ b_i   subject to   conv(a,b) ≤ M,   b ≥ 0.
    Returns the optimal b (may be all zeros if the LP fails).
    """
    n = a.shape[0]
    A = _build_conv_matrix(a)
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n
    # maximise Σb → minimise -Σb
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": 20},
    )
    if not res.success:
        # In rare cases the LP may be infeasible; return a zero vector.
        return np.zeros_like(a)
    b_opt = np.maximum(res.x, 0.0)
    return b_opt


def _refine_sequence(seq: np.ndarray, deadline: float) -> np.ndarray:
    """
    Apply a few LP‑based refinement steps to lower the maximal self‑convolution.
    Returns a possibly improved sequence (still satisfies Σa = √(2n)).
    """
    n = len(seq)
    S = np.sqrt(2.0 * n)
    a = np.clip(seq, 0.0, None)                 # safety
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())

    while time.time() < deadline:
        # current peak
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # LP to maximise total mass under the current peak bound
        b = _solve_lp_max_sum(a, M)

        sum_b = b.sum()
        if sum_b < 1e-12:
            break

        # Normalise b to the required L¹‑norm
        g = b * (S / sum_b)

        improved = False
        # try a few step sizes; the larger ones produce faster progress
        for alpha in (0.5, 0.3, 0.1, 0.05):
            a_new = (1.0 - alpha) * a + alpha * g
            # evaluate – this also clips values, which is fine.
            score_new = evaluate_sequence(a_new.tolist())
            if score_new < best_score - 1e-12:
                best_score = score_new
                best_seq = a_new.copy()
                a = a_new
                improved = True
                print(f"[{time.time():.2f}] Refinement improved to {best_score:.6f}")
                break
        if not improved:
            # No progress – quit refinement.
            break

        if best_score < 1.503:           # target already reached
            break

    return best_seq

        init = np.maximum(init, 1e-12)

        # ---------- log‑it preparation ----------
        target_sum = np.sqrt(2.0 * n)                # Σ a_i = √(2 n)
        init = init / init.sum() * target_sum        # scale to the correct sum
        x0 = np.log(init + 1e-12)                    # logits (avoid log(0))

        # allocate a time slice for this Adam run (≈ 8‑12 s)
        slice_time = min(12.0, remaining * 0.35)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            alpha=0.05,
            beta1=0.9,
            beta2=0.999,
        )

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:   # early exit if we already beat the target
                break

    # --------------------------------------------------------------
    # Phase 2 – LP refinement (if time permits)
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        # give the remaining budget to refinement (but keep a small safety margin)
        refine_deadline = deadline - 0.05
        refined = _refine_sequence(best_seq, refine_deadline)
        refined_score = evaluate_sequence(refined.tolist())
        if refined_score < best_score:
            best_score = refined_score
            best_seq = refined
            print(f"[{time.time():.2f}] AFTER REFINEMENT best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform sequence if nothing was generated (unlikely).
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    return best_seq.tolist()
