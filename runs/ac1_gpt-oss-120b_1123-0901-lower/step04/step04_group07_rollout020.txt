"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Construct a *Golomb ruler* (a set of indices whose pairwise
  differences are all distinct).  With a binary (0/1) vector whose
  1‑entries are exactly the marks of a Golomb ruler the convolution
  ``a*a`` never exceeds 1 – the maximum is attained at the squares of
  the 1‑entries.
* To increase the ℓ₁‑norm (and therefore reduce the objective)
  fill the remaining positions with a small constant ϵ.  The constant is
  chosen as large as possible while keeping every convolution entry ≤
  1:
      – w·ϵ ≤ 1      (a 1‑entry never meets more than w small entries)
      – (n‑w)·ϵ² ≤ 1 (overlap of two small entries)
  Hence ϵ = 0.99·min(1/w , 1/√(n‑w)).
* Try several lengths (the original length and a few larger ones) and
  a few random greedy constructions of Golomb rulers.  Keep the vector
  with the smallest value of ``evaluate_sequence``.
* The whole search finishes well inside the 1000 s budget and returns
  a plain Python list that can be evaluated by the harness.
"""

import math
import time
import numpy as np

# ----------------------------------------------------------------------
# Helper: greedy construction of a Golomb ruler (differences unique).
# ----------------------------------------------------------------------
def greedy_golomb(n: int, rng: np.random.Generator) -> list[int]:
    """
    Return a list of marks (indices) in {0,…,n‑1} such that all
    positive differences are distinct.
    A random permutation of the candidate indices is used to obtain
    diverse solutions.
    """
    # random order of all possible positions
    candidates = rng.permutation(np.arange(n))
    marks: list[int] = []
    used: set[int] = set()                     # already used differences
    for c in candidates:
        # test whether c can be added without creating a duplicate difference
        ok = True
        for m in marks:
            d = abs(int(c) - int(m))
            if d in used:
                ok = False
                break
        if ok:
            # record the new differences
            for m in marks:
                used.add(abs(int(c) - int(m)))
            marks.append(int(c))
    marks.sort()
    return marks


# ----------------------------------------------------------------------
# Build a candidate vector for a given length.
# ----------------------------------------------------------------------
def build_candidate(n: int, marks: list[int]) -> np.ndarray:
    """
    Build a vector of length n.
      * entries at ``marks`` are 1.0
      * all other entries are a small constant ϵ, chosen as large as
        possible while guaranteeing that the maximum convolution value
        stays ≤ 1.
    Returns the vector (numpy array of float64).
    """
    w = len(marks)
    if w == 0:                     # safety – should not happen
        return np.full(n, 0.0)

    # maximal safe epsilon (see docstring)
    eps_bound1 = 1.0 / w
    eps_bound2 = 1.0 / math.sqrt(max(1, n - w))
    eps = 0.99 * min(eps_bound1, eps_bound2)

    a = np.full(n, eps, dtype=np.float64)
    a[marks] = 1.0
    return a


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """Search for a sequence of non‑negative heights that minimises
    ``evaluate_sequence``.  Returns the best sequence as a plain Python
    list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Base length – use the known construction if available, otherwise
    # fall back to the classic size 2100.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        n0 = base.shape[0]
    except Exception:
        base = None
        n0 = 2100

    # ------------------------------------------------------------------
    # Lengths we will explore (original and a couple of larger ones,
    # never larger than 8000).
    length_factors = [1.0, 1.5, 2.0]
    lengths = sorted({int(n0 * f) for f in length_factors})
    lengths = [max(300, min(l, 8000)) for l in lengths]

    best_seq = None
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Try each length.  For each we generate several random Golomb rulers
    # (including the deterministic one) and keep the best result.
    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"[search] length = {n}")

        # deterministic greedy (takes candidates in natural order)
        deterministic_marks = greedy_golomb(n, np.random.default_rng(0))
        candidates_marks = [deterministic_marks]

        # a few random attempts
        for _ in range(4):
            if time.time() >= deadline:
                break
            candidates_marks.append(greedy_golomb(n, rng))

        # also try to interpolate the previous best if it exists and size differs
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            # turn into a binary mask by keeping the largest entries (roughly sqrt(2n) of them)
            k = int(math.sqrt(2 * n))
            idx = np.argpartition(interp, -k)[-k:]
            marks = sorted(int(i) for i in idx)
            candidates_marks.append(marks)

        # evaluate all candidates for this length
        for marks in candidates_marks:
            if time.time() >= deadline:

            a = build_candidate(n, marks)

            score = evaluate_sequence(a.tolist())
            # sanity check – sometimes numerical issues may give inf
            if math.isfinite(score) and score < best_score:
                best_score = score
                best_seq = a.copy()
                print(f"  new global best = {best_score:.6f} (len={n}, marks={len(marks)})")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should never happen).
    if best_seq is None:

        n = n0
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        best_seq = uniform
        best_score = evaluate_sequence(best_seq.tolist())

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return plain Python list as required
    return best_seq.tolist()
