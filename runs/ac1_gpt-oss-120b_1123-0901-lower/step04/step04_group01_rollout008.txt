"""
Propose a step‑function (non‑negative vector) that minimises
    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm is a blend of:
* Sub‑gradient descent on the simplex with Adam (soft‑max re‑parameterisation).
* Heavy restarts, jitter on stagnation and long runs for the most promising
  length (1024).
* A cheap simulated‑annealing refinement after the gradient phase.

The search is time‑bounded; it stops early if the 1000 s budget is exhausted.
The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np


def _softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that yields a low value of
    evaluate_sequence().  Returns the best sequence (list of floats) found
    before the time budget expires.
    """
    np.random.seed(seed)
    # a small safety margin so the interpreter can finish cleanly
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # will hold a NumPy array
    best_val = float("inf")

    # ------------------------------------------------------------
    # 1. Problem sizes.  We spend most time on n=1024 – the size that
    #    gave the previous best result – but also explore a few neighbours.
    # ------------------------------------------------------------
    lengths = [1024, 2048, 4096, 512, 256]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes – cached for the inner loops
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # Heuristic number of restarts per length
        if n <= 1024:
            restarts = 12
            max_iters = 200_000
        elif n <= 2048:
            restarts = 8
            max_iters = 120_000
        else:
            restarts = 5
            max_iters = 80_000

        for r in range(restarts):
            if time.time() > deadline:
                break

            # --------------------------------------------------------
            # 2. Initialise a probability vector (sum = 1)
            # --------------------------------------------------------
            init_a = None
            # warm start from global best of the same length, if any
            if best_seq is not None and len(best_seq) == n:
                init_a = best_seq.copy()
                init_a /= init_a.sum()
                print(f"[n={n} restart={r}] warm start from previous best")
            else:
                # try to reuse the previously published height_sequence_1
                cand = globals().get("height_sequence_1")
                if cand is not None:
                    cand = np.array(cand, dtype=np.float64)
                    if cand.shape[0] >= n:
                        cand = cand[:n]
                    else:
                        cand = np.pad(cand, (0, n - cand.shape[0]))
                    cand = np.maximum(cand, 0.0)
                    if cand.sum() >= 0.01:
                        init_a = cand / cand.sum()
                        print(f"[n={n} restart={r}] start from height_sequence_1")
            # fallback to a random Dirichlet vector
            if init_a is None:
                init_a = np.random.dirichlet(np.ones(n))

            # logits for Adam (soft‑max parametrisation)
            x = np.log(init_a + 1e-15)

            # --------------------------------------------------------
            # 3. Adam optimiser on the logits
            # --------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.5                     # start learning rate
            decay_every = 1500           # learning‑rate decay interval
            last_improve = time.time()

            t = 0
            while time.time() < deadline and t < max_iters:
                # (a) soft‑max → probability vector
                a = _softmax(x)                     # a_i ≥ 0, Σ a_i = 1

                # (b) linear self‑convolution
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c                # exact objective (Σa=1)

                # (c) keep the best seen so far

                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()

                # (e) back‑propagation through soft‑max
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)

                if t % decay_every == 0:

                if time.time() - last_improve > 15.0:
                    x += (np.random.rand(n) - 0.5) * 5e-3

            # --------------------------------------------------------
            # 4. Simulated‑annealing style local refinement (optional)
            # --------------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_cur = best_seq.copy()
                val_cur = best_val
                temp = 0.01 * val_cur            # initial temperature
                cooling = 0.9995
                sa_steps = 30_000 if n <= 1024 else 15_000

                for step in range(sa_steps):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # magnitude of the move adapts to the current temperature
                    eps_move = min(a_cur[i],
                                   max(1e-7, temp) * np.random.rand())
                    if eps_move <= 0:
                        continue
                    a_new = a_cur.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move

                    # evaluate new candidate
                    conv_new = _full_convolution(a_new, fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    delta = val_new - val_cur

                    if delta < 0 or np.random.rand() < np.exp(-delta / max(temp, 1e-12)):
                        a_cur = a_new
                        val_cur = val_new
                        if val_cur < best_val - 1e-12:
                            best_val = val_cur
                            best_seq = a_cur.copy()
                            print(f"[SA n={n} step={step}] new best {val_cur:.6f}")

                    temp *= cooling

    # -----------------------------------------------------------------
    # 5. Fallback – very unlikely, but guarantees a valid return value.
    # -----------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = evaluate_sequence(best_seq.tolist())
        print(f"Fallback uniform vector used (value {best_val:.6f})")

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
