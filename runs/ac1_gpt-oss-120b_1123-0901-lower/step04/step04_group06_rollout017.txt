def max_conv(a):
    n = a.shape[0]

    fa = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(fa * fa, n=size)[:2*n-1]
    return conv.max()

def _project_simplex(v, s):
    # ensures nonnegative and sum == s
    # algorithm by Duchi et al.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The value 2·n/(sum(a)²) equals 1 when we enforce  Σa_i = √(2 n),
hence the problem is equivalent to **minimising the peak of the
self‑convolution** of a probability vector.

Algorithm
---------
1. **Parameterisation** – a is represented by a soft‑max of logits x,
   guaranteeing a_i ≥ 0 and Σa_i = √(2 n) without any projection.

2. **Smooth surrogate** – instead of the non‑smooth max we minimise a
   temperature‑controlled log‑sum‑exp of the convolution,
   i.e. `τ·log Σ exp(conv/τ)`.  The gradient of this surrogate is a
   weighted convolution that can be evaluated with a single FFT.

3. **Adam optimiser** – we run Adam on the logits.  The temperature τ
   decays exponentially from a moderate value (≈ 0.1·max) down to a
   tiny fraction (≈ 0.001·max), so the optimisation first explores then
   focuses on the true max.

4. **Post‑optimisation refinement** – after the Adam run we perform a
   few projected‑gradient steps on the *exact* max (τ≈0).  The gradient
   of the peaked convolution entry is known analytically; after each
   step we project back onto the simplex.

5. **Multi‑restart search** – the whole procedure is repeated with
   random lengths (≈ 800 – 3000) and random initial shapes (power‑law,
   exponential taper, Dirichlet‑like).  The best sequence found within
   the time budget (1000 s) is returned.

All heavy work uses NumPy + FFT; a single convolution for n≈2500 costs
well below a millisecond, allowing thousands of optimisation steps in
the allotted time.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative round‑off
    np.maximum(conv, 0.0, out=conv)
    return conv


def _max_convolution(a: np.ndarray) -> float:
    """Return max(conv(a, a)) – the quantity we actually minimise."""
    n = a.shape[0]
    conv = _full_convolution_fft(a, a)
    return float(conv.max())


def _project_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """ Euclidean projection of v onto the simplex {x≥0, Σx = s}. """
    # based on Duchi et al. (2008)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero – return uniform vector
        return np.full_like(v, s / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Adam optimisation on the soft‑max (log‑it) representation
# ----------------------------------------------------------------------


def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 0.001,
    alpha: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the log‑sum‑exp surrogate.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- (1) soft‑max mapping -------------------------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numerical stability
        p = exp_shift / np.sum(exp_shift)      # probability vector, Σp = 1
        a = target_sum * p                     # sequence on the required simplex

        # ----- (2) convolution & current max ----------------------------------------
        conv = _full_convolution_fft(a, a)     # length 2n‑1
        max_val = float(conv.max())
        # update global best (the true objective) -----------------------------------
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ----- (3) smooth‑max weights ------------------------------------------------
        # exponential decay of the temperature
        tau = max(
            init_tau_factor * max_val * (0.95 ** iteration),
            min_tau_factor * max_val,
        )
        # log‑sum‑exp weights (numerically stable)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- (4) gradient w.r.t. a -------------------------------------------------
        # grad_a = ∂/∂a Σ w_k conv_k = 2 * (a ⊛ w_rev)
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # positions producing Σ_j a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # ----- (5) chain rule to logits ---------------------------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- (6) Adam update -------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random kick to escape plateaus
        if iteration % 400 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate
        alpha *= 0.9996

    # ----------------------------------------------------------------------
    # Final refinement on the true max (τ → 0) using projected gradient
    # ----------------------------------------------------------------------
    remaining = run_deadline - time.time()
    if remaining > 0.5:
        best_seq, _ = _refine_max_gradient(
            best_seq,
            target_sum,
            time.time() + remaining * 0.6,   # spend most of the spare time here
        )

    # ----------------------------------------------------------------------
    # Final evaluation through the provided evaluator (ensures clipping)
    # ----------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


def _refine_max_gradient(
    a: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_step: float = None,
    min_step_factor: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Perform a few projected‑gradient steps on the *exact* max entry
    (τ essentially 0).  This often squeezes a few percent out of the
    Adam result.
    Returns the refined sequence and its max‑convolution value.
    """
    n = a.shape[0]
    cur_a = a.copy()
    cur_score = _max_convolution(cur_a)

    # initial step size – a few percent of the total mass
    step = init_step if init_step is not None else target_sum * 0.02

    while time.time() < run_deadline:
        # evaluate full convolution to locate the active index
        conv = _full_convolution_fft(cur_a, cur_a)
        kmax = int(np.argmax(conv))

        # gradient of conv[kmax] w.r.t a_i is 2 * a_{kmax-i}
        lo = max(0, kmax - (n - 1))
        hi = min(n - 1, kmax)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad = np.zeros_like(cur_a)
        grad[idx] = 2.0 * cur_a[kmax - idx]

        # take a step opposite to the gradient and project back onto the simplex
        cand = cur_a - step * grad
        cand = _project_simplex(cand, target_sum)

        cand_score = _max_convolution(cand)
        if cand_score < cur_score:
            cur_a, cur_score = cand, cand_score
            step *= 0.9                    # keep moving, but slowly shrink
        else:
            step *= 0.5                    # back‑off when we overshoot

        if step < target_sum * min_step_factor:
            break

    return cur_a, cur_score
