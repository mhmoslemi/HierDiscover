def propose_candidate(seed=42, budget_s=1000):

    np.random.seed(seed)

    start = time.time()
    deadline = start + budget_s - 0.5

    best_val = np.inf
    best_seq = None

    lengths = [2000,2500,3000,3500,4000] # maybe 5 values

    restart = 0
    while time.time() < deadline:
        n = lengths[restart % len(lengths)]

        # random init shape varied:
        init_type = restart % 9
        if init_type == 0: a0 = np.random.dirichlet(np.ones(n))
        elif init_type == 1: a0 = np.random.exponential(scale=1.0,size=n); a0/=a0.sum()
        elif init_type == 2: r = np.random.uniform(0.2,0.95); a0 = r**np.arange(n); a0/=a0.sum()
        elif init_type == 3: # half-sine
            i = np.arange(1,n+1)
            a0 = np.sin(np.pi * i / (n+1))
            a0 = np.maximum(a0,0); a0/=a0.sum()
        elif init_type == 4: # cosine bump
            x = np.linspace(0, np.pi, n)
            a0 = np.cos(x)**2
            a0 = np.maximum(a0,0); a0/=a0.sum()
        elif init_type == 5: # quadratic - high in middle
            i = np.arange(n)
            a0 = -(i - n/2)**2 + (n/2)**2
            a0 = np.maximum(a0,0); a0/=a0.sum()
        elif init_type == 6: # decreasing polynomial
            i = np.arange(n)
            a0 = (n - i)**2
            a0 = np.maximum(a0,0); a0/=a0.sum()
        elif init_type == 7: # random uniform within [0,1] and normalize
            a0 = np.random.rand(n); a0/=a0.sum()
        else: # previous best if same length
            if best_seq is not None and len(best_seq)==n and np.random.rand()<0.2:
                a0 = np.array(best_seq)
            else:
                a0 = np.random.dirichlet(np.ones(n))

        # Logits param
        u = np.log(a0 + 1e-30)

        # Adam variables
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # hyperparameters
        base_lr = 0.03
        lam = 0.5
        lam_factor = 1.02
        max_iter = 15000
        it = 0
        last_improve = time.time()

        # Determine fft size for this n
        conv_fft_size = 1 << int(np.ceil(np.log2(2*n-1)))
        size_corr = 1 << int(np.ceil(np.log2(3*n)))

        while time.time() < deadline and it < max_iter:
            it += 1

            # compute probabilities a
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a sum: a/=a.sum()

            # compute conv and gradient
            if lam < 8.0:
                conv = _full_convolution(a, size=conv_fft_size)
                w = _softmax_weights(conv, lam)
                grad_a = _softmax_gradient(a, w, size_corr=size_corr)
            else:
                # use hard max subgradient
                conv = _full_convolution(a, size=conv_fft_size)
                # compute hard max weight
                max_val = conv.max()
                # identify indices within tolerance
                idx = np.flatnonzero(conv >= max_val - 1e-15)  # all max indices
                w = np.zeros_like(conv)
                w[idx] = 1.0 / len(idx)
                grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # map to log-space gradient
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9*m + 0.1*grad_u
            v = 0.999*v + 0.001* (grad_u**2)
            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart} iter {it}] new best {val:.6f}")

            if time.time() - last_improve > 3.0:

"""
Hybrid optimizer for the self‑convolution bound.

The search proceeds in three phases:

1.  Warm‑up – many short Adam runs on a smooth‑max surrogate of
    F(a)=2·n·max(a∗a)/(∑a)².  The surrogate is a soft‑max with a
    temperature λ slowly increased.  Several random families of
    initial vectors (Dirichlet, exponential, half‑sine, polynomial,
    etc.) are tried for a few thousand iterations each.

2.  Hard‑max refinement – after λ becomes large we switch to the exact
    sub‑gradient of the max‑convolution (a one‑hot weight vector).  This
    sharpens the descent and removes the smoothing bias.

3.  Local stochastic hill‑climb (simulated‑annealing style) that
    perturbs the current best sequence by moving a tiny amount of mass
    between two coordinates and accepts worsening moves with a
    temperature schedule.  The final few Adam steps (hard‑max) polish
    the result.

The algorithm respects a user supplied time budget (default 1000 s) and
returns the best sequence (plain Python list) found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·(convₖ‑max(conv)))/Σⱼexp(…)  (numerically stable)."""
    m = conv.max()
    # (conv‑m) ≤ 0, the exponent is never positive → no overflow.
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:               # extreme under‑flow – fall back to uniform weights
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a∗a)ₖ w.r.t. a.
    Closed form: ∇ = 2·(w ★ a)   (cross‑correlation).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # the slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 12345, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    best_val = float("inf")
    best_seq = None
    restart = 0

    # a modest set of lengths – longer vectors give more freedom but cost a bit more
    lengths = [2000, 2500, 3000, 3500, 4000]

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # pick a length (cycle through the list)
        n = lengths[restart % len(lengths)]

        # pre‑compute FFT sizes that stay constant through this inner run
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # generous bound

        # ---------- diverse random initialisations ----------
        mode = restart % 9
        if mode == 0:                               # Dirichlet (almost uniform)
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:                             # Exponential decay
            r = np.random.uniform(0.2, 0.9)
            a0 = r ** np.arange(n, dtype=np.float64)
            a0 /= a0.sum()
        elif mode == 2:                             # Half‑sine
            i = np.arange(1, n + 1, dtype=np.float64)
            a0 = np.sin(np.pi * i / (n + 1))
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 3:                             # Sine^p
            p = np.random.uniform(0.5, 2.0)
            a0 = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 4:                             # Quadratic hill
            i = np.arange(n, dtype=np.float64)
            a0 = -(i - n / 2) ** 2 + (n / 2) ** 2
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 5:                             # Decreasing polynomial
            i = np.arange(n, dtype=np.float64)
            a0 = (n - i) ** 2
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 6:                             # Random uniform and normalise
            a0 = np.random.rand(n)
            a0 /= a0.sum()
        elif mode == 7:                             # Bimodal half‑sine glued together
            half = n // 2
            left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
            right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
            a0 = np.concatenate([left, right])
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        else:                                        # Seed from the current best (if compatible)
            if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.2:
                a0 = np.array(best_seq, copy=True)
                a0 /= a0.sum()
            else:
                a0 = np.random.dirichlet(np.ones(n))

        # ------------------------------------------------------------------
        # soft‑max (logits) parametrisation – guarantees a_i ≥ 0, Σa_i = 1
        u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # continuation parameters for the smooth‑max surrogate
        lam = 0.5            # start fairly smooth
        lam_factor = 1.02   # a bit faster growth than the original
        base_lr = 0.035

        it = 0
        max_iter = 18000
        last_improve = time.time()

        # --------------------------------------------------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- probabilities from logits (soft‑max) -----
            u_shift = u - np.max(u)          # numerical stability
            a = np.exp(u_shift)
            a /= a.sum()

            # ----- convolution (FFT) -----
            conv = _full_convolution(a, size=conv_fft_size)

            # ----- gradient depending on the current λ -----
            if lam < 8.0:                     # still in the smooth regime
                w = _softmax_weights(conv, lam)
                grad_a = _softmax_gradient(a, w, size_corr=size_corr)
            else:                             # λ large → approximate true max
                max_val = conv.max()
                idx = np.flatnonzero(conv >= max_val - 1e-15)
                w = np.zeros_like(conv)
                w[idx] = 1.0 / len(idx)      # uniform weight over all maximal indices
                grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # ----- map to log‑space gradient (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9998 ** it)    # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- increase smoothness factor -----
            lam *= lam_factor

            # ----- exact evaluation every few steps -----
            if it % 25 == 0:
                val = evaluate_sequence(a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- shake if stalled -----
            if time.time() - last_improve > 3.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # end of inner loop for this restart
        restart += 1
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣  Hard‑max fine‑tuning from the best candidate found so far.
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))
        u = np.log(best_seq + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 12.0          # stay in the hard‑max regime
        base_lr = 0.018
        it = 0
        max_iter = 25000
        last_improve = time.time()

        while time.time() < deadline and it < max_iter:
            it += 1

            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, size=conv_fft_size)
            max_val = conv.max()
            idx = np.flatnonzero(conv >= max_val - 1e-15)
            w = np.zeros_like(conv)
            w[idx] = 1.0 / len(idx)
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            if it % 20 == 0:
                val = evaluate_sequence(a.tolist())
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")

            # occasional shake if nothing improves for a while
            if time.time() - last_improve > 2.5:
                u += (np.random.rand(n) - 0.5) * 3e-4
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

    # ------------------------------------------------------------------
    # 3️⃣  Simple stochastic local search (simulated annealing) on the best vector.
    # ------------------------------------------------------------------
    if best_seq is not None:
        a = np.array(best_seq, copy=True)
        cur_val = evaluate_sequence(a.tolist())

        # temperature schedule
        T_start = 0.02
        T_end = 5e-6
        total_steps = 20000
        start_time = time.time()

        for step in range(total_steps):
            if time.time() > deadline:
                break
            # geometric cooling
            T = T_start * (T_end / T_start) ** (step / total_steps)

            i = np.random.randint(n)
            j = np.random.randint(n)
            if i == j:
                continue

            # propose moving a small random fraction of a[i] to a[j]
            delta = np.random.exponential(scale=0.005) * a[i]  # typical size ~0.5 % of a[i]
            if delta > a[i]:
                delta = a[i] * np.random.rand()

            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta

            val_new = evaluate_sequence(a_new.tolist())
            accept = False
            if val_new < cur_val:
                accept = True
            else:
                # metropolis acceptance
                if np.random.rand() < np.exp(-(val_new - cur_val) / T):
                    accept = True

            if accept:
                a = a_new
                cur_val = val_new
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    print(f"[Anneal step {step:05d}] new best = {cur_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
