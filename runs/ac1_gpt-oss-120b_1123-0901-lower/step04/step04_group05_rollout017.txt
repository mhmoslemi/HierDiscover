   a = np.exp(u_shift)
   a_sum = a.sum()

def adam_optimize(a, n, conv_fft_size, size_corr, max_iter, lam_start, lam_factor, lr):
    u = np.log(a + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    lam = lam_start
    best_val = 2*n * _full_convolution(a, conv_fft_size).max()
    best_seq = a.copy()
    last_improve = time.time()
    for it in range(max_iter):
        # compute a from u
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)
        # map to log-space
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam update
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u**2)
        it_f = float(it+1)
        m_hat = m / (1.0 - 0.9**it_f)
        v_hat = v / (1.0 - 0.999**it_f)
        lr_eff = lr * (0.9999**it)
        u -= lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)
        lam *= lam_factor
        # evaluate occasionally
        if it % 20 == 0:
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
        # optional shake
        if time.time() - last_improve > 1.5:
            u += (np.random.rand(n) - 0.5) * 1e-4
            m.fill(0); v.fill(0)
            last_improve = time.time()
        # check deadline externally.
    return best_seq, best_val

The program searches for a non‑negative step function (a vector `a`) that minimises  

\[
F(a)=\frac{2\,n\;\max_k (a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\]

where `*` denotes the linear convolution.  
Because the objective is scale‑invariant we can force `∑a_i = 1`.  
The search consists of three stages

1. **Diverse random initialisations** – quick evaluation of many candidates.
2. **Deterministic optimisation** – a smooth approximation of the `max`
   (log‑sum‑exp) is minimised with **L‑BFGS‑B** on the log‑soft‑max
   parametrisation `u = log a`.  The gradient is obtained analytically via
   FFT‑based convolution, so each iteration is cheap.
3. **Fine‑tuning** – the best sequence from stage 2 is refined with an
   Adam optimiser that slowly sharpens the smoothing parameter, exactly as
   in the reference code but with a tighter schedule.

All helpers are defined at the top level (no closures) and the routine
stops automatically when the allotted time (default 1000 s) is exhausted.
The returned object is a plain Python list, ready for `evaluate_sequence`.

```python
"""Hybrid optimisation for the self‑convolution bound.

The algorithm combines:
  * random Dirichlet / sine / exponential seeds,
  * L‑BFGS‑B on a smooth log‑sum‑exp surrogate (log‑soft‑max parametrisation),
  * a final Adam fine‑tune with increasing smoothness.
It respects a user‑supplied time budget and returns the best non‑negative
sequence (plain list of floats) found.
"""

import time
import numpy as np
import scipy.optimize as opt


# ----------------------------------------------------------------------
# FFT based convolution utilities
def full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max (log‑sum‑exp) weights for a given smoothing λ."""
    m = conv.max()
    shifted = conv - m               # avoid overflow
    w = np.exp(lam * shifted)
    s = w.sum()
    if s == 0.0:                     # safety – should never happen
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w


def softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the probability vector a.
    Closed form: ∇ = 2·(w ★ a)  (correlation, not convolution).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Smooth objective (log‑sum‑exp) and its gradient in log‑soft‑max space
def smooth_obj_and_grad(
    u: np.ndarray,
    n: int,
    conv_fft_size: int,
    size_corr: int,
    lam: float,
):
    """
    Returns (f, g) where
      f = 2·n·smooth_max(conv)   (smooth_max = log‑sum‑exp / λ)
      g = ∂f/∂u  (gradient w.r.t. logits u)
    The implementation uses FFTs for O(n log n) work.
    """
    # a = soft‑max(u)  (guarantees a ≥ 0, Σa = 1)
    u_shift = u - np.max(u)
    a = np.exp(u_shift)
    a = a / a.sum()

    # convolution a*a
    conv = full_convolution(a, size=conv_fft_size)

    # smooth max via log‑sum‑exp
    m = conv.max()
    shifted = conv - m
    exp_term = np.exp(lam * shifted)
    sum_exp = exp_term.sum()
    smooth_max = m + np.log(sum_exp) / lam
    f = 2.0 * n * smooth_max

    # soft‑max weights for gradient
    w = exp_term / sum_exp

    # gradient w.r.t. a
    grad_a = softmax_gradient(a, w, size_corr=size_corr)

    # full gradient w.r.t. a (the factor 2·n from f)
    grad_a_f = 2.0 * n * grad_a

    # map to logits space (Jacobian of the soft‑max)
    dot = np.dot(a, grad_a_f)
    grad_u = a * (grad_a_f - dot)

    return f, grad_u


# ----------------------------------------------------------------------
# Random initialisation families
def generate_sequence(n: int, mode: int) -> np.ndarray:
    """Return a non‑negative vector of length n (Σ = 1) for a given mode."""
    if mode == 0:                         # Dirichlet
        a = np.random.dirichlet(np.ones(n))
    elif mode == 1:                       # exponential decay
        r = np.random.uniform(0.2, 0.9)
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
    elif mode == 2:                       # half‑sine
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 3:                       # sineⁿ
        p = np.random.uniform(0.5, 2.0)
        a = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 4:                       # perturbed exponential
        r = np.random.uniform(0.3, 0.85)
        a = r ** np.arange(n, dtype=np.float64)
        a += 0.01 * np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 5:                       # inverted parabola
        x = np.linspace(-1.0, 1.0, n)
        a = 1.0 - x ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:                                 # bimodal half‑sine
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam fine‑tuning (hard max, increasing λ)
def adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_size: int,
    size_corr: int,
    deadline: float,
    max_iter: int = 8000,
    lam_start: float = 5.0,
    lam_factor: float = 1.001,
    lr: float = 0.02,
):
    """Adam optimisation of the exact max‑convolution objective."""
    u = np.log(a0 + 1e-30)
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    lam = lam_start

    best_val = float("inf")
    best_a = a0.copy()
    last_improve = time.time()

    for it in range(max_iter):
        if time.time() > deadline:
            break

        # probabilities
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()

        # convolution & smooth weights (still used for the gradient)
        conv = full_convolution(a, size=conv_fft_size)
        w = softmax_weights(conv, lam)
        grad_a = softmax_gradient(a, w, size_corr=size_corr)

        # map to log‑space (soft‑max Jacobian)
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # Adam moments
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)

        it_f = float(it + 1)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)

        lr_eff = lr * (0.9999 ** it)
        u -= lr_eff * m_hat / (np.sqrt(v_hat) + 1e-8)

        lam *= lam_factor

        # exact evaluation every few steps
        if it % 20 == 0:
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_a = a.copy()
                last_improve = time.time()

        # shake if stuck for > 1 s
        if time.time() - last_improve > 1.0:
            u += (np.random.rand(n) - 0.5) * 1e-4
            m.fill(0.0)
            v.fill(0.0)
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
```
