"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

under the constraints a_i ≥ 0.  The implementation follows three
co‑operating phases that respect a total time budget (default 1000 s):

1. **Adam on a smooth approximation of the max** – the exact max of the
   self‑convolution is replaced by a soft‑max with a temperature β
   that is increased during the run.  This yields a smooth gradient
   (2·a∗w) that is used in a mirror‑descent (soft‑max) Adam optimiser
   on the simplex.

2. **Differential Evolution (DE) global search** – a larger population
   (default 30) explores the simplex using the classic “best/1/bin”
   mutation strategy.  The DE loop is stopped early when no improvement
   has been seen for a few seconds.

3. **(1+1) Evolution Strategy (ES) fine‑tuning** – a simple Gaussian
   perturbation of the current best solution with 1/5‑rule step‑size
   control.

After the three phases a cheap local hill‑climber swaps tiny amounts
of mass between two coordinates; it is only run if time is still left.
The best sequence found (across all tried lengths) is returned as a
plain Python list.

All heavy operations (convolutions) are performed with FFTs for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (all pure top‑level functions)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps logits to a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def fft_convolve(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Linear convolution via FFT (real input, real output)."""
    n = x.shape[0] + y.shape[0] - 1
    L = 1 << int(np.ceil(np.log2(n)))
    fx = np.fft.rfft(x, L)
    fy = np.fft.rfft(y, L)
    conv = np.fft.irfft(fx * fy, L)[:n]
    return conv.real


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)−1)."""
    return fft_convolve(a, a)


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence (sum(a) is assumed > 0)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# 1️⃣  Adam optimiser on a smooth approximation of max(a*a)
# ----------------------------------------------------------------------
def run_adam_smooth_max(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    beta_start: float = 5.0,
    beta_end: float = 80.0,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the simplex minimizing 2·n·softmax_β(conv(a,a)).
    Returns the best (a, actual objective) encountered before *deadline*.
    """
    # initialise logits from a strictly positive probability vector
    x = np.log(np.maximum(init_a, 1e-15))

    m = np.zeros(n)
    v = np.zeros(n)
    am_beta1, am_beta2, am_eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥0, Σa = 1
        conv = full_convolution(a)          # length 2n‑1
        max_c = conv.max()                  # exact max (for bookkeeping)

        # ---- smooth “max” via soft‑max with temperature β ----------
        beta = beta_start + (beta_end - beta_start) * min(1.0, t / max_iters)
        z = beta * conv
        z_max = z.max()
        w = np.exp(z - z_max)               # stability shift
        w_sum = w.sum()
        w = w / w_sum                       # weights sum to 1

        # smooth objective (≈ 2·n·max(conv))
        smooth_obj = 2.0 * n * np.dot(w, conv)

        # actual objective (what evaluate_sequence will see)
        actual_obj = 2.0 * n * max_c

        if actual_obj < best_val:
            best_val = actual_obj
            best_a = a.copy()
            last_improve = time.time()
            print(f"[AdamSmooth n={n} iter={t:05d}] new best: {best_val:.6f}")

        # ---- gradient of the smooth objective w.r.t. a --------------
        # ∂/∂a_i  ( Σ_k w_k (a*a)_k ) = 2 Σ_k w_k a_{k‑i}  = 2·(a * w)[i]
        grad_a = 2.0 * fft_convolve(a, w)[:n]

        # back‑propagation through the softmax (mirror descent on simplex)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = am_beta1 * m + (1.0 - am_beta1) * grad_x
        v = am_beta2 * v + (1.0 - am_beta2) * (grad_x ** 2)
        m_hat = m / (1.0 - am_beta1 ** t)
        v_hat = v / (1.0 - am_beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + am_eps)

        # keep logits bounded to avoid numerical blow‑up
        x = np.clip(x, -50.0, 50.0)

        # occasional random “shake” when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[AdamSmooth n={n} iter={t}] shake (stagnation)")

    if best_a is None:                     # fallback (should never happen)
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# 2️⃣  Differential Evolution (DE) – “best/1/bin” strategy
# ----------------------------------------------------------------------
def run_de_opt_large(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 30,
    max_gens: int = 20_000,
) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex with a larger population.
    Returns the best (a, objective) discovered before *deadline*.
    """
    # initialise population (list of numpy arrays)
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))
    pop = np.stack(pop)                      # shape (popsize, n)

    obj_vals = np.empty(popsize, dtype=np.float64)
    for i in range(popsize):
        obj_vals[i] = compute_objective(pop[i])

    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8      # differential weight
    CR = 0.9     # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # choose two distinct donors different from i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2 = np.random.choice(idxs, 2, replace=False)

            # mutation: best + F·(pop[r1] – pop[r2])
            mutant = best_a + F * (pop[r1] - pop[r2])
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover (guarantee at least one mutant component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DElarge n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # stop early if we have been idle for a while
        if time.time() - last_improve > 15.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# 3️⃣  (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """Simple (1+1) ES with 1/5‑rule step‑size adaptation."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # slightly increase step size
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay on failure

        # reset if stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# 4️⃣  Tiny local hill‑climber (pairwise mass exchange)
# ----------------------------------------------------------------------
def local_refine(a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """Try very small pairwise swaps while time permits."""
    best = a.copy()
    best_val = compute_objective(best)
    n = len(a)
    attempts = 0
    while time.time() < deadline and attempts < 50_000:
        i, j = np.random.randint(n, size=2)
        if i == j or best[i] <= 0.0:
            attempts += 1
            continue
        # move a tiny random fraction of the mass from i to j
        delta = np.random.rand() * best[i] * 0.1   # ≤10 % of a[i]
        trial = best.copy()
        trial[i] -= delta
        trial[j] += delta
        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best = trial
            # occasional progress print (very cheap)
            if attempts % 5000 == 0:
                print(f"[Refine] improved to {best_val:.6f}")
        attempts += 1
    return best, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects a total time budget and returns the
    best sequence (as a plain Python list) found across several lengths.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # small safety margin

    best_seq = None
    best_val = np.inf

    # try a few lengths – larger n gives more freedom but costs more
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------- build initial candidates -------------------
        init_cands = []

        # uniform distribution
        init_cands.append(np.full(n, 1.0 / n, dtype=np.float64))

        # exponential decay (r ≈ 0.5)
        r = 0.5
        exp_seq = np.power(r, np.arange(n, dtype=np.float64))
        exp_seq = exp_seq / exp_seq.sum()
        init_cands.append(exp_seq)

        # linearly decreasing sequence
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_cands.append(dec)

        # warm‑start from the current global best (if same length)
        if best_seq is not None and len(best_seq) == n:
            init_cands.append(np.asarray(best_seq, dtype=np.float64))

        # historic construction, if it exists
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 1e-6:
                hs = hs / s
            else:
                hs = np.full(n, 1.0 / n)
            init_cands.append(hs)

        # ----------------------------------------------------------------
        # Phase 1 – Adam on smooth max
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        phase1_time = min(remaining * 0.30, 80.0)          # at most ~80 s
        phase1_deadline = time.time() + phase1_time
        for restart_idx, cand in enumerate(init_cands[:2]):   # only a couple of restarts
            if time.time() > phase1_deadline:
                break
            a_opt, val_opt = run_adam_smooth_max(
                n, cand, phase1_deadline,
                beta_start=5.0, beta_end=80.0,
                max_iters=200_000,
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[Phase1 n={n} restart={restart_idx}] GLOBAL best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Phase 2 – Differential Evolution (larger population)
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        phase2_time = min(remaining * 0.40, 120.0)
        phase2_deadline = time.time() + phase2_time
        a_de, val_de = run_de_opt_large(
            n,
            init_pop=init_cands[:3],   # a few warm starts
            deadline=phase2_deadline,
            popsize=30,
            max_gens=20_000,
        )
        if val_de < best_val:
            best_val, best_seq = val_de, a_de.copy()
            print(f"[Phase2 DE n={n}] GLOBAL best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Phase 3 – (1+1) ES fine‑tuning
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining <= 0:
            break
        phase3_time = min(remaining * 0.30, 60.0)
        phase3_deadline = time.time() + phase3_time
        if best_seq is not None and len(best_seq) == n:
            a_es, val_es = run_es_opt(
                n,
                best_seq,
                deadline=phase3_deadline,
                max_iter=100_000,
            )
            if val_es < best_val:
                best_val, best_seq = val_es, a_es.copy()
                print(f"[Phase3 ES n={n}] GLOBAL best = {best_val:.6f}")

        # ----------------------------------------------------------------
        # Optional cheap local refinement (if we still have time left)
        # ----------------------------------------------------------------
        remaining = deadline - time.time()
        if remaining > 2.0 and best_seq is not None and len(best_seq) == n:
            a_ref, val_ref = local_refine(best_seq, deadline=time.time() + remaining)
            if val_ref < best_val:
                best_val, best_seq = val_ref, a_ref.copy()
                print(f"[Refine n={n}] GLOBAL best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed but kept for safety.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nFinished. Best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
