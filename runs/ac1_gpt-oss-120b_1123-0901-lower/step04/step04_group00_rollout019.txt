```
def interpolate_seq(seq, new_n):
    old_n = len(seq)
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    seq_interp = np.interp(x_new, x_old, seq)
    seq_interp = np.maximum(seq_interp, 0)
    seq_interp /= seq_interp.sum()
    return seq_interp
```

```
def smooth_max_weights(conv, lam):
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0:
        w = np.full_like(conv, 1.0/conv.size)
    else:
        w /= w_sum
    return w, max_c
```

```
def adam_optimize(a0, n, fft_len_conv, fft_len_corr, deadline):
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03
    best_local_a = a.copy()
    best_local_val = compute_obj(a, n, fft_len_conv)
    last_improve = time.time()
    max_iters = 20000  # adapt to n maybe
    for t in range(1, max_iters+1):
        if time.time() > deadline:
            break
        # softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv and weights
        conv = full_convolution(a, fft_len_conv)
        w, max_c = smooth_max_weights(conv, lam)
        # gradient
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1-beta1) * grad_log
        v = beta2 * v + (1-beta2) * (grad_log**2)
        m_hat = m / (1-beta1**t)
        v_hat = v / (1-beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        lam = min(lam * lam_factor, 1e6)
        # compute val
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
        # stagnation check
        if time.time() - last_improve > 4.0:
            # small pert
            logits += (np.random.rand(n)-0.5) * 1e-2
            last_improve = time.time()
    return best_local_a, best_local_val
```

```
def subgrad_optimize(a0, n, fft_len_conv, fft_len_corr, deadline, init_val=None):
    a = a0 / a0.sum()
    best_a = a.copy()
    best_val = init_val if init_val is not None else compute_obj(a, n, fft_len_conv)
    sub_last_improve = time.time()
    eta = 0.6
    for it in range(2000):
        if time.time() > deadline:
            break
        conv = full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        # evaluate
        conv = full_convolution(a, fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            sub_last_improve = time.time()
        if time.time() - sub_last_improve > 4.0:
            break
    return best_a, best_val
```

```
    best_a = a.copy()
    best_val = compute_obj(a, n, fft_len_conv)

            break
        # propose random redistribution: pick idx1 random, idx2 random, move epsilon fraction from idx1 to idx2
        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue
        eps = np.random.rand() * 0.1 * a[i1]  # up to 10% of source mass

            best_val = val
            best_a = new_a
            a = new_a
```

```
def full_convolution(a, fft_len):
    n = a.shape[0]
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
```

```
def cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n-1
    return corr[start:start+n]
```

"""
Optimises a non‑negative step‑function sequence to minimise
    F(a) = 2·n·max(a∗a) / (sum a)²

using a multi‑stage scheme:
1. Generate diverse initial guesses (warm‑start, random Dirichlet,
   edge‑concentrated power‑laws, edge Gaussians, inverse‑root shapes,
   multi‑peak Gaussians).
2. Refine each guess with Adam on a smooth‑max surrogate of the max‑convolution.
3. Further improve with an exact sub‑gradient (exponentiated‑gradient) method.
4. Perform a short hill‑climbing local search.
5. Keep the best sequence across all lengths and restart a long‑run Adam
   on the current global best when time permits.

The algorithm respects a total time budget (default 1000 s) and returns
a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float('inf')
    return 2.0 * n * max_c / (s * s)

def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    total = new.sum()
    if total < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / total
    return new

def _generate_initial_candidates(n: int, best_seq: np.ndarray,
                                 hist_seq: np.ndarray) -> list:
    """
    Produce a list of diverse starting points for length n.
    Each candidate is a probability vector (sum = 1, non‑negative).
    """
    candidates = []

    # warm start from the globally best (if length matches)
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.copy())

    # warm start by interpolating the best to this length
    if best_seq is not None and best_seq.shape[0] != n:
        candidates.append(_interpolate_seq(best_seq, n))

    # historic sequence (height_sequence_1)
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            cand = hist_seq[:n]
        else:
            cand = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 1e-12:
            cand = np.random.dirichlet(np.ones(n))
        else:
            cand = cand / cand.sum()
        candidates.append(cand)

    # pure random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes   ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # two‑Gaussian edge shapes  exp(-γ·i²)+exp(-γ·(n‑1‑i)²)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # inverse‑root edge shape   (i+1)**‑α + (n‑i)**‑α
    for alpha in [0.5, 0.8, 1.0]:
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        shape = shape / shape.sum()
        candidates.append(shape)

    # three Gaussian peaks at 0.25, 0.5, 0.75 of the interval
    for sigma in [0.05, 0.1, 0.2]:
        i = np.arange(n, dtype=np.float64) / (n - 1)
        pts = [0.25, 0.5, 0.75]
        shape = sum(np.exp(-0.5 * ((i - p) / sigma) ** 2) for p in pts)
        shape = shape / shape.sum()
        candidates.append(shape)

    return candidates

def _adam_optimize(a0: np.ndarray, n: int, fft_len_conv: int,
                   fft_len_corr: int, deadline: float) -> tuple:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) found before deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03
    max_iters = 25000  # generous upper bound, will be cut by deadline

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max projection onto simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # evaluate exact objective (cheap, we already have max_c)
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam iter={t:06d}] new best {cur_val:.6f}")

        # stagnation handling – a tiny random shake after a few seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val

def _subgrad_optimize(a0: np.ndarray, n: int, fft_len_conv: int,
                      fft_len_corr: int, deadline: float,
                      init_val: float = None) -> tuple:
    """
    Exact sub‑gradient optimisation (exponentiated gradient on the simplex).
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(3000):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # evaluation
        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val

def _hill_climb_optimize(a0: np.ndarray, n: int, fft_len_conv: int,
                         deadline: float, init_val: float = None) -> tuple:
    """
    Very cheap local search: repeatedly move a small amount of probability
    from one index to another and keep the move if it improves the objective.
    """

    if init_val is None:

        else:

    return best_a, best_val

"""
Optimises a non‑negative step‑function sequence to minimise

        F(a) = 2·n·max(a∗a) / (sum a)²

using a multi‑stage scheme:

1.  Generate diverse initial guesses (warm‑starts, random Dirichlet,
    edge‑concentrated power‑laws, edge Gaussians, inverse‑root edges,
    multi‑peak Gaussians).
2.  Refine each guess with Adam on a smooth‑max surrogate of the
    max‑convolution.
3.  Further improve with an exact sub‑gradient (exponentiated‑gradient)
    method that works directly on the true maximum.
4.  Apply a cheap hill‑climbing local search.
5.  Keep the best sequence across all tested lengths;
    if time remains, perform a final long‑run Adam refinement on the
    current global best.

The algorithm respects a total time budget (default 1000 s) and returns
a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) using an FFT of length fft_len."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) = n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _interpolate_seq(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Resize a 1‑D sequence by linear interpolation and renormalise."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, seq)
    new = np.maximum(new, 0.0)
    tot = new.sum()
    if tot < 1e-12:
        new = np.random.dirichlet(np.ones(new_n))
    else:
        new = new / tot
    return new


def _generate_initial_candidates(n: int,
                                 best_seq: np.ndarray | None,
                                 hist_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a modestly sized, diverse set of starting points for length n.
    All returned arrays are non‑negative and sum to 1.
    """
    cand = []

    # warm start from the global best (exact length)
    if best_seq is not None and best_seq.shape[0] == n:
        cand.append(best_seq.copy())
    # warm start by interpolating the global best to this length
    if best_seq is not None and best_seq.shape[0] != n:
        cand.append(_interpolate_seq(best_seq, n))

    # historic sequence if available
    if hist_seq is not None:
        if hist_seq.shape[0] > n:
            tmp = hist_seq[:n]
        else:
            tmp = np.pad(hist_seq, (0, n - hist_seq.shape[0]))
        tmp = np.maximum(tmp, 0.0)
        tot = tmp.sum()
        if tot < 1e-12:
            tmp = np.random.dirichlet(np.ones(n))
        else:
            tmp = tmp / tot
        cand.append(tmp)

    # pure random Dirichlet
    cand.append(np.random.dirichlet(np.ones(n)))

    # edge‑concentrated power‑law shapes ((i+1)*(n-i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 5):
        shape = np.power(prod, alpha, where=prod > 0,
                         out=np.zeros_like(prod))
        if shape.sum() > 0:
            cand.append(shape / shape.sum())

    # two‑Gaussian edge shapes
    for gamma in (0.2, 0.5, 1.0, 2.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        cand.append(shape / shape.sum())

    # inverse‑root edge shapes (i+1)**‑α + (n‑i)**‑α
    for alpha in (0.5, 0.8, 1.0):
        i = np.arange(n, dtype=np.float64) + 1.0
        shape = np.power(i, -alpha) + np.power(n - i + 1.0, -alpha)
        cand.append(shape / shape.sum())

    # three Gaussian peaks (0.25,0.5,0.75) with various widths
    for sigma in (0.05, 0.10, 0.20):
        x = np.arange(n, dtype=np.float64) / (n - 1)
        peaks = sum(np.exp(-0.5 * ((x - p) / sigma) ** 2) for p in (0.25, 0.5, 0.75))
        cand.append(peaks / peaks.sum())

    # shuffle and keep a limited number (12) to stay within budget
    np.random.shuffle(cand)
    return cand[:12]


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best (a, value) encountered before the deadline.
    """
    a = a0 / a0.sum()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.03

    best_a = a.copy()
    conv = _full_convolution(a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # simplex projection via soft‑max
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        ws = w.sum()
        if ws == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= ws

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        cur_val = 2.0 * n * max_c / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Adam it={t:06d}] new best {cur_val:.6f}")

        # small random shake after a few seconds without improvement
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def _subgrad_optimize(a0: np.ndarray,
                      n: int,
                      fft_len_conv: int,
                      fft_len_corr: int,
                      deadline: float,
                      init_val: float | None = None,
                      max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Exact sub‑gradient optimisation (exponentiated‑gradient) on the true max.
    """
    a = a0 / a0.sum()
    best_a = a.copy()
    if init_val is None:
        conv = _full_convolution(a, fft_len_conv)
        best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    else:
        best_val = init_val
    last_improve = time.time()
    eta = 0.6

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        conv = _full_convolution(a, fft_len_conv)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[n={n} Subgrad it={it}] new best {cur_val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


def _hill_climb_optimize(a0: np.ndarray,
                         n: int,
                         fft_len_conv: int,
                         deadline: float,
                         init_val: float | None = None,
                         max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Simple cheap local search: repeatedly shift a small amount of mass between
    two random coordinates and keep the move if it improves the objective.
    """
    a = a0.copy()
    cur_val = _objective(a, n, fft_len_conv) if init_val is None else init_val
    best_a = a.copy()
    best_val = cur_val

    for it in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        i1, i2 = np.random.randint(0, n, size=2)
        if i1 == i2:
            continue

        # move up to 5 % of the probability mass from i1 to i2
        eps = np.random.rand() * 0.05 * a[i1]
        if eps <= 0.0:
            continue
        new_a = a.copy()
        new_a[i1] -= eps
        new_a[i2] += eps
        if new_a[i1] < -1e-15:
            continue
        new_a /= new_a.sum()

        new_val = _objective(new_a, n, fft_len_conv)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a.copy()
            a = new_a
            cur_val = new_val
            if it % 50 == 0:
                print(f"[n={n} HillClimb it={it}] new best {new_val:.6f}")
        else:
            # very occasional uphill move to escape tiny plateaus
            if np.random.rand() < 0.001:
                a = new_a

    return best_a, best_val
