"""
Propose a candidate step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm is based on the linear‑programming “mix‑with‑g” scheme
described in the problem statement:

* Start from a (normalised) vector a (sum = 1).
* Let M = max(a*a).  Build the convolution matrix T so that
      (a * g) = T @ g .
* Solve the LP
      maximise   sum(g)
      subject to g ≥ 0 ,  T @ g ≤ M .
  The optimal g puts mass where a is small, thereby flattening the
  convolution.
* Combine the current a with the normalised g (convex combination)
  and repeat.  After a few LP steps we perform a cheap random‑swap
  local refinement.

The whole procedure respects the 1000 s budget, works for modest
sizes (n≈1024–2048) and returns the best sequence encountered.
"""

import time
import numpy as np
import cvxpy as cp
from scipy import sparse


def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return the dense linear convolution matrix T such that
        np.convolve(a, g) == (T @ g)
    for any vector g of the same length as a.
    Shape of T is (2·n‑1, n).
    """
    n = a.shape[0]
    # indices for rows (k) and columns (j)
    rows = np.arange(2 * n - 1)[:, None]          # (2n‑1,1)
    cols = np.arange(n)[None, :]                  # (1,n)
    idx = rows - cols                              # (2n‑1,n)  = k‑j
    mask = (idx >= 0) & (idx < n)
    # Where mask is True, take a[idx]; elsewhere 0.
    T = np.where(mask, a[idx], 0.0)
    return T


def solve_lp(T: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve the linear program:
        maximise   sum(g)
        subject to g >= 0 ,  T @ g <= M
    Returns the optimal g as a 1‑d ndarray, or None on failure.
    """
    n = T.shape[1]
    g = cp.Variable(n, nonneg=True)
    constraints = [T @ g <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)

    try:
        # ECOS is a fast open‑source interior‑point solver.
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except (cp.SolverError, Exception):
        return None

    if g.value is None:
        return None
    return np.asarray(g.value).reshape(-1)


def evaluate_obj(a: np.ndarray) -> float:
    """
    Compute 2·n·max(conv(a,a)) for a normalised vector (sum = 1).
    """
    n = a.shape[0]
    conv = np.convolve(a, a)
    return 2.0 * n * conv.max()


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑value step‑function within the given time budget.
    The function returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Choose a problem size.  1024 is a sweet spot for the LP approach.
    # Larger sizes give a lower theoretical bound but make each LP slower.
    # ------------------------------------------------------------------
    n = 1024

    # ------------------------------------------------------------------
    # Initialise a from the previous best if it exists, otherwise random.
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.array(height_sequence_1, dtype=np.float64)
        if cand.shape[0] >= n:
            a = cand[:n].copy()
        else:
            a = np.pad(cand, (0, n - cand.shape[0]), "constant")
    else:
        a = np.random.dirichlet(np.ones(n))

    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()               # ensure sum = 1

    # ------------------------------------------------------------------
    # Main LP‑mixing loop.
    # ------------------------------------------------------------------
    max_outer_iters = 30        # each iteration solves one LP
    t_candidates = [0.2, 0.5, 0.8]   # mixing fractions to try

    for outer in range(max_outer_iters):
        if time.time() > deadline:
            break

        # Current objective value
        cur_val = evaluate_obj(a)
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[LP iter {outer}] new best {best_val:.6f}")

        # Stop early if we already beat the target
        if best_val < 1.503:
            break

        # Compute convolution matrix for the current a
        T = build_conv_matrix(a)          # shape (2n‑1, n)
        M = np.convolve(a, a).max()       # current max convolution value

        # Solve the LP to obtain a candidate g
        g_opt = solve_lp(T, M)
        if g_opt is None:
            print("[LP iter] LP failed – aborting LP phase")
            break

        # Normalise g to be a probability vector
        g_opt = np.maximum(g_opt, 0.0)
        sum_g = g_opt.sum()
        if sum_g == 0.0:
            print("[LP iter] Obtained zero g – aborting LP phase")
            break
        g_norm = g_opt / sum_g

        # Try a handful of mixing coefficients and keep the best result
        best_mix_a = a
        best_mix_val = cur_val

        for t in t_candidates:
            a_candidate = (1.0 - t) * a + t * g_norm
            a_candidate = np.maximum(a_candidate, 0.0)
            a_candidate /= a_candidate.sum()
            val_candidate = evaluate_obj(a_candidate)
            if val_candidate < best_mix_val - 1e-12:
                best_mix_val = val_candidate
                best_mix_a = a_candidate

        a = best_mix_a   # continue from the best mixed point

    # ------------------------------------------------------------------
    # Optional local random‑swap refinement (cheap hill‑climbing).
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_local = best_seq.copy()
        local_iters = 3000
        for it in range(local_iters):
            if time.time() > deadline:
                break
            i, j = np.random.choice(n, 2, replace=False)
            # Move a small random amount from i to j
            eps = min(a_local[i], 0.01) * (np.random.rand() * 0.5 + 0.5)
            if eps <= 0:
                continue
            a_new = a_local.copy()
            a_new[i] -= eps
            a_new[j] += eps
            val_new = evaluate_obj(a_new)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_seq = a_new.copy()
                a_local = a_new
                print(f"[local] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never be needed, but guarantees a valid output.
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = evaluate_obj(best_seq)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
