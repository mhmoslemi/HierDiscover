def _combine_sequences(f, g, target_sum, t=0.5):

def _adam_optimize(a, n, deadline, max_iters, best_seq, best_score):
    target_sum = math.sqrt(2.0 * n)
    # normalize a
    a = a.astype(float)
    s = a.sum()
    if s > 0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr0 = 0.05
    jitter_interval = 2000
    tau_factor = 0.2
    # decay_rate = max_iters / 10.0  # ensure not zero
    decay_rate = max(1.0, max_iters / 10.0)
    Nfft = 1 << ((2 * n - 2).bit_length())
    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        # convolution f*f
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[:2 * n - 1]
        M = float(conv.max())
        # temperature
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-8:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)
        # weighted reverse conv (gradient)
        rev = _weighted_rev(a, w, Nfft)

        coeff = 4.0 * n / (target_sum ** 2)
        # constant term for projection onto sum
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        m = beta1 * m + (1 - beta1) * grad

"""
propose_candidate
=================

Search for a non‑negative step‑function (vector of heights) that minimizes the
provided ``evaluate_sequence`` objective.

The algorithm combines three ideas:

1. **Diverse initialisation** – uniform, Dirichlet, exponential, Hann/Blackman,
   Gaussian and random ramps.
2. **Two‑phase Adam optimisation** – a soft‑max smoothing of the max‑convolution
   supplies a smooth surrogate gradient, which is gradually turned off.
   After every step the vector is projected onto the feasible set
   (non‑negative, ℓ₁‑norm = √(2 n)).
3. **Linear‑programming refinement** – after a good Adam run, we solve a linear
   program that maximises the ℓ₁‑norm of a vector `g` under the constraint
   `f* g ≤ M` (where `M = max(f*f)`).  Mixing the current solution `f` with this
   `g` (convex combination) provably reduces the peak of the autocorrelation.
   A few LP‐guided mixing steps are performed for the smallest problem size
   (≤ 2500) where the LP is cheap.

The routine respects a wall‑clock budget (default 1000 s) and returns the
best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the Adam optimiser
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max over a vector of convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Weighted reverse convolution (gradient of the soft‑maxed max‑convolution).

    Returns a vector ``rev`` of length ``len(a)`` where
    rev[i] = Σ_k w[k] * a[k-i]   (indices outside the range are ignored).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Maximum value of the (auto)convolution a*a."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())

# Initial vector generator
def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Create a small, diverse set of feasible start vectors."""
    rng = np.random.default_rng()
    init = []

    # uniform
    init.append(np.full(n, target_sum / n))

    # Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(1.0, 1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # reversed exponential
    init.append(x[::-1] / x.sum() * target_sum)

    # Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    init.append(black / black.sum() * target_sum)

    # Gaussian bell (centered)
    sigma = n / 8.0
    xg = np.arange(n) - (n - 1) / 2.0
    gauss = np.exp(-0.5 * (xg / sigma) ** 2)
    init.append(gauss / gauss.sum() * target_sum)

    # Linear ramp up
    ramp_up = np.linspace(0.0, 1.0, n)
    init.append(ramp_up / ramp_up.sum() * target_sum)

    # Linear ramp down
    ramp_down = ramp_up[::-1]
    init.append(ramp_down / ramp_down.sum() * target_sum)

    # small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


# ----------------------------------------------------------------------
# Adam optimiser (soft‑max phase → exact sub‑gradient phase)
def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Run Adam optimisation on a single start vector.
    Returns the best sequence seen inside this run together with its exact objective
    (max convolution after normalisation, which equals the evaluate_sequence score).
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a.astype(float)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / n)

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 2000

    # temperature schedule parameters
    tau_factor = 0.20
    decay_rate = max(1.0, max_iters / 10.0)

    # FFT length (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 2).bit_length())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact self‑convolution & peak -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & soft‑max weighting -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-8:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- gradient of the smoothed maximum -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam gradient step (after normalisation the objective is max(conv)) -----
        coeff = 4.0 * n / (target_sum ** 2)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto the feasible set -----
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ----- exact objective after the projection -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        cur_score = float(conv.max())

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

# Simple hill‑climbing local search (random jitter)
def _local_hill_climb(a: np.ndarray,
                      n: int,
                      target_sum: float,
                      deadline: float,
                      max_steps: int) -> tuple[np.ndarray, float]:
    """Random perturbations with greedy acceptance."""
    best_a = a.copy()
    best_score = _max_convolution(best_a, 1 << ((2 * n - 2).bit_length()))
    Nfft = 1 << ((2 * n - 2).bit_length())

    for step in range(max_steps):
        if time.time() >= deadline:
            break
        # decreasing noise magnitude
        sigma = 0.012 * target_sum / n * (1.0 - step / max_steps)
        cand = best_a + np.random.randn(n) * sigma
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s > 0.0:
            cand = cand * (target_sum / s)
        else:
            cand = np.full_like(cand, target_sum / n)

        cur = _max_convolution(cand, Nfft)
        if cur < best_score - 1e-12:
            best_score = cur
            best_a = cand
    return best_a, best_score

# Linear‑programming refinement (if cvxpy is available)
def _solve_lp_maxsum(f: np.ndarray, M: float):
    """Solve   maximise Σ g_j   s.t.   g≥0  and  conv(f,g) ≤ M.
    Returns the optimal g as a numpy array, or None on failure."""
    try:
        import cvxpy as cp
    except Exception as exc:
        print("[LP] cvxpy not available:", exc)
        return None

    n = len(f)
    g = cp.Variable(n, nonneg=True)
    # convolution of the constant vector f with variable g
    conv_expr = cp.conv(f, g)                # length 2n‑1
    constraints = [conv_expr <= M]
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000, feastol=1e-9, abstol=1e-9, reltol=1e-9)
    except Exception as exc:
        print("[LP] solve error:", exc)
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        print("[LP] not optimal – status:", prob.status)
        return None

    g_val = np.asarray(g.value).flatten()
    g_val = np.maximum(g_val, 0.0)
    return g_val


def _combine_sequences(f: np.ndarray, g: np.ndarray, target_sum: float, t: float = 0.5) -> np.ndarray:
    """Convex combine f with the (ℓ₁‑normalised) g."""
    if g.sum() > 0.0:
        g_norm = g * (target_sum / g.sum())
    else:
        g_norm = np.full_like(g, target_sum / len(g))
    new = (1.0 - t) * f + t * g_norm
    new = np.maximum(new, 0.0)
    s = new.sum()
    if s > 0.0:
        new = new * (target_sum / s)
    else:
        new = np.full_like(new, target_sum / len(new))
    return new
