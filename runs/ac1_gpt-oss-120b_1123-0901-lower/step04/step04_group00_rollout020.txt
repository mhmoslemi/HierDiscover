"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* A *coarse* representation on a small number of bins (≈ n/32) is
  optimised with Adam on a smooth‑max surrogate of the auto‑convolution.
  The bin‑weights are kept on the probability simplex via a soft‑max
  parametrisation, and the sequence is forced to be symmetric after
  each iteration (a = (a+rev(a))/2).
* The best coarse solution is expanded to a full vector and refined
  with an *exponentiated‑gradient sub‑gradient* method that works on the
  exact max‑convolution.
* Finally a small stochastic “hill‑climbing’’ phase swaps tiny amounts
  of mass between random positions.
* Several initialisations (warm‑start, historic, random Dirichlet,
  edge‑concentrated power‑laws, double‑exponential edges, two‑Gaussian
  edges, spiky seeds) are tried for a few lengths (1024…16384).  The
  overall time budget (≈1000 s) is respected; the best sequence found
  is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the next power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT of given length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()


def _objective_from_conv(conv: np.ndarray, n: int, total_sum: float) -> float:
    """Exact objective 2·n·max(conv) / (total_sum²)."""
    return 2.0 * n * conv.max() / (total_sum ** 2)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5               # safety margin

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # lengths to try (shorter lengths are cheaper, longer ones give
    # more “resolution’’).  The loop will stop early if we run out of time.
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ----------- FFT lengths for this size --------------------
        conv_len = 2 * n - 1
        fft_len_conv = _next_pow2(conv_len)
        # correlation needs length conv_len + n - 1 ≤ 3·n
        fft_len_corr = _next_pow2(conv_len + n - 1)

        # ----------- Binning (coarse representation) -------------
        # use ≈ n/32 bins, at least 32 and at most 256 bins.
        B = max(32, min(256, n // 32))
        edges = np.arange(B + 1) * n // B
        edges[-1] = n
        bin_widths = edges[1:] - edges[:-1]                # (B,)
        # map every index to its bin id
        bin_assign = np.empty(n, dtype=np.int32)
        for b in range(B):
            bin_assign[edges[b]:edges[b + 1]] = b

        # ----------- Helper to evaluate the exact objective --------
        def exact_val(a_vec: np.ndarray) -> float:
            conv = _full_convolution(a_vec, fft_len_conv)
            return _objective_from_conv(conv, n, a_vec.sum())

        # ----------- Build a list of initial candidates ------------
        candidates_bins = []   # each entry is a (B,) array of bin weights (sum=1)

        # Warm start from the globally best (if same length)
        if best_seq is not None and len(best_seq) == n:
            w = np.bincount(bin_assign, weights=np.asarray(best_seq, dtype=float), minlength=B)
            w = w / w.sum()
            candidates_bins.append(w)

        # Historic solution if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            h = np.asarray(hseq, dtype=float)
            if h.shape[0] > n:
                h = h[:n]
            elif h.shape[0] < n:
                h = np.pad(h, (0, n - h.shape[0]))
            h = np.maximum(h, 0.0)
            w = np.bincount(bin_assign, weights=h, minlength=B)
            if w.sum() < 1e-12:
                w = np.random.dirichlet(np.ones(B))
            else:
                w = w / w.sum()
            candidates_bins.append(w)

        # Pure random Dirichlet on the bins
        candidates_bins.append(np.random.dirichlet(np.ones(B)))

        # Edge‑concentrated power‑law shapes
        idx = np.arange(n, dtype=float) + 1.0
        prod = idx * (n - idx + 1.0)                     # maximum at centre
        for alpha in np.linspace(-3.0, -0.2, 7):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() == 0:
                continue
            shape /= shape.sum()
            w = np.bincount(bin_assign, weights=shape, minlength=B)
            w = w / w.sum()
            candidates_bins.append(w)

        # Double‑exponential edge shape (∝ e^{-β i}+e^{-β (n-1-i)})
        for beta in [0.1, 0.3, 0.7, 1.5, 3.0]:
            i = np.arange(n, dtype=float)
            shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
            shape /= shape.sum()
            w = np.bincount(bin_assign, weights=shape, minlength=B)
            w = w / w.sum()
            candidates_bins.append(w)

        # Two‑Gaussian edge shape (as in the previous code)
        for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
            i = np.arange(n, dtype=float)
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            w = np.bincount(bin_assign, weights=shape, minlength=B)
            w = w / w.sum()
            candidates_bins.append(w)

        # Sparse spiky seeds: pick a few random bins, give them all mass
        for _ in range(3):
            w = np.zeros(B, dtype=float)
            k = max(1, B // 10)                     # about 10 % active bins
            idxs = np.random.choice(B, size=k, replace=False)
            w[idxs] = np.random.rand(k) + 0.1
            w = w / w.sum()
            candidates_bins.append(w)

        # Uniform bins
        candidates_bins.append(np.full(B, 1.0 / B, dtype=float))

        # ----------------------------------------------------------
        # Optimise every coarse candidate
        for cand_id, w0 in enumerate(candidates_bins):
            if time.time() > deadline:
                break

            # initialise logits for the Adam phase
            logits = np.log(w0 + 1e-15)

            # Adam hyper‑parameters
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.4                # start temperature for smooth‑max
            lam_factor = 1.025

            # Helper to build the full vector from current logits
            def _logits_to_full(logits_vec: np.ndarray) -> np.ndarray:
                y = _softmax(logits_vec)                     # bin weights, sum=1
                a = y[bin_assign] / bin_widths[bin_assign]   # expand to length n
                # enforce symmetry (greatly reduces the maximum)
                a = (a + a[::-1]) * 0.5
                return a, y

            # ------------------------------------------------------
            # Phase 1 – Adam on the smooth‑max surrogate
            last_improve = time.time()
            max_iters1 = int(8000 * 1024 / n) + 1500   # a few thousand per length
            for t in range(1, max_iters1 + 1):
                a, y = _logits_to_full(logits)

                # compute convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # smoother weights (log‑sum‑exp trick)
                w_s = np.exp(lam * (conv - max_c))
                w_sum = w_s.sum()
                if w_sum == 0:
                    w_s = np.full_like(conv, 1.0 / conv.size)
                else:
                    w_s /= w_sum

                # gradient of the surrogate w.r.t a
                grad_a = 2.0 * _cross_correlation(w_s, a, fft_len_corr)

                # aggregate to bin‑level gradient
                # grad_w[b] = Σ_i∈b grad_a[i] / width_b
                grad_w = np.bincount(bin_assign, weights=grad_a, minlength=B) / bin_widths

                # back‑prop through the soft‑max (logits → y)
                avg = np.dot(y, grad_w)
                grad_log = y * (grad_w - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule (smooth‑max becomes tighter)
                lam = min(lam * lam_factor, 1e6)

                # occasional exact evaluation (cheap, no extra FFT)
                if t % 200 == 0 or t == max_iters1:
                    val = exact_val(a)
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # stagnation handling – small random shake
                if time.time() - last_improve > 3.0:
                    logits += (np.random.rand(B) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ------------------------------------------------------
            # Phase 2 – Exact sub‑gradient on the full vector
            a, _ = _logits_to_full(logits)   # final a from the coarse phase
            a = a / a.sum()                 # safety normalisation
            eta = 0.6                       # initial exponentiated‑gradient step
            sub_last_improve = time.time()
            max_sub_iters = int(3000 * 1024 / n) + 500
            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # active set (indices attaining the maximum)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w_active = np.zeros_like(conv)
                    w_active[active] = 1.0 / active.sum()
                else:
                    w_active = np.zeros_like(conv)

                # sub‑gradient of max(conv) w.r.t a
                grad_a = 2.0 * _cross_correlation(w_active, a, fft_len_corr)

                # exponentiated‑gradient update (keeps a ≥ 0)
                a = a * np.exp(-eta * grad_a)
                s = a.sum()
                if s < 1e-12:                      # avoid collapse to zero
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                # enforce symmetry after each step
                a = (a + a[::-1]) * 0.5

                # evaluate
                val = exact_val(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # simple step‑size decay
                eta *= 0.9995

                # early exit on stagnation
                if time.time() - sub_last_improve > 3.5:
                    break
                if time.time() > deadline:
                    break

            # ------------------------------------------------------
            # Phase 3 – tiny stochastic hill‑climbing (local swaps)
            if time.time() < deadline:
                hill_steps = min(2000, int(5000 * 1024 / n))
                for _ in range(hill_steps):
                    i, j = np.random.randint(0, n, size=2)
                    if i == j:
                        continue
                    eps_move = 0.02 * a[i]               # move up to 2 % of mass at i
                    if eps_move <= 0:
                        continue
                    new_a = a.copy()
                    transfer = min(eps_move, new_a[i])
                    new_a[i] -= transfer
                    new_a[j] += transfer
                    # keep symmetry after the move
                    new_a = (new_a + new_a[::-1]) * 0.5
                    val = exact_val(new_a)
                    if val < best_val:
                        best_val = val
                        best_seq = new_a.copy()
                        a = new_a
                        # optional: print improvement
                        print(f"[n={n} cand={cand_id} Hillclimb] new best {val:.6f}")

            # ------------------------------------------------------
            # Respect the overall deadline before moving to the next candidate
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Final long‑run refinement on the very best vector (if time permits)
    if best_seq is not None and (time.time() < deadline):
        n = len(best_seq)
        print("\n=== Final refinement on the global best (n=%d) ===" % n)

        # recompute FFT lengths for the chosen size
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n - 2)

        # start from the current best – we operate directly on the full vector
        a = best_seq.copy()
        a = a / a.sum()
        # Adam on the full vector (no binning) – longer schedule
        logits = np.log(a + 1e-15)
        m = np.zeros_like(logits)
        v = np.zeros_like(logits)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.07
        lam = 0.3
        lam_factor = 1.02
        max_iters = 15000
        last_imp = time.time()
        for t in range(1, max_iters + 1):
            # soft‑max → simplex (keeps positivity and sum=1)
            mx = logits.max()
            exp_x = np.exp(logits - mx)
            a = exp_x / exp_x.sum()
            a = (a + a[::-1]) * 0.5   # enforce symmetry each step

            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0:
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum

            grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

            # soft‑max backward
            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad_log
            v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

            lam = min(lam * lam_factor, 1e6)

            if t % 300 == 0:
                val = _objective_from_conv(conv, n, a.sum())
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_imp = time.time()
                    print(f"[Refine Adam iter={t:05d}] new best {val:.6f}")

            # stagnation handling
            if time.time() - last_imp > 5.0:
                logits += (np.random.rand(n) - 0.5) * 5e-3
                last_imp = time.time()

            if time.time() > deadline:
                break

        # final sub‑gradient phase (longer)
        eta = 0.5
        sub_last_imp = time.time()
        max_sub = 5000
        for sub_it in range(max_sub):
            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
            w_act = np.zeros_like(conv)
            if active.any():
                w_act[active] = 1.0 / active.sum()
            grad_a = 2.0 * _cross_correlation(w_act, a, fft_len_corr)

            a = a * np.exp(-eta * grad_a)
            a = a / a.sum()
            a = (a + a[::-1]) * 0.5

            val = exact_val(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                sub_last_imp = time.time()
                print(f"[Refine Subgrad iter={sub_it}] new best {val:.6f}")

            eta *= 0.999
            if time.time() - sub_last_imp > 4.0:
                break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=float)

    # Return as plain Python list (required by the external evaluator)
    return best_seq.tolist()
