"""
Propose a non‑negative step function that minimises the provided
``evaluate_sequence`` objective.

Algorithm (≈ 100 s budget):
--------------------------------------------------------------
1. Normalise any starting vector so that Σa = √(2·n)  (n = length).
   The objective then equals the maximal entry of the full convolution
   a∗a, i.e.  max_k (a∗a)_k.

2. **Linear‑programming mixing phase**
   * For the current vector f compute M = max(f∗f).
   * Solve a linear program
         maximise   Σ g_i
         subject to  f ∗ g ≤ M   (element‑wise)  and   g_i ≥ 0.
     This LP has n variables and (2·n‑1) linear constraints.
   * Scale g to the required ℓ₁‑norm (√(2·n)) and update
         f ← (1‑t)·f + t·g   with a small mixing weight t≈0.15.
   * Repeat until M stops improving or time runs out.
   The process drives the convolution toward a flat shape and
   typically reduces the bound well below the L‑BFGS‑only result.

3. **Smooth‑max L‑BFGS refinement**
   * Re‑parameterise the probability vector p = f / Σf via a soft‑max
     (logits y) so that non‑negativity and the sum‑1 constraint are
     satisfied automatically.
   * minimise the smooth approximation
         L_τ(p) = τ·log Σ_k exp((p∗p)_k / τ)
     using limited‑memory BFGS (L‑BFGS‑B).  τ is decreased
     geometrically (τ ← 0.7·τ) until it reaches 10⁻¹².
   * After each L‑BFGS run evaluate the exact objective; keep the best
     vector found.

4. Perform the whole pipeline from a few different starts
   (the supplied ``height_sequence_1`` if present, a uniform start,
   and a random Dirichlet start).  The best sequence over all starts is
   returned.

The implementation respects the 1000 s time limit, prints progress,
and works with only NumPy and SciPy (which are available in the
challenge environment).
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy imports – the environment used by the challenge provides
# them, but we guard against their absence.
try:
    from scipy.optimize import minimize, linprog
    from scipy import sparse
    _SCIPY = True
except Exception:  # pragma: no cover
    _SCIPY = False

# ----------------------------------------------------------------------
# Helper functions for the smooth‑max formulation (identical to the
# previous state‑of‑the‑art code).

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smooth maximum L_τ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Weighted reverse convolution needed for the gradient."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[:len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p∗p (real FFT based). Returns length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """Smooth‑max value and its gradient w.r.t. the logits y."""
    # soft‑max (probability vector)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max and its weights
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # push back through soft‑max Jacobian
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """Repeated L‑BFGS runs with decreasing temperature τ."""
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.7
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the smooth objective -----
        def fun(vec):
            L, g = _objective_and_grad(vec, tau, Nfft)
            return L, g

        # generous limit on function evaluations
        opts = {'maxfun': 1500, 'maxiter': 1500, 'ftol': 1e-12, 'gtol': 1e-12}
        try:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
        except Exception:  # pragma: no cover
            break
        y = res.x

        # ----- exact evaluation -----
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # equal to evaluate_sequence output

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[L‑BFGS improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule -----
        tau *= tau_decay
        # occasional jitter to leave plateaus
        if np.random.rand() < 0.12:
            y += np.random.randn(y.size) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Functions for the linear‑programming mixing phase.

def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Maximum entry of the full convolution a∗a computed via FFT."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _build_conv_matrix(f: np.ndarray):
    """Build a sparse matrix A such that A @ g = f∗g (full convolution)."""
    n = f.shape[0]
    # rows = i + j, cols = i for each i,j∈[0,n)
    rows = np.repeat(np.arange(n), n) + np.tile(np.arange(n), n)
    cols = np.tile(np.arange(n), n)
    data = np.tile(f, n)
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))
    return A


def _solve_lp(f: np.ndarray, M: float, remaining: float):
    """Linear programme:
           maximise   Σ g_i
           subject to   f ∗ g ≤ M   (element‑wise)
                        g_i ≥ 0
       Returns the optimal g (numpy array) or None on failure.
    """
    if not _SCIPY:          # pragma: no cover
        return None

    n = f.size
    A = _build_conv_matrix(f)
    c = -np.ones(n)                     # linprog minimises → maximise Σg
    b = np.full(2 * n - 1, M, dtype=float)

    # Bounds: g_i ≥ 0
    bounds = [(0.0, None) for _ in range(n)]

    try:
        # 'highs' is the default efficient solver; we give it a time budget.
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'time_limit': remaining})
    except Exception:  # pragma: no cover
        return None

    if res.success:
        return res.x
    return None


def _lp_improve(f0: np.ndarray, target_sum: float,
                Nfft: int, deadline: float) -> (np.ndarray, float):
    """Iteratively improve f using the LP‑mixing step described above."""
    f = f0.copy()
    best_f = f.copy()
    best_score = _max_convolution(f, Nfft)   # exact objective (since Σf = target_sum)

    iteration = 0
    max_iters = 30
    t = 0.15                     # mixing weight (small enough for monotone descent)

    while time.time() < deadline and iteration < max_iters:
        M = _max_convolution(f, Nfft)

        # update best record
        if M < best_score:
            best_score = M
            best_f = f.copy()
            print(f"[LP improve] iter={iteration}  max={best_score:.6f}")

        # solve the LP for a partner vector g
        remaining = max(0.0, deadline - time.time() - 0.1)  # tiny safety margin
        g = _solve_lp(f, M, remaining)
        if g is None:
            break

        s = g.sum()
        if s < 1e-12:                     # degenerate solution – stop
            break
        # scale g to the required ℓ₁‑norm
        g = g * (target_sum / s)

        # convex combination
        f = (1 - t) * f + t * g
        iteration += 1

        # early stop if improvement stalls
        if iteration > 5 and abs(M - best_score) < 1e-5:
            break

    return best_f, best_score

def propose_candidate(seed: int = 42, budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    ``evaluate_sequence`` within the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # 5 s safety margin

    # --------------------------------------------------------------
    # Load an existing construction if it exists, otherwise start from uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)                # fallback length

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm.
    base = base * (target_sum / base.sum())

    # Initialise best record.
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size for all convolutions (next power‑of‑two ≥ 2·n‑1).
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Prepare a few different starting points.
    starts = []

    # 1) the known construction (if any)
    starts.append(base.copy())

    # 2) random Dirichlet (exponential sampling) – nice spread
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(rand_exp * (target_sum / rand_exp.sum()))

    # 3) uniform start
    starts.append(np.full(n, target_sum / n, dtype=float))

    # 4) a perturbed version of the known solution
    starts.append(base + np.random.randn(n) * 0.03)

    # ------------------------------------------------------------------
    # Run the two‑phase optimisation from each start while respecting the deadline.
    for idx, start_vec in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # ----- Phase 1: LP‑based mixing -----
        f_lp, score_lp = _lp_improve(start_vec, target_sum, Nfft, deadline)
        if score_lp < best_score:
            best_score = score_lp
            best_seq = f_lp.copy()
            print(f"[phase‑1 best] score={best_score:.6f}")

        # ----- Phase 2: Smooth‑max L‑BFGS refinement -----
        # Convert to probability vector and then to logits.
        p0 = f_lp / target_sum
        eps_log = 1e-12
        y0 = np.log(p0 + eps_log)

        # initial temperature – a few times the current smooth max

        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y0,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Final sanity check
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
