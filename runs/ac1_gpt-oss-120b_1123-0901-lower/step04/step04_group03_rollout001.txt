assistantfinal**Solution Overview**

The code below implements a multi‑stage optimiser for the problem  

\[
F(a)=\frac{2n\;\max (a\star a)}{(\sum a)^2},\qquad a_i\ge 0 .
\]

* **Re‑parameterisation** – the vector is stored as a soft‑max of free
  logits `x`.  This guarantees `a≥0` and `∑a = 1`, so the objective reduces to `2·n·max(a⋆a)`.

* **Smooth Adam** – instead of the non‑differentiable max we minimise a
  *smoothed* version  
  \[
  \ell_\beta(a)=\frac{1}{\beta}\log\Bigl(\sum_k e^{\beta\,c_k}\Bigr)
  \qquad c=a\star a .
  \]
  The gradient is obtained analytically (a correlation of `a` with the
  soft‑max weights).  A continuation on the temperature `β` (from `1` to
  `200`) guides the optimiser from a gentle landscape to the true max.

* **Differential Evolution (DE)** – a global, population‑based search on the
  simplex.  Population size is larger than in the reference solution and the
  mutation/crossover parameters are tuned for faster convergence.

* **(1+1) Evolution Strategy (ES)** – a fine‑grained Gaussian perturbation
  optimiser that works directly on the simplex.

* **Hill‑Climbing (HC)** – a cheap coordinate‑wise move (`ε` mass transferred
  from one index to another).  It exploits the exact objective and can
  squeeze a few extra percent out of a good candidate.

All stages respect the overall time budget (`budget_s`).  The algorithm
tries the longest feasible dimension first (`8192`), falling back to
smaller lengths if time runs out.  The best sequence found is returned as a
plain Python list.

```python
"""
Multi‑stage optimiser for   F(a)=2·n·max(a⋆a)/(∑a)²   with a_i≥0.
- Soft‑max parametrisation (∑a=1).
- Smooth Adam (log‑sum‑exp surrogate) with temperature continuation.
- Differential Evolution (larger population, more generations).
- (1+1) Evolution Strategy.
- Simple hill‑climbing (mass transfer between two coordinates).
All stages stop automatically when the allocated time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (pure functions, no closures)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of a*a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence.
    The vector a is assumed to be non‑negative; sum may be arbitrary.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# 1. Smooth Adam (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def run_smooth_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 120_000,
    beta_start: float = 1.0,
    beta_end: float = 200.0,
):
    """
    Adam optimiser on the smoothed objective
        ℓ_β(a) = (1/β) log Σ exp(β·(a⋆a))
    The temperature β is increased from beta_start to beta_end.
    Returns (best_a, best_val) discovered before *deadline*.
    """
    # initialise logits
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta = beta_start

    beta_factor = (beta_end / beta_start) ** (1.0 / max_iters)

    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                        # a on the simplex
        conv = full_convolution(a)            # length 2n‑1
        max_c = conv.max()

        # ----- smoothed max (log‑sum‑exp) -----
        # stable exponentiation
        shift = max_c
        w = np.exp(beta * (conv - shift))
        w_sum = w.sum()
        smooth_val = shift + np.log(w_sum) / beta

        # ----- gradient of smooth_val w.r.t. a -----
        weights = w / w_sum                    # soft‑max weights, sum=1
        a_rev = a[::-1]
        # correlation a_rev ⋆ weights  → slice [n‑1 : 2n‑1)
        corr = np.convolve(a_rev, weights, mode="full")
        grad_a = 2.0 * corr[n - 1 : 2 * n - 1]

        # exact objective for bookkeeping
        exact_val = 2.0 * n * max_c

        if exact_val < best_val:
            best_val = exact_val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t:06d}] new best: {best_val:.6f}")

        # ----- back‑prop through softmax -----
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # ----- Adam update -----
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # temperature schedule
        beta *= beta_factor

        # shake if stagnating
        if time.time() - last_improve > 6.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:          # fallback – uniform simplex point
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# 2. Differential Evolution on the simplex
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 30,
    max_gens: int = 30_000,
    F: float = 0.5,
    CR: float = 0.9,
):
    """
    Simple DE that stays inside the probability simplex.
    """
    # ---- initialise population ----
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)                # non‑negativity

            # binomial crossover
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # early stop if no progress for a while
        if time.time() - last_improve > 12.0:
            break

    return best_a, best_val

# 3. (1+1) Evolution Strategy (Gaussian mutations)
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 40_000,
):
    """
    Simple (1+1) ES on the simplex.
    """
    a = init_a.copy()
    best_a = a.copy()
    best_val = compute_objective(a)

    sigma = 0.025
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.06, 0.25)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 6.0:
            a = best_a.copy()
            sigma = 0.025
            last_improve = time.time()

    return best_a, best_val

# 4. Simple hill‑climbing (mass transfer between two coordinates)
# ----------------------------------------------------------------------
def run_hc_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    init_step: float = 0.02,
    min_step: float = 1e-4,
):
    """
    Randomly transfer a fraction of mass from one coordinate to another.
    Accepts only improving moves.
    """
    a = init_a.copy()
    best_val = compute_objective(a)
    best_a = a.copy()
    step = init_step
    last_improve = time.time()

    while time.time() < deadline:
        i, j = np.random.choice(n, 2, replace=False)
        delta = step * a[i]          # amount to move from i to j
        if delta <= 0:
            continue
        trial = a.copy()
        trial[i] = max(0.0, trial[i] - delta)
        trial[j] += delta
        # renormalise (sum unchanged, but numerical drift may appear)
        s = trial.sum()
        if s == 0.0:
            continue
        trial /= s

        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial.copy()
            a = trial
            step = min(step * 1.03, 0.1)
            last_improve = time.time()
            print(f"[HC n={n}] new best: {best_val:.6f} step={step:.5f}")
        else:
            # slowly decay step size when stuck
            if time.time() - last_improve > 4.0:
                step = max(step * 0.95, min_step)

    return best_a, best_val
```
