"""
Multi‑phase optimizer for the step‑function objective

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,  a_i ≥ 0

The algorithm respects a total time budget (default 1000 s) and returns
the best sequence found as a plain Python list.  It consists of the
following stages for a series of lengths (large → small):

1. **Simple analytic families** – a few handcrafted shapes (uniform,
   linear, quadratic, power‑law) and a coarse sweep over a power‑law
   exponent.  The best of these is kept.

2. **Adam on the exact sub‑gradient of max‑convolution** – a few
   restarts from the candidates above.  The variables are represented
   in log‑softmax space to stay on the simplex.

3. **Differential Evolution (DE)** – a population‑based global search
   on the simplex (population size ≈ 30).  The DE loop is stopped
   early when the time budget is exhausted.

4. **(1+1) Evolution Strategy (ES)** – fine‑tuning with adaptive
   mutation strength.

5. **Linear interpolation with the uniform distribution** – a cheap
   “mix‑with‑uniform” post‑processing step that occasionally improves
   the peak‑to‑average ratio.

All heavy work (convolution, gradient) is performed with FFTs, so even
vectors of length 8192 are cheap to evaluate.  The code is fully
self‑contained, uses only NumPy and the standard library, and prints
progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – returns a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    s = e.sum()
    return e / s if s != 0 else np.full_like(x, 1.0 / x.size)


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returns a real vector.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence.
    The vector a is assumed to be a probability vector (sum = 1).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def grad_submax(a: np.ndarray):
    """
    Sub‑gradient of max(a∗a) w.r.t. a.
    Returns (gradient, current maximal convolution value).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick a random index among all maxima (break ties uniformly)
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    # for j such that 0 ≤ j < n and 0 ≤ k‑j < n we have
    #   ∂/∂a_j max = 2·a_{k‑j}
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def run_adam_opt(n: int, init_a: np.ndarray, deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the sub‑gradient of max(a∗a) while staying on the simplex.
    Returns (best_a, best_value) discovered before *deadline*.
    """
    # initialise logits from the probability vector
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥ 0, Σa = 1
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient of max(conv)
        val = 2.0 * n * max_c               # objective (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:                      # fallback (should not happen)
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


def run_de_opt(n: int,
               init_pop: list,
               deadline: float,
               popsize: int = 30,
               max_gens: int = 20_000) -> tuple[np.ndarray, float]:
    """
    Differential Evolution on the simplex.
    Returns (best_a, best_value) found before *deadline*.
    """
    # --------------------------------------------------------------
    # initialise population (probability vectors)
    # --------------------------------------------------------------
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    # evaluate initial population
    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8   # differential weight
    CR = 0.9  # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)               # enforce non‑negativity

            # binomial crossover (ensure at least one mutant component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # stop early if no improvement for a while
        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


def run_es_opt(n: int,
               init_a: np.ndarray,
               deadline: float,
               max_iter: int = 30_000) -> tuple[np.ndarray, float]:
    """
    (1+1) Evolution Strategy – Gaussian mutations with simple step‑size control.
    Returns the best vector discovered before *deadline*.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                         # success → keep
            sigma = min(sigma * 1.05, 0.2)     # slightly increase step size
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                     # decay on failure

        # reset when stuck
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
