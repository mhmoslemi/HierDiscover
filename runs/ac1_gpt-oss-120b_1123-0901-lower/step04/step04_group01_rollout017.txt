"""
Propose a candidate step‑function (non‑negative vector) that minimizes

    F(a) = 2·n·max(a*a) / (sum(a)**2)

The algorithm combines three ideas:

1. **Smooth‑max Adam optimisation** – we represent `a` by soft‑max
   logits, evaluate a differentiable approximation of the max via
   a log‑sum‑exp (temperature `alpha`) and update the logits with Adam.
   The temperature is increased during training so the optimisation
   first follows a smooth landscape and later concentrates on the true
   max.

2. **Simulated‑annealing local search** – starting from the best point
   found by Adam we perform many small mass‑transfer moves
   (move `δ` from index `i` to index `j`).  Moves that increase the
   objective are accepted with a probability that decays with a linear
   temperature schedule, allowing the search to escape shallow local
   minima.

3. **Multi‑scale restarts** – the procedure is run for several problem
   sizes (1024, 2048, 4096).  For each size we try a few restarts,
   warm‑starting from the current global best when possible.  Random
   Dirichlet initialisations are used as well.

All calculations are performed on the simplex (∑a = 1) so the
evaluate_sequence objective reduces to `2·n·max(conv)`.  The routine
returns the best sequence found before the time budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT‑based convolution utilities
# ----------------------------------------------------------------------
def next_pow2(x: int) -> int:
    """Small helper: next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using a pre‑computed FFT length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Core optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for at most ``budget_s`` seconds and
    returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    # give a tiny safety margin so we can finish cleanly
    deadline = time.time() + budget_s - 0.5

    best_seq = None          # numpy array with the best candidate
    best_val = float("inf")  # objective value of the best candidate

    # ------------------------------------------------------------------
    # Problem sizes – larger n gives a lower theoretical lower bound.
    # We explore a few sizes; the search for each size is limited by the
    # remaining time budget.
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = next_pow2(2 * n - 1)
        fft_len_corr = next_pow2(3 * n)          # ≥ (2n‑1)+n‑1

        # Number of independent restarts for the current length
        n_restarts = 3

        for restart in range(n_restarts):
            if time.time() > deadline:
                break

            # ------------------------------------------------------------------
            # Initialise logits (soft‑max parameters) for the Adam phase.
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                # warm start from the current global best
                a0 = best_seq.copy()
                a0 /= a0.sum()
                logits = np.log(a0 + 1e-15)
                print(f"[n={n} restart={restart}] warm start from global best")
            elif 'height_sequence_1' in globals():
                # use the provided reference sequence, resized by linear interpolation
                ref = np.array(height_sequence_1, dtype=np.float64)
                if ref.shape[0] != n:
                    x_old = np.linspace(0.0, 1.0, ref.shape[0])
                    x_new = np.linspace(0.0, 1.0, n)
                    ref = np.interp(x_new, x_old, ref)
                ref = np.maximum(ref, 0.0)
                if ref.sum() < 0.01:
                    a0 = np.random.dirichlet(np.ones(n))
                else:
                    a0 = ref / ref.sum()
                logits = np.log(a0 + 1e-15)
                print(f"[n={n} restart={restart}] start from height_sequence_1 (interpolated)")
            else:
                # completely random start
                a0 = np.random.dirichlet(np.ones(n))
                logits = np.log(a0 + 1e-15)
                print(f"[n={n} restart={restart}] random start")

            # ------------------------------------------------------------------
            # Adam optimiser state
            # ------------------------------------------------------------------
            m = np.zeros_like(logits)          # 1st moment
            v = np.zeros_like(logits)          # 2nd moment
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07                          # initial learning rate
            decay_interval = 5000
            decay_factor = 0.7

            # parameters for the smooth‑max temperature
            alpha_start = 5.0
            alpha_end = 200.0

            # iteration budget depends on n (larger n → fewer iterations)
            max_iters = 60000 if n <= 1024 else 40000 if n <= 2048 else 30000

            last_improve = time.time()
            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ---------- soft‑max → probability vector ----------
                max_log = np.max(logits)
                exp_log = np.exp(logits - max_log)    # numerical stability
                a = exp_log / exp_log.sum()           # Σa = 1, a_i ≥ 0

                # ---------- convolution ----------
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_conv = conv.max()

                # ---------- smooth‑max weights ----------
                # temperature grows linearly from alpha_start to alpha_end
                alpha = alpha_start + (alpha_end - alpha_start) * min(1.0, t / max_iters)
                # shift by max_conv to avoid overflow
                w = np.exp(alpha * (conv - max_conv))
                w_sum = w.sum()
                if w_sum == 0.0:                      # numerical safeguard
                    w = np.zeros_like(w)
                    w[conv.argmax()] = 1.0
                else:
                    w /= w_sum

                # ---------- gradient w.r.t. a (sub‑gradient of smooth‑max) ----------
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # ---------- back‑propagation through the soft‑max ----------
                inner = np.dot(a, grad_a)
                grad_logit = a * (grad_a - inner)

                # ---------- Adam update ----------
                m = beta1 * m + (1.0 - beta1) * grad_logit
                v = beta2 * v + (1.0 - beta2) * (grad_logit ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits = logits - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ---------- learning‑rate decay ----------
                if t % decay_interval == 0:
                    lr *= decay_factor

                # ---------- evaluate current objective (real max, not smoothed) ----------
                cur_val = 2.0 * n * max_conv      # because Σa = 1

                if cur_val < best_val - 1e-12:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t:05d}] new best {cur_val:.6f}")

                # ---------- stagnation shake ----------
                if time.time() - last_improve > 30.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={t}] shake (stagnation)")

            # ------------------------------------------------------------------
            # Simulated‑annealing local refinement starting from the best point
            # found by Adam for this restart.
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a_local = best_seq.copy()
                cur_conv = full_convolution(a_local, fft_len=fft_len_conv)
                cur_val = 2.0 * n * cur_conv.max()

                # SA schedule – linear cooling
                sa_iters = 20000 if n <= 1024 else 15000 if n <= 2048 else 12000
                temp_start = cur_val * 0.1 + 1e-6
                temp_end = cur_val * 0.001 + 1e-8

                for it in range(sa_iters):
                    if time.time() > deadline:
                        break
                    # linear temperature
                    temp = temp_start * (1 - it / sa_iters) + temp_end * (it / sa_iters)

                    # propose a tiny mass transfer
                    i, j = np.random.choice(n, 2, replace=False)
                    if a_local[i] <= 0.0:
                        continue
                    # move a random fraction (up to 2 % of the current mass at i)
                    delta = min(a_local[i], 0.02) * np.random.rand()
                    if delta <= 0.0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= delta
                    a_new[j] += delta

                    # compute the new objective (full convolution)
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    new_val = 2.0 * n * conv_new.max()
                    d_val = new_val - cur_val

                    # Metropolis accept/reject
                    if d_val < 0.0 or np.random.rand() < np.exp(-d_val / max(temp, 1e-12)):
                        a_local = a_new
                        cur_val = new_val
                        if new_val < best_val - 1e-12:
                            best_val = new_val
                            best_seq = a_new.copy()
                            print(f"[n={n} SA iter={it}] improved to {new_val:.6f}")

            # ------------------------------------------------------------------
            # Short random sampling phase – occasional pure‑random Dirichlet draws.
            # ------------------------------------------------------------------
            sample_time = min(8.0, (deadline - time.time()) * 0.05)   # at most ~8 s
            start_sampling = time.time()
            while time.time() - start_sampling < sample_time and time.time() < deadline:
                cand = np.random.dirichlet(np.ones(n))
                conv_cand = full_convolution(cand, fft_len=fft_len_conv)
                val_cand = 2.0 * n * conv_cand.max()
                if val_cand < best_val - 1e-12:
                    best_val = val_cand
                    best_seq = cand.copy()
                    print(f"[n={n} random sample] new best {val_cand:.6f}")

    # ----------------------------------------------------------------------
    # Fallback (unlikely) – uniform vector of length 2000 (objective = 2)
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2000
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
