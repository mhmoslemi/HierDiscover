"""
Propose a step‑function sequence minimizing F(a) = 2·n·max(a*a)/(sum a)^2.
Algorithm:
1. Work on several lengths (1024 … 16384).
2. For each length generate many initial shapes:
   - warm‑start from the best seen,
   - random Dirichlet,
   - product‑power‑law ( (i+1)*(n‑i) )^α with a fine grid α∈[‑5,‑0.1],
   - two‑exponential edge shapes exp(‑γ·i²)+exp(‑γ·(n‑1‑i)²) with γ on a log‑grid,
   - sum‑of‑inverse‑power edges (i+1)^‑p+(n‑i)^‑p with p on a grid.
3. Each candidate is refined with two phases:
   a) Adam on a smooth‑max (log‑sum‑exp) surrogate of the max‑convolution.
      The temperature λ grows exponentially, logits are updated via the
      soft‑max trick, and we run a number of iterations proportional to
      1024/n (more iterations for short vectors).
   b) A short exponentiated‑gradient sub‑gradient descent on the exact
      max‑convolution.
4. The current global best is updated after every exact evaluation.
5. After all lengths have been processed we take the best sequence,
   optionally up‑sample it to the next larger length and run a final
   long Adam run to polish the solution.
The function respects the time budget (default 1000 s) and returns a plain
list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT of given length."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]   (length = len(a)).
    This is the gradient of Σ_k w[k]·(a*a)[k] w.r.t. a.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _exact_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/(sum a)^2 (sum a ≈ 1)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


def _adam_phase(a0: np.ndarray, n: int,
                fft_len_conv: int,
                fft_len_corr: int,
                deadline: float,
                max_iters: int) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate. Returns best a and its exact value."""
    # Initialise logits = log a (soft‑max representation)
    logits = np.log(a0 + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature
    lam_factor = 1.02

    best_local_a = None
    best_local_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (exact max)
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()

        # shake if stagnating for a few seconds
        if time.time() - last_improve > 4.0:
            # small random jitter on logits
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    # Return the best a we have seen
    if best_local_a is None:
        best_local_a = a0
        best_local_val = _exact_objective(best_local_a, n, fft_len_conv)
    return best_local_a, best_local_val


def _subgrad_phase(a0: np.ndarray, n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on max(conv)."""
    a = a0 / a0.sum()
    eta = 0.6
    best_local_a = a.copy()
    best_local_val = _exact_objective(a, n, fft_len_conv)
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps non‑negativity)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:            # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _exact_objective(a, n, fft_len_conv)

        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 4.0:
            break

    return best_local_a, best_local_val


# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse list of initial probability vectors for length n."""
    candidates = []

    # Warm start from global best if same length
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.copy())

    # Random Dirichlet
    candidates.append(np.random.dirichlet(np.ones(n)))

    # Product‑power‑law ((i+1)*(n‑i))^α, α negative
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-5.0, -0.1, 15):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # Two‑exponential edge shapes exp(-γ·i²)+exp(-γ·(n‑1‑i)²)
    i = np.arange(n, dtype=np.float64)
    for gamma in np.logspace(-2, 1, 12):   # 0.01 … 10
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # Sum of inverse‑power edges (i+1)^‑p + (n‑i)^‑p
    for p in np.linspace(0.2, 5.0, 12):
        shape = np.power(idx, -p) + np.power(n - idx + 1.0, -p)
        shape = np.maximum(shape, 0.0)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            candidates.append(shape)

    # Linear ramp – high at edges, zero in the centre (triangular)
    ramp = np.minimum(idx, n - idx + 1.0).astype(np.float64)
    ramp = ramp / ramp.max()
    ramp = np.where(ramp > 0, ramp, 0.0)
    ramp /= ramp.sum()
    candidates.append(ramp)

    return candidates

    """

    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # Lengths to explore – we spend more time on the largest lengths.
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for convolution and correlation.
    fft_len_conv_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        fft_len_conv_cache[n] = _next_pow2(2 * n - 1)
        fft_len_corr_cache[n] = _next_pow2(3 * n)  # safe upper bound

    # ------------------------------------------------------------------
    # Main loop over lengths
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Optimising length n={n} ---")
        fft_len_conv = fft_len_conv_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # Build candidate pool
        candidates = _make_initial_candidates(n, best_seq)

        # Prioritise candidates that have already shown promise
        # (short‑list the first few for a quick test)
        quick_iters = max(2000, int(3000 * 1024 / n))

        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector
            a = a0 / a0.sum()
            # ---- Phase 1: Adam on smooth‑max ----
            max_iters = int(6000 * 1024 / n) + 500   # more for small n
            a_opt, val_opt = _adam_phase(a, n, fft_len_conv, fft_len_corr,
                                         deadline, max_iters)

            # Update global best if improved
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cand_id} Adam] new best {best_val:.6f}")

            # ---- Phase 2: exact sub‑gradient refinement ----
            # Only invoke if we have still enough time left
            remaining = deadline - time.time()
            if remaining < 2.0:

            a_opt2, val_opt2 = _subgrad_phase(a_opt, n, fft_len_conv,
                                              fft_len_corr, deadline,
                                              max_iters=2000)

            if val_opt2 < best_val:
                best_val = val_opt2
                best_seq = a_opt2.copy()
                print(f"[n={n} cand={cand_id} Subgrad] new best {best_val:.6f}")

            print(f"\n*** Upscaling best sequence {cur_n} → {target_n} for final polishing ***")
            # Simple linear interpolation to create a longer vector

            a_up = np.maximum(a_up, 0.0)
            a_up /= a_up.sum()
            # Run a longer Adam optimisation on the up‑scaled vector

            fft_len_corr = _next_pow2(3 * target_n)
            max_iters = int(15000 * 1024 / target_n) + 3000
            a_final, val_final = _adam_phase(a_up, target_n,
                                             fft_len_conv, fft_len_corr,
                                             deadline, max_iters)
            # One final sub‑gradient pass
            a_final, val_final = _subgrad_phase(a_final, target_n,
                                                fft_len_conv, fft_len_corr,
                                                deadline, max_iters=3000)
            if val_final < best_val:
                best_val = val_final
                best_seq = a_final.copy()

    # Fallback – uniform vector if nothing was found (should never happen)

    print(f"\n=== Finished. Best objective ≈ {best_val:.6f} (n={best_seq.shape[0]}) ===")

    return best_seq.tolist()

   • warm‑start from the current global best,  
   • random Dirichlet vectors,  
   • product‑power‑law shapes ((i+1)*(n‑i))^α with a fine grid α∈[‑5,‑0.1],  
   • two‑exponential edge shapes exp(‑γ·i²)+exp(‑γ·(n‑1‑i)²) with γ on a log‑grid,  
   • sums of inverse‑power edges (i+1)^‑p+(n‑i)^‑p (p‑grid),  
   • a simple linear‑ramp (triangular) profile.  
3. Every candidate is refined in two phases:  
   a) **Adam on a smooth‑max surrogate** of the max‑convolution.  
      The surrogate uses a log‑sum‑exp weighting with a temperature λ
      that grows geometrically.  The variables are kept on the simplex
      via the soft‑max trick.  The number of iterations scales with
      1024/n so that short vectors receive more updates.  
   b) **Exact sub‑gradient polishing** using an exponentiated‑gradient
      update on the true max‑convolution (active‑set gradient).  
4. The global best is updated after every exact evaluation.  
5. After the length loop, the best sequence (if any) is up‑sampled to the
   next larger length and polished once more with a long Adam run
   followed by a final sub‑gradient pass.  
6. The function respects the supplied time budget (default 1000 s) and
   returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length `fft_len`."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)]  (length = len(a)).
    This is the gradient of Σ_k w[k]·(a*a)[k] with respect to a.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = n - 1
    return corr[start:start + n]


def _exact_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv)/(sum a)² (sum a ≈ 1)."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _adam_phase(a0: np.ndarray, n: int,
                fft_len_conv: int,
                fft_len_corr: int,
                deadline: float,
                max_iters: int) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate.
    Returns the best vector seen and its exact objective value.
    """
    logits = np.log(a0 + 1e-15)          # soft‑max representation

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    lam = 0.5          # temperature for the log‑sum‑exp
    lam_factor = 1.02

    best_a = None
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # ---- soft‑max onto the simplex ----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ---- smooth‑max of the convolution ----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of the surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (max_c already computed)
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        # shake if stagnating for a few seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    if best_a is None:
        best_a = a0
        best_val = _exact_objective(best_a, n, fft_len_conv)

    return best_a, best_val


def _subgrad_phase(a0: np.ndarray, n: int,
                   fft_len_conv: int,
                   fft_len_corr: int,
                   deadline: float,
                   max_iters: int = 2000) -> tuple[np.ndarray, float]:
    """
    Short exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its exact value.
    """
    a = a0 / a0.sum()
    eta = 0.6
    best_a = a.copy()
    best_val = _exact_objective(a, n, fft_len_conv)
    last_improve = time.time()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                  # avoid collapse to zero
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        val = _exact_objective(a, n, fft_len_conv)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()

        if time.time() - last_improve > 4.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def _make_initial_candidates(n: int, best_seq: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a diverse list of starting probability vectors for a given length.
    """
    candidates = []

    # warm‑start from the global best (if the length matches)
    if best_seq is not None and best_seq.shape[0] == n:
        candidates.append(best_seq.copy())

    # pure random start
    candidates.append(np.random.dirichlet(np.ones(n)))

    idx = np.arange(n, dtype=np.float64) + 1.0               # 1 … n
    prod = idx * (n - idx + 1.0)                             # (i+1)*(n-i)

    # product‑power‑law ((i+1)*(n-i))^α, α negative
    for alpha in np.linspace(-5.0, -0.1, 15):
        shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
        if shape.sum() > 0:
            shape /= shape.sum()
            candidates.append(shape)

    # two‑exponential edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in np.logspace(-2, 1, 12):   # 0.01 … 10
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape /= shape.sum()
        candidates.append(shape)

    # sum of inverse‑power edges: (i+1)^‑p + (n-i)^‑p
    for p in np.linspace(0.2, 5.0, 12):
        shape = np.power(idx, -p) + np.power(n - idx + 1.0, -p)
        shape = np.maximum(shape, 0.0)
        s = shape.sum()
        if s > 0:
            shape /= s
            candidates.append(shape)

    # simple linear ramp (triangular) – high at edges, zero centre
    ramp = np.minimum(idx, n - idx + 1.0)
    ramp = ramp / ramp.max()
    ramp = np.where(ramp > 0, ramp, 0.0)
    ramp /= ramp.sum()
    candidates.append(ramp)

    return candidates


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence that minimises the evaluation
    function.  The search respects a wall‑clock budget (default 1000 s)
    and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = float("inf")

    # lengths to investigate – we invest a bit more time on the larger ones
    lengths = [1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT sizes
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        fft_len_conv[n] = _next_pow2(2 * n - 1)
        fft_len_corr[n] = _next_pow2(3 * n)   # safe upper bound

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # generate candidates
        candidates = _make_initial_candidates(n, best_seq)

        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            a0 = a0 / a0.sum()               # make it a probability vector

            # ----- Phase 1: Adam on smooth‑max -----
            max_iters = int(6000 * 1024 / n) + 500   # more iterations for short n
            a_adam, val_adam = _adam_phase(a0, n, conv_len, corr_len,
                                          deadline, max_iters)

            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[n={n} cand={cid} Adam] new best {best_val:.6f}")

            # ----- Phase 2: exact sub‑gradient polishing -----
            # run only if we still have a reasonable amount of time left
            if deadline - time.time() < 2.0:
                continue
            a_sub, val_sub = _subgrad_phase(a_adam, n, conv_len, corr_len,
                                            deadline, max_iters=2000)

            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[n={n} cand={cid} Subgrad] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final polishing: upscale the best vector (if possible) and run a long Adam pass
    if best_seq is not None and time.time() < deadline - 5.0:
        cur_n = best_seq.shape[0]
        bigger = [L for L in lengths if L > cur_n]
        if bigger:
            target_n = bigger[0]
            print(f"\n*** Upscaling {cur_n} → {target_n} for final polishing ***")
            # linear interpolation to the larger grid
            x_old = np.linspace(0.0, 1.0, cur_n)
            x_new = np.linspace(0.0, 1.0, target_n)
            a_big = np.interp(x_new, x_old, best_seq)
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()

            conv_len_big = _next_pow2(2 * target_n - 1)
            corr_len_big = _next_pow2(3 * target_n)

            max_iters = int(15000 * 1024 / target_n) + 3000
            a_big, val_big = _adam_phase(a_big, target_n,
                                         conv_len_big, corr_len_big,
                                         deadline, max_iters)

            a_big, val_big = _subgrad_phase(a_big, target_n,
                                            conv_len_big, corr_len_big,
                                            deadline, max_iters=3000)

            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Final] new best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _exact_objective(best_seq, fallback_n,
                                    _next_pow2(2 * fallback_n - 1))
        print(f"[Fallback] uniform sequence, value {best_val:.6f}")

    print(f"\n=== Finished. Best objective ≈ {best_val:.6f} (length {best_seq.shape[0]}) ===")
    # Return as a plain Python list
    return best_seq.tolist()
