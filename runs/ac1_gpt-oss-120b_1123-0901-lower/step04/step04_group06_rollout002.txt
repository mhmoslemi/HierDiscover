```
def project_simplex(v, s=1):
    # v: array, ensure nonnegative after projecting onto simplex
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, len(v)+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _optimise_adam_simplex(n, a0, target_sum, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    best_seq = a.copy()
    best_score = float('inf')
    iteration = 0
    alpha0 = 0.05
    while time.time() < deadline:
        # conv
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
        # compute weights
        tau = max(0.1 * max_val * (0.99**iteration), 1e-4 * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled - np.max(scaled))   # for stability
        w /= w.sum()
        # gradient
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)
        idx = (2 * n - 2) - np.arange(n)
        grad_a = 2.0 * conv2[idx]
        # Adam
        iteration += 1
        m = 0.9 * m + 0.1 * grad_a
        v = 0.999 * v + 0.001 * grad_a**2
        m_hat = m / (1 - 0.9**iteration)
        v_hat = v / (1 - 0.999**iteration)
        step = alpha0 * m_hat / (np.sqrt(v_hat) + 1e-8)
        a_new = a - step
        # symmetrize
        a_new = (a_new + a_new[::-1]) / 2.0
        # project
        a = _project_to_simplex(a_new, target_sum)
        # occasional random jitter
        if iteration % 200 == 0 and np.random.rand() < 0.3:
            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0)
            a = _project_to_simplex(a, target_sum)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The function reduces to `max(conv(a, a))` when we enforce
∑a = √(2 n).  The search therefore tries to flatten the
self‑convolution of a sequence.

Algorithmic ideas
-----------------
* **Simplex parametrisation** – the sequence is kept on the
  simplex (non‑negative, prescribed sum) by an Euclidean projection
  after each Adam update.  This avoids the soft‑max wrapper and gives
  cleaner gradients.
* **Symmetry** – after every update the sequence is symmetrised
  (`a ← (a + a[::-1])/2`).  A symmetric sequence yields a symmetric
  autocorrelation, which empirically lowers the maximal convolution
  value.
* **Smoothed max surrogate** – instead of the non‑differentiable `max`,
  we optimise a temperature‑controlled log‑sum‑exp of the convolution
  entries.  The temperature τ is exponentially cooled, so the optimiser
  gradually focuses on the true peak.
* **Adam optimiser** – adaptive first‑ and second‑moment estimates give
  robust steps even for large dimensions.
* **Multi‑restart + up‑sampling** – many short runs with random
  lengths (500 – 8000) explore the space; the best candidate is later
  up‑sampled (linear interpolation) to a longer length and refined
  further.
* **Local hill‑climbing** – after Adam finishes we perform a cheap
  random‑mass‑swap local search to squeeze out a few more improvements.

The routine respects a 1000 s budget, prints progress, and finally
returns the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution via FFT (real input)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real 1‑D arrays."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative rounding artefacts
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection of a vector onto the simplex {z ≥ 0, Σz = s}
# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Project v onto the simplex of non‑negative vectors with sum s."""
    if s <= 0:
        return np.zeros_like(v)
    # Duchi et al. (2008) algorithm
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, len(v) + 1) > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (with symmetry and smoothed‑max surrogate)
# ----------------------------------------------------------------------
def _optimise_adam_simplex(
    n: int,
    a0: np.ndarray,
    target_sum: float,
    run_deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Perform Adam optimisation of a sequence of length ``n`` while
    staying on the simplex and keeping the vector symmetric.
    Returns the best sequence encountered and its evaluation score.
    """
    a = a0.copy()
    # Adam moments
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_seq = a.copy()
    best_score = float("inf")
    iteration = 0

    # hyper‑parameters
    alpha = 0.07          # base learning rate
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8

    while time.time() < run_deadline:
        # --------------------------------------------------------------
        # convolution and max (exact objective for the current a)
        # --------------------------------------------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # --------------------------------------------------------------
        # smoothed‑max weights (log‑sum‑exp surrogate)
        # --------------------------------------------------------------
        # exponential cooling of the temperature τ
        tau = max(
            max_val * max(0.12 * (0.95 ** iteration), 0.001),
            1e-12,
        )
        scaled = (conv - max_val) / tau          # ≤ 0
        # stability: subtract max before exponentiation
        w = np.exp(scaled - np.max(scaled))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # --------------------------------------------------------------
        # gradient of the surrogate w.r.t. a
        # --------------------------------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)      # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)              # positions that give Σ a_j w_{i+j}
        grad_a = 2.0 * conv2[idx]

        # --------------------------------------------------------------
        # Adam update on the primal variables (a)
        # --------------------------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_a
        v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step

        # --------------------------------------------------------------
        # enforce symmetry and project back onto the simplex
        # --------------------------------------------------------------
        a_new = (a_new + a_new[::-1]) * 0.5
        a = _project_to_simplex(a_new, target_sum)

        # occasional random jitter to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 0.001
            a = np.maximum(a, 0.0)
            a = _project_to_simplex(a, target_sum)

        # slowly decay the base learning‑rate (helps final fine‑tuning)
        alpha *= 0.9995

    # final evaluation using the provided evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Optional cheap local refinement (random mass swaps)
# ----------------------------------------------------------------------
def _local_hill_climb(a: np.ndarray, budget: float) -> np.ndarray:
    """
    Perform a very cheap stochastic hill‑climb: repeatedly move a tiny
    amount of mass from a random index to another and keep the move
    if it improves the max‑convolution.
    """
    target_sum = a.sum()
    n = a.shape[0]
    start = time.time()
    best = a.copy()
    best_score = evaluate_sequence(best.tolist())

    while time.time() - start < budget:
        i, j = np.random.randint(0, n, size=2)
        if i == j or best[i] < 1e-12:
            continue
        delta = best[i] * np.random.uniform(1e-4, 1e-3)
        cand = best.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project onto the simplex (tiny numerical drift)
        cand = _project_to_simplex(cand, target_sum)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best = cand
    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step function that minimises the given
    evaluation function within the allotted time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main multi‑restart loop
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # pick a length – we favour larger lengths when we have time
        if remaining > 600:
            n = np.random.randint(3000, 8000 + 1)
        elif remaining > 200:
            n = np.random.randint(2000, 6000 + 1)
        else:
            n = np.random.randint(800, 3000 + 1)

        # ----------------------------------------------------------
        # initial shape (several families to diversify the search)
        # ----------------------------------------------------------
        shape_type = np.random.choice(["power", "tri", "exp", "rand"])
        idx = np.arange(1, n + 1, dtype=float)

        if shape_type == "power":
            p = np.random.uniform(0.2, 0.8)
            init = idx ** (-p)
        elif shape_type == "tri":
            half = (n + 1) // 2
            inc = np.arange(1, half + 1, dtype=float)
            if n % 2 == 0:
                dec = inc[::-1]
            else:
                dec = inc[-2::-1]
            init = np.concatenate([inc, dec])
        elif shape_type == "exp":
            tau = np.random.uniform(0.3, 2.0) * n
            init = np.exp(-idx / tau)
        else:  # random jittered uniform
            init = np.random.rand(n) + 1e-5

        # occasionally inject the current global best (cropped or padded)
        if best_seq is not None and np.random.rand() < 0.4:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        # ensure positivity and correct ℓ₁‑norm
        target_sum = np.sqrt(2.0 * n)
        init = np.maximum(init, 1e-12)
        init = init / init.sum() * target_sum

        # ------------------------------------------------------------------
        # Run the Adam‑on‑simplex optimiser for a modest time slice
        # ------------------------------------------------------------------
        slice_time = min(11.0, remaining * 0.6)
        run_deadline = min(deadline, time.time() + slice_time)

        seq_local, score_local = _optimise_adam_simplex(
            n, init, target_sum, run_deadline
        )

        # optional cheap local improvement (uses at most 0.8 s)
        if time.time() < deadline - 1.0:
            seq_local = _local_hill_climb(
                seq_local, budget=min(0.7, deadline - time.time())
            )
            score_local = evaluate_sequence(seq_local.tolist())

        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            if best_score < 1.503:                     # target achieved
                break

    # --------------------------------------------------------------
    # Final refinement on a longer up‑sampled version of the best
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:

        # up‑sample by a factor of ~1.4 but keep within a reasonable bound

        if n_up > n_best:
            # linear interpolation (numpy)
            x_old = np.linspace(0.0, 1.0, n_best)
            x_new = np.linspace(0.0, 1.0, n_up)
