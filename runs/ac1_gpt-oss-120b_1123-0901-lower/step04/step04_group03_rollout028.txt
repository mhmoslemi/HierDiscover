"""
Propose a step‑function (non‑negative vector a) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm consists of two phases

1. **LP flattening** – given the current a we compute M = max(a∗a) and
   solve the linear programme  

        maximise  Σ b_j
        subject   a∗b ≤ M   (element‑wise)
                   b ≥ 0

   The matrix a∗b is linear in the unknown vector b, therefore the
   constraints are linear.  The LP tries to put as much mass as possible
   into b while respecting the same upper bound M on the convolution.
   The solution b is normalised to a probability vector g (Σg = 1) and
   mixed with the present a (a ← (1‑t)·a + t·g) with a small step size.
   Repeating the LP step quickly reduces the peak of the convolution.

2. **Fine‑tuning** – after the LP phase a short (1+1)‑Evolution‑Strategy
   is applied.  Gaussian perturbations are accepted only when they
   improve the exact objective.  This removes the remaining tiny
   irregularities.

Both phases respect the global time budget (default 1000 s) and keep the
best vector discovered.  If ``cvxpy`` is unavailable the routine falls
back to the differential‑evolution/ES approach from the reference
implementation.
"""

import time
import numpy as np

# Optional import of cvxpy – used for the LP phase

    _CVXPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective used by evaluate_sequence."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Build the (constant‑a) convolution matrix C such that C @ b = a*b
# ----------------------------------------------------------------------
def build_conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Return a dense matrix C of shape (2·n‑1, n) with
        (C @ b)[k] = (a ∗ b)[k] .
    Column j of C contains a shifted copy of a, i.e.
    C[j : j + n, j] = a.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    C = np.zeros((rows, n), dtype=np.float64)
    for j in range(n):
        C[j:j + n, j] = a
    return C


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float,
               max_iter: int = 30_000) -> tuple[np.ndarray, float]:
    """Simple (1+1)‑ES on the simplex."""
    a = init_a.copy()
    best_val = compute_objective(a)
    best_a = a.copy()
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.clip(trial, 0.0, None)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial /= s

        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES] it={it:05d} → new best {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()
    return best_a, best_val

    return best_a.tolist()
