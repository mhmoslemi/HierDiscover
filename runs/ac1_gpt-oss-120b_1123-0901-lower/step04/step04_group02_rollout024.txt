def upscale_sequence(seq, new_n):
    old_n = len(seq)
    # simple linear interpolation via np.interp
    x_old = np.linspace(0,1,old_n)
    x_new = np.linspace(0,1,new_n)
    seq_new = np.interp(x_new, x_old, seq)
    seq_new = np.maximum(seq_new,0)
    seq_new /= seq_new.sum()
    return seq_new

"""
propose_candidate
-----------------
Searches a non‑negative step‑function (a probability vector) that
minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

where ``a*a`` is the linear convolution of ``a`` with itself.
The optimiser works in three stages:

1. **Coarse optimisation on a small size** – many random restarts,
   Adam updates on the simplex with the exact sub‑gradient of the
   current maximum convolution entry.

2. **Upscaling** – the best small‑size vector is linearly interpolated
   to a larger length (default 1024) and refined further with the same
   optimiser.

3. **Intensive local refinement** – pairwise mass transfers and small
   random “shakes’’ are performed until the time budget expires.

All operations stay on the probability simplex (non‑negative,
entries sum to one) using a Euclidean projection.
The routine respects the supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT.
    Returns the (2·len(a)-1)‑length result.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the cross‑correlation (w ★ a) used for the gradient of
    Σ_k w_k·(a*a)_k.
    Result has length len(a) and corresponds to the gradient of that
    linear form w.r.t. a (up to a factor 2, applied outside).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    Implements the O(n) algorithm of Duchi et al.
    """
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zero -> uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective value 2·n·max(conv)/sum(a)^2.
    ``a`` is assumed non‑negative; scaling does not matter.
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_b = conv.max()
    s = a.sum()
    if s < 1e-12:  # safeguard (should never happen)
        return np.inf
    return 2.0 * n * max_b / (s * s)


def _gradient_max_entry(a: np.ndarray, fft_len_corr: int) -> np.ndarray:
    """
    Sub‑gradient of the maximal convolution entry.
    Finds the index k where the convolution is maximal (any tie is fine)
    and returns the gradient of that entry (times 2, as required by the
    chain rule of a*a).
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    k = int(conv.argmax())
    w = np.zeros_like(conv)
    w[k] = 1.0
    grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
    return grad, conv.max()


def _upscale_vector(vec: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from len(vec) to new_n while preserving
    non‑negativity and renormalising to sum = 1.
    """
    old_n = len(vec)
    if old_n == new_n:
        return vec.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, vec)
    up = np.maximum(up, 0.0)
    up /= up.sum()
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises the evaluation function.
    The search respects ``budget_s`` seconds and prints progress.
    Returns the best sequence as a plain list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, sum == 1
    best_val = np.inf

    # ------------------------------------------------------------------
    # Helper to initialise a probability vector of length n.
    def _initial_vector(n, restart_idx):
        # warm‑start from global best if same length
        if restart_idx == 0 and best_seq is not None and best_seq.shape[0] == n:
            print("[restart 0] warm start from global best")
            return best_seq.copy()
        # use historic construction if present
        historic = globals().get('height_sequence_1')
        if restart_idx == 0 and historic is not None:
            cand = np.asarray(historic, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-6:
                return np.random.dirichlet(np.ones(n))
            return cand / s
        # a few structured random attempts
        if np.random.rand() < 0.2:
            # monotone shape – large at left, small at right
            v = np.random.dirichlet(np.ones(n))
            left = np.sort(v[: n // 2])[::-1]
            right = np.sort(v[n // 2 :])[::-1]
            cand = np.concatenate([left, right])
            cand = np.maximum(cand, 0.0)
            cand /= cand.sum()
            return cand
        # pure Dirichlet
        return np.random.dirichlet(np.ones(n))

    # ------------------------------------------------------------------
    # Phase 1 – optimise a modest size to obtain a good “shape’’.
    base_n = 256
    max_restarts_small = 30
    print(f"\n=== Phase 1: coarse optimisation on n={base_n} ===")
    for r in range(max_restarts_small):
        if time.time() > deadline:
            break
        a = _initial_vector(base_n, r)

        # Adam parameters on the simplex
        m = np.zeros_like(a)
        v_vec = np.zeros_like(a)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.15
        fft_len_corr = _next_pow2(3 * base_n)   # enough for (2n‑1)+n

        last_improve = time.time()
        max_iters = 6000
        t = 0
        while time.time() < deadline and t < max_iters:
            grad, cur_max = _gradient_max_entry(a, fft_len_corr)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v_vec / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a = _project_simplex(a - step)

            # Exact objective for monitoring
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[small n={base_n} restart={r} iter={t}] new best {val:.6f}")

            # Simple stagnation handling
            if time.time() - last_improve > 4.0:
                noise = (np.random.rand(base_n) - 0.5) * 1e-2
                a = _project_simplex(a + noise)
                last_improve = time.time()
                print(f"[small n={base_n} restart={r} iter={t}] shake")

    # ------------------------------------------------------------------
    # Phase 2 – upscale to the target length (default 1024) and refine.
    target_n = 1024
    if best_seq is None:
        # fallback uniform if Phase 1 never succeeded
        best_seq = np.full(target_n, 1.0 / target_n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"\n=== Phase 2: upscale to n={target_n} and refine ===")
    a = _upscale_vector(best_seq, target_n)
    # Adam state for the larger problem
    m = np.zeros_like(a)
    v_vec = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12
    fft_len_corr = _next_pow2(3 * target_n)
    last_improve = time.time()
    max_iters = 25000
    t = 0
    while time.time() < deadline and t < max_iters:
        grad, cur_max = _gradient_max_entry(a, fft_len_corr)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad
        v_vec = beta2 * v_vec + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v_vec / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a - step)

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[target n={target_n} iter={t}] new best {val:.6f}")

        if time.time() - last_improve > 5.0:
            noise = (np.random.rand(target_n) - 0.5) * 5e-3
            a = _project_simplex(a + noise)
            last_improve = time.time()
            print(f"[target n={target_n} iter={t}] shake")

    # ------------------------------------------------------------------
    # Phase 3 – intensive local random refinement on the current best.
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len = _next_pow2(2 * n - 1)
        print("\n=== Phase 3: intensive local refinement ===")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move a fraction of mass from i to j (up to 5 %)
            delta = a[i] * (np.random.rand() * 0.05)
            if delta <= 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)  # numerical safety
            val = _objective(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was ever found.
    if best_seq is None:
        n_fallback = 512
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain list as required by the external evaluator.
    return best_seq.tolist()
