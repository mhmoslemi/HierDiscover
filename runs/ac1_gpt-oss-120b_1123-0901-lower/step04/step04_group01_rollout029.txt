# ----------------------------------------------------------------------
# FFT‑based linear convolution and cross‑correlation (re‑used many times)
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT.  Length = 2·len(a)‑1."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

def generate_initial_candidates(n: int, rng: np.random.Generator):
    """Return a list of candidate probability vectors for the given length."""
    cand = []

    # 1) uniform
    cand.append(np.full(n, 1.0 / n))

    # 2) linear “mountain” (symmetric)
    idx = np.arange(n)
    lin = np.minimum(idx, n - 1 - idx) + 1.0
    cand.append(lin / lin.sum())

    # 3) quadratic mountain
    quad = (np.minimum(idx, n - 1 - idx) + 1.0) ** 2
    cand.append(quad / quad.sum())

    # 4) arcsine‑like (coefficients of (1‑x)^{-1/2})
    i = np.arange(n, dtype=np.float64)
    # central binomial coefficient / 4^i = comb(2i,i)/4**i
    # use logarithms for stability
    log_bc = (np.log(np.arange(1, 2 * n, 2)) + np.log(np.arange(2, 2 * n, 2))).cumsum()
    # directly compute via recurrence b0=1, b_{i+1}=b_i*(2*(2*i+1))/(i+2)
    b = np.empty(n, dtype=np.float64)
    b[0] = 1.0
    for k in range(1, n):
        b[k] = b[k - 1] * (2.0 * (2 * k - 1)) / (k + 1.0)
    cand.append(b / b.sum())

    # 5) Dirichlet draws with several concentration parameters
    for alpha in (0.2, 0.5, 0.8, 1.2, 2.0):
        cand.append(rng.dirichlet(np.full(n, alpha)))

    # 6) warm start from a previously discovered best sequence, if present
    if 'height_sequence_1' in globals():
        old = np.array(height_sequence_1, dtype=np.float64)
        if old.size >= n:
            tmp = old[:n]
        else:
            tmp = np.pad(old, (0, n - old.size))
        tmp = np.maximum(tmp, 0.0)
        if tmp.sum() > 0.01:
            cand.append(tmp / tmp.sum())

    return cand

            # -------- smooth‑max (soft‑max over convolution values) -----
            tau = tau_start + (tau_end - tau_start) * (t / max_iters)
            # stability: shift by max_c
            exp_w = np.exp(tau * (conv - max_c))
            w = exp_w / exp_w.sum()                    # Σw = 1

            # -------- gradient of smooth‑max wrt a (2·(w ★ a)) ---------
            grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

            # -------- back‑prop through soft‑max -----------------------
            inner = np.dot(a, grad_a)
            grad_x = a * (grad_a - inner)              # ∂L/∂x

            # -------- Adam update ------------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # -------- learning‑rate decay -----------------------------
            if t % decay_every == 0:
                lr *= 0.7

            # -------- shake if stuck ----------------------------------
            if time.time() - last_improve > 30.0:
                # tiny random jitter in the logits
                x += (rng.random(n) - 0.5) * 1e-2
                last_improve = time.time()
                print(
                    f"[n={n} restart={restart} iter={t}] shake due to stagnation"
                )

    # ------------------------------------------------------------------
    # Optional cheap greedy refinement (pairwise mass transfer)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_local = best_seq.copy()
        ft = full_convolution(a_local, fft_len=fft_len_conv)
        val_local = 2.0 * n * ft.max()

        for _ in range(3000):
            if time.time() > deadline:
                break
            i, j = rng.integers(0, n, size=2)
            if i == j:
                continue
            move = min(a_local[i], 0.01) * (rng.random() * 0.5 + 0.5)
            if move <= 0:
                continue
            a_new = a_local.copy()
            a_new[i] -= move
            a_new[j] += move
            ft_new = full_convolution(a_new, fft_len=fft_len_conv)
            val_new = 2.0 * n * ft_new.max()
            if val_new < val_local - 1e-12:
                a_local = a_new
                val_local = val_new
                if val_new < best_val:
                    best_val = val_new
                    best_seq = a_new.copy()
                print(f"[local refine] improved to {val_new:.6f}")

    return best_seq, best_val


# ----------------------------------------------------------------------
# Public entry point required by the competition framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best probability vector found.
    """
    rng = np.random.default_rng(seed)
    # give the optimiser a small safety margin so the interpreter can exit cleanly
    deadline = time.time() + budget_s - 0.5

    # problem sizes – bigger n give a lower theoretical lower bound
    lengths = [1024, 2048, 4096, 8192, 16384]

    global_best_seq = None
    global_best_val = float("inf")

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        # allocate a fraction of the remaining time to this size (roughly equal)
        remaining = deadline - time.time()
        # give at least 40 s to the largest sizes; otherwise split evenly
        allocate = max(40.0, remaining / (len(lengths) - lengths.index(n)))
        sub_deadline = time.time() + allocate

        seq, val = optimise_length(
            n=n,
            deadline=sub_deadline,
            rng=rng,
            max_restarts=8 if n <= 4096 else 6,
        )
        if seq is not None and val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"*** New overall best: {val:.6f} (length {n}) ***")

    # ------------------------------------------------------------------
    # Fallback (unlikely) – uniform vector of moderate length
    # ------------------------------------------------------------------
    if global_best_seq is None:
        n_fallback = 2048
        global_best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        global_best_val = 2.0
        print("Fallback to uniform vector.")

    print(f"\nFinal best value: {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
