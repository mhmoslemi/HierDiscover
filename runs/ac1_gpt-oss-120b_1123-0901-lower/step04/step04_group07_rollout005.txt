The goal is to output a non‑negative vector `a` (step‑function heights) that minimises  

\[
\text{score}(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

where `n = len(a)` and `*` denotes discrete convolution.  
If the vector is scaled so that  

\[
\sum a = \sqrt{2n},
\]

the score equals simply `max(a*a)`.  
The search therefore consists of two stages  

1. **Parameter sweep over classic window functions** – Hann, Blackman, Kaiser,
   Tukey, Gaussian, exponential, power‑law, etc.  Each window is generated,
   normalised to the required ℓ₁‑norm and evaluated with a fast FFT based
   convolution.  The best vector among all lengths (300 … 8000) is kept.

2. **Local refinement with a projected Adam optimiser** – starting from the
   best vector found in stage 1 we run a (soft‑max) Adam optimiser that works
   directly on the exact objective.  After each step the vector is projected
   onto the feasible set `{a_i ≥ 0,  Σa_i = √(2n)}`.  The optimiser stops when
   the time budget is exhausted or when no improvement has been seen for a
   while.

Both stages respect the 1000 s wall‑clock limit, keep a small safety margin,
and print progress information.  The implementation uses only NumPy and
(optionally) SciPy for a few window definitions; it contains **no closures**,
all helper functions are defined at the top level, and the required function
signature `propose_candidate` is provided.

```python
"""
candidate_generator.py

search for a non‑negative step‑function that minimises
evaluate_sequence(sequence). The algorithm combines a wide
parameter sweep over classic window shapes with a final
projected Adam optimisation.  It works for lengths up to 8000,
keeps the L1‑norm equal to sqrt(2·n), and respects the 1000 s
budget.
"""

from __future__ import annotations
import time
import math
from typing import List, Tuple, Iterable

import numpy as np

# ----------------------------------------------------------------------
# Helper functions for the objective (FFT based convolution)
# ----------------------------------------------------------------------


def _max_convolution(a: np.ndarray, Nfft: int = None) -> float:
    """Maximum value of the discrete autocorrelation a*a."""
    n = a.shape[0]
    if Nfft is None:
        Nfft = 1 << ((2 * n - 1).bit_length())  # next power‑of‑two
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Stable soft‑max of a convolution vector.
    Returns (weights, smooth_max) where weights sum to 1.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Gradient of the (soft‑max) max‑convolution.
    Implements the "reverse weighted convolution" needed by Adam.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    rev_full = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep only the relevant slice of length n
    rev = rev_full[: a.shape[0]]
    return rev


def _project_to_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a non‑negative vector onto the L1‑ball of radius target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.size)
    return a


# ----------------------------------------------------------------------
# Window generation (parameter sweep)
# ----------------------------------------------------------------------


def _tukey_window(N: int, alpha: float) -> np.ndarray:
    """Manual implementation of scipy.signal.tukey (avoids a hard dependency)."""
    if N == 0:
        return np.array([], dtype=float)
    if alpha <= 0:
        return np.ones(N)
    if alpha >= 1:
        return np.hanning(N)
    t = np.arange(N) / (N - 1)
    w = np.ones(N)
    first = t < alpha / 2
    w[first] = 0.5 * (1 + np.cos(2 * np.pi / alpha * (t[first] - alpha / 2)))
    last = t > 1 - alpha / 2
    w[last] = 0.5 * (1 + np.cos(2 * np.pi / alpha * (t[last] - 1 + alpha / 2)))
    return w


def _candidate_windows(n: int, target_sum: float) -> Iterable[np.ndarray]:
    """
    Yield a collection of candidate windows (already normalised to the required
    L1‑norm).  The set is deliberately diverse but kept reasonably small so that
    the full sweep fits well within the time budget.
    """
    # ----- deterministic cheap families -----
    # uniform
    yield np.full(n, target_sum / n)

    # Hann (Cosine) window
    hann = np.hanning(n)
    yield hann / hann.sum() * target_sum

    # Blackman window
    black = np.blackman(n)
    yield black / black.sum() * target_sum

    # Kaiser windows (beta = 0 … 12)
    for beta in np.linspace(0.0, 12.0, 7):
        kaiser = np.kaiser(n, beta)
        yield kaiser / kaiser.sum() * target_sum

    # Tukey windows (alpha = 0 … 1)
    for alpha in np.linspace(0.0, 1.0, 6):
        tuk = _tukey_window(n, alpha)
        yield tuk / tuk.sum() * target_sum

    # Gaussian windows (sigma as a fraction of half‑length)
    for sigma_factor in np.linspace(0.2, 1.0, 5):
        i = np.arange(n)
        centre = (n - 1) / 2.0
        sigma = sigma_factor * centre
        gauss = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        if gauss.sum() > 0:
            yield gauss / gauss.sum() * target_sum

    # Exponential decay (forward and reversed)
    for rate in np.linspace(0.001, 0.02, 6):
        exp_fwd = np.exp(-rate * np.arange(n))
        exp_rev = exp_fwd[::-1]
        for v in (exp_fwd, exp_rev):
            yield v / v.sum() * target_sum

    # Power‑law (i+1)^{-p}
    for p in [0.5, 1.0, 1.5, 2.0]:
        pw = (np.arange(1, n + 1, dtype=float) ** -p)
        yield pw / pw.sum() * target_sum

    # Small random perturbations of the uniform vector (3 examples)
    rng = np.random.default_rng(12345 + n)
    for _ in range(3):
        pert = np.full(n, target_sum / n)
        pert += rng.normal(scale=0.01 * target_sum / n, size=n)
        pert = np.maximum(pert, 0.0)
        pert = pert / pert.sum() * target_sum
        yield pert


# ----------------------------------------------------------------------
# Projected Adam optimiser (exact sub‑gradient when temperature is tiny)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
) -> Tuple[np.ndarray, float]:
    """
    Adam optimiser for the exact objective (max‑convolution after normalisation).
    Returns the best vector found during this run together with its exact score.
    """
    target_sum = math.sqrt(2.0 * n)

    # normalise the start vector
    a = _project_to_simplex(a0, target_sum)

    # Adam bookkeeping
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.04
    jitter_interval = 3000

    # temperature schedule
    tau_factor = 0.25          # initial τ as a fraction of the current max
    decay_rate = max_iters / 12.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    # current best (starts with the initial point)
    best_seq = a.copy()
    best_score = _max_convolution(a, Nfft)

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact autocorrelation -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & (soft‑)max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-10:
            # exact sub‑gradient: uniform weight on all active constraints
            mask = conv >= M - 1e-12
            w = mask.astype(float) / max(mask.sum(), 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution (gradient of smoothed max) -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam gradient step -----
        # The exact gradient of 2·n·max / (sum)^2 after normalisation
        # reduces to 4·n·rev / target_sum^2  - 4·n·Ltau·a / target_sum^3
        coeff = 4.0 * n / (target_sum * target_sum)
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam state update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = _project_to_simplex(a, target_sum)

        # occasional tiny jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project_to_simplex(a + jitter, target_sum)

        # ----- keep the best exact score seen so far -----
        cur_score = _max_convolution(a, Nfft)  # exact max after projection
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam n={n:5d} it={it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function that minimises ``evaluate_sequence``.
    The function runs a cheap exhaustive sweep over classic window families,
    then refines the best candidate with a projected Adam optimiser.
    The returned value is a plain Python list of non‑negative floats.
    """
    np.random.seed(seed)
    start_time = time.time()
    # Keep a small safety margin for the surrounding harness.
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Try to recover a previously discovered construction, if any.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    global_best_seq: np.ndarray | None = None
    global_best_score = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore (capped at 8000, never smaller than 300)
    if base is not None:
        n0 = base.shape[0]
    else:
        n0 = 2100
    length_factors = [1.0, 1.5, 2.0]                    # original, 1.5×, 2×
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in length_factors})

    # ------------------------------------------------------------------
    # Stage 1 – exhaustive sweep over window families
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)
        print(f"[search] exploring length {n}")

        # Include a linearly interpolated version of the old base vector,
        # if its size differs from ``n``.
        extra_candidates = []
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            extra_candidates.append(interp)

        # iterate over the generic families
        for cand in _candidate_windows(n, target_sum):
            # evaluate the exact objective (max convolution after normalisation)
            score = _max_convolution(cand, None)
            if score < global_best_score:
                global_best_score = score
                global_best_seq = cand.copy()
                print(f"[update] len={n:4d} new best = {global_best_score:.6f}")

        # add the interpolated old construction (if any)
        for cand in extra_candidates:
            score = _max_convolution(cand, None)
            if score < global_best_score:
                global_best_score = score
                global_best_seq = cand.copy()
                print(f"[update] (interpolated) len={n:4d} new best = {global_best_score:.6f}")

        # ------------------------------------------------------------------
        # Stage 2 – local refinement with Adam (only if time permits)
        if global_best_seq is not None and time.time() < deadline:
            # Prepare a start vector of exactly length ``n``
            if global_best_seq.shape[0] != n:
                # linear interpolation to the required length
                old = global_best_seq
                old_x = np.linspace(0.0, 1.0, old.shape[0])
                new_x = np.linspace(0.0, 1.0, n)
                start_vec = np.interp(new_x, old_x, old)
            else:
                start_vec = global_best_seq.copy()
            start_vec = _project_to_simplex(start_vec, target_sum)

            # Rough iteration budget: more steps for smaller n
            max_iters = int(1_800_000 * 2100 / n)
            max_iters = max(200_000, max_iters)   # at least 200 k steps

            print(f"[adam] length={n} max_iters≈{max_iters}")
            refined_seq, refined_score = _adam_optimize(
                a0=start_vec,
                n=n,
                deadline=deadline,
                max_iters=max_iters,
            )
            if refined_score < global_best_score:
                global_best_score = refined_score
                global_best_seq = refined_seq.copy()
                print(f"[global] after Adam: len={n} best = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely event that nothing was found
    if global_best_seq is None:
        n = max(300, min(n0, 8000))
        target_sum = math.sqrt(2.0 * n)
        global_best_seq = np.full(n, target_sum / n)
        global_best_score = _max_convolution(global_best_seq, None)
        print("[fallback] using uniform vector")

    elapsed = time.time() - start_time
    print(f"[done] best score = {global_best_score:.6f}, length = {global_best_seq.shape[0]}, time = {elapsed:.1f}s")
    return global_best_seq.tolist()
```
