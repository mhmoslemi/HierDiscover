"""
propose_candidate
-----------------

Search for a non‑negative step‑function (a probability vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)^2)

where ``a*a`` denotes the linear convolution of ``a`` with itself.
The algorithm is a multi‑scale projected Adam optimiser:

* start from several random Dirichlet vectors (or warm‑start from a
  previously found construction);
* run Adam updates on the simplex using the exact sub‑gradient of
  ``max(conv)`` (the gradient of the active convolution entries);
* periodically inject a small random “shake’’ to escape plateaus;
* after the main phase perform a cheap random‑local‑search that moves a
  tiny amount of mass between two coordinates;
* finally return the best sequence found as a plain Python list.

The routine respects a global time budget (default 1000 s).  It
automatically adapts the amount of work to the remaining time and
prints progress information.  All helper functions are top‑level as
required.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    # ``1 << (x-1).bit_length()`` works for any x ≥ 1.
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using an FFT of size ``fft_len``.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)],  i = 0 … n-1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0 , Σ_i x_i = 1}
    Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be one‑dimensional")
    n = v.shape[0]
    u = np.sort(v)[::-1]                 # descending
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):                 # all zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Return a list of non‑negative numbers (the step function heights) that
    yields a low value of the evaluation function.  The search uses a
    multi‑restart Adam optimiser on several problem sizes and finishes
    with a short random hill‑climbing refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None          # np.ndarray, always normalised to sum == 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # lengths to explore – we start modest and finish with a large instance.
    lengths = [1024, 2048, 4096, 8192, 16384, 32768, 65536]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT sizes (powers of two) – large enough for the operations we need.
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)      # sufficient for (2n‑1)+n

        # How many independent restarts per length – more for the cheap sizes.
        restarts = 6 if n <= 8192 else 4

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---------- initialise a (probability vector) ----------
            if r == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart] warm start from current global best")
            elif 'height_sequence_1' in globals():
                # use the historic construction if it exists; truncate/pad.
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from height_sequence_1")
            else:
                # Pure Dirichlet draw (uniform on the simplex).
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random Dirichlet start")

            # Adam state on the simplex.
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.20                     # relatively aggressive initial LR

            # Stopping criteria for the inner loop.
            t = 0
            max_iters = 25000 if n <= 8192 else 12000
            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # ----- compute convolution and locate the active set -----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # Boolean mask of entries that are (almost) maximal.
                # A tiny tolerance prevents missing entries due to round‑off.
                mask = conv >= max_c - 1e-12
                w = mask.astype(np.float64)

                # ----- sub‑gradient of max(conv) w.r.t. a -----
                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- Adam update (pre‑projection) -----
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- project back onto the simplex -----
                a = _project_simplex(a_new)

                # ----- simple learning‑rate decay (every few thousand iters) -----
                if t % 5000 == 0 and lr > 0.02:
                    lr *= 0.7

                # ----- monitor exact objective (sum is 1 by construction) -----
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best: {val:.6f}")

                # ----- stagnation handling – small random shake -----
                if time.time() - last_improve > 5.0:
                    perturb = (np.random.rand(n) - 0.5) * 1e-3
                    a = _project_simplex(a + perturb)
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # end of one restart

        # end of lengths loop

    # ------------------------------------------------------------------
    # Final random hill‑climbing refinement (if time permits)
    # -------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move up to 0.2 % of the total mass from i to j.
            delta = 0.002 * a[i] * np.random.rand()
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            conv = _full_convolution(a_prop, fft_len_conv)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = 2.0 * n_fallback * (1.0 / n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return as plain Python list.
    return best_seq.tolist()
