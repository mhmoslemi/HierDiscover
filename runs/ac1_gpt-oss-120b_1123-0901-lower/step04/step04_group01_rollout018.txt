import time, numpy as np

def full_convolution(a):
    n = a.shape[0]
    fft_len = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
    return conv

def cross_correlation(w, a):
    n = a.shape[0]
    rev_a = a[::-1]
    fft_len = 1 << int(np.ceil(np.log2(3*n)))
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n-1
    return corr[start:start+n]

def upsample(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n-1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    # ensure non-neg
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total < 1e-12:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq = new_seq / total
    return new_seq

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The implementation follows a multi‑scale optimisation:
* start from a small problem (n≈256) and progressively increase the length,
  up‑sampling the current best vector at each scale.
* each scale is optimised by Adam on the soft‑max logits of the vector.
* the non‑smooth “max” is replaced by a temperature‑controlled soft‑max
  (log‑sum‑exp) which yields a smooth gradient; the temperature is
  slowly annealed so the optimisation first sees a smooth landscape and
  later concentrates on the true maximum.
* occasional random jitter prevents stagnation and a cheap local swap
  refinement is performed after each scale.
* the best vector found over all scales (and all restarts) is returned.

The routine respects the 1000 s time budget and never performs any I/O.
"""

import time
import numpy as np

def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(seq) to new_n, preserving the L1‑norm."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.arange(old_n, dtype=np.float64)
    x_new = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    total = new_seq.sum()
    if total < 1e-12:
        new_seq[:] = 1.0 / new_n
    else:
        new_seq /= total
    return new_seq


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # best sequence seen so far (numpy array)
    best_val = float("inf")  # its objective value (using sum==1)

    # progressive problem sizes – larger n gives a lower theoretical bound
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # initialise a from the previous best (upsampled) or a fallback
        if best_seq is not None:
            a0 = upsample_sequence(best_seq, n)
        elif 'height_sequence_1' in globals():
            cand = np.array(height_sequence_1, dtype=np.float64)
            if cand.shape[0] >= n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            a0 = np.maximum(cand, 0.0)
            a0_sum = a0.sum()
            a0 = a0 / a0_sum if a0_sum > 1e-12 else np.full(n, 1.0 / n)
        else:
            a0 = np.random.dirichlet(np.ones(n))

        # work with logits (soft‑max) to enforce positivity & Σa=1
        x = np.log(a0 + 1e-15)

        # Adam state
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.05          # initial learning rate

        # pre‑compute FFT lengths used many times
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # optimisation loop parameters

        t = 0
