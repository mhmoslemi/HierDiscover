"""
Hybrid optimizer for the self‑convolution bound.

The routine consists of three stages:

1. **Stochastic Adam optimisation** – many short runs with diverse random
   initialisations on a smooth‑max surrogate of  max(a*a).  A soft‑max
   temperature λ is increased slowly (continuation) and a small random
   “shake’’ is applied when an iteration stalls.

2. **Fine‑tuning** – a longer Adam run with a large λ and a very small
   learning‑rate to converge to the true max.

3. **Linear‑program refinement** – given the current best vector f,
   solve the LP   maximise Σb   s.t.   f∗b ≤ M (where M = max(f∗f))
   and b≥0.  Rescale b to the original ℓ₁‑norm, mix it with f, and repeat.
   The LP step is cheap (a few iterations) and provably reduces the bound.

Optionally a tiny random hill‑climb is performed on the final vector.

The whole procedure respects a user supplied time budget (default 1000 s)
and returns the best sequence found as a plain Python list.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse


def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.
    The closed form is  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function (list of heights) that minimises
    ``evaluate_sequence``.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3  # small safety margin

    best_seq = None
    best_val = float("inf")
    restart = 0

    # ------------------------------------------------------------------
    # Phase 1 – many short Adam runs with diverse initialisations
    # ------------------------------------------------------------------
    lengths = [2000, 2500, 3000, 3500, 4000]  # reasonable trade‑off
    while time.time() < deadline:
        n = lengths[restart % len(lengths)]

        # ---- random initialisation families ---------------------------------
        mode = restart % 7
        if mode == 0:
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            r = np.random.uniform(0.2, 0.9)
            a0 = r ** np.arange(n, dtype=np.float64)
            a0 /= a0.sum()
        elif mode == 2:
            i = np.arange(1, n + 1, dtype=np.float64)
            a0 = np.sin(np.pi * i / (n + 1))
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()
        elif mode == 3:

    # ------------------------------------------------------------------
    # Optional tiny random hill‑climb on the final vector
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        a = np.array(best_seq)
        n = len(a)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        # simple temperature schedule
        step_scale = 0.02
        while time.time() < deadline - 0.2:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a small fraction of mass from i to j
            delta = step_scale * (np.random.rand() - 0.5) * a[i]
            if a[i] - delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv = _full_convolution(a_new, size=conv_fft_size)
            val = 2.0 * n * conv.max()
            if val < best_val:
                a = a_new
                best_seq = a_new.copy()
                best_val = val
                print(f"[Hill‑climb] new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)
        best_val = evaluate_sequence(best_seq.tolist())

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
