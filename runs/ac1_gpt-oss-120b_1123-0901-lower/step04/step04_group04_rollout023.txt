def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns the weight vector (summing to 1) and the smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Linear convolution p∗p using real FFTs."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(p: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """Compute rev_i = Σ_k w_k·p_{k‑i} efficiently via FFTs."""
    # FFT of p (zero‑padded)
    P = np.fft.rfft(p, n=Nfft)
    # FFT of reversed weight vector
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    rev_full = np.fft.irfft(W_rev * P, n=Nfft)
    rev_full = rev_full[: len(w) + p.shape[0] - 1]

    # Pick the slice that corresponds to the required indices
    start = 2 * p.shape[0] - 2
    rev = rev_full[start - np.arange(p.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """Smooth max Lτ and its gradient w.r.t. logits y."""
    # soft‑max to obtain a probability vector
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, Ltau = _softmax_weights(conv, tau)

    # gradient of Lτ w.r.t. p: 2·rev
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # chain rule through the soft‑max
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y

            eps = 1e-8
            for _ in range(200):
                L, g = _objective_and_grad(y, tau, Nfft)
                m = beta1 * m + (1 - beta1) * g
                v = beta2 * v + (1 - beta2) * (g ** 2)
                m_hat = m / (1 - beta1)
                v_hat = v / (1 - beta2)
                y -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- evaluate exact objective ------------------------------------
        # probability vector
        exp_y = np.exp(y - np.max(y))
        p = exp_y / exp_y.sum()
        # un‑scaled sequence (still sums to ``target_sum``)
        a = p * target_sum

        # fast exact evaluation (no security checks)
        conv = _conv_full(p, Nfft)          # conv of probabilities
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # identical to evaluate_sequence

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] τ={tau:.2e} score={best_score:.6f}")

        # ---- temperature schedule & occasional jitter --------------------
        tau *= tau_decay
        if np.random.rand() < 0.12:            # jitter to escape shallow minima
            y += np.random.randn(y.size) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _evaluate_fast(seq: np.ndarray, Nfft: int) -> float:
    """Exact objective without the heavy security checks of ``evaluate_sequence``."""
    s = seq.sum()
    if s < 0.01:                     # same guard used in the original function
        return np.inf
    A = np.fft.rfft(seq, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * seq.size - 1]
    max_b = conv.max()
    return 2.0 * seq.size * max_b / (s * s)


def _simulated_annealing(seq: np.ndarray,
                         score: float,
                         Nfft: int,
                         deadline: float) -> (np.ndarray, float):
    """
    Simple coordinate‑descent simulated annealing that preserves the total sum.
    Works on the original (scaled) sequence.
    """
    n = seq.size
    cur_seq = seq.copy()
    cur_score = score
    best_seq = cur_seq.copy()
    best_score = cur_score

    rng = np.random.default_rng()
    # initial temperature – a fraction of the current score
    temp = max(1.0, cur_score) * 0.2
    min_temp = 1e-4
    decay = 0.9995

    # typical step size is a small fraction of the mean entry
    step_scale = max(cur_seq.mean(), 1e-6) * 0.1

    while time.time() < deadline and temp > min_temp:
        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue

        # amount to transfer from i to j (never exceeds the source entry)
        max_transfer = min(cur_seq[i], step_scale)
        if max_transfer <= 0.0:
            continue
        delta = rng.uniform(0.0, max_transfer)

        new_seq = cur_seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta

        new_score = _evaluate_fast(new_seq, Nfft)

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            prob = math.exp(-(new_score - cur_score) / (temp + 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_seq = new_seq
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = new_seq.copy()
                print(f"[SA improve] score={best_score:.6f}")

        # geometric cooling
        temp *= decay

        # occasional global jitter (small random perturbation, re‑projected onto simplex)
        if rng.random() < 0.01:
            perturb = rng.normal(scale=step_scale * 0.05, size=n)
            tmp = cur_seq + perturb
            tmp = np.maximum(0.0, tmp)
            # re‑scale to keep the sum unchanged
            tmp *= cur_seq.sum() / (tmp.sum() + 1e-12)
            cur_seq = tmp
            cur_score = _evaluate_fast(cur_seq, Nfft)

    return best_seq, best_score


def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the allotted time budget.

    The function follows the three‑stage strategy described in the module
    docstring.  The returned list contains the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Load a starting construction if available, otherwise fall back to uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:                                   # pragma: no cover
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required L1‑norm (the optimiser works with probabilities).
    base = base * (target_sum / base.sum())
    p0 = base / target_sum
    eps_log = 1e-12
    y0 = np.log(p0 + eps_log)

    # Record the best sequence found so far.
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # FFT size – next power‑of‑two ≥ 2·n‑1 (used throughout the run).
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Prepare several diverse starts.
    starts = []

    # 1) the known construction
    starts.append(y0)

    # 2) random Dirichlet (exponential samples)
    rand_exp = np.random.exponential(scale=1.0, size=n)
    starts.append(np.log(rand_exp + eps_log))

    # 3) uniform logits (all zeros ⇒ uniform probabilities)
    starts.append(np.zeros(n, dtype=float))

    # 4) a perturbed version of the known solution
    starts.append(y0 + np.random.randn(n) * 0.05)

    # ------------------------------------------------------------------
    # Run the L‑BFGS optimisation from each start.
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run] start {idx + 1}/{len(starts)}")

        # Initialise the temperature from the current smooth max.
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Fine‑tune the best solution with simulated annealing.
    if time.time() < deadline:
        best_seq, best_score = _simulated_annealing(
            seq=best_seq,
            score=best_score,
            Nfft=Nfft,
            deadline=deadline,
        )

    # Final sanity check using the official evaluator.
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
