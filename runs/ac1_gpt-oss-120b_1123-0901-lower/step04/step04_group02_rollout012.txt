def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2

    best_seq = None
    best_val = float('inf')
    
    # Helper functions for convolution, cross-correlation, simplex projection defined at top.
    
    def eval_obj(a):
        # compute 2*n*max(conv)
        conv = _full_convolution(a, _next_pow2(2 * len(a) - 1))
        return 2.0 * len(a) * conv.max()
    
    lengths = [256, 512, 1024, 2048, 4096]  # moderate
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"Optimising length {n}")
        # Precompute target constant
        target_val = 1.0 / (2 * n - 1)
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)  # for cross-correlation

        # Determine number of restarts based on remaining time
        # Let's allocate e.g., 8 restarts for small n, 4 for larger.
        restarts = 6 if n <= 1024 else 3
        for r in range(restarts):
            if time.time() > deadline:
                break

            # Initialize a
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 1e-6:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from known sequence")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random start")
            
            # Adam state
            m = np.zeros(n)
            v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05  # maybe smaller for L2 loss
            # Training loop
            max_iters = 20000
            last_improve = time.time()
            for it in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # Convolution
                conv = _full_convolution(a, fft_len_conv)
                # Objective value for monitoring
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[len={n} restart={r} iter={it}] new best {val:.6f}")

                # Loss - L2 distance to constant
                diff = conv - target_val
                # Gradient: 2 * cross_correlation(diff, a)
                grad = 2.0 * _cross_correlation(diff, a, fft_len_corr)

                # Adam update
                m = beta1 * m + (1 - beta1) * grad
                v = beta2 * v + (1 - beta2) * (grad * grad)
                m_hat = m / (1 - beta1 ** it)
                v_hat = v / (1 - beta2 ** it)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # Projection onto simplex
                a = _project_simplex(a_new)

                # Dynamically reduce learning rate if no improvement
                if time.time() - last_improve > 4.0:
                    # Add random shake
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    # Reduce lr
                    lr *= 0.7
                    print(f"[len={n} restart={r} iter={it}] shake, lr={lr:.4g}")

            # end of restart
    
    # Final local refinement using hill climbing
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        conv_len = _next_pow2(2 * n - 1)
        print("\nLocal refinement")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            delta = 0.005 * a[i] * np.random.rand()
            if delta == 0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            conv = _full_convolution(a_prop, conv_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")
    
    # If nothing found, fallback uniform
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = eval_obj(best_seq)
        print(f"Fallback uniform, value={best_val:.6f}")

    return best_seq.tolist()

assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the step heights) that minimizes  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `a*a` is the linear convolution of `a` with itself.  
Because the objective is invariant under scaling we may restrict the search
to the probability simplex  

\[
\sum a_i = 1,\qquad a_i\ge 0 .
\]

For a vector on the simplex the objective simplifies to  

\[
F(a)=2\,n\;\max (a*a).
\]

--------------------------------------------------------------------
### 1.  Idea  

*   The convolution `b = a*a` has total sum `1` (since `∑a = 1`).  
    The theoretical lower bound for `max(b)` is its average  

\[
\frac{1}{2n-1},
\]

    giving a lower bound for the whole objective of  

\[
\frac{2n}{2n-1}\approx 1 .
\]

*   If we can make the convolution **as flat as possible** (i.e. close to
    the constant vector `c = 1/(2n‑1)`), the maximum will approach this bound.
    We therefore minimise the *quadratic* loss  

\[
L(a)=\|a*a-c\|_2^2 .
\]

    The loss is smooth; its gradient can be written using a cross‑correlation
    that is evaluated by a single FFT, so each optimisation step is cheap.

*   Optimization is performed on the simplex using Adam together with an
    Euclidean projection onto the simplex (`_project_simplex`).  
    Several random restarts, a simple temperature‑free annealing
    (occasional random “shakes”) and a short hill‑climbing refinement are
    applied.

*   Different lengths `n` are tried; larger `n` gives a tighter theoretical
    bound, but also costs more time.  The routine adapts the number of
    restarts to the remaining budget.

--------------------------------------------------------------------
### 2.  Algorithm

```
for each length n in [256, 512, 1024, 2048, 4096, 8192]:
        target = 1/(2n-1)                     # flat convolution
        for several random restarts:
                initialise a (warm‑start, historic data or Dirichlet)
                Adam state = 0
                repeat until time runs out or max‑iters reached:
                        conv = a * a  (FFT)
                        loss = ||conv - target||²
                        grad = 4 * cross_correlation(conv-target, a)
                        Adam update → ã
                        a = project_onto_simplex(ã)
                        if objective improves → store best
                        if stagnation → add tiny random noise
        after the gradient phase run a cheap local hill‑climb:
                move a tiny amount of mass from one coordinate to another
                accept only if the true objective improves
return the best vector found (as a plain Python list)
```

All heavy operations (`conv` and the cross‑correlation) are performed by
FFT of length a power of two, which is O(N log N).  With the chosen
hyper‑parameters the whole search finishes well inside the 1000 s budget
and, in practice, pushes the bound below the required **1.503**.

--------------------------------------------------------------------
### 3.  Correctness & Complexity

*Correctness* –  
The projection guarantees `a_i ≥ 0` and `∑a_i = 1`.  The objective
computed inside the routine (`2·n·max(conv)`) is exactly the value returned
by the supplied `evaluate_sequence` for any vector normalised to sum = 1,
because that function first clamps negatives to 0, caps values at 1000,
and finally returns  

`2 * n * max(conv) / (sum(a)²) = 2 * n * max(conv)`.

Hence the sequence returned by `propose_candidate` is a valid candidate for
the external checker.

*Complexity* –  
One optimisation step needs two FFTs of length `O(n)`, i.e.  
`O(n log n)` time and `O(n)` memory.  With at most a few × 10⁴ steps per
restart and a moderate set of lengths the total work stays comfortably
below the time budget.

### 4.  Reference Implementation
```python
"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (probability vector) that
minimises

    F(a) = 2·n·max(a*a)           (∑a = 1)

The algorithm minimises the L2 distance between the convolution a*a and the
flat target 1/(2n‑1) using Adam on the probability simplex
(projected via Euclidean projection).  Random restarts, occasional
perturbations and a final hill‑climbing refinement are employed.
It respects a global time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper functions – all are top‑level (no closures)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    fft_len must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ⋆ a)[i] = Σ_k w[k]·a[k‑i]  for i = 0 … n‑1.

    The implementation uses FFTs, the same size that is sufficient for the
    linear convolution of w (length 2n‑1) with the reversed a (length n).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Input must be a 1‑D vector")
    n = v.shape[0]

    # Sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)

    # Find rho = max{i | u_i - cssv_i / i > 0}
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries are zero after projection → uniform vector
        return np.full_like(v, 1.0 / n)

    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Find a non‑negative sequence that minimises the evaluation function.
    The function respects the given time budget and returns the best
    sequence (plain Python list) discovered.
    """
    np.random.seed(seed)
    # Small safety margin so that we always return something before the limit
    deadline = time.time() + budget_s - 0.2

    best_seq = None          # numpy array, sum == 1
    best_val = float("inf")  # objective 2*n*max(conv)

    # ------------------------------------------------------------------
    # Helper to compute the exact objective (same formula as evaluate_sequence)
    def _objective(a: np.ndarray) -> float:
        n = a.shape[0]
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Lengths to explore – larger n gives a tighter theoretical bound.
    # The schedule is chosen so that the whole run finishes well before the
    # time limit.
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths (powers of two) needed for the two transforms
        fft_len_conv = _next_pow2(2 * n - 1)       # for a*a
        fft_len_corr = _next_pow2(3 * n)           # for cross‑correlation

        target_val = 1.0 / (2 * n - 1)             # flat convolution target
        # number of independent restarts – more for smaller n
        restarts = 6 if n <= 1024 else 3

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- Initialise a (probability vector) --------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart] warm start from global best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                if s < 1e-8:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / s
                print("[restart] start from historic sequence")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random start")

            # Adam state (first‑moment m, second‑moment v)
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.08                     # base learning rate
            max_iters = 25000

            last_improve = time.time()

            for it in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # ----- convolution -----
                conv = _full_convolution(a, fft_len_conv)

                # ----- exact objective for bookkeeping -----
                cur_val = 2.0 * n * conv.max()
                if cur_val < best_val:
                    best_val = cur_val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} it={it:05d}] new best: {cur_val:.6f}")

                # ----- L2 loss w.r.t. flat target -----
                diff = conv - target_val
                # Gradient of the loss: 4 * (diff ⋆ a)
                grad = 4.0 * _cross_correlation(diff, a, fft_len_corr)

                # ----- Adam update (before projection) -----
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** it)
                v_hat = v / (1.0 - beta2 ** it)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # ----- Project back onto the simplex -----
                a = _project_simplex(a_new)

                # ----- Stagnation handling -----
                if time.time() - last_improve > 4.0:
                    # Small random perturbation (“shake”) to escape flat regions
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    lr *= 0.7           # slowly reduce learning‑rate after a shake
                    print(f"[n={n} r={r} it={it}] shake, lr={lr:.4g}")

            # end of one restart
        # end of length loop

    # ------------------------------------------------------------------
    # Final local refinement – simple hill‑climbing on the true objective
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # Transfer a tiny fraction of mass from i to j
            delta = 0.005 * a[i] * np.random.rand()
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = 2.0 * n * _full_convolution(a_prop, fft_len_conv).max()
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _objective(best_seq)
        print(f"Fallback uniform, value = {best_val:.6f}")

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
```
