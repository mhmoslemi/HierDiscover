def grad_smooth_logits(x, p):
    a = softmax(x)
    conv = full_convolution(a)
    conv_p = conv ** p
    sum_p = conv_p.sum()
    if sum_p <= 0:
        grad_x = np.zeros_like(x)
        return grad_x, conv.max()
    factor = sum_p ** (1.0/p - 1.0)
    w = p * (conv ** (p-1))
    # compute conv2 = np.convolve(a, w[::-1])
    # using FFT for speed
    # Choose L = next_pow2 (3*n -2)
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(3*n - 2)))
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w[::-1], L)
    conv2 = np.fft.irfft(fa * fw, L)[: 3*n - 2]
    # slice indices:
    slice_idx = (2*n - 2) - np.arange(n)
    s = conv2[slice_idx]
    grad_a = 4.0 * n * factor * s
    # backprop through softmax
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)
    return grad_x, conv.max()

def run_hc_opt(n, init_a, deadline, steps=100000):
    a = init_a.copy()
    cur_val = compute_objective(a)
    best_a = a.copy()
    best_val = cur_val
    last_improve = time.time()
    it = 0
    while time.time() < deadline and it < steps:
        it += 1
        # pick two indices i,j distinct
        i = np.random.randint(n)
        j = np.random.randint(n-1)
        if j >= i:
            j += 1
        # propose moving small delta mass from i to j (or vice versa)
        # Ensure a_i >= delta
        delta = np.random.exponential(scale=0.01)  # some small amount
        # random direction: move from i to j or j to i
        if np.random.rand() < 0.5:
            # i→j
            move = min(delta, a[i])
            new_a = a.copy()
            new_a[i] -= move
            new_a[j] += move
        else:
            # j→i
            move = min(delta, a[j])
            new_a = a.copy()
            new_a[j] -= move
            new_a[i] += move
        # renormalize? Since we moved mass preserving total sum, sum unchanged; we keep as is.
        # However ensure no negative: done.
        new_val = compute_objective(new_a)
        if new_val < cur_val:
            a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()
                last_improve = time.time()
                print(f"[HC n={n} it={it}] new best {best_val:.6f}")
        # optional occasional perturb full: random Dirichlet mixture if stagnate
        if time.time() - last_improve > 10.0:
            # restart to best and jitter
            a = best_a.copy()
            # add Gaussian jitter then renorm
            jitter = np.random.randn(n) * 0.01
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0:
                a /= s
            else:
                a = np.full(n, 1.0 / n)
            cur_val = compute_objective(a)
            last_improve = time.time()
    return best_a, best_val

def run_adam_multi_phase(n, init_a, deadline):
    # allocate fractions of time
    total_time = deadline - time.time()
    if total_time <= 0:
        return compute_objective(init_a), init_a
    # allocate time per smoothing stage (including submax)
    p_vals = [4, 8, 16, 32, 64]
    n_phases = len(p_vals) + 1
    time_per = total_time / n_phases
    best_a = init_a.copy()
    best_val = compute_objective(best_a)
    # smoothing phases
    for p in p_vals:
        if time.time() >= deadline:
            break
        sub_dead = min(deadline, time.time() + time_per)
        a_opt, val_opt = run_adam_smooth_phase(n, best_a, sub_dead, p)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
    # final submax phase
    if time.time() < deadline:
        sub_dead = deadline
        a_opt, val_opt = run_adam_submax_phase(n, best_a, sub_dead)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
    return best_a, best_val

Now produce final message with code.assistantfinal**Solution Overview**

The goal is to find a non‑negative vector `a` (the heights of a step function) that minimises  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2},
\]

where `*` denotes the linear convolution.  
Because the objective is *scale‑invariant* we may restrict the search to the simplex
\(\sum_i a_i = 1,\;a_i\ge0\).  
Thus the problem is equivalent to minimising the maximum entry of the self‑convolution.

The algorithm below combines three complementary optimisation stages, each of which
is aware of the remaining time budget (`budget_s` ≈ 1000 s):

1. **Smooth Adam optimisation** – the non‑smooth `max` is replaced by an
   \(p\)‑norm (`p` = 4, 8, 16, 32, 64).  The gradient of the smooth surrogate is
   computed analytically, allowing fast Adam updates on the simplex (logits → softmax).

2. **Exact Adam (sub‑gradient)** – after the smoothing schedule we finish with the
   true sub‑gradient of the `max` (as in the reference code), to polish the solution.

3. **Differential‑Evolution (DE)** – a global population‑based search on the simplex;
   the best individual from step 1 is used as a warm‑start.

4. **Local hill‑climbing (HC)** – small random mass transfers between coordinates,
   accepted only if they improve the true objective.  This fine‑tunes the final vector.

The routine works for any length; we start with the most promising size (`n=1024`)
and try larger lengths only if time allows.  All helper functions are pure,
top‑level, and contain no closures.

```python
"""Step‑function optimisation.

Searches for a non‑negative sequence a (∑a = 1) that minimises
    2·n·max(a*a) / (∑a)²

The algorithm uses:
  1. Adam on a smooth p‑norm surrogate for the maximum (p = 4,8,16,32,64).
  2. Adam on the exact sub‑gradient of max (final polishing).
  3. Differential‑Evolution on the simplex (global search).
  4. Simple (1+1) hill‑climbing with random mass transfers (fine‑tuning).

All phases respect the overall time budget; the best sequence found is returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic vector utilities
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def next_pow2(x: int) -> int:
    """Next power of two (≥ x)."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence (sum(a) is assumed >0).
    Returns 2·n·max(a*a) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    return 2.0 * n * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Gradient of the smooth surrogate  (p‑norm of the convolution)
# ----------------------------------------------------------------------
def grad_smooth_logits(x: np.ndarray, p: int) -> tuple[np.ndarray, float]:
    """
    Gradient of   2·n·(∑ conv^p)^{1/p}   w.r.t. the logits x (softmax(x)=a).
    Returns (grad_x, max(conv)).
    """
    a = softmax(x)
    n = a.shape[0]

    # convolution a*a
    conv = full_convolution(a)                     # length 2n‑1
    max_c = conv.max()

    # p‑norm surrogate
    conv_p = conv ** p
    sum_p = conv_p.sum()
    if sum_p <= 0.0:                               # should not happen
        grad_a = np.zeros_like(a)
        grad_x = np.zeros_like(x)
        return grad_x, max_c

    factor = sum_p ** (1.0 / p - 1.0)               # (∑c^p)^{1/p-1}
    w = p * (conv ** (p - 1))                      # weight for each conv entry

    # We need g_j = Σ_k w_k ∂conv_k/∂a_j = 2 Σ_i a_i w_{i+j}
    # This is a convolution of a with the reversed w.
    w_rev = w[::-1]
    L = next_pow2(3 * n - 2)
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w_rev, L)
    conv2 = np.fft.irfft(fa * fw, L)[: 3 * n - 2]   # a * w_rev

    # slice that corresponds to Σ_i a_i w_{i+j}
    s = conv2[(2 * n - 2) - np.arange(n)]          # length n

    # gradient w.r.t. a (including the outer factor 2·n)
    grad_a = 4.0 * n * factor * s                  # 2 (from ∂conv) × 2n (outer)

    # back‑propagation through softmax
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)
    return grad_x, max_c


# ----------------------------------------------------------------------
# Gradient of the true max (sub‑gradient) – same as the reference code
# ----------------------------------------------------------------------
def grad_submax(a: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient of max(a*a) w.r.t. a.
    Returns (grad_a, max(conv)).
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)                     # pick a maximal index

    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


def grad_submax_logits(x: np.ndarray) -> tuple[np.ndarray, float]:
    """Sub‑gradient w.r.t. logits (softmax parametrisation)."""
    a = softmax(x)
    grad_a, max_c = grad_submax(a)
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)
    return grad_x, max_c


# ----------------------------------------------------------------------
# Adam optimiser – smooth phases
# ----------------------------------------------------------------------
def run_adam_smooth_phase(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    p: int,
    lr: float = 0.05,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth p‑norm surrogate.
    Returns (best_a, best_value) found before *deadline*.
    """
    x = np.log(init_a + 1e-15)                     # logits
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        grad_x, max_c = grad_smooth_logits(x, p)
        # because a sums to 1
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam‑smooth p={p} iter={t:05d}] new best {val:.6f}")

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnant
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam‑smooth p={p} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)
    return best_a, best_val


def run_adam_submax_phase(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    lr: float = 0.05,
    max_iters: int = 50000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the exact sub‑gradient of max(a*a).
    Returns (best_a, best_value) found before *deadline*.
    """
    x = np.log(init_a + 1e-15)
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)
        grad_x, max_c = grad_submax_logits(x)
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam‑submax iter={t:05d}] new best {val:.6f}")

        t += 1
        m = beta1 * m + (1 - beta1) * grad_x
        v = beta2 * v + (1 - beta2) * (grad_x ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)

        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam‑submax iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)
    return best_a, best_val


def run_adam_multi_phase(
    n: int,
    init_a: np.ndarray,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Cascade of smooth Adam stages (p = 4,8,16,32,64) followed by an exact sub‑gradient stage.
    The time budget is split evenly among the six phases.
    """
    total_time = max(0.0, deadline - time.time())
    if total_time <= 0.0:
        return init_a, compute_objective(init_a)

    p_schedule = [4, 8, 16, 32, 64]
    n_phases = len(p_schedule) + 1
    time_per = total_time / n_phases

    best_a = init_a.copy()
    best_val = compute_objective(best_a)

    # smooth phases
    for p in p_schedule:
        if time.time() >= deadline:
            break
        sub_dead = min(deadline, time.time() + time_per)
        a_opt, val_opt = run_adam_smooth_phase(n, best_a, sub_dead, p)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Adam‑multi] after p={p} best {best_val:.6f}")

    # final exact sub‑gradient phase
    if time.time() < deadline:
        a_opt, val_opt = run_adam_submax_phase(n, best_a, deadline)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Adam‑multi] final sub‑max best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Differential Evolution on the simplex
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list[np.ndarray],
    deadline: float,
    popsize: int = 30,
    max_gens: int = 5000,
) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex (population of probability vectors).
    Returns the best individual discovered before *deadline*.
    """
    # initialise population
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0.0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(p) for p in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8
    CR = 0.9
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val, best_a = val, trial.copy()
                    last_improve = time.time()
                    print(f"[DE gen={gen:04d} i={i}] new best {best_val:.6f}")

        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple (1+1) hill‑climbing (mass transfer) refinement
# ----------------------------------------------------------------------
def run_hc_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_trials: int = 200000,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climbing.
    Moves a small amount of probability from one index to another;
    accepts only improving moves.
    """
    a = init_a.copy()
    cur_val = compute_objective(a)
    best_a, best_val = a.copy(), cur_val
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_trials:
        it += 1
        i, j = np.random.randint(n), np.random.randint(n - 1)
        if j >= i:
            j += 1

        # propose a transfer
        delta = np.random.exponential(scale=0.02)
        if np.random.rand() < 0.5:
            move = min(delta, a[i])
            a2 = a.copy()
            a2[i] -= move
            a2[j] += move
        else:
            move = min(delta, a[j])
            a2 = a.copy()
            a2[j] -= move
            a2[i] += move

        val2 = compute_objective(a2)
        if val2 < cur_val:
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val, best_a = val2, a2.copy()
                last_improve = time.time()
                print(f"[HC it={it}] new best {best_val:.6f}")

        # occasional escape from stagnation
        if time.time() - last_improve > 8.0:
            a = best_a.copy()
            jitter = np.random.randn(n) * 0.01
            a = np.maximum(a + jitter, 0.0)
            s = a.sum()
            if s > 0.0:
                a = a / s
            else:
                a = np.full(n, 1.0 / n)
            cur_val = compute_objective(a)
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the provided evaluation
    function.  The routine respects the total time budget and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = np.inf

    # Work primarily with n = 1024 (good trade‑off between quality and speed)
    lengths = [1024, 2048, 4096, 8192]

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # build a handful of sensible initial candidates
        # ------------------------------------------------------------------
        init_candidates = []

        # warm‑start from the current global best if length matches
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.array(best_seq, dtype=np.float64))

        # historic construction if available
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 1e-8:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # simple monotone decreasing linear shape
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # uniform distribution
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # a couple of random Dirichlet draws
        for _ in range(2):
            init_candidates.append(np.random.dirichlet(np.ones(n)))

        # ------------------------------------------------------------------
        # 1️⃣ Adam (smooth + exact) – use the best candidate as start
        # ------------------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        if remaining > 0.0:
            adam_time = remaining * 0.35
            slice_per = adam_time / len(init_candidates)
            for cand in init_candidates:
                if time.time() >= deadline:
                    break
                sub_dead = min(deadline, time.time() + slice_per)
                a_opt, val_opt = run_adam_multi_phase(n, cand, sub_dead)
                if val_opt < best_val:
                    best_val, best_seq = val_opt, a_opt.copy()
                    print(f"[Adam] global best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣ Differential Evolution (global) – warm start from the best few
        # ------------------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        if remaining > 0.0:
            de_time = remaining * 0.30
            sub_dead = min(deadline, time.time() + de_time)
            de_pop = init_candidates[:5]          # small warm‑start set
            a_de, val_de = run_de_opt(n, de_pop, sub_dead, popsize=30, max_gens=4000)
            if val_de < best_val:
                best_val, best_seq = val_de, a_de.copy()
                print(f"[DE] global best now {best_val:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣ Hill‑climbing fine‑tuning
        # ------------------------------------------------------------------
        remaining = max(0.0, deadline - time.time())
        if remaining > 0.0 and best_seq is not None and len(best_seq) == n:
            hc_dead = min(deadline, time.time() + remaining)
            a_hc, val_hc = run_hc_opt(n, best_seq, hc_dead)
            if val_hc < best_val:
                best_val, best_seq = val_hc, a_hc.copy()
                print(f"[HC] global best now {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector of size 1024
    # ----------------------------------------------------------------------
    if best_seq is None:
        n = 1024
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
```
