def solve_lp_max_sum(f, M):
    n = len(f)
    # number of constraints conv_len = 2*n-1
    conv_len = 2*n-1
    # Build constraint matrix in dense.
    A = np.zeros((conv_len, n))
    for k in range(conv_len):
        start = max(0, k-(n-1))
        end = min(k, n-1)
        # indices i from start..end correspond to f_i coefficients.
        # for each i, g_{k-i} appears with coefficient f_i.
        # Equivalent: for j from start..end, set A[k, j] = f_{k-j} ??? Actually rows correspond to output index k; we sum over i (index in f) as f_i * g_{k-i}, where g index = k-i
        # So j = k-i => i = k-j.
        # So for each i (range start..end):
        #   g_{k-i} where index = k-i
        #   coefficient f_i.
        # So in row k (constraint for conv element k), coefficient for variable g_{k-i} is f_i.
        # So we can fill using loops:
        for i in range(start, end+1):
            j = k - i
            A[k, j] = f[i]
    b = np.full(conv_len, M)
    # bounds: g_i >=0
    bounds = [(0, None)] * n
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True, 'time_limit': 0.1})
    if res.success:
        return res.x
    else:
        # fallback: return current f
        return f
```

def conv_ab(a, b):
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2*n-1)))
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[:2*n-1]
    return conv.real

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in three stages:

1. **Linear‑programming (LP) improvement** –  for a current vector `f`
   we solve the LP

        max   Σ g_j
        s.t.  (f ∗ g)_k ≤ M   ∀k          (M = max(f∗f))
              g_j ≥ 0

   The LP finds a direction that does not increase the current
   convolution bound `M`.  A convex combination `f ← (1‑α)f + α·ĝ`
   (with `ĝ = g / Σg`) strictly reduces `max(f∗f)`.  Repeating a few
   LP steps quickly pushes the objective down.

2. **(1+1) Evolution Strategy (ES)** –  after the LP phase a simple
   Gaussian mutation/selection loop refines the solution on the simplex.

3. **Multiple restarts & length schedule** –  the search is performed for
   several lengths (default 2048 → 1024 → 512) and a few random initial
   “exponential” shapes.  The best sequence found before the time budget
   expires is returned.

All convolution operations use an FFT implementation (`full_convolution`)
so even n = 2048 is very fast.  If SciPy is unavailable the LP stage is
skipped and the method falls back to the ES phase only.

The routine respects a total time budget (default 1000 s) and never
returns an empty or invalid list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import – LP is used only when available.
# ----------------------------------------------------------------------
try:
    from scipy.optimize import linprog
    _SCIPY_AVAILABLE = True
except Exception:                     # pragma: no cover
    _SCIPY_AVAILABLE = False

# ----------------------------------------------------------------------
# Convolution utilities (FFT based – O(n log n))
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_ab(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """Linear convolution a*b using FFT (same length convention as full_convolution)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    fb = np.fft.rfft(b, L)
    conv = np.fft.irfft(fa * fb, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution a*a."""
    return full_convolution(a).max()


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for a normalised vector)
# ----------------------------------------------------------------------
def compute_objective(a: np.ndarray) -> float:
    """2·n·max(a∗a) / (sum(a)²) – a is assumed non‑negative."""
    s = a.sum()
    if s <= 1e-12:
        return math.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Linear‑programming step (only if SciPy is present)
# ----------------------------------------------------------------------
def solve_lp_max_sum_g(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ g_j  s.t. (f∗g)_k ≤ M  ∀k ,  g ≥ 0 .
    Returns the optimal g (may be None if the LP fails).
    """
    if not _SCIPY_AVAILABLE:
        return None

    n = f.shape[0]
    conv_len = 2 * n - 1

    # Build the constraint matrix A where A[k, j] = f_{k‑j} (zero when out of range).
    # This can be filled efficiently by putting the vector f on each diagonal.
    A = np.zeros((conv_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j + n, j] = f

    # Inequality RHS – a tiny slack avoids numerical infeasibility.
    b = np.full(conv_len, M + 1e-14, dtype=np.float64)

    # Objective: minimise -Σ g_j  (i.e. maximise Σ g_j).
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None)] * n

    # HiGHS (default) is fast; we give it a small per‑LP time limit.
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"time_limit": 0.5},
    )
    if res.success:
        return res.x
    return None


# ----------------------------------------------------------------------
# Simple (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float,
               max_iter: int = 30_000) -> tuple[np.ndarray, float]:
    """
    (1+1)‑ES on the simplex.  Returns the best vector found before *deadline*.
    """
    a = init_a.copy()
    best_val = compute_objective(a)
    best_a = a.copy()
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val - 1e-12:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                             # success → keep
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
        else:
            sigma *= 0.99

        # Reset if we stall for a while.
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Initialiser – random exponential shape (a simple, well‑behaved family)
# ----------------------------------------------------------------------
def init_exponential(n: int) -> np.ndarray:
    """Return a normalised decreasing exponential distribution of length n."""
    # λ is sampled so that the decay speed adapts to n.
    lam = np.random.uniform(0.0005, 0.02) * n
    idx = np.arange(n, dtype=np.float64)
    a = np.exp(-lam * idx)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main entry point required by the competition harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative height sequence that minimises `evaluate_sequence`.
    The function respects the given time budget and returns a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq: np.ndarray | None = None
    best_val: float = math.inf

    # Length schedule: larger n gives more freedom but is more expensive.
    # 2048 works well with the LP routine while staying within memory limits.
    lengths = [2048, 1024, 512]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # Perform a few independent restarts (different random seeds).
        # ------------------------------------------------------------------
        restarts = 3
        for rs in range(restarts):
            if time.time() > deadline:
                break

            # ---- initialise a hopeful candidate --------------------------------
            if best_seq is not None and len(best_seq) == n:
                f = np.asarray(best_seq, dtype=np.float64)
                f = np.maximum(f, 0.0)
                f /= f.sum()
            else:
                f = init_exponential(n)

            # ------------------------------------------------------------------
            # Phase 1 – LP driven improvement (only if SciPy is available).
            # ------------------------------------------------------------------
            lp_iters = 0
            max_lp_iters = 25
            while time.time() < deadline and lp_iters < max_lp_iters:
                conv_f_f = full_convolution(f)
                M = conv_f_f.max()
                g = solve_lp_max_sum_g(f, M)
                if g is None:
                    break
                sum_g = g.sum()
                # If the LP cannot increase the total mass we are stuck.
                if sum_g <= 1.0 + 1e-12:
                    break
                # Normalise g to the simplex.
                g_norm = g / sum_g

                # Simple line‑search on the mixing parameter α.
                alphas = [0.1, 0.25, 0.5, 0.75]
                improved = False
                cur_val = compute_objective(f)
                best_local = f
                best_local_val = cur_val

                for alpha in alphas:
                    cand = (1.0 - alpha) * f + alpha * g_norm
                    cand = np.maximum(cand, 0.0)
                    cand /= cand.sum()
                    val_cand = compute_objective(cand)
                    if val_cand < best_local_val - 1e-12:
                        best_local_val = val_cand
                        best_local = cand
                        improved = True

                if not improved:
                    break

                # accept the best mixing we found
                f = best_local
                lp_iters += 1

                if best_local_val < best_val - 1e-12:
                    best_val = best_local_val
                    best_seq = f.copy()
                    print(f"[LP n={n} restart={rs} iter={lp_iters}] new global best {best_val:.6f}")

            # ------------------------------------------------------------------
            # Phase 2 – (1+1) ES fine‑tuning (always executed if time permits)
            # ------------------------------------------------------------------
            if time.time() < deadline:
                # give the ES a modest time slice (up to a minute)
                es_time = min(60.0, deadline - time.time())
                es_deadline = time.time() + es_time
                a_es, val_es = run_es_opt(n, f, es_deadline)
                if val_es < best_val - 1e-12:
                    best_val = val_es
                    best_seq = a_es.copy()
                    print(f"[ES n={n} restart={rs}] improved global best to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – if nothing was discovered we return a uniform vector.
    # ----------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = compute_objective(best_seq)

    # Ensure non‑negativity before returning (evaluate_sequence will clip anyway).
    best_seq = np.maximum(best_seq, 0.0)
    # Normalise once more (tiny numerical drift may have happened).
    if best_seq.sum() == 0.0:
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)
    else:
        best_seq = best_seq / best_seq.sum()

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
