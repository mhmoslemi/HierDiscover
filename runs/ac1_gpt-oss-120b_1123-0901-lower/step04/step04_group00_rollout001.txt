def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    best_seq = None
    best_val = float('inf')
    lengths = [128,256,512,1024,2048,4096,8192,16384] # maybe start smaller for quick search.
    # but avoid too large for early phase.

    # caches for FFT lengths
    fft_len_cache = {}
    for n in lengths:
        conv_len = 2*n-1
        fft_len_cache[n] = 1 << int(np.ceil(np.log2(conv_len)))

    # iterate lengths from small to large, bring best up.
    for n in lengths:
        if time.time() >= deadline:
            break
        conv_len = fft_len_cache[n]
        candidates = generate_initial_candidates(n, best_seq)
        # include upsampling of best_seq if exists
        for cand_idx, init in enumerate(candidates):
            if time.time() >= deadline: break
            a = init / init.sum()
            seq, val = optimise(a, n, conv_len, deadline)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"New global best (len {n}) {val:.6f}")
        # At end of length, optionally run random perturb local search on best_seq of that length
        if best_seq is not None and len(best_seq) == n:
            seq = best_seq.copy()
            seq, val = local_search(seq, n, conv_len, deadline, best_val)
            if val < best_val:
                best_val = val
                best_seq = seq
                print(f"Improved after local search len {n} {val:.6f}")

    if best_seq is None:
        # fallback uniform length 1024
        n=1024
        best_seq = np.full(n, 1.0/n)
    return best_seq.tolist()

"""
propose_candidate
=================

Search for a non‑negative step‑function (probability vector) that
minimises

    F(a) = 2·n·max(conv(a,a)) / (sum(a)²)

under a_i ≥ 0.  The objective is scale‑invariant, therefore the
search is performed on the simplex (∑a_i = 1).  The algorithm mixes
several ideas that proved useful in earlier attempts:

* **Multi‑scale exploration** – lengths 256 … 16384 are tried
  sequentially; a good solution for a small length is up‑sampled
  and refined for the next larger length.

* **Rich initial pool** – uniform, several Dirichlet draws,
  power‑law edge‑concentrated families, double‑exponential edge
  families and the up‑sampled best solution.

* **Two‑phase local optimisation**
  1. *Smooth‑max* Adam optimisation.  A temperature λ controls a
     soft‑max surrogate of the max‑convolution; λ is increased
     geometrically so the surrogate gradually approaches the true
     max.
  2. *Exact sub‑gradient* descent on the true max using an
     Euclidean projection onto the simplex (mirror‑descent style).

* **Local random walk** – after the deterministic phases a cheap
  stochastic “shake‑and‑accept” walk is performed while time
  permits.

* **Time‑budget awareness** – the routine stops as soon as the
  remaining budget (1000 s total) is exhausted; intermediate best
  solutions are always kept.

The function returns a plain Python list of floats that can be fed
directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)],  i = 0 … n‑1,
    where n = len(a).  The result has length n.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start : start + n]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex.
    Implements the algorithm of Duchi et al. (2008).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.nonzero(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) (sum(a) is assumed 1)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    best_val: float,
):
    """
    Adam optimiser on a soft‑max surrogate of max(conv).
    Returns the best pair (a, value) encountered during the run.
    """
    a = a0.copy()
    # initialise logits for a = softmax(logits)
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07

    lam = 0.5               # start temperature
    lam_factor = 1.09
    lam_max = 2e5

    best_local_a = a.copy()
    best_local_val = best_val
    last_improve = time.time()
    max_iters = int(14000 * 1024 / n) + 1500

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- soft‑max ↦ simplex -----
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # ----- convolution and smooth‑max weight -----
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient wrt a (smooth max) -----
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, lam_max)

        # ----- exact objective (sum ≈ 1) -----
        val = 2.0 * n * max_c  # no division by sum^2 because sum≈1

        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            best_val = val
            last_improve = time.time()
            print(f"[len={n} Adam it={it:05d}] new best {val:.6f}")

        # stagnation handling – a tiny random shake
        if time.time() - last_improve > 3.5:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_local_a, best_local_val


def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    best_val: float,
):
    """
    Sub‑gradient descent on the exact max‑convolution with Euclidean
    projection onto the simplex.
    """
    a = a0.copy()
    best_local_a = a.copy()
    best_local_val = best_val
    last_improve = time.time()
    eta0 = 0.55

    for it in range(2000):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        # uniform distribution over the active indices
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        step = eta0 / np.sqrt(it + 1.0)
        a = a - step * grad_a
        a = _project_onto_simplex(a)

        val = 2.0 * n * max_c
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            best_val = val
            last_improve = time.time()
            print(f"[len={n} Subgrad it={it:04d}] new best {val:.6f}")

        if time.time() - last_improve > 3.0:
            break

    return best_local_a, best_local_val


def _random_local_search(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    deadline: float,
    current_best_val: float,
):
    """
    Very cheap random walk: move a tiny amount of mass between two
    randomly chosen positions.  Accept only improvements.
    """
    a = a0.copy()
    best_val = current_best_val
    best_seq = a.copy()
    eps = 0.02  # typical magnitude of a single move

    while time.time() < deadline:
        # pick source and target
        i, j = np.random.choice(n, size=2, replace=False)
        if a[i] < 1e-12:
            continue
        delta = eps * a[i] * np.random.rand()
        a[i] -= delta
        a[j] += delta
        # re‑normalise (numerically safe)
        a = _project_onto_simplex(a)

        conv = _full_convolution(a, fft_len_conv)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[len={n} Random walk] new best {val:.6f}")

    return best_seq, best_val


def _upsample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a simplex vector to a larger size."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    up_sum = up.sum()
    if up_sum < 1e-12:
        up = np.full(new_n, 1.0 / new_n)
    else:
        up /= up_sum
    return up


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine respects ``budget_s`` seconds
    and returns the best sequence found (plain Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4  # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore (small → large).  Very large lengths are cheap
    # because the algorithm quickly converges to a flat solution.
    lengths = [256, 512, 1024, 2048, 4096, 8192, 16384]

    # pre‑compute FFT lengths for each size (both conv and cross‑corr)
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        fft_len_conv[n] = _next_pow2(2 * n - 1)
        # cross‑correlation needs length conv_len + n - 1 ≤ 3·n
        fft_len_corr[n] = _next_pow2(3 * n)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break

        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ------------------- build initial candidate pool ----------------
        candidates = []

        # 1) warm start if we already have a solution of the same size
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # 2) up‑sample previous best (if any) to current size
        if best_seq is not None and len(best_seq) < n:
            candidates.append(_upsample(np.asarray(best_seq, dtype=np.float64), n))

        # 3) historic global variable (if present)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            cand_sum = cand.sum()
            if cand_sum < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand_sum
            candidates.append(cand)

        # 4) a few Dirichlet draws (different concentration parameters)
        for alpha in (0.2, 0.5, 1.0, 2.0):
            candidates.append(
                np.random.dirichlet(np.full(n, alpha))
            )

        # 5) power‑law edge‑concentrated families
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 11):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 6) double‑exponential edge families
        i = np.arange(n, dtype=np.float64)
        for gamma in (0.3, 0.7, 1.2, 2.5, 4.0):
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise every candidate (deterministic + stochastic phases)
        for cand_id, a_start in enumerate(candidates):
            if time.time() >= deadline:
                break

            # ensure we start on the simplex
            a_start = np.maximum(a_start, 0.0)
            a_start_sum = a_start.sum()
            if a_start_sum < 1e-12:
                a_start = np.full(n, 1.0 / n)
            else:
                a_start = a_start / a_start_sum

            # ----- Phase 1 : Adam on smooth‑max -----
            a_adam, val_adam = _adam_smooth_max(
                a_start,
                n,
                conv_len,
                corr_len,
                deadline,
                best_val,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[len={n} cand={cand_id}] Adam best {val_adam:.6f}")

            # ----- Phase 2 : exact sub‑gradient -----
            a_sub, val_sub = _subgradient_descent(
                a_adam,
                n,
                conv_len,
                corr_len,
                deadline,
                best_val,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[len={n} cand={cand_id}] Sub‑grad best {val_sub:.6f}")

            # ----- Phase 3 : cheap random walk (if time left) -----
            if time.time() < deadline:
                a_rw, val_rw = _random_local_search(
                    a_sub,
                    n,
                    conv_len,
                    deadline,
                    best_val,
                )
                if val_rw < best_val:
                    best_val = val_rw
                    best_seq = a_rw.copy()
                    print(f"[len={n} cand={cand_id}] Random walk best {val_rw:.6f}")

        # End of length loop – keep the best sequence found so far
        if best_seq is not None:
            print(f"*** After length {n} best value = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    # Return as plain Python list (as required by the external evaluator)
    return best_seq.tolist()
