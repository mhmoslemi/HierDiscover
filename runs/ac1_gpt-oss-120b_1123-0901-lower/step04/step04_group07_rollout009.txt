Now function to run LP iteration:

def lp_step(a, M, target_sum):
    # Build matrix A
    A = build_constraint_matrix(a)
    n = len(a)
    # constraints: A @ b <= M
    b_ub = np.full(2*n -1, M, dtype=float)
    # objective
    c = -np.ones(n, dtype=float)  # maximize sum b
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0:
        return None
    g = b * (target_sum / sum_b)
    return g

"""
Search for a non‑negative step‑function that minimises ``evaluate_sequence``.

The algorithm is a two‑phase hybrid:
  1.  Small‑scale optimisation using a linear‑programming “mass‑redistribution”
      iteration (the classic LP‑rebalancing scheme from the problem statement).
      Starting from a diverse set of seeds we repeatedly
          – compute M = max(a∗a),
          – solve   maximise 1ᵀb   subject to   a∗b ≤ M   and   b ≥ 0,
          – rescale the solution to the required ℓ₁‑norm and mix it with a.
      This rapidly drives down the peak of the self‑convolution.
  2.  Large‑scale refinement with a projected Adam optimiser (soft‑max
      smoothing of the max‑convolution).  The Adam loop is run on the best
      candidate from phase 1 (interpolated to the larger length) and on a few
      random starts, always respecting the simplex constraint Σa = √(2 n).

Both phases respect the 1000 s wall‑clock limit, keep the best sequence
found throughout, and fall back to a uniform vector if nothing else succeeds.
"""

import time
import math
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of appropriate length."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _build_constraint_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the Toeplitz matrix A such that (A @ b) == a∗b.
    Shape: (2n‑1, n).
    """
    n = a.shape[0]
    col = np.concatenate([a, np.zeros(n - 1)])
    row = np.concatenate([np.array([a[0]]), np.zeros(n - 1)])
    return toeplitz(col, row)


def _lp_step(a: np.ndarray, M: float, target_sum: float):
    """
    One LP iteration: given current a and its peak M, find a vector g
    with larger ℓ₁‑norm under the same peak bound and return the rescaled
    version g (so that Σg = target_sum).  Returns None on failure.
    """
    n = a.shape[0]
    A = _build_constraint_matrix(a)          # (2n‑1) × n
    b_ub = np.full(2 * n - 1, M, dtype=float)
    c = -np.ones(n, dtype=float)             # maximise Σb  → minimise -Σb
    bounds = [(0.0, None)] * n

    # Using HiGHS via scipy (fast for dense problems of this size)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds,
                  method='highs',
                  options={'presolve': True, 'time_limit': 30})
    if not res.success:
        return None
    b = res.x
    sum_b = b.sum()
    if sum_b <= 0.0:
        return None
    # Rescale so that the ℓ₁‑norm matches the required target_sum
    scale = target_sum / sum_b
    return b * scale


def _run_lp_optimisation(a0: np.ndarray,
                         n: int,
                         target_sum: float,
                         deadline: float,
                         max_iters: int,
                         best_seq: np.ndarray,
                         best_score: float):
    """
    Run a few LP‑based rebalancing steps starting from a0.
    Returns the (possibly) improved best sequence and its score.
    """
    a = a0.copy()
    # Ensure feasibility (already normalised but we enforce it once)
    a = np.maximum(a, 0.0)
    a = a * (target_sum / a.sum())

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        M = _max_convolution(a)          # current peak = objective after normalisation
        if M < best_score:
            best_score = M
            best_seq = a.copy()
            print(f"[LP n={n:5d} iter={it:3d}] new best = {best_score:.6f}")

        g = _lp_step(a, M, target_sum)
        if g is None:
            # LP failed – stop this run
            break

        # Simple averaging update (mixing factor 0.5 works well in practice)
        a = 0.5 * a + 0.5 * g

        # Projection onto the feasible simplex
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a *= (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

    return best_seq, best_score


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution entries."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Sub‑gradient of the soft‑max max‑convolution.
    Computes rev_i = Σ_k w_k * a_{k‑i}.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _adam_optimize(a: np.ndarray,
                   n: int,
                   deadline: float,
                   max_iters: int,
                   best_seq: np.ndarray,
                   best_score: float) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser with a soft‑max temperature schedule.
    Returns the best sequence (and its exact max‑convolution) found during
    the run.
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the starting vector
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # Temperature schedule parameters
    tau_factor = 0.20
    decay_rate = max_iters / 10.0

    Nfft = 1 << ((2 * n - 1).bit_length())

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # --- exact convolution and its peak ---
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # --- temperature & soft‑max approximation ---
        tau = tau_factor * M * math.exp(-iteration / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # --- weighted reverse convolution (gradient of smoothed max) ---
        rev = _weighted_rev(a, w, Nfft)

        # --- Adam gradient step ---
        coeff = 4.0 * n / (target_sum * target_sum)          # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        lr = lr0 / (1.0 + 1e-4 * iteration)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # --- projection onto feasible set ---
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # --- exact objective (after projection it equals M) ---
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam n={n:5d} iter={iteration:7d}] new best = {best_score:.6f}")

    return best_seq, best_score
