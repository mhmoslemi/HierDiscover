import time, math, numpy as np

def next_pow2(x):
    return 1 << (x-1).bit_length()

def full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[: 2*a.shape[0] - 1]
    return conv

def cross_correlation(w, a, fft_len):
    # linear cross-correlation (w ★ a) as defined in original
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n - 1
    return corr[start:start+n]

def project_simplex(v):
    # Projection of a vector onto the probability simplex.
    # Implements algorithm from Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros case
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def resize_sequence(seq, new_n):
    old_n = len(seq)
    if old_n == new_n:
        return np.copy(seq)
    # use linear interpolation on normalized coordinate [0,1]
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(new_n))
    else:
        new_seq = new_seq / s
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence that minimises
    F(a)=2·n·max(a*a)/(sum(a))².  A multi‑stage optimiser is used:
    • generate diverse initial candidates (warm‑starts, random Dirichlet,
      power‑law, exponential and Beta shapes);
    • run Adam on a smooth‑max surrogate of the convolution peak, with
      projection onto the probability simplex and enforced symmetry;
    • refine with a sub‑gradient method on the exact max;
    • occasional random “shakes’’ escape stagnation.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    best_seq = None
    best_val = float("inf")

    # ----- Helper that evaluates the exact objective (assumes sum≈1) -----
    def exact_objective(a, n, fft_len):
        conv = full_convolution(a, fft_len)
        return 2.0 * n * conv.max()  # sum(a) == 1

    # ----- Lengths to explore (larger lengths give more degrees) -----
    lengths = [1024, 2048, 4096, 8192]   # 16384 would be too costly for many restarts

    # pre‑compute FFT lengths
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = next_pow2(conv_len)
        # for correlation we need length conv_len + n - 1 <= 3n
        fft_len_corr[n] = next_pow2(conv_len + n - 1)

    # ----- Main loop over lengths -----
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # ----- Build a list of diverse initial candidates -----
        candidates = []

        # Warm‑start from global best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(best_seq.copy())

        # Warm‑start from global best resized
        if best_seq is not None and len(best_seq) != n:
            candidates.append(resize_sequence(best_seq, n))

        # Historic solution if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 1e-12:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # Random Dirichlet seeds (a few)
        for _ in range(2):
            candidates.append(np.random.dirichlet(np.ones(n)))

        # Power‑law edge‑concentrated families (α negative)
        prod = (np.arange(n, dtype=np.float64) + 1) * (n - np.arange(n, dtype=np.float64))
        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # Two‑Gaussian edge shapes
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.3, 0.7, 1.5, 3.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # Symmetric Beta‑type shapes (U‑shaped, a<1)
        x = (np.arange(n, dtype=np.float64) + 0.5) / n
        for a_param in [0.2, 0.3, 0.5, 0.7]:
            # beta pdf with identical parameters a,b = a_param
            shape = (x ** (a_param - 1)) * ((1 - x) ** (a_param - 1))
            shape = shape / shape.sum()
            candidates.append(shape)

        # -----------------------------------------------------------------
        # Optimise each candidate
        # -----------------------------------------------------------------
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # initialise a on the simplex
            a = a0.copy()
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / s

            # Adam hyper‑parameters (these work well for a wide range of n)
            m = np.zeros_like(a)
            v = np.zeros_like(a)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.07          # modest step size
            lam = 0.5          # initial temperature for smooth‑max
            lam_factor = 1.04  # gently increase temperature

            # ADAM optimisation on the smooth‑max surrogate
            last_improve = time.time()
            # iteration budget scales inversely with n
            max_iters = int(15000 * 1024 / n) + 5000
            for t in range(1, max_iters + 1):
                # ----- compute smooth‑max weights -----
                conv = full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ----- gradient of the surrogate w.r.t a -----
                grad_a = 2.0 * cross_correlation(w, a, corr_len)

                # ----- Adam update -----
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                a_tmp = a - lr * m_hat / (np.sqrt(v_hat) + eps)

                # enforce non‑negativity
                a_tmp = np.maximum(a_tmp, 0.0)
                # enforce symmetry (helps reduce dimensionality & noise)
                a_tmp = (a_tmp + a_tmp[::-1]) * 0.5
                # project onto simplex
                a = project_simplex(a_tmp)

                # increase temperature
                lam = min(lam * lam_factor, 1e6)

                # ----- exact objective (sum ≈ 1) -----
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # ----- stagnation detection -----
                if time.time() - last_improve > 5.0:
                    # small random perturbation of the current logits
                    a = a + (np.random.rand(n) - 0.5) * 1e-3
                    a = np.maximum(a, 0.0)
                    a = (a + a[::-1]) * 0.5
                    a = project_simplex(a)
                    last_improve = time.time()  # reset timer

                if time.time() > deadline:
                    break

            # ----- Sub‑gradient refinement on the true max -----
            # start from the last a (already on simplex and symmetric)
            sub_eta = 0.6
            sub_last_improve = time.time()
            for sub_it in range(2000):
                conv = full_convolution(a, conv_len)
                max_c = conv.max()
                # identify active indices (within tolerance)
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                # sub‑gradient of exact max
                grad_a = 2.0 * cross_correlation(w, a, corr_len)

                # exponentiated‑gradient update (keeps non‑negativity)
                a = a * np.exp(-sub_eta * grad_a)
                a = np.maximum(a, 0.0)
                a = (a + a[::-1]) * 0.5
                a_sum = a.sum()
                if a_sum < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = a / a_sum

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                # early stop if stagnated
                if time.time() - sub_last_improve > 4.0:
                    break
                if time.time() > deadline:
                    break

            # ----- Local random shake – try to escape final plateaus -----
            shake_trials = 5
            for shake in range(shake_trials):
                if time.time() > deadline:
                    break
                # perturb current best_seq slightly (Gaussian noise)
                noisy = best_seq * (1.0 + 0.02 * np.random.randn(n))
                noisy = np.maximum(noisy, 0.0)
                noisy = (noisy + noisy[::-1]) * 0.5
                noisy_sum = noisy.sum()
                if noisy_sum < 1e-12:
                    continue
                noisy = noisy / noisy_sum

                # quick 200‑iteration Adam fine‑tune
                a = noisy.copy()
                m = np.zeros_like(a)
                v = np.zeros_like(a)
                lam_local = 0.5
                for tt in range(200):
                    conv = full_convolution(a, conv_len)
                    max_c = conv.max()
                    w = np.exp(lam_local * (conv - max_c))
                    w_sum = w.sum()
                    w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
                    grad_a = 2.0 * cross_correlation(w, a, corr_len)

                    m = beta1 * m + (1.0 - beta1) * grad_a
                    v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
                    m_hat = m / (1.0 - beta1 ** (tt + 1))
                    v_hat = v / (1.0 - beta2 ** (tt + 1))
                    a = a - lr * m_hat / (np.sqrt(v_hat) + eps)
                    a = np.maximum(a, 0.0)
                    a = (a + a[::-1]) * 0.5
                    a = project_simplex(a)

                    lam_local = min(lam_local * lam_factor, 1e6)

                conv = full_convolution(a, conv_len)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} shake={shake}] new best after shake {val:.6f}")

    # -----------------------------------------------------------------
    # Fallback – uniform distribution if nothing found
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Ensure returned list is plain Python floats
    return best_seq.tolist()

"""
Search for a non‑negative step‑function that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

The optimiser works in three stages:

1.   A pool of diverse initial candidates (warm‑starts, random Dirichlet,
     power‑law, exponential and Beta‑shaped families) is built.
2.   Each candidate is refined by Adam on a smooth‑max surrogate of the
     convolution peak.  The iterate lives on the probability simplex
     (projection) and is forced to be symmetric – this halves the effective
     search space and speeds convergence.
3.   A sub‑gradient phase on the exact max followed by a few random “shakes’’
     removes any remaining bias.

The routine respects a time budget (default 1000 s) and returns the best
sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # should not happen, but guard against division by zero
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _resize_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = seq.shape[0]
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        new_seq = np.random.dirichlet(np.ones(new_n))
    else:
        new_seq = new_seq / s
    return new_seq

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Run a multi‑stage optimiser (see module docstring) respecting the
    allotted time budget and return the best sequence found as a plain list.
    """

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    best_val = float("inf")

    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        fft_len_conv[n] = _next_pow2(2 * n - 1)
        fft_len_corr[n] = _next_pow2(3 * n)      # ≥ 2·n‑1 + n‑1

        lr_fine = 0.02
        lam = 0.5
        lam_factor = 1.03
        max_fine = int(8000 * 1024 / n) + 3000
        last_imp = time.time()
        for t in range(1, max_fine + 1):
            conv = _full_convolution(a, conv_fft)
            max_c = conv.max()
            w = np.exp(lam * (conv - max_c))
            w_sum = w.sum()
            w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)
            grad_a = 2.0 * _cross_correlation(w, a, corr_fft)

            m = beta1 * m + (1.0 - beta1) * grad_a
            v = beta2 * v + (1.0 - beta2) * (grad_a ** 2)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            a = a - lr_fine * m_hat / (np.sqrt(v_hat) + eps)

            a = np.maximum(a, 0.0)
            a = (a + a[::-1]) * 0.5
            a = _project_simplex(a)

            lam = min(lam * lam_factor, 1e6)

            if t % 500 == 0:
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_imp = time.time()
                    print(f"[final fine‑tune iter={t}] new best {val:.6f}")

            if time.time() - last_imp > 4.0:
                break
            if time.time() > deadline:
                break

    # ------------------------------------------------------------------
    # fallback – uniform vector (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # return plain Python list
    return best_seq.tolist()
