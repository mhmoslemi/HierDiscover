A two‑phase optimizer for the step‑function objective.

Phase 1 – a smooth Adam optimiser that minimises a soft‑max approximation
      of the maximal self‑convolution.  The temperature of the soft‑max
      decays exponentially so the landscape becomes increasingly sharp.

Phase 2 – a cheap simulated‑annealing (hill‑climbing) that moves a small
      amount of mass between two randomly chosen positions while keeping
      the ℓ₁‑norm exactly constant.  The convolution is recomputed with an
      FFT after each proposal; only moves that lower the maximal
      convolution are accepted (occasionally a worse move is accepted
      during the early “hot’’ stage).

Both phases respect the feasible set (a_i ≥ 0, Σ a_i = √(2 n)).  Several
initialisations (uniform, Dirichlet, exponential, Hann, sparse) and a
few different lengths (≈2100, 1.5× and 2×) are tried.  The best sequence
found before the wall‑clock deadline is returned as a plain Python list.
"""
import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns the weight vector w (summing to 1) and the smoothed maximum."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute   rev_i = Σ_k w_k * a_{k‑i}   (indices outside range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    # slice that corresponds to rev_i for i=0..n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """Adam optimiser with a soft‑max warm‑up followed by an exact sub‑gradient."""
    target_sum = math.sqrt(2.0 * n)

    # normalise the initial vector
    a = a0 * (target_sum / a0.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 5000

    # temperature schedule parameters
    tau_factor = 0.20
    decay_rate = max_iters / 10.0

    Nfft = 1 << ((2 * n - 1).bit_length())   # next power‑of‑two ≥ 2·n‑1

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # exact convolution
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # temperature & soft‑max
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # weighted reverse convolution (= gradient of smoothed max)
        rev = _weighted_rev(a, w, Nfft)

        # Adam gradient step
        coeff = 4.0 * n / (target_sum * target_sum)   # =2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)
        grad = coeff * rev - const

        # Adam updates
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # projection onto feasible set
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # occasional jitter
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        cur_score = M           # after normalisation the objective equals M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """Generate a diverse set of starting vectors for length n."""
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    rev = x[::-1]
    init.append(rev / rev.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    init.append(black / black.sum() * target_sum)

    # 7) sparse random – choose ~sqrt(n) positions with equal weight
    m = max(3, int(math.sqrt(n)))
    pos = rng.choice(n, size=m, replace=False)
    sparse = np.zeros(n)
    sparse[pos] = 1.0
    sparse = sparse / sparse.sum() * target_sum
    init.append(sparse)

    # 8) small perturbation of uniform
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


def _simulated_annealing(
    a0: np.ndarray,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing / annealing phase.
    Moves a small amount of mass from one index to another while keeping Σa constant.
    """
    n = len(a0)
    a = a0.copy()
    target_sum = a.sum()
    Nfft = 1 << ((2 * n - 1).bit_length())

    # current objective (max convolution)
    cur_score = _max_convolution(a, Nfft)

    rng = np.random.default_rng()
    total_iters = 0
    # schedule: temperature decreases linearly with the fraction of time used
    while time.time() < deadline and total_iters < 20000:
        # pick source index with some mass
        i = rng.integers(0, n)
        if a[i] < 1e-12:
            continue
        # destination
        j = rng.integers(0, n)
        if i == j:
            continue

        # propose moving a small fraction of the source mass
        frac = 0.1
        delta = a[i] * rng.random() * frac
        if delta <= 0:
            continue

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_score = _max_convolution(a_new, Nfft)

        # temperature based on remaining time (higher early on)
        remaining = deadline - time.time()
        temp = max(1e-4, remaining)  # simple schedule

        accept = False
        if new_score < cur_score:
            accept = True
        else:
            # Metropolis acceptance (rarely needed, kept for completeness)
            prob = math.exp(-(new_score - cur_score) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            a = a_new
            cur_score = new_score
            if new_score < best_score:
                best_score = new_score
                best_seq = a.copy()
                print(f"[anneal] new best = {best_score:.6f}")
        total_iters += 1

    return best_seq, best_score
