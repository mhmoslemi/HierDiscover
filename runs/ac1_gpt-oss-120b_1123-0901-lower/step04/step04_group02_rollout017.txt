def cross_correlation(w, a, fft_len):
    # w length = 2n-1, a length = n
    # we want correlation length n (same as a)
    # Compute via convolution: w * rev_a (full) then appropriate slice.
    rev_a = a[::-1]

"""
propose_candidate
-----------------

Search for a non‑negative step function that minimises

        F(a) = 2·n·max(a*a) / (sum(a)^2)

where ``a*a`` is the linear convolution of ``a`` with itself.
The method consists of three parts

1. **Exponentiated‑gradient descent on the simplex**  
   The non‑smooth max‑convolution is replaced by a smooth‑max
   (softmax) with a temperature λ that is increased
   geometrically (annealing).  The gradient of the smooth surrogate
   is obtained by the cross‑correlation of the weight vector with the
   current candidate.  An exponentiated‑gradient (mirror‑descent) step
   keeps the iterate on the probability simplex.

2. **Random “shake’’** – if the current run stagnates for a few
   seconds a tiny random perturbation is added before continuing.

3. **Local random refinement** – after the main optimisation a fast
   hill‑climbing phase moves a tiny amount of mass between two
   coordinates.  Improvements are kept.

The routine tries several problem sizes (from a few hundred up to
≈ 30 000 entries) with a few independent restarts per size.  
The best sequence discovered before the allotted time (default 1000 s)
is returned as a plain Python list.  If nothing is found a uniform
fallback is produced.

All heavy work is done with FFT ( O(n log n) ) and the code respects
the global time‑budget.  No closures or lambdas are used – all helpers
are top‑level functions.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used directly but kept for safety


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length *fft_len* (power of two)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    This is exactly the gradient of the smooth‑max surrogate
    (up to the factor 2).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr_full[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex Σx_i = 1, x_i≥0."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _upscale(seq: np.ndarray, new_len: int) -> np.ndarray:
    """Linear interpolation of a (short) sequence to a longer length."""
    old_len = seq.shape[0]
    if old_len == new_len:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_len, endpoint=False)
    x_new = np.linspace(0.0, 1.0, new_len, endpoint=False)
    return np.interp(x_new, x_old, seq)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Return a list of non‑negative heights (a step function) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget (default 1000 s) and prints progress information.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_seq = None          # numpy array, sum == 1
    best_val = float("inf")
    target_val = 1.503       # stop early if we ever beat this

    # ------------------------------------------------------------------
    # Helper that returns the *exact* objective for a probability vector.
    def _exact_obj(a: np.ndarray) -> float:
        n = a.shape[0]
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Problem sizes we will try.  Small sizes are cheap and give quick
    # feedback, large sizes give the theoretical advantage.
    lengths = [512, 1024, 2048, 4096, 8192, 16384, 32768]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)     # sufficient for (2n‑1)+n

        # number of independent random restarts for this length
        restarts = 4 if n <= 4096 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # -------------------- initialise a (probability vector) --------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm‑start from global best")
            elif best_seq is not None and best_seq.shape[0] < n:
                a = _upscale(best_seq, n)
                a = np.maximum(a, 0.0)
                a /= a.sum()
                print("[restart] up‑scaled previous best")
            elif 'height_sequence_1' in globals():
                cand = np.asarray(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                s = cand.sum()
                a = cand / s if s > 0 else np.random.dirichlet(np.ones(n))
                print("[restart] start from height_sequence_1")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random Dirichlet start")

            # ----- exponentiated‑gradient state -----
            eta0 = 0.4                 # base step‑size
            lam = 0.3                  # initial smooth‑max temperature (λ)
            lam_factor = 1.12          # geometric increase per iteration
            t = 0
            last_improve = time.time()
            max_iters = 30000 if n <= 4096 else 15000

            while time.time() < deadline and t < max_iters:
                t += 1

                # ----- convolution and smooth‑max weights -----
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # soft‑max over all convolution entries
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum

                # ----- gradient of the smooth surrogate -----
                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ----- exponentiated‑gradient update -----
                eta = eta0 / np.sqrt(t)            # simple decay
                # limit the magnitude to avoid overflow in the exponent
                grad = np.clip(grad, -50.0, 50.0)
                a = a * np.exp(-eta * grad)
                a = np.maximum(a, 1e-15)           # keep positivity
                a /= a.sum()                       # project back onto simplex

                # ----- anneal temperature -----
                lam = min(lam * lam_factor, 1e6)

                # ----- exact objective for monitoring -----
                val = _exact_obj(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t:05d}] new best: {val:.6f}")
                    if best_val < target_val:
                        print("Target reached – terminating early.")
                        deadline = time.time()      # force outer loops to stop
                        break

                # ----- stagnation handling -----
                if time.time() - last_improve > 6.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = a + noise
                    a = np.maximum(a, 0.0)
                    a /= a.sum()
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake (stagnation)")

            # end of one restart
        # end of lengths loop

    # ------------------------------------------------------------------
    # Final random refinement (if any time left)
    # --------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local random refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move a tiny fraction of the total mass from i to j
            delta = 0.003 * a[i] * np.random.rand()
            if delta <= 0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = np.maximum(a_prop, 0.0)
            a_prop /= a_prop.sum()
            val = _exact_obj(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")
                if best_val < target_val:
                    print("Target reached during refinement.")
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return a plain Python list as required by the external evaluator.
    return best_seq.tolist()
