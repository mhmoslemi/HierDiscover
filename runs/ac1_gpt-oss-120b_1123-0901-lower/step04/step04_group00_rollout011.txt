def surrogate_f(logits, lam, n, fft_len_conv, fft_len_corr):
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a = exp_x / exp_x.sum()
    conv = _full_convolution_fft(a, fft_len_conv)
    # log-sum-exp:
    shifted = conv - conv.max()  # max_c
    exp_shifted = np.exp(lam * shifted)
    sum_exp = exp_shifted.sum()
    smooth_max = conv.max() + (1.0/lam) * np.log(sum_exp)
    obj = 2.0 * n * smooth_max   # sum a = 1
    # compute weights
    w = exp_shifted / sum_exp
    grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)
    return obj, grad_logits

def run_l_bfgs(a0, n, lam, fft_len_conv, fft_len_corr, max_iter):
    logits0 = np.log(a0 + 1e-15)
    def fun(logits):
        obj, grad = surrogate_f(logits, lam, n, fft_len_conv, fft_len_corr)
        return obj, grad
    res = scipy.optimize.minimize(fun, logits0, method='L-BFGS-B', jac=True,
                                  options={'maxiter': max_iter, 'ftol': 1e-12, 'gtol':1e-12})
    logits_opt = res.x
    # compute final a
    exp_x = np.exp(logits_opt - np.max(logits_opt))
    a_opt = exp_x / exp_x.sum()
    return a_opt, res.fun

"""
Propose a low‑value step‑function sequence.

The algorithm works on several lengths (4096, 8192, 16384).  For
each length it builds a handful of plausible initial vectors (warm‑start,
historic solution, random Dirichlet, edge‑concentrated power‑law,
double‑Gaussian).  Every candidate is refined by a two‑stage procedure:

1.  A smooth‑max surrogate of  max(a*a) is minimised with a
    L‑BFGS‑B optimizer.  The surrogate is
        2·n·[max(conv) + (1/λ)·log Σ exp(λ·conv)]
    and its gradient w.r.t. the soft‑max logits is obtained by a
    cross‑correlation (FFT‑based, O(n log n)).  λ is increased in a
    schedule so that the optimisation gradually focuses on the true
    maximum.

2.  A short exponentiated‑gradient sub‑gradient phase on the exact
    max‑convolution removes the remaining smoothing bias.

All evaluations are made with a fast FFT convolution.  The best sequence
found (smallest exact objective) is stored in the global variable
`height_sequence_1` for warm‑starts in later calls and finally returned as
a plain Python list.
"""

import time
import numpy as np
import functools
import scipy.optimize as opt

# ----------------------------------------------------------------------
def _fft_len_pow2(x: int) -> int:
    """Return the smallest power‑of‑two ≥ x."""
    return 1 << int(np.ceil(np.log2(x)))


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a) and equals Σ_k w[k]·a[i+k‑(n‑1)].
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _smooth_max_obj_and_grad(
    logits: np.ndarray,
    n: int,
    lam: float,
    fft_len_conv: int,
    fft_len_corr: int,
):

    Smooth‑max surrogate of the objective and its gradient w.r.t. logits.
    logits are the unconstrained parameters; a = softmax(logits) is a
    probability vector (∑a = 1, a_i ≥ 0).
    """

    exp_x = np.exp(logits - mx)
    a = exp_x / exp_x.sum()

    conv = _full_convolution_fft(a, fft_len_conv)

    sum_exp = exp_shifted.sum()
    smooth_max = max_c + (1.0 / lam) * np.log(sum_exp)

    f = 2.0 * n * smooth_max                       # surrogate objective

    # soft‑max weights for the smooth max
    w = exp_shifted / sum_exp

    # gradient w.r.t. a (2·cross‑correlation)
    grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)

    # back‑propagation through the soft‑max
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    return f, grad_logits


def _optimize_l_bfgs(
    a0: np.ndarray,
    n: int,
    lam: float,
    fft_len_conv: int,
    fft_len_corr: int,
    max_iter: int = 200,
):
    """
    Run L‑BFGS‑B on the smooth‑max surrogate starting from a0.
    Returns the refined probability vector and the surrogate value.
    """
    logits0 = np.log(a0 + 1e-15)

    fun = functools.partial(
        _smooth_max_obj_and_grad,
        n=n,
        lam=lam,
        fft_len_conv=fft_len_conv,
        fft_len_corr=fft_len_corr,
    )
    res = opt.minimize(
        fun,
        logits0,
        method="L-BFGS-B",
        jac=True,
        options={"maxiter": max_iter, "disp": False},
    )
    exp_x = np.exp(res.x - np.max(res.x))
    a_opt = exp_x / exp_x.sum()
    return a_opt, res.fun


def _subgradient_refine(
    a: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    max_iter: int = 500,
    eta: float = 0.5,
    deadline: float = None,
):
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Stops early if no improvement for a few seconds.
    Returns the best vector seen and its exact objective value.
    """
    best_a = a.copy()
    conv = _full_convolution_fft(a, fft_len_conv)
    best_val = 2.0 * n * conv.max()
    last_improve = time.time()

    for it in range(max_iter):
        if deadline is not None and time.time() > deadline:
            break

        conv = _full_convolution_fft(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_corr_fft(w, a, fft_len_corr)

        # exponentiated‑gradient update (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = 2.0 * n * _full_convolution_fft(a, fft_len_conv).max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 3.0:
            break

    return best_a, best_val


def _stretch(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation of a probability vector to a new length.
    The result is re‑normalised to sum to 1.
    """
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    stretched = np.interp(x_new, x_old, seq)
    stretched = np.maximum(stretched, 0.0)
    if stretched.sum() < 1e-12:
        stretched = np.random.dirichlet(np.ones(new_n))
    else:
        stretched /= stretched.sum()
    return stretched


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths to explore (larger lengths often give a lower bound)
    lengths = [4096, 8192, 16384]

    # Pre‑compute FFT lengths for each problem size
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = _fft_len_pow2(conv_len)
        corr_len = conv_len + n - 1           # ≤ 3·n
        fft_len_corr[n] = _fft_len_pow2(corr_len)

    # Schedule for the smoothing temperature λ (increasing → closer to true max)
    lambda_sched = [5.0, 20.0, 80.0, 300.0, 1200.0]

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_c = fft_len_conv[n]
        fft_x = fft_len_corr[n]

        # ----- build a small pool of starting points --------------------
        candidates = []

        # warm start from the current global best if it has the same length
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # historic solution (height_sequence_1) – trimmed / padded as needed
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        # stretch previous best (if any) to this length
        if best_seq is not None and len(best_seq) != n:
            stretched = _stretch(np.asarray(best_seq, dtype=np.float64), n)
            candidates.append(stretched)

        # pure random Dirichlet start
        candidates.append(np.random.dirichlet(np.ones(n)))

        # edge‑concentrated power‑law family  ((i+1)*(n-i))**α , α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 7):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # two‑Gaussian edge shapes  exp(-γ·i²)+exp(-γ·(n‑1‑i)²)
        i = np.arange(n, dtype=np.float64)
        for gamma in [0.2, 0.5, 1.0, 2.0]:
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise each candidate independently
        for cid, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Ensure a proper probability vector
            a = a0 / a0.sum()

            # ----- smooth‑max L‑BFGS optimisation with increasing λ  -----
            for lam in lambda_sched:
                if time.time() > deadline:
                    break

                # L‑BFGS – a small number of iterations is enough for a good
                # descent direction because the surrogate is smooth.
                a, _ = _optimize_l_bfgs(
                    a,
                    n,
                    lam,
                    fft_len_conv=fft_c,
                    fft_len_corr=fft_x,
                    max_iter=150,
                )

                # Exact evaluation (sum(a) is ≈ 1)
                conv = _full_convolution_fft(a, fft_c)
                val = 2.0 * n * conv.max() / (a.sum() ** 2)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    globals()["height_sequence_1"] = best_seq.tolist()
                    print(
                        f"[n={n} cand={cid} λ={lam:.0f}] new best {val:.6f}"
                    )

                # Early termination of λ‑loop if we are already out of time
                if time.time() > deadline:
                    break

            # ----- final exact sub‑gradient phase -------------------------
            a_refined, sub_val = _subgradient_refine(
                a,
                n,
                fft_len_conv=fft_c,
                fft_len_corr=fft_x,
                max_iter=800,
                eta=0.5,
                deadline=deadline,
            )
            if sub_val < best_val:
                best_val = sub_val
                best_seq = a_refined.copy()
                globals()["height_sequence_1"] = best_seq.tolist()
                print(f"[n={n} cand={cid} subgrad] new best {sub_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list
    return best_seq.tolist()
