"""
propose_candidate
-----------------
A hybrid optimizer for the “max‑convolution” problem.

The optimiser works in three stages for each problem size *n*:

1. **Adam on a smooth‑max surrogate** – fast gradient descent on the
   simplex (probability vector) that drives the maximum entry of the
   linear convolution a∗a down.

2. **Frequency‑magnitude equalisation** – an alternating‑projection
   scheme that forces the magnitude of the (zero‑padded) FFT of *a* to
   become flat.  A flatter spectrum makes the autocorrelation (the
   convolution) more uniform, which directly reduces the target
   ``2·n·max(a∗a)``.

3. **Local stochastic hill‑climbing** – tiny mass transfers between
   random coordinates are accepted only if they improve the exact
   objective.  This polishes the solution found by the first two
   phases.

The routine respects a global time budget (default 1000 s) and
maintains a global best sequence across all explored lengths.
It can be warm‑started from the historic construction ``height_sequence_1``,
or from the best solution found for a smaller length.
The final result is returned as a plain Python list of non‑negative
floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k divided by 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

    The function respects the time budget ``budget_s`` (seconds) and prints
    progress information.  The best sequence found is returned as a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None               # numpy array, sum == 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Exact objective for a probability vector (sum = 1)
    # ------------------------------------------------------------------
    def _exact_obj(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, _next_pow2(2 * n - 1))
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Problem sizes – we start small and increase the dimension if time permits
    # ------------------------------------------------------------------
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        fft_len_conv = _next_pow2(2 * n - 1)     # for a∗a
        fft_len_corr = _next_pow2(3 * n)        # for gradient (w★a)
        fft_len_eq   = _next_pow2(2 * n)        # zero‑padded FFT for magnitude equalisation

        # ------------------------------------------------------------------
        # Multiple independent restarts per length
        # ------------------------------------------------------------------
        restarts = 5
        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------------------------------------------------------
            # Initialise a (probability vector)
            # ------------------------------------------------------------------
            if best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print(f"[n={n} restart={r}] warm start from global best")
            else:
                # Try to reuse the historic construction if it exists
                try:
                    cand = np.asarray(height_sequence_1, dtype=np.float64)
                    if cand.shape[0] >= n:
                        a = cand[:n]
                    else:
                        a = np.pad(cand, (0, n - cand.shape[0]))
                    a = np.maximum(a, 0.0)
                    s = a.sum()
                    a = a / s if s > 1e-8 else np.random.dirichlet(np.ones(n))
                    print(f"[n={n} restart={r}] start from height_sequence_1")
                except NameError:
                    a = np.random.dirichlet(np.ones(n))
                    print(f"[n={n} restart={r}] random start")

            # ------------------------------------------------------------------
            # Phase 1 – Adam on smooth‑max surrogate
            # ------------------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.20
            lam = 0.5
            lam_factor = 1.10
            t = 0
            max_iters = 25000
            last_improve = time.time()

            while time.time() < deadline and t < max_iters:
                # convolution and smooth‑max weights
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum

                # gradient of the smooth surrogate
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # Adam update (pre‑projection)
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_a
                v = beta2 * v + (1.0 - beta2) * (grad_a * grad_a)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a_new = a - step

                # projection back onto the simplex
                a = _project_simplex(a_new)

                # temperature schedule (approaches true max)
                lam = min(lam * lam_factor, 1e6)

                # exact evaluation for bookkeeping
                val = _exact_obj(a, n)

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best = {val:.6f}")

                # occasional shake to escape stagnation
                if time.time() - last_improve > 6.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # ------------------------------------------------------------------
            # Phase 2 – Frequency‑magnitude equalisation (Gerchberg‑Saxton style)
            # ------------------------------------------------------------------
            eq_iters = 2000
            gamma = 0.20           # how aggressively we move toward a flat spectrum
            while time.time() < deadline and eq_iters > 0:
                eq_iters -= 1
                # FFT of current a (zero‑padded to fft_len_eq)
                F = np.fft.rfft(a, fft_len_eq)
                mags = np.abs(F)
                target = mags.mean()
                new_mags = (1.0 - gamma) * mags + gamma * target
                new_F = new_mags * np.exp(1j * np.angle(F))
                a_eq = np.fft.irfft(new_F, fft_len_eq)[:n]
                a_eq = _project_simplex(np.maximum(a_eq, 0.0))
                a = a_eq

                val = _exact_obj(a, n)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} equalisation] improved to {val:.6f}")

            # ------------------------------------------------------------------
            # Phase 3 – Local random hill‑climbing (transfer mass between two bins)
            # ------------------------------------------------------------------
            while time.time() < deadline:
                i, j = np.random.choice(n, 2, replace=False)
                # move a small fraction of the mass on i towards j
                delta = a[i] * np.random.rand() * 0.005   # ≤0.5 % of a[i]
                if delta <= 0.0:
                    continue
                a_prop = a.copy()
                a_prop[i] -= delta
                a_prop[j] += delta
                a_prop = _project_simplex(a_prop)   # safety against rounding errors
                val = _exact_obj(a_prop, n)
                if val < best_val:
                    best_val = val
                    best_seq = a_prop.copy()
                    a = a_prop
                    print(f"[n={n} hill‑climb] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\n=== Search finished ===\nBest value = {best_val:.6f}")
    return best_seq.tolist()
```

This implementation combines a fast gradient‐based optimiser, a spectral‑flattening step, and a final stochastic polishing phase while respecting the time budget and the non‑negativity constraints required by `evaluate_sequence`.
