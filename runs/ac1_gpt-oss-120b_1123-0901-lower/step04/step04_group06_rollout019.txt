import time, math, numpy as np
def _full_convolution_fft(x, y):
    n = len(x) + len(y) - 1
    size = 1 << ((n-1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx*fy, n=size)[:n]
    # due to floating errors maybe tiny negative, clip
    np.maximum(conv, 0.0, out=conv)
    return conv
```

    p0 = init_seq / init_seq.sum()
    x = np.log(p0 + 1e-12)  # logits

    v = np.zeros_like(x)

        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        a = target_sum * p

        # compute conv and max
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_local:

def _simulated_annealing(seq, cur_score, deadline):

    n = len(a)
    S = a.sum()
    best_seq = a.copy()
    best_score = cur_score

        if delta == 0: continue

        if new_score < best_score:
            best_score = new_score
            best_seq = a_new.copy()
            a = a_new
            # optionally print

            prob = math.exp(-(new_score - best_score) / (temp + 1e-9))

    return best_seq, best_score

"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The optimizer works in three stages:

1. **Smart initialisation** – a power‑law shape a_i ∝ (i+1)^‑p with p
   close to 0.5 (the analytically flat continuous solution) is used.
   Random jitter and optional exponential taper enrich the search space.

2. **Annealed Adam on the soft‑max parametrisation** – we map an
   unrestricted vector `x` to a feasible sequence via a soft‑max
   (guaranteeing non‑negativity and the exact sum Σa_i = √(2 n)).
   The objective is a smoothed max obtained by a temperature‑controlled
   log‑sum‑exp of the full self‑convolution.  Adam updates with a
   slowly decreasing temperature and learning‑rate drive the solution
   towards a flat convolution.

3. **Short simulated‑annealing polishing** – on the best sequence of
   each restart we perform a cheap random‑mass‑swap hill‑climb with a
   temperature schedule, always keeping the sum constant.
   This helps to escape the narrow basins that gradient methods may miss.

The routine runs for at most ``budget_s`` seconds (default 1000 s),
keeps the best sequence seen across many independent restarts (different
lengths, different random seeds) and finally returns it as a plain
list.  A previously discovered construction can be used as a warm‑start
via the global variable ``height_sequence_1``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper – fast full linear convolution using real FFT (no external libs)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # Numerical noise may produce tiny negatives – clip them.
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) parametrisation.
# ----------------------------------------------------------------------
def _adam_optimize(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.25,
    min_tau_factor: float = 0.001,
    lr: float = 0.08,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> np.ndarray:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best feasible sequence encountered before *deadline*.
    The objective is a smooth approximation of ``max(conv(a,a))``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_val = float("inf")
    it = 0

    while time.time() < deadline:
        # ---- 1) map logits → simplex (soft‑max) -----------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # stable soft‑max
        p = exp_shift / np.sum(exp_shift)        # probabilities, sum = 1
        a = target_sum * p                       # feasible sequence

        # ---- 2) convolution and current (smooth) max ----------------
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_val:
            best_val = cur_max
            best_seq = a.copy()

        # ---- 3) smooth‑max weights (log‑sum‑exp) --------------------
        # temperature decays exponentially, never below `min_tau_factor`.
        tau = max(init_tau_factor * cur_max * (0.97 ** it), min_tau_factor * cur_max)
        w = np.exp((conv - cur_max) / tau)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4) gradient w.r.t. a (2 * a ⊛ rev(w)) ------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)            # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                    # positions for grad_a
        grad_a = 2.0 * conv2[idx]

        # ---- 5) back‑prop through soft‑max ----------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)           # shape (n,)

        # ---- 6) Adam update -----------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random jitter to shake out plateaus
        if it % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # very slow learning‑rate decay (helps final fine‑tuning)
        lr *= 0.9996

    # ensure a feasible return even if the loop never improved
    if best_seq is None:
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)
        p = exp_shift / np.sum(exp_shift)
        best_seq = target_sum * p
    return best_seq


# ----------------------------------------------------------------------
# Light‑weight simulated‑annealing polishing step.
# ----------------------------------------------------------------------
def _simulated_annealing(
    seq: np.ndarray,
    cur_val: float,
    deadline: float,
    *,
    max_moves: int = 5000,
    init_temp: float = 0.5,
    min_temp: float = 0.001,
) -> np.ndarray:
    """
    Perform a short random‑mass‑swap hill‑climb with temperature decay.
    Returns a (possibly) improved sequence.
    """
    a = seq.copy()
    n = a.shape[0]
    best_seq = a.copy()
    best_val = cur_val

    start = time.time()
    for move in range(max_moves):
        if time.time() >= deadline:
            break

        # linear temperature schedule
        t = max(init_temp * (1 - move / max_moves), min_temp)

        # propose a transfer of a random amount from i → j
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # amount is a fraction of the source value
        delta = (np.random.rand() - 0.5) * 0.1 * a[i]   # up to ±5 % of a[i]
        if delta == 0.0:
            continue
        if a[i] - delta < 0.0:
            delta = a[i]   # clamp to keep non‑negative

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        conv = _full_convolution_fft(a_new, a_new)
        new_val = conv.max()

        # Metropolis acceptance
        if new_val < best_val:
            best_val = new_val
            best_seq = a_new.copy()
            a = a_new
            # occasional progress report
            if move % 500 == 0:
                print(f"[{time.time():.2f}] SA improvement → {best_val:.6f}")
        else:
            if np.random.rand() < math.exp(-(new_val - best_val) / (t + 1e-12)):
                a = a_new

    return best_seq


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the provided
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly select a length ``n`` (up to ≈ 15000) – larger ``n`` gives
      a finer discretisation of the analytically flat 1/√x shape.
    * Initialise with the power‑law a_i ∝ (i+1)^{-p} where p≈0.5,
      optionally perturbed by an exponential taper and a tiny jitter.
    * Optimise the shape with an annealed Adam descent on the soft‑max
      parametrisation (see ``_adam_optimize``).
    * Polish the result with a short simulated‑annealing mass‑swap.
    * Keep the overall best across many independent restarts.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2   # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction, if present.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # Choose problem size. Early on we try larger n to gain resolution.
        if remaining > 400:
            n = np.random.randint(8000, 15000 + 1)
        elif remaining > 150:
            n = np.random.randint(4000, 10000 + 1)
        else:
            n = np.random.randint(1200, 5000 + 1)

        # ------- 1) build initial shape --------------------------------
        # Power‑law exponent centred at 0.5 (the analytically flat case)
        p_exp = np.random.normal(loc=0.5, scale=0.12)
        p_exp = np.clip(p_exp, 0.25, 0.75)

        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # occasional exponential taper (adds extra flexibility)
        if np.random.rand() < 0.4:
            tau_exp = np.random.uniform(0.5, 2.5) * n
            init *= np.exp(-idx / tau_exp)

        # tiny jitter to break perfect symmetry
        init += np.random.rand(n) * 1e-5
        init = np.maximum(init, 1e-12)                # avoid exact zeros

        # ------- 2) warm‑start injection (optional) --------------------
        if best_seq is not None and np.random.rand() < 0.45:
            # reuse a prefix of the current global best (or pad with noise)
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # ------- 3) scale to the required sum -------------------------
        target_sum = math.sqrt(2.0 * n)               # Σ a_i = √(2 n)
        init = init / init.sum() * target_sum

        # ------- 4) convert to logits for Adam -----------------------
        x0 = np.log(init + 1e-12)

        # allocate a time slice for this run (roughly 8‑12 s)
        slice_time = min(12.0, remaining * 0.55)
        run_deadline = min(deadline, time.time() + slice_time)

        # ------- 5) Adam optimisation ---------------------------------
        seq_local = _adam_optimize(
            n=n,
            x0=x0,
            target_sum=target_sum,
            deadline=run_deadline,
            init_tau_factor=0.25,
            min_tau_factor=0.001,
            lr=0.08,
        )

        # evaluate the candidate with the official evaluator
        score_local = evaluate_sequence(seq_local.tolist())
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})")
            if best_score < 1.503:                     # early exit if we beat the target
                break

        # ------- 6) short simulated‑annealing polishing ---------------
        # Only attempt if we still have enough time left.
        if time.time() < deadline - 5.0:
            sa_deadline = min(deadline, time.time() + 4.5)
            seq_sa = _simulated_annealing(
                seq=seq_local,
                cur_val=score_local,
                deadline=sa_deadline,
                max_moves=3000,
            )
            score_sa = evaluate_sequence(seq_sa.tolist())
            if score_sa < best_score:
                best_score = score_sa
                best_seq = seq_sa.copy()
                print(f"[{time.time():.2f}] GLOBAL SA improvement = {best_score:.6f} (n={n})")
                if best_score < 1.503:
                    break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Return a plain Python list – the evaluator expects this format.
    return best_seq.tolist()
