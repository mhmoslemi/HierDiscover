"""
Propose a step‑function (non‑negative vector) with a low value of

        F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm combines three ideas:

1. **Projected gradient descent** on the probability simplex.
   - The objective is 2·n·max(conv) with conv = a*a.
   - We compute a sub‑gradient of max(conv) by averaging the
     gradients of all lags attaining the maximum.
   - After a gradient step we *project* back onto the simplex,
     which naturally creates zeros (sparsity) that the previous
     soft‑max based method could not reach.

2. **Annealed learning‑rate** – the step size is increased when a
   step improves the objective and reduced otherwise; this gives a
   simple line‑search without expensive back‑tracking.

3. **Local greedy refinement** – after the gradient phase we repeatedly
   move a small amount of mass from one coordinate to another and
   keep the move if it improves the objective.  The refinement can
   escape shallow local minima left behind by the gradient stage.

The routine works for any problem size up to several × 10⁴,
adapting the number of iterations to the remaining time budget.
It returns the best sequence found before the 1000 s deadline.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for the gradient.
    Returns a vector of length n equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def project_simplex(v: np.ndarray, s: float = 1.0) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex
        { x | x_i ≥ 0, Σ x_i = s }.
    Returns a new array (the original is not modified).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # all components are zero, return the origin (it satisfies the simplex)
        return np.zeros_like(v)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine respects the 1000 s time budget and returns the
    best sequence (as a plain Python list) found so far.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # problem sizes – larger n gives a lower theoretical lower‑bound.
    # we start with a modest size to obtain a good pattern quickly.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths that will be reused many times (powers of two)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))   # ≥ (2n‑1)+n‑1

        # number of independent restarts – fewer for the biggest sizes
        if n <= 1024:
            restarts = 8
        elif n <= 4096:
            restarts = 5
        elif n <= 8192:
            restarts = 3
        else:
            restarts = 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initialise a (probability vector) -----------------------
            if best_seq is not None and len(best_seq) == n:
                a = best_seq.copy()
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    a = np.random.dirichlet(np.ones(n))
                else:
                    a = cand / cand.sum()
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                # random Dirichlet – we sample a concentration that favours
                # a bit of sparsity (α in [0.3, 2]).
                alpha = np.random.uniform(0.3, 2.0)
                a = np.random.dirichlet(alpha * np.ones(n))
                print(f"[n={n} restart={r}] random Dirichlet α={alpha:.2f}")

            # ----- optimisation loop (projected sub‑gradient) --------------
            lr = 0.5                     # initial learning rate
            min_lr = 1e-8
            decay_factor = 0.7
            decay_every = 3000
            last_improve = time.time()
            max_iters = 30000 if n <= 2048 else 20000 if n <= 8192 else 12000
            t = 0

            while time.time() < deadline and t < max_iters:
                # convolution and current objective
                conv = full_convolution(a, fft_len=fft_len_conv)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t:05d}] new best {val:.6f}")

                # sub‑gradient of the max‑function (average over all active lags)
                tol = max_c * 1e-12 + 1e-15
                active = np.where(conv >= max_c - tol)[0]
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.size

                # gradient of the *objective* (2·n·max_c) w.r.t. a
                grad = 4.0 * n * cross_correlation(w, a, fft_len=fft_len_corr)

                # gradient step with simple line‑search‑style acceptance
                a_candidate = a - lr * grad
                a_candidate = project_simplex(a_candidate, s=1.0)

                # evaluate candidate
                conv_cand = full_convolution(a_candidate, fft_len=fft_len_conv)
                val_cand = 2.0 * n * conv_cand.max()

                if val_cand <= val:
                    # accept, maybe increase step size a little
                    a = a_candidate
                    lr = min(lr * 1.05, 1.0)
                else:
                    # reject – shrink the step size
                    lr = lr * 0.5
                    if lr < min_lr:
                        # step size too small → break out of the gradient loop
                        break

                t += 1
                # learning‑rate decay (deterministic)
                if t % decay_every == 0:
                    lr = max(lr * decay_factor, min_lr)

                # stagnation handling – add a tiny random shake
                if time.time() - last_improve > 30.0:
                    eps = (np.random.rand(n) - 0.5) * 1e-2
                    a = project_simplex(a + eps, s=1.0)
                    last_improve = time.time()
                    print(f"[n={n} restart={r} iter={t}] shake (stagnation)")

            # ----- local greedy refinement -----------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_loc = best_seq.copy()
                local_iters = 4000
                for it in range(local_iters):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    if a_loc[i] <= 1e-15:
                        continue
                    # move a random fraction (20‑80 %) of a_loc[i] to j
                    frac = np.random.rand() * 0.6 + 0.2
                    delta = a_loc[i] * frac
                    a_new = a_loc.copy()
                    a_new[i] -= delta
                    a_new[j] += delta
                    a_new = project_simplex(a_new, s=1.0)

                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_loc = a_new
                        print(f"[local refine n={n}] improved to {val_new:.6f}")

    # ----------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform vector (objective = 2)
        fallback_n = 2000
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
