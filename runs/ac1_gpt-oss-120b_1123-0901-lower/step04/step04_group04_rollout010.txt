def simplex_projection(v):
    """Project vector v onto the probability simplex {x >=0, sum x = 1}."""
    # Reference: Duchi et al. 2008.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive, return uniform distribution
        return np.full_like(v, 1.0/n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    # numerical issues: adjust sum to exactly 1
    w_sum = w.sum()
    if w_sum == 0:
        return np.full_like(v, 1.0/n)
    w = w / w_sum
    return w

Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load a known good construction (``height_sequence_1``) if it exists,
   otherwise start from a uniform vector.
2. Normalise to the required L1‑norm `Σ a = sqrt(2·n)` and work in the
   probability simplex `p = a / Σ a` (`Σ p = 1 , p ≥ 0`).
3. **Phase A – sub‑gradient descent**  
   Repeatedly:
   * compute the full auto‑convolution `c = p * p`,
   * locate the index `k*` where `c` attains its maximum,
   * the (sub‑)gradient of `max(p*p)` at `p` is `g_i = 2·p_{k*‑i}`
     (zero when the index is out of range),
   * take a small step `p ← p − η·g` and project back onto the simplex.
   The step size `η` decays slowly.
4. **Phase B – simulated‑annealing local search**  
   Starting from the best point of Phase A, repeatedly move a random
   amount of probability mass from a source index to a destination index.
   Accept the move if it improves the exact objective, otherwise accept
   with probability `exp(−Δ/temperature)`.  The temperature and the
   transferred mass shrink over time.
5. Convert the best probability vector back to the required heights,
   perform a final exact evaluation and return the list.

Both phases are cheap (FFT of length ≈ 2·n, O(n log n) per iteration)
and comfortably fit into the 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------


def simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project a vector v onto the probability simplex {x ≥ 0, Σ x = 1}.

    Uses the O(n log n) algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    # numerical safety: renormalise exactly to one
    w_sum = w.sum()
    if w_sum == 0.0:
        return np.full_like(v, 1.0 / n)
    return w / w_sum


def conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p * p (real FFT). Returns length 2·n‑1 array."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)
    return conv[: 2 * p.shape[0] - 1]


def rev_from_max(p: np.ndarray, k_max: int) -> np.ndarray:
    """Sub‑gradient component for the active maximum index k_max."""
    n = p.shape[0]
    rev = np.zeros_like(p)
    idx = k_max - np.arange(n)
    mask = (idx >= 0) & (idx < n)
    rev[mask] = p[idx[mask]]
    return rev
