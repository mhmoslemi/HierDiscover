"""
Improved step‑function search (≈1 000 s budget)

The optimiser works in three stages for each problem size *n*:

1. **Smooth‑max Adam** – Optimises a differentiable surrogate of
   max(a∗a) obtained with a temperature τ (log‑sum‑exp).  The surrogate
   is gradually sharpened (τ → large) while Adam updates the logits of
   a = softmax(x).  This gives a much more stable descent direction
   than the raw sub‑gradient of the max and typically finds lower peaks.

2. **Differential Evolution** – A classic population‑based global search
   on the simplex (same implementation as the reference code).  It
   refines the shape found by stage 1 and can escape local minima.

3. **(1+1) Evolution Strategy** – Simple Gaussian perturbations used for
   final fine‑tuning.

The algorithm starts with a modest size (n=512) where the search is cheap,
then upsamples the current best vector to larger *n* (1024 → 2048 → 4096 → 8192)
by linear interpolation and continues optimisation.  This multiscale
approach dramatically reduces the dimensionality of the early search while
still allowing very large final sequences.

All operations respect the 1000 s time budget; the function returns the
best sequence (as a plain Python list) discovered before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities – pure top‑level functions
# ----------------------------------------------------------------------
def softmax(logits: np.ndarray) -> np.ndarray:
    """Numerically stable softmax (maps logits → probability vector)."""
    m = np.max(logits)
    e = np.exp(logits - m)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear self‑convolution a*a (length 2·len(a)−1) using FFT.
    Returns a real‐valued array.
    """
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence`` for an arbitrary vector.
    Equivalent to 2·n·max(a∗a) / (sum(a)²).  The vector *a* need not be
    normalised – the formula is invariant to scaling.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation from the old vector to size ``new_n``.
    The resulting vector is renormalised to sum = 1.
    """
    old_n = seq.size
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0, 1, old_n)
    x_new = np.linspace(0, 1, new_n)
    up = np.interp(x_new, x_old, seq)
    up = up / up.sum()
    return up

# ----------------------------------------------------------------------

    n: int,

    max_iters: int = 30_000,
    lr: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
    tau_start: float = 5.0,
    tau_end: float = 80.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the *log‑sum‑exp* surrogate of max(a∗a).
    The probabilities a are represented as softmax(logits) – this
    automatically enforces positivity and Σa = 1.
    Returns the best vector and its exact objective.
    """
    # initialise logits from the starting probabilities
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)

    best_val = np.inf
    best_a = None

    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                           # a ∈ simplex
        conv = full_convolution(a)               # length 2n‑1
        max_c = conv.max()                       # true max (for reporting)

        # exact objective (sum(a)=1)
        obj = 2.0 * n * max_c

        if obj < best_val:
            best_val = obj
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SmoothAdam n={n} it={t:05d}] new best: {obj:.6f}")

        # ---------- smooth max (log‑sum‑exp) ----------
        # temperature schedule (linear in iteration)
        tau = tau_start + (tau_end - tau_start) * min(1.0, t / max_iters)

        # w_i ∝ exp(tau * conv_i)  – numerically stable
        shift = conv.max()
        exp_vals = np.exp(tau * (conv - shift))
        w = exp_vals / exp_vals.sum()            # probability over convolution indices

        # gradient of the smoothed max w.r.t. a:
        #   ∂/∂a_j  Σ_i w_i (a∗a)_i  = 2 Σ_i w_i a_{i‑j}
        grad_a = 2.0 * full_convolution(w, a[::-1])[n - 1 : n - 1 + n]

        # back‑propagation through the softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)               # keep logits bounded

        # shake if stuck
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} it={t}] shake (stagnation)")

    # safety fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# 2️⃣  Differential Evolution (global search on simplex)
# ----------------------------------------------------------------------
def run_de_opt(
    n: int,
    init_pop: list,
    deadline: float,
    popsize: int = 12,
    max_gens: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple DE on the simplex.  Individuals are projected back onto
    Σa = 1 after each trial.  Returns the best individual found.
    """
    # initialise population
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0.0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))

    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8
    CR = 0.9
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # pick three distinct donors
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)

            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)

            # binomial crossover (ensure at least one donor component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# 3️⃣  (1+1) Evolution Strategy (fine‑tuning)
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) ES with adaptive step‑size.  Operates directly on the
    simplex (probability vector).  Returns the best vector found.
    """
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function (list of heights) that
    minimises the supplied evaluation function.  The routine respects the
    time budget and returns the best sequence discovered as a plain list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    best_seq = None
    best_val = np.inf

    # multiscale list – we start small and grow
    lengths = [512, 1024, 2048, 4096, 8192]

    for idx, n in enumerate(lengths):
        if time.time() > deadline:
            break
        print(f"\n=== Optimising size n={n} ===")

        # ------------------------------------------------------------------
        # Build a few sensible initial candidates for the current size.
        # ------------------------------------------------------------------
        init_candidates = []

        # Warm‑start from the best sequence found so far (up‑sample if needed)
        if best_seq is not None:
            if len(best_seq) == n:
                init_candidates.append(np.asarray(best_seq, dtype=np.float64))
            else:
                up = upsample_sequence(np.asarray(best_seq, dtype=np.float64), n)
                init_candidates.append(up)

        # Historic construction if present
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.01:
                hs = hs / s
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # Simple decreasing linear shape (biased towards low max convolution)
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec = dec / dec.sum()
        init_candidates.append(dec)

        # Uniform distribution (baseline)
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # ------------------------------------------------------------------
        # 1️⃣ Smooth‑max Adam (several restarts)
        # ------------------------------------------------------------------
        restarts = 4
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r < len(init_candidates):
                a0 = init_candidates[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a modest slice of remaining time
            slice_time = min((deadline - time.time()) * 0.12, 35.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_smooth_adam_opt(
                n, a0, sub_deadline, max_iters=25_000,
                tau_start=4.0, tau_end=90.0, lr=0.04
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # 2️⃣ Differential Evolution (global search)
        # ------------------------------------------------------------------
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.30, 120.0)
            sub_deadline = time.time() + de_time
            # Warm start population – take the three most recent candidates
            de_pop = init_candidates[:3]
            a_de, val_de = run_de_opt(n, de_pop, sub_deadline,
                                      popsize=12, max_gens=50_000)
            if val_de < best_val:
                best_val = val_de
                best_seq = a_de.copy()
                print(f"[n={n}] DE GLOBAL best now {best_val:.6f}")

        # ------------------------------------------------------------------
        # 3️⃣ (1+1) ES fine‑tuning
        # ------------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.25, 80.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[n={n}] ES GLOBAL best now {best_val:.6f}")

        # ------------------------------------------------------------------
        # Prepare for the next (larger) size: up‑sample current best.
        # ------------------------------------------------------------------
        if idx + 1 < len(lengths):
            next_n = lengths[idx + 1]
            if best_seq is not None:
                best_seq = upsample_sequence(np.asarray(best_seq, dtype=np.float64), next_n)

    # ----------------------------------------------------------------------
    # Fallback – if nothing succeeded, return a uniform vector (len 1024)
    # ----------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
