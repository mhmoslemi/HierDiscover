def proj_simplex(v):
    """Project onto the probability simplex."""
    # From literature: sort descending, compute cumulative sum.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _run_l_bfgs(y0, init_tau, target_sum, Nfft, deadline, best_seq, best_score):
    y = y0.copy()
    tau = init_tau
    tau_decay = 0.7
    min_tau = 1e-12
    while time.time() < deadline and tau > min_tau:
        # define objective
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g
        options = {'maxfun': 300, 'maxiter': 300, 'ftol': 1e-12, 'gtol': 1e-12}
        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)
            y = res.x
        else:
            # simple gradient descent if SciPy not present
            L, g = _objective_and_grad(y, tau, Nfft)
            step = 0.1
            y = y - step * g

        # evaluate exact objective
        p = np.exp(y - np.max(y))
        p = p / p.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = target_sum**2 * max_conv

            best_score = cur_score
            best_seq = (p * target_sum).copy()

        # schedule update
        tau *= tau_decay
        # jitter
        if np.random.rand() < 0.2:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break
    return best_seq, best_score

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Work with a probability vector `p = a / Σa` (so Σp = 1).  The
  exact objective after normalisation equals `max_k (p∗p)_k`,
  i.e. the maximum entry of the self‑convolution of `p`.

* Optimise a smooth approximation of the maximum:
        L_τ(p) = τ·log Σ_k exp((p∗p)_k / τ) .
  Its gradient can be expressed via a *weighted* reverse convolution,
  which we compute efficiently with FFTs.

* Perform a continuation L‑BFGS‑B run:
    - start with a relatively large temperature `τ`,
    - after each L‑BFGS stage shrink `τ`,
    - after every stage evaluate the *exact* objective and keep the best
      sequence found,
    - add small random jitter to escape shallow basins.

* After the smooth optimisation finish a few iterations of *exact* sub‑
  gradient descent on the true max‑convolution (gradient of a single
  active constraint).  Each sub‑gradient step is projected onto the
  probability simplex.

* Several random starts (uniform, Dirichlet, etc.) are tried within the
  time budget (default 1000 s).  The best sequence found is returned.

The code is pure NumPy / SciPy (if available) and respects the
``evaluate_sequence`` security checks.
"""

import time
import math
import numpy as np
from typing import List

# ---------- optional SciPy ----------
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ---------- utility functions ----------
def proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project onto the probability simplex (non‑negative, sum = 1)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns (weights, smooth‑max value)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p∗p using real FFT (length 2·n‑1)."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k a_{k‑i}
    using FFTs (indices outside the support are ignored).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the valid part
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective L_τ and its gradient w.r.t. the log‑probability
    parameters `y`.
    """
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    # full convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p: 2 * rev, where rev_i = Σ_k w_k p_{k‑i}
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # back‑prop through the softmax
    dot = np.dot(p, grad_p)           # scalar = Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Repeated L‑BFGS minimisation of the smooth max, decreasing τ each
    stage.  Returns the best sequence (as a *scaled* vector a) and the
    best objective value seen so far.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.70      # shrink factor per stage
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # modest budget per stage – enough for a good direction
        options = {
            'maxfun': 300,
            'maxiter': 300,
            'ftol': 1e-12,
            'gtol': 1e-12,
            'disp': False,
        }

        if _SCIPY_AVAILABLE:
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)
            y = res.x
        else:
            # simple gradient step when SciPy unavailable
            L, g = _objective_and_grad(y, tau, Nfft)
            y = y - 0.1 * g

        # ----- evaluate the *exact* objective -----
        p = np.exp(y - np.max(y))
        p = p / p.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # = evaluate_sequence(a)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule update & jitter -----
        tau *= tau_decay

        # occasional random jitter to avoid plateaus
        if np.random.rand() < 0.20:
            y += np.random.randn(y.shape[0]) * 0.01

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _subgradient_descent(p: np.ndarray, Nfft: int,
                         max_iter: int = 200, step0: float = 0.5) -> np.ndarray:
    """
    Exact sub‑gradient descent on the true max‑convolution.
    Each iteration moves away from the currently active sum index.
    The iterate is projected onto the probability simplex.
    """
    n = p.shape[0]
    K = 2 * n - 1

    for it in range(max_iter):
        # full convolution and current worst index
        conv = _conv_full(p, Nfft)
        k_max = int(np.argmax(conv))
        max_val = conv[k_max]

        # weight vector = unit mass at the active index
        w = np.zeros(K, dtype=float)
        w[k_max] = 1.0

        # reverse‑weighted vector gives the gradient of the active max
        rev = _weighted_rev(p, w, Nfft)
        grad = 2.0 * rev

        # diminishing step size
        step = step0 / np.sqrt(it + 1.0)
        p_new = p - step * grad
        p_new = proj_simplex(p_new)

        # check improvement
        conv_new = _conv_full(p_new, Nfft)
        if conv_new.max() < max_val - 1e-12:
            p = p_new
        else:
            # No immediate improvement → shrink step size a bit
            step0 *= 0.5
            if step0 < 1e-6:
                break

    return p
