"""
Step‑function optimizer for the self‑convolution bound.

The routine works in three stages:

1. **Broad exploration** – a collection of parametric families
   (exponential, Gaussian, Laplace, half‑sine, power‑sine, quadratic,
   Dirichlet draws, plus the previously discovered best sequence) is
   generated for several lengths (≈2k–4k).  Each candidate is normalised
   to sum = 1 and evaluated with a cheap FFT‑based convolution.
   The best few candidates are retained.

2. **Projected sub‑gradient refinement** – starting from each retained
   candidate a simple sub‑gradient descent on the exact objective
   ``F(a)=2·n·max(a*a)/(∑a)^2`` is performed.
   The gradient of the max‑convolution is obtained analytically
   (∇ max = 2·a shifted by the arg‑max indices).  After every step the
   vector is projected onto the probability simplex (non‑negative,
   unit sum) so the scaling invariance is respected.  A decaying step
   size and periodic random “shakes’’ avoid stagnation.

3. **Final polishing** – a short fine‑tuning pass with a smaller step
   size is run on the overall best result.

All work respects a user supplied time budget (default 1000 s) and
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import numpy as np
import random

# -------------------------------------------------------------
# Helper 1 – convolution via FFT (linear self‑convolution)
def conv_self_fft(a: np.ndarray) -> np.ndarray:
    """Return the linear convolution a*a using an FFT of appropriate size."""
    n = a.shape[0]
    # next power of two ≥ 2·n‑1
    size = 1 << ((2 * n - 2).bit_length())
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# -------------------------------------------------------------
# Helper 2 – fast objective (identical to evaluate_sequence up to clipping)
def fast_objective(a: np.ndarray) -> float:
    """2·n·max(a*a)/(∑a)^2; a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    conv = conv_self_fft(a)
    return float(2.0 * a.shape[0] * conv.max() / (s * s))


# -------------------------------------------------------------
# Helper 3 – projection onto the simplex {x ≥ 0, ∑x = 1}
def project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    # sort in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# -------------------------------------------------------------
# Helper 4 – gradient of the max‑convolution (sub‑gradient)
def max_conv_subgradient(a: np.ndarray, max_idx: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max_k (a*a)[k] w.r.t. a.
    max_idx contains all indices where the max is attained.
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    for k in max_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        grad[i] += 2.0 * a[k - i]
    return grad


# -------------------------------------------------------------
# Helper 5 – one refinement run (projected sub‑gradient descent)
def refine_subgradient(
    a0: np.ndarray,
    deadline: float,
    max_iter: int = 20000,
) -> tuple[np.ndarray, float]:
    """
    Refine a candidate using projected sub‑gradient descent.
    Returns (best_sequence, best_objective).
    """
    a = a0.copy()
    n = a.shape[0]

    # hyper‑parameters
    eta = 0.4               # initial step size
    decay = 0.9995          # geometric decay
    best_val = fast_objective(a)
    best_seq = a.copy()
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1

        conv = conv_self_fft(a)
        mval = conv.max()
        # indices where the max is attained (allow tiny tolerance)
        idxs = np.where(conv >= mval - 1e-12)[0]

        # sub‑gradient of the numerator
        grad_num = max_conv_subgradient(a, idxs)

        # full gradient of 2·n·max/∑a² using quotient rule
        sum_a = a.sum()
        denom = sum_a * sum_a
        grad = (2.0 * n / denom) * grad_num - (
            4.0 * n * mval / (denom * sum_a)
        ) * a

        # gradient step and projection onto simplex
        a = a - eta * grad
        a = project_simplex(a)

        # step‑size schedule
        eta *= decay

        # occasional exact evaluation
        if it % 5 == 0:
            val = fast_objective(a)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Refine] iter {it:05d}  new best = {val:.6f}")

        # shake if stagnated
        if time.time() - last_improve > 2.0:
            a = best_seq + 0.001 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = project_simplex(a)
            last_improve = time.time()

    return best_seq, best_val


# -------------------------------------------------------------
# Helper 6 – final fine‑tuning (same algorithm with smaller step)
def fine_tune(a0: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """A short run with a smaller step size to polish the result."""
    a = a0.copy()
    n = a.shape[0]
    eta = 0.1
    decay = 0.9999
    best_val = fast_objective(a)
    best_seq = a.copy()
    last_improve = time.time()
    it = 0

    while time.time() < deadline:
        it += 1
        conv = conv_self_fft(a)
        mval = conv.max()
        idxs = np.where(conv >= mval - 1e-12)[0]
        grad_num = max_conv_subgradient(a, idxs)

        sum_a = a.sum()
        denom = sum_a * sum_a
        grad = (2.0 * n / denom) * grad_num - (
            4.0 * n * mval / (denom * sum_a)
        ) * a

        a = a - eta * grad
        a = project_simplex(a)
        eta *= decay

        if it % 10 == 0:
            val = fast_objective(a)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")

        if time.time() - last_improve > 2.0:
            a = best_seq + 0.0003 * np.random.randn(n)
            a = np.maximum(a, 0.0)
            a = project_simplex(a)
            last_improve = time.time()
    return best_seq, best_val


# -------------------------------------------------------------
# Candidate generators (various parametric families)
def generate_candidates(budget_deadline: float) -> list[tuple[np.ndarray, float]]:
    """

    cand = []
    lengths = [1800, 2000, 2200, 2500, 2800, 3000, 3500, 4000]

    for n in lengths:

        # 3. Gaussian shapes
        mu = (n - 1) / 2.0

        # 4. Laplace (double‑exponential)
        for tau_fac in (0.10, 0.12, 0.15):

            a = np.maximum(a, 0.0)
            a /= a.sum()
            cand.append((a, fast_objective(a)))

        # 5. pure exponential
        for r in (0.30, 0.50, 0.70, 0.85):

        cand.append((a, fast_objective(a)))

            cand.append((a, fast_objective(a)))

        for _ in range(8):

    if "height_sequence_1" in globals():

def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises `evaluate_sequence`.  The routine respects the given
    time budget (seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # -----------------------------------------------------------------
    # Stage 1 – broad exploration
    print("[Stage 1] generating candidates …")
    candidates = generate_candidates(deadline)

    # keep only the top few (by objective)
    candidates.sort(key=lambda kv: kv[1])
    top_k = min(5, len(candidates))
    best_seq, best_val = candidates[0]
    print(f"[Stage 1] best initial val = {best_val:.6f} (len {len(best_seq)})")

    # -----------------------------------------------------------------
    # Stage 2 – projected sub‑gradient refinement on the top candidates
    for idx in range(top_k):
        if time.time() >= deadline:
            break
        a0, _ = candidates[idx]
        remaining = deadline - time.time()
        # allocate a fair share of the remaining time
        slice_deadline = time.time() + remaining / (top_k - idx)
        refined_seq, refined_val = refine_subgradient(a0, slice_deadline)
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined_seq.copy()
            print(f"[Stage 2] candidate {idx} improved to {refined_val:.6f}")

    # -----------------------------------------------------------------
    # Stage 3 – final fine‑tuning (small step size, more iterations)
    if time.time() < deadline:
        print("[Stage 3] final fine‑tuning …")
        fine_seq, fine_val = fine_tune(best_seq, deadline)
        if fine_val < best_val:
            best_val = fine_val
            best_seq = fine_seq.copy()
            print(f"[Stage 3] final best = {fine_val:.6f}")

    # -----------------------------------------------------------------
    # Return the best sequence as a plain Python list (scaling is irrelevant)
    print(f"Finished – best objective ≈ {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
