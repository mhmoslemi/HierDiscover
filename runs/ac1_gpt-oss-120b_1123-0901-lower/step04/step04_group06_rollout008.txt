"""
Propose a non‑negative step‑function that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluation simplifies to `max(conv(a, a))` when the sum is fixed to
√(2 n).  Hence the objective is to make the self‑convolution as flat as
possible.

Algorithm
---------
1. **Coarse parametric search** – Randomly sample a few thousand
   candidate shapes from simple parametric families (power‑law,
   exponential, sine‑power, Gaussian, triangular and “U”‑shaped).  For
   each length `n` the shape is normalised to the required sum
   √(2 n) and its maximal convolution value is computed with an FFT‑based
   convolution.  The best few candidates are stored.

2. **Refinement** – For the top‑k stored candidates an Adam optimiser
   works on the soft‑max parametrisation of the vector (log‑its `x`).
   The optimiser minimises a smoothed version of the max (log‑sum‑exp)
   and therefore converges to a local optimum of the true objective.

3. **Warm‑start** – If a global variable `height_sequence_1` is present
   it is used as an additional starting point.

The whole routine respects a 1000 s time budget; progress and improving
bounds are printed.  The final sequence (as a plain list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by floating‑point errors
    np.maximum(conv, 0.0, out=conv)
    return conv


def _conv_max(a: np.ndarray) -> float:
    """Return max(conv(a, a)) for a non‑negative vector a."""
    return _full_convolution_fft(a, a).max()


# ----------------------------------------------------------------------
# Adam optimiser on the log‑softmax (log‑it) representation.
# The objective is the smoothed max of the self‑convolution.
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best sequence encountered (as a NumPy array) and its
    evaluation score obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- 1. map logits to the simplex (a = target_sum·softmax(x)) -----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)                # numerical stability
        p = exp_shift / np.sum(exp_shift)            # probabilities, sum=1
        a = target_sum * p                            # enforce Σa = target_sum

        # ----- 2. compute (smoothed) max of the convolution --------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ----- 3. smooth‑max weights ------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- 4. gradient w.r.t. a ------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)               # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                       # positions of interest
        grad_a = 2.0 * conv2[idx]

        # ----- 5. gradient w.r.t. logits x (chain rule) ---------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- 6. Adam update -------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if iteration % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning‑rate
        alpha *= 0.9995

    # final evaluation with the official evaluator (identical rounding)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

            target_sum = np.sqrt(2.0 * n)

            # Convert to log‑it representation required by the optimiser
            x0 = np.log(seq + 1e-12)

            slice_deadline = time.time() + per_candidate * 0.9   # tiny safety margin
            seq_opt, score_opt = _optimise_adam_softmax(
                n,
                x0,
                target_sum,
                slice_deadline,
                init_tau_factor=0.15,
                min_tau_factor=0.0005,
                alpha=0.07,
                beta1=0.9,
                beta2=0.999,
            )

            if score_opt < best_val:
                best_val = score_opt
                best_seq = seq_opt.copy()
                print(f"[{time.time():.2f}] REFINED GLOBAL BEST n={n} "
                      f"val={score_opt:.6f}")

            # stop early if we already beat the target bound
            if best_val < 1.503:
                break

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # Convert to plain Python list for the external evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    print(f"[{time.time():.2f}] FINAL best score = {final_score:.6f} "
          f"(length={len(best_seq)})")
    return best_seq.tolist()
