def local_search(a, budget):
    n = len(a)
    conv = np.convolve(a, a)
    sum_a = a.sum()
    cur_obj = 2 * n * conv.max() / (sum_a**2)
    best_obj = cur_obj
    best_seq = a.copy()
    start = time.time()
    step_factor = 0.2
    temperature = 1e-3  # small; maybe not used
    while time.time() < deadline:
        idx_max = np.argmax(conv)
        # compute gradient slice
        low = max(0, idx_max - (n-1))
        high = min(idx_max, n-1)
        # build weights
        weights = np.zeros(n)
        # for i in range(low, high+1):
        #   weights[i] = a[i] * a[idx_max - i]
        i_range = np.arange(low, high+1)
        j_range = idx_max - i_range
        weights[i_range] = a[i_range] * a[j_range]

        total_weight = weights.sum()
        if total_weight == 0 or np.isnan(total_weight):
            # fallback random source
            i_src = np.random.randint(n)
        else:
            # sample source according to weights
            probs = weights / total_weight
            i_src = np.random.choice(n, p=probs)

        # choose destination uniformly different
        j_dst = np.random.randint(n - 1)
        if j_dst >= i_src:
            j_dst += 1

        delta = step_factor * a[i_src]  # move fraction
        if delta <= 0:
            continue

        # compute new conv via copy and updates
        conv_new = conv.copy()
        # apply updates using original a (before shift)
        conv_new[j_dst:j_dst+n] += 2 * delta * a
        conv_new[i_src:i_src+n] -= 2 * delta * a
        conv_new[2*j_dst] += delta * delta
        conv_new[2*i_src] += delta * delta
        conv_new[i_src + j_dst] -= 2 * delta * delta

        # compute new objective
        new_obj = 2 * n * conv_new.max() / (sum_a**2)

        # accept if improvement; else maybe accept with small prob
        if new_obj < cur_obj or np.random.rand() < np.exp((cur_obj - new_obj) / (temperature + 1e-12)):
            # accept move
            a[i_src] -= delta
            a[j_dst] += delta
            conv = conv_new
            cur_obj = new_obj
            if new_obj < best_obj:
                best_obj = new_obj
                best_seq = a.copy()
        # optionally adapt step_factor or temperature
        # e.g., decay step_factor slowly
        step_factor *= 0.999999  # extremely slow decay to maintain small steps
        # optionally anneal temperature
        temperature *= 0.9999

    return best_seq, best_obj

    import time, numpy as np
    deadline = time.time() + budget_s - 0.1
    # set random seed using time for variability
    np.random.seed(int(time.time()) % 2**32)

    # Obtain initial sequence
    try:
        init_arr = np.array(height_sequence_1, dtype=float)
        n = len(init_arr)

        init_arr = init_arr / init_arr.sum()
    except NameError:
        n = 2500
        init_arr = np.random.dirichlet(np.ones(n))

    # Compute initial conv
    conv = np.convolve(init_arr, init_arr)
    sum_a = init_arr.sum()
    cur_obj = 2.0 * n * conv.max() / (sum_a**2)
    best_seq = init_arr.copy()
    best_val = cur_obj
    best_iter = 0

    # algorithm parameters
    step_factor = 0.2
    temperature = 0.001

    iter_cnt = 0
    last_improve_time = time.time()

    while time.time() < deadline:
        iter_cnt += 1
        # find index with maximum conv
        idx_max = int(conv.argmax())
        # compute weight array for source selection
        low = max(0, idx_max - (n - 1))

        contrib = init_arr[idx_i] * init_arr[idx_j]
        total = contrib.sum()

        else:

        j_dst = np.random.randint(n - 1)

def propose_candidate(budget_s: float = 1000):

    Starts from a known construction (height_sequence_1) or a random Dirichlet.
    Repeatedly selects a source index that contributes heavily to the current

    incrementally (O(n) per move) and moves are accepted if they do not increase
    the exact objective (or with a small probability for uphill steps).
    The best sequence found before the time budget expires is returned.
    """

    np.random.seed(int(time.time()) % (2**32 - 1))

    except Exception:
        n = 2500

    # algorithmic parameters
    step_factor = 0.5       # maximal fraction of source mass that may be moved
    temperature = 1e-3
    last_improve = time.time()

    accept_cnt = 0

    # ------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # index where convolution attains its maximum
        idx_max = int(conv.argmax())

        contrib = a[idx_i] * a[idx_j]
        total = contrib.sum()
        if total > 0.0:

        j_dst = np.random.randint(n - 1)

"""
Local‑search optimiser for ``evaluate_sequence``.

The algorithm works with a fixed length (default 2500).  It starts from the
pre‑computed construction ``height_sequence_1`` if it exists, otherwise from a
random Dirichlet vector (unit sum).  The self‑convolution is kept as a NumPy
array of length ``2·n‑1`` and updated **incrementally** when a small amount of
mass (Δ) is transferred from a source index *i* to a destination index *j*:

    conv←conv+2·Δ·shift(j)−2·Δ·shift(i)+Δ²·e_{2j}+Δ²·e_{2i}−2·Δ²·e_{i+j}

where ``shift(k)`` is the vector ``a`` placed at positions ``k … k+n‑1``.
Only O(n) work is required per move.

At each iteration the index where the convolution attains its maximum is
identified.  The source index is sampled proportionally to its contribution
to that peak (i.e. proportional to ``a[i]·a[idx_max‑i]``) – this concentrates
the search on the most influential coefficients.  A random fraction of the
source mass (up to a configurable *step_factor*) is moved to a uniformly
chosen destination.  Moves are accepted greedily or with a tiny uphill
probability (simulated‑annealing style).  The step size and a not‑zero
temperature are slowly decayed; after a period without improvement a small
random perturbation (re‑normalisation) is applied to escape plateaus.

The routine runs for at most ``budget_s`` seconds and returns the best
sequence (plain ``list[float]``) discovered.
"""

import time
import numpy as np

def propose_candidate(budget_s: float = 1000) -> list[float]:

    Search for a non‑negative step function that minimises
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence found.
    """
    # ------------------------------------------------------------------

    np.random.seed(int(time.time()) % (2**32 - 1))

    last_improve  = time.time()

    accept_cnt    = 0

        it += 1
