def convolve_fft(a):
    n = a.shape[0]
    conv_len = 2*n - 1
    fft_len = 1 << (conv_len-1).bit_length()  # next power of 2
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)[:conv_len]
    return conv

def cross_corr_fft(w, a):
    n = a.shape[0]
    fft_len = 1 << ((2*n-1 + n-1)-1).bit_length() # maybe 3n -2 length; but we can compute conv_len + n -1
    # simpler: pad to power of 2 of (len(w)+len(a)-1) which is (2n-1)+(n)-1 = 3n-2
    conv_len = len(w) + len(a) - 1
    fft_len = 1 << (conv_len-1).bit_length()
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = n-1
    return corr[start:start+n]

def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = np.inf

    lengths = [1024, 2048, 4096, 8192, 16384]
    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimizing length {n}")
        # candidate list
        candidates = []
        # warm start if matching length
        if best_seq is not None and len(best_seq)==n:
            candidates.append(np.array(best_seq))
        # global history
        hseq = globals().get('height_sequence_1')
        if hseq is not None:
            cand = np.asarray(hseq, dtype=float)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n-cand.shape[0]))
            cand = np.maximum(cand,0)
            if cand.sum()<0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)
        # uniform
        candidates.append(np.full(n, 1.0/n))
        # random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))
        # power law shapes
        idx = np.arange(n)+1
        prod = idx*(n-idx+1)
        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            shape = shape/shape.sum()
            candidates.append(shape)
        # arcsine shape
        i = np.arange(n)+0.5
        shape = 1/np.sqrt(i*(n+0.5-i))
        shape = shape/shape.sum()
        candidates.append(shape)
        # double Gaussian
        for gamma in [0.4,0.8,1.5,3.0]:
            i = np.arange(n)
            shape = np.exp(-gamma*i**2)+np.exp(-gamma*(n-1-i)**2)
            shape=shape/shape.sum()
            candidates.append(shape)
        # relaxed tent shape
        shape = np.concatenate([np.linspace(0,1, n//2, endpoint=False), np.linspace(1,0,n - n//2)])
        shape = shape/shape.sum()
        candidates.append(shape)

        # iterate over candidates
        for cand_id, a0 in enumerate(candidates):
            if time.time()>deadline: break
            a = a0.copy()
            # ensure non-negative and normalized
            a = np.maximum(a,0)
            total = a.sum()
            if total < 1e-12:
                a = np.random.dirichlet(np.ones(n))
                total = a.sum()
            a = a/total
            # symmetrize
            a = symmetrize(a)
            # Adam optimization
            logits = np.log(a + 1e-12)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1=0.9; beta2=0.999; eps=1e-8
            lr = 0.05
            lam = 0.2
            lam_factor = 1.04
            patience = time.time()
            max_iters = int(25000 * 1024 / n) + 5000
            for t in range(1, max_iters+1):
                if time.time()>deadline: break
                # compute a via softmax
                mx = logits.max()
                exp_logits = np.exp(logits - mx)
                a = exp_logits/exp_logits.sum()
                # symmetrize to keep mirror property
                a = symmetrize(a)
                # convolution
                conv = convolve_fft(a)
                max_c = conv.max()
                # smooth max weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(w, 1.0/w.size)
                else:
                    w /= w_sum
                # gradient
                grad = 2.0 * cross_corr_fft(w, a)
                # backprop through softmax
                avg = np.dot(a, grad)
                grad_log = a * (grad - avg)
                # Adam update
                m = beta1*m + (1-beta1)*grad_log
                v = beta2*v + (1-beta2)*(grad_log**2)
                m_hat = m/(1-beta1**t)
                v_hat = v/(1-beta2**t)
                logits -= lr * m_hat / (np.sqrt(v_hat)+eps)
                # temperature increase
                lam = min(lam*lam_factor, 1e6)
                # compute objective
                val = 2.0*n*max_c/(a.sum()**2)
                if val<best_val:
                    best_val = val
                    best_seq = a.copy()
                    patience = time.time()
                    print(f"[n={n} cand={cand_id} Adam {t}] new best {val:.6f}")
                # patience check
                if time.time() - patience > 5.0:
                    # random shake of logits
                    logits += (np.random.rand(n)-0.5)*1e-2
                    patience = time.time()
            # subgradient
            a = a/ a.sum()
            a = symmetrize(a)
            eta = 0.5
            sub_patience = time.time()
            for sub_it in range(3000):
                if time.time()>deadline: break
                conv = convolve_fft(a)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0/active.sum()
                else:
                    w = np.zeros_like(conv)
                grad = 2.0 * cross_corr_fft(w, a)
                a = a * np.exp(-eta*grad)
                # renormalize
                s = a.sum()
                if s<1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a/s

                conv = convolve_fft(a)

                    best_seq = a.copy()
                    sub_patience = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad {sub_it}] new best {val:.6f}")
                if time.time() - sub_patience > 5.0:
                    break

    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
    return best_seq.tolist()

def cross_corr_fft(w, a):
    n = a.shape[0]
    # Compute linear cross-correlation of w with a
    # Equivalent to convolution of w with reversed a.
    # length of w is 2n-1 (since conv of a with a is that length)
    conv_len = len(w) + n - 1   # w length + a length -1
    fft_len = 1 << (conv_len - 1).bit_length()
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)
    corr_full = np.fft.irfft(fw*fr, fft_len)
    start = n - 1
    return corr_full[start:start+n]  # length n

"""
Hybrid Adam‑subgradient optimizer for the max‑convolution objective.

The algorithm searches over several sequence lengths (1024 … 16384) and
over a diverse set of initial shapes (random Dirichlet, power‑law,
arcsine‑like, double‑Gaussian, linear “tent”, etc.).  For each candidate
the following two‑phase local optimisation is performed:

1. **Adam on a smooth‑max surrogate** – the max in the objective
   2·n·max(a*a)/(∑a)² is approximated by a soft‑max with a temperature
   λ that is increased geometrically.  The variables live on the
   probability simplex via the soft‑max (logits) parametrisation.
   The gradient of the surrogate is obtained from a weighted
   cross‑correlation (convolution * weighting*).  Symmetry
   a[i]=a[n‑1‑i] is enforced after each update.

2. **Exact sub‑gradient descent** – after the surrogate phase the true
   max is used.  The sub‑gradient is the cross‑correlation with a
   uniform distribution on the active (max‑attaining) convolution
   entries.  An exponentiated‑gradient step keeps the iterate non‑negative
   and on the simplex.

Throughout the search the best sequence (lowest exact objective) is
kept and returned.  The code respects a user‑supplied time budget and
uses only NumPy and the standard library (no external LP solver)."""

import time
import numpy as np

# ----------------------------------------------------------------------
def convolve_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑2 length."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    fft_len = 1 << (conv_len - 1).bit_length()   # next power of two
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:conv_len]
    return conv


def cross_corr_fft(w: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Both w and a are real vectors; w has length 2·n‑1, a has length n.
    Returns a vector of length n:
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    # length of the full linear convolution of w and reversed a
    conv_len = len(w) + n - 1          # = (2n‑1) + n‑1 = 3n‑2
    fft_len = 1 << (conv_len - 1).bit_length()
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(a[::-1], fft_len)   # reversed a
    corr_full = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr_full[start:start + n]   # slice of length n


def symmetrize(a: np.ndarray) -> np.ndarray:
    """Enforce a[i] = a[n‑1‑i] (mirror symmetry)."""
    return (a + a[::-1]) * 0.5


def objective(a: np.ndarray) -> float:
    """Exact evaluation of 2·n·max(a*a) / (sum a)²."""
    n = a.shape[0]
    conv = convolve_fft(a)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function sequence with a low
    evaluate_sequence() value.  The function respects the supplied time
    budget and returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # helper to possibly update the global best
    def maybe_update(a: np.ndarray, val: float):
        nonlocal best_seq, best_val
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"*** new global best: {val:.6f} (len={a.shape[0]})")

    # ------------------------------------------------------------------
    # list of lengths to explore (small to large)
    lengths = [1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        candidates = []

        # 1. Warm‑start from the current global best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.array(best_seq, dtype=np.float64))

        # 2. Historic solution (global variable height_sequence_1)
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            if cand.sum() < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / cand.sum()
            candidates.append(cand)

        # 3. Uniform distribution
        candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # 4. Random Dirichlet
        candidates.append(np.random.dirichlet(np.ones(n)))

        # 5. Power‑law edge‑concentrated shapes ((i+1)*(n-i))**α , α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                candidates.append(shape)

        # 6. Arcsine‑like shape (inverse‑square‑root, high near the edges)
        i = np.arange(n, dtype=np.float64) + 0.5
        shape = 1.0 / np.sqrt(i * (n + 0.5 - i))
        shape = shape / shape.sum()
        candidates.append(shape)

        # 7. Two‑Gaussian edge shapes
        for gamma in [0.4, 0.8, 1.5, 3.0]:
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape = shape / shape.sum()
            candidates.append(shape)

        # 8. Linear “tent” shape (rise then fall)
        half = n // 2
        left = np.linspace(0, 1, half, endpoint=False)
        right = np.linspace(1, 0, n - half)
        shape = np.concatenate([left, right])
        shape = shape / shape.sum()
        candidates.append(shape)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Start from a proper probability vector (non‑negative, sum=1)
            a = np.maximum(a0, 0.0)
            if a.sum() < 1e-12:
                a = np.random.dirichlet(np.ones(n))
            else:
                a = a / a.sum()
            a = symmetrize(a)

            # --------------------------------------------------------------
            # Phase 1 – Adam on smooth‑max surrogate
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.05
            lam = 0.2               # initial temperature
            lam_factor = 1.04
            patience_time = time.time()
            # iteration budget scales inversely with n (more steps for small n)
            max_iters = int(25000 * 1024 / n) + 5000

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                # soft‑max → simplex, then enforce symmetry
                mx = logits.max()
                exp_logits = np.exp(logits - mx)
                a = exp_logits / exp_logits.sum()
                a = symmetrize(a)

                # convolution and smooth‑max weights
                conv = convolve_fft(a)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (via cross‑correlation)
                grad_a = 2.0 * cross_corr_fft(w, a)

                # back‑prop through soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update on logits
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # increase temperature
                lam = min(lam * lam_factor, 1e6)

                # exact objective (cheap to compute)
                val = 2.0 * n * max_c / (a.sum() ** 2)
                maybe_update(a, val)

                # patience – shake if stuck for >5 s
                if time.time() - patience_time > 5.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    patience_time = time.time()

            # --------------------------------------------------------------
            # Phase 2 – Sub‑gradient descent on the true max
            a = a / a.sum()
            a = symmetrize(a)
            eta = 0.5
            sub_patience = time.time()
            for sub_it in range(3000):
                if time.time() > deadline:
                    break

                conv = convolve_fft(a)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
                if active.any():
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()
                else:
                    w = np.zeros_like(conv)

                # sub‑gradient of the true max
                grad_a = 2.0 * cross_corr_fft(w, a)

                # exponentiated‑gradient step (keeps non‑negative)
                a = a * np.exp(-eta * grad_a)

                # renormalise, guard against collapse
                s = a.sum()
                if s < 1e-12:
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s
                a = symmetrize(a)

                conv = convolve_fft(a)
                max_c = conv.max()
                val = 2.0 * n * max_c / (a.sum() ** 2)
                maybe_update(a, val)

                if time.time() - sub_patience > 5.0:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (unlikely)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return as plain Python list (required by the external evaluator)
    return best_seq.tolist()
