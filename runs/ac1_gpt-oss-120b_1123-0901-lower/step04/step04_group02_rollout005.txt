def _next_pow2(x: int) -> int:
    """Smallest power of two  ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Result has length len(a) and corresponds to
        (w ★ a)[i] = Σ_k w[k]·a[i + k - (n‑1)],   i = 0 … n‑1 .
    """
    n = a.shape[0]
    rev_a = a[::-1]

    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _smooth_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """
    Soft‑max (smooth‑max) weights for a convolution vector.
    w_i ∝ exp(lam·(conv_i – max(conv))).
    """
    max_c = conv.max()
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0.0:
        # fall back to a uniform distribution (should hardly happen)
        return np.full_like(conv, 1.0 / conv.size)
    return w / w_sum


def _eg_update(a: np.ndarray, grad: np.ndarray, eta: float) -> np.ndarray:

    Exponentiated‑gradient step on the simplex.
    a ← a ⊙ exp(–η·grad)   ;   finally renormalise to sum = 1.

    if s == 0.0:

        a = np.full_like(a, 1.0 / a.size)

    if s == 0.0:

                eta0 = 0.6                       # initial step‑size
                lam = 0.5                        # smooth‑max temperature
                lam_factor = 1.35                # geometric increase
                max_iters = max(2000, int(3e6 / (n * math.log2(n + 1))))
                # ensure we do not waste time on the very largest lengths
                max_iters = min(max_iters, 20000)

                last_improve = time.time()

                for it in range(1, max_iters + 1):
                    if time.time() > deadline:
                        break

                    # ----- convolution and smooth‑max weights -----
                    conv = _conv_fft(a, fft_len_conv)
                    w = _smooth_weights(conv, lam)

                    # ----- gradient of the surrogate -----
                    grad = 2.0 * _corr_fft(w, a, fft_len_corr)

                    # ----- EG update (learning‑rate decays like 1/√t) -----
                    eta = eta0 / math.sqrt(it)
                    a = _eg_update(a, grad, eta)

                    # ----- temperature schedule -----
                    lam = min(lam * lam_factor, 1e6)

                    # ----- monitor exact objective every few iterations -----
                    if it % 250 == 0 or it == max_iters:
                        _monitor(a)

                    # ----- stagnation handling -----
                    if time.time() - last_improve > 8.0:
                        # small random “shake’’ to escape flat regions
                        noise = (np.random.rand(n) - 0.5) * 1e-3
                        a = a + noise
                        a = np.maximum(a, 0.0)
                        s = a.sum()
                        a = a / s if s > 0 else np.full(n, 1.0 / n)
                        last_improve = time.time()
                        print(f"[restart {restart_idx} it={it}] shake")
    except StopIteration:
        # early exit when target reached
        pass

    # ------------------------------------------------------------------
    # Final local random refinement (tiny mass transfers)
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Local refinement phase ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move at most 0.5 % of the total mass from i to j
            delta = 0.005 * a[i] * np.random.rand()
            if delta <= 0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            if a_prop[i] < 0.0:
                continue
            a_prop = a_prop / a_prop.sum()
            val = _exact_objective(a_prop)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 2048
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_objective(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    # Return as plain Python list as required by the outer evaluator.
    return best_seq.tolist()
