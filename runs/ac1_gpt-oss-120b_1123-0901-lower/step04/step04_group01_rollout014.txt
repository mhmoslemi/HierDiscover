def lp_mix(a):
    n = len(a)
    M = full_convolution(a, fft_len_conv).max()
    # Build A matrix as dense? Could be large but we can build sparse.
    rows = []
    cols = []
    data = []
    for k in range(2*n-1):
        # range for i from max(0, k-(n-1)) to min(n-1,k)
        i_min = max(0, k-(n-1))
        i_max = min(n-1, k)
        for i in range(i_min, i_max+1):
            # second index j = k - i
            rows.append(k)
            cols.append(i)
            data.append(a[k-i])   # f_j = a_{k-i}
    A = scipy.sparse.csr_matrix((data, (rows, cols)), shape=(2*n-1, n))
    b = np.full(2*n-1, M)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        return a
    g = res.x
    if g.sum() == 0:
        return a
    g_norm = g / g.sum()
    # mixing
    tau = 0.5
    a_new = (1 - tau) * a + tau * g_norm
    # renorm
    a_new = np.maximum(a_new, 0)
    a_new /= a_new.sum()
    return a_new

"""
propose_candidate – search for a non‑negative step‑function that minimizes

    F(a) = 2·n·max(a * a) / (sum(a)²)

The algorithm builds on three ideas:

1. **Projected Adam on the simplex** – parameters are logits `x`,
   `softmax(x)` gives a probability vector `a`.  The Adam optimiser
   works on the logits, automatically respecting non‑negativity and the
   normalization ∑a = 1.

2. **Smooth max‑approximation** – instead of using a hard active set,
   we weight the convolution lags by a log‑sum‑exp kernel.
   `w_k ∝ exp(β·(conv_k‑max(conv)))`.  The temperature `β` grows
   slowly, starting with a very smooth objective and ending with an
   almost exact max.  This yields a more stable descent and avoids
   premature clipping to a tiny active set.

3. **Local simulated‑annealing refinement** – after the Adam phase we
   perform a cheap annealing walk that moves a small amount of mass
   between two random indices.  Accepted moves either improve the
   objective or are accepted with probability `exp(Δ/τ)` where the
   temperature `τ` decays exponentially.

The search proceeds over several problem sizes (512 … 4096).  For each
size a few restarts are performed; warm‑starts use the best sequence
found so far or the globally supplied `height_sequence_1`.  Progress
(and any new best value) is printed.  The routine stops after the
budget (`budget_s`) expires and returns the best sequence found as a
plain Python list.
"""

import time
import numpy as np
from math import log, exp

# ----------------------------------------------------------------------
# Helper functions (no closures, no lambdas, top‑level as required)
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using the FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]

# Main optimisation entry point
# ----------------------------------------------------------------------
