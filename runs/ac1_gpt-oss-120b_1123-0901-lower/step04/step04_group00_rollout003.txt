def refine_with_adam(a0, n, fft_len, max_iters, deadline):
    a = a0.copy()
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.01
    best_local_val = evaluate_objective(a, n, fft_len)
    best_local_a = a.copy()
    last_improve = time.time()
    for t in range(1, max_iters+1):
        # compute softmax
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()
        # conv
        conv = full_convolution(a, fft_len)
        max_c = conv.max()
        # surrogate weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum
        # gradient
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)  # need corr length maybe compute accordingly
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature
        lam = min(lam * lam_factor, 1e7)  # high max
        # evaluate exact
        val = evaluate_objective(a, n, fft_len)
        if val < best_local_val:
            best_local_val = val
            best_local_a = a.copy()
            last_improve = time.time()
        if time.time() - last_improve > 2.0:
            # random perturbation
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
        if time.time() > deadline:
            break
    return best_local_a, best_local_val

def cross_correlation(w,a, fft_len_corr):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len_corr)
    fr = np.fft.rfft(rev_a, fft_len_corr)
    corr = np.fft.irfft(fw * fr, fft_len_corr)
    start = n-1
    return corr[start:start+n]

"""
Propose a step‑function (non‑negative vector) that minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

using a two‑stage strategy:

1.  Fast parametric search.
    * Several families of analytic shapes (power‑law, sine‑power,
      exponential‑edge, Gaussian‑edge, product‑powers) are sampled on a
      grid of parameters.
    * A few random Dirichlet draws and the historic “height_sequence_1”
      (if present) are also added.
    * Each candidate is normalised to sum = 1 and evaluated exactly with a
      FFT based convolution.  The best K≈20 candidates are kept.

2.  Local refinement.
    * The retained candidates are optimised with Adam on a smooth‑max
      surrogate of the convolution, keeping the vector in the simplex by
      the soft‑max trick.
    * After the Adam phase a few exponentiated‑gradient sub‑gradient
      steps on the true max are performed.
    * The best vector found is kept as the global solution.

If time permits the solution for n=4096 is up‑sampled (linear
interpolation) to larger lengths (8192, 16384) and refined again with a
smaller number of iterations – larger n often yields a slightly lower
constant in the objective.

The function respects the supplied time budget (default 1000 s) and
returns a plain Python list of floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for gradients.

    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


def _objective(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²)."""
    conv = _full_convolution(a, fft_len_conv)
    max_c = conv.max()
    s = a.sum()
    # protect against division by zero (should never happen)
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def _generate_candidates(n: int, best_vec: np.ndarray | None) -> list[np.ndarray]:
    """
    Produce a list of candidate vectors (not normalised yet) for length n.
    Includes warm‑starts, analytic families and random draws.
    """
    candidates = []

    # warm start from the global best (if length matches)
    if best_vec is not None and best_vec.shape[0] == n:
        candidates.append(best_vec.copy())

    # historic solution if present
    hist = globals().get("height_sequence_1")
    if hist is not None:
        h = np.asarray(hist, dtype=np.float64)
        if h.shape[0] > n:
            h = h[:n]
        elif h.shape[0] < n:
            h = np.pad(h, (0, n - h.shape[0]))
        candidates.append(h)

    # uniform (useful as baseline)
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)        # (i+1)*(n-i)

    # ---------- power‑law edge‑concentrated shapes ----------
    for beta in np.linspace(-5.0, 2.0, 71):          # fine grid
        shape = np.where(prod > 0,
                         np.power(prod, beta, out=np.zeros_like(prod)),
                         0.0)
        if shape.sum() > 0:
            candidates.append(shape)

    # ---------- sine‑power shapes (zero at edges, peak at centre) ----------
    angle = np.pi * (idx - 0.5) / n
    for p in np.linspace(0.5, 6.0, 56):
        shape = np.sin(angle) ** p
        if shape.sum() > 0:
            candidates.append(shape)

    # ---------- exponential edge shapes ----------
    for a in np.linspace(0.01, 5.0, 50):
        shape = np.exp(-a * np.arange(n)) + np.exp(-a * np.arange(n - 1, -1, -1))
        candidates.append(shape)

    # ---------- Gaussian edge shapes ----------
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        candidates.append(shape)

    # ---------- product‑power shapes ----------
    for p in np.linspace(0.0, 2.0, 5):
        for q in np.linspace(0.0, 2.0, 5):
            shape = (idx ** p) * ((n - idx + 1) ** q)
            candidates.append(shape)

    # ---------- random Dirichlet draws ----------
    rng = np.random.default_rng()
    for _ in range(30):
        cand = rng.dirichlet(np.ones(n))
        candidates.append(cand)

    return candidates


def _refine_adam(a0: np.ndarray,
                 n: int,
                 fft_len_conv: int,
                 fft_len_corr: int,
                 max_iters: int,
                 deadline: float,
                 global_best_val: float,
                 global_best_seq: np.ndarray | None):
    """
    Adam on smooth‑max surrogate, followed by a short sub‑gradient phase.
    Returns the (possibly) improved vector and its objective value.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5                # temperature for smooth‑max
    lam_factor = 1.02

    best_val = _objective(a, n, fft_len_conv)
    best_a = a.copy()
    last_improve = time.time()

    for t in range(1, max_iters + 1):
        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w.fill(1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # raise temperature
        lam = min(lam * lam_factor, 1e6)

        # evaluate exact objective
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # also update global best if needed
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[n={n} Adam] new global best {val:.6f}")

        # stagnation handling – occasional shake
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

        if time.time() > deadline:
            break

    # ---- Sub‑gradient stage (exponentiated gradient) ----
    a = best_a / best_a.sum()
    eta = 0.6
    sub_last_improve = time.time()
    for sub_it in range(1500):
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            sub_last_improve = time.time()
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[n={n} Subgrad] new global best {val:.6f}")

        if time.time() - sub_last_improve > 3.0:
            break
        if time.time() > deadline:
            break

    return best_a, best_val, global_best_seq, global_best_val


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector onto a larger size."""
    old_n = seq.shape[0]
    old_x = np.arange(old_n, dtype=np.float64)
    new_x = np.linspace(0, old_n - 1, new_n, dtype=np.float64)
    up = np.interp(new_x, old_x, seq)
    # re‑normalise (tiny numerical drift may appear)
    up = np.maximum(up, 0.0)
    if up.sum() == 0.0:
        up = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        up = up / up.sum()
    return up

            a_big,
