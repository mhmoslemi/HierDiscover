"""
Propose a step‑function (non‑negative vector) that minimizes the
``evaluate_sequence`` objective.

Method
------
* Build a diverse pool of initial vectors for each problem size.
  Besides the usual uniform/Dirichlet/exponential windows we also add
  several Beta‑distribution shapes (including the arcsine shape
  β(0.5,0.5)) and a simple triangular ramp.  All vectors are projected on
  the ℓ₁‑sphere Σ aᵢ = √(2 n).

* For every start we run a *projected Adam* optimiser.
  The non‑smooth max‑convolution is approximated by a soft‑max with a
  temperature τ that decays slowly (τ ≈ 0.3·M·exp(‑t·5/|max_iters|)).
  When τ becomes tiny we fall back to the exact sub‑gradient (uniform
  weights on the active set).

* Adam’s learning‑rate schedule is slightly more aggressive than the
  previous baseline (initial lr = 0.07, decay ∝1/√t) and we jitter the
  iterate every 2000 steps to escape shallow plateaus.

* The whole search respects the 1000 s wall‑clock limit.  It explores
  three problem sizes (≈2100, ≈4000 and ≈8000) and keeps the globally best
  sequence (measured by the exact objective, i.e. the max of the
  self‑convolution after normalisation).

The function ``propose_candidate`` implements the whole pipeline and
returns a plain Python list of floats ready for ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities for the smoothed max‑convolution
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values.
    Returns a weight vector w (summing to 1) and the smoothed maximum Lτ.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i}   (indices outside the range ignored)
    using FFTs.  This is the (sub‑)gradient of the (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    # reverse the weight vector before FFT
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the relevant part (length = len(w)+len(a)-1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # The slice that maps to rev_i for i=0..n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Exact max of a∗a using an FFT of length Nfft."""
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * a.shape[0] - 1]
    return float(conv.max())


# ----------------------------------------------------------------------
# Adam optimiser with soft‑max temperature schedule
# ----------------------------------------------------------------------


def _adam_optimize(
    a: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Run a projected Adam optimiser on a single start.
    Returns the best sequence (as a NumPy array) discovered in this run
    together with its exact objective value (max convolution after normalisation).
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector to satisfy the ℓ₁‑constraint
    a = a * (target_sum / a.sum())

    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.07                       # a bit larger than the previous 0.05
    jitter_interval = 2000

    # Temperature schedule parameters
    tau_factor = 0.30                # start τ = τ_factor·M
    decay_rate = max_iters / 5.0     # slower decay → smoother landscape longer

    # FFT length (next power‑of‑two ≥ 2·n−1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    for iteration in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ---- exact self‑convolution ---------------------------------
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ---- temperature and soft‑max -------------------------------
        tau = tau_factor * M * math.exp(-iteration / decay_rate)

        if tau < 1e-10:
            # Switch to exact sub‑gradient (uniform weight on the active set)
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ---- gradient of the smoothed max ---------------------------
        rev = _weighted_rev(a, w, Nfft)

        # The exact objective after normalisation is simply M.
        # The gradient of 2 n·max_b / Σa² (with Σa = √(2 n)) simplifies to:
        coeff = 4.0 * n / (target_sum * target_sum)          # = 2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # projects onto Σa = const
        grad = coeff * rev - const

        # ---- Adam update --------------------------------------------
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        # Learning‑rate decay ~ 1/√t
        lr = lr0 / (1.0 + 1e-4 * math.sqrt(iteration))
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ---- projection onto the feasible set -----------------------
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0.0:
            a = a * (target_sum / s)
        else:
            a = np.full_like(a, target_sum / n)

        # ---- occasional jitter ---------------------------------------
        if iteration % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = np.maximum(a + jitter, 0.0)
            a = a * (target_sum / a.sum())

        # ---- record improvement --------------------------------------
        cur_score = M
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(
                f"[len {n:5d} iter {iteration:8d}]  new best = {best_score:.6f}"
            )

    return best_seq, best_score

def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:

    black = (

    init.append(black / black.sum() * target_sum)

    # 5) Beta‑distribution shapes (including the arcsine β=0.5,0.5)
    betas = [
