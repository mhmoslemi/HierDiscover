"""
Optimisation via Simulated‑Annealing + a quick sub‑gradient finetune.

* Build a handful of diverse initial vectors (uniform, Dirichlet,
  cosine‑U‑shape, Hann, Blackman, small perturbations).
* For each initialisation run a simple simulated‑annealing loop that
  preserves the ℓ₁‑norm (∑a = √(2 n)) by moving a random amount of mass
  from one index to another.  The temperature is slowly decreased; uphill
  moves are accepted with probability exp(−Δ/τ).  The convolution maximum
  (i.e. the evaluation objective after normalisation) is computed with a
  single FFT, which is fast enough to allow >10⁵ proposals per second.
* The best sequence seen so far is kept globally.  After the annealing
  phase a very short Adam‑style sub‑gradient descent (using the exact
  max‑convolution gradient) cleans up any remaining noise.
* The whole routine respects the 1000 s wall‑clock limit and returns the
  best vector as a plain Python list.
"""

import time
import math
import numpy as np


def _max_convolution(a: np.ndarray, fft_len: int) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of length fft_len."""
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * a.size - 1]
    return float(conv.max())


def _project_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project onto the set {x ≥ 0,  ∑x = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        # extremely unlikely, but keep feasibility
        a = np.full_like(a, target_sum / a.size)
    return a


def _subgradient_maxconv(a: np.ndarray, fft_len: int, eps: float = 1e-12):
    """
    Compute a (sub‑)gradient of max_{k}(a∗a)[k].
    The active set is all indices whose value is within eps of the maximum.
    """
    A = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(A * A, n=fft_len)[: 2 * a.size - 1]
    M = conv.max()
    active = np.where(conv >= M - eps)[0]

    # Gradient of conv_k w.r.t a is 2 * reversed a shifted by k
    # We average over the active set.
    grad = np.zeros_like(a)
    n = a.size
    for k in active:
        # contribution of index k:
        # grad_j = a_{k-j} (if 0 ≤ k-j < n) plus a_{j-(k-n)} when k-j<0?
        # The easiest way: compute reverse convolution of the indicator.
        # Here we use a direct loop (fast because active set is tiny).
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        # for j in [lo, hi]: term a_{k-j}
        grad[lo:hi + 1] += a[k - np.arange(lo, hi + 1)]
    grad *= 2.0 / max(1, active.size)   # average
    return grad, M


def _adam_refine(a: np.ndarray, target_sum: float, deadline: float,
                 max_iters: int = 20000) -> np.ndarray:
    """
    A few Adam steps using the exact sub‑gradient of the max‑convolution.
    This cleans up the final annealed candidate.
    """
    n = a.size
    fft_len = 1 << ((2 * n - 1).bit_length())
    # Adam state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break
        grad, _ = _subgradient_maxconv(a, fft_len)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        lr = lr0 / (1.0 + 1e-4 * it)
        a -= lr * m_hat / (np.sqrt(v_hat) + eps)
        a = _project_simplex(a, target_sum)
    return a


def propose_candidate(seed: int = 12345, budget_s: float = 1000, **_kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` (i.e. 2 n·max(a∗a) / (∑a)²).
    Returns the best sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Base length (use previously discovered one if available)
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
        n0 = base.shape[0]
    except Exception:
        base = None
        n0 = 2100                     # fallback length used previously

    # ------------------------------------------------------------------
    # Search parameters
    lengths = [n0]                    # we concentrate on a single size for speed
    best_score = float("inf")
    best_seq = None

    # ------------------------------------------------------------------
    # Helper to build diverse initial vectors
    def _make_initials(n: int, target: float):
        rng = np.random.default_rng()
        init = []

        # 1) uniform
        init.append(np.full(n, target / n))

        # 2) random Dirichlet
        g = rng.gamma(1.0, 1.0, size=n)
        init.append(g / g.sum() * target)

        # 3) cosine‑U‑shape (high at ends, low in centre)
        u = 1.0 + np.cos(np.pi * np.arange(n) / (n - 1))
        init.append(u / u.sum() * target)

        # 4) reversed U‑shape (identical to 3 but shuffled)
        init.append(u[::-1] / u.sum() * target)

        # 5) Hann window
        hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
        init.append(hann / hann.sum() * target)

        # 6) Blackman window
        a0, a1, a2 = 0.42, 0.5, 0.08
        idx = np.arange(n)
        black = (a0
                 - a1 * np.cos(2.0 * np.pi * idx / (n - 1))
                 + a2 * np.cos(4.0 * np.pi * idx / (n - 1)))
        init.append(black / black.sum() * target)

        # 7) small Gaussian perturbation of uniform
        pert = np.full(n, target / n)
        pert += rng.normal(scale=0.02 * target / n, size=n)
        init.append(_project_simplex(pert, target))

        # If we have a base of different size, interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            init.append(_project_simplex(interp, target))

        return init

    # ------------------------------------------------------------------
    # Main optimisation loops
    for n in lengths:
        if time.time() >= deadline:
            break

        target_sum = math.sqrt(2.0 * n)
        fft_len = 1 << ((2 * n - 1).bit_length())
        rng = np.random.default_rng()

        # Pre‑compute a cheap function for the objective (max convolution)
        def max_conv(a_vec):
            return _max_convolution(a_vec, fft_len)

        initial_vectors = _make_initials(n, target_sum)

        for start_id, a0 in enumerate(initial_vectors, 1):
            if time.time() >= deadline:
                break

            a = a0.copy()
            cur_score = max_conv(a)

            # ------- simulated annealing ----------
            # temperature proportional to the current score
            T = cur_score * 0.3
            Tmin = 1e-6
            cooling = 0.9995
            # step scale (max fraction of transferred mass)
            max_step_frac = 0.25

            # estimate how many proposals we can afford for this start
            # (roughly 0.1 ms per proposal → 10 k per second)
            max_proposals = 200_000

            for it in range(max_proposals):
                if time.time() >= deadline:
                    break
                # propose a move that preserves the ℓ₁‑norm
                i, j = rng.integers(0, n, size=2)
                if i == j:
                    continue
                if a[i] <= 0.0:
                    continue
                # magnitude of transfer – scaled with temperature
                step_frac = max_step_frac * (T / (cur_score * 0.3 + 1e-12))
                delta = step_frac * a[i] * rng.random()
                if delta <= 0.0:
                    continue

                a_new = a.copy()
                a_new[i] -= delta
                a_new[j] += delta
                new_score = max_conv(a_new)
                diff = new_score - cur_score

                if diff < 0.0 or rng.random() < math.exp(-diff / max(T, 1e-12)):
                    # accept
                    a = a_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = a.copy()
                        print(f"[len {n:5d}] new global best = {best_score:.6f}")

                # annealing schedule
                T = max(T * cooling, Tmin)

                # occasional tiny jitter to avoid stagnation
                if (it + 1) % 5000 == 0:
                    jitter = rng.normal(scale=1e-4 * target_sum / n, size=n)
                    a = _project_simplex(a + jitter, target_sum)
                    cur_score = max_conv(a)

                # stop early if temperature is already negligible
                if T <= Tmin:
                    break

            # ------- quick Adam refinement ---------
            # run a few Adam steps on the best candidate of this start
            a = _adam_refine(a, target_sum, deadline, max_iters=5000)
            final_score = max_conv(a)
            if final_score < best_score:
                best_score = final_score
                best_seq = a.copy()
                print(f"[len {n:5d}] post‑refine best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _max_convolution(best_seq, 1 << ((2 * n - 1).bit_length()))
        print("[fallback] using uniform sequence")

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
