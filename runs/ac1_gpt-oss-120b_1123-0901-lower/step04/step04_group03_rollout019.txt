def solve_lp_for_g(a, M):
    n = a.shape[0]
    k_len = 2*n - 1
    # Build A matrix dense
    A = np.zeros((k_len, n), dtype=np.float64)
    for j in range(n):
        A[j:j+n, j] = a
    # Linear program
    c = -np.ones(n)
    bounds = [(0, None)] * n
    b_ub = np.full(k_len, M, dtype=np.float64)
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to uniform
        return np.full(n, 1.0/n)
    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s <= 0:
        return np.full(n, 1.0/n)
    return b / s
```

def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the self‑convolution of a."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) / (sum(a)²).  a is assumed non‑negative."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient for the Adam phase
# ----------------------------------------------------------------------
def grad_submax(a: np.ndarray):
    """Sub‑gradient of max(a∗a) w.r.t. a, returns (g, max_value)."""
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick one maximal index uniformly at random
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    # g_j = 2 * a_{k‑j} whenever the index exists
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax – maps arbitrary logits to a probability vector."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the simplex
# ----------------------------------------------------------------------
def run_adam_opt(n: int, init_a: np.ndarray, deadline: float, max_iters: int = 200_000):
    """Adam optimisation of the sub‑gradient while staying on the simplex."""
    # initialise logits
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.07                # a bit larger than before

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # simplex
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient
        val = 2.0 * n * max_c               # objective (sum = 1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        x = np.clip(x, -50.0, 50.0)   # avoid overflow

        if time.time() - last_improve > 5.0:
            # small random shake when we are stuck
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def solve_lp_for_g(f: np.ndarray, M: float):
    """
    Solve  max Σ b_j   s.t.  b ≥ 0, (f * b)_k ≤ M   (k = 0…2n‑2).
    Returns a normalized probability vector g (=b / Σb).
    """
    n = f.shape[0]
    k_len = 2 * n - 1

    # Build dense constraint matrix A where A[k, j] = f[k‑j] (or 0 outside)
    A = np.zeros((k_len, n), dtype=np.float64)
    # column j contributes f[0..n‑1] to rows j … j+n‑1
    for j in range(n):
        A[j : j + n, j] = f

    c = -np.ones(n)                     # maximise Σb  → minimise –Σb
    bounds = [(0, None)] * n
    b_ub = np.full(k_len, M, dtype=np.float64)

    # high‑s than as default solver
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # fallback to a simple uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)

    b = np.maximum(res.x, 0.0)
    s = b.sum()
    if s <= 0.0:
        return np.full(n, 1.0 / n, dtype=np.float64)

    return b / s


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy – final fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float, max_iter: int = 30_000):
    """Simple (1+1) ES on the simplex."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                       # success – move centre
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val

    best_val = np.inf

            if hs.size >= n:

            s = hs.sum()
            if s > 1e-12:

            else:

            init_pool.append(hs)

        # 3. Simple decreasing linear profile
        dec = np.arange(n, 0, -1, dtype=np.float64)
        dec /= dec.sum()
        init_pool.append(dec)

        # 4. Uniform
        init_pool.append(np.full(n, 1.0 / n, dtype=np.float64))

        # Keep the best of the pool as the starting point for the subsequent phases
        cur_a = None
        cur_val = np.inf
        for cand in init_pool:
            v = compute_objective(cand)
            if v < cur_val:

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\n=== Finished ===\nBest value found: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain list (required by the external evaluator)
    return best_seq.tolist()
