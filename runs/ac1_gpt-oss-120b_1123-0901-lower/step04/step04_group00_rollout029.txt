"""
Propose a step‑function (non‑negative list) that approximately minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where “a*a” is the linear convolution (numpy.convolve).
The implementation follows the ideas from the reference code but adds
   * a richer set of initial guesses (including spiky Dirichlet samples
     and two‑sided exponential shapes),
   * a much longer Adam optimisation phase,
   * a more aggressive temperature schedule for the smooth‑max,
   * an extended exponentiated‑gradient (sub‑gradient) phase,
   * a final fine‑tuning pass on the best candidate found for each length.
The routine respects a user‑supplied time budget (default 1000 s) and
returns the best sequence discovered as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of the smooth‑max.
    Returns a vector of length len(a) equal to

        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]

    where n = len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Lengths for the search – the larger the length the more expressive
    # the sequence, but also the slower the FFTs.
    lengths = [2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths that are powers of two (fast for FFTW/NumPy)
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # For cross‑correlation we need conv_len + n – 1 ≤ 3·n
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Helper that computes the exact objective for a *probability* vector.
    # Because we keep the sum equal to 1, the denominator disappears.
    def _exact_obj(a: np.ndarray, n: int) -> float:
        conv = _full_convolution(a, fft_len_conv[n])
        return 2.0 * n * conv.max() / (a.sum() ** 2)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        conv_len = fft_len_conv[n]
        corr_len = fft_len_corr[n]

        # --------------------- build a list of initial candidates ----------
        candidates = []

        # warm start from the current best (if length matches)
        if best_seq is not None and len(best_seq) == n:
            candidates.append(np.asarray(best_seq, dtype=np.float64))

        # historic solution if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            cand = np.asarray(hseq, dtype=np.float64)
            if cand.shape[0] > n:
                cand = cand[:n]
            else:
                cand = np.pad(cand, (0, n - cand.shape[0]))
            cand = np.maximum(cand, 0.0)
            s = cand.sum()
            if s < 0.01:
                cand = np.random.dirichlet(np.ones(n))
            else:
                cand = cand / s
            candidates.append(cand)

        # pure random Dirichlet (concentration 1)
        candidates.append(np.random.dirichlet(np.ones(n)))

        # spiky Dirichlet (concentration <1) – tends to concentrate mass
        for concentration in [0.2, 0.5, 0.8]:
            candidates.append(np.random.dirichlet(np.full(n, concentration)))

        # Edge‑concentrated power‑law shapes ((i+1)*(n-i))**α, α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.1, 13):
            shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # Two‑sided exponential shape:
        # a_i ∝ exp(-γ·i) + exp(-γ·(n‑1‑i))
        for gamma in [0.0005, 0.001, 0.002, 0.005]:
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * i) + np.exp(-gamma * (n - 1 - i))
            shape /= shape.sum()
            candidates.append(shape)

        # Slightly smoothed “tent’’ shape (linear rise then fall)
        left = np.linspace(0, 1, n // 2, endpoint=False)
        right = left[::-1]
        tent = np.concatenate([left, right])
        tent = np.maximum(tent, 0)
        tent /= tent.sum()
        candidates.append(tent)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # Normalise to a probability vector (sum = 1)
            a = a0 / a0.sum()
            logits = np.log(a + 1e-15)                 # soft‑max parameters

            # ---- Adam on a smooth‑max surrogate of the max convolution -----
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.06                                 # a tad larger learning rate
            lam = 0.3                                 # start temperature
            lam_factor = 1.015                         # slower increase -> smoother

            last_improve = time.time()
            # iteration budget grows for larger n, but we cap it to keep time under control
            max_iters = int(25000 * 1024 / n) + 5000

            for t in range(1, max_iters + 1):
                # soft‑max ⇒ simplex
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                # convolution and smooth‑max weights
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # gradient of the surrogate (thanks to linearity of convolution)
                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                # back‑prop through the soft‑max
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                # Adam update
                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                # temperature schedule (smooth‑max becomes closer to true max)
                lam = min(lam * lam_factor, 1e6)

                # exact objective (a.sum() == 1)
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Adam iter={t:05d}] new best {val:.6f}")

                # Stagnation handling – a small random shake
                if time.time() - last_improve > 8.0:
                    logits += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()

                if time.time() > deadline:
                    break

            # ---- Sub‑gradient phase on the exact max (exponentiated gradient) ----
            a = a / a.sum()
            eta = 0.55                                 # step size for EG
            sub_last_improve = time.time()
            max_sub_iters = 5000

            for sub_it in range(max_sub_iters):
                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

                if not active.any():
                    w = np.zeros_like(conv)
                else:
                    w = np.zeros_like(conv)
                    w[active] = 1.0 / active.sum()   # uniform over the active set

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)

                a = a * np.exp(-eta * grad_a)         # exponentiated‑gradient step
                s = a.sum()
                if s < 1e-12:                         # protect against collapse
                    a = np.random.dirichlet(np.ones(n))
                    s = a.sum()
                a = a / s

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                val = 2.0 * n * max_c

                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last_improve = time.time()
                    print(f"[n={n} cand={cand_id} Subgrad iter={sub_it}] new best {val:.6f}")

                if time.time() - sub_last_improve > 8.0:
                    break
                if time.time() > deadline:
                    break

        # ------------------------------------------------------------------
        # Optional fine‑tuning of the *global* best for this length
        if best_seq is not None and len(best_seq) == n:
            print(f"[n={n}] fine‑tuning the current best (val={best_val:.6f})")
            a = best_seq.copy()
            logits = np.log(a + 1e-15)
            m = np.zeros_like(logits)
            v = np.zeros_like(logits)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.04
            lam = 0.5
            lam_factor = 1.01
            last_improve = time.time()
            fine_iters = int(40000 * 1024 / n) + 5000

            for t in range(1, fine_iters + 1):
                mx = logits.max()
                exp_x = np.exp(logits - mx)
                a = exp_x / exp_x.sum()

                conv = _full_convolution(a, conv_len)
                max_c = conv.max()
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                w = w / w_sum if w_sum != 0 else np.full_like(w, 1.0 / w.size)

                grad_a = 2.0 * _cross_correlation(w, a, corr_len)
                avg = np.dot(a, grad_a)
                grad_log = a * (grad_a - avg)

                m = beta1 * m + (1.0 - beta1) * grad_log
                v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

                lam = min(lam * lam_factor, 1e6)

                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} fine‑tune iter={t:05d}] new best {val:.6f}")

                if time.time() - last_improve > 10.0:
                    # final small random shake before convergence
                    logits += (np.random.rand(n) - 0.5) * 5e-3
                    last_improve = time.time()

                if time.time() > deadline:
                    break

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    # Return a plain Python list as required by the external evaluator
    return best_seq.tolist()
