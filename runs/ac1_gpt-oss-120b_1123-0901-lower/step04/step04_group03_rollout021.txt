"""step‑function optimiser

This module searches a non‑negative step‑function (list of heights) that
minimises the evaluation function

    F(a) = 2·n·max(a∗a) / (sum(a)²)

where “∗” is the linear convolution.  The search consists of three
independent ideas that complement the earlier ADAM+DE+ES pipeline:

1. **Parametric scans** – cheap scans over simple families (power‑law
   and exponential decay) give good starting points for the subsequent
   optimisation.
2. **Smooth‑max Adam** – we replace the non‑differentiable
   max‑operator by a temperature‑controlled log‑sum‑exp
   (soft‑max) and perform Adam on the resulting smooth objective.
   The temperature (α) is doubled every few thousand iterations,
   gradually turning the soft‑max into the true maximum.
3. **Fine‑tuning (1+1) ES)** – a short evolution‑strategy run refines the
   best sequence found so far.

All heavy work (FFT based convolutions and correlations) is O(n log n)
and comfortably fits into the 1000 s budget.  The algorithm respects the
time limit, prints progress, and finally returns the best sequence as a
plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities – pure, top‑level
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax, maps logits to a probability vector (sum=1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def next_pow2(k: int) -> int:
    """Small helper – smallest power of two ≥ k."""
    return 1 << (k - 1).bit_length()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    n = a.shape[0]
    L = next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def correlation_full(a: np.ndarray, w: np.ndarray) -> np.ndarray:
    """
    Full correlation (a ⋆ w) for lag 0 … n‑1.
    a has length n, w has length 2·n‑1.
    Returns an array of length n:  (a ⋆ w)[ℓ] = Σ_i a_i w_{i+ℓ}.
    """
    n = a.shape[0]
    m = w.shape[0]           # = 2·n‑1
    L = next_pow2(n + m - 1)  # FFT length
    A = np.fft.fft(a, L)
    W = np.fft.fft(w, L)
    corr = np.fft.ifft(A * np.conj(W), L).real  # length L
    start = m - 1            # index where lag 0 sits
    return corr[start:start + n]


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``.
    Equivalent to 2·n·max(a∗a) / (sum(a)²).
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    max_c = np.max(full_convolution(a))
    return 2.0 * a.shape[0] * max_c / (s * s)


# ----------------------------------------------------------------------
# Parametric families (used for cheap warm‑starts)
# ----------------------------------------------------------------------
def power_law_seq(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^{-p}, normalised to sum‑1."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def exponential_seq(n: int, r: float) -> np.ndarray:
    """a_i ∝ r^{i}, normalised to sum‑1 (0<r<1)."""
    idx = np.arange(n, dtype=np.float64)
    a = r ** idx
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser
# ----------------------------------------------------------------------
def run_smooth_adam(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iters: int = 200_000,
    init_alpha: float = 2.0,
    alpha_mult: float = 2.0,
    alpha_step: int = 2_500,
):
    """
    Adam optimiser on the *smoothed* max of the self‑convolution.
    ``alpha`` controls the temperature of the soft‑max; it is doubled
    every ``alpha_step`` iterations, making the objective gradually
    approach the true max.
    Returns (best_a, best_val) discovered before *deadline*.
    """
    # logits – keep the variables unconstrained, the soft‑max does the
    # normalisation and guarantees a_i ≥ 0, Σa_i = 1.
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()
    alpha = init_alpha

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                       # a ≥ 0, Σa = 1
        conv = full_convolution(a)           # length 2·n‑1
        # ----- smoothed max (log‑sum‑exp) -----
        # we work with a shifted version to avoid overflow
        shift = conv - np.max(conv)
        exp_alpha = np.exp(alpha * shift)
        sum_exp = exp_alpha.sum()
        weight = exp_alpha / sum_exp         # ∑ weight = 1
        smmax = np.max(conv) + np.log(sum_exp) / alpha

        # exact objective (for logging / book‑keeping)
        val_exact = 2.0 * n * np.max(conv) / (1.0)   # sum(a)=1 ⇒ denominator =1
        if val_exact < best_val:
            best_val = val_exact
            best_a = a.copy()
            last_improve = time.time()
            print(
                f"[SmoothAdam n={n:04d} iter={t:05d} α={alpha:.1f}] "
                f"new best exact F={best_val:.6f}"
            )

        # ----- gradient of smoothed max w.r.t. a -----
        # d smmax / d conv = weight
        # d conv / d a = 2 * (a ⋆ weight)
        grad_a = 2.0 * correlation_full(a, weight)

        # back‑prop through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)
        x = np.clip(x, -50.0, 50.0)          # keep logits numerically stable

        # temperature schedule
        if t % alpha_step == 0 and alpha < 1_024:
            alpha *= alpha_mult

        # shake (helps escape plateaus)
        if time.time() - last_improve > 8.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t}] shake (stagnation)")

    if best_a is None:                      # fallback – uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val

# (1+1) Evolution‑Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
    init_sigma: float = 0.02,
):
    """Simple Gaussian (1+1) ES on the simplex (exact objective)."""
    a = init_a.copy()
    best_a, best_val = a.copy(), compute_objective(a)
    sigma = init_sigma
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val = compute_objective(trial)
        if val < best_val:
            best_val, best_a = val, trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.25)
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best F={best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 6.0:
            a, sigma = best_a.copy(), init_sigma
            last_improve = time.time()

    return best_a, best_val

                print(f"[scan Exp r={r:.2f}] new global best F={val:.6f}")

        # --------------------------------------------------------------
        # 3️⃣  smooth‑max Adam optimisation (multiple restarts)
        # --------------------------------------------------------------
        restarts = 3
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r < len(init_cands):
                a0 = init_cands[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a modest slice of the remaining time
            slice_time = min((deadline - time.time()) * 0.25, 120.0)
            sub_deadline = time.time() + slice_time

            a_opt, val_opt = run_smooth_adam(
                n,
                init_a=a0,
                deadline=sub_deadline,
                max_iters=150_000,
                init_alpha=2.0,
                alpha_mult=2.0,
                alpha_step=3_000,
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[Adam n={n} restart={r}] GLOBAL best F={val_opt:.6f}")

        # --------------------------------------------------------------
        # 4️⃣  final fine‑tuning with (1+1) ES
        # --------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.20, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(
                n,
                init_a=best_seq,
                deadline=sub_deadline,
                max_iter=20_000,
                init_sigma=0.018,
            )
            if val_es < best_val:
                best_val, best_seq = val_es, a_es.copy()
                print(f"[ES n={n}] GLOBAL best improved to F={val_es:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list for the external evaluator
    return best_seq.tolist()
