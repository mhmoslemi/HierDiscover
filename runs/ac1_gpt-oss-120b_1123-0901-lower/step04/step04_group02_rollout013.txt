def solve_max_sum_lp(f, M):
    n = f.shape[0]
    rev = f[::-1]
    A = build_toeplitz_rev_f(rev)   # shape (2n-1, n)

def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    best_seq = None
    best_val = float('inf')
    lengths = [256, 512, 1024, 2048, 4096]  # maybe drop large if time.
    for n in lengths:
        if time.time() > deadline: break
        print(f"Optimizing length {n}")
        fft_len = _next_pow2(2*n-1)
        corr_fft_len = _next_pow2(3*n)
        restarts = 3
        for r in range(restarts):
            if time.time() > deadline: break
            if best_seq is not None and best_seq.shape[0]==n:
                a = best_seq.copy()
                print("[restart] warm start")
            else:
                a = np.random.dirichlet(np.ones(n))
            # Adam state
            m = np.zeros(n); v = np.zeros(n)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2
            lam = 0.5
            lam_factor = 1.2
            max_iters = 20000
            best_local_val = float('inf')
            best_local_seq = a.copy()
            last_improve = time.time()
            for t in range(1, max_iters+1):
                if time.time() > deadline: break
                conv = _full_convolution(a, fft_len)
                max_c = conv.max()
                # compute softmax weights
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0:
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum
                grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
                # Adam update
                m = beta1 * m + (1-beta1) * grad
                v = beta2 * v + (1-beta2) * grad * grad
                m_hat = m / (1 - beta1**t)
                v_hat = v / (1 - beta2**t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a = _project_simplex(a - step)
                lam = min(lam * lam_factor, 1e6)
                # evaluate objective
                val = 2.0 * n * conv.max()
                # Note: sum=1 ensures denominator=1
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[n={n} r={r} iter={t}] new global best {val:.6f}")
                if val < best_local_val:
                    best_local_val = val
                    best_local_seq = a.copy()
                    last_improve = time.time()
                # stagnation shake
                if time.time() - last_improve > 5.0:
                    # add small random noise
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    # print maybe
            # after gradient loop, optionally keep best_local_seq
            a = best_local_seq.copy()
            # LP refinement loop
            for lp_iter in range(10):
                if time.time() > deadline: break
                conv = _full_convolution(a, fft_len)
                M = conv.max()
                # Solve LP
                b = _lp_solve_max_sum(a, M)
                sum_b = b.sum()
                if sum_b <= 1.0 + 1e-8:
                    break
                b_norm = b / sum_b
                # line search
                best_eta = 0.0
                best_val_local = 2.0 * n * M
                best_f = a
                for eta in np.linspace(0.1, 0.9, 9):
                    cand = (1-eta)*a + eta*b_norm
                    conv_cand = _full_convolution(cand, fft_len)
                    val_cand = 2.0*n*conv_cand.max()
                    if val_cand < best_val_local:
                        best_val_local = val_cand
                        best_f = cand
                        best_eta = eta
                if best_eta == 0.0:
                    break
                a = best_f
                # update global best if improved
                if best_val_local < best_val:
                    best_val = best_val_local
                    best_seq = a.copy()
                    print(f"[n={n} LP iter={lp_iter}] new global best {best_val_local:.6f}")
            # local random refinement (fine)
            if time.time() < deadline:
                a = best_seq.copy()
                ncur = a.shape[0]
                # hill climbing with small delta
                while time.time() < deadline:
                    i, j = np.random.choice(ncur, 2, replace=False)
                    # transfer fraction up to 1% of total mass from i to j
                    delta = 0.01 * a[i] * np.random.rand()
                    if delta <= 0: continue
                    a_prop = a.copy()
                    a_prop[i] -= delta
                    a_prop[j] += delta
                    a_prop = _project_simplex(a_prop)  # safety
                    conv_prop = _full_convolution(a_prop, fft_len)
                    val_prop = 2.0*ncur*conv_prop.max()
                    if val_prop < best_val:
                        best_val = val_prop
                        best_seq = a_prop.copy()
                        a = a_prop
                        print(f"[refine] global best improved to {best_val:.6f}")
                        # maybe break from loop if improvement small?
        # after lengths loop, maybe break if reached target
        if best_val < 1.503:
            break
    if best_seq is None:
        # fallback to uniform 256
        n_fallback = 256
        best_seq = np.full(n_fallback, 1.0/n_fallback)
        best_val = 2.0*n_fallback*(1.0/n_fallback)
    print(f"Best value found: {best_val:.6f} length {len(best_seq)}")
    return best_seq.tolist()

"""
Propose a low‑value step‑function.

The algorithm works in three stages:

1. **Adam on the simplex** – a smooth‑max surrogate of
   max(a∗a) is minimised with Adam (as in the previous baseline).
   This quickly finds a decent shape.

2. **Linear‑programming refinement** – for the current vector `a`
   we solve the LP

        maximise   1ᵀ b
        subject to  (a ★ b)[k] ≤ M  for all k
                    b ≥ 0

   where `M = max(a∗a)`.  The optimal `b` carries the largest possible
   total mass while respecting the same convolution bound.
   After normalising `b` we perform a short line‑search on the mixing
   coefficient `t ∈ [0,1]` for the convex combination
   `c = (1‑t)a + t·b̂`.  Any reduction in `max(c∗c)` is kept.
   Repeating this LP‑step a few times drives the objective down
   aggressively.

3. **Local random hill‑climbing** – tiny transfers of mass between
   two coordinates are tried and kept if they improve the exact
   objective.  This finishes the optimisation.

The whole search respects the global time budget (default 1000 s) and
stores the best sequence seen over all restarts and lengths.
"""

import time
import numpy as np
import scipy.optimize
import scipy.linalg

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)], i = 0 … n‑1.
    This is the gradient of Σ_k w_k·(a*a)_k / 2.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _build_toeplitz_rev_f(f: np.ndarray) -> np.ndarray:
    """Build the dense Toeplitz matrix of the reversed f.
    For any b, (f ★ b) = A @ b where A has shape (2n‑1, n)."""
    n = f.size
    rev = f[::-1]
    c = np.concatenate([rev, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate([rev[:1], np.zeros(n - 1, dtype=np.float64)])
    return scipy.linalg.toeplitz(c, r)


def _lp_max_sum_under_M(f: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   max 1ᵀ b   s.t.   (f ★ b) ≤ M ,  b ≥ 0.
    Returns the optimal b (may be all‑zero if LP fails).
    """
    n = f.shape[0]
    A = _build_toeplitz_rev_f(f)               # (2n‑1) × n
    b_vec = np.full(2 * n - 1, M * (1.0 + 1e-12))
    # linprog minimises, so we flip the sign of the objective.
    res = scipy.optimize.linprog(
        c=-np.ones(n),
        A_ub=A,
        b_ub=b_vec,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": 5},
    )
    if not res.success:
        # Fall back to a zero vector – it will not improve anything.
        return np.zeros_like(f)
    return res.x


def _objective_from_conv(n: int, conv_full: np.ndarray) -> float:
    """Exact objective for a probability vector (sum = 1)."""
    return 2.0 * n * np.max(conv_full)


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Convenient wrapper – a is assumed to sum to 1."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    return _objective_from_conv(n, conv)


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step sequence that minimises
    F(a)=2·n·max(a*a)/[sum(a)]² .
    The function respects the time budget and returns the best list found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # numpy array, sum == 1
    best_val = float("inf")
    target = 1.503          # early‑stop threshold

    # ------------------------------------------------------------------
    # Helper to evaluate a sequence (sum is forced to 1).
    def _eval(a: np.ndarray) -> float:
        n = a.shape[0]
        fft_len = _next_pow2(2 * n - 1)
        return _objective(a, fft_len)

    # ------------------------------------------------------------------
    # Potential starting point supplied by the host (if any).
    historic = None
    if "height_sequence_1" in globals():
        try:
            cand = np.asarray(height_sequence_1, dtype=np.float64)
            if cand.ndim == 1 and np.all(cand >= 0):
                historic = cand / max(cand.sum(), 1e-12)
        except Exception:
            historic = None

    # ------------------------------------------------------------------
    # Lengths we will explore – keep the LP feasible (n ≤ 1024 for LP).
    lengths = [256, 512, 1024, 2048]   # larger lengths give more freedom

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)  # safe length for cross‑correlation

        # Number of independent restarts per length.
        restarts = 3 if n <= 1024 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ------------------- initialise a -------------------
            if r == 0 and best_seq is not None and best_seq.shape[0] == n:
                a = best_seq.copy()
                print("[restart 0] warm start from global best")
            elif historic is not None and historic.shape[0] == n:
                a = historic.copy()
                print("[restart] warm start from historic sequence")
            elif historic is not None and historic.shape[0] > n:
                a = historic[:n].copy()
                a = a / max(a.sum(), 1e-12)
                print("[restart] trimmed historic sequence")
            else:
                a = np.random.dirichlet(np.ones(n))
                print("[restart] random Dirichlet start")

            # ------------------- Adam optimisation -------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.18                     # a bit more aggressive
            lam = 0.5
            lam_factor = 1.15
            max_iters = 20000
            last_improve = time.time()
            best_local_val = float("inf")
            best_local_seq = a.copy()

            for t in range(1, max_iters + 1):
                if time.time() > deadline:
                    break

                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()
                # smooth‑max weights (softmax with temperature 1/lam)
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.shape[0])
                else:
                    w /= w_sum

                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # Adam update (before projection)
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                step = lr * m_hat / (np.sqrt(v_hat) + eps)
                a = _project_simplex(a - step)

                # temperature schedule
                lam = min(lam * lam_factor, 1e6)

                # exact objective for monitoring
                val = _objective_from_conv(n, conv)

                if val < best_local_val:
                    best_local_val = val
                    best_local_seq = a.copy()
                    last_improve = time.time()
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        print(f"[n={n} r={r} it={t:05d}] new global best {val:.6f}")

                # stagnation handling – small random “shake”
                if time.time() - last_improve > 5.0:
                    noise = (np.random.rand(n) - 0.5) * 1e-2
                    a = _project_simplex(a + noise)
                    last_improve = time.time()
                    # optional print
                    # print(f"[n={n} r={r} it={t}] shake")

                # early exit if we already beat the target
                if best_val < target:
                    deadline = time.time()
                    break

            # keep the best local sequence found by Adam
            a = best_local_seq.copy()

            # ------------------- LP‑based refinement (n ≤ 1024) -------------------
            if n <= 1024:
                for lp_iter in range(8):
                    if time.time() > deadline:
                        break
                    conv = _full_convolution(a, fft_len_conv)
                    M = conv.max()
                    b = _lp_max_sum_under_M(a, M)
                    sum_b = b.sum()
                    if sum_b <= 1.0 + 1e-8:
                        # cannot increase total mass – stop LP phase
                        break
                    b_norm = b / sum_b

                    # line‑search on mixing coefficient t
                    best_t = 0.0
                    best_val_lp = _objective_from_conv(n, conv)  # current value
                    best_candidate = a
                    for t_mix in np.linspace(0.1, 0.9, 9):
                        cand = (1.0 - t_mix) * a + t_mix * b_norm
                        cand = _project_simplex(cand)   # numerical safety
                        conv_cand = _full_convolution(cand, fft_len_conv)
                        val_cand = _objective_from_conv(n, conv_cand)
                        if val_cand < best_val_lp - 1e-12:
                            best_val_lp = val_cand
                            best_candidate = cand
                            best_t = t_mix

                    if best_t == 0.0:
                        # no improvement found – stop LP refinement
                        break

                    a = best_candidate
                    if best_val_lp < best_val:
                        best_val = best_val_lp
                        best_seq = a.copy()
                        print(f"[n={n} LP={lp_iter}] improved to {best_val_lp:.6f}")

                    if best_val < target:
                        deadline = time.time()
                        break

            # ------------------- final random hill‑climbing -------------------
            if time.time() < deadline:
                a = best_seq.copy() if best_seq is not None else a
                ncur = a.shape[0]
                fft_len_cur = _next_pow2(2 * ncur - 1)
                while time.time() < deadline:
                    i, j = np.random.choice(ncur, 2, replace=False)
                    # move a modest fraction of the mass from i to j
                    delta = 0.005 * a[i] * np.random.rand()
                    if delta <= 0.0:
                        continue
                    a_prop = a.copy()
                    a_prop[i] -= delta
                    a_prop[j] += delta
                    a_prop = _project_simplex(a_prop)
                    conv_prop = _full_convolution(a_prop, fft_len_cur)
                    val_prop = _objective_from_conv(ncur, conv_prop)
                    if val_prop < best_val - 1e-12:
                        best_val = val_prop
                        best_seq = a_prop.copy()
                        a = a_prop
                        print(f"[refine] new best {best_val:.6f}")
                    if best_val < target:
                        deadline = time.time()
                        break

        # ------------------------------------------------------------
        if best_val < target:
            break

    # ------------------------------------------------------------------
    # Fallback if nothing succeeded (should never happen).
    if best_seq is None:
        n_fallback = 256
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _eval(best_seq)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\n*** Finished: best objective = {best_val:.6f} (length {len(best_seq)}) ***")
    return best_seq.tolist()
