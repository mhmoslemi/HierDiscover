"""
Propose a step‑function (non‑negative vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
* Use three complementary search phases inside the allowed wall‑clock time:
    1. **Projected Adam optimisation** – a smooth‐max (softmax) approximation
       of the peak of the auto‑convolution is minimised.  The temperature is
       exponentially decayed, and after each update the vector is projected
       onto the feasible set (a_i ≥ 0, Σ a_i = √(2 n)).  Several diverse initial
       vectors are tried for each length.
    2. **Massive random sampling** – generate many candidate vectors from a
       mixture of Dirichlet, exponential, reversed‑exponential and power‑law
       distributions (the concentration / rate parameters are sampled
       randomly).  Each candidate is evaluated with a fast FFT based
       convolution.  The best sample is kept.
    3. **Local hill‑climbing** – starting from the current best vector,
       repeatedly add small Gaussian jitter, re‑project onto the simplex and
       keep the move if it improves the exact objective.  The jitter amplitude
       is slowly reduced.

* All three phases share a common helper that computes the exact objective
  `2 n·max(a∗a)/Σa²` using FFTs (much faster than the reference
  ``evaluate_sequence`` while mathematically equivalent because the
  candidates are already non‑negative and bounded by 1000).

* The code respects the 1000 s budget, breaks early on a deadline, prints
  progress and finally returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------


def _softmax_weights(conv: np.ndarray, tau: float):
    """
    Stable soft‑max of the convolution values.
    Returns (weights, smoothed_max) where weights sum to 1.
    """
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * np.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute the weighted reverse convolution
        rev_i = Σ_k w_k * a_{k-i}
    (indices outside the range are ignored).  This is the sub‑gradient of the
    (soft‑max) max‑convolution.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]

    # The slice that corresponds to rev_i for i = 0 … n‑1
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _max_convolution_fft(a: np.ndarray) -> float:
    """Maximum value of the full auto‑convolution a∗a (FFT implementation)."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _project_onto_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Project a onto {a_i ≥ 0, Σ a_i = target_sum}."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_iters: int,
    best_seq: np.ndarray,
    best_score: float,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser with a soft‑max phase that gradually turns into the exact
    sub‑gradient of the max‑convolution.
    """
    target_sum = math.sqrt(2.0 * n)

    # Normalise the initial vector to the required ℓ₁‑norm
    a = _project_onto_simplex(a0, target_sum)

    # Adam internal state
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr0 = 0.05
    jitter_interval = 4000

    # Temperature schedule parameters
    tau_factor = 0.30               # initial τ = τ_factor * max(conv)
    decay_rate = max_iters / 9.0    # slower decay → smoother early phase

    Nfft = 1 << ((2 * n - 1).bit_length())   # power‑of‑two FFT length

    for it in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        # ----- exact convolution & its current maximum -----
        A_fft = np.fft.rfft(a, n=Nfft)
        conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
        M = float(conv.max())

        # ----- temperature & (soft‑)max -----
        tau = tau_factor * M * math.exp(-it / decay_rate)
        if tau < 1e-9:
            # Exact sub‑gradient: uniform weight over the active set
            mask = conv >= M - 1e-12
            cnt = mask.sum()
            w = mask.astype(float) / max(cnt, 1)
            Ltau = M
        else:
            w, Ltau = _softmax_weights(conv, tau)

        # ----- weighted reverse convolution → (sub‑)gradient of the max -----
        rev = _weighted_rev(a, w, Nfft)

        # ----- Adam gradient step (derived from the exact objective) -----
        # After normalisation the objective equals the convolution peak M.
        coeff = 4.0 * n / (target_sum * target_sum)          # = 2 after normalisation
        const = 4.0 * n * Ltau / (target_sum ** 3)           # projection term
        grad = coeff * rev - const

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        lr = lr0 / (1.0 + 1e-4 * it)
        a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

        # ----- projection onto feasible set -----
        a = _project_onto_simplex(a, target_sum)

        # occasional jitter to escape plateaus
        if it % jitter_interval == 0:
            jitter = np.random.randn(n) * (1e-3 * target_sum / n)
            a = _project_onto_simplex(a + jitter, target_sum)

        # ----- record improvement -----
        cur_score = M                     # equal to the exact objective after projection
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[Adam len {n:5d} iter {it:8d}] new best = {best_score:.6f}")

    return best_seq, best_score


def _make_initial_vectors(n: int, target_sum: float) -> list[np.ndarray]:
    """
    Produce a small, diverse set of starting vectors for a given length n.
    All vectors are already normalised to have ℓ₁‑norm = target_sum.
    """
    rng = np.random.default_rng()
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1) i.i.d.)
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay (random rate)
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    rev = x[::-1]
    init.append(rev / rev.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = a0 - a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) + a2 * np.cos(4.0 * np.pi * n_arr / (n - 1))
    init.append(black / black.sum() * target_sum)

    # 7) small random perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.01 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


def _objective_fft(a: np.ndarray) -> float:
    """
    Exact objective : 2·n·max(a∗a) / (sum a)².
    The implementation uses FFTs and assumes a_i ≥ 0 (no clipping needed).
    """
    n = a.shape[0]
    s = a.sum()
    if s < 0.01:
        return float("inf")
    Nfft = 1 << ((2 * n - 1).bit_length())
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    max_b = float(conv.max())
    return float(2.0 * n * max_b / (s * s))

def propose_candidate(seed: int = 12345, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    Returns the best sequence found as a plain Python list.
    """
    # ------------------------------------------------------------------
    # Basic housekeeping
    # ------------------------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 999)  # separate RNG for sampling
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # safety margin

    # ------------------------------------------------------------------
    # Load any previously discovered construction (if present)
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = None

    # ------------------------------------------------------------------
    # Choose a set of lengths to explore (bounded by 300 … 8000)
    # ------------------------------------------------------------------
    if base is not None:
        n0 = int(base.shape[0])
    else:
        n0 = 2100                         # sensible historic fallback
    factors = [0.5, 0.75, 1.0, 1.25, 1.5, 2.0]
    lengths = sorted({max(300, min(int(n0 * f), 8000)) for f in factors})

    # ------------------------------------------------------------------
    # Global best tracker
    # ------------------------------------------------------------------
    best_seq = None          # numpy array
    best_score = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – projected Adam optimisation from diverse starts
    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() >= deadline:
            break
        target_sum = math.sqrt(2.0 * n)

        init_vectors = _make_initial_vectors(n, target_sum)

        # If we have a base vector of a different size, linearly interpolate it
        if base is not None and base.shape[0] != n:
            old_x = np.linspace(0.0, 1.0, base.shape[0])
            new_x = np.linspace(0.0, 1.0, n)
            interp = np.interp(new_x, old_x, base)
            interp = np.maximum(interp, 0.0)
            interp = interp / interp.sum() * target_sum
            init_vectors.append(interp)

        # Rough iteration budget – more steps for smaller problems
        max_iters = int(3_000_000 * 2100 / n)
        max_iters = max(400_000, max_iters)   # at least 400 k steps

        print(f"[run] length={n}, max_iters≈{max_iters}, target_sum={target_sum:.3f}")

        for idx, vec in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            print(f"   start {idx+1}/{len(init_vectors)} for length {n}")
            best_seq, best_score = _adam_optimize(
                a0=vec.copy(),
                n=n,
                deadline=deadline,
                max_iters=max_iters,
                best_seq=best_seq if best_seq is not None else vec,
                best_score=best_score,
            )

    # ------------------------------------------------------------------
    # Phase 2 – massive random sampling (Dirichlet / exponential / power‑law)
    # ------------------------------------------------------------------
    rand_iters = 0
    max_rand_iters = 200_000
    while time.time() < deadline and rand_iters < max_rand_iters:
        # pick a length at random (the same distribution as above)
        n = rng.choice(lengths)
        target_sum = math.sqrt(2.0 * n)

        mode = rng.integers(0, 5)   # 0 … 4
        if mode == 0:               # uniform
            a = np.full(n, target_sum / n)
        elif mode == 1:             # Dirichlet with random concentration
            alpha = rng.uniform(0.1, 5.0)
            g = rng.gamma(alpha, 1.0, size=n)
            a = g / g.sum() * target_sum
        elif mode == 2:             # exponential decay
            rate = rng.uniform(0.0005, 0.01)
            x = np.exp(-rate * np.arange(n))
            a = x / x.sum() * target_sum
        elif mode == 3:             # reversed exponential
            rate = rng.uniform(0.0005, 0.01)
            x = np.exp(-rate * np.arange(n))[::-1]
            a = x / x.sum() * target_sum
        else:                       # power‑law tail
            p = rng.uniform(0.3, 2.0)
            x = 1.0 / (np.arange(1, n + 1) ** p)
            a = x / x.sum() * target_sum

        score = _objective_fft(a)

        if score < best_score:
            best_score = score
            best_seq = a.copy()
            print(f"[rand] new best = {best_score:.6f} (len {n})")
        rand_iters += 1

    # ------------------------------------------------------------------
    # Phase 3 – local hill‑climbing (Gaussian jitter + projection)
    # ------------------------------------------------------------------
    if best_seq is not None:
        a_cur = best_seq.copy()
        n = a_cur.shape[0]
        target_sum = math.sqrt(2.0 * n)
        sigma = target_sum * 0.02 / n            # initial jitter scale

        while time.time() < deadline:
            # propose a perturbed candidate
            a_new = a_cur + rng.normal(scale=sigma, size=n)
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_onto_simplex(a_new, target_sum)

            score = _objective_fft(a_new)

            if score < best_score:
                best_score = score
                best_seq = a_new.copy()
                a_cur = a_new
                sigma *= 0.98                # slowly shrink the step size
                print(f"[local] new best = {best_score:.6f} (len {n})")
            else:
                # occasional acceptance of neutral moves to keep exploring
                if rng.random() < 0.001:
                    a_cur = a_new
                    sigma *= 0.99

    # ------------------------------------------------------------------
    # Fallback in the extremely unlikely case that nothing was found
    # ------------------------------------------------------------------
    if best_seq is None:
        n = n0
        target_sum = math.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n)
        best_score = _objective_fft(best_seq)
        print("[fallback] using uniform sequence")

    # ------------------------------------------------------------------
    # Final report
    # ------------------------------------------------------------------
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # The harness expects a plain Python list
    return best_seq.tolist()
