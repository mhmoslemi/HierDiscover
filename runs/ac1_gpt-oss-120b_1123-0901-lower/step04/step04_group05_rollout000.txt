"""
Hybrid edge‑biased Adam optimiser for the self‑convolution peak.

The routine searches for a non‑negative step function (a list of heights)
that minimises

    2·n·max_k (a * a)[k] / (∑ a)^2

which is invariant under scaling, so we fix ∑ a = 1.
The objective is therefore `2·n·max(conv)`.  Reducing the peak of the
auto‑convolution is the same as flattening the correlation of the
distribution.

Key ideas
---------

* **Multiple initial families** – besides plain Dirichlet draws we also
  use an “edge‑decay” shape (mass concentrated near the ends) and
  half‑sine bumps.  This diversifies the search and often finds a
  better basin than the pure uniform‑like starts used before.

* **Smoothed max surrogate** – we replace `max(conv)` by a
  log‑sum‑exp (soft‑max) with a temperature λ.  λ is increased
  gradually (continuation) so optimisation first sees a smooth landscape
  and later focuses on the true maximum.

* **Adam optimiser in log‑space** – the parameters are logits `u`,
  `a = softmax(u)`, guaranteeing non‑negativity and unit sum.
  Adam with bias‑correction gives a robust, cheap first‑order method.

* **Adaptive perturbations** – when progress stalls for a few seconds
  we (i) apply a small random shake to the logits and (ii) move a tiny
  amount of mass from the centre of the distribution to the edges.
  This helps escape shallow local minima.

* **Two‑phase schedule** – the outer loop performs many short Adam
  runs (different lengths and initialisations).  A final long fine‑tune
  run starts from the best sequence found so far with a larger λ and a
  slower learning‑rate.

The whole algorithm respects the supplied `budget_s` (default 1000 s)
and returns the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    # Numerical stabilisation
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """Gradient of the smooth‑max surrogate w.r.t. a.
    The analytical form is  ∇_a = 2·(w ★ a),  where ★ denotes correlation."""
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # The relevant slice corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _init_edge_decay(n: int, decay: float) -> np.ndarray:
    """Mass decays exponentially from the two ends towards the centre."""
    i = np.arange(n)
    dist = np.minimum(i, n - 1 - i)
    a = np.exp(-decay * dist)
    a /= a.sum()
    return a


def _init_half_sine(n: int) -> np.ndarray:
    """A single half‑sine bump centred in the interval."""
    i = np.arange(1, n + 1)
    a = np.sin(np.pi * i / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _init_bimodal_sine(n: int) -> np.ndarray:
    """Two half‑sine bumps glued back‑to‑back (symmetrical)."""
    half = n // 2
    left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
    right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
    a = np.concatenate([left, right])
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

                    best_seq = a.copy()
                    print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform (should never be needed)
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    # Final report
    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")

    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
