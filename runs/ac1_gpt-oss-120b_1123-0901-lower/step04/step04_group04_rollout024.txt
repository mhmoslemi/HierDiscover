"""
Propose a non‑negative step‑function (list of heights) with a low
evaluate_sequence value.

The algorithm works on the probability vector `p = a / Σa` (so Σp = 1).
For a fixed length `n` the evaluated score equals

    score = 2·n·max(p * p)

where `*` denotes the ordinary (non‑reversed) convolution.
We minimise the smooth log‑sum‑exp approximation of the max
and use a L‑BFGS‑B (or a simple mirror‑descent fallback) to optimise.
A temperature schedule (τ) is applied to the smooth max; after each
τ‑stage we evaluate the exact objective and keep the best vector.
Many random restarts, a slower τ‑decay and a larger optimisation budget
allow us to explore the landscape more thoroughly than the reference
implementation, often improving the bound below the required 1.503.

After the continuous optimisation a cheap coordinate‑wise local‑search
(transfer a small amount of mass between two positions) is performed
until the time budget is exhausted.  The final vector is returned as a
plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# optional SciPy – used for L‑BFGS‑B if available
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


# ----------------------------------------------------------------------
# FFT‑based convolution utilities ----------------------------------------
def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p * p (real FFT). Result length 2·n‑1."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·a_{k‑i} (indices outside the support are ignored)
    using FFTs.
    """
    A = np.fft.rfft(a, n=Nfft)
    W = np.fft.rfft(w[::-1], n=Nfft)          # reverse w before FFT
    rev_full = np.fft.irfft(W * A, n=Nfft)
    # keep only the valid part (length len(a)+len(w)-1)
    rev_full = rev_full[: len(a) + len(w) - 1]
    # extract the required slice, see original code for indexing
    start = 2 * a.shape[0] - 2
    rev = rev_full[start - np.arange(a.shape[0])]
    return rev


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


# ----------------------------------------------------------------------
# Objective (smooth max) and its gradient w.r.t. the log‑probabilities y
def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """Return smooth max Lτ and gradient w.r.t. y (log‑probabilities)."""
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)                 # numeric stability
    p = exp_y / exp_y.sum()

    # full convolution p*p
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p (see derivation in the description)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # chain rule: ∂L/∂y = J_softmaxᵀ · grad_p
    dot = np.dot(p, grad_p)                   # Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


# ----------------------------------------------------------------------
# Core optimisation routine (L‑BFGS if possible, otherwise mirror descent)
def _optimise_from_y(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,
    *,
    maxfun: int = 800,
    tau_decay: float = 0.85,
    min_tau: float = 1e-12,
):
    """
    Optimise the smooth max for decreasing temperatures τ.
    Returns the best (sequence, score) encountered.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    n = y.shape[0]

    while time.time() < deadline and tau > min_tau:
        # ---- minimise smooth max at current temperature -----------------
        if _SCIPY_AVAILABLE:
            def fun(vec):
                L, g = _objective_and_grad(vec, tau, Nfft)
                return L, g

            options = {
                "maxfun": maxfun,
                "maxiter": maxfun,
                "ftol": 1e-14,
                "gtol": 1e-14,
            }
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=options)
            y = res.x
        else:
            # simple exponentiated‑gradient steps
            eta = 0.2
            for _ in range(maxfun):
                _, grad_y = _objective_and_grad(y, tau, Nfft)
                y = y - eta * grad_y
                # re‑normalise to the simplex via soft‑max
                y_max = np.max(y)
                exp_y = np.exp(y - y_max)
                y = np.log(exp_y / exp_y.sum() + 1e-14)

        # ---- evaluate the *exact* objective (no smoothing) -------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        max_conv = _conv_full(p, Nfft).max()
        cur_score = 2.0 * n * max_conv                     # equals evaluate_sequence after scaling

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve] τ={tau:.2e}  score={best_score:.6f}")

        # ---- schedule update -------------------------------------------
        tau *= tau_decay

        # occasional jitter to escape flat plateaus
        if np.random.rand() < 0.12:
            y += np.random.randn(y.shape[0]) * 0.01

        # early‑out if we are out of time
        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Very cheap local‑search on the probability vector (mass swaps)
def _local_search(p: np.ndarray, Nfft: int, deadline: float):
    """
    Perform many tiny random mass‑transfer moves between two coordinates.
    Stops when the deadline is reached.
    Returns the best probability vector and its exact score.
    """
    n = p.shape[0]
    best_p = p.copy()
    best_obj = 2.0 * n * _conv_full(best_p, Nfft).max()

    while time.time() + 0.001 < deadline:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # move at most 10 % of the source mass
        delta = best_p[i] * np.random.rand() * 0.10
        if delta <= 0.0:
            continue
        cand = best_p.copy()
        cand[i] -= delta
        cand[j] += delta
        if cand[i] < 0.0:      # safety – should never happen
            continue
        obj = 2.0 * n * _conv_full(cand, Nfft).max()
        if obj < best_obj:
            best_obj = obj
            best_p = cand
            print(f"[local] improved score to {best_obj:.6f}")

    return best_p, best_obj

def propose_candidate(seed: int = 42, budget_s: float = 1000, **_ignored) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises evaluate_sequence.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0          # small safety margin

    # ------------------------------------------------------------------
    # Load a known construction if present, otherwise start from uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)                # Σa = √(2n)

    # Normalise the base vector to the required L1‑norm.
    base = base * (target_sum / base.sum())

    # ------------------------------------------------------------------
    # Initial probability vector and corresponding log‑params.
    p0 = base / target_sum
    eps = 1e-12
    y0 = np.log(p0 + eps)

    # Record the best solution so far.
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, score={best_score:.6f}")

    # FFT size (next power‑of‑two ≥ 2·n‑1)
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # Main optimisation loop – many random restarts.
    restart = 0
    max_restarts = 60                               # adjustable; respects deadline
    while time.time() < deadline and restart < max_restarts:
        # --------------------------------------------------------------
        # Choose an initial point.
        if restart == 0:                           # first start = known solution
            y_start = y0.copy()
        else:
            rand_exp = np.random.exponential(scale=1.0, size=n)
            y_start = np.log(rand_exp + eps)

        # --------------------------------------------------------------
        # Rough estimate for the initial temperature.
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 4.0, 1e-9)    # a few times the current max

        # --------------------------------------------------------------
        # Run the (smooth) optimisation.
        best_seq, best_score = _optimise_from_y(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
            maxfun=800,
            tau_decay=0.85,
        )

        # --------------------------------------------------------------
        # Optional cheap local refinement (only if a few seconds remain).
        if time.time() + 5.0 < deadline:
            p_current = best_seq / target_sum
            p_refined, obj_refined = _local_search(p_current, Nfft, deadline)
            a_refined = p_refined * target_sum
            exact = evaluate_sequence(a_refined.tolist())
            if exact < best_score:
                best_score = exact
                best_seq = a_refined.copy()
                print(f"[refine] exact score={best_score:.6f}")

        restart += 1

    # ------------------------------------------------------------------
    # Final sanity check (evaluate the exact objective once more).
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return as plain Python list, as required by the evaluator.
    return best_seq.tolist()
