"""
Propose a non‑negative step‑function (height sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

using a *softmax* parametrisation and an Adam optimiser that targets a
*smoothed max* together with a *variance* penalty of the self‑convolution.
The variance term forces the convolution to become flatter, which usually
lowers the true max beyond what the pure smoothed‑max objective can reach.
The search runs many independent restarts with different lengths (up to a
few × 10⁴) and different initial shapes (U‑shaped power‑law, exponential
tails, random jitter etc.).  The current global best sequence is kept and
occasionally injected as a warm‑start for new runs.  A tiny hill‑climbing
post‑processing (mass transfer between two positions) is performed while
time permits.

The code respects the 1000 s budget, prints progress information and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – linear convolution via real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())      # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny numerical negatives
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (softmax) representation with a
# smooth‑max + variance surrogate.
# ----------------------------------------------------------------------
def _optimise_adam_conv(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    var_weight: float = 300.0,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it space.
    The surrogate loss is  smooth_max(conv)  +  var_weight * Var(conv).

    Returns the best sequence (as a NumPy array) encountered during the run
    and its true evaluation as returned by ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    it = 0

    while time.time() < run_deadline:
        # ---- 1. map logits to the simplex ---------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # numerical stability
        p = exp_shift / np.sum(exp_shift)      # probabilities, sum = 1
        a = target_sum * p                     # a lives on the simplex

        # ---- 2. convolution and basic statistics -------------------------
        conv = _full_convolution_fft(a, a)                # length = 2n-1
        max_val = conv.max()
        mean_val = conv.mean()
        var_val = ((conv - mean_val) ** 2).mean()

        # keep the best (true) score seen so far
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {best_score:.6f}")

        # ---- 3. smooth‑max weights ---------------------------------------
        tau = max(init_tau_factor * max_val * (0.95 ** it),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w_s = np.exp(scaled)
        w_s_sum = w_s.sum()
        if w_s_sum == 0.0:
            w_s = np.full_like(w_s, 1.0 / w_s.size)
        else:
            w_s /= w_s_sum                                    # ∑w_s = 1

        # ---- 4. variance weights (gradient of var(conv)) -----------------
        m_len = conv.size
        w_v = (2.0 / m_len) * (conv - mean_val)               # ∑w_v = 0

        # ---- 5. combine the two weight vectors ---------------------------
        w = w_s + var_weight * w_v

        # ---- 6. gradient w.r.t. a ----------------------------------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                 # slice for grad_a
        grad_a = 2.0 * conv2[idx]

        # ---- 7. gradient w.r.t. logits (chain rule) ----------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 8. Adam update ----------------------------------------------
        it += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random perturbation to escape plateaus
        if it % 250 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # modest decay of the base learning‑rate
        alpha *= 0.9995

    # ------------------------------------------------------------------
    # Final evaluation using the official evaluator (identical rounding)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Different initial shape generators (all produce positive vectors)
# ----------------------------------------------------------------------
def _init_powerlaw(n, p):
    """a_i ∝ (i+1)^{-p} (decreasing towards the end)."""
    idx = np.arange(1, n + 1, dtype=float)
    return idx ** (-p)


def _init_ushape_powerlaw(n, p):
    """U‑shaped: larger at both ends, minimal in the middle."""
    idx = np.arange(1, n + 1, dtype=float)
    left  = idx ** (-p)
    right = (n - idx + 1) ** (-p)
    return left + right


def _init_exponential_tails(n, tau):
    """Exponential decay from both ends (symmetric)."""
    idx = np.arange(n, dtype=float)
    left  = np.exp(-idx / tau)
    right = np.exp(-(n - 1 - idx) / tau)
    return left + right


def _init_random_jitter(base, scale=1e-5):
    """Add a tiny uniform jitter to break perfect symmetry."""
    return base + np.random.rand(base.size) * scale


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative height sequence that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.

    Strategy
    --------
    * Randomly pick a length ``n`` (≈ 3000 – 8000) – larger lengths give a
      lower theoretical bound.
    * Initialise ``a`` with one of several U‑shaped / power‑law shapes,
      occasionally injecting the currently best known sequence (interpolated
      to the new length).
    * Optimise the shape with Adam on the log‑softmax parametrisation,
      targeting a surrogate loss = smooth_max(conv) + λ·Var(conv).
      The variance weight λ is drawn randomly for each restart to promote
      diversity.
    * Keep the overall best across many independent restarts.
    * When time permits, apply a cheap hill‑climbing phase that moves a tiny
      amount of mass from a larger‑conv contributing index to a smaller‑conv
      one.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # small safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a previously known construction (if available)
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start present

    # --------------------------------------------------------------
    # Main restart loop – each restart obtains a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()

        # ----- choose problem size ------------------------------------
        if remaining > 400:
            n = np.random.randint(4000, 8000 + 1)
        elif remaining > 200:
            n = np.random.randint(2500, 6000 + 1)
        else:
            n = np.random.randint(1500, 3500 + 1)

        # ----- initial shape -------------------------------------------
        shape_type = np.random.choice(["powerlaw", "ushape", "exp", "mix"])
        if shape_type == "powerlaw":
            p = np.random.uniform(0.2, 0.8)
            init = _init_powerlaw(n, p)
        elif shape_type == "ushape":
            p = np.random.uniform(0.2, 0.8)
            init = _init_ushape_powerlaw(n, p)
        elif shape_type == "exp":
            tau = np.random.uniform(0.05, 0.3) * n
            init = _init_exponential_tails(n, tau)
        else:  # mix – start from uniform plus a tiny random bump
            init = np.full(n, 1.0, dtype=float)
            init += np.random.rand(n) * 0.1

        # occasional smoothing by a short moving‑average
        if np.random.rand() < 0.18:
            klen = np.random.randint(3, 9)
            kernel = np.ones(klen) / klen
            init = np.convolve(init, kernel, mode='same')

        # occasionally inject the current global best (interpolated)
        if best_seq is not None and np.random.rand() < 0.45:
            # up‑sample / down‑sample via linear interpolation
            old = best_seq
            old_n = old.size
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            interpolated = np.interp(x_new, x_old, old)
            # mix a little bit of the random shape to keep diversity
            init = 0.6 * interpolated + 0.4 * init
            init = np.maximum(init, 1e-12)

        # ensure strictly positive values for log‑it conversion
        init = np.maximum(init, 1e-12)

        # ----- map to the simplex (sum = sqrt(2n)) --------------------
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)  # logits

        # ----- allocate a time slice for this run ----------------------
        slice_time = min(18.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        # ----- randomly pick a variance weight for this restart -------
        var_w = 10.0 ** np.random.uniform(1.5, 3.0)   # between ~30 and ~1000

        # ----- run the optimiser ---------------------------------------
        seq_local, score_local = _optimise_adam_conv(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.12,
            min_tau_factor=0.001,
            var_weight=var_w,
            alpha=0.05,
            beta1=0.9,
            beta2=0.999,
            eps=1e-8,
        )

        # ----- update global best ---------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n}, var_w={var_w:.1f})"
            )
            if best_score < 1.503:          # early exit if target already beaten
                break

    # ------------------------------------------------------------------
    # Optional cheap hill‑climbing refinement while time permits
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.size
        a = best_seq.copy()
        target_sum = a.sum()                 # should be ≈ sqrt(2n)
        conv = _full_convolution_fft(a, a)
        best_val = conv.max()
        while time.time() < deadline - 0.5:
            # pick two indices, move a tiny amount from the larger‑conv side
            i, j = np.random.choice(n, 2, replace=False)
            if a[i] < 1e-12:
                continue
            delta = min(a[i] * 0.02, 1e-3)   # move at most 2 % of a[i] or 1e‑3
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # re‑normalise to keep sum (tiny numerical drift possible)
            a_new *= target_sum / a_new.sum()
            conv_new = _full_convolution_fft(a_new, a_new)
            val_new = conv_new.max()
            if val_new < best_val:
                a = a_new
                best_val = val_new
                best_seq = a.copy()
                print(f"[{time.time():.2f}] hill‑climbing improvement = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was generated (should never happen)
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 2000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
