"""
Hybrid local‑search optimizer for the self‑convolution bound.

The algorithm works directly on the discrete “step‑function’’ a (non‑negative,
∑a=1).  At each iteration it:
  1. finds the convolution index k where (a*a)[k] is maximal,
  2. computes the contribution of each coordinate a_i to that peak,
  3. moves a small amount of mass from the coordinate that contributes most
     to the peak to a coordinate that contributes least,
  4. updates the convolution exactly by a rank‑1 update (O(n) work),
  5. accepts the move only if the peak value decreases; otherwise a random
     perturbation or a global “shake’’ is tried.

All updates preserve the simplex constraints, the convolution is kept in
memory, and the objective 2·n·max(a*a) (which equals the value returned by
evaluate_sequence for ∑a=1) is tracked.  The routine respects the supplied
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

def _convolution(a: np.ndarray) -> np.ndarray:
    """Full linear convolution a*a (len = 2·len(a)-1)."""
    return np.convolve(a, a)

def _make_padded(a: np.ndarray) -> np.ndarray:
    """Pad a with n‑1 zeros on each side; used for fast shift extraction."""
    n = a.shape[0]
    return np.concatenate((np.zeros(n - 1, dtype=a.dtype), a,
                           np.zeros(n - 1, dtype=a.dtype)))

def _contrib_vector(a: np.ndarray, k: int) -> np.ndarray:
    """
    Contribution of each a_i to convolution entry k:
        c_i = a_{k-i}  (0 ≤ k-i < n)  else 0.
    Returned as a length‑n vector.
    """
    n = a.shape[0]
    offset = n - 1 - k
    c = np.empty(n, dtype=a.dtype)
    if offset >= 0:
        # first part non‑zero
        c[: n - offset] = a[::-1][offset:]
        c[n - offset:] = 0.0
    else:
        # shift left
        off = -offset
        c[off:] = a[::-1][: n - off]
        c[:off] = 0.0
    return c

    """
    Search for a non‑negative step function (list of heights) that minimises

    and returns the best sequence found.

    # basic bookkeeping
    conv = _convolution(a)               # length 2*n-1
    conv_len = conv.shape[0]
    curr_max = conv.max()
    best_val = 2.0 * n * curr_max         # because ∑a = 1
    best_seq = a.copy()
    print(f"[init] objective = {best_val:.6f}")

    # padded version for O(1) shift extraction
    a_padded = _make_padded(a)

    # algorithmic parameters
    step_factor = 0.4      # fraction of the source coordinate moved each try
    eps_min = 1e-12
    tolerance = 1e-15      # improvement tolerance
    no_improve_cnt = 0
    max_no_improve = 4000  # after this we shake the whole vector

    it = 0
    # ----------------------------------------------------------------------
    while time.time() < deadline:
        it += 1

        # 1) index of the current worst (largest) convolution entry
        k_max = int(np.argmax(conv))

        # 2) contribution of each coordinate to that entry
        contrib = _contrib_vector(a, k_max)

        # 3) pick source (i) with large contribution and positive mass
        mask = a > eps_min
        if not np.any(mask):
            break
        contrib_src = np.where(mask, contrib, -1.0)
        i_src = int(np.argmax(contrib_src))

        # 4) pick target (j) with the smallest contribution
        j_tgt = int(np.argmin(contrib))

        # make sure i != j
        if i_src == j_tgt:
            # pick a random different target
            j_tgt = int(np.random.choice(np.delete(np.arange(n), i_src)))
        # ensure we do not try to move more than we have
        eps = a[i_src] * step_factor
        if eps < eps_min:
            eps = a[i_src] * 0.1

        # limit epsilon to the available mass at the source
        eps = min(eps, a[i_src])

        # pre‑fetch the two shift vectors (views on the padded array)
        shift_i = a_padded[i_src : i_src + conv_len]
        shift_j = a_padded[j_tgt : j_tgt + conv_len]

        # attempt a sequence of decreasing epsilons until an improvement is seen
        improved = False
        eps_try = eps
        while eps_try > eps_min:
            # rank‑1 update of the convolution
            new_conv = conv + 2.0 * eps_try * (shift_j - shift_i)

            # add the quadratic correction (epsilon² terms)
            eps2 = eps_try * eps_try
            idx2i = 2 * i_src
            if idx2i < conv_len:
                new_conv[idx2i] += eps2
            idx2j = 2 * j_tgt
            if idx2j < conv_len:
                new_conv[idx2j] += eps2
            idxij = i_src + j_tgt
            if idxij < conv_len:
                new_conv[idxij] -= 2.0 * eps2

            new_max = new_conv.max()
            if new_max + tolerance < curr_max:
                # ----- ACCEPT -----
                a[i_src] -= eps_try
                a[j_tgt] += eps_try
                conv = new_conv
                curr_max = new_max
                # update padded view
                a_padded[n - 1 : n - 1 + n] = a
                # record best if improved
                val = 2.0 * n * curr_max
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[iter {it:08d}] new best = {best_val:.6f}")
                no_improve_cnt = 0
                improved = True
                break
            else:
                eps_try *= 0.5   # try a smaller step

        if not improved:
            # ------------------------------------------------------------------
            # Random exploratory move (small mass shuffle)
            i_r = np.random.randint(0, n)
            j_r = np.random.randint(0, n)
            if i_r != j_r and a[i_r] > eps_min:
                eps_r = a[i_r] * np.random.uniform(0.02, 0.15)
                shift_i = a_padded[i_r : i_r + conv_len]
                shift_j = a_padded[j_r : j_r + conv_len]
                new_conv = conv + 2.0 * eps_r * (shift_j - shift_i)
                eps2 = eps_r * eps_r
                idx2i = 2 * i_r
                if idx2i < conv_len:
                    new_conv[idx2i] += eps2
                idx2j = 2 * j_r
                if idx2j < conv_len:
                    new_conv[idx2j] += eps2
                idxij = i_r + j_r
                if idxij < conv_len:
                    new_conv[idxij] -= 2.0 * eps2
                new_max = new_conv.max()
                if new_max + tolerance < curr_max:
                    a[i_r] -= eps_r
                    a[j_r] += eps_r
                    conv = new_conv
                    curr_max = new_max
                    a_padded[n - 1 : n - 1 + n] = a
                    val = 2.0 * n * curr_max
                    if val < best_val:
                        best_val = val
                        best_seq = a.copy()
                        print(f"[iter {it:08d}] new best (rand) = {best_val:.6f}")
                    no_improve_cnt = 0
                    continue
            # if the random move did not help count a failure
            no_improve_cnt += 1

        # ----------------------------------------------------------------------
        # Global shake if we have been stuck for a while
        if no_improve_cnt >= max_no_improve:
            # add a tiny zero‑mean noise, re‑project onto simplex
            noise = np.random.randn(n) * 1e-4
            a += noise
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
                s = 1.0
            a /= s
            conv = _convolution(a)
            curr_max = conv.max()
            a_padded = _make_padded(a)
            no_improve_cnt = 0
            print(f"[shake] iter {it:08d}  max after shake = {curr_max:.8f}")

    # ----------------------------------------------------------------------
    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
