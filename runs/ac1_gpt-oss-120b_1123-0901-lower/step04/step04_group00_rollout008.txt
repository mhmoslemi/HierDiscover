def generate_initial_candidates(n, best_seq):
    candidates = []
    # from best_seq of same length if available
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.array(best_seq, dtype=np.float64))
    # previous best upsampled
    if best_seq is not None and len(best_seq) < n:
        # linear interpolation upsample
        old = np.array(best_seq)
        old_n = len(old)
        x_old = np.linspace(0, 1, old_n)
        x_new = np.linspace(0, 1, n)
        up = np.interp(x_new, x_old, old)
        candidates.append(up)
    # random Dirichlet
    candidates.append(rng.dirichlet(np.ones(n)))
    # edge concentration law
    idx = np.arange(n) + 1
    prod = idx * (n - idx + 1)
    for alpha in np.linspace(-2, -0.2, 7):
        shape = np.where(prod>0, prod**alpha, 0.0)
        if shape.sum()>0:
            shape/=shape.sum()
            candidates.append(shape)
    # two Gaussian edge
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2,0.5,1.0,2.0,5.0]:
        shape = np.exp(-gamma*i*i) + np.exp(-gamma*(n-1-i)*(n-1-i))
        shape/=shape.sum()
        candidates.append(shape)
    # interior Gaussian (center peaked)
    for sigma in [0.5, 1.0, 2.0]:
        shape = np.exp(- (i - (n-1)/2)**2 / (2*sigma**2))
        shape/=shape.sum()
        candidates.append(shape)
    # sinusoidal
    for power in [1,2,3]:
        shape = np.sin(np.pi * (i+0.5) / n) ** power
        shape+=1e-12
        shape/=shape.sum()
        candidates.append(shape)
    # monotonic decreasing power law
    for gamma in [0.5, 1.0, 1.5, 2.0]:
        shape = 1.0/ ( (i+1) ** gamma )
        shape/=shape.sum()
        candidates.append(shape)
    # choose limited number
    if len(candidates) > 20:
        # random sample
        rng.shuffle(candidates)
        candidates = candidates[:20]
    return candidates

def adam_smooth_max(a0, n, fft_len_conv, fft_len_corr, deadline, max_iter=5000):
    a = a0.copy()
    # initial logits
    logits = np.log(a + 1e-12)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.5
    lam_factor = 1.02
    best_local_a = a.copy()
    best_local_val = np.inf
    last_improve = time.time()
    for t in range(1, max_iter+1):
        if time.time() > deadline:
            break
        # softmax for simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv
        conv = full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w = w / w_sum
        # gradient
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)
        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # temperature update
        lam = min(lam * lam_factor, 1e6)
        # evaluate objective
        obj = 2.0 * n * max_c / (a.sum() ** 2)

        max_c = conv.max()
        # active mask
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)
        # exponentiated gradient
        a = a * np.exp(-eta * grad_a)
        s = a.sum()

            s = a.sum()

        if obj < best_local_val:
            best_local_val = obj
            best_local_a = a.copy()
            last_improve = time.time()
        # stagnation
        if time.time() - last_improve > 2.0:
            break
    return best_local_a, best_local_val

"""
Propose a step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm
---------
* Multi‑scale evolutionary search:
    - Work on several lengths (1024 … 16384).  For each length a
      diverse set of initial shapes is generated (random Dirichlet,
      edge‑concentrated power‑law, edge Gaussians, centre Gaussians,
      sinusoids, monotone power‑laws and up‑sampled best from a
      smaller length).
* For every candidate two optimisation phases are applied:
    1. Adam on a smooth‑max surrogate of the convolution.
       The temperature λ is increased slowly; the iterate lives on
       the probability simplex via the soft‑max trick.
    2. A short exact sub‑gradient phase (exponentiated‑gradient
       update) that refines the solution w.r.t. the true max‑convolution.
* The best sequence across all lengths and candidates is kept.
* A fixed 1000 s time budget is honoured; time is split roughly
  proportionally to the sequence length, guaranteeing that the longer
  sequences receive enough optimisation steps.
* The function returns a plain Python list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]  # length n


# ----------------------------------------------------------------------
# Generation of initial candidates
def generate_initial_candidates(n: int, best_seq, rng: np.random.Generator) -> list:
    """Return a list of diverse initial probability vectors of length n."""
    candidates = []

    # Warm start from current global best (if same length)
    if best_seq is not None and len(best_seq) == n:
        candidates.append(np.array(best_seq, dtype=np.float64))

    # Historic solution (height_sequence_1) if it exists
    hseq = globals().get("height_sequence_1")
    if hseq is not None:
        cand = np.asarray(hseq, dtype=np.float64)
        if cand.shape[0] > n:
            cand = cand[:n]
        elif cand.shape[0] < n:
            cand = np.pad(cand, (0, n - cand.shape[0]))
        cand = np.maximum(cand, 0.0)
        s = cand.sum()
        if s < 0.01:
            cand = rng.dirichlet(np.ones(n))
        else:
            cand = cand / s
        candidates.append(cand)

    # Pure random Dirichlet
    candidates.append(rng.dirichlet(np.ones(n)))

    # Edge‑concentrated power‑law shapes ((i+1)*(n‑i))**α , α<0
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.2, 7):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            candidates.append(shape)

    # Two‑Gaussian edge shapes
    i = np.arange(n, dtype=np.float64)
    for gamma in [0.2, 0.5, 1.0, 2.0, 5.0]:
        shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # Interior centred Gaussian shapes
    for sigma in [0.3, 0.5, 1.0, 2.0]:
        shape = np.exp(- ((i - (n - 1) / 2) ** 2) / (2 * sigma ** 2))
        shape = shape / shape.sum()
        candidates.append(shape)

    # Sinusoidal families
    for power in [1, 2, 3, 4]:
        shape = np.sin(np.pi * (i + 0.5) / n) ** power
        shape = shape + 1e-12
        shape = shape / shape.sum()
        candidates.append(shape)

    # Monotonically decreasing power‑law tails
    for gamma in [0.5, 1.0, 1.5, 2.0]:
        shape = 1.0 / ((i + 1) ** gamma)
        shape = shape / shape.sum()
        candidates.append(shape)

    # Shuffle and prune to a reasonable size
    rng.shuffle(candidates)
    if len(candidates) > 25:

def upsample_sequence(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linearly up‑sample a probability vector to length new_n."""
    old_n = len(seq)
    if old_n == new_n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, old_n)
    x_new = np.linspace(0.0, 1.0, new_n)
    up = np.interp(x_new, x_old, seq)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s < 1e-12:
        up = np.random.rand(new_n)
        up /= up.sum()
    else:
        up /= s
    return up


# ----------------------------------------------------------------------
# Optimisation phases
def adam_smooth_max(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iter: int,
) -> tuple:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector encountered and its exact objective value.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.5
    lam_factor = 1.02

    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()

    for t in range(1, max_iter + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and smooth‑max weights
        conv = full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

        # exact objective (sum(a) == 1)
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # stagnation handling
        if time.time() - last_improve > 3.0:
            logits += np.random.randn(n) * 1e-2
            last_improve = time.time()

    return best_a, best_val


def subgradient_optimize(
    a0: np.ndarray,
    n: int,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
    max_iter: int = 2000,
) -> tuple:
    """
    Exact sub‑gradient optimisation of max(conv) with an
    exponentiated‑gradient update that keeps the iterate in the simplex.
    Returns the best vector found and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    last_improve = time.time()
    eta = 0.3

    for it in range(max_iter):
        if time.time() > deadline:
            break

        conv = full_convolution(a, fft_len_conv)
        max_c = conv.max()

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad_a = 2.0 * cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.rand(n)
            s = a.sum()
        a /= s

        # exact objective
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} iter={it}] new best {val:.6f}")

        # early stop if stuck
        if time.time() - last_improve > 2.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Multi‑scale evolutionary search for a non‑negative step‑function
    sequence that minimises the evaluation score.  The function respects
    the given time budget (≈ 1000 s) and returns the best sequence found
    as a plain Python list.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    best_seq = None
    best_val = float("inf")
    best_per_len = {}                           # keep best per length

    # lengths are processed from small to large
    lengths = [1024, 2048, 4096, 8192, 16384]

    # allocate a minimal slice per length, the rest proportionally to n
    min_slice = 5.0
    total_weight = sum(lengths)
    for n in lengths:
        if time.time() > deadline:
            break

        # time slice for this length
        slice_budget = max(min_slice, budget_s * n / total_weight)
        slice_deadline = time.time() + slice_budget
        if slice_deadline > deadline:
            slice_deadline = deadline

        print(f"\n=== Optimising length n={n} (budget≈{slice_budget:.1f}s) ===")

        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(2 * n - 1 + n - 1)))

        # collect initial candidates
        candidates = generate_initial_candidates(n, best_seq, rng)

        # add an up‑sampled version of the best solution from a smaller length
        smaller_lengths = [l for l in best_per_len if l < n]
        if smaller_lengths:
            l0 = max(smaller_lengths)
            up = upsample_sequence(best_per_len[l0], n)
            candidates.append(up)

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cid, cand in enumerate(candidates):
            if time.time() > slice_deadline:
                break

            # Normalise to a probability vector
            a0 = cand.astype(np.float64)
            if a0.sum() < 1e-12:
                a0 = rng.dirichlet(np.ones(n))
            else:
                a0 = a0 / a0.sum()

            # Adam – smooth‑max phase
            max_iter_adam = int(5000 * 1024 / n) + 1200
            a_adam, val_adam = adam_smooth_max(
                a0, n, fft_len_conv, fft_len_corr, slice_deadline, max_iter_adam
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                best_per_len[n] = best_seq.copy()
                print(f"[n={n} cand={cid} Adam] GLOBAL BEST {best_val:.6f}")

            # Sub‑gradient refinement (if budget permits)
            if time.time() < slice_deadline:
                a_sub, val_sub = subgradient_optimize(
                    a_adam, n, fft_len_conv, fft_len_corr, slice_deadline, max_iter=2000
                )
                if val_sub < best_val:
                    best_val = val_sub
                    best_seq = a_sub.copy()
                    best_per_len[n] = best_seq.copy()
                    print(f"[n={n} cand={cid} Subgrad] GLOBAL BEST {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final fallback (should never be needed)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = 2.0

    # Optional verification with the provided evaluator (if imported)
    try:
        # evaluate_sequence is assumed to be available in the global namespace
        val_check = evaluate_sequence(best_seq.tolist())
        print(f"\nVerified evaluation (external): {val_check:.6f}")
    except Exception:
        pass

    print(f"\nBest value overall: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
