"""
propose_candidate
-----------------

Searches for a non‑negative step‑function (a probability vector) that
minimises

    F(a) = 2·n·max(a*a) / (∑a)²

where ``a*a`` is the linear convolution of ``a`` with itself.
The optimiser works directly on the probability simplex (∑a = 1) and
uses an Adam optimiser on a smooth‑max surrogate of the max‑convolution.
A temperature schedule drives the surrogate towards the true max,
random “shakes’’ avoid stagnation and a final hill‑climbing phase
refines the best vector found.

The routine respects a 1000 s global time budget (default) and returns
the best sequence (as a plain Python list) discovered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    ``fft_len`` must be a power of two ≥ 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Computes the gradient of  Σ_k w_k·(a*a)_k  w.r.t. a.
    The result has length len(a) and equals
        2·(w ★ a)[i] = 2·Σ_k w_k·a_{i+k-(n-1)} .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ_i x_i = 1}
    """
    if v.ndim != 1:
        raise ValueError("Simplex projection works on 1‑D arrays")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries would become zero – return the uniform point
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """
    Exact objective for a vector that already lies on the simplex
    (∑a == 1).  Returns  2·n·max(conv(a,a)).
    """
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max()

    Find a non‑negative sequence with a low evaluation score.
    The function respects ``budget_s`` seconds, prints progress and
    returns the best sequence as a plain Python list.
    """
