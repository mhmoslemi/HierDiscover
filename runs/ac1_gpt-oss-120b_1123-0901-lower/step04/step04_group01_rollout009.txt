def simplex_projection(v):
    """Return the projection of v onto the probability simplex."""
    # v: 1D array
    # from https://github.com/indyk/fastproj
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all negative? then return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
propose_candidate
-----------------

An alternative to the previous soft‑max Adam approach.
The algorithm works directly on the probability simplex
using a projected Adam optimiser and a cheap sub‑gradient
for the max‑convolution term.

Key ideas
---------

1. **Simplex projection** – the variables are kept on the
   simplex (non‑negative, sum = 1) by an O(n) Euclidean
   projection after each Adam step.  Scaling does not affect
   the objective, so we can safely enforce Σa = 1.

2. **Sub‑gradient of max‑convolution** – if the current
   convolution `c = a * a` attains its maximum at a set of
   lags `L`, the sub‑gradient of `max(c)` is the average of the
   gradients of those active lags:
   ```
   ∇_a max(c) = 2 * (w ★ a)      with  w[l]=1/|L| for l∈L
   ```
   When the active set is small (typically ≤ 30) the convolution
   `w ★ a` is evaluated by simple shifted copies of `a`,
   which is O(|L| · n) and much faster than an FFT.  For larger
   active sets we fall back to an FFT‑based cross‑correlation.

3. **Multiscale optimisation** – we start on a modest size
   (e.g. n = 256), optimise, then up‑sample the best vector to a
   larger size and continue optimisation.  This gives a good
   initial shape for large problems without excessive work.

4. **Adam with learning‑rate decay** – Adam provides adaptive
   step sizes; we decay the base learning‑rate every few thousand
   iterations and occasionally “shake’’ the iterate if no
   improvement has been seen for 30 s.

5. **Local greedy refinement** – after the Adam phase we perform
   a cheap random‑pair transfer move (move a small amount of mass
   from one coordinate to another) to squeeze out any remaining
   improvement.

The whole search runs for the allocated time budget (default
1000 s) and returns the best sequence found.
"""

import time

def _next_pow2(x: int) -> int:
    """Small helper – next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
        {x | x_i ≥ 0, Σ x_i = 1}
    Implementation of the algorithm from Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _shift_gradient(a: np.ndarray, active: np.ndarray) -> np.ndarray:
    """
    Compute 2 * (w ★ a) when the weight vector w is uniform on the
    active indices.  For a small active set we do it with explicit
    shifts (O(|active|·n)).
    """
    n = a.shape[0]
    grad = np.zeros_like(a)
    inv = 1.0 / max(1, active.size)  # avoid divide‑by‑zero
    for lag in active:
        offset = lag - (n - 1)          # signed shift amount
        if offset >= 0:
            # contribution of a shifted right by `offset`
            grad[: n - offset] += a[offset:]
        else:
            off = -offset
            grad[off:] += a[: n - off]
    grad *= 2.0 * inv
    return grad


def _objective(a: np.ndarray, n: int, conv_cache: np.ndarray = None) -> Tuple[float, np.ndarray]:
    """
    Compute the exact objective value 2·n·max(a*a) (Σa = 1).
    Returns (value, convolution_array).  The convolution is also
    returned to avoid recomputation in the calling loop.
    """
    if conv_cache is None:
        conv = _full_convolution(a)
    else:
        conv = conv_cache
    max_c = conv.max()
    val = 2.0 * n * max_c
    return val, conv


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """
    Linear interpolation to obtain a sequence of length `new_n`
    from a shorter sequence `seq`.  The result is projected onto
    the simplex so that it is a valid probability vector.
    """
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    if new_seq.sum() < 1e-12:
        new_seq = np.full(new_n, 1.0 / new_n)
    else:
        new_seq /= new_seq.sum()
    return new_seq

def _optimise_fixed_length(
    a0: np.ndarray,
    n: int,
    deadline: float,
    global_best_val: float,
    global_best_seq: np.ndarray,
) -> Tuple[float, np.ndarray]:
    """
    Adam optimiser on the simplex for a fixed length `n`.
    Returns possibly updated (global_best_val, global_best_seq).
    """
    # FFT sizes (re‑used each iteration)
    fft_len_conv = _next_pow2(2 * n - 1)
    fft_len_corr = _next_pow2(3 * n)            # ≥ (2n‑1)+n‑1

    # Adam state
    m = np.zeros(n, dtype=np.float64)
    v = np.zeros(n, dtype=np.float64)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.12                                   # a bit larger than before
    decay_every = 2000
    max_iters = (
        50000 if n <= 4096 else 30000 if n <= 8192 else 15000
    )
    last_improve = time.time()
    a = a0.copy()
    a = _simplex_projection(a)

    # main optimisation loop
    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # --- convolution & objective ---
        conv = _full_convolution(a, fft_len=fft_len_conv)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # Σa = 1 by construction
        if val < global_best_val - 1e-12:
            global_best_val = val
            global_best_seq = a.copy()
            last_improve = time.time()
            print(f"[n={n} iter={t:05d}] new global best {val:.6f}")

        # --- sub‑gradient of max(conv) ---
        tol = max_c * 1e-12 + 1e-15
        active = np.where(conv >= max_c - tol)[0]
        if active.size <= 30:
            grad = _shift_gradient(a, active)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.size
            grad = _cross_correlation_fft(w, a, fft_len=fft_len_corr)
            grad *= 2.0

        # --- Adam step ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # projection back onto simplex
        a = _simplex_projection(a)

        # learning‑rate decay
        if t % decay_every == 0:
            lr *= 0.7

        # stagnation handling
        if time.time() - last_improve > 30.0:
            a += (np.random.rand(n) - 0.5) * 1e-2
            a = _simplex_projection(a)
            last_improve = time.time()
            print(f"[n={n} iter={t}] shake due to stagnation")

    return global_best_val, global_best_seq

def _local_refinement(
    seq: np.ndarray,
    n: int,
    deadline: float,
    current_best_val: float,
) -> Tuple[np.ndarray, float]:
    """
    Simple random pairwise mass transfer; accepts only improving moves.
    """
    a_local = seq.copy()
    fft_len = _next_pow2(2 * n - 1)
    max_local_iters = 4000
    last_improve = time.time()

    for it in range(max_local_iters):
        if time.time() > deadline:
            break
        i, j = np.random.choice(n, 2, replace=False)
        # transfer a small random fraction of a_i to a_j
        eps_move = min(a_local[i], 0.01) * (0.5 + np.random.rand() * 0.5)
        if eps_move <= 0:
            continue
        a_new = a_local.copy()
        a_new[i] -= eps_move
        a_new[j] += eps_move
        # evaluate
        conv = _full_convolution(a_new, fft_len=fft_len)
        val = 2.0 * n * conv.max()
        if val < current_best_val - 1e-12:
            current_best_val = val
            a_local = a_new
            seq = a_new.copy()
            last_improve = time.time()
            print(f"[local refine] improved to {val:.6f}")

        # occasional shuffling if stuck
        if time.time() - last_improve > 20.0:
            # add a tiny jitter and re‑project
            a_local += (np.random.rand(n) - 0.5) * 5e-4
            a_local = _simplex_projection(a_local)
            last_improve = time.time()

    return seq, current_best_val

                a0 /= max(a0.sum(), 1e-12)

            best_val, best_seq = _optimise_fixed_length(
                a0, n, deadline, best_val, best_seq
            )

            # early exit if target is already beaten
            if best_val < 1.503:
                print(f"Target reached (val={best_val:.6f}) – stopping early.")
                return best_seq.tolist()

        # After the Adam phase, try a cheap local refinement
        if best_seq is not None and best_seq.shape[0] == n and time.time() < deadline:
            best_seq, best_val = _local_refinement(
                best_seq, n, deadline, best_val
            )

    # Fallback – uniform vector (objective = 2) – should never happen
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = 2.0

    # Final report
    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
