"""
Propose a non‑negative step‑function (height sequence) that minimises

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

Strategy
--------
* Represent a as a soft‑max of unconstrained logits x.  This automatically
  enforces a ≥ 0 and the fixed sum S = √(2 n) (a lies on the probability
  simplex multiplied by S).

* Optimise a smooth approximation of max(conv) using Adam:
    - smooth‑max is realised via a temperature τ (τ → 0 → true max).
    - the gradient of the smoothed objective is obtained analytically:
        grad_x = S·p·(grad_a – ⟨p,grad_a⟩)   where p = softmax(x) and
        grad_a = 2·(a * w_rev)  (w are the soft‑max weights of the
        convolution values).

* After the Adam phase, perform a short projected‑gradient polishing that
  targets the *actual* max entry of the convolution (sub‑gradient descent).

* Run many independent restarts with random lengths (500–8000) and varied
  initial shapes (power‑law, symmetric, random‑smooth).  Keep the overall
  best sequence and stop early if the bound drops below the target
  1.503.

The code respects the 1000 s budget, prints progress, and finally returns
the best sequence as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based full linear convolution (real input → real output)
# ----------------------------------------------------------------------
def full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    np.maximum(conv, 0.0, out=conv)             # kill tiny negatives
    return conv


# ----------------------------------------------------------------------
# Projection of a vector onto the simplex {z ≥ 0, sum(z) = s}
# ----------------------------------------------------------------------
def project_onto_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """Euclidean projection of v onto the simplex of size s."""
    if s < 0:
        raise ValueError("Simplex size must be non‑negative")
    n = v.shape[0]
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(1, n + 1)
    # Find the rho index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries will be zero
        return np.zeros_like(v)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimisation on the log‑it (soft‑max) parametrisation
# ----------------------------------------------------------------------
def optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.07,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> (np.ndarray, float):
    """
    Adam optimisation of a soft‑max parametrised step‑function.
    Returns the best sequence (numpy array) and its *evaluate_sequence* score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0
    # pre‑allocate arrays for speed
    while time.time() < deadline:
        # 1) soft‑max mapping
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # stability
        p = exp_shift / np.sum(exp_shift)           # probabilities
        a = target_sum * p

        # 2) convolution and current max
        conv = full_convolution_fft(a, a)
        max_val = conv.max()

        # update best if improved
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(
                f"[{time.time():.2f}] n={n}  new local best (Adam) = {max_val:.6f}"
            )

        # 3) smooth‑max weights (temperature τ)
        tau = max(init_tau_factor * max_val * (0.95 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # 4) gradient wrt a   (cross‑correlation a ⊛ w_rev)
        w_rev = w[::-1]
        conv2 = full_convolution_fft(a, w_rev)          # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # positions aligning a
        grad_a = 2.0 * conv2[idx]

        # 5) chain rule to logits
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # 6) Adam update
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny perturbation (helps escape plateaus)
        if iteration % 300 == 0 and np.random.rand() < 0.2:
            x += np.random.randn(n) * 0.001

        # slow learning‑rate decay
        alpha *= 0.9995

    # final evaluation of the best sequence we saw
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Local polishing using the true max (sub‑gradient descent + simplex proj)
# ----------------------------------------------------------------------
def polish_max_gradient(
    a: np.ndarray,
    target_sum: float,
    deadline: float,
    *,
    max_iters: int = 250,
    init_step: float = 0.05,
    step_decay: float = 0.97,
    tol: float = 1e-8,
) -> (np.ndarray, float):
    """
    Perform a short projected‑gradient descent that directly targets
    the *actual* maximal convolution entry.
    """
    n = a.shape[0]
    best_seq = a.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    step = init_step
    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = full_convolution_fft(best_seq, best_seq)
        i_max = int(np.argmax(conv))
        max_val = conv[i_max]

        # gradient of the *max* entry (2 * shifted a)
        # Build a unit‑vector w at the max index and reuse the convolution trick.
        w = np.zeros(2 * n - 1)
        w[i_max] = 1.0
        conv2 = full_convolution_fft(best_seq, w[::-1])
        idx = (2 * n - 2) - np.arange(n)
        grad = 2.0 * conv2[idx]

        # take a step opposite the gradient, then project onto the simplex
        cand = best_seq - step * grad
        cand = project_onto_simplex(cand, target_sum)

        # evaluate candidate
        cand_score = evaluate_sequence(cand.tolist())
        if cand_score + tol < best_score:
            best_score = cand_score
            best_seq = cand
            # aggressive step increase when we improve
            step = min(step / step_decay, 0.5)
        else:
            # otherwise shrink step
            step *= step_decay
        # tiny random jitter to break ties
        if it % 50 == 0 and np.random.rand() < 0.15:
            jitter = np.random.randn(n) * 1e-4
            best_seq = project_onto_simplex(best_seq + jitter, target_sum)

    return best_seq, best_score
