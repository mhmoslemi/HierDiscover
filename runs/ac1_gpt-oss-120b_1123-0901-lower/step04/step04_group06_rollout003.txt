"""
A three‑phase optimisation for the step‑function attack.

Phase 1 – many short Adam runs on a soft‑max parametrisation.
        Random lengths (≈1 500‑3 500) and a variety of power‑law / exponential
        initialisations are explored.  The surrogate objective is a smoothed
        maximum using a temperature that is annealed during the run.

Phase 2 – refine the current best with a Frank‑Wolfe style step.
        For the current shape `f` we solve the linear programme

            maximise    Σ b_j
            subject to  f * b ≤ M      (M = max(f*f) )
                         b ≥ 0

        (convolution is linear in `b`).  The optimiser `b` is scaled to the
        required ℓ₁‑norm √(2 n) and mixed with `f`.  Repeating a few times
        provably never increases the maximum convolution value and often
        improves it.

Phase 3 – a final longer Adam run starting from the refined sequence.

All steps respect the 1000 s budget, log progress and finally return the
best sequence found as a plain Python list.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Helper – linear convolution via FFT (fast and works for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Full linear convolution of two real vectors using a real FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power‑of‑two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative artefacts caused by floating point errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) representation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.15,
    min_tau_factor: float = 5e-4,
    alpha: float = 0.05,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a length‑`n` vector `a` (non‑negative, Σa = target_sum) by
    maximising a smooth approximation of `max(conv(a,a))`.  The variables are
    logits `x` such that `a = target_sum * softmax(x)`.  Returns the best
    sequence encountered and its *exact* evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ----- map logits to the simplex ---------------------------------
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)               # numerical stability
        p = exp_shift / np.sum(exp_shift)           # probabilities (sum = 1)
        a = target_sum * p

        # ----- convolution & current (true) max -------------------------
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n} new local best = {max_val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -------------------------
        tau = max(init_tau_factor * max_val * (0.94 ** iteration),
                  min_tau_factor * max_val)
        scaled = (conv - max_val) / tau           # ≤ 0
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient w.r.t. a (see derivation in analysis) ----------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)   # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)           # extract needed slice
        grad_a = 2.0 * conv2[idx]

        # ----- gradient w.r.t. logits x (chain rule) -------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ----- Adam update -----------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny jitter to escape plateaus
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # decay the base learning‑rate a little
        alpha *= 0.9995

    # final exact evaluation (the evaluator does its own clipping, etc.)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Linear‑programming direction for the Frank‑Wolfe step
# ----------------------------------------------------------------------
def _solve_lp_direction(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Given a current shape `f` (non‑negative) and a bound `M = max(conv(f,f))`,
    solve

        maximise    Σ b_j
        subject to  conv(f, b) ≤ M   (elementwise)
                    b ≥ 0

    The returned vector `b` satisfies the constraints; None is returned on
    failure.
    """
    n = f.shape[0]
    # Build the convolution matrix: each column j contains f shifted by j.
    offsets = np.arange(n)                         # diagonal offsets 0 … n‑1
    A = sp.diags(f, offsets, shape=(2 * n - 1, n), format='csc')
    c = -np.ones(n)                                # minimise -Σb  ⇒ maximise Σb
    b_ub = np.full(2 * n - 1, M)
    bounds = [(0, None)] * n

    try:
        res = opt.linprog(
            c,
            A_ub=A,
            b_ub=b_ub,
            bounds=bounds,
            method="highs",
            options={"presolve": True},
        )
        if res.success:
            # numerical noise might give tiny negatives → clip
            return np.maximum(res.x, 0.0)
    except Exception as e:
        print(f"LP solve failed: {e}")
    return None

def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """

    """

    if "height_sequence_1" in globals():

    # ==============================================================

    # ==============================================================
    while time.time() < deadline:
        remaining = deadline - time.time()
        # Choose a length; larger lengths are tried early on.

            beta1=0.9,

        # update global best if improved

            best_score = score_local
            best_seq = seq_local.copy()
            print(f"[{time.time():.2f}] Phase 1 global best = {best_score:.6f} (n={n})")

        # early exit if we are already beneath the target
        if best_score < 1.51:
            break

    # ==============================================================
    # Phase 2 – Frank‑Wolfe style refinement using a linear programme
    # ==============================================================

    if best_seq is not None and (time.time() < deadline):
        f = best_seq.copy()
        n = f.shape[0]

                if sum_b <= 0:
