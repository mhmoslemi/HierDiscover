"""
Propose a non‑negative step‑function that minimizes

    2 * n * max(conv(a, a)) / (sum(a) ** 2)

The evaluator automatically normalises the sum, so the objective
effectively equals the maximal value of the self‑convolution.
The search consists of three stages:

1. **Adam on a soft‑max parametrisation**  
   – a = S·softmax(x) with S = √(2 n) (enforces non‑negativity and the
     simplex constraint).  
   – A smoothed max (“log‑sum‑exp”) surrogate is optimised with Adam.
   – Many independent restarts with random lengths (≈ 500‑3000) and
     diverse initial shapes (power‑law, tapered, smoothed) are performed.
   – The current global best is occasionally injected as a warm‑start.

2. **Local stochastic refinement**  
   – Starting from the best sequence found, Gaussian noise is added,
     the vector is re‑normalised, and the noisy candidate is accepted
     if it improves the true objective.  The noise scale is annealed
     during the run.

3. **Fallback** – If no candidate is produced, a uniform sequence is
   returned (guaranteed to have a finite score).

The whole routine respects a user‑specified time budget (default 1000 s)
and prints progress information.  It only uses NumPy and the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper – full linear convolution using real FFT (fast for any length)
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of real vectors x and y via FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative values caused by numerical noise
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Adam optimiser on the log‑it (soft‑max) representation
# ----------------------------------------------------------------------
def _optimise_adam_softmax(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.20,
    min_tau_factor: float = 5e-4,
    alpha_start: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Optimise a sequence of length ``n`` using Adam on the log‑it variables.
    Returns the best sequence encountered (as a NumPy array) and its
    true evaluation score.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0
    alpha = alpha_start

    while time.time() < run_deadline:
        # ---- 1. map logits -> simplex (a = target_sum * softmax(x)) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)            # numerical stability
        p = exp_shift / np.sum(exp_shift)        # soft‑max probabilities
        a = target_sum * p

        # ---- 2. convolution and current max (the true objective) ----
        conv = _full_convolution_fft(a, a)
        max_val = conv.max()
        if max_val < best_score:
            best_score = max_val
            best_seq = a.copy()
            print(f"[{time.time():.2f}] n={n}  new local best = {max_val:.6f}")

        # ---- 3. smooth‑max weights (log‑sum‑exp) --------------------
        # temperature decays geometrically but never below a fraction of max_val
        tau = max(init_tau_factor * max_val * (0.96 ** iteration),
                  min_tau_factor * max_val)
        # w_i ∝ exp((c_i - max)/τ)  (already shifted by -max to avoid overflow)
        w_exp = np.exp((conv - max_val) / tau)
        w_sum = w_exp.sum()
        w = w_exp / w_sum if w_sum > 0 else np.full_like(w_exp, 1.0 / w_exp.size)

        # ---- 4. gradient w.r.t. a (cross‑correlation) ----------------
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)            # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                    # pick the right slice
        grad_a = 2.0 * conv2[idx]

        # ---- 5. chain rule to obtain gradient w.r.t. logits x --------
        dotp = np.dot(p, grad_a)                            # <p, grad_a>
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update ------------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random jitter to escape plateaus
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # very slow decay of the base learning‑rate
        alpha *= 0.9997

    # Final evaluation with the official evaluator (guard against NaNs)
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Stochastic local refinement (simple simulated‑annealing style)
# ----------------------------------------------------------------------
def _refine_local_search(
    seq: np.ndarray,
    target_sum: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Starting from ``seq``, perform cheap random perturbations while time
    permits.  The sum is re‑normalised after each perturbation.
    Returns the best sequence discovered and its evaluation.
    """
    best_seq = seq.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    n = seq.shape[0]

    # initial noise scale – a few percent of the typical coefficient size
    sigma = max(best_seq.mean() * 0.05, 1e-6)

    iteration = 0
    while time.time() < deadline:
        # propose a noisy candidate
        cand = best_seq + np.random.randn(n) * sigma
        # enforce non‑negativity
        np.maximum(cand, 0.0, out=cand)
        # re‑normalise to the required sum
        s = cand.sum()
        if s < 1e-12:            # degenerate – retry
            continue
        cand *= target_sum / s

        # evaluate (the evaluator already caps values at 0–1000)
        score = evaluate_sequence(cand.tolist())
        if score < best_score:
            best_score = score
            best_seq = cand.copy()
            print(
                f"[{time.time():.2f}] refinement iter={iteration}  new best = {score:.6f}"
            )
            # reduce noise when we improve
            sigma = max(sigma * 0.85, 1e-9)
        else:
            # slowly increase the noise to explore more when stuck
            sigma = min(sigma * 1.002, best_seq.mean() * 0.2)

        iteration += 1

    return best_seq, best_score

    Strategy
    --------
    * Randomly pick a length ``n`` (≈ 600‑3200) and initialise a
      power‑law / tapered shape (sometimes smoothed, sometimes taken

    * Optimise the shape with Adam on the soft‑max parametrisation.

    """

    # Warm‑start from a previously known construction, if available.
    # --------------------------------------------------------------

        score = evaluate_sequence(hs_arr.tolist())
        if score < best_score:
            best_score = score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start from global seed = {score:.6f}")
    except Exception:
        pass  # no warm‑start present

    # --------------------------------------------------------------
    # Main multi‑restart optimisation loop
    # --------------------------------------------------------------
    while time.time() < main_deadline:
        remaining = main_deadline - time.time()
        # Choose a length. Early on we explore a wide range,
        # later we concentrate on a cheaper size.
        if remaining > 300:
            n = np.random.randint(1800, 3400 + 1)
        elif remaining > 120:
            n = np.random.randint(1200, 2600 + 1)
        else:
            n = np.random.randint( 800, 1800 + 1)

        # ---------- initialise a candidate shape --------------------
        # power‑law backbone
        p_exp = np.random.uniform(0.20, 0.80)          # exponent
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # optional exponential taper
        if np.random.rand() < 0.4:
            tau_exp = np.random.uniform(0.4, 2.5) * n
            init *= np.exp(-idx / tau_exp)

        # optional moving‑average smoothing
        if np.random.rand() < 0.25:
            klen = np.random.randint(3, 9)
            kernel = np.ones(klen) / klen
            init = np.convolve(init, kernel, mode='same')

        # inject the current best with some probability (helps escape plateaus)
        if best_seq is not None and np.random.rand() < 0.45:
            bs = best_seq
            if bs.size >= n:
                init = bs[:n].copy()
            else:
                pad = np.random.rand(n - bs.size) * 1e-5
                init = np.concatenate([bs, pad])

        init = np.maximum(init, 1e-12)   # avoid exact zeros for log‑softmax

        # scale to the required simplex (Σ a_i = √(2 n))
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum
        x0 = np.log(init + 1e-12)       # logits for the Adam optimiser

        # Allocate a time slice for this run (≈ 10‑15 s, never exceed deadline)
        slice_time = min(15.0, remaining * 0.55)
        run_deadline = min(main_deadline, time.time() + slice_time)

        # ---- run the Adam optimiser on the soft‑max parametrisation ----
        seq_local, score_local = _optimise_adam_softmax(
            n,
            x0,
            target_sum,
            run_deadline,
            init_tau_factor=0.20,
            min_tau_factor=5e-4,
            alpha_start=0.03,
            beta1=0.9,
            beta2=0.999,
        )

        # ---- update global best ---------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} (n={n})"
            )
            # early exit if we already beat the target bound
            if best_score < 1.503:
                break

    # ------------------------------------------------------------------
    # Local stochastic refinement (if we have a candidate)
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = best_seq.shape[0]
        target_sum = np.sqrt(2.0 * n)
        refine_deadline = start_time + budget_s - 0.05   # final safety margin
        best_seq, best_score = _refine_local_search(
            best_seq, target_sum, refine_deadline
        )
        print(f"[{time.time():.2f}] FINAL best after refinement = {best_score:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the very unlikely case that no candidate was generated.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1200
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
```

This implementation builds upon the previous Adam‑based search, uses a smoother annealing schedule, spends a bit more time per restart, and adds a simple stochastic refinement phase that often pushes the final score a little lower.  It respects the 1000 s budget, prints progress messages, and returns a plain list of non‑negative heights ready for `evaluate_sequence`.
