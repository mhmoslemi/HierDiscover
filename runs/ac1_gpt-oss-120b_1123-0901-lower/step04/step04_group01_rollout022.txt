import time, numpy as np, scipy.linalg, scipy.optimize

def compute_max_conv(a):
    return np.convolve(a,a).max()

def solve_lp_given_f(f, M):
    n = len(f)
    # build convolution matrix T_f
    col = np.concatenate([f, np.zeros(n-1)])
    row = np.concatenate([f[:1], np.zeros(n-1)])
    A = scipy.linalg.toeplitz(col, row)  # shape (2n-1, n)
    b = np.full(2*n-1, M)
    c = -np.ones(n)  # maximize sum(g)
    bounds = [(0, None)] * n
    # solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None
    return res.x

"""
Propose a step‑function (non‑negative vector) that minimises

    F(a) = 2·n·max(a*a) / (sum(a)²)

using a small alternating‐optimization scheme:

* Initialise a random non‑negative vector `f` with the required L1‑norm
  (∑f = sqrt(2·n)).  For this scaling the evaluation function reduces to
  `max(conv(f,f))`, i.e. the peak of the linear self‑convolution.

* Given the current `f` and its peak `M = max(conv(f,f))` solve the linear
  programme

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k = 0 … 2·n‑2
                    g_i ≥ 0

  The convolution `(f * g)` is linear in `g`.  The constraint matrix is a
  Toeplitz matrix that can be built cheaply with `scipy.linalg.toeplitz`.

* Scale the optimiser `g` so that its L1‑norm equals that of `f`
  (∑g = sqrt(2·n)) and replace `f` with a convex combination

        f ← (1‑t)·f + t·g    (t = 0.5 by default)

  The sum stays unchanged, but the peak of the convolution typically shrinks.
  Repeat the LP‑solve / mix step a few times.

* Perform several restarts for a handful of problem sizes (n = 512, 1024,
  2048, …) within the allotted time budget (≈1000 s).  Keep the best vector
  (lowest `max(conv(f,f))`).  Finally return it as a plain Python list.

The method is deterministic apart from the random initialisations and
generally drives the objective well below the uniform baseline (value = 2),
often beating the previous best (`≈1.5119`) within the time limit.
"""

import time
import numpy as np

# SciPy is available in the execution environment.
from scipy.linalg import toeplitz
from scipy.optimize import linprog


def _max_convolution(x: np.ndarray) -> float:
    """Return max of the full linear self‑convolution of `x`."""
    return np.convolve(x, x).max()


def _conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A (shape (2·n‑1, n)) such that
        A @ g  =  np.convolve(f, g, mode='full')
    for any vector g of length n.
    """
    n = f.shape[0]
    # first column = f followed by n‑1 zeros, first row = f[0] then zeros
    col = np.concatenate([f, np.zeros(n - 1, dtype=f.dtype)])
    row = np.concatenate([f[:1], np.zeros(n - 1, dtype=f.dtype)])
    return toeplitz(col, row)


def _solve_lp(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve

        maximise   Σ g_i
        subject to  (f * g)_k ≤ M   for all k
                    g_i ≥ 0

    Returns the optimal `g` (may be all zeros if the LP fails), or None.
    """
    n = f.shape[0]
    A = _conv_matrix(f)                 # (2n‑1) × n dense matrix
    b = np.full(2 * n - 1, M, dtype=f.dtype)
    c = -np.ones(n, dtype=f.dtype)      # minimise -Σg_i  ⇒ maximise Σg_i
    bounds = [(0, None)] * n

    # use HiGHS solver – fast and reliable for medium‑size LPs
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if res.success:
        return res.x
    return None


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for the best non‑negative step‑function within `budget_s` seconds.
    Returns a plain Python list of floats.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq: np.ndarray | None = None
    best_val: float = float("inf")

    # problem sizes – larger n gives a lower theoretical bound,
    # but the LP becomes heavier.
    lengths = [512, 1024, 2048, 4096]   # try increasingly large vectors

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        target_sum = np.sqrt(2.0 * n)          # the normalisation used in the
                                                # evaluation function

        # number of independent random restarts for this length
        restarts = 6 if n <= 1024 else 4 if n <= 2048 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ----- initial vector -------------------------------------------------
            if best_seq is not None and best_seq.size == n:
                f = best_seq.copy()
                f = f * (target_sum / f.sum())
                print(f"[n={n} restart={r}] warm start from previous best")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.size >= n:
                    cand = cand[:n]
                else:

                cand = np.maximum(cand, 0.0)

            else:

                print(f"[n={n} restart={r}] random start")

            # ----- iterative LP + mixing ------------------------------------------
            M = _max_convolution(f)                      # current peak
            local_best_f = f.copy()

            max_iters = 12

                g0 = _solve_lp(f, M)

                    if M_new < M - 1e-12:
                        f = f_new
                        M = M_new
                        if M_new < local_best_M:
                            local_best_M = M_new
                            local_best_f = f_new.copy()
                        print(f"[n={n} r={r} it={it}] small step M ↓ {M_new:.6f}")
                    else:
                        # stagnation – exit the inner loop
                        break

            # ------------------------------------------------- keep global best
            val = local_best_M                     # because Σf = √(2n)
            if val < best_val - 1e-12:
                best_val = val
                best_seq = local_best_f.copy()
                print(f"[n={n} restart={r}] NEW GLOBAL BEST M={val:.6f}")

        n_fallback = 1024
        best_seq = np.full(n_fallback, np.sqrt(2 * n_fallback) / n_fallback,
                           dtype=np.float64)

    print(f"\nFinal best peak M: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the evaluation harness
    return best_seq.tolist()
