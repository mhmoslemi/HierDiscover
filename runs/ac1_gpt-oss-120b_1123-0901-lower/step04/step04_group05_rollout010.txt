Propose a step‑function (non‑negative height vector) that minimizes

    F(a) = 2·n·max_k (a*a)[k] / (sum_i a_i)^2 .

The algorithm combines several ideas that proved effective in earlier
experiments and adds a few new ingredients:

* **Rich set of random initialisations** – besides Dirichlet and
  exponential families we also use the *arcsine* shape
  a_i ∝ 1 / sqrt((i+½)(n‑½‑i)).  This shape is known to flatten the
  self‑convolution in the continuous setting and often yields a lower
  max‑convolution than plain sine‑shapes.

* **Adam optimisation on a soft‑max surrogate** – the objective
  `max(conv)` is replaced by a smooth approximation
  `λ⁻¹·log∑exp(λ·conv)`.  The gradient of this surrogate can be written
  in closed form as `∇_a = 2·(w ★ a)` where `w` are the soft‑max weights.
  We keep the parameters in log‑space (`u = log a`) so that the simplex
  constraint Σa = 1 is enforced automatically by the soft‑max.

* **Continuation on the smoothness parameter λ** – λ starts small
  (very smooth) and is multiplied by a modest factor each iteration.
  This lets the optimiser first move in a friendly landscape and later
  focus on the true max‑value.

* **Multiple lengths & progressive up‑sampling** – the search is run on
  several problem sizes (500 … 5000).  Whenever a good solution is found
  for a given length we optionally up‑sample it (linear interpolation)
  to the next larger length and continue optimisation.  Larger sequences
  give the algorithm more degrees of freedom.

* **Stagnation handling** – if the best value has not improved for a few
  seconds we add a small random jitter to the logits and reset the Adam
  moments.

* **Final fine‑tuning** – after the time budget is almost exhausted we
  switch to a higher λ (sharper surrogate) and a smaller learning rate
  to polish the best solution found.

The routine respects the supplied time budget (default 1000 s) and

def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    # use the shifted version for numerical stability
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate with respect to the original
    probabilities ``a``.
    Closed‑form:  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _initialise(n: int, mode: str) -> np.ndarray:
    """Return a non‑negative vector of length ``n`` whose entries sum to 1."""
    i = np.arange(n, dtype=np.float64)

    if mode == "dirichlet":
        a = np.random.dirichlet(np.ones(n))
    elif mode == "exponential":
        r = np.random.uniform(0.2, 0.9)
        a = r ** i
        a /= a.sum()
    elif mode == "sine":
        a = np.sin(np.pi * (i + 1) / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == "arcsine":
        # a_i ∝ 1 / sqrt((i+½)*(n‑½‑i))
        a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == "quadratic":
        a = (1.0 - i / (n - 1.0)) ** 2
        a += 0.001 * np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == "bimodal":
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:  # fallback uniform
        a = np.full(n, 1.0 / n)

    return a


def _upscale(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to a new length ``new_n``."""
    old_n = seq.shape[0]
    if new_n == old_n:
        return seq.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_seq = np.interp(new_x, old_x, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function that minimizes ``evaluate_sequence``.
    The routine respects a time budget (default 1000 s) and returns the best
    sequence discovered as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    best_val = float("inf")
    best_seq = None

    # --------------------------------------------------------------
    # progression of problem sizes – start small and gradually go larger
    lengths = [500, 1000, 2000, 3000, 4000, 5000]

    # several initialisation families
    init_modes = ["dirichlet", "exponential", "sine", "arcsine",
                  "quadratic", "bimodal"]

    restart = 0
    while time.time() < deadline:
        # ------------------------------------------------------------------
        # pick length in a round‑robin fashion
        n = lengths[restart % len(lengths)]
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound

        # ------------------------------------------------------------------
        # pick an initialisation mode (cycling through the list)
        mode = init_modes[restart % len(init_modes)]

        # base random start
        a0 = _initialise(n, mode)

        # occasionally start from the current best (if same length)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.2:
            a0 = 0.5 * a0 + 0.5 * best_seq
            a0 = np.maximum(a0, 0.0)
            a0 /= a0.sum()

        # ------------------------------------------------------------------
        # log‑space parametrisation (soft‑max guarantees positivity & sum=1)
        u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # continuation parameters (smooth‑max)
        lam = 0.2                     # start very smooth
        lam_factor = 1.015            # slow increase
        base_lr = 0.07                # initial learning rate for logits

        it = 0
        last_improve = time.time()
        max_iter = 18000

        # ------------------------------------------------------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- probabilities from logits (soft‑max) -----
            u_shift = u - np.max(u)          # numerical stability
            a = np.exp(u_shift)
            a /= a.sum()                     # Σa = 1

            # ----- convolution and soft‑max weights -----
            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # ----- gradient w.r.t a (smooth‑max surrogate) -----
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # ----- map to log‑space gradient (soft‑max Jacobian) -----
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update ----------------------------------
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9997 ** it)    # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- increase smoothness (λ continuation) -----
            lam *= lam_factor

            # ----- exact evaluation every few steps ------------
            if it % 25 == 0:
                max_conv = conv.max()
                val = 2.0 * n * max_conv          # because Σa = 1
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] n={n:04d} iter={it:05d}  "
                          f"new best = {val:.6f}")

            # ----- stagnation handling ------------------------
            if time.time() - last_improve > 2.5:
                # small random shake of logits + reset Adam moments
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # ------------------------------------------------------------------
        # after the inner optimisation loop we may up‑sample the best vector
        # to the next larger length (if any) – this gives the next run a head start.
        if best_seq is not None:
            # try to up‑scale only when the current length is the smallest
            # among the schedule and we still have room for a larger problem.
            next_len_idx = (restart % len(lengths)) + 1
            if next_len_idx < len(lengths):
                new_n = lengths[next_len_idx]
                if new_n > n:
                    best_seq = _upscale(best_seq, new_n)

        restart += 1
        print(f"[Restart {restart:03d}] finished n={n}, best so far = {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Final fine‑tuning phase – use a sharper surrogate (larger λ) and a
    # smaller learning rate to polish the best candidate.
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        u = np.log(best_seq + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 6.0           # fairly sharp surrogate
        lam_factor = 1.001
        base_lr = 0.022

        it = 0
        max_iter = 40000
        last_improve = time.time()

        while time.time() < deadline and it < max_iter:
            it += 1

            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            lam *= lam_factor

            if it % 40 == 0:
                max_conv = conv.max()
                val = 2.0 * n * max_conv
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")

    # ----------------------------------------------------------------------
    # Safety fallback (should never be needed)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length = {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
