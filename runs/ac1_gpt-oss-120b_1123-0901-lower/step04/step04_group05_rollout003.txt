def _project_simplex(v):
    # return projection onto {x >=0, sum(x)=1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _subgradient_max_conv(a, conv):
    # conv = full convolution a*a
    max_val = conv.max()
    # get one index where max occurs (first)
    m = np.argmax(conv)
    n = a.shape[0]
    # compute subgradient
    grad = np.zeros_like(a)
    # For i from 0 to n-1, if 0 <= m-i < n:
    lo = max(0, m - (n-1))
    hi = min(m, n-1)
    # i in [lo, hi] yields m-i in [0, n-1]
    # then grad[i] = 2*a[m-i]
    # but if i==m-i (i == m/2) -> grad[i] = a[i]
    # vectorized:
    idx = np.arange(lo, hi+1)
    rev_idx = m - idx
    grad[idx] = 2 * a[rev_idx]
    if m % 2 == 0:
        i_mid = m // 2
        if lo <= i_mid <= hi:
            grad[i_mid] = a[i_mid]  # correct factor
    return grad, max_val

def _run_adam(a0, n, deadline, best_val, best_seq):
    # Setup
    size_fft = 1 << int(np.ceil(np.log2(2*n-1)))
    u = np.log(a0 + 1e-12)  # logits
    m = np.zeros_like(u)
    v = np.zeros_like(u)
    lam = 0.3
    lam_factor = 1.025
    base_lr = 0.04
    it = 0
    max_iter = 12000
    last_improve = time.time()
    while time.time() < deadline and it < max_iter:
        it += 1
        # softmax
        u_shift = u - np.max(u)
        a = np.exp(u_shift)
        a /= a.sum()
        # conv
        conv = _full_convolution(a, size=size_fft)
        # smooth max weights
        w = _softmax_weights(conv, lam)
        # gradient wrt a: 2 * (w * a) convolution? Already provided _softmax_gradient
        grad_a = _softmax_gradient(a, w, size_corr=size_fft*2)  # maybe bigger
        # map to logits
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)
        # Adam updates
        m = 0.9 * m + 0.1 * grad_u
        v = 0.999 * v + 0.001 * (grad_u ** 2)
        it_f = float(it)
        m_hat = m / (1.0 - 0.9 ** it_f)
        v_hat = v / (1.0 - 0.999 ** it_f)
        lr = base_lr * (0.9998 ** it)
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        lam *= lam_factor
        # evaluate exact every 20 steps
        if it % 20 == 0:
            conv = _full_convolution(a, size=size_fft)  # recalc maybe but we have conv already
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam restart {restart}] iter {it} new best = {val:.6f}")
        # random jitter if stagnated
        if time.time() - last_improve > 2.5:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m.fill(0)
            v.fill(0)
            last_improve = time.time()
    return a, best_val, best_seq

"""
Propose a step‑function (non‑negative heights) that minimises
    evaluate_sequence(sequence).

The search combines three complementary strategies

1. **Adam on a smooth‑max surrogate** – the objective
   2·n·max(a*a) is approximated by a soft‑max over the full self‑convolution.
   The logits of a soft‑max parametrisation guarantee a≥0, Σa=1.
   A continuation on the temperature λ slowly sharpens the approximation.

2. **Projected sub‑gradient descent on the true max‑convolution** – after the
   Adam phase the exact max index is identified and a sub‑gradient
   (2·a shifted) is taken.  The iterate is projected back onto the simplex.

3. **Local random refinement** – final short hill‑climbing with random
   perturbations, again projected onto the simplex.

The routine runs many short restarts with different lengths (1000–4000)
and random initial families (Dirichlet, exponential, sine‑shapes, etc.).
If the global variable ``height_sequence_1`` exists it is occasionally used
as a warm start.  All candidates are kept normalised (Σa=1) so the true
objective equals 2·n·max(conv).  The best sequence found before the time
budget expires is returned as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: linear self‑convolution using FFT (fast for all lengths)
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) using FFT."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Soft‑max over convolution values (smooth approximation of max)
def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """wₖ = exp(lam·(convₖ‑max(conv)))  –  a probability distribution."""
    m = conv.max()
    # shift to avoid overflow
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray,
                      size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate ∑ₖ wₖ·(a*a)[k] w.r.t. a.
    Closed form:  ∇ = 2·(w ★ a)  (linear convolution of w with a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice indices that correspond to the original length
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
# Projection onto the probability simplex {x ≥ 0, Σx = 1}
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Sub‑gradient of the exact max‑convolution (used in Phase 2)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray):
    """
    Returns a sub‑gradient of max(conv) w.r.t. a and the value max(conv).
    The gradient corresponds to the (unique) index achieving the maximum;
    if several indices tie the first one is used.
    """
    max_val = conv.max()
    m = int(np.argmax(conv))               # index where max occurs
    n = a.shape[0]
    grad = np.zeros_like(a)

    # i contributes if 0 ≤ m‑i < n  → i ∈ [max(0,m‑(n‑1)), min(m,n‑1)]
    lo = max(0, m - (n - 1))
    hi = min(m, n - 1)
    idx = np.arange(lo, hi + 1, dtype=int)
    rev_idx = m - idx                     # indices in a that pair with idx

    # generic case: factor 2
    grad[idx] = 2.0 * a[rev_idx]

    # correction when i = m‑i (i.e. m even and i = m/2)
    if (m % 2) == 0:
        i_mid = m // 2
        if lo <= i_mid <= hi:
            grad[i_mid] = a[i_mid]       # factor should be 1 instead of 2

    return grad, max_val


# ----------------------------------------------------------------------
# Phase 1 – Adam on smooth‑max surrogate
def _run_adam(a0: np.ndarray, n: int, deadline: float,
              best_val: float, best_seq: np.ndarray,
              restart_id: int) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Adam optimisation on the soft‑max approximation of max(a*a).
    Returns the final a, the possibly improved best_val and best_seq.
    """
    # FFT size for convolutions (fixed during this run)
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    size_corr = 1 << int(np.ceil(np.log2(conv_fft_size + n)))

    # log‑softmax parametrisation (guarantees a ≥ 0, Σa = 1)
    u = np.log(a0 + 1e-30)      # logits
    m1 = np.zeros_like(u)      # Adam first moment
    v1 = np.zeros_like(u)      # Adam second moment

    lam = 0.35                  # start with a smooth landscape
    lam_factor = 1.025
    base_lr = 0.045
    it = 0
    max_iter = 12000
    last_improve = time.time()

    while time.time() < deadline and it < max_iter:
        it += 1

        # ----- soft‑max – obtains a probability vector -----
        u_shift = u - np.max(u)            # improve numerical stability
        a = np.exp(u_shift)
        a /= a.sum()

        # ----- convolution and smooth‑max weights -----
        conv = _full_convolution(a, size=conv_fft_size)
        w = _softmax_weights(conv, lam)

        # ----- gradient w.r.t. a (smooth‑max surrogate) -----
        grad_a = _softmax_gradient(a, w, size_corr=size_corr)

        # ----- map gradient to log‑space (Jacobian of soft‑max) -----
        dot = np.dot(a, grad_a)
        grad_u = a * (grad_a - dot)

        # ----- Adam updates -----
        m1 = 0.9 * m1 + 0.1 * grad_u
        v1 = 0.999 * v1 + 0.001 * (grad_u ** 2)

        it_f = float(it)
        m_hat = m1 / (1.0 - 0.9 ** it_f)
        v_hat = v1 / (1.0 - 0.999 ** it_f)

        lr = base_lr * (0.9997 ** it)      # gentle decay
        u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

        # ----- temperature continuation -----
        lam *= lam_factor

        # ----- exact evaluation every few steps -----
        if it % 20 == 0:
            # recompute conv for safety (already computed)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[Adam  R{restart_id:03d}] it {it:05d}  new best = {val:.6f}")

        # ----- escape stagnation -----
        if time.time() - last_improve > 2.5:
            u += (np.random.rand(n) - 0.5) * 1e-3
            m1.fill(0.0)
            v1.fill(0.0)
            last_improve = time.time()

    # Return a (still normalised) for the next phase
    a = np.exp(u - np.max(u))
    a /= a.sum()
    return a, best_val, best_seq


# ----------------------------------------------------------------------
# Phase 2 – Projected sub‑gradient descent on the true max‑convolution
def _run_subgradient(a0: np.ndarray, n: int, deadline: float,
                     best_val: float, best_seq: np.ndarray,
                     restart_id: int) -> tuple[np.ndarray, float, np.ndarray]:
    """
    Sub‑gradient descent on  max(a*a) with projection onto the simplex.
    """
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    a = a0.copy()
    t = 0
    eta0 = 0.7                         # initial step size
    last_improve = time.time()
    max_iter = 6000

    while time.time() < deadline and t < max_iter:
        t += 1

        conv = _full_convolution(a, size=conv_fft_size)
        grad, max_val = _subgradient_max_conv(a, conv)

        # Diminishing step size (1/√t) works well for sub‑gradients
        eta = eta0 / math.sqrt(t)
        a = a - eta * grad

        # Project back onto the simplex Σa = 1, a ≥ 0
        a = _project_simplex(a)

        val = 2.0 * n * max_val
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            last_improve = time.time()
            print(f"[Subgrad R{restart_id:03d}] it {t:05d}  new best = {val:.6f}")

        # Random shake if we are stuck for a while
        if time.time() - last_improve > 2.0:
            a += 1e-4 * np.random.randn(n)
            a = np.clip(a, 0.0, None)
            a = _project_simplex(a)
            last_improve = time.time()

    return a, best_val, best_seq


# ----------------------------------------------------------------------
# Phase 3 – Simple local random refinement (hill climbing)
def _run_local_search(a0: np.ndarray, n: int, deadline: float,
                      best_val: float, best_seq: np.ndarray,
                      restart_id: int) -> tuple[float, np.ndarray]:
    """
    Tiny stochastic refinement: small Gaussian perturbations,
    projection onto the simplex and acceptance only on improvement.
    """
    conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    a = a0.copy()
    last_improve = time.time()
    attempts = 0

    while time.time() < deadline and attempts < 20000:
        attempts += 1
        # propose a slight random move (relative to current magnitude)
        eps = 0.005 * (np.random.randn(n) * a)
        a_new = a + eps
        a_new = np.clip(a_new, 0.0, None)
        a_new = _project_simplex(a_new)

        conv = _full_convolution(a_new, size=conv_fft_size)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a_new.copy()
            a = a_new
            last_improve = time.time()
            print(f"[Local R{restart_id:03d}] it {attempts:05d}  new best = {val:.6f}")

        # stop early if no improvement for a couple of seconds
        if time.time() - last_improve > 1.5:
            break

    return best_val, best_seq
