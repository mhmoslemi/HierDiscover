def run_cma_es_opt(n, init_candidates, deadline, popsize=None, max_gen=None):
    # returns best a (1D np.ndarray) and best_val

   # if init_candidates is None or empty: generate a random one based on Dirichlet.
   # else choose first.

   n_dims = n

   # set popsize lambda
   if popsize is None:
       lam = max(10, 4 + int(3 * np.log(n_dims)))  # typical CMA-ES
   else:
       lam = popsize
   mu = lam // 2
   # compute recombination weights
   w = np.log(mu + 0.5) - np.log(np.arange(1, mu+1))
   w = w / w.sum()
   mu_eff = 1.0 / np.sum(w**2)

   # initial mean in logit space
   if init_candidates:
       logs = [np.log(c + 1e-15) for c in init_candidates]
       m = np.mean(logs, axis=0)
   else:
       m = np.zeros(n_dims)   # uniform

   sigma = 0.5
   # ensure sigma not too small or large
   sigma = max(1e-4, min(sigma, 2.0))

   best_a = None
   best_val = np.inf
   # baseline objective for sigma adaptation: use current mean's softmax for parent
   parent_a = softmax(m[np.newaxis,:])[0]
   parent_val = compute_objective(parent_a)
   if parent_val < best_val:
       best_val = parent_val
       best_a = parent_a.copy()

   t = 0

   while time.time() < deadline and (max_gen is None or t < max_gen):
       t += 1
       # sample population
       Y = m + sigma * np.random.randn(lam, n_dims)   # shape (lam, n)
       # compute softmax per row
       maxY = np.max(Y, axis=1, keepdims=True)
       E = np.exp(Y - maxY)
       A = E / np.sum(E, axis=1, keepdims=True)   # shape (lam, n)
       # compute objectives
       vals = np.empty(lam, dtype=np.float64)
       for i in range(lam):
           vals[i] = compute_objective(A[i])
       # sort indices
       idx_sorted = np.argsort(vals)
       # select top mu
       best_inds = idx_sorted[:mu]
       # recombination to update mean
       m = np.dot(w, Y[best_inds])   # w shape (mu,) dot (mu,n) => (n,)

       # update best global
       if vals[idx_sorted[0]] < best_val:
           best_val = vals[idx_sorted[0]]
           best_a = A[idx_sorted[0]].copy()
           # maybe also update parent for sigma adaptation
           parent_val = best_val

       # adapt sigma using 1/5 success rule, success defined as #offspring improving over parent_val
       # success count from previous parent
       successes = np.sum(vals < parent_val)
       success_rate = successes / lam
       # adapt sigma
       if success_rate > 0.2:
           sigma *= 1.2
       else:
           sigma *= 0.8
       sigma = max(1e-5, min(sigma, 5.0))

   return best_a, best_val

"""
CMA‑ES based optimizer for the step‑function problem.

The algorithm works in three progressive stages:

1.  A cheap CMA‑ES run on a very small dimension (n≈256) to discover a
    good “shape’’ of the distribution.  The simplex constraint (a_i≥0,
    Σa_i=1) is enforced by representing the vector in log‑it space and
    applying a softmax.

2.  The best small‑dimensional vector is interpolated to a medium size
    (n≈1024) and refined again with CMA‑ES.

3.  The medium solution is up‑scaled to a large size (n≈4096) and a final
    CMA‑ES run is performed.  After the evolutionary phase a short
    pair‑wise local search (mass transfers between two coordinates) is
    applied to squeeze out a final improvement.

The whole process respects the global time budget (default 1000 s) and
always keeps the best vector discovered.  The returned list is a plain
Python list of non‑negative floats whose sum is 1, ready to be fed to
`evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – identical to the evaluator’s internal logic
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of suitable length."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))  # next power of two
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective 2·n·max(a∗a) / (sum(a)²) for a non‑negative vector.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    n = a.shape[0]
    max_c = full_convolution(a).max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Simple helper: stable softmax for a 2‑D array (rows = individuals)
# ----------------------------------------------------------------------
def softmax_rows(Y: np.ndarray) -> np.ndarray:
    """Row‑wise softmax, numerically stable."""
    maxY = np.max(Y, axis=1, keepdims=True)
    E = np.exp(Y - maxY)
    return E / np.sum(E, axis=1, keepdims=True)


# ----------------------------------------------------------------------
# Upscaling of a low‑dimensional candidate to a higher dimension
# ----------------------------------------------------------------------
def upscale_vector(v_small: np.ndarray, n_large: int) -> np.ndarray:
    """
    Linear interpolation of a simplex vector to a larger size.
    The result is re‑normalised to sum to 1.
    """
    n_small = v_small.shape[0]
    x_small = np.linspace(0.0, 1.0, n_small)
    x_large = np.linspace(0.0, 1.0, n_large)
    v_large = np.interp(x_large, x_small, v_small)
    v_large = np.maximum(v_large, 0.0)
    total = v_large.sum()
    if total > 0.0:
        v_large /= total
    else:
        v_large = np.full(n_large, 1.0 / n_large)
    return v_large

# ----------------------------------------------------------------------
# Core optimiser – a simplified (μ/λ)‑CMA‑ES with isotropic mutation
# ----------------------------------------------------------------------
def run_cma_es_opt(
    n: int,
    init_candidates: list,
    deadline: float,
    popsize: int = None,
    max_gen: int = None,
) -> tuple[np.ndarray, float]:
    """
    Simple CMA‑ES on the probability simplex.
    Returns the best vector and its objective value discovered before *deadline*.
    """
    # population size (λ) – classic CMA‑ES rule
    if popsize is None:
        lam = max(10, 4 + int(3 * np.log(n)))
    else:
        lam = popsize
    mu = lam // 2

    # recombination weights (log‑weights, normalised)
    w = np.log(mu + 0.5) - np.log(np.arange(1, mu + 1))
    w /= np.sum(w)

    # initialise mean in log‑it space
    if init_candidates:
        logs = [np.log(c + 1e-15) for c in init_candidates]
        m = np.mean(logs, axis=0)
    else:
        m = np.zeros(n, dtype=np.float64)

    sigma = 0.5                       # initial global step size
    sigma = np.clip(sigma, 1e-5, 5.0)

    # store the best elite ever seen
    best_a = None
    best_val = np.inf

    # current “parent’’ (softmax of the mean) – used for the 1/5 success rule
    parent_a = softmax_rows(m[np.newaxis, :])[0]
    parent_val = compute_objective(parent_a)
    if parent_val < best_val:
        best_val = parent_val
        best_a = parent_a.copy()

    gen = 0
    while time.time() < deadline and (max_gen is None or gen < max_gen):
        gen += 1

        # --------------------------------------------------------------
        # Sample λ offspring in log‑it space and map to the simplex
        # --------------------------------------------------------------
        Y = m + sigma * np.random.randn(lam, n)        # (λ , n)
        A = softmax_rows(Y)                            # (λ , n)

        # --------------------------------------------------------------
        # Evaluate objective for every offspring
        # --------------------------------------------------------------
        vals = np.empty(lam, dtype=np.float64)
        for i in range(lam):
            vals[i] = compute_objective(A[i])

        # --------------------------------------------------------------
        # Select the μ best individuals and recombine the mean
        # --------------------------------------------------------------
        idx = np.argsort(vals)
        elite_idx = idx[:mu]
        m = np.dot(w, Y[elite_idx])     # new mean (log‑it)

        # --------------------------------------------------------------
        # Track the global best
        # --------------------------------------------------------------
        if vals[idx[0]] < best_val:
            best_val = vals[idx[0]]
            best_a = A[idx[0]].copy()
            parent_val = best_val          # treat new elite as next parent

        # --------------------------------------------------------------
        # 1/5‑success rule to adapt sigma
        # --------------------------------------------------------------
        successes = np.sum(vals < parent_val)
        success_rate = successes / lam
        if success_rate > 0.2:
            sigma *= 1.2
        else:
            sigma *= 0.8
        sigma = np.clip(sigma, 1e-6, 5.0)

        # occasional “shake’’ when stagnating for a few seconds
        if gen % 20 == 0 and time.time() < deadline:
            # add a tiny random offset to avoid exact convergence
            m += (np.random.rand(n) - 0.5) * 1e-3

        # optional progress output
        if gen % 100 == 0:
            print(f"[CMA‑ES n={n} gen={gen}] best={best_val:.6f} sigma={sigma:.4f}")

    return best_a, best_val

def run_pairwise_local_search(a: np.ndarray, deadline: float, step: float = 0.02) -> tuple[np.ndarray, float]:
    """
    Simple (1+1) style hill‑climber: repeatedly move a fraction *step* of
    mass from a random source index to a random target index and keep
    the move if the objective improves.
    """
    best_a = a.copy()
    best_val = compute_objective(best_a)
    n = best_a.shape[0]

    while time.time() < deadline:
        i = np.random.randint(n)          # source
        if best_a[i] < 1e-14:
            continue
        j = np.random.randint(n - 1)      # target (different)
        if j >= i:
            j += 1

        delta = step * best_a[i]          # mass to transfer
        trial = best_a.copy()
        trial[i] -= delta
        trial[j] += delta
        # sum unchanged, non‑negativity preserved by construction
        val = compute_objective(trial)
        if val < best_val:
            best_val = val
            best_a = trial
            # slow down step size a bit after a success
            step = min(step * 1.05, 0.3)

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises the provided
    evaluation function.  The routine works in three progressive steps,
    each using a lightweight CMA‑ES optimiser followed by a short local
    refinement.  It respects the total time budget and returns a plain
    Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Stage 1 – tiny dimensionality (fast, gives a coarse shape)
    # ------------------------------------------------------------------
    n1 = 256
    t1_end = time.time() + min(0.2 * (deadline - time.time()), 120.0)
    cand1 = warm_start_candidates(n1)
    best1, val1 = run_cma_es_opt(n1, cand1, t1_end)
    print(f"[Stage 1] n={n1} best={val1:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – medium dimensionality (refinement of the coarse shape)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n2 = 1024
        upscale1 = upscale_vector(best1, n2)
        cand2 = warm_start_candidates(n2) + [upscale1]
        t2_end = time.time() + min(0.3 * (deadline - time.time()), 250.0)
        best2, val2 = run_cma_es_opt(n2, cand2, t2_end)
        print(f"[Stage 2] n={n2} best={val2:.6f}")
    else:
        best2, val2 = best1, val1
        n2 = n1

    # ------------------------------------------------------------------
    # Stage 3 – large dimensionality (the final high‑resolution optimisation)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        n3 = 4096
        upscale2 = upscale_vector(best2, n3)
        cand3 = warm_start_candidates(n3) + [upscale2]
        t3_end = time.time() + min(0.4 * (deadline - time.time()), 400.0)
        best3, val3 = run_cma_es_opt(n3, cand3, t3_end)
        print(f"[Stage 3] n={n3} best={val3:.6f}")
    else:
        best3, val3 = best2, val2
        n3 = n2

    # ------------------------------------------------------------------
    # Final local fine‑tuning (if any time remains)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        local_end = deadline
        best3, val3 = run_pairwise_local_search(best3, local_end, step=0.02)
        print(f"[Local search] final best={val3:.6f}")

    # ------------------------------------------------------------------
    # Choose the overall best among the three stages
    # ------------------------------------------------------------------
    candidates = [(best1, val1, n1), (best2, val2, n2), (best3, val3, n3)]
    best_vec, best_val, best_n = min(candidates, key=lambda x: x[1])

    # Safeguard: ensure non‑negative, sum to 1 and clip extreme values
    best_vec = np.maximum(best_vec, 0.0)
    s = best_vec.sum()
    if s > 0.0:
        best_vec /= s
    else:
        best_vec = np.full(best_n, 1.0 / best_n)

    print(f"\n=== DONE === best value {best_val:.6f} with length {best_n}")
    return best_vec.tolist()
