def simplex_projection(v):
    # Project v onto simplex {x>=0, sum=1}
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid symmetric Adam + sub‑gradient optimizer.

The search works in three stages.

1. **Exploratory Adam runs**  
   * Each candidate is parametrised by logits `u` → `a = softmax(u)`.  
   * After the soft‑max we explicitly enforce symmetry  
     `a = (a + a[::-1]) / 2` – this reduces the effective
     dimension and concentrates the search on the most promising
     family of step functions.  
   * A smooth‑max surrogate of the true objective is built with a
     temperature `λ`.  `λ` starts small (very smooth) and is increased
     exponentially until it reaches a large value (`≈30`), at which
     point the surrogate approximates the true `max` very well.  
   * Adam (β₁=0.9, β₂=0.999) updates the logits.  When progress stalls a
     tiny random shake is added to the logits to escape shallow plateaus.
   * Several random initialisation families (Dirichlet, exponential,
     half‑sine, powered‑sine, quadratic, power‑law, bimodal) are used.
   * The best sequence seen so far (lowest `evaluate_sequence` value) is
     stored globally.

2. **Fine‑tuning with a hard max sub‑gradient**  
   * Starting from the best sequence we minimise the *exact* objective
     `2·n·max(conv)` (the sum is kept equal to 1).  
   * The sub‑gradient of the `max` is simply a shifted copy of the current
     vector: if `k` is the index of the maximal convolution entry then
     `∇_a max(conv) = 2·a shifted by k`.  
   * After a gradient step we project the vector back onto the probability
     simplex (non‑negative, sum = 1) using the O(n) algorithm of Duchi et al.
   * The step size decays slowly; the procedure runs until the overall time
     budget is exhausted.

3. **Fallback**  
   * If nothing has been found (should never happen) we return the uniform
     vector.

All heavy work (FFT‑based convolutions, soft‑max weighting and gradients)
is performed with NumPy’s fast FFT.  The whole routine respects the user
supplied time budget (default 1000 s) and prints occasional progress
messages.

The returned object is a plain Python list of non‑negative heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights wₖ = exp(lam·convₖ) / Σⱼ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the original probabilities ``a``.
    The closed form is  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # indices n‑1 … n‑1+n‑1 correspond to the convolution with the reversed vector
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x ≥ 0, Σ x = 1}.
    Implementation of Duchi et al. (2008).
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function (list of heights) that minimises
    ``evaluate_sequence``.  The routine respects ``budget_s`` seconds and
    returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.4   # safety margin

    best_val = float("inf")
    best_seq = None
    restart = 0

    # ------------------------------------------------------------------
    # Lengths to try – larger n gives more flexibility but also higher cost
    lengths = [2000, 2500, 3000, 3500, 4000]

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # pick a length (cycle through the list)
        n = lengths[restart % len(lengths)]

        # Pre‑compute FFT sizes that stay constant for the whole inner run
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound

        # ------------ random initialisation (several families) ----------
        mode = restart % 8
        if mode == 0:                                 # Dirichlet
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:                               # pure exponential
            r = np.random.uniform(0.2, 0.9)
            a0 = r ** np.arange(n, dtype=np.float64)
        elif mode == 2:                               # half‑sine
            i = np.arange(1, n + 1, dtype=np.float64)
            a0 = np.sin(np.pi * i / (n + 1))
        elif mode == 3:                               # powered half‑sine
            p = np.random.uniform(0.5, 3.0)
            a0 = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
        elif mode == 4:                               # exponential with a tiny floor
            r = np.random.uniform(0.3, 0.85)
            a0 = r ** np.arange(n, dtype=np.float64) + 0.01 * np.random.rand(n)
        elif mode == 5:                               # quadratic (parabolic) bump
            x = np.linspace(-1.0, 1.0, n)
            a0 = 1.0 - x ** 2
        elif mode == 6:                               # power‑law decay
            p = np.random.uniform(0.5, 2.5)
            a0 = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        else:                                         # bimodal half‑sine glued together
            half = n // 2
            left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
            right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
            a0 = np.concatenate([left, right])

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()
        # enforce symmetry from the start
        a0 = (a0 + a0[::-1]) * 0.5

        # Occasionally seed from the current best (if size matches)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.15:
            a0 = np.array(best_seq, copy=True)

        # Log‑space (soft‑max) parametrisation
        u = np.log(a0 + 1e-30)

        # Adam bookkeeping
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # λ‑continuation: start smooth, end sharp
        lam_start = 0.5
        lam_target = 30.0
        max_iter = 20000
        lam_factor = (lam_target / lam_start) ** (1.0 / max_iter)
        lam = lam_start

        # learning‑rate schedule
        base_lr = 0.08

        it = 0
        last_improve = time.time()

        # --------------------------------------------------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            # ----- obtain probabilities from logits --------------------
            u_shift = u - np.max(u)          # improve numerical stability
            a = np.exp(u_shift)
            a /= a.sum()                     # Σ a = 1
            a = (a + a[::-1]) * 0.5          # enforce symmetry

            # ----- convolution and smooth‑max weights ------------------
            conv = _full_convolution(a, size=conv_fft_size)
            w = _softmax_weights(conv, lam)

            # ----- gradient w.r.t. a (smooth‑max surrogate) ----------
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # ----- map to log‑space gradient (soft‑max Jacobian) -------
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # ----- Adam update -----------------------------------------
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9998 ** it)    # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # ----- raise smoothness ------------------------------------
            lam *= lam_factor

            # ----- exact evaluation every few steps --------------------
            if it % 20 == 0:
                # because Σ a = 1, exact objective = 2·n·max(conv)
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # ----- shake if we have stalled -----------------------------
            if time.time() - last_improve > 3.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        restart += 1
        print(f"[Restart {restart:03d}] finished it={it}, best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # ---------- Sub‑gradient fine‑tuning on the exact max ----------
    # ------------------------------------------------------------------
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        a = best_seq.copy()
        a = np.maximum(a, 0.0)
        a /= a.sum()                         # ensure exact simplex

        sub_deadline = time.time() + max(5.0, deadline - time.time())
        step0 = 0.12
        it = 0
        last_improve = time.time()

        while time.time() < sub_deadline:
            it += 1
            conv = _full_convolution(a, size=conv_fft_size)
            k = int(np.argmax(conv))

            # sub‑gradient of the max entry (2·a shifted by k)
            grad = np.zeros_like(a)
            start = max(0, k - (n - 1))
            end = min(n, k + 1)
            idx = np.arange(start, end, dtype=int)
            grad[idx] = 2.0 * a[k - idx]

            # simple decreasing step size
            eta = step0 / (1.0 + it * 1e-3)
            a = a - eta * grad
            a = _project_onto_simplex(a)

            if it % 10 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Sub‑grad] iter {it:05d}  new best = {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback: uniform vector (should never be used)
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
