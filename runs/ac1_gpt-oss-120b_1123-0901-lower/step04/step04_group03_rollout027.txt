def smoothed_adam_opt(n, init_a, deadline):
    x = np.log(init_a + 1e-15)
    best_val = np.inf
    best_a = None
    t = 0
    beta1, beta2 = 0.9, 0.999
    m = np.zeros(n)
    v = np.zeros(n)
    gamma_schedule = [1, 2, 4, 8, 16, 32, 64]
    for gamma in gamma_schedule:
        # allocate time slice maybe
        while time.time() < deadline:
            a = softmax(x)
            conv = full_convolution(a)
            # compute smoothed max contribution
            max_c = np.max(conv)  # for logging maybe
            # compute p softmax of gamma * conv
            tmp = gamma * conv
            max_tmp = np.max(tmp)
            exp_vals = np.exp(tmp - max_tmp)
            p = exp_vals / np.sum(exp_vals)

            # gradient: corr = np.correlate(p, a, mode='valid')
            corr = np.correlate(p, a, mode='valid')
            grad_a = 4.0 * n * corr  # shape n

            # objective value (exact) for logging
            obj_val = 2.0 * n * max_c

            if obj_val < best_val:
                best_val = obj_val
                best_a = a.copy()
                print(f"[SmoothedAdam gamma={gamma} iter={t}] new best: {obj_val:.6f}")

            # backprop through softmax
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # Adam update
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1 - beta1 ** t)
            v_hat = v / (1 - beta2 ** t)
            lr = 0.05 / (np.sqrt(gamma) + 1e-12)  # maybe adjust lr
            x = x - lr * m_hat / (np.sqrt(v_hat) + 1e-8)
            # keep x bounded
            x = np.clip(x, -50.0, 50.0)

            if time.time() > deadline - 5.0:
                break
    # after loop, return best
    return best_a, best_val

"""
A three‑phase optimiser for the step‑function problem.

1. **Smoothed Adam** – minimise a smooth approximation of the
   objective  F(a)=2·n·max(a∗a)/(∑a)² .
   The smooth max is a log‑sum‑exp with a continuation on the
   temperature γ.  Gradients are obtained analytically (a correlation
   of the softmax weights with a) and Adam updates are performed
   on the log‑probability (softmax) parametrisation, guaranteeing
   non‑negativity and ∑a=1.

2. **Incremental local search** – a fast hill‑climber that moves a
   random amount of mass from one coordinate to another.
   The convolution update is performed analytically, i.e. in
   O(n) time per trial, so millions of trials fit easily into the 1000 s
   budget.

3. **(1+1)‑ES fine‑tuning** – a brief Gaussian mutation phase to polish
   the final result.

The routine respects a global time budget, tries a few different
sequence lengths (larger n gives a better asymptotic bound) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (pure, top‑level)
# ----------------------------------------------------------------------
def softmax(x: np.ndarray) -> np.ndarray:
    """Stable softmax – maps logits to a probability vector (sum=1)."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


def full_convolution(a: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    Returns a real array.
    """
    n = a.shape[0]
    m = 2 * n - 1
    L = 1 << int(np.ceil(np.log2(m + n - 1)))   # power of two for speed
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[:m]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a∗a) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * np.max(full_convolution(a)) / (s * s)


def correlation(p: np.ndarray, a: np.ndarray) -> np.ndarray:
    """
    Correlation r[j] = Σ_i p[i+j] * a[i]  for j = 0…n‑1.
    Implemented via FFT → O(n log n).
    """
    n = a.shape[0]
    # reverse a for convolution
    a_rev = a[::-1]
    L = 1 << int(np.ceil(np.log2(p.shape[0] + a_rev.shape[0] - 1)))
    fp = np.fft.rfft(p, L)
    fa = np.fft.rfft(a_rev, L)
    conv = np.fft.irfft(fp * fa, L)
    start = n - 1
    return conv[start:start + n]


# ----------------------------------------------------------------------
# Phase 1 – Smoothed Adam on the softmax‑parametrised vector
# ----------------------------------------------------------------------
def smoothed_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    gamma_schedule: list = [1, 2, 4, 8, 16, 32, 64],
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector seen (sum=1) and its exact objective.
    """
    # initialise logits from the probability vector
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    t = 0

    best_val = np.inf
    best_a = None

    for gamma in gamma_schedule:
        # Give each γ a modest time slice
        slice_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.15)
        while time.time() < slice_deadline:
            a = softmax(x)                         # a ≥0, Σa = 1
            conv = full_convolution(a)             # linear self‑convolution
            max_c = conv.max()
            # exact objective for logging (sum(a)=1)
            val = 2.0 * n * max_c

            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[SmoothedAdam γ={gamma} it={t}] new best: {val:.6f}")

            # ------- gradient of smooth max -------
            # p_i = softmax(γ·conv_i)
            tmp = gamma * conv
            tmp_max = np.max(tmp)
            exp_vals = np.exp(tmp - tmp_max)
            p = exp_vals / exp_vals.sum()

            # correlation(p, a) = Σ_i p_i a_{i‑j}
            corr = correlation(p, a)                 # length n
            grad_a = 4.0 * n * corr                  # ∂F/∂a

            # back‑prop through softmax
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            # Adam step
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad_x
            v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate scaled with γ (large γ → sharper landscape)
            lr = 0.05 / (np.sqrt(gamma) + 1e-12)
            x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

            # keep logits bounded to avoid overflow
            x = np.clip(x, -50.0, 50.0)

            if time.time() > deadline - 0.5:
                break

    # fallback – uniform if nothing succeeded
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 2 – Fast hill‑climbing with exact convolution updates
# ----------------------------------------------------------------------
def run_local_search(
    a0: np.ndarray,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple stochastic local search.
    Moves a random amount of mass from i→j and updates the convolution
    analytically (O(n) per trial). Accepts only improving moves,
    occasionally accepting a worsening move with a tiny probability
    for escaping shallow local minima.
    Returns the best vector found.
    """
    n = a0.shape[0]
    m = 2 * n - 1
    a = a0.copy()
    conv = full_convolution(a)
    best_a = a.copy()
    best_val = compute_objective(a)

    # step size (max fraction of the source coordinate that may be moved)
    max_frac = 0.25
    it = 0
    last_improve = time.time()

    while time.time() < deadline and it < max_iters:
        it += 1

        i = np.random.randint(n)
        if a[i] <= 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # random fraction of mass at i
        frac = np.random.rand() * max_frac
        delta = frac * a[i]
        if delta <= 0.0:
            continue

        # ----- compute updated convolution (exact) -----
        new_conv = conv.copy()
        # shift contributions of the source index i
        new_conv[i:i + n] -= 2.0 * delta * a
        # shift contributions of the destination index j
        new_conv[j:j + n] += 2.0 * delta * a

        # quadratic terms from the self‑product of the moved mass
        ii = 2 * i
        jj = 2 * j
        ij = i + j
        new_conv[ii] += delta * delta
        new_conv[jj] += delta * delta
        new_conv[ij] -= 2.0 * delta * delta

        val = 2.0 * n * new_conv.max()

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # commit the move permanently
            a[i] -= delta
            a[j] += delta
            conv = new_conv
            max_frac = max_frac * 0.9995          # gradually tighten the step size
            last_improve = time.time()
            print(f"[Local it={it}] new best: {best_val:.6f}")
        else:
            # tiny Metropolis acceptance to keep moving
            if np.random.rand() < 1e-3:
                a[i] -= delta
                a[j] += delta
                conv = new_conv

        # Periodically re‑compute the convolution from scratch to avoid drift
        if it % 5000 == 0:
            conv = full_convolution(a)

        # abandon very long stagnation
        if time.time() - last_improve > 20.0:
            max_frac = min(max_frac * 1.5, 0.5)

    return best_a, best_val


# ----------------------------------------------------------------------
# Phase 3 – (1+1) Evolution Strategy (fine‑tuning)
# ----------------------------------------------------------------------
def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """(1+1)‑ES with adaptive step size on the exact objective."""
    a = init_a.copy()
    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial
            sigma = min(sigma * 1.05, 0.2)
            last_improve = time.time()
            print(f"[ES it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val
