"""
Optimiser for the self‑convolution bound.

The routine combines three ideas:

1. **Smooth‑max Adam** – a soft‑max surrogate of the max‑convolution is
   minimised with Adam on the log‑softmax parametrisation (guarantees a≥0,
   Σa=1).  The temperature λ starts small and is increased slowly, giving a
   coarse‑to‑fine optimisation.

2. **Projected sub‑gradient refinement** – after the Adam phase we switch
   to a pure sub‑gradient method for the exact max.  The sub‑gradient is
   an average of the gradients of the active convolution entries.
   A simple Euclidean projection onto the probability simplex enforces
   the constraints.

3. **Multi‑restart + fine‑tune** – many short restarts with diverse
   random initialisations (Dirichlet, exponential, half‑sine, bimodal,
   etc.) explore the landscape.  The current best sequence is used as a
   warm‑start for a final long Adam run with a high λ.

The optimiser respects a user supplied time budget (default 1000 s) and
returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Soft‑max weights wₖ = exp(lam·convₖ) / Σ exp(lam·convⱼ)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int = None) -> np.ndarray:
    """
    Gradient of Σₖ wₖ·(a*a)[k] w.r.t. a.
    Closed form: ∇ = 2·(w ★ a), where ★ is correlation.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σ x = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection only works on 1‑D arrays")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _random_initialisation(mode: int, n: int) -> np.ndarray:
    """Generate a random non‑negative vector of length n and unit sum."""
    if mode == 0:
        a = np.random.dirichlet(np.ones(n))
    elif mode == 1:
        r = np.random.uniform(0.2, 0.9)
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
    elif mode == 2:
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 3:
        p = np.random.uniform(0.5, 2.0)
        a = (np.sin(np.pi * np.arange(1, n + 1) / (n + 1))) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 4:
        r = np.random.uniform(0.3, 0.85)
        a = r ** np.arange(n, dtype=np.float64)
        a += 0.01 * np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 5:
        x = np.linspace(-1.0, 1.0, n)
        a = 1.0 - x ** 2
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:
        # bimodal half‑sine glued together
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a


def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step function (list of heights) that minimizes
    ``evaluate_sequence``.  The routine respects the time budget and returns
    the best sequence discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    best_val = float("inf")
    best_seq = None
    restart = 0

    # ------------------------------------------------------------------
    # candidate lengths – longer gives more freedom, but FFT cost grows as ~n log n
    lengths = [2000, 2500, 3000, 3500, 4000]

    # ------------------------------------------------------------------
    while time.time() < deadline:
        n = lengths[restart % len(lengths)]

        # --------------------------------------------------------------
        # random initialisation – pick a family cyclically
        mode = restart % 7
        a0 = _random_initialisation(mode, n)

        # Occasionally perturb the current global best if it matches length
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.15:
            a0 = np.array(best_seq) * (1.0 + 0.02 * (np.random.rand(n) - 0.5))
            a0 = np.clip(a0, 0.0, None)
            a0 /= a0.sum()

        # Log‑space parameterisation (soft‑max guarantees Σa=1, a≥0)
        u = np.log(a0 + 1e-30)

        # Adam state for the smooth‑max phase
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # FFT sizes for the whole run (they stay constant)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        # ------------------------------------------------------------------
        # Phase 1 – smooth‑max Adam
        lam = 0.3                     # start temperature
        lam_factor = 1.018            # growth per iteration
        base_lr = 0.05                # initial learning‑rate in log‑space
        max_iter1 = 8000
        last_improve = time.time()

        for it in range(1, max_iter1 + 1):
            if time.time() >= deadline:
                break

            # from logits to probabilities
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # convolution and soft‑max weights
            conv = _full_convolution(a, conv_fft_size)
            w = _softmax_weights(conv, lam)

            # gradient of the smooth‑max surrogate w.r.t. a
            grad_a = _softmax_gradient(a, w, size_corr)

            # map to gradient w.r.t. logits (Jacobian of soft‑max)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9995 ** it)          # gentle decay
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # increase smoothness
            lam *= lam_factor

            # exact evaluation every few steps
            if it % 20 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] Phase1 it {it:05d}  new best = {val:.6f}")

            # shake if stagnated
            if time.time() - last_improve > 3.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # ------------------------------------------------------------------
        # Phase 2 – exact max sub‑gradient with projection onto simplex
        max_iter2 = 20000
        step0 = 0.25                     # initial step size
        last_improve2 = time.time()

        for it2 in range(1, max_iter2 + 1):
            if time.time() >= deadline:
                break

            # obtain current a from logits (logits are kept for convenience)
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, conv_fft_size)
            max_val = conv.max()

            # active set (indices attaining the maximum, tolerance for numerics)
            eps = 1e-12
            active = np.where(conv >= max_val - eps)[0]

            # uniform weights over the active set – a valid sub‑gradient
            w = np.zeros_like(conv)
            if len(active) > 0:
                w[active] = 1.0 / len(active)
            else:
                w[np.argmax(conv)] = 1.0

            grad_a = _softmax_gradient(a, w, size_corr)

            # diminishing step size
            eta = step0 / np.sqrt(it2)

            a_new = a - eta * grad_a
            a_new = _proj_simplex(a_new)

            # refresh the logit vector to reflect the projected point
            u = np.log(a_new + 1e-30)

            # periodic exact evaluation
            if it2 % 100 == 0:
                val = 2.0 * n * max_val
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve2 = time.time()
                    print(f"[Restart {restart:03d}] Phase2 it {it2:05d}  new best = {val:.6f}")

            # shake if no improvement for a while
            if time.time() - last_improve2 > 5.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                last_improve2 = time.time()

        # ------------------------------------------------------------------
        restart += 1
        # early stop if we already beat the target bound
        if best_val < 1.503:
            break

    # ----------------------------------------------------------------------
    # Final fine‑tuning on the globally best sequence (high λ, long run)
    if best_seq is not None:
        n = len(best_seq)
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))

        u = np.log(best_seq + 1e-30)
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        lam = 5.0                     # sharper surrogate
        lam_factor = 1.001
        base_lr = 0.018
        max_iter_fine = 40000
        last_improve_f = time.time()

        for it in range(1, max_iter_fine + 1):
            if time.time() >= deadline:
                break

            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            conv = _full_convolution(a, conv_fft_size)
            w = _softmax_weights(conv, lam)
            grad_a = _softmax_gradient(a, w, size_corr)

            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            lr = base_lr * (0.9999 ** it)
            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            lam *= lam_factor

            if it % 200 == 0:
                val = 2.0 * n * conv.max()
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Fine‑tune] iter {it:05d}  new best = {val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        n = 3000
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best bound = {best_val:.6f} (length {len(best_seq)})")
    # ``evaluate_sequence`` expects a plain Python list.
    return best_seq.tolist()
