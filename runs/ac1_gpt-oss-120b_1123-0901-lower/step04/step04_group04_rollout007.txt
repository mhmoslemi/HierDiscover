def proj_simplex(v, s):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

Works for s>0.

def proj_simplex(v, s):
    """Project onto the simplex {x>=0, sum x = s}."""
    n = v.shape[0]
    if s < 0:
        raise ValueError
    # shift by median?
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all zeros? set to uniform?
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

Projected sub‑gradient descent + simulated‑annealing refinement.

* Start from the best known construction (``height_sequence_1`` if
  available) and normalise the L¹‑norm to √(2 n).
* Perform several restarts of a simple projected sub‑gradient method:
    – Compute the full self‑convolution by FFT.
    – Identify the shift where the convolution attains its maximum.
    – The sub‑gradient of the max‑convolution at this shift is
      g_j = 2·a_{k*‑j} (or 0 when the index is out of range).
    – Update a ← a – η·g with a decreasing step size η and project onto
      the probability simplex of total mass √(2 n).
* After the sub‑gradient phase run a cheap simulated‑annealing /
  hill‑climbing phase that repeatedly transfers a small amount of mass
  between two positions and accepts the move if it improves the exact
  objective (or with a Metropolis probability at a decreasing
  temperature).
* Throughout the search keep the best sequence seen (according to
  ``evaluate_sequence``) and return it as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper functions (top‑level, no closures)
# ----------------------------------------------------------------------


def _conv_full(a: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Full convolution a * a using real FFT.
    Returns an array of length 2·len(a)‑1.
    """
    A = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * a.shape[0] - 1]
    return conv


def _proj_simplex(v: np.ndarray, s: float) -> np.ndarray:
    """
    Euclidean projection of vector v onto the simplex {x ≥ 0, sum x = s}.
    Implements the algorithm of Duchi et al. (2008).
    """
    if s < 0:
        raise ValueError("Simplex radius must be non‑negative")
    n = v.shape[0]
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(n) + 1  # 1‑based index
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero (should not happen for s>0)
        theta = cssv[-1] / n
    else:
        rho = ind[cond].max()
        theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(a: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (a * a)_k."""
    conv = _conv_full(a, Nfft)
    return float(conv.max())


def _subgradient(a: np.ndarray, kstar: int) -> np.ndarray:
    """
    Sub‑gradient of max_{k}(a*a)_k at the current argmax shift kstar.
    g_j = 2 * a_{kstar - j}   if 0 ≤ kstar - j < n, else 0.
    """
    n = a.shape[0]
    idx = np.arange(n)
    valid = (kstar - idx >= 0) & (kstar - idx < n)
    g = np.zeros_like(a)
    g[valid] = 2.0 * a[kstar - idx[valid]]
    return g

    start_time = time.time()

    best_score = best_val  # because denominator cancels out after normalisation
    print(f"[init] n={n}, initial objective={best_score:.6f}")

    # step‑size base (tuned experimentally)

            kstar = int(np.argmax(conv))

    return best_seq.tolist()
