"""
Propose a low‑value step‑function using annealed Adam on the soft‑max
parameterisation. The algorithm works on several problem sizes (powers of 2)
and progressively refines the distribution:

* a = softmax(x) guarantees a_i ≥ 0 and Σ a_i = 1 (the scale does not affect
  the objective).
* At each iteration we minimise a smoothed version of
        F(a) = 2·n·max_k (a*a)[k]
  i.e.   2·n·T·log∑_k exp(conv_k/T).  The temperature T is decreased
  following a deterministic‑annealing schedule, making the landscape
  gradually more “max‑like”.
* Gradients are obtained analytically:
        ∂F/∂a = 2· Σ_k w_k ∂conv_k/∂a,
  where w_k = softmax(conv_k/T).  The term ∂conv_k/∂a is the
  cross‑correlation 2·(w ★ a) and is computed by FFT.
* Adam (β1=0.9, β2=0.999) updates the logits x.  Stagnation is broken by
  small random “shakes”.
* After the annealed phase a cheap local greedy refinement (pairwise mass
  transfers) is performed.
* Results from smaller lengths are up‑sampled to initialise larger
  lengths, allowing the search to propagate a good shape to higher
  resolutions.

The function returns the best sequence found before the time budget
expires.  All numbers are non‑negative, the sum is ≥ 0.01, and the
sequence length is unrestricted (but limited by the time budget).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) equal to
        (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)]   for i=0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _softmax(z: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    m = np.max(z)
    e = np.exp(z - m)
    return e / e.sum()


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list:
    """
    Search for a non‑negative step function (probability vector) that minimises
    evaluate_sequence.  The search uses several lengths, annealed Adam, and a
    cheap local refinement.  The function returns the best sequence found
    before the time budget expires.
    """
    np.random.seed(seed)
    # ---------- time handling ----------
    safety = 0.5          # keep a small margin for clean exit
    deadline = time.time() + budget_s - safety

    # ---------- bookkeeping ----------
    best_seq = None            # numpy array, not a list yet
    best_val = float("inf")    # smallest objective seen so far

    # ---------- problem sizes ----------
    lengths = [256, 512, 1024, 2048, 4096, 8192]

    # ----- temperature schedule (high -> low) -----
    # T is the smoothing temperature; smaller T approximates the true max.
    temp_schedule = [5.0, 2.0, 1.0, 0.5, 0.2, 0.1, 0.05]

    # ----- optimizer hyper‑parameters -----
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    init_lr = 0.2
    decay_factor = 0.7           # decay after each temperature phase
    max_stagnation = 25.0        # seconds without improvement before a shake

    # ----- helper to evaluate true objective (sum of a is 1) -----
    def _objective(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec, fft_len=_next_pow2(2 * a_vec.shape[0] - 1))
        return float(2 * a_vec.shape[0] * conv.max())

    # -------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n = {n} ===")

        # FFT lengths that will be reused many times
        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)         # ≥ (2n‑1)+n‑1

        # Number of independent restarts – more for smaller n (cheaper)
        if n <= 512:
            restarts = 10
        elif n <= 1024:
            restarts = 8
        elif n <= 2048:
            restarts = 6
        elif n <= 4096:
            restarts = 4
        else:
            restarts = 3

        for restart in range(restarts):
            if time.time() > deadline:
                break

            # ----------------- initialise logits x -----------------
            # Warm‑start from a previously found sequence if available
            if best_seq is not None and best_seq.shape[0] == n:
                init_a = best_seq.copy()
                init_a /= init_a.sum()
                x = np.log(init_a + 1e-15)
                print(f"[n={n} restart={restart}] warm start from best of same length")
            elif best_seq is not None and best_seq.shape[0] < n:
                # Upsample smaller best_seq to length n (linear interpolation)
                src = best_seq
                src_n = src.shape[0]
                xs = np.linspace(0, 1, src_n)
                xt = np.linspace(0, 1, n)
                upsampled = np.interp(xt, xs, src)
                upsampled = np.maximum(upsampled, 0.0)
                if upsampled.sum() < 1e-12:
                    upsampled = np.random.dirichlet(np.ones(n))
                else:
                    upsampled /= upsampled.sum()
                x = np.log(upsampled + 1e-15)
                print(f"[n={n} restart={restart}] upsampled from length {src_n}")
            elif 'height_sequence_1' in globals():
                # Use the provided global as a fallback (crop / pad)
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] > n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]))
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand.sum()
                x = np.log(cand + 1e-15)
                print(f"[n={n} restart={restart}] start from global height_sequence_1")
            else:
                # Pure random start
                a0 = np.random.dirichlet(np.ones(n))
                x = np.log(a0 + 1e-15)
                print(f"[n={n} restart={restart}] random start")

            # ----------------- Adam state -----------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            lr = init_lr

            # Tracking for stagnation detection
            last_improve = time.time()
            local_best_val = float("inf")
            local_best_seq = None

            # -------------------------------------------------
            # Max iterations are set generously; actual stop is time‑based.
            max_iters = 30000 if n <= 1024 else 20000 if n <= 2048 else 15000
            # Phase boundaries for temperature schedule (equally sized)
            phase_len = max_iters // len(temp_schedule)

            for it in range(max_iters):
                if time.time() > deadline:
                    break

                # ------- soft‑max to obtain a -------
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()          # Σ a_i = 1

                # ------- convolution -------
                conv = _full_convolution(a, fft_len_conv)
                max_c = conv.max()

                # ------- true objective for bookkeeping -------
                val = 2.0 * n * max_c
                if val < best_val - 1e-12:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={it:05d}] NEW GLOBAL BEST {val:.6f}")

                # also keep best for this run (local)
                if val < local_best_val - 1e-12:
                    local_best_val = val
                    local_best_seq = a.copy()

                # ------- temperature for this iteration -------
                temp_idx = min(it // phase_len, len(temp_schedule) - 1)
                T = temp_schedule[temp_idx]

                # ------- compute smoothed weights w = softmax(conv / T) -------
                # Use numerically stable scaling
                scaled = conv / T
                scaled -= np.max(scaled)           # shift for stability
                w = np.exp(scaled)
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(w, 1.0 / w.size)
                else:
                    w /= w_sum

                # ------- gradient w.r.t. a -------
                grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

                # ------- back‑prop through soft‑max (gradient w.r.t logits x) -------
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # ------- Adam update -------
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** (it + 1))
                v_hat = v / (1.0 - beta2 ** (it + 1))
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # ------- learning‑rate decay at the end of each temperature phase -------
                if (it + 1) % phase_len == 0:
                    lr *= decay_factor

                # ------- stagnation handling -------
                if time.time() - last_improve > max_stagnation:
                    # small random perturbation on logits
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} restart={restart} iter={it}] shake (stagnation)")

            # ------------------ local greedy refinement ------------------
            if local_best_seq is not None:
                a_local = local_best_seq.copy()
                # Simple pairwise mass transfer moves
                for _ in range(3000):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # transfer a fraction of mass from i to j
                    move = min(a_local[i], 0.01) * (np.random.rand() * 0.5 + 0.5)
                    if move <= 0.0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= move
                    a_new[j] += move
                    conv_new = _full_convolution(a_new, fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        print(f"[local refine] NEW GLOBAL BEST {val_new:.6f}")
                    if val_new < local_best_val - 1e-12:
                        local_best_val = val_new
                        a_local = a_new.copy()

    # -----------------------------------------------------------------
    if best_seq is None:
        # Fallback: uniform distribution of moderate size
        fallback_n = 2000
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the harness
    return best_seq.tolist()
