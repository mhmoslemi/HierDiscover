def solve_lp(a, M):
    n = a.shape[0]
    # A matrix shape (2n-1) x n.
    # Build Toeplitz matrix: each row k corresponds to shift.
    # First column: a reversed? Let's construct:
    # For conv f*b: conv = np.convolve(a,b). So conv[k] = sum_i a[i] * b[k-i]
    # For each k from 0 to 2n-2, the coefficient b[j] contributes a[k-j] if 0<=k-j<n.
    # Equivalent to A[k, j] = a[k-j] if in range.
    # We can build using np.zeros.
    rows = 2*n - 1
    A = np.zeros((rows, n), dtype=float)
    for j in range(n):
        # Fill column j with a[0: rows] shifted
        # row index k = j + i where i ranges from 0..n-1 => a[i]
        # So for i in range(n): set A[j+i, j] = a[i]
        A[j:j + n, j] = a
    # Now solve
    c = -np.ones(n) # maximize sum => minimize -sum
    bounds = [(0, None)]*n
    # Use linprog with method 'highs'
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=np.full(rows, M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
    else:
        b = np.zeros_like(a)
    return b

#!/usr/bin/env python
"""Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

The search consists of three phases:
1.  A set of diverse initial vectors (uniform, Dirichlet, exponential …) is
    generated for a few problem sizes (n ≈ 1500‑3000).  Each vector is
    normalised so that its ℓ₁‑norm equals √(2 n).
2.  For each start we run the *linear‑program mixing* scheme described in the
    statement.  With a fixed sequence f we compute M = max(f∗f) and solve the
    linear program

        maximise   Σ b_i
        subject to   b_i ≥ 0 ,    f∗b ≤ M

    (the convolution constraints are linear in b).  The optimal b is then
    rescaled to the required ℓ₁‑norm and mixed with f:
        f ← (1‑t)·f + t·b .
    The mixing parameter t is gradually reduced.  The process is repeated a
    handful of times, each iteration reducing the peak of the self‑convolution.
3.  A cheap local refinement (pairwise mass transfers) and a final
    interpolation with the uniform vector are applied to the best candidate.
    The overall best sequence found before the deadline is returned.

The algorithm works with only NumPy and SciPy (no file or network access) and
respects the 1000 s wall‑clock budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def _max_convolution(a: np.ndarray) -> float:
    """Return max_{k} (a∗a)[k] using an FFT of appropriate length."""
    n = a.shape[0]
    Nfft = 1 << ((2 * n - 1).bit_length())          # next power of two ≥ 2·n‑1
    A_fft = np.fft.rfft(a, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * n - 1]
    return float(conv.max())


def _project_to_simplex(a: np.ndarray, target_sum: float) -> np.ndarray:
    """Clip negatives to zero and scale to have ℓ₁‑norm = target_sum."""
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full_like(a, target_sum / a.shape[0])
    return a


def _make_initial_vectors(n: int, target_sum: float, rng: np.random.Generator) -> list[np.ndarray]:
    """Produce a diverse pool of initial vectors for length n."""
    init = []

    # 1) uniform
    init.append(np.full(n, target_sum / n))

    # 2) Dirichlet (Gamma(1,1))
    g = rng.gamma(shape=1.0, scale=1.0, size=n)
    init.append(g / g.sum() * target_sum)

    # 3) exponential decay with random rate
    rate = rng.uniform(0.0005, 0.01)
    x = np.exp(-rate * np.arange(n))
    init.append(x / x.sum() * target_sum)

    # 4) reversed exponential
    rev = x[::-1]
    init.append(rev / rev.sum() * target_sum)

    # 5) Hann window
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * np.arange(n) / (n - 1))
    init.append(hann / hann.sum() * target_sum)

    # 6) Blackman window
    a0, a1, a2 = 0.42, 0.5, 0.08
    n_arr = np.arange(n)
    black = (a0 -
             a1 * np.cos(2.0 * np.pi * n_arr / (n - 1)) +
             a2 * np.cos(4.0 * np.pi * n_arr / (n - 1)))
    init.append(black / black.sum() * target_sum)

    # 7) small perturbation of the uniform vector
    pert = np.full(n, target_sum / n)
    pert += rng.normal(scale=0.02 * target_sum / n, size=n)
    pert = np.maximum(pert, 0.0)
    pert = pert / pert.sum() * target_sum
    init.append(pert)

    return init


# ----------------------------------------------------------------------
# Linear‑program (LP) that computes the partner sequence g

def _conv_matrix_dense(a: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix A such that A @ b = a ∗ b (linear convolution).
    Shape of A is (2n‑1) × n.
    """
    n = a.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    # column j contributes a to rows j … j+n‑1
    for j in range(n):
        A[j:j + n, j] = a
    return A


def _solve_lp(a: np.ndarray, M: float) -> np.ndarray:
    """
    Solve   maximise Σ b_i   subject to   b_i ≥ 0  and  a∗b ≤ M.
    Returns the optimal b (may be all zeros if the solver fails).
    """
    import scipy.optimize

    n = a.shape[0]
    A = _conv_matrix_dense(a)                 # (2n‑1) × n
    c = -np.ones(n, dtype=np.float64)         # maximise Σ b  → minimise -Σ b
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # ‘highs’ is robust and handles dense matrices efficiently.
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b_ub,
                                bounds=bounds, method='highs',
                                options={'presolve': True, 'time_limit': 30})
    if res.success:
        return res.x
    # fallback – return a zero vector (the caller will ignore it)
    return np.zeros_like(a)


# ----------------------------------------------------------------------
# Main optimisation loop for a single start vector

def _lp_mix_optimize(a: np.ndarray, target_sum: float,
                     deadline: float, max_outer: int = 12) -> tuple[np.ndarray, float]:
    """
    Perform the LP‑mixing scheme described in the statement.
    Returns the best sequence found in this run and its exact objective.
    """
    best_seq = a.copy()
    best_val = _max_convolution(a)
    t = 0.5                       # initial mixing weight

    for it in range(max_outer):
        if time.time() >= deadline:
            break

        M = _max_convolution(a)

        # Solve the LP → g
        g = _solve_lp(a, M)
        if g.sum() == 0.0:
            break

        # Rescale g to the required ℓ₁‑norm
        g = g * (target_sum / g.sum())

        # Mix and re‑project
        a = (1.0 - t) * a + t * g
        a = np.maximum(a, 0.0)
        a = a * (target_sum / a.sum())
        # Optional symmetry (helps stability)
        a = 0.5 * (a + a[::-1])

        cur_val = _max_convolution(a)
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"   [LP mix iter {it}] new best = {best_val:.6f}")

        t *= 0.9                    # gradually reduce the step size

    return best_seq, best_val


# ----------------------------------------------------------------------
# Light local refinement – move a tiny amount of mass between two positions

def _local_refine(a: np.ndarray, target_sum: float,
                  deadline: float, steps: int = 2000) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    rng = np.random.default_rng()
    best_seq = a.copy()
    best_val = _max_convolution(a)

    eps0 = 0.01 * target_sum / n   # starting magnitude of a transfer

    for s in range(steps):
        if time.time() >= deadline:
            break

        i, j = rng.integers(0, n, size=2)
        if i == j:
            continue
        # slowly decay the transferred mass
        eps = eps0 * (0.9 ** (s // 500))
        if a[i] < eps:
            continue

        a_candidate = a.copy()
        a_candidate[i] -= eps
        a_candidate[j] += eps

        val = _max_convolution(a_candidate)
        if val < best_val:
            best_val = val
            best_seq = a_candidate.copy()
            a = a_candidate          # continue from the improved point

    return best_seq, best_val


# ----------------------------------------------------------------------
# Interpolate a candidate with the uniform vector – often helps smooth out peaks

def _mix_with_uniform(a: np.ndarray, target_sum: float,
                      deadline: float) -> tuple[np.ndarray, float]:
    n = a.shape[0]
    uniform = np.full(n, target_sum / n)
    best_seq = a.copy()
    best_val = _max_convolution(a)

    for t in np.linspace(0.0, 1.0, 21):
        if time.time() >= deadline:
            break
        mix = (1.0 - t) * a + t * uniform
        mix = mix * (target_sum / mix.sum())
        val = _max_convolution(mix)
        if val < best_val:
            best_val = val
            best_seq = mix.copy()

    return best_seq, best_val

    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # keep a small safety margin

    # ------------------------------------------------------------------
    # Try to load the previously discovered construction, if available.
    try:

    for n in lengths:
