def propose_candidate(seed=0, budget_s=1000, **kwargs):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0  # margin

    # baseline sequence
    try:
        base_seq = np.array(height_sequence_1, dtype=float)
    except Exception:
        base_seq = np.ones(2100, dtype=float)

    best_seq = base_seq
    best_score = evaluate_sequence(best_seq.tolist())
    print("[global] initial score:", best_score)

    lengths = [800,1200,1600,2100,2600,3200,4000,5000]   # could adapt based on time
    # shuffle lengths for randomness

    for n in lengths:
        if time.time() >= deadline: break
        # produce initial sequence
        if n == base_seq.shape[0]:
            init_seq = base_seq.copy()
        else:
            # interpolate base seq to length n using linear interpolation
            x_old = np.linspace(0, 1, base_seq.shape[0])
            x_new = np.linspace(0, 1, n)
            init_seq = np.interp(x_new, x_old, base_seq)
        # ensure nonnegative
        init_seq = np.clip(init_seq, 0, None)

        # transform to y (log-softmax)
        eps_log = 1e-12
        p0 = init_seq / init_seq.sum()
        y0 = np.log(p0 + eps_log)

        # Optimization
        seq_opt, score_opt = optimise_sequence_len(y0, n, deadline, best_score, best_seq)

        # Evaluate
        if score_opt < best_score:
            best_score = score_opt
            best_seq = seq_opt.copy()
            print("[global] new best length", n, "score", best_score)

    # final sanity
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print("[final] best score:", best_score, "len:", best_seq.shape[0])

    # Convert to list
    return best_seq.tolist()

def local_search(seq, time_limit, best_score):
    n = len(seq)
    seq = seq.copy()
    while time.time() < time_limit:
        # pick random indices i,j
        i, j = np.random.randint(0, n, size=2)
        if i == j: continue
        if seq[i] == 0: continue
        delta = np.random.rand() * seq[i] * 0.1   # move up to 10% of source mass
        new_seq = seq.copy()
        new_seq[i] -= delta
        new_seq[j] += delta
        # ensure no negative due to numeric
        # evaluate
        cur = evaluate_sequence(new_seq.tolist())
        if cur < best_score:
            seq = new_seq
            best_score = cur
            best_seq_local = seq.copy()
            # optional print
    return seq, best_score

def simulated_annealing(a, Nfft, n, deadline, best_score):
    # a is numpy array
    cur = a.copy()
    cur_score = compute_score(cur, Nfft, n)
    best_seq = cur.copy()
    best = cur_score
    T0 = 1.0
    T_end = 1e-4
    start = time.time()
    while time.time() < deadline:
        # linear schedule based on elapsed time
        elapsed = time.time() - start
        T = T0 * (1 - elapsed / (deadline - start)) + T_end * (elapsed / (deadline - start))
        # propose move
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # swap
        cur[i], cur[j] = cur[j], cur[i]
        # compute new score
        new_score = compute_score(cur, Nfft, n)
        delta = new_score - cur_score
        if delta < 0 or np.random.rand() < np.exp(-delta / T):
            cur_score = new_score
            if new_score < best:
                best = new_score
                best_seq = cur.copy()
                print(f"[SA improve] n={n} new best {best:.6f}")
        else:
            # revert swap
            cur[i], cur[j] = cur[j], cur[i]
        # check timeout
        if time.time() >= deadline:
            break
    return best_seq, best

"""
step‑function optimiser
=======================

This module provides a single public entry‑point ``propose_candidate`` that
searches for a non‑negative vector (a “step function”) minimising the
``evaluate_sequence`` objective.

Algorithmic outline
-------------------
1. **Smooth‑max L‑BFGS**  
   The maximum of the self‑convolution is approximated by a smooth log‑sum‑exp
   (temperature τ).  The optimisation variables are the logits *y* of a
   probability vector `p = softmax(y)`.  A limited‑memory BFGS (SciPy if
   available, otherwise a tiny Adam implementation) minimises the smooth
   objective for a decreasing schedule of τ.  The schedule is slower than the
   reference implementation (τ‑decay = 0.85) and each inner optimisation is
   allowed many more function evaluations (``maxfun=1500``) – this gives the
   optimiser more chance to escape shallow local minima.

2. **Permutation‑based simulated annealing**  
   After the continuous stage the values of the vector are fixed but their
   ordering is still free.  Swapping two entries only changes the ordering,
   not the multiset of heights.  A fast simulated‑annealing loop (≈ 2 s per
   length) searches over permutations, accepting worsening swaps with a
   temperature that cools linearly to 1e‑4.  Because the objective depends only
   on the convolution, each swap is evaluated by a single FFT‑based convolution
   (≈ 0.5 ms for n≈2000).

3. **Multiple restarts & lengths**  
   Several initialisations are tried (the known construction, a random Dirichlet,
   a uniform vector and a perturbed copy).  The whole procedure is repeated for
   a few different lengths (the original length and a few larger ones) as long
   as the global time budget permits.  The best sequence encountered is
   returned.

All numbers are kept non‑negative, values are clipped to [0, 1000] only inside
``evaluate_sequence`` (the optimiser never exceeds those limits).  The code
uses only NumPy and (optionally) SciPy and runs comfortably within the
1000‑second budget.

import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Optional SciPy import
try:
    from scipy.optimize import minimize
    _SCIPY = True
except Exception:               # pragma: no cover
    _SCIPY = False

# ----------------------------------------------------------------------
# Utility FFT helpers ---------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())

def _conv_full(a: np.ndarray, N: int) -> np.ndarray:
    """
    Full (real) convolution of 'a' with itself using FFT.
    Returns an array of length 2·len(a)‑1.
    """
    A = np.fft.rfft(a, n=N)
    c = np.fft.irfft(A * A, n=N)
    return c[: 2 * a.shape[0] - 1]

def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """
    Stable soft‑max of the convolution values.
    Returns (weights, smooth‑max value).
    """
    if tau <= 0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, conv.max()
    m = conv.max()
    ex = np.exp((conv - m) / tau)
    s = ex.sum()
    w = ex / s
    Ltau = m + tau * math.log(s)
    return w, Ltau

def _weighted_rev(a: np.ndarray, w: np.ndarray, N: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k * a_{k‑i}
    using FFTs (indices outside the support are ignored).
    """
    A = np.fft.rfft(a, n=N)
    Wr = np.fft.rfft(w[::-1], n=N)          # reversed weight vector
    conv_rev = np.fft.irfft(A * Wr, n=N)
    # keep only the valid part of the reverse‑convolution
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    start = 2 * a.shape[0] - 2
    rev = conv_rev[start - np.arange(a.shape[0])]
    return rev

# ----------------------------------------------------------------------
# Smooth‑max objective (used by L‑BFGS) ---------------------------------
def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int) -> Tuple[float, np.ndarray]:
    """
    Returns (smooth‑max value, gradient w.r.t. logits y) for temperature τ.
    """
    # p = softmax(y)
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)          # stabilise
    p = exp_y / exp_y.sum()

    # full convolution p * p
    conv = _conv_full(p, Nfft)

    # smooth max of convolution
    w, Ltau = _softmax_weights(conv, tau)

    # gradient dL/dp = 2 * rev (see derivation in the reference code)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # back‑propagate through the soft‑max
    dot = np.dot(p, grad_p)            # Σ_i p_i * grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y

# ----------------------------------------------------------------------
# Exact objective (used for bookkeeping & final check) ------------------
def _exact_score(a: np.ndarray, Nfft: int) -> float:
    """
    Compute the exact value returned by ``evaluate_sequence`` for a vector a.
    a must be non‑negative.
    """
    n = a.shape[0]
    # convolution a * a
    conv = _conv_full(a, Nfft)
    max_conv = conv.max()
    # 2·n·max_conv / (sum(a)^2)
    s = a.sum()
    if s == 0:
        return float("inf")
    return float(2.0 * n * max_conv / (s * s))

# ----------------------------------------------------------------------
# Simple Adam optimiser (fallback when SciPy is absent) ----------------
def _adam_optimize(y0: np.ndarray,
                   tau: float,
                   Nfft: int,
                   max_iter: int = 2000,
                   lr: float = 0.1) -> np.ndarray:
    """
    Very thin Adam implementation on the smooth‑max objective.
    Returns the final logits y.
    """
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    m = np.zeros_like(y0)
    v = np.zeros_like(y0)
    y = y0.copy()
    for t in range(1, max_iter + 1):
        loss, grad = _objective_and_grad(y, tau, Nfft)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        y -= lr * m_hat / (np.sqrt(v_hat) + eps)
        # optional early stop if gradient is tiny
        if np.linalg.norm(grad) < 1e-12:
            break
    return y

# ----------------------------------------------------------------------
# Core L‑BFGS (or Adam) optimisation loop -------------------------------
def _continuous_opt(y_start: np.ndarray,
                    n: int,
                    deadline: float,
                    best_seq: np.ndarray,
                    best_score: float) -> Tuple[np.ndarray, float]:
    """
    Runs the smooth‑max optimisation (L‑BFGS or Adam) with a cooling schedule.
    Returns the best sequence (scaled to the required L1 norm) and its score.
    """
    Nfft = _next_pow2(2 * n - 1)
    target_sum = math.sqrt(2.0 * n)

    # initial temperature (a few times the current smooth max)
    p0 = np.exp(y_start - np.max(y_start))
    p0 /= p0.sum()
    conv0 = _conv_full(p0, Nfft)
    tau = max(conv0.max() * 5.0, 1e-12)      # a little larger than the reference
    tau_decay = 0.85
    min_tau = 1e-12

    y = y_start.copy()

    while time.time() < deadline and tau > min_tau:
        # ----- optimisation of the smooth problem ------------------------
        if _SCIPY:
            def fun(v):
                L, g = _objective_and_grad(v, tau, Nfft)
                return L, g
            opts = {"maxfun": 1500, "maxiter": 1500, "ftol": 1e-12, "gtol": 1e-12}
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)
            y = res.x
        else:
            # simple Adam fallback
            y = _adam_optimize(y, tau, Nfft, max_iter=1500, lr=0.2)

        # ----- evaluate exact objective ---------------------------------
        p = np.exp(y - np.max(y))
        p /= p.sum()
        a = p * target_sum
        cur_score = _exact_score(a, Nfft)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[improve] n={n:4d} τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule & jitter ----------------------------------------
        tau *= tau_decay
        if np.random.rand() < 0.2:
            y += np.random.randn(y.shape[0]) * 0.02

        # early exit if time is exhausted
        if time.time() >= deadline:
            break

    return best_seq, best_score


# ----------------------------------------------------------------------
# Simple simulated‑annealing over permutations -------------------------
def _permute_anneal(a: np.ndarray,
                    deadline: float,
                    best_score: float) -> Tuple[np.ndarray, float]:
    """
    Given a vector a (already L1‑normalised), try to improve the max‑convolution
    by swapping entries.  The method is a lightweight simulated annealing
    routine that runs until the supplied deadline.
    """
    n = a.shape[0]
    Nfft = _next_pow2(2 * n - 1)
    cur = a.copy()
    cur_score = _exact_score(cur, Nfft)

    # linear temperature schedule
    T_start, T_end = 1.0, 1e-4
    start_time = time.time()
    total_time = deadline - start_time
    if total_time <= 0:
        return cur, best_score

    best_local = cur.copy()
    best_local_score = cur_score

    it = 0
    while time.time() < deadline:
        it += 1
        # progress‑based temperature
        elapsed = time.time() - start_time
        T = T_start * (1 - elapsed / total_time) + T_end * (elapsed / total_time)

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue

        # swap and evaluate
        cur[i], cur[j] = cur[j], cur[i]
        new_score = _exact_score(cur, Nfft)
        Δ = new_score - cur_score

        if Δ < 0 or np.random.rand() < math.exp(-Δ / T):
            cur_score = new_score
            if new_score < best_local_score:
                best_local_score = new_score
                best_local = cur.copy()
                if new_score < best_score:
                    best_score = new_score
                    print(f"[SA improve] n={n:4d} iter={it} score={new_score:.6f}")
        else:
            # revert the swap
            cur[i], cur[j] = cur[j], cur[i]

        # safety break
        if it % 1000 == 0 and time.time() >= deadline:
            break

    return best_local, best_score


# ----------------------------------------------------------------------
# Main public entry point -----------------------------------------------
def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step function that minimises the
    ``evaluate_sequence`` objective.

    The routine combines a smooth‑max L‑BFGS (or Adam) phase with a
    permutation‑based simulated annealing phase.  Several random restarts and
    a few different sequence lengths are tried while respecting the overall
    time budget.  The best sequence found is returned as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0          # a tiny safety margin

    # ------------------------------------------------------------------
    # Base construction (provided by the platform) – fall back to uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    # ------------------------------------------------------------------
    # Helper to initialise logits y from a raw vector
    eps_log = 1e-12
    def _logits_from_vec(v):
        p = v / v.sum()
        return np.log(p + eps_log)

    # ------------------------------------------------------------------
    # Set of lengths we will try (original plus a few larger ones).
    lengths = [base.shape[0], 2500, 3000, 3500, 4000]
    # shuffle to avoid systematic bias if time runs out early
    np.random.shuffle(lengths)

    # Global best record
    global_best_seq = base.copy()
    global_best_score = _exact_score(global_best_seq, _next_pow2(2 * global_best_seq.shape[0] - 1))
    print(f"[init] length={global_best_seq.shape[0]}  score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Main search over lengths
    for n in lengths:
        if time.time() >= deadline:
            break

        # ----------------------------------------------------------------
        # Build an initial vector for this length.
        # If the length differs from the base, linearly interpolate:
        if n == base.shape[0]:
            init_vec = base.copy()
        else:
            x_old = np.linspace(0.0, 1.0, base.shape[0])
            x_new = np.linspace(0.0, 1.0, n)
            init_vec = np.interp(x_new, x_old, base)

        # Normalise to the required L1‑norm (any scaling works, but we keep it)
        target_sum = math.sqrt(2.0 * n)
        init_vec = init_vec * (target_sum / init_vec.sum())

        # --------------------------------------------------------------
        # Prepare several starting points for the continuous phase.
        starts = []

        # 1) the interpolated solution itself (converted to logits)
        starts.append(_logits_from_vec(init_vec))

        # 2) random Dirichlet (exponential) – same L1‑norm
        rand_dir = np.random.exponential(scale=1.0, size=n)
        rand_dir = rand_dir * (target_sum / rand_dir.sum())
        starts.append(_logits_from_vec(rand_dir))

        # 3) pure uniform
        uniform_vec = np.full(n, target_sum / n)
        starts.append(_logits_from_vec(uniform_vec))

        # 4) a slightly perturbed copy of the first start
        pert = starts[0] + np.random.randn(n) * 0.03
        starts.append(pert)

        # --------------------------------------------------------------
        # Run the optimisation from each start while time permits.
        for idx, y0 in enumerate(starts):
            if time.time() >= deadline:
                break
            print(f"[run] length={n} start={idx+1}/{len(starts)}")

            # Continuous optimisation (L‑BFGS + temperature schedule)
            seq_cont, score_cont = _continuous_opt(
                y_start=y0,
                n=n,
                deadline=deadline,
                best_seq=global_best_seq,
                best_score=global_best_score,
            )

            # Update global records
            if score_cont < global_best_score:
                global_best_score = score_cont
                global_best_seq = seq_cont.copy()
                print(f"[global] new best after continuous: len={n} score={global_best_score:.6f}")

            # ----------------------------------------------------------
            # Short simulated‑annealing phase on the ordering.
            # Give each length at most ~5 s for this phase.
            anneal_deadline = min(deadline, time.time() + 5.0)
            seq_perm, score_perm = _permute_anneal(
                a=seq_cont,
                deadline=anneal_deadline,
                best_score=global_best_score,
            )
            if score_perm < global_best_score:
                global_best_score = score_perm
                global_best_seq = seq_perm.copy()
                print(f"[global] new best after SA: len={n} score={global_best_score:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check (use the official evaluator once more)
    final_score = evaluate_sequence(global_best_seq.tolist())
    if final_score < global_best_score:
        global_best_score = final_score
        global_best_seq = global_best_seq.copy()
        print(f"[final] score improved by official call: {final_score:.6f}")

    print(f"[done] best score = {global_best_score:.6f}, length = {global_best_seq.shape[0]}")
    return global_best_seq.tolist()
