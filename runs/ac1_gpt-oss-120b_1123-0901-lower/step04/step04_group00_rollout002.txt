def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.1
    best_seq = None
    best_val = float('inf')
    # lengths ascend
    lengths = [1024, 2048, 4096, 8192, 16384]
    # Precompute FFT lens: conv_len = 2*n-1, fft_len = next_pow2(conv_len)
    for n in lengths:
        # compute fft lengths once
        conv_fft_len = next_pow2(2*n-1)
        corr_fft_len = next_pow2(3*n)
        # generate candidates:
        cand_list = []

        # warm previous best (if any) interpolated
        if best_seq is not None:
            if len(best_seq) != n:
                # stretch
                x_old = np.linspace(0, 1, len(best_seq))
                x_new = np.linspace(0, 1, n)
                cand = np.interp(x_new, x_old, best_seq)
            else:
                cand = np.array(best_seq)
            cand = np.maximum(cand, 0)
            cand = cand / cand.sum()
            cand_list.append(cand)

        # random Dirichlet with alpha=0.5 (spiky) and alpha=2.0 (smooth)
        for alpha in [0.5, 1.0, 2.0]:
            cand = np.random.dirichlet(np.full(n, alpha))
            cand_list.append(cand)

        # power law shape
        idx = np.arange(n) + 1
        prod = idx * (n - idx +1)
        for alpha in np.linspace(-2.0, -0.4, 5):
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            if shape.sum()>0:
                shape = shape/shape.sum()
                cand_list.append(shape)

        # two gaussian edges
        for gamma in [0.2, 0.5, 1.0]:
            i = np.arange(n)
            shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n-1-i)**2))
            if shape.sum()>0:
                shape = shape/shape.sum()
                cand_list.append(shape)

        # maybe mixture of random weight noise on power shape
        # for each shape create a noisy version
        for base in cand_list[:]:
            noisy = base * (1 + 0.2*np.random.randn(n))
            noisy = np.clip(noisy, 0, None)
            if noisy.sum()>0:
                noisy = noisy/noisy.sum()
                cand_list.append(noisy)

        # Iterate candidates
        for ci, a0 in enumerate(cand_list):
            if time.time()>deadline:
                break
            # ensure simplex
            a = a0/np.sum(a0)
            # Adam phase
            a = adam_optimize(a, n, max_iters=8000, lr=0.05, lam_start=0.5, lam_factor=1.02,
                              fft_len_conv=conv_fft_len, fft_len_corr=corr_fft_len,
                              deadline=deadline)
            # subgradient phase
            a = subgrad_optimize(a, n, max_iters=2000, eta=0.6,
                                 fft_len_conv=conv_fft_len, fft_len_corr=corr_fft_len,
                                 deadline=deadline)
            # hill climbing refinement
            a = hill_climb_optimize(a, n, max_iters=5000, epsilon=0.005, deadline=deadline)

            # evaluate
            val = 2.0 * n * full_conv_fft(a, conv_fft_len).max() / (np.sum(a)**2)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={ci}] new global best {val:.6f}")

    if best_seq is None:
        best_seq = np.full(2048, 1.0/2048)
    return best_seq.tolist()

def _cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n-1
    return corr[start:start+n]
```

Note: w length 2n-1, rev_a length n => convolution length 2n-1 + n -1 = 3n-2? The result in time domain length conv_len = len(w)+len(a)-1 = (2n-1)+n-1 = 3n-2. The linear cross-correlation for each shift index i (0..n-1) is sum_k w[k] * a[i+k-(n-1)] as earlier. So we slice start = n-1; length n.

def _adam_optimize(a, n, max_iters, lr, lam_start, lam_factor,
                   fft_len_conv, fft_len_corr, deadline):
    # a on simplex
    logits = np.log(a + 1e-15)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start
    for t in range(1, max_iters+1):
        if time.time()>deadline:
            break
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        # compute softmax weights using lam
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(w, 1.0/w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # backprop through softmax (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # lambda increase
        lam = min(lam * lam_factor, 1e6)
    # return final a in simplex
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a = exp_x / exp_x.sum()
    return a
```

def _subgrad_optimize(a, n, max_iters, eta,
                      fft_len_conv, fft_len_corr, deadline):
    for it in range(max_iters):
        if time.time()>deadline:
            break
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)
        # exponentiated gradient step
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
    return a

def _hill_climb_optimize(a, n, max_iters, epsilon, deadline):
    # use evaluate_sequence for exact objective
    best_a = a.copy()
    best_val = 2.0*n*_full_convolution(best_a, _next_pow2(2*n-1)).max()/(best_a.sum()**2)

    for it in range(max_iters):
        if time.time()>deadline:
            break
        # propose move: choose i, j distinct
        i, j = np.random.randint(0, n, size=2)
        if i==j:
            continue
        # delta amount
        delta = epsilon * (np.random.rand() - 0.5)

"""
Propose a non‑negative step‑function sequence that (approximately) minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

where a*a is the linear convolution of a with itself.
The algorithm is a multi‑stage stochastic search:

1.  Choose several lengths (1024 … 16384).  For each length generate a diverse
    set of initial candidates (warm‑started from the current best,
    Dirichlet draws, power‑law edge shapes, double‑Gaussian edges and noisy
    perturbations).

2.  Refine every candidate with three optimisation phases:
    * Adam on a smooth‑max (soft‑max) surrogate of the max‑convolution.
    * Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    * A cheap hill‑climbing local search that moves a tiny amount of mass
      between two coordinates.

3.  Keep the globally best sequence (across all lengths) and stop as soon as
    the time budget (≈ 1000 s) is exhausted or the target value 1.503 is reached.

All heavy operations (convolution / cross‑correlation) use FFTs for speed,
and the code checks the wall‑clock frequently to respect the budget.
The returned object is a plain Python list of non‑negative floats.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) using FFT.
    The returned array has exactly length 2·n‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute the linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a).  Implements the formula used in the reference code.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start : start + n]


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    max_iters: int,
    lr: float,
    lam_start: float,
    lam_factor: float,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
) -> np.ndarray:
    """
    Adam optimisation on the smooth‑max surrogate.
    The variable is kept on the probability simplex via a log‑softmax parametrisation.
    """
    logits = np.log(a0 + 1e-15)          # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lam = lam_start

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max → simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # back‑prop through the soft‑max (logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # temperature schedule
        lam = min(lam * lam_factor, 1e6)

    # final projection back to the simplex
    mx = logits.max()
    exp_x = np.exp(logits - mx)
    a = exp_x / exp_x.sum()
    return a


def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    max_iters: int,
    eta: float,
    fft_len_conv: int,
    fft_len_corr: int,
    deadline: float,
) -> np.ndarray:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        # uniform distribution over the active set
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, fft_len_corr)

        # exponentiated‑gradient step (keeps a ≥ 0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12:                     # avoid collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

    return a


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    max_iters: int,
    epsilon: float,
    fft_len_conv: int,
    deadline: float,
) -> np.ndarray:
    """
    Very cheap stochastic local search.
    Each iteration moves a tiny random amount of mass from one coordinate to another.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(best_a, fft_len_conv)
    best_val = 2.0 * n * conv.max() / (best_a.sum() ** 2)

    for it in range(max_iters):
        if time.time() > deadline:
            break

        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        delta = epsilon * (np.random.rand() - 0.5)  # can be positive or negative

        # stay feasible
        if best_a[i] - delta < 0:
            continue

        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta

        conv = _full_convolution(cand, fft_len_conv)
        val = 2.0 * n * conv.max() / (cand.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = cand

    return best_a


def _resize_sequence(seq: np.ndarray, n: int) -> np.ndarray:
    """
    Resize a sequence to length n by linear interpolation,
    then re‑normalise it to a probability vector.
    """
    if seq.shape[0] == n:
        return seq.copy()
    x_old = np.linspace(0.0, 1.0, seq.shape[0])
    x_new = np.linspace(0.0, 1.0, n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    s = new_seq.sum()
    if s < 1e-12:
        return np.full(n, 1.0 / n, dtype=np.float64)
    return new_seq / s


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The function respects the given time budget and returns the best
    sequence found as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2   # safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # lengths to explore (powers of two)
    lengths = [1024, 2048, 4096, 8192, 16384]

    # Pre‑compute FFT lengths for each size
    fft_len_conv_cache = {}
    fft_len_corr_cache = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv_cache[n] = _next_pow2(conv_len)
        # correlation needs length conv_len + n - 1 ≤ 3·n
        fft_len_corr_cache[n] = _next_pow2(conv_len + n - 1)

    # ------------------------------------------------------------------
    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")
        fft_len_conv = fft_len_conv_cache[n]
        fft_len_corr = fft_len_corr_cache[n]

        # --------------------- build initial candidates --------------------
        candidates = []

        # warm‑start from the global best (interpolated if needed)
        if best_seq is not None:
            candidates.append(_resize_sequence(np.asarray(best_seq, dtype=np.float64), n))

        # random Dirichlet draws with several concentration parameters
        for alpha in (0.3, 0.8, 2.0):
            cand = np.random.dirichlet(np.full(n, alpha))
            candidates.append(cand)

        # power‑law edge‑concentrated shapes: ((i+1)*(n-i))**α  with α<0
        idx = np.arange(n, dtype=np.float64) + 1.0
        prod = idx * (n - idx + 1.0)
        for alpha in np.linspace(-2.0, -0.2, 5):
            shape = np.where(prod > 0, prod ** alpha, 0.0)
            if shape.sum() > 0:
                shape /= shape.sum()
                candidates.append(shape)

        # two‑Gaussian edge shapes
        for gamma in (0.1, 0.3, 0.6, 1.2):
            i = np.arange(n, dtype=np.float64)
            shape = np.exp(-gamma * (i ** 2)) + np.exp(-gamma * ((n - 1 - i) ** 2))
            shape /= shape.sum()
            candidates.append(shape)

        # a few noisy perturbations of the above (to increase diversity)
        base_cands = list(candidates)   # snapshot
        for base in base_cands:
            for _ in range(2):
                noise = 0.2 * np.random.randn(n)
                cand = np.maximum(base + noise, 0.0)
                if cand.sum() > 1e-12:
                    cand /= cand.sum()
                    candidates.append(cand)

        # limit the number of candidates per length to keep the budget reasonable
        max_candidates = 15
        if len(candidates) > max_candidates:
            np.random.shuffle(candidates)
            candidates = candidates[:max_candidates]

        # ------------------------------------------------------------------
        # Optimise each candidate
        for cand_id, a0 in enumerate(candidates):
            if time.time() > deadline:
                break

            # ensure a0 is a proper probability vector
            a = a0 / a0.sum()

            # ---- Adam phase (smooth‑max surrogate) --------------------
            # iteration budget scales inversely with n (≈ constant total work)
            base_iters = 8000
            max_iters_adam = max(1, int(base_iters * 1024 / n))
            a = _adam_optimize(
                a0=a,
                n=n,
                max_iters=max_iters_adam,
                lr=0.05,
                lam_start=0.5,
                lam_factor=1.02,
                fft_len_conv=fft_len_conv,
                fft_len_corr=fft_len_corr,
                deadline=deadline,
            )

            # ---- Sub‑gradient phase (exact max) -----------------------
            max_iters_sub = max(1, int(2000 * 1024 / n))
            a = _subgrad_optimize(
                a0=a,
                n=n,
                max_iters=max_iters_sub,
                eta=0.6,
                fft_len_conv=fft_len_conv,
                fft_len_corr=fft_len_corr,
                deadline=deadline,
            )

            # ---- Hill‑climbing local refinement -----------------------
            max_iters_hc = max(1, int(1500 * 1024 / n))
            a = _hill_climb_optimize(
                a0=a,
                n=n,
                max_iters=max_iters_hc,
                epsilon=0.008,
                fft_len_conv=fft_len_conv,
                deadline=deadline,
            )

            # ---- Exact evaluation --------------------------------------
            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[n={n} cand={cand_id}] NEW GLOBAL BEST = {val:.6f}")

            # early exit if we already hit the target bound
            if best_val <= 1.503:
                print("Target bound reached – stopping early.")
                return best_seq.tolist()

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing was found (should never happen)
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)

    return best_seq.tolist()
