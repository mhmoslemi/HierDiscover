import time, numpy as np

def full_convolution(a, fft_len=None):
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2*n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[:2*n-1]
    return conv

def cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start+n]

def symmetrize_and_normalize(a):
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a

def compute_objective(a):
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

def propose_candidate(seed=0, budget_s=1000):

    deadline = time.time() + budget_s - 0.5   # safety

    best_seq = None
    best_val = float('inf')

    # Precompute fft lengths for lengths up to max possible chosen
    lengths = [1024, 2048, 4096, 8192, 16384]
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2*n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # Helper to possibly update global best
    def try_update(a):
        nonlocal best_seq, best_val
        val = compute_objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"New global best {val:.6f} for n={a.shape[0]}")
    
    # Phase 1: quick search on small lengths to get a reasonable shape
    # We'll only use n=2048 for initial seed
    n_small = 2048
    if time.time() < deadline:
        print(f"=== Initial optimisation on n={n_small} ===")
        # generate few seeds
        seeds = []
        # warm start from global best if has same length
        if best_seq is not None and best_seq.shape[0] == n_small:
            seeds.append(best_seq.copy())
        # historic variable if exists
        hseq = globals().get("height_sequence_1")
        if hseq is not None:
            hseq_arr = np.asarray(hseq, dtype=np.float64)
            if hseq_arr.shape[0] > n_small:
                hseq_arr = hseq_arr[:n_small]
            elif hseq_arr.shape[0] < n_small:
                hseq_arr = np.pad(hseq_arr, (0, n_small - hseq_arr.shape[0]))
            hseq_arr = np.maximum(hseq_arr, 0.0)
            hseq_arr = hseq_arr / max(hseq_arr.sum(), 1e-12)
            seeds.append(hseq_arr)
        seeds.append(np.random.dirichlet(np.ones(n_small)))
        # power law
        idx = np.arange(n_small, dtype=np.float64) + 1.0
        prod = idx * (n_small - idx + 1.0)
        for alpha in np.linspace(-2.5, -0.2, 7):
            shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                seeds.append(shape)
        # two Gaussian
        i = np.arange(n_small, dtype=np.float64)
        for gamma in [0.3, 0.7, 1.5, 3.0]:
            shape = np.exp(-gamma * (i**2)) + np.exp(-gamma * ((n_small-1-i)**2))
            shape = shape / shape.sum()
            seeds.append(shape)

        # run optimization on each seed
        for sid, a0 in enumerate(seeds):
            if time.time() > deadline:

            # Adam-like smooth-max with exponentiated updates

            lam_factor = 1.03
            eta = 0.15  # step size

                else:

                grad = 2.0 * cross_correlation(w, a, fft_len_corr[n_small])
                a = a * np.exp(-eta_sub * grad)
                a = symmetrize_and_normalize(a)
                val = compute_objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    sub_last = time.time()
                    print(f"[n={n_small} seed={sid} sub={sub_it}] new best {val:.6f}")
                if time.time() - sub_last > 3.0:
                    break

    # Phase 2: Upscale and fine‑tune on larger lengths
    # Use the best sequence obtained so far (if any) as bootstrap
    for n in [8192, 16384]:
        if time.time() > deadline:
            break
        print(f"=== Upscaling to length n={n} ===")
        # interpolate best_seq to new length
        if best_seq is None:
            # fallback random
            a0 = np.random.dirichlet(np.ones(n))
        else:
            # linear interpolation of probability density
            old_n = best_seq.shape[0]
            x_old = np.linspace(0, 1, old_n)
            x_new = np.linspace(0, 1, n)
            a0 = np.interp(x_new, x_old, best_seq)
        a = symmetrize_and_normalize(a0)
        try_update(a)
        # run a compact optimization loop

        lam_factor = 1.04

"""
Step‑function optimiser

The optimiser works in two stages.

1.  A modest search on a medium size (n≈2000) builds a decent
    “edge‑concentrated’’ shape.  The optimisation is performed by
    exponentiated‑gradient updates on a *smooth‑max* surrogate of the
    maximal auto‑convolution.  The temperature λ is increased
    geometrically; after the surrogate phase a few exact sub‑gradient
    steps are executed.

2.  The obtained shape is *up‑sampled* to larger lengths (8192 and
    16384) and refined again with the same two‑phase scheme.  The
    sequence is always projected onto the probability simplex and
    symmetrised (a[i]=a[n‑1‑i]), which is known to be optimal for the
    objective

        F(a)=2·n·max(a∗a)/(∑a)² .

The code respects a global time budget (default 1000 s), prints progress
and finally returns the best sequence found as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n‑1)] for i=0…n‑1.
    Implemented with FFT; the output length is len(a).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def symmetrize_and_normalise(a: np.ndarray) -> np.ndarray:
    """Enforce a[i]=a[n‑1‑i] and ∑a=1 (probability simplex)."""
    a = (a + a[::-1]) * 0.5
    s = a.sum()
    if s <= 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a = a / s
    return a


def objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a)/(∑a)²."""
    n = a.shape[0]
    conv = full_convolution(a)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation function defined in the challenge statement.
    The function returns the best sequence discovered before the time
    budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for the lengths we intend to examine.
    lengths = [1024, 2048, 4096, 8192, 16384]
    fft_len_conv = {}
    fft_len_corr = {}
    for n in lengths:
        conv_len = 2 * n - 1
        fft_len_conv[n] = 1 << int(np.ceil(np.log2(conv_len)))
        # cross‑correlation needs a length at most conv_len + n - 1 ≤ 3·n
        fft_len_corr[n] = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and possibly update the global best.
    def consider(a: np.ndarray):
        nonlocal best_seq, best_val
        val = objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[global] new best {val:.6f} (n={a.shape[0]})")

    # ------------------------------------------------------------------
    # Phase 1 – medium size optimisation (n≈2048) to obtain a good shape.
    n_small = 2048
    if time.time() < deadline:
        print(f"=== Phase 1 – optimisation on n={n_small} ===")
        # Build an initial candidate pool.
        candidates = []

        # warm start from a globally best sequence of the same length
        if best_seq is not None and best_seq.shape[0] == n_small:
            candidates.append(best_seq.copy())

        # historic solution, if present
        hseq = globals().get("height_sequence_1")
        if hseq is not None:

            h = np.maximum(h, 0.0)
            h = h / max(h.sum(), 1e-12)
            candidates.append(h)

        # random Dirichlet start
        candidates.append(np.random.dirichlet(np.ones(n_small)))

        # edge‑concentrated power‑law family
        idx = np.arange(n_small, dtype=np.float64) + 1.0

            lam = 0.5

                    best_seq = a.copy()

                # stagnation handling – small random shake

                else:
