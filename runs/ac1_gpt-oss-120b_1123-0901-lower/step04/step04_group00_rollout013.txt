def cross_corr(w,a,fft_len):

   n=len(a)

   rev_a = a[::-1]

   fw = np.fft.rfft(w,fft_len)

   fr = np.fft.rfft(rev_a,fft_len)

   corr = np.fft.irfft(fw*fr,fft_len)

   start = n-1

   return corr[start:start+n]

   v = np.zeros_like(logits)

   lr = 0.05
   lam = 0.5
   lam_factor = 1.02
   best_local_a = a.copy()
   best_local_val = compute_obj(a, n, fft_len_conv)
   last_improve = time.time()
   max_iters = 8000   # maybe adapt based on n

   for t in range(1, max_iters+1):
       if time.time() > deadline:
           break

       # compute a via softmax
       mx = logits.max()
       exp_x = np.exp(logits - mx)
       a = exp_x / exp_x.sum()

       # convolution
       conv = conv_fft(a, fft_len_conv)
       max_c = conv.max()
       # smoothing weights
       # We'll use lam large but not infinite
       w = np.exp(lam * (conv - max_c))
       w_sum = w.sum()
       if w_sum == 0:
           w = np.full_like(w, 1.0 / w.size)
       else:
           w = w / w_sum
       
       # gradient
       grad_a = 2.0 * corr_fft(w, a, fft_len_corr)

       # backprop through softmax:
       avg = np.dot(a, grad_a)
       grad_log = a * (grad_a - avg)

       # update Adam
       m = beta1 * m + (1 - beta1) * grad_log
       v = beta2 * v + (1 - beta2) * (grad_log ** 2)
       m_hat = m / (1 - beta1 ** t)
       v_hat = v / (1 - beta2 ** t)
       logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

       # schedule lambda
       lam = min(lam * lam_factor, 1e6)

       # evaluate objective
       val = compute_obj(a, n, fft_len_conv)

       if val < best_local_val:
           best_local_val = val
           best_local_a = a.copy()
           last_improve = time.time()
           # optionally print

       # stagnation check
       if time.time() - last_improve > 5.0:
           # small shake
           logits += (np.random.rand(n) - 0.5) * 1e-2
           last_improve = time.time()
           # continue

def shape_powerlaw(n, alpha):
    idx = np.arange(n, dtype=np.float64) + 1.0  # from 1 to n
    prod = idx * (n - idx + 1.0)
    # raise to alpha:
    shape = np.power(prod, alpha, where=prod>0, out=np.zeros_like(prod))
    if shape.sum() == 0:
        return np.full(n, 1.0/n)
    return shape / shape.sum()

def shape_exponential_edge(n, beta):
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    # shape symmetrical edges
    shape /= shape.sum()
    return shape

def shape_beta(n, p, q):
    # x values at i+0.5 positions:
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # Beta pdf (unnormalized)
    shape = x**(p - 1) * (1 - x)**(q - 1)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    # ensure non-negative
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0/n)
    shape /= shape.sum()
    return shape

def shape_combined(n, alpha, beta):
    # product of powerlaw and exponential edges
    a = shape_powerlaw(n, alpha)
    b = shape_exponential_edge(n, beta)
    shape = a * b
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0/n)
    shape /= shape.sum()
    return shape

"""
Propose a step‑function (a non‑negative vector that sums to one) that
minimises

        F(a) = 2·n·max(a*a) / (sum(a)²)

under a_i ≥ 0.

Algorithm – high‑level
----------------------
1. Choose a few lengths (4096, 8192, 16384).  For each length:
   * Scan a few *parameterised families* (edge‑exponential, power‑law,
     Beta‑distribution, and a product of the two).  Because the families
     are very low‑dimensional we can afford a dense grid and obtain a
     good starting shape quickly.
   * Optionally add a random Dirichlet sample.
   * Refine the currently best shape with two optimisation phases:
        – Adam on the smooth‑max surrogate (soft‑max of the convolution).
        – Exponentiated‑gradient (sub‑gradient) on the exact max.
   * Keep the best sequence found over all lengths.

2. The whole routine is time‑bounded (default 1000 s).  The search stops
   a little before the deadline, prints progress, and returns the best
   sequence as a plain Python list.

All heavy‑weight helper functions are defined at module level (no closures)
and use only NumPy and the standard library.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT based convolution / cross‑correlation (linear, not circular)
# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _corr_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a):
        (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)]  for i = 0…n‑1.
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence for non‑negative inputs)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len_conv: int) -> float:
    """Return 2·n·max(conv(a,a)) / (sum(a)²).  a is assumed non‑negative."""
    conv = _conv_fft(a, fft_len_conv)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Parametric families of edge‑concentrated shapes
# ----------------------------------------------------------------------
def _shape_powerlaw(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ ((i+1)*(n-i))^α   (α < 0 concentrates mass on the edges)."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    prod = idx * (n - idx + 1.0)
    shape = np.power(prod, alpha, where=prod > 0, out=np.zeros_like(prod))
    if shape.sum() == 0:
        shape[:] = 1.0 / n
    else:
        shape /= shape.sum()
    return shape


def _shape_exponential_edge(n: int, beta: float) -> np.ndarray:
    """a_i ∝ exp(-β·i) + exp(-β·(n‑1‑i)).  Symmetric exponential edges."""
    i = np.arange(n, dtype=np.float64)
    shape = np.exp(-beta * i) + np.exp(-beta * (n - 1 - i))
    shape /= shape.sum()
    return shape


def _shape_beta(n: int, p: float, q: float) -> np.ndarray:
    """
    Discrete Beta‑distribution (symmetric when p == q).
    Uses points i+0.5 over the interval (0,1).
    """
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    # avoid 0⁰ or 1⁰ issues – NumPy handles them as 0/inf correctly
    shape = np.power(x, p - 1.0, where=(x > 0), out=np.zeros_like(x))
    shape *= np.power(1.0 - x, q - 1.0, where=(x < 1), out=np.ones_like(x))
    shape = np.where(np.isfinite(shape), shape, 0.0)
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape[:] = 1.0 / n
    else:
        shape /= shape.sum()
    return shape


def _shape_combined(n: int, alpha: float, beta: float) -> np.ndarray:
    """Product of the power‑law and exponential‑edge families."""
    a = _shape_powerlaw(n, alpha)
    b = _shape_exponential_edge(n, beta)
    shape = a * b
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape[:] = 1.0 / n
    else:
        shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Small grid search inside a given length
# ----------------------------------------------------------------------
def _grid_search(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Scan several low‑dimensional families and keep the best shape.
    Returns (best_sequence, best_value).  All sequences are already
    normalised (sum = 1) and contain non‑negative entries.
    """
    conv_len = 2 * n - 1
    fft_len_conv = 1 << int(np.ceil(np.log2(conv_len)))

    best_seq = None
    best_val = float("inf")

    # 1) power‑law family
    for alpha in np.linspace(-4.5, -0.1, 44):
        if time.time() > deadline:
            return best_seq, best_val
        a = _shape_powerlaw(n, alpha)
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # 2) exponential‑edge family
    for beta in np.logspace(-2, 1.0, 30):        # 0.01 … 10
        if time.time() > deadline:
            return best_seq, best_val
        a = _shape_exponential_edge(n, beta)
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # 3) symmetric Beta‑distribution (p == q)
    for p in np.linspace(0.05, 0.5, 20):
        if time.time() > deadline:
            return best_seq, best_val
        a = _shape_beta(n, p, p)
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val, best_seq = val, a.copy()

    # 4) product of power‑law and exponential
    alphas = np.linspace(-3.0, -0.5, 13)
    betas = np.logspace(-2, 0.5, 15)          # 0.01 … ≈ 3
    for alpha in alphas:
        for beta in betas:
            if time.time() > deadline:
                return best_seq, best_val
            a = _shape_combined(n, alpha, beta)
            val = _objective(a, n, fft_len_conv)
            if val < best_val:
                best_val, best_seq = val, a.copy()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_refine(a0: np.ndarray,
                 n: int,
                 fft_len_conv: int,
                 fft_len_corr: int,
                 deadline: float) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns (best_sequence, best_value) found during the run.
    """
    a = a0.copy()
    logits = np.log(a + 1e-15)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5
    lam_factor = 1.02

    best_seq = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()

    # iteration budget loosely proportional to n (more steps for larger n)
    max_iters = int(12000 * 1024 / n) + 3000

    for t in range(1, max_iters + 1):
        if time.time() > deadline:
            break

        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_x = np.exp(logits - mx)
        a = exp_x / exp_x.sum()

        conv = _conv_fft(a, fft_len_conv)
        max_c = conv.max()

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _corr_fft(w, a, fft_len_corr)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        logits -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # increase temperature -> sharper approximation of the max
        lam = min(lam * lam_factor, 1e6)

        # evaluate exact objective
        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best {val:.6f}")

        # stagnation handling – small random shake after a few quiet seconds
        if time.time() - last_improve > 4.0:
            logits += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()

    return best_seq, best_val


# ----------------------------------------------------------------------
# Sub‑gradient refinement on the *exact* max (exponentiated‑gradient)
# ----------------------------------------------------------------------
def _subgrad_refine(a0: np.ndarray,
                    n: int,
                    fft_len_conv: int,
                    fft_len_corr: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient updates using the true max‑convolution.
    Returns the best sequence found and its value.
    """
    a = a0.copy()
    best_seq = a.copy()
    best_val = _objective(a, n, fft_len_conv)
    last_improve = time.time()
    eta = 0.6                     # fixed step size for the EG update

    for it in range(2000):
        if time.time() > deadline:
            break

        conv = _conv_fft(a, fft_len_conv)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)

        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _corr_fft(w, a, fft_len_corr)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12:                     # rescue from collapse
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective(a, n, fft_len_conv)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            last_improve = time.time()
            print(f"[Subgrad n={n} it={it:04d}] new best {val:.6f}")

        if time.time() - last_improve > 4.0:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Main entry point required by the competition platform
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative “height” sequence that minimises
    the evaluation function.  The routine works under a hard time
    budget (default 1000 s) and returns the best sequence found as a
    plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.2      # tiny safety margin

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # If a historic solution exists, start from it
    # ------------------------------------------------------------------
    if "height_sequence_1" in globals():
        cand = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if cand.ndim != 1:
            cand = cand.ravel()
        s = cand.sum()
        if s > 0.0:
            cand = cand / s
        else:
            cand = np.random.dirichlet(np.ones(cand.shape[0]))
        n0 = cand.shape[0]
        fft_len_conv0 = 1 << int(np.ceil(np.log2(2 * n0 - 1)))
        val0 = _objective(cand, n0, fft_len_conv0)
        best_seq, best_val = cand.copy(), val0
        print(f"Warm start from height_sequence_1 (n={n0}) → value {val0:.6f}")

    # ------------------------------------------------------------------
    # Lengths we will explore – 4096 gives a good trade‑off between
    # quality and speed; larger lengths are tried if time permits.
    # ------------------------------------------------------------------
    lengths = [4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break

        print(f"\n=== Optimising length n={n} ===")

        conv_len = 2 * n - 1
        fft_len_conv = 1 << int(np.ceil(np.log2(conv_len)))
        fft_len_corr = 1 << int(np.ceil(np.log2(conv_len + n - 1)))

        # --------------------------------------------------------------
        # 1) Grid search over low‑dimensional families
        # --------------------------------------------------------------
        seq_ps, val_ps = _grid_search(n, deadline)
        if seq_ps is not None and val_ps < best_val:
            best_seq, best_val = seq_ps.copy(), val_ps
            print(f"[grid search] new best for n={n} → {val_ps:.6f}")

        if time.time() > deadline:
            break

        # --------------------------------------------------------------
        # 2) A fresh random Dirichlet sample – cheap, sometimes jumps
        # --------------------------------------------------------------
        rand_seq = np.random.dirichlet(np.ones(n))
        val_rand = _objective(rand_seq, n, fft_len_conv)
        if val_rand < best_val:
            best_seq, best_val = rand_seq.copy(), val_rand
            print(f"[random init] new best for n={n} → {val_rand:.6f}")

        # --------------------------------------------------------------
        # 3) Adam refinement (smooth‑max)
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            seq_adam, val_adam = _adam_refine(best_seq, n,
                                              fft_len_conv, fft_len_corr,
                                              deadline)
            if val_adam < best_val:
                best_seq, best_val = seq_adam.copy(), val_adam
                print(f"[Adam refine] new best for n={n} → {val_adam:.6f}")

        # --------------------------------------------------------------
        # 4) Sub‑gradient (exact max) refinement
        # --------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n:
            seq_sub, val_sub = _subgrad_refine(best_seq, n,
                                               fft_len_conv, fft_len_corr,
                                               deadline)
            if val_sub < best_val:
                best_seq, best_val = seq_sub.copy(), val_sub
                print(f"[Subgrad refine] new best for n={n} → {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 2048
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = _objective(best_seq, fallback_n,
                              1 << int(np.ceil(np.log2(2 * fallback_n - 1))))
        print("Fallback to uniform sequence.")

    print(f"\nBest value found: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
