def run_smooth_adam_opt(n, init_a, deadline, beta_start=1.0, beta_max=50.0, max_iters=200000, lr=0.05):
    # Convert init_a to probability (sum=1)
    a0 = np.asarray(init_a, dtype=np.float64)
    if a0.sum() <= 0:
        a0 = np.full(n, 1.0/n)
    else:
        a0 = a0 / a0.sum()
    x = np.log(a0 + 1e-15)
    m = np.zeros(n); v = np.zeros(n)
    beta1=0.9; beta2=0.999; eps=1e-8
    best_val=np.inf; best_a=None
    # schedule for beta increase
    beta = beta_start
    beta_inc = (beta_max - beta_start) / max_iters if max_iters>0 else 0
    last_improve = time.time()
    t=0
    while time.time()<deadline and t<max_iters:
        a = softmax(x)
        # compute smooth gradient and max conv
        grad_a, max_b = smooth_grad_and_max(a, beta)
        val = 2.0 * n * max_b
        if val < best_val:
            best_val = val; best_a = a.copy(); last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t}] new best: {val:.6f}")
        # backprop through softmax (same as before)
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)
        t+=1
        # Adam update
        m = beta1 * m + (1-beta1) * grad_x
        v = beta2 * v + (1-beta2) * (grad_x*grad_x)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        x = x - lr * m_hat/(np.sqrt(v_hat)+eps)
        # clip logits to moderate range
        x = np.clip(x, -50.0, 50.0)
        # schedule beta
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)
        # occasional shake
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n)-0.5)*1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} iter={t}] shake")
    if best_a is None:
        best_a = np.full(n, 1.0/n)
        best_val = compute_objective(best_a)
    return best_a, best_val

def smooth_grad_and_max(a: np.ndarray, beta: float):
    # a sum to 1
    n = a.shape[0]
    # compute convolution a*a
    b = conv_fft(a, a)  # length 2n-1
    max_b = b.max()
    # compute weights softmax (with temperature 1/beta)
    # if beta large, use max for numeric stability
    # compute w = exp(beta*(b - max_b)); then normalize
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / exp_term.sum()
    # compute gradient = 2 * conv(w_rev, a) slice
    w_rev = w[::-1]
    conv_full = conv_fft(a, w_rev)  # length 3n-2
    # take first n elements and reverse
    grad = 2.0 * conv_full[:n][::-1]
    return grad, max_b

def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    # compute full linear convolution using FFT (real)
    n = a.shape[0] + b.shape[0] - 1
    L = 1 << int(np.ceil(np.log2(n)))
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    C = A * B
    conv = np.fft.irfft(C, L)[:n]
    return conv.real

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The implementation builds on three ideas that proved useful in earlier
experiments:

1. **Smooth‑max Adam** – the non‑smooth max in the objective is replaced
   by a temperature‑controlled soft‑max (log‑sum‑exp).  The gradient of the
   smooth objective can be written analytically as a convolution, which
   is evaluated with an FFT.  A schedule on the temperature β gradually
   turns the smooth objective into the true max, giving a stable trajectory

2. **Hand‑crafted seeds** – a small gallery of sensible shapes

def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax that maps logits to a probability vector."""
    m = np.max(x)
    e = np.exp(x - m)
    return e / e.sum()


def next_pow2(x: int) -> int:
    """Small helper returning the next power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def conv_fft(a: np.ndarray, b: np.ndarray) -> np.ndarray:
    """
    Linear convolution a*b using a real FFT.
    Returns a real 1‑D array of length len(a)+len(b)-1.
    """
    n = a.size + b.size - 1
    L = next_pow2(n)
    A = np.fft.rfft(a, L)
    B = np.fft.rfft(b, L)
    C = A * B
    conv = np.fft.irfft(C, L)[:n]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by evaluate_sequence for a probability vector.
    The sum of a is assumed to be 1, therefore
        F(a) = 2·n·max(a∗a)
    """
    n = a.size
    conv = conv_fft(a, a)
    return 2.0 * n * conv.max()


def smooth_grad_and_max(a: np.ndarray, beta: float):
    """
    Returns (gradient, max_conv) of the smooth approximation

        f_β(a) = (1/β)·log Σ_k exp(β·(a∗a)_k)

    The gradient w.r.t. a is 2·conv(w_rev, a) where
    w_k = exp(β·b_k)/Σ_j exp(β·b_j)  (soft‑max weights) and
    b = a∗a.  The implementation uses FFTs and returns only the first
    n entries needed for the simplex gradient.
    """
    n = a.size
    # 1. convolution a*a
    b = conv_fft(a, a)                     # length 2n‑1
    max_b = b.max()

    # 2. soft‑max weights (numerically stable)
    # subtract max_b so the largest exponent is 0 → avoid overflow
    exp_term = np.exp(beta * (b - max_b))
    w = exp_term / exp_term.sum()          # still length 2n‑1, sum(w)=1

    # 3. gradient = 2·conv(w_rev, a) ; keep only first n entries reversed
    w_rev = w[::-1]
    full = conv_fft(a, w_rev)              # length 3n‑2
    grad = 2.0 * full[:n][::-1]            # g_j = 2·Σ_t a_t·w_{j+t}
    return grad, max_b

# Optimisation building blocks
# ----------------------------------------------------------------------


def run_smooth_adam_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    beta_start: float = 1.0,
    beta_max: float = 50.0,
    max_iters: int = 120_000,
    lr: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate.
    The internal variable is a log‑probability vector `x` (softmax(x) = a).
    Returns the best (probability) vector and its true objective
    (i.e. 2·n·max(a∗a)).
    """
    a0 = np.asarray(init_a, dtype=np.float64)
    if a0.sum() <= 0.0:
        a0 = np.full(n, 1.0 / n)
    else:
        a0 = a0 / a0.sum()

    x = np.log(a0 + 1e-15)                 # logits
    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()
    beta = beta_start
    beta_inc = (beta_max - beta_start) / max_iters if max_iters > 0 else 0.0

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥ 0, Σa = 1

        # smooth gradient & true max value
        grad_a, max_b = smooth_grad_and_max(a, beta)

        # true objective (no smoothness here)
        val = 2.0 * n * max_b

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[SmoothAdam n={n:04d} it={t:06d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update on logits
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded (prevents overflow in the softmax)
        x = np.clip(x, -50.0, 50.0)

        # temperature schedule
        if beta < beta_max:
            beta = min(beta_max, beta + beta_inc)

        # small random shake when we have not improved for a while
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[SmoothAdam n={n} it={t}] shake (stagnation)")

    if best_a is None:                     # safety net
        best_a = np.full(n, 1.0 / n)
        best_val = compute_objective(best_a)

    return best_a, best_val


def run_es_opt(
    n: int,
    init_a: np.ndarray,
    deadline: float,
    max_iter: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    (1+1) Evolution Strategy – Gaussian mutations on the simplex with
    a simple 1/5‑success‑rule for step‑size adaptation.
    Returns the best vector found and its true objective.
    """
    a = np.asarray(init_a, dtype=np.float64)
    if a.sum() <= 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    val = compute_objective(a)
    best_a, best_val = a.copy(), val
    sigma = 0.02
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                       # success → keep
            sigma = min(sigma * 1.05, 0.2)   # increase step size a little
            last_improve = time.time()
            print(f"[ES n={n:04d} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                   # decay on failure

        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators (hand‑crafted shapes)
# ----------------------------------------------------------------------


def generate_initial_candidates(n: int) -> list[np.ndarray]:
    """Return a modest list of diverse feasible vectors for length `n`."""
    cand = []

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # linear decreasing (a_i ∝ n‑i)
    dec = np.arange(n, 0, -1, dtype=np.float64)
    cand.append(dec / dec.sum())

    # quadratic decreasing (a_i ∝ (n‑i)²)
    quad = (np.arange(n, 0, -1, dtype=np.float64)) ** 2
    cand.append(quad / quad.sum())

    # exponential families (several decay rates)
    for base in [0.99, 0.97, 0.95, 0.93]:
        ex = np.exp(np.log(base) * np.arange(n, dtype=np.float64))
        cand.append(ex / ex.sum())

    # raised‑cosine / sinusoidal shape
    x = np.arange(n, dtype=np.float64) + 0.5
    sin = np.sin(np.pi * x / n)  # positive on (0,n)
    for p in [1.0, 1.5, 2.0]:
        s = sin ** p
        cand.append(s / s.sum())

    return cand


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function sequence that minimises the
    provided evaluation function.  The routine respects the supplied
    time budget (default 1000 s) and returns the best sequence found as
    a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # a tiny safety margin

    best_seq = None
    best_val = np.inf

    # Try a handful of lengths – smaller dimensions are cheaper to optimise.
    # The constant 2·n·max(a∗a) is (approximately) independent of n,
    # therefore any size that yields a low value is acceptable.
    lengths = [1024, 512, 256, 128]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 1) build a pool of sensible initial vectors
        # ------------------------------------------------------------------
        init_pool = generate_initial_candidates(n)

        # warm‑start from the globally best sequence if it matches the length
        if best_seq is not None and len(best_seq) == n:
            init_pool.append(np.asarray(best_seq, dtype=np.float64))

        # optional historic variable supplied by the platform
        if "height_sequence_1" in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            s = hs.sum()
            if s > 0.0:
                hs = hs / s
                init_pool.append(hs)

        # ------------------------------------------------------------------
        # 2) Smooth‑max Adam (multiple restarts)
        # ------------------------------------------------------------------
        restarts = 6
        for r in range(restarts):
            if time.time() > deadline:
                break
            if r < len(init_pool):
                a0 = init_pool[r]
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a slice of the remaining budget for this run
            slice_time = min((deadline - time.time()) * 0.2, 30.0)
            sub_deadline = time.time() + slice_time

            a_opt, val_opt = run_smooth_adam_opt(
                n, a0, sub_deadline,
                beta_start=1.0,
                beta_max=45.0,
                max_iters=80_000,
                lr=0.04,
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # ------------------------------------------------------------------
        # 3) Local ES fine‑tuning (if time permits)
        # ------------------------------------------------------------------
        if best_seq is not None and len(best_seq) == n and time.time() < deadline:
            es_slice = min((deadline - time.time()) * 0.3, 60.0)
            sub_deadline = time.time() + es_slice
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline, max_iter=40_000)
            if val_es < best_val:
                best_val, best_seq = val_es, a_es.copy()
                print(f"[n={n}] ES improved global best to {best_val:.6f}")

    # ----------------------------------------------------------------------
    # Fallback – in the unlikely event nothing was found.
    # ----------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the outer evaluator
    return best_seq.tolist()
