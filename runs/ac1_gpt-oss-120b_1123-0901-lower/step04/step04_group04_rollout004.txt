def lp_update(f, target_sum, t=0.5):
    n = len(f)
    M = np.max(np.convolve(f,f))
    # Build constraint matrix A
    A = sparse.lil_matrix((2*n-1, n))
    for i in range(n):
        A[i:i+n, i] = f  # assign column
    A = A.tocsr()
    # Solve LP: maximize sum(g) s.t. A @ g <= M
    c = -np.ones(n)  # minimize -sum
    res = linprog(c, A_ub=A, b_ub=M*np.ones(2*n-1), bounds=(0, None), method='highs')
    if not res.success:
        return f  # keep f unchanged
    g0 = res.x
    sum_g0 = np.sum(g0)
    if sum_g0 == 0:
        return f
    # Scale g0 to target_sum
    g = g0 * (target_sum / sum_g0)
    # Update f
    f_new = (1 - t) * f + t * g
    # optionally renormalize due to floating error
    f_new = np.clip(f_new, 0, None)
    # Normalize to sum target_sum
    sum_f_new = np.sum(f_new)
    if sum_f_new > 0:
        f_new = f_new * (target_sum / sum_f_new)
    return f_new

"""
Propose a step‑function (non‑negative vector) that minimises
``evaluate_sequence``.

Algorithm
---------
* Load the previous best construction (global ``height_sequence_1``) and
  normalise it so that the L1‑norm equals √(2·n) – this makes the
  objective exactly the maximal entry of the self‑convolution.
* **Frank‑Wolfe / linear‑programming phase**  
  Repeatedly solve a linear programme  
  ``max Σ g_j   s.t.   g ≥ 0  and   f * g ≤ M``  
  where *f* is the current sequence, *M* = max(f*f) and “*f*g*’’ denotes
  the ordinary discrete convolution.  The LP (≈2100 variables,
  4200 constraints) is solved with SciPy's high‑performance ``linprog``.
  The solution is scaled to the required L1‑norm and mixed with *f*
  (convex combination).  If the mixture improves the exact objective,
  it is kept; otherwise a tiny random jitter is added to escape a
  plateau.
* **L‑BFGS fine‑tuning** – after the LP phase we switch to the smooth‑max
  L‑BFGS approach used previously: optimise a temperature‑controlled
  soft‑max of the convolution, gradually cooling the temperature.
  Several random restarts are performed.
* The whole search respects the 1000 s budget; the best sequence found
  (as a plain ``list[float]``) is returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy imports
try:
    from scipy.optimize import minimize, linprog
    import scipy.sparse as sp
    _SCIPY_OPT_AVAILABLE = True

# Helper functions – all top‑level (no closures)

def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max over the convolution values."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i}   (reverse‑weighted convolution).
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    # keep the valid part (length = len(w)+len(a)-1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p (real FFT based). Returns array of length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective Lτ and its gradient w.r.t. the log‑probability
    parameters y (soft‑max is the change‑of‑variables to enforce p≥0,
    Σp=1).
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)          # p∗p
    w, Ltau = _softmax_weights(conv, tau)

    # ∇_p Lτ = 2 * rev (see paper)
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # Transform gradient to y – Jacobian of soft‑max
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray,
                init_tau: float,
                target_sum: float,
                Nfft: int,
                deadline: float,
                best_seq: np.ndarray,
                best_score: float):

    Repeatedly minimise the smooth maximum with decreasing temperature.
    Returns the (potentially) improved sequence and its exact score.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65          # shrink factor after each stage
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the smooth objective -------------------------
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # modest limits – enough for a quick descent at each temperature
        options = {'maxfun': 250, 'maxiter': 250,
                   'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=options)
        y = res.x

        # ----- evaluate exact objective --------------------------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = (target_sum ** 2) * max_conv   # equals evaluate_sequence

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[L‑BFGS] τ={tau:.2e} improved score={best_score:.6f}")

        # ----- schedule update -----------------------------------------
        tau *= tau_decay

        # occasional jitter to hop out of plateaus
        if np.random.rand() < 0.15:
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break
    return best_seq, best_score


def _build_conv_constraints(f: np.ndarray):
    """
    Build the (2·n‑1) × n matrix A for the constraints (f * g)_k ≤ M,
    i.e. A @ g ≤ M·1.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = sp.lil_matrix((rows, n), dtype=np.float64)
    # Column j contains the vector f shifted down by j positions.
    for j in range(n):
        A[j:j + n, j] = f
    return A.tocsr()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0   # safety margin

    # ------------------------------------------------------------------
    # Load a starting construction; fall back to a uniform vector.
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:                                    # pragma: no cover
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to required L1‑norm
    base = base * (target_sum / base.sum())
    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, L1={best_seq.sum():.6f}, score={best_score:.6f}")

    # FFT size for all convolution calls
    Nfft = 1 << ((2 * n - 1).bit_length())

    # ------------------------------------------------------------------
    # 1) Frank‑Wolfe / LP phase (only if SciPy's linprog is available)
    f = best_seq.copy()
    no_improve = 0
    max_no_improve = 12
    iter_lp = 0

    while (time.time() < deadline) and (_SCIPY_OPT_AVAILABLE):
        iter_lp += 1
        # current max‑convolution M
        conv_f = np.convolve(f, f)
        M = conv_f.max()

        # Build constraint matrix A (sparse)
        A = _build_conv_constraints(f)

        # Linear programme: maximise Σ g_j  → minimise -Σ g_j
        c = -np.ones(n, dtype=float)
        b = np.full(2 * n - 1, M, dtype=float)

        # Solve LP
        try:
            lp_res = linprog(c, A_ub=A, b_ub=b,
                             bounds=(0, None),
                             method='highs',
                             options={'presolve': True, 'time_limit': max(0, deadline - time.time())})
        except Exception as e:  # pragma: no cover
            print("[LP] linprog exception:", e)
            break

        if not lp_res.success:
            print("[LP] failed:", lp_res.message)
            break

        g0 = lp_res.x
        sum_g0 = g0.sum()
        if sum_g0 <= 0:
            break

        # Scale to target L1‑norm
        g_scaled = g0 * (target_sum / sum_g0)

        # Try a few mixing weights and keep the best improvement
        improved = False
        for t in (0.2, 0.4, 0.6, 0.8):
            candidate = (1.0 - t) * f + t * g_scaled
            candidate = np.clip(candidate, 0.0, None)
            # renormalise (numerical drift)
            candidate = candidate * (target_sum / candidate.sum())
            score = evaluate_sequence(candidate.tolist())
            if score < best_score - 1e-9:
                best_score = score
                best_seq = candidate.copy()
                f = candidate.copy()
                improved = True
                no_improve = 0
                print(f"[LP] iter {iter_lp} τmix={t:.2f} → score={best_score:.6f}")
                break

        if not improved:
            no_improve += 1
            # small random perturbation to attempt escape
            jitter = np.random.randn(n) * 1e-3 * (f.mean() + 1e-12)
            f = np.clip(f + jitter, 0.0, None)
            f = f * (target_sum / f.sum())
            # evaluate jitter (optional)
            score_j = evaluate_sequence(f.tolist())
            if score_j < best_score - 1e-12:
                best_score = score_j
                best_seq = f.copy()
                print(f"[jitter] iter {iter_lp} → score={best_score:.6f}")
                no_improve = 0

        # early exit if progress stalls
        if no_improve >= max_no_improve:
            print("[LP] no improvement for several rounds – exiting LP phase")
            break

        # cut off if we already beat the target
        if best_score < 1.503:
            print("[LP] reached target bound, exiting early")
            break

    # ------------------------------------------------------------------
    # 2) L‑BFGS refinement (always available – uses only NumPy/FFT)
    # Convert the best sequence to log‑probabilities
    p0 = best_seq / target_sum
    eps = 1e-12
    y0 = np.log(p0 + eps)

    # initial temperature – a couple of times the current smooth max
    conv0 = _conv_full(p0, Nfft)
    init_tau = max(conv0.max() * 2.0, 1e-12)

    # generate a few start points (including the current one)
    starts = [
        y0,
        np.log(np.random.dirichlet(alpha=np.ones(n)) + eps),
        np.zeros(n, dtype=float),
        y0 + np.random.randn(n) * 0.05,
    ]

    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[L‑BFGS] start {idx + 1}/{len(starts)}")
        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Final sanity check
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print(f"[done] best score = {best_score:.6f}, length = {len(best_seq)}")
    return best_seq.tolist()
