def _simplex_projection(v, target_sum):
    """Projection onto the simplex {x >=0, sum(x)=target_sum}."""
    if np.sum(v) == target_sum and np.all(v >= 0):
        return v
    # sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - target_sum
    rho = np.where(u - cssv / np.arange(1, len(v)+1) > 0)[0]
    if rho.size == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _optimise_lp_iter(n, init_a, target_sum, run_deadline):
    import scipy.optimize as opt
    f = init_a.copy()
    best_seq = f.copy()
    best_score = np.max(full_convolution_fft(f,f))
    iteration = 0
    t_mix = 0.3 # mixing parameter
    while time.time() < run_deadline:
        conv = full_convolution_fft(f,f)
        M = conv.max()
        # construct A matrix
        A = np.zeros((2*n-1, n), dtype=float)
        for j in range(n):
            A[j:j+n, j] = f
        # solve LP max sum(g) => minimize -sum(g)
        c = -np.ones(n)
        b = np.full(2*n-1, M)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            break
        g = res.x
        sum_g = g.sum()
        if sum_g <= 1e-12:
            break
        # rescale to target sum
        g = g * (target_sum / sum_g)
        # mix
        f = (1 - t_mix) * f + t_mix * g
        # project onto simplex (to correct numerical errors)
        f = _simplex_projection(f, target_sum)
        # evaluate new
        new_score = np.max(full_convolution_fft(f,f))
        if new_score < best_score:
            best_score = new_score
            best_seq = f.copy()
            print(f"[{time.time():.2f}] n={n} LP new local best = {new_score:.6f}")
        # check improvement minimal
        if iteration > 0 and best_score/ new_score < 1.0001:
            # small improvement
            pass
        iteration += 1
        # optional early break if improvement small
        if iteration >= 30:
            break
    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score

def _full_convolution_fft(x, y):
    n = len(x) + len(y) -1
    size = 1 << ((n-1).bit_length())
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    conv = np.maximum(conv, 0.0, out=conv)
    return conv
```

Propose a non‑negative step‑function minimizing

    2 * n * max(conv(a,a)) / (sum(a)**2)

The code combines three independent optimisation strategies and
keeps the best sequence found within the allotted time budget.
* **Adam on a soft‑max parametrisation** – smooth‑max surrogate and
  adaptive step sizes (similar to the reference implementation but
  with a longer run per restart and a slightly more aggressive
  temperature schedule).
* **Sub‑gradient descent on the true max** – directly minimises the
  peak of the self‑convolution using a diminishing step‑size and a
  projection onto the simplex (sum = √(2 n)).
* **Linear‑programming “boost”** – given a current shape f, solves
  a LP that maximises the ℓ₁‑norm of a partner g under the
  convolution bound ‖f∗g‖∞ ≤ M = ‖f∗f‖∞; the rescaled partner is then
  mixed back into f (as described in the paper).  This step often
  yields a noticeable drop in the maximum.

The routine picks a random length (≈ 800 – 3500), creates a power‑law
initial shape, and runs one of the three optimisation loops for a
time slice.  The global best is retained and warm‑started whenever
possible.  The search stops early if the target bound 1.503 is
surpassed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Optional SciPy import (required only for the LP boost)
# ----------------------------------------------------------------------
try:
    import scipy.optimize as _opt
    _HAS_SCIPY = True
except Exception:                     # pragma: no cover
    _HAS_SCIPY = False

# ----------------------------------------------------------------------
# Fast full linear convolution via real FFT
# ----------------------------------------------------------------------
def _full_convolution_fft(x: np.ndarray, y: np.ndarray) -> np.ndarray:
    """Return the full linear convolution of two real vectors using FFT."""
    n = x.shape[0] + y.shape[0] - 1
    size = 1 << ((n - 1).bit_length())          # next power of two
    fx = np.fft.rfft(x, n=size)
    fy = np.fft.rfft(y, n=size)
    conv = np.fft.irfft(fx * fy, n=size)[:n]
    # eliminate tiny negative round‑off errors
    np.maximum(conv, 0.0, out=conv)
    return conv


# ----------------------------------------------------------------------
# Euclidean projection onto the simplex {z ≥ 0, sum(z) = s}
# ----------------------------------------------------------------------
def _simplex_projection(v: np.ndarray, s: float) -> np.ndarray:
    """Project vector v onto the simplex of non‑negative vectors with sum s."""
    if s < 0:
        raise ValueError("Simplex sum must be non‑negative")
    # If already feasible, return a copy to avoid side‑effects
    if np.all(v >= 0) and np.isclose(v.sum(), s):
        return v.copy()
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - s
    ind = np.arange(len(v)) + 1
    # Find the largest rho such that u[rho] > cssv[rho]/rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Adam optimiser on the soft‑max (log‑it) parametrisation
# ----------------------------------------------------------------------
def _optimise_adam(
    n: int,
    x0: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    init_tau_factor: float = 0.12,
    min_tau_factor: float = 0.001,
    alpha: float = 0.03,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smoothed‑max surrogate.
    Returns the best sequence (as a NumPy array) and the exact
    evaluation obtained from ``evaluate_sequence``.
    """
    x = x0.copy()
    m = np.zeros_like(x)
    v = np.zeros_like(x)

    best_seq = None
    best_score = float("inf")
    iteration = 0

    while time.time() < run_deadline:
        # ---- 1. soft‑max mapping (ensures non‑negativity & sum) ----
        max_x = np.max(x)
        exp_shift = np.exp(x - max_x)          # stability trick
        p = exp_shift / np.sum(exp_shift)      # simplex probabilities
        a = target_sum * p                     # actual heights

        # ---- 2. convolution & current max (exact objective) ----
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] Adam n={n} new local best = {cur_max:.6f}")

        # ---- 3. smooth‑max weights (softmax over conv entries) ----
        tau = max(init_tau_factor * cur_max * (0.93 ** iteration),
                  min_tau_factor * cur_max)
        scaled = (conv - cur_max) / tau
        w = np.exp(scaled)
        w_sum = w.sum()
        if w_sum == 0.0:
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- 4. gradient w.r.t. a (see derivation in the paper) ----
        w_rev = w[::-1]
        conv2 = _full_convolution_fft(a, w_rev)        # length = 3n‑2
        idx = (2 * n - 2) - np.arange(n)                # pick the right slice
        grad_a = 2.0 * conv2[idx]

        # ---- 5. chain rule back to logits x ------------------------
        dotp = np.dot(p, grad_a)
        grad_x = target_sum * p * (grad_a - dotp)

        # ---- 6. Adam update ----------------------------------------
        iteration += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x ** 2)

        m_hat = m / (1.0 - beta1 ** iteration)
        v_hat = v / (1.0 - beta2 ** iteration)

        step = alpha * m_hat / (np.sqrt(v_hat) + eps)
        x -= step

        # occasional tiny random kick to escape plateaus
        if iteration % 300 == 0 and np.random.rand() < 0.25:
            x += np.random.randn(n) * 0.001

        # slowly decay the base learning rate
        alpha *= 0.9995

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Sub‑gradient descent on the true max (no smoothing)
# ----------------------------------------------------------------------
def _optimise_subgradient(
    n: int,
    init_a: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    step0: float = 0.6,
    eps: float = 1e-12,
) -> tuple[np.ndarray, float]:
    """
    Simple sub‑gradient descent on max(conv(a,a)).
    Returns the best sequence and its evaluation.
    """
    a = init_a.copy()
    best_seq = a.copy()
    conv = _full_convolution_fft(a, a)
    best_score = conv.max()
    print(f"[{time.time():.2f}] SubGrad n={n} start score = {best_score:.6f}")

    iteration = 0
    while time.time() < run_deadline:
        conv = _full_convolution_fft(a, a)
        cur_max = conv.max()
        if cur_max < best_score:
            best_score = cur_max
            best_seq = a.copy()
            print(f"[{time.time():.2f}] SubGrad n={n} new local best = {cur_max:.6f}")

        # indices where the max is attained (allow small tolerance)
        max_idx = np.where(conv >= cur_max - 1e-12)[0]

        # ---- gradient of max (sum of gradients of active constraints) ----
        grad = np.zeros_like(a)
        for k in max_idx:
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            i = np.arange(start, end + 1, dtype=int)
            j = k - i
            grad[j] += 2.0 * a[i]

        # diminishing step size
        step = step0 / np.sqrt(iteration + 1)
        a = a - step * grad
        # project back onto the simplex (maintains sum = target_sum)
        a = _simplex_projection(a, target_sum)

        iteration += 1

        # occasional random perturbation to avoid cycles
        if iteration % 400 == 0 and np.random.rand() < 0.15:
            a += np.random.randn(n) * 1e-4
            a = _simplex_projection(a, target_sum)

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# LP boost: maximise ℓ₁‑norm of g under convolution bound, then mix
# ----------------------------------------------------------------------
def _optimise_lp_iter(
    n: int,
    init_a: np.ndarray,
    target_sum: float,
    run_deadline: float,
    *,
    mix_frac: float = 0.33,
    max_inner_iters: int = 30,
) -> tuple[np.ndarray, float]:
    """
    Run the linear‑programming iteration described in the paper.
    Returns the best sequence found and its evaluator score.
    """
    if not _HAS_SCIPY:                         # pragma: no cover
        raise RuntimeError("SciPy not available – LP boost cannot be used")

    a = init_a.copy()
    best_seq = a.copy()
    best_score = np.max(_full_convolution_fft(a, a))

    inner_iter = 0
    while time.time() < run_deadline and inner_iter < max_inner_iters:
        conv = _full_convolution_fft(a, a)
        M = conv.max()

        # Build the convolution matrix A where A[k, j] = a_{k-j}
        # (zero outside the valid range)
        A = np.zeros((2 * n - 1, n), dtype=float)
        for j in range(n):
            A[j:j + n, j] = a

        # Linear program: maximise sum(g)  ⇔ minimise -sum(g)
        c = -np.ones(n)
        b = np.full(2 * n - 1, M)
        bounds = [(0.0, None)] * n

        res = _opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # Something went wrong – abort the inner loop
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= eps:
            break

        # Rescale g so that it has the required ℓ₁‑norm
        g = g * (target_sum / sum_g)

        # Mix the new partner into the current sequence
        a = (1.0 - mix_frac) * a + mix_frac * g
        # Numerical errors may accumulate – re‑project onto simplex
        a = _simplex_projection(a, target_sum)

        cur_score = np.max(_full_convolution_fft(a, a))
        if cur_score < best_score:
            best_score = cur_score
            best_seq = a.copy()
            print(f"[{time.time():.2f}] LP n={n} new local best = {cur_score:.6f}")

        # tiny random jitter every few iterations to shake out of stagnation
        if inner_iter % 10 == 0 and np.random.rand() < 0.2:
            a += np.random.randn(n) * 1e-4
            a = _simplex_projection(a, target_sum)

        inner_iter += 1

    final_score = evaluate_sequence(best_seq.tolist())
    return best_seq, final_score


# ----------------------------------------------------------------------
# Main entry point expected by the outer evaluator
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation function.  The routine runs for at most ``budget_s`` seconds
    and returns the best sequence discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1  # tiny safety margin

    best_seq = None
    best_score = float("inf")

    # --------------------------------------------------------------
    # Warm‑start from a globally provided construction, if available.
    # --------------------------------------------------------------
    try:
        hs = globals()["height_sequence_1"]
        hs_arr = np.array(hs, dtype=float)
        hs_arr = np.clip(hs_arr, 0.0, 1000.0)
        # ensure the sum constraint is respected (the evaluator will
        # re‑normalise anyway, but keeping it close helps the optimiser)
        cur_score = evaluate_sequence(hs_arr.tolist())
        if cur_score < best_score:
            best_score = cur_score
            best_seq = hs_arr.copy()
            print(f"[{time.time():.2f}] Warm‑start global = {cur_score:.6f}")
    except Exception:
        pass  # no warm‑start available

    # --------------------------------------------------------------
    # Main restart loop – each restart receives a time slice.
    # --------------------------------------------------------------
    while time.time() < deadline:
        remaining = deadline - time.time()
        if remaining < 5.0:                     # not enough time for another run
            break

        # Choose a problem size adaptively (larger early, smaller later)
        if remaining > 300:
            n = np.random.randint(2000, 3500 + 1)
        elif remaining > 120:
            n = np.random.randint(1500, 3000 + 1)
        else:
            n = np.random.randint(800, 2000 + 1)

        # ----- initialise a power‑law shape (with optional taper) -----
        p_exp = np.random.uniform(0.25, 0.75)
        idx = np.arange(1, n + 1, dtype=float)
        init = idx ** (-p_exp)

        # optional exponential decay
        if np.random.rand() < 0.33:
            tau_exp = np.random.uniform(0.5, 2.0) * n
            init *= np.exp(-idx / tau_exp)

        # optional tiny smoothing
        if np.random.rand() < 0.20:
            klen = np.random.randint(3, 9)
            kernel = np.ones(klen) / klen
            init = np.convolve(init, kernel, mode='same')

        # add a little jitter to break symmetry
        init += np.random.rand(n) * 1e-5
        init = np.maximum(init, 1e-12)

        # ------------------------------------------------------------------
        # Warm‑start from the current global best with some probability
        # ------------------------------------------------------------------
        if best_seq is not None and np.random.rand() < 0.45:
            if best_seq.size >= n:
                init = best_seq[:n].copy()
            else:
                pad = np.random.rand(n - best_seq.size) * 1e-5
                init = np.concatenate([best_seq, pad])

        # Normalise to the required sum √(2 n)
        target_sum = np.sqrt(2.0 * n)
        init = init / init.sum() * target_sum

        # --------------------------------------------------------------
        # Pick an optimisation method for this restart
        # --------------------------------------------------------------
        method_choices = ["adam", "subgrad"]
        if _HAS_SCIPY:
            method_choices.append("lp")
        method = np.random.choice(method_choices, p=[0.45, 0.35, 0.20] if _HAS_SCIPY else [0.55, 0.45])

        # Allocate a time slice (roughly 8‑30 s depending on remaining budget)
        slice_time = min(30.0, remaining * 0.5)
        run_deadline = min(deadline, time.time() + slice_time)

        if method == "adam":
            # Transform to log‑it variables for Adam
            x0 = np.log(init + 1e-12)
            seq_local, score_local = _optimise_adam(
                n=n,
                x0=x0,
                target_sum=target_sum,
                run_deadline=run_deadline,
                init_tau_factor=0.12,
                min_tau_factor=0.001,
                alpha=0.03,
                beta1=0.9,
                beta2=0.999,
                eps=1e-8,
            )
        elif method == "subgrad":
            seq_local, score_local = _optimise_subgradient(
                n=n,
                init_a=init.copy(),
                target_sum=target_sum,
                run_deadline=run_deadline,
                step0=0.6,
            )
        else:  # LP boost
            seq_local, score_local = _optimise_lp_iter(
                n=n,
                init_a=init.copy(),
                target_sum=target_sum,
                run_deadline=run_deadline,
                mix_frac=0.33,
                max_inner_iters=30,
            )

        # --------------------------------------------------------------
        # Update global best if improvement was found
        # --------------------------------------------------------------
        if score_local < best_score:
            best_score = score_local
            best_seq = seq_local.copy()
            print(
                f"[{time.time():.2f}] GLOBAL new best = {best_score:.6f} "
                f"(n={n}, method={method})"
            )
            if best_score < 1.503:            # early exit if target beaten
                break

    # ------------------------------------------------------------------
    # Fallback – in the unlikely case no candidate was produced.
    # ------------------------------------------------------------------
    if best_seq is None:
        n = 1000
        target_sum = np.sqrt(2.0 * n)
        best_seq = np.full(n, target_sum / n, dtype=float)

    # The evaluator expects a plain Python list.
    return best_seq.tolist()
