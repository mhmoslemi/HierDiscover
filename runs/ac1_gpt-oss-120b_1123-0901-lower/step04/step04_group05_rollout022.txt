Pseudo:

def project_simplex(v):
    # returns x >=0, sum x =1 and close to v in L2.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def conv_fft(a):

    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[:2*n-1]
    return conv

def local_descent(a, max_iter, deadline):
    n = len(a)
    conv_fft_size = 1 << int(np.ceil(np.log2(2*n - 1)))
    best_local_val = evaluate_sequence(a.tolist())
    best_local_a = a.copy()
    last_improve_time = time.time()
    lr = 0.05
    for it in range(max_iter):
        if time.time() >= deadline:
            break
        # compute conv
        fa = np.fft.rfft(a, conv_fft_size)
        conv = np.fft.irfft(fa * fa, conv_fft_size)[:2*n-1]
        max_val = conv.max()
        # objective
        obj = 2.0 * n * max_val / (np.sum(a)**2)  # sum should be 1
        # compute gradient at max index
        k = int(np.argmax(conv))
        grad = gradient_maxconv(a, k)  # returns shape n

        # Optional: normalizing gradient? Use step size.
        # Compute descent direction
        direction = -grad

        # step size schedule
        step = lr / np.sqrt(it+1)

        # tentative update
        a_new = a + step * direction

        # projection onto simplex
        a_new = project_simplex(a_new)

        # compute new objective via evaluate_sequence, but for speed we compute via conv.
        fa_new = np.fft.rfft(a_new, conv_fft_size)
        conv_new = np.fft.irfft(fa_new * fa_new, conv_fft_size)[:2*n-1]

"""
Hybrid stochastic‑gradient & projected sub‑gradient search.

* Randomly draws many asymmetric initial step‑functions on several
  lengths (≈2–4k).  Each candidate is improved by a fast projected
  sub‑gradient descent that works directly on the true objective
  `2·n·max(conv(a,a))/(sum a)^2` (the max is taken, not a smooth‑max).
* The gradient of the maximal convolution entry is computed analytically
  (O(n) slicing) and the iterate is projected onto the probability simplex
  to keep all heights non‑negative and Σa = 1.
* If the algorithm stalls (no improvement for a few iterations) a tiny
  random shake is added to escape plateaus.
* A cheap FFT based convolution (`np.fft.rfft`) is used for every step;
  the whole search respects the user supplied time budget (default 1000 s).
* The best sequence found (as a plain Python list) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _fft_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using a real FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x | x_i ≥ 0, Σ_i x_i = 1}.
    """
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v.copy()
    n = v.shape[0]
    u = np.sort(v)[::-1]                     # sorted descending
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    # Numerical safety – renormalise to exactly 1
    w /= w.sum() if w.sum() > 0 else 1.0
    return w


def _grad_max_conv(a: np.ndarray, k: int) -> np.ndarray:
    """
    Gradient of the linear convolution entry conv[k] = Σ_i a_i a_{k-i}
    with respect to a, for a non‑negative vector a of length n.
    Returns a vector of length n.
    """
    n = a.shape[0]
    g = np.zeros_like(a)

    if k < n:
        # term a_{k-i}
        i = np.arange(k + 1)               # 0 .. k
        g[i] += a[k - i]

        # term a_{i-k}
        i2 = np.arange(k, n)               # k .. n‑1
        g[i2] += a[i2 - k]
    else:
        start = k - (n - 1)                 # smallest i with k-i ≤ n‑1
        i = np.arange(start, n)            # start .. n‑1
        g[i] += a[k - i]                   # only the first term exists

    return g


def _local_descent(
    a0: np.ndarray,
    n: int,
    deadline: float,
    global_best: float,
    max_iter: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective
    `obj(a) = 2·n·max(conv(a,a)) / (Σa)²`  (Σa is kept equal to 1).
    Returns the best vector found and its exact objective value
    as evaluated by the supplied `evaluate_sequence`.
    """
    a = a0.copy()
    size_fft = 1 << int(np.ceil(np.log2(2 * n - 1)))

    # initial exact evaluation (slow but cheap compared to many runs)
    best_val = float("inf")
    best_a = a.copy()
    try:
        cur_val = evaluate_sequence(a.tolist())
    except Exception:                     # fallback if evaluate_sequence missing
        conv = _fft_convolution(a, size_fft)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    if cur_val < best_val:
        best_val, best_a = cur_val, a.copy()

    last_improve = time.time()
    stall_counter = 0
    step0 = 0.07                     # initial step size (will decay)

    for it in range(1, max_iter + 1):
        if time.time() >= deadline:
            break

        # --- convolution & max position ------------------------------------
        fa = np.fft.rfft(a, size_fft)
        conv = np.fft.irfft(fa * fa, size_fft)[: 2 * n - 1]
        max_idx = int(np.argmax(conv))
        obj = 2.0 * n * conv[max_idx]          # Σa = 1 ⇒ denominator = 1

        # --- gradient of the active entry ---------------------------------
        grad = _grad_max_conv(a, max_idx)

        # --- projected sub‑gradient step -----------------------------------
        step = step0 / np.sqrt(it)            # simple 1/√t decay
        a_new = a - step * grad
        a_new = _project_simplex(a_new)

        # --- inexpensive objective check -----------------------------------
        fa_new = np.fft.rfft(a_new, size_fft)
        conv_new = np.fft.irfft(fa_new * fa_new, size_fft)[: 2 * n - 1]
        obj_new = 2.0 * n * conv_new.max()

        # accept if we made progress
        if obj_new < obj - 1e-13:
            a = a_new
            obj = obj_new
            stall_counter = 0
            if obj_new < best_val - 1e-13:
                best_val = obj_new
                best_a = a_new.copy()
                last_improve = time.time()
                print(f"   [local] iter {it:04d}  new local best = {best_val:.6f}")
        else:
            stall_counter += 1
            # occasional shake when stuck for a while
            if stall_counter >= 30:
                a = a + (np.random.rand(n) - 0.5) * 1e-3
                a = _project_simplex(a)
                stall_counter = 0

        # early exit if we are already better than the global target
        if obj < global_best and obj < 1.503:
            break

    # final exact evaluation (to avoid any subtle rounding differences)
    final_val = evaluate_sequence(best_a.tolist())
    return best_a, final_val

        elif mode == 4:
            r = np.random.uniform(0.3, 0.85)
            a0 = r ** np.arange(n, dtype=np.float64) + 0.01 * np.random.rand(n)
        elif mode == 5:
            x = np.linspace(-1.0, 1.0, n)
            a0 = 1.0 - x ** 2
        else:                                   # bimodal half‑sine
            half = n // 2
            left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
            right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
            a0 = np.concatenate([left, right])

        a0 = np.maximum(a0, 0.0)
        a0 /= a0.sum()

        # Occasionally seed from the best sequence found so far (if compatible)
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.15:
            a0 = np.array(best_seq, copy=True)

        # Occasionally use the global historic construction (if sizes match)
        if seed_seq is not None and seed_seq.shape[0] == n and np.random.rand() < 0.05:
            a0 = seed_seq.copy()

        # ----- local improvement -------------------------------------------
        a_opt, val_opt = _local_descent(
            a0, n, deadline, best_val, max_iter=1800
        )

        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(
                f"[Restart {restart:03d}] new GLOBAL best = {best_val:.6f} length {n}"
            )

        restart += 1

        # Stop early if we already beat the target
        if best_val < 1.503:
            break

    # ------------------------------------------------------------------
    # Fallback – should never be used, but guarantees a valid return value
    if best_seq is None:
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
