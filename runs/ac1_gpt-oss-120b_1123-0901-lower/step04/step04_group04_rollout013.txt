def simplex_projection(v):
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

4. From the best L‑BFGS result start a long sub‑gradient descent / simulated‑
   annealing polishing phase:
   * The exact sub‑gradient of F at p is 4·n·rev where
         rev_i = p_{k*‑i}  for the index k* attaining the maximal convolution.
   * Perform a projected gradient step (project onto the probability simplex)
   * Occasionally make random “mass‑swap’’ moves and accept them with a
     temperature schedule (simulated annealing) to escape shallow local minima.
   * Continue until the time budget expires.
5. Convert the best probability vector back to heights by multiplying with
   √(2 n) and return it as a plain Python list.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns weight vector w (summing to 1) and the smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """Compute rev_i = Σ_k w_k·a_{k‑i} using FFTs (indices out of range ignored)."""
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p (real FFT based)."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Smooth‑max objective Lτ(p) and its gradient w.r.t. the log‑probability
    parameters y (p = softmax(y)).
    """
    # softmax
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    # convolution
    conv = _conv_full(p, Nfft)

    # smooth max
    w, Ltau = _softmax_weights(conv, tau)

    # gradient wrt p
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # back‑prop to y (softmax Jacobian)
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries will be zero
        return np.zeros_like(v)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _max_convolution(p: np.ndarray, Nfft: int) -> float:
    """Return max_{k} (p∗p)_k."""
    conv = np.fft.irfft(np.fft.rfft(p, n=Nfft) ** 2, n=Nfft)[: 2 * len(p) - 1]
    return float(conv.max())


def _objective_p(p: np.ndarray, Nfft: int) -> float:
    """Exact objective F(p) = 2·n·max(p∗p)."""
    n = len(p)
    return 2.0 * n * _max_convolution(p, Nfft)


def _subgradient_step(p: np.ndarray, lr: float, Nfft: int) -> np.ndarray:
    """One projected sub‑gradient step for the exact objective."""
    n = len(p)
    # convolution and argmax
    conv = np.fft.irfft(np.fft.rfft(p, n=Nfft) ** 2, n=Nfft)[: 2 * n - 1]
    kmax = int(np.argmax(conv))

    # build rev = (p shifted by kmax)
    rev = np.zeros_like(p)
    lo = max(0, kmax - (n - 1))
    hi = min(n - 1, kmax)
    if lo <= hi:
        idx = np.arange(lo, hi + 1)
        rev[idx] = p[kmax - idx]

    # gradient of F is 4·n·rev; absorb 4·n into learning rate
    p_new = p - lr * rev
    # project back onto simplex
    p_new = _project_onto_simplex(p_new)
    return p_new


def _random_dirichlet(n: int) -> np.ndarray:
    """Draw a random probability vector from Dirichlet(1,…,1) (uniform on simplex)."""
    x = np.random.exponential(scale=1.0, size=n)
    return x / x.sum()


# ----------------------------------------------------------------------
# Core search routines
# ----------------------------------------------------------------------
def _run_l_bfgs(y0: np.ndarray, init_tau: float, target_sum: float,
                Nfft: int, deadline: float,
                best_seq: np.ndarray, best_score: float) -> (np.ndarray, float):
    """Continuation L‑BFGS optimisation (same spirit as the reference code)."""
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the smooth max ----- #
        def fun(z):
            L, g = _objective_and_grad(z, tau, Nfft)
            return L, g

        # give the inner optimiser a few hundred calls
        opts = {"maxfun": 400, "maxiter": 400, "ftol": 1e-12, "gtol": 1e-12}
        try:
            from scipy.optimize import minimize
            res = minimize(fun, y, method="L-BFGS-B", jac=True, options=opts)
            y = res.x
        except Exception:
            # fall back to a tiny number of gradient steps if SciPy missing
            L, g = fun(y)
            y = y - 0.01 * g
            y = _project_onto_simplex(np.exp(y))  # keep roughly feasible

        # ----- evaluate exact objective ----- #
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        cur_score = (target_sum ** 2) * _max_convolution(p, Nfft) * 2 * len(p)
        # cur_score equals evaluate_sequence after scaling

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve L‑BFGS] τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule update ----- #
        tau *= tau_decay
        if np.random.rand() < 0.15:  # occasional jitter
            y += np.random.randn(y.shape[0]) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _polish_subgradient(p_start: np.ndarray, target_sum: float,
                        deadline: float, best_seq: np.ndarray,
                        best_score: float) -> (np.ndarray, float):
    """Long polishing phase using projected sub‑gradient descent + occasional swaps."""
    n = len(p_start)
    Nfft = 1 << ((2 * n - 1).bit_length())
    p = p_start.copy()
    cur_score = _objective_p(p, Nfft)

    it = 0
    # learning‑rate schedule parameters
    init_lr = 0.5   # raw step size for the rev vector
    decay = 0.0005  # slower decay gives many small steps

    while time.time() < deadline:
        it += 1
        lr = init_lr / (1.0 + decay * it)

        # ----- projected sub‑gradient step ----- #
        p = _subgradient_step(p, lr, Nfft)
        cur_score = _objective_p(p, Nfft)

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[polish] it={it}  score={best_score:.6f}")

        # ----- occasional random mass‑swap move (simulated annealing) ----- #
        if np.random.rand() < 0.03:
            i, j = np.random.randint(0, n, size=2)
            while i == j:
                j = np.random.randint(0, n)
            max_transfer = p[i]
            if max_transfer > 0:
                delta = np.random.rand() * max_transfer
                p_new = p.copy()
                p_new[i] -= delta
                p_new[j] += delta
                # projection not needed – sum stays 1 and we kept non‑negativity
                new_score = _objective_p(p_new, Nfft)
                # simple annealing schedule
                temp = max(0.01, (best_score - cur_score) * 0.5)
                if new_score < cur_score or np.random.rand() < math.exp(-(new_score - cur_score) / temp):
                    p = p_new
                    cur_score = new_score
                    if cur_score < best_score:
                        best_score = cur_score
                        best_seq = (p * target_sum).copy()
                        print(f"[swap] it={it}  score={best_score:.6f}")

        # occasional global jitter to shake the state
        if np.random.rand() < 0.005:
            jitter = np.random.randn(n) * 0.005
            p = _project_onto_simplex(p + jitter)

    return best_seq, best_score


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000.0, **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    Uses a combination of continuation L‑BFGS on a smooth max‑approximation
    and a long projected sub‑gradient polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 2.0  # small safety margin

    # ----- initialise base construction ----- #
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        # fallback – uniform of length 2100
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # normalise to the required L1‑norm
    base = base * (target_sum / base.sum())
    p0 = base / target_sum               # probability vector ∑p = 1

    # ----- best record ----- #
    best_seq = base.copy()
    # compute exact objective for the initial sequence using the known formula
    Nfft = 1 << ((2 * n - 1).bit_length())
    best_score = 2.0 * n * _max_convolution(p0, Nfft)
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ----- prepare several starting points for L‑BFGS ----- #
    eps_log = 1e-12
    y0 = np.log(p0 + eps_log)            # log‑softmax parameters

    starts = []
    starts.append(y0)                     # deterministic start (base)
    starts.append(np.log(_random_dirichlet(n) + eps_log))  # random Dirichlet
    starts.append(np.zeros(n, dtype=float))                # uniform in log‑space
    starts.append(y0 + np.random.randn(n) * 0.05)          # perturbed base

    # ----- run L‑BFGS from each start ----- #
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[run L‑BFGS] start {idx + 1}/{len(starts)}")
        # initial temperature roughly twice the current smooth max
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ----- polishing phase: projected sub‑gradient descent ----- #
    if time.time() < deadline:
        print("[polish] starting sub‑gradient phase")
        p_best = best_seq / target_sum
        best_seq, best_score = _polish_subgradient(
            p_start=p_best,
            target_sum=target_sum,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ----- final evaluation (safety) ----- #
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()
    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    return best_seq.tolist()
