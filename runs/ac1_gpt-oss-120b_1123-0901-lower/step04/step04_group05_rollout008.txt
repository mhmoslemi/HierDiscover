    restart = 0

    while time.time() < deadline:
        n = lengths[restart % len(lengths)]

        # Precompute FFT sizes

        size_corr = 1 << int(np.ceil(np.log2(3*n)))

        # Init a0

        a0 = init_sequence(n, init_type)
        # optionally try to use best_seq if same length
        if best_seq is not None and len(best_seq) == n and np.random.rand() < 0.2:
            a0 = np.array(best_seq)

        # convert to logits
        u = np.log(a0 + 1e-30)

        # Adam state

        # hyperparameters
        lam = 0.5
        lam_target = 30.0  # final smoothing param for exact max (like large)
        lam_factor = 1.011
        base_lr = 0.05
        it = 0
        max_iter = 20000

        # inner loop
        while time.time() < deadline and it < max_iter:
            it += 1

            # compute prob a via softmax
            u_shift = u - np.max(u)
            a = np.exp(u_shift)

"""
Multi‑restart Adam optimiser with smooth‑max continuation,
exact‑max fine‑tuning and a cheap random‑walk refinement.

The algorithm works with a probability vector a (Σa = 1) represented by
logits u = log a.  For each restart a length *n* is chosen and a is
initialised from one of several parametric families (Dirichlet,
exponential, half‑sine, polynomial, bimodal …).  While time remains we
run an Adam optimiser on the surrogate

    F_λ(a) = 2·n·∑_k w_k·(a∗a)[k]   with   w_k ∝ exp(λ·(a∗a)[k])

where λ starts small (smooth landscape) and is multiplied by a factor
> 1 every iteration (continuation).  When λ exceeds a large target we
switch to the exact sub‑gradient (w is a Kronecker‑delta at the maximal
convolution index).  The best sequence over all restarts is stored.

When the global budget is almost exhausted a simple random‑walk /
simulated‑annealing phase perturbs the current best sequence while
maintaining non‑negativity and Σa = 1, accepting any improvement of the
true objective

    obj(a) = 2·n·max_k (a∗a)[k] / (∑a)² .

The returned value is a plain Python list of non‑negative floats ready for
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray, size: int = None) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    n = a.shape[0]
    if size is None:
        size = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * n - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Smooth‑max weights  w_k = exp(lam·conv_k) / Σ_j exp(lam·conv_j)."""
    m = conv.max()
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:                     # safety – should not happen
        w.fill(1.0 / w.shape[0])
    else:
        w /= s
    return w


def _softmax_gradient(a: np.ndarray, w: np.ndarray,
                      size_corr: int = None) -> np.ndarray:
    """
    Gradient of  Σ_k w_k·(a∗a)[k]  w.r.t. a.
    Closed form:  ∇ = 2·(w ★ a) .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    if size_corr is None:
        size_corr = 1 << int(np.ceil(np.log2(len(w) + n - 1)))
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # elements n‑1 … 2·n‑2 correspond to the correlation needed for grad
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad


# ----------------------------------------------------------------------
def _init_sequence(n: int, mode: int) -> np.ndarray:
    """Generate a random non‑negative vector of length n and normalise it."""
    if mode == 0:                      # Dirichlet (almost uniform)
        a = np.random.dirichlet(np.ones(n))
    elif mode == 1:                    # Simple exponential decay
        r = np.random.uniform(0.2, 0.9)
        a = r ** np.arange(n, dtype=np.float64)
        a /= a.sum()
    elif mode == 2:                    # Half‑sine
        i = np.arange(1, n + 1, dtype=np.float64)
        a = np.sin(np.pi * i / (n + 1))
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 3:                    # Sine‑power
        p = np.random.uniform(0.5, 2.0)
        a = np.sin(np.pi * np.arange(1, n + 1) / (n + 1)) ** p
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 4:                    # Polynomial decay  a_i ∝ (n‑i)^p
        p = np.random.uniform(1.0, 3.0)
        a = (np.arange(n, 0, -1, dtype=np.float64) ** p)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    elif mode == 5:                    # Bimodal half‑sine glued together
        half = n // 2
        left = np.sin(np.pi * (np.arange(half) + 1) / (half + 1))
        right = np.sin(np.pi * (np.arange(n - half) + 1) / (n - half + 1))
        a = np.concatenate([left, right])
        a = np.maximum(a, 0.0)
        a /= a.sum()
    else:                              # Small uniform noise on top of uniform
        a = np.full(n, 1.0 / n)
        a += 0.001 * np.random.rand(n)
        a = np.maximum(a, 0.0)
        a /= a.sum()
    return a


def _resample(seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of ``seq`` to length ``new_n`` and renorm."""
    old_n = len(seq)
    x_old = np.arange(old_n)
    x_new = np.linspace(0, old_n - 1, new_n)
    new_seq = np.interp(x_new, x_old, seq)
    new_seq = np.maximum(new_seq, 0.0)
    new_seq /= new_seq.sum()
    return new_seq


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: float = 1000) -> list[float]:
    """
    Search for a non‑negative step function (list of heights) that
    minimises ``evaluate_sequence``.  The routine respects the supplied
    time budget (seconds) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # small safety margin

    best_val = float("inf")
    best_seq = None
    restart = 0

    # lengths to try – larger n gives a larger “design space”
    lengths = [2000, 2500, 3000, 3500, 4000, 5000]

    # ------------------------------------------------------------------
    while time.time() < deadline:
        # choose a length (cycling)
        n = lengths[restart % len(lengths)]

        # FFT sizes – stay constant inside one inner run
        conv_fft_size = 1 << int(np.ceil(np.log2(2 * n - 1)))
        size_corr = 1 << int(np.ceil(np.log2(3 * n)))   # safe upper bound

        # ----- initialise -------------------------------------------------
        mode = restart % 7
        a0 = _init_sequence(n, mode)

        # Occasionally inject the current global best (resampled if needed)
        if best_seq is not None and np.random.rand() < 0.15:
            if len(best_seq) == n:
                a0 = np.array(best_seq)
            else:
                a0 = _resample(np.array(best_seq), n)

        # Logits (soft‑max parametrisation)
        u = np.log(a0 + 1e-30)

        # Adam state
        m = np.zeros_like(u)
        v = np.zeros_like(u)

        # Continuation parameters
        lam = 0.5               # start smooth
        lam_target = 30.0       # switch to exact max around here
        lam_factor = 1.012
        base_lr = 0.045

        it = 0
        max_iter = 20000
        last_improve = time.time()

        # ------------------------------------------------------------------
        while time.time() < deadline and it < max_iter:
            it += 1

            # soft‑max to obtain a probability vector (Σa = 1)
            u_shift = u - np.max(u)
            a = np.exp(u_shift)
            a /= a.sum()

            # convolution
            conv = _full_convolution(a, size=conv_fft_size)

            # pick surrogate weights
            if lam < lam_target:
                w = _softmax_weights(conv, lam)
            else:
                idx = int(np.argmax(conv))
                w = np.zeros_like(conv)
                w[idx] = 1.0

            # gradient w.r.t. a (smooth‑max or exact‑max)
            grad_a = _softmax_gradient(a, w, size_corr=size_corr)

            # map to logits gradient (soft‑max Jacobian)
            dot = np.dot(a, grad_a)
            grad_u = a * (grad_a - dot)

            # Adam update
            m = 0.9 * m + 0.1 * grad_u
            v = 0.999 * v + 0.001 * (grad_u ** 2)

            it_f = float(it)
            m_hat = m / (1.0 - 0.9 ** it_f)
            v_hat = v / (1.0 - 0.999 ** it_f)

            # learning‑rate schedule – a bit more aggressive early on
            if lam < lam_target:
                lr = base_lr * (0.9997 ** it)
            else:
                lr = (base_lr * 0.5) * (0.9999 ** it)

            u -= lr * m_hat / (np.sqrt(v_hat) + 1e-8)

            # raise smoothness
            lam = min(lam_target, lam * lam_factor)

            # exact evaluation every few steps
            if it % 20 == 0:
                val = 2.0 * n * conv.max() / (a.sum() ** 2)   # a.sum()≈1
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    last_improve = time.time()
                    print(f"[Restart {restart:03d}] iter {it:05d}  new best = {val:.6f}")

            # shake if stagnating
            if time.time() - last_improve > 3.0:
                u += (np.random.rand(n) - 0.5) * 1e-3
                m.fill(0.0)
                v.fill(0.0)
                last_improve = time.time()

        # end of inner optimisation
        restart += 1
        print(f"[Restart {restart:03d}] finished it={it},  best so far = {best_val:.6f}")

    # ------------------------------------------------------------------
    # ---------- cheap random‑walk / simulated‑annealing refinement --------
    # Use the remaining time (if any) for a simple local search.
    if best_seq is not None:
        time_left = deadline - time.time()
        if time_left > 5.0:
            a = best_seq.copy()
            n = len(a)
            best_val_local = best_val
            # Pre‑compute a convolution once per trial – cheap for small move sets
            while time.time() < deadline:
                # propose a tiny redistribution of mass
                i, j = np.random.randint(0, n, size=2)
                if i == j:
                    continue
                # magnitude of the move – a few percent of the source entry
                eps = (np.random.rand() * 0.15 + 0.01) * a[i]
                if eps <= 0.0:
                    continue
                new_a = a.copy()
                new_a[i] -= eps
                new_a[j] += eps
                # safeguard non‑negativity
                if new_a[i] < 0.0:
                    continue
                # evaluate exact objective
                conv_new = _full_convolution(new_a)
                val_new = 2.0 * n * conv_new.max() / (new_a.sum() ** 2)
                if val_new + 1e-9 < best_val_local:      # strict improvement
                    a = new_a
                    best_val_local = val_new
                    best_seq = a.copy()
                    best_val = val_new
                    print(f"[Local] improvement → {val_new:.6f}")

    # ------------------------------------------------------------------
    # final sanity clamp (required by evaluate_sequence)
    if best_seq is None:
        # fallback – uniform distribution
        n = 2500
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(np.maximum(best_seq, 0.0), 0.0, 1000.0).tolist()
    print(f"Final best objective = {best_val:.6f} (length {len(best_seq)})")
    return best_seq
