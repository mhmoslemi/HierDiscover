"""
A fast multi‑phase optimizer for the step‑function problem.

The objective is

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,  a_i ≥ 0

where “∗” is the linear convolution.
The algorithm works completely on the probability simplex
(∑a_i = 1) – the scaling cancels out in the objective.

Phases
-------
1. **Parametric warm‑start** – a small grid of power‑law shapes
   a_i ∝ (i+offset)^{-p} (both one‑sided and symmetric) is evaluated
   on the chosen length.  The best of those cheap candidates seeds the
   later optimisation.

2. **Adam on the exact sub‑gradient of the max‑convolution** – the
   vector is kept as soft‑max logits so positivity and the simplex
   constraint are satisfied automatically.  Several restarts with a
   generous iteration budget further explore the neighbourhood.

3. **Differential Evolution (DE)** – a classic global search on the
   simplex (population size 10–20).  The best individual from the
   Adam phase is injected as a warm‑start; the DE run stops early if
   no improvement is observed for 10 s.

4. **(1+1) Evolution Strategy (ES)** – fine‑tuning with Gaussian
   perturbations and a self‑adjusting mutation strength.

The whole routine respects a total time budget (default 1000 s).  The
best sequence found before the deadline is returned as a plain
Python list, ready for the external ``evaluate_sequence`` call.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Convolution utilities – FFT based for speed
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using a power‑of‑2 FFT."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def conv_max(a: np.ndarray) -> float:
    """Maximum entry of the linear self‑convolution of ``a``."""
    return full_convolution(a).max()


def compute_objective(a: np.ndarray) -> float:
    """
    Exact objective used by ``evaluate_sequence``.
    The vector ``a`` is assumed to be non‑negative; its sum may be any
    positive number – the formula is scale‑invariant.
    """
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * conv_max(a) / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient of the max‑convolution (used by Adam)
# ----------------------------------------------------------------------
def grad_submax(a: np.ndarray):
    """
    Sub‑gradient of max(a∗a) w.r.t. ``a``.
    Returns (gradient, current max value).
    The gradient follows ∂max/∂a_j = 2·a_{k*‑j},
    where k* is a random index attaining the maximum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick a random maximal index to avoid bias
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


# ----------------------------------------------------------------------
# Adam optimiser on the simplex (logits + softmax)
# ----------------------------------------------------------------------
def run_adam_opt(n: int, init_a: np.ndarray, deadline: float,
                 max_iters: int = 200_000):
    """Adam on the exact sub‑gradient while staying on the simplex."""
    # initialise logits from the probability vector (add eps to avoid log‑0)
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = np.exp(x - np.max(x))          # softmax (stable)
        a = a / a.sum()

        grad_a, max_c = grad_submax(a)
        val = 2.0 * n * max_c               # sum(a) == 1 → exact objective

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            print(f"[Adam n={n} iter={t:05d}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded to avoid overflow
        x = np.clip(x, -50.0, 50.0)

        # small random shake if we stagnate
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} iter={t}] shake (stagnation)")

    # fallback – uniform vector (should never happen)
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple Differential Evolution on the simplex
# ----------------------------------------------------------------------
def run_de_opt(n: int, init_pop: list, deadline: float,
               popsize: int = 10, max_gens: int = 20_000):
    """
    Differential Evolution (DE/best/1/bin) on the simplex.
    The population is kept normalised after every trial.
    """
    # ---- initialise population ------------------------------------------------
    pop = []
    for cand in init_pop:
        a = np.asarray(cand, dtype=np.float64)
        s = a.sum()
        if s > 0.0:
            pop.append(a / s)
        else:
            pop.append(np.full(n, 1.0 / n))
    while len(pop) < popsize:
        pop.append(np.random.dirichlet(np.ones(n)))
    # ---- evaluate -------------------------------------------------------------
    obj_vals = [compute_objective(a) for a in pop]
    best_idx = int(np.argmin(obj_vals))
    best_a = pop[best_idx].copy()
    best_val = obj_vals[best_idx]

    F = 0.8   # differential weight
    CR = 0.9  # crossover probability
    last_improve = time.time()
    gen = 0

    while time.time() < deadline and gen < max_gens:
        gen += 1
        for i in range(popsize):
            # three distinct donors ≠ i
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = np.random.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.maximum(mutant, 0.0)               # enforce ≥0

            # binomial crossover (ensure at least one mutant component)
            cross = np.random.rand(n) < CR
            cross[np.random.randint(n)] = True
            trial = np.where(cross, mutant, pop[i])

            s = trial.sum()
            if s == 0.0:
                trial = np.full(n, 1.0 / n)
            else:
                trial = trial / s

            val = compute_objective(trial)
            if val < obj_vals[i]:
                pop[i] = trial
                obj_vals[i] = val
                if val < best_val:
                    best_val = val
                    best_a = trial.copy()
                    last_improve = time.time()
                    print(f"[DE n={n} gen={gen:04d} i={i}] new best: {val:.6f}")

        # stop early if stagnant for a while
        if time.time() - last_improve > 10.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# (1+1) Evolution Strategy – fine‑tuning
# ----------------------------------------------------------------------
def run_es_opt(n: int, init_a: np.ndarray, deadline: float,
               max_iter: int = 30_000):
    """Gaussian (1+1)‑ES on the simplex with simple 1/5‑success rule."""
    a = init_a.copy()
    best_a = a.copy()
    best_val = compute_objective(a)

    sigma = 0.02                     # initial mutation strength
    last_improve = time.time()
    it = 0

    while time.time() < deadline and it < max_iter:
        it += 1
        trial = a + sigma * np.random.randn(n)
        trial = np.maximum(trial, 0.0)
        s = trial.sum()
        if s == 0.0:
            trial = np.full(n, 1.0 / n)
        else:
            trial = trial / s

        val_trial = compute_objective(trial)
        if val_trial < best_val:
            best_val = val_trial
            best_a = trial.copy()
            a = trial                               # success → keep
            sigma = min(sigma * 1.05, 0.2)           # slightly increase
            last_improve = time.time()
            print(f"[ES n={n} it={it}] new best: {best_val:.6f}")
        else:
            sigma *= 0.99                           # decay

        # occasional reset if we have been stuck for a while
        if time.time() - last_improve > 5.0:
            a = best_a.copy()
            sigma = 0.02
            last_improve = time.time()

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point required by the evaluation framework
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the
    evaluation function.  The routine respects the given time budget
    and returns the best sequence (plain Python list) discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    best_seq = None
    best_val = np.inf

    # Try a few lengths – larger n gives a lower theoretical bound.
    # We start with the largest feasible size and fall back if time runs out.
    lengths = [8192, 4096, 2048, 1024]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # --------------------------------------------------------------
        # 1) cheap parametric warm‑starts (power‑law families)
        # --------------------------------------------------------------
        exponents = np.arange(0.45, 0.55 + 1e-9, 0.02)      # 0.45,0.47,…0.55
        offsets   = np.arange(1.0, 6.0 + 1e-9, 1.0)       # 1,…,6
        for p in exponents:
            for off in offsets:
                if time.time() > deadline:
                    break
                idx = np.arange(n, dtype=np.float64)
                a = (idx + off) ** (-p)                # decreasing power law
                a = a / a.sum()
                val = compute_objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[grid] n={n} p={p:.3f} off={off:.1f} val={val:.6f}")

        # a symmetric version – mirror the decreasing shape
        for p in exponents:
            for off in offsets:
                if time.time() > deadline:
                    break
                idx = np.arange(n, dtype=np.float64)
                left = (idx + off) ** (-p)
                right = left[::-1]
                a = left + right
                a = a / a.sum()
                val = compute_objective(a)
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[grid‑sym] n={n} p={p:.3f} off={off:.1f} val={val:.6f}")

        # --------------------------------------------------------------
        # 2) Adam refinement (multiple restarts)
        # --------------------------------------------------------------
        restarts = 4
        for r in range(restarts):
            if time.time() > deadline:
                break
            # choose a starting point
            if best_seq is not None and len(best_seq) == n:
                a0 = best_seq.copy()
            else:
                a0 = np.random.dirichlet(np.ones(n))

            # allocate a slice of the remaining budget for this Adam run
            slice_time = min((deadline - time.time()) * 0.15, 30.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = run_adam_opt(n, a0, sub_deadline,
                                          max_iters=300_000)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Adam n={n} restart={r}] GLOBAL best improved to {best_val:.6f}")

        # --------------------------------------------------------------
        # 3) Differential Evolution (global search)
        # --------------------------------------------------------------
        if time.time() < deadline:
            de_time = min((deadline - time.time()) * 0.35, 120.0)
            sub_deadline = time.time() + de_time
            # warm start population – the current best and a few random draws
            init_pop = [best_seq] if (best_seq is not None and len(best_seq) == n) else []
            while len(init_pop) < 5:
                init_pop.append(np.random.dirichlet(np.ones(n)))
            a_de, val_de = run_de_opt(n, init_pop, sub_deadline,
                                      popsize=12, max_gens=25_000)
            if val_de < best_val:
                best_val = val_de
                best_seq = a_de.copy()
                print(f"[DE n={n}] GLOBAL best now {best_val:.6f}")

        # --------------------------------------------------------------
        # 4) (1+1) ES fine‑tuning
        # --------------------------------------------------------------
        if time.time() < deadline and best_seq is not None and len(best_seq) == n:
            es_time = min((deadline - time.time()) * 0.30, 60.0)
            sub_deadline = time.time() + es_time
            a_es, val_es = run_es_opt(n, best_seq, sub_deadline,
                                      max_iter=40_000)
            if val_es < best_val:
                best_val = val_es
                best_seq = a_es.copy()
                print(f"[ES n={n}] GLOBAL best now {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – if nothing was found (should not happen) return uniform.
    # ------------------------------------------------------------------
    if best_seq is None:
        fallback_n = 1024
        best_seq = np.full(fallback_n, 1.0 / fallback_n, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\nBest value discovered: {best_val:.6f} (length {len(best_seq)})")
    # convert to plain Python list as required by the external evaluator
    return best_seq.tolist()
