"""
Hybrid annealed soft‑max Adam optimizer with diversified initialisations
including a novel “U‑shaped” start.  The routine runs for at most
``budget_s`` seconds, repeatedly runs short Adam runs with a smooth‑max
surrogate of the true objective
    F(a)=2·n·max(a*a)/(∑a)²,
gradually increases the smoothing parameter λ and decays the learning‑rate.
Whenever progress stalls a small random perturbation of the logits is added.
After the main phase a fine‑tuning phase with a higher λ and a smaller
learning‑rate is executed.  The best sequence found is returned as a plain
list of floats.
"""

import time
import numpy as np

def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT of given length."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv

def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Weights wₖ = exp(lam·(convₖ‑max(conv))) / Σⱼ … ."""
    m = conv.max()
    # for very large λ the exponent can under‑flow → fall back to hard max
    if lam > 5.0:
        # exact max (or a tiny tolerance for numerical noise)
        eps = 1e-12
        idx = np.where(conv >= m - eps)[0]
        w = np.zeros_like(conv)
        w[idx] = 1.0 / len(idx)
        return w
    w = np.exp(lam * (conv - m))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w

def _softmax_gradient(a: np.ndarray, w: np.ndarray, size_corr: int) -> np.ndarray:
    """
    Gradient of the smooth‑max surrogate w.r.t. the original probabilities a.
    Closed form: ∇ = 2·(w ★ a) where ★ denotes cross‑correlation.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, size_corr)
    frev = np.fft.rfft(rev_a, size_corr)
    corr = np.fft.irfft(fw * frev, size_corr)
    # slice that corresponds to indices 0 … n‑1 of the original vector
    n = a.shape[0]
    grad = 2.0 * corr[n - 1 : n - 1 + n]
    return grad

def _local_refine(a: np.ndarray, time_budget: float) -> tuple[np.ndarray, float]:
    """
    Very cheap coordinate‑wise hill‑climbing that preserves the ℓ₁‑norm.
    Randomly transfers a tiny amount of mass between two coordinates.
    """
    start = time.time()

    # pre‑compute FFT size for the fast evaluation of the objective
    fft_sz = 1 << int(np.ceil(np.log2(2 * n - 1)))
    def obj(x):
        conv = _full_convolution(x, fft_sz)
        return float(2.0 * n * conv.max())
    best_val = obj(a)
    best_a = a.copy()
    while time.time() - start < time_budget:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # random delta, keep both entries non‑negative
        delta = (np.random.rand() - 0.5) * 1e-4

            continue
        cand = best_a.copy()
        cand[i] += delta
        cand[j] -= delta
        val = obj(cand)
        if val < best_val:
            best_val = val
            best_a = cand
    return best_a, best_val

Hybrid annealed soft‑max Adam optimizer with diversified initialisations.

The routine searches for a non‑negative step function (list of heights) that
minimises ``evaluate_sequence`` within a user‑supplied time budget.  It works
as follows:

*   A soft‑max parametrisation (logits ``u``) guarantees non‑negativity and
    Σa = 1.
*   Several families of random starts are tried; one novel family is a
    “U‑shaped” distribution (high at the ends, low in the middle).
*   While the optimisation runs, a smooth‑max surrogate of the true
    objective  F(a)=2·n·max(a∗a) is used.  The surrogate’s temperature λ
    starts small (smooth) and is increased slowly (continuation) so that the
    optimiser first sees a friendly landscape and gradually focuses on the
    true maximum.
*   Adam with bias‑correction updates the logits.  If progress stalls for a
    few seconds the logits receive a tiny random shake and the Adam moments
    are reset (helps escape plateaus).
*   After the main phase a fine‑tuning stage starts from the best sequence
    found so far, with a larger λ and a smaller learning‑rate.
*   Finally, a cheap coordinate‑wise hill‑climbing refinement is performed
    if time permits.
*   The best sequence (plain Python list) is returned.
"""

import time
import numpy as np


def _full_convolution(a: np.ndarray, size: int) -> np.ndarray:
    """Linear self‑convolution a*a using an FFT of length ``size``."""
    fa = np.fft.rfft(a, size)
    conv = np.fft.irfft(fa * fa, size)[: 2 * a.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, lam: float) -> np.ndarray:
    """Weights wₖ = exp(lam·(convₖ‑max(conv))) / Σⱼ … .
    For large λ the exact max (hard‑max) is used."""
    max_val = conv.max()
    if lam > 5.0:                                   # hard max – avoid under‑flow
        eps = 1e-12
        idx = np.where(conv >= max_val - eps)[0]
        w = np.zeros_like(conv)
        w[idx] = 1.0 / len(idx)
        return w
    w = np.exp(lam * (conv - max_val))
    s = w.sum()
    if s == 0.0:
        w.fill(1.0 / w.size)
    else:
        w /= s
    return w
