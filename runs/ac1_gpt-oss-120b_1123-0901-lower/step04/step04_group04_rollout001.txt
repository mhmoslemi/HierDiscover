def project_simplex(v):
    # v: array, we want w >= 0, sum w = 1.
    # Standard algorithm using sorting.
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Propose a non‑negative step function that (approximately) minimises
``evaluate_sequence``.

Algorithm
---------
* Work with the probability vector p = a / Σa, so Σp = 1 and a = S·p
  with S = √(2·n) (the required L1‑norm).
* Phase 1 – smooth optimisation:
  minimise the smooth maximum
       Lτ(p) = τ·log Σ exp((p∗p)_k/τ)
  with L‑BFGS‑B (variables y = log p) for a decreasing temperature τ.
  Several diverse initialisations (uniform, Dirichlet, perturbed
  known solution, sparse random) are tried; the best result is kept.
* Phase 2 – exact sub‑gradient refinement:
  starting from the best p, repeatedly perform a projected sub‑gradient
  step on the true objective  max_k (p∗p)_k.
  The sub‑gradient for the maximal index k is
        g_i = 2·p_{k‑i}   (out‑of‑range entries are zero).
  After each step p is projected onto the probability simplex.
  A diminishing step size α_t = 0.3 / √t is used.
* The best sequence a = S·p found during the whole run is returned.

Both phases use FFT‑based convolutions (O(n log n)).  The code respects
the 1000 s budget, prints progress, and works with any length n
(e.g. the supplied ``height_sequence_1``).  No external files are read.
"""

import time
import math
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Optional SciPy import for L‑BFGS‑B.
try:
    from scipy.optimize import minimize
    _SCIPY_AVAILABLE = True
except Exception:   # pragma: no cover
    _SCIPY_AVAILABLE = False


def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values.
    Returns weight vector w (summing to 1) and the smooth max Lτ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full circular convolution p∗p (real FFT based). Returns length 2·n‑1."""
    A = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A * A, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·a_{k‑i} using FFT.
    ``a`` and ``w`` are real vectors; the result has the same length as ``a``.
    """
    # FFT of a
    A = np.fft.rfft(a, n=Nfft)
    # FFT of reversed weight
    W_rev = np.fft.rfft(w[::-1], n=Nfft)
    # convolution of w_rev with a
    conv_rev = np.fft.irfft(W_rev * A, n=Nfft)
    # keep the valid part (length len(w)+len(a)-1)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    # the required slice (see original implementation)
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Compute smooth max Lτ and its gradient w.r.t. y (log‑softmax parameters).
    """
    # softmax → probability vector p
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)               # numerically stable
    p = exp_y / exp_y.sum()

    # full convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max and weights
    w, Ltau = _softmax_weights(conv, tau)

    # gradient of Lτ wrt p : 2·rev
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # convert to gradient wrt y (soft‑max Jacobian)
    dot = np.dot(p, grad_p)                 # Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float,
                best_seq: np.ndarray, best_score: float) -> (np.ndarray, float):
    """
    Repeatedly minimise the smooth maximum with decreasing temperature.
    Returns the best (sequence, score) seen during the run.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.85            # slower cooling than the baseline
    min_tau = 1e-12
    n = y.shape[0]

    while time.time() < deadline and tau > min_tau:
        # ----- L‑BFGS on the current temperature -----------------------
        if _SCIPY_AVAILABLE:
            def fun(y_vec):
                L, g = _objective_and_grad(y_vec, tau, Nfft)
                return L, g

            opts = {'maxfun': 500, 'maxiter': 500, 'ftol': 1e-12, 'gtol': 1e-12}
            res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
            y = res.x
        else:
            # Very simple gradient descent fallback
            for _ in range(200):
                L, g = _objective_and_grad(y, tau, Nfft)
                y -= 0.1 * g

        # ----- evaluate the exact objective (no smoothing) -------------
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        # exact evaluate = 2·n·max_conv (because target_sum² = 2·n)
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve L‑BFGS] τ={tau:.2e}  score={best_score:.6f}")

        # ----- schedule ------------------------------------------------
        tau *= tau_decay

        # occasional jitter to avoid shallow plateaus
        if np.random.rand() < 0.12:
            y += np.random.randn(n) * 0.02

        if time.time() >= deadline:
            break

    return best_seq, best_score


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex:
    {w | w_i ≥ 0, Σ w_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform simplex
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


def _subgradient_refine(p: np.ndarray, target_sum: float, Nfft: int,
                        deadline: float,
                        best_seq: np.ndarray, best_score: float) -> (np.ndarray, float):
    """
    Projected sub‑gradient descent on the true objective
        f(p) = max_k (p∗p)_k .
    """
    n = p.shape[0]
    it = 0
    while time.time() < deadline:
        conv = _conv_full(p, Nfft)
        max_conv = conv.max()
        cur_score = 2.0 * n * max_conv
        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[subgrad improve] iter={it}  score={best_score:.6f}")

        # index where the maximum is attained (first one)
        k = int(conv.argmax())

        # sub‑gradient g_i = 2·p_{k−i} (zero outside the support)
        grad = np.zeros_like(p)
        i_start = max(0, k - (n - 1))
        i_end = min(n, k + 1)          # exclusive
        # vectorised fill
        idx = np.arange(i_start, i_end, dtype=int)
        grad[idx] = 2.0 * p[k - idx]

        # diminishing step size
        alpha = 0.30 / math.sqrt(it + 1.0)

        # gradient step + simplex projection
        p = _project_simplex(p - alpha * grad)

        # occasional random jitter to shake out of plateaus
        if it % 80 == 0 and np.random.rand() < 0.25:
            p += np.random.randn(n) * 0.001
            p = _project_simplex(p)

        it += 1

    return best_seq, best_score
