            break
        x = res.x
        sum_x = x.sum()
        if sum_x <= 0:
            break
        # Compute g scaled to target sum
        g = x * (target_sum / sum_x)

        # blending factor t: start larger, then decay
        if iter_num < 5:
            t = 0.5
        else:
            t = 0.2
        a = (1 - t) * a + t * g
        # renormalize (just in case)
        a = a * (target_sum / a.sum())

        val = compute_objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve_time = time.time()
        iter_num += 1
        # stop if no improvement for certain time
        if time.time() - last_improve_time > 10.0:
            break
    return best_a, best_val

"""
Step‑function optimizer for the evaluation function

    F(a) = 2·n·max(a∗a) / (sum(a)²) ,   aᵢ ≥ 0

The algorithm works in three stages:

1. **Iterated linear‑programming (LP) refinement** –  
   For a fixed length *n* we normalise the candidate so that
   Σa = √(2n).  Let M = max(a∗a).  The LP

        maximise   Σ gⱼ
        subject to  (a * g)ₖ ≤ M   for all convolution indices k
                    gⱼ ≥ 0

 finds the most massive vector *g* that still respects the bound M.
   After scaling *g* back to the required sum we blend it with the
   current candidate (large step early, smaller later).  Repeating this
   loop monotonically reduces the objective.

2. **Multiple random restarts** –  uniform, linearly decreasing,
   Dirichlet‑sampled and (if available) the historic `height_sequence_1`
   give diverse starting points.

3. **Fine‑tuning with Adam on a sub‑gradient of the max‑convolution** –  
   The best vector from the LP phase is polished by a few Adam steps
   on the exact sub‑gradient of the `max(a∗a)` term (soft‑max parametrisation).

The whole search respects the supplied time budget (default 1000 s);
any remaining time at the end is used for the final Adam polishing.
The returned list contains only non‑negative numbers and can be fed
directly to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using FFT (real output)."""
    n = a.shape[0]
    L = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, L)
    conv = np.fft.irfft(fa * fa, L)[: 2 * n - 1]
    return conv.real


def compute_objective(a: np.ndarray) -> float:
    """Exact objective: 2·n·max(a∗a) / (sum(a)²)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * a.shape[0] * full_convolution(a).max() / (s * s)


# ----------------------------------------------------------------------
# Sub‑gradient of max‑convolution (used by Adam)
# ----------------------------------------------------------------------
def grad_submax(a: np.ndarray):
    """
    Sub‑gradient of max(a∗a) w.r.t. a.
    Returns (grad, current max value).  The gradient obeys
        ∂max/∂aⱼ = 2·a_{k*‑j}
    where k* is a convolution index attaining the maximum.
    """
    n = a.shape[0]
    conv = full_convolution(a)
    max_c = conv.max()
    # pick a random maximiser to avoid bias
    idxs = np.flatnonzero(np.isclose(conv, max_c, atol=1e-14))
    k = np.random.choice(idxs)

    g = np.zeros_like(a)
    # indices i where a_i participates in (a*a)_k
    start = max(0, k - (n - 1))
    end = min(k, n - 1)
    i = np.arange(start, end + 1, dtype=int)
    j = k - i
    g[i] = 2.0 * a[j]
    return g, max_c


def softmax(x: np.ndarray) -> np.ndarray:
    """Numerically stable softmax."""
    mx = np.max(x)
    e = np.exp(x - mx)
    return e / e.sum()


# ----------------------------------------------------------------------
# Adam optimiser on the sub‑gradient (used for final polishing)
# ----------------------------------------------------------------------
def run_adam_opt(n: int, init_a: np.ndarray, deadline: float,
                 max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the sub‑gradient of max(a∗a) while keeping a on the simplex.
    Returns (best_a, best_value) discovered before *deadline*.
    """
    # initialise logits from the probability vector
    x = np.log(init_a + 1e-15)

    m = np.zeros(n)
    v = np.zeros(n)
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    lr = 0.05

    best_val = np.inf
    best_a = None
    t = 0
    last_improve = time.time()

    while time.time() < deadline and t < max_iters:
        a = softmax(x)                     # a ≥0, Σa = 1
        grad_a, max_c = grad_submax(a)     # exact sub‑gradient
        val = 2.0 * n * max_c               # exact objective (sum(a)=1)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve = time.time()
            # occasional progress print – cheap
            print(f"[Adam n={n} it={t}] new best: {val:.6f}")

        # back‑propagation through softmax
        avg = np.dot(a, grad_a)
        grad_x = a * (grad_a - avg)

        # Adam update
        t += 1
        m = beta1 * m + (1.0 - beta1) * grad_x
        v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)
        x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

        # keep logits bounded
        x = np.clip(x, -50.0, 50.0)

        # shake when stagnating
        if time.time() - last_improve > 5.0:
            x += (np.random.rand(n) - 0.5) * 1e-2
            last_improve = time.time()
            print(f"[Adam n={n} it={t}] shake (stagnation)")

    if best_a is None:                      # very unlikely fallback
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = compute_objective(best_a)
    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------
def lp_optimize(n: int, init_a: np.ndarray, deadline: float) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a candidate by solving the LP
        maximise Σ gⱼ  s.t. (a * g)ₖ ≤ M, g ≥ 0,
    where M = max(a*a).  After each LP we blend the solution back into a.
    Returns (best_a, best_val) discovered before *deadline*.
    """
    # Normalise to Σa = √(2n) – this makes the objective equal to max(a*a)
    target_sum = np.sqrt(2.0 * n)
    a = np.maximum(init_a, 0.0).astype(np.float64)
    s = a.sum()
    if s > 0.0:
        a = a * (target_sum / s)
    else:
        a = np.full(n, target_sum / n)

    best_a = a.copy()
    best_val = compute_objective(a)

    it = 0
    last_improve_time = time.time()

    while time.time() < deadline:
        # current bound M
        conv = full_convolution(a)
        M = conv.max()

        # build the convolution matrix A (2n‑1)×n : (A·g)_k = (a*g)_k
        A = np.zeros((2 * n - 1, n), dtype=np.float64)
        for j in range(n):
            A[j:j + n, j] = a

        # Linear programme: maximise Σ g ⟹ minimise –1ᵀg
        c = -np.ones(n, dtype=np.float64)
        b = np.full(2 * n - 1, M, dtype=np.float64)
        bounds = [(0.0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                      method='highs', options={'presolve': True, 'dual_feasibility_tolerance': 1e-9})
        if not res.success:
            # LP failed – abort this run
            break

        g = res.x
        sum_g = g.sum()
        if sum_g <= 0.0:
            break

        # scale g back to the required total mass
        g = g * (target_sum / sum_g)

        # blending factor – aggressive at start, then gentle
        t = 0.5 if it < 5 else 0.2
        a = (1.0 - t) * a + t * g

        # re‑normalise (numerical drift)
        a = a * (target_sum / a.sum())

        val = compute_objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            last_improve_time = time.time()
            print(f"[LP n={n} iter={it}] new best: {val:.6f}")

        it += 1

        # stop if we have stalled for a while
        if time.time() - last_improve_time > 12.0:
            break

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs) -> list[float]:
    """
    Search for a non‑negative step‑function that minimises ``evaluate_sequence``.
    The routine combines:
      • several random / heuristic initialisations,
      • an LP‑based refinement loop,
      • a short Adam fine‑tuning.
    It respects the supplied time budget and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + max(0.0, budget_s - 0.5)   # safety margin

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1️⃣ Choose problem sizes – 1024 is a sweet spot for speed / quality.
    # ------------------------------------------------------------------
    lengths = [1024, 2048]   # try larger size if time permits

    for n in lengths:
        if time.time() >= deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # ------------------------------------------------------------------
        # 2️⃣ Build a small pool of sensible initial candidates.
        # ------------------------------------------------------------------
        init_candidates = []

        # uniform
        init_candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

        # linearly decreasing (positive)
        dec = np.arange(n, 0, -1, dtype=np.float64)
        init_candidates.append(dec / dec.sum())

        # random Dirichlet
        init_candidates.append(np.random.dirichlet(np.ones(n)))

        # warm‑start from the currently best sequence (if same length)
        if best_seq is not None and len(best_seq) == n:
            init_candidates.append(np.asarray(best_seq, dtype=np.float64))

        # historic construction, if the notebook provided it
        if 'height_sequence_1' in globals():
            hs = np.asarray(height_sequence_1, dtype=np.float64)
            if hs.size >= n:
                hs = hs[:n]
            else:
                hs = np.pad(hs, (0, n - hs.size))
            hs = np.maximum(hs, 0.0)
            if hs.sum() > 0.0:
                hs = hs / hs.sum()
            else:
                hs = np.random.dirichlet(np.ones(n))
            init_candidates.append(hs)

        # ------------------------------------------------------------------
        # 3️⃣ Run LP‑based refinement for each candidate (time‑sliced).
        # ------------------------------------------------------------------
        for cid, cand in enumerate(init_candidates):
            if time.time() >= deadline:
                break
            # allocate a modest slice of remaining time for this candidate
            slice_time = min((deadline - time.time()) * 0.4, 180.0)
            sub_deadline = time.time() + slice_time
            a_opt, val_opt = lp_optimize(n, cand, sub_deadline)
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[n={n} cand={cid}] GLOBAL best improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final fine‑tuning with Adam (if any time left).
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline:
        remaining = deadline - time.time()
        # Adam expects a probability vector (sum = 1)
        a_norm = best_seq / best_seq.sum()
        adam_deadline = time.time() + min(remaining, 40.0)
        a_ad, val_ad = run_adam_opt(len(best_seq), a_norm, adam_deadline)
        # scale back to Σ = √(2n) so that the objective matches our definition
        a_ad_scaled = a_ad * np.sqrt(2.0 * len(best_seq))
        val_scaled = compute_objective(a_ad_scaled)
        if val_scaled < best_val:
            best_val = val_scaled
            best_seq = a_ad_scaled.copy()
            print(f"[Adam fine‑tune] new global best: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Fallback – uniform vector if nothing was discovered.
    # ------------------------------------------------------------------
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = compute_objective(best_seq)

    print(f"\n*** Final best value: {best_val:.6f} (length {len(best_seq)})")
    # Convert to plain Python list as required by the external evaluator.
    return best_seq.tolist()
