def exp_gradient_refine(p0, Nfft, target_sum, deadline, best_seq, best_score):
    p = p0.copy()
    eta0 = 0.2
    iteration = 0
    tol = 1e-12
    while time.time() < deadline:
        conv = _conv_full(p, Nfft)
        max_val = conv.max()
        # Determine active set
        active = np.where(conv >= max_val - 1e-12)[0]  # tolerance
        w = np.zeros_like(conv)
        if len(active) > 0:
            w[active] = 1.0 / len(active)   # uniform weights among maxima
        rev = _weighted_rev(p, w, Nfft)
        grad = 2.0 * rev
        # Learning rate schedule
        eta = eta0 / np.sqrt(iteration + 1)
        # Exponentiated update
        p = p * np.exp(-eta * grad)
        # Normalize
        total = p.sum()
        if total == 0:
            p = np.full_like(p, 1.0 / p.shape[0])
        else:
            p /= total
        # Occasionally evaluate exact
        if iteration % 100 == 0 or time.time() > deadline - 1.0:
            seq = (p * target_sum).tolist()
            cur_score = evaluate_sequence(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = np.array(seq)
                print(f"[expgrad improve] iter={iteration} score={best_score:.6f}")
        iteration += 1
    return best_seq, best_score
```

"""
Propose a non‑negative step function (vector) that minimises the
``evaluate_sequence`` objective.

Algorithm
---------
1. Load the supplied construction ``height_sequence_1`` (fallback to a
   uniform vector of length 2100).  Normalise it so that
   ``sum(a)=sqrt(2·n)`` – with this normalisation the objective equals the
   maximum value of the self‑convolution of the corresponding probability
   vector ``p = a / sum(a)``.

2. Several *starting points* are generated:
   * the known construction,
   * a uniform vector,
   * a dense Dirichlet sample,
   * a few sparse Dirichlet samples (few non‑zero entries).

3. For each start a *smooth‑max* L‑BFGS optimisation is performed.
   The smooth objective is the log‑sum‑exp approximation of the
   max‑convolution, with a temperature ``τ`` that is decreased
   gradually (factor 0.8).  The optimisation is carried out on the
   unconstrained log‑probability parameters ``y`` (soft‑max guarantees
   non‑negativity and unit sum).  More L‑BFGS iterations and a slower
   temperature schedule than the reference code are used.

4. After the L‑BFGS phase a large number of additional random restarts
   (≈30) are tried while time permits, using the same optimisation pipeline.

5. Finally a *mirror‑descent* (exponentiated‑gradient) refinement is run
   on the best solution found.  This uses the exact hard max
   (sub‑gradient) of the convolution and a diminishing step size.

6. The best sequence encountered (according to the exact
   ``evaluate_sequence`` call) is returned.

All helper functions are defined at top level; the search respects the
time budget (default 1000 s) and prints progress information.
"""

import time

import numpy as np
from typing import List, Tuple

# Optional SciPy import (used for L‑BFGS‑B)

# FFT‑based convolution utilities

def _conv_full(p: np.ndarray, Nfft: int) -> np.ndarray:
    """Full convolution p * p using real FFT. Returns array of length 2·n‑1."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _softmax_weights(conv: np.ndarray, tau: float) -> Tuple[np.ndarray, float]:
    """Stable soft‑max of convolution entries.
    Returns weight vector w (summing to 1) and the smoothed max L_τ."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(p: np.ndarray, w: np.ndarray, Nfft: int) -> np.ndarray:
    """
    Compute rev_i = Σ_k w_k·p_{k‑i}
    (reverse‑weighted convolution).  ``w`` is the weight vector of the
    smooth‑max and has length 2·n‑1.
    """
    # FFT of p (zero‑padded)
    A_fft = np.fft.rfft(p, n=Nfft)
    # FFT of reversed weight vector (also zero‑padded)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    # Convolution of w_rev with p
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + p.shape[0] - 1]  # keep valid part
    # Extract the reversed part needed for the gradient
    idx_start = 2 * p.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(p.shape[0])]
    return rev


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int) -> Tuple[float, np.ndarray]:
    """
    Smooth‑max objective L_τ(p) and its gradient w.r.t. the log‑probabilities y.
    ``p = softmax(y)``.
    """
    # soft‑max
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)
    p = exp_y / exp_y.sum()

    # convolution p∗p
    conv = _conv_full(p, Nfft)

    # smooth max and weights
    w, Ltau = _softmax_weights(conv, tau)

    # gradient w.r.t. p
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # push gradient through soft‑max (Jacobian)
    dot = np.dot(p, grad_p)                # scalar Σ_i p_i·grad_p_i
    grad_y = p * (grad_p - dot)

    return Ltau, grad_y

def _random_logit_start(n: int, sparsity: int | None = None) -> np.ndarray:
    """
    Generate a random logit vector ``y`` of length ``n``.
    If ``sparsity`` is given, only that many components receive a
    non‑negligible mass (the rest are set to a large negative constant).
    """
    eps = 1e-12

def _run_l_bfgs(
    y0: np.ndarray,
    init_tau: float,
    target_sum: float,
    Nfft: int,
    deadline: float,
    best_seq: np.ndarray,
    best_score: float,

    """
    Repeated L‑BFGS optimisation while decreasing the temperature τ.
    Returns the best sequence (as a raw ``np.ndarray`` of heights) and its
    (exact) score.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.80
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        def fun(y_vec: np.ndarray) -> Tuple[float, np.ndarray]:
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # allow more function evaluations than the reference implementation

        if _SCIPY_AVAILABLE:

            y = res.x
        else:

            L, g = _objective_and_grad(y, tau, Nfft)
            step = 0.1
            y = y - step * g

        y_max = np.max(y)

            cur_score = evaluate_sequence(seq)
            if cur_score < best_score:
                best_score = cur_score
                best_seq = np.array(seq)
                print(f"[refine improve] iter={iteration} score={best_score:.6f}")

        iteration += 1
        if time.time() >= deadline:
            break

    return best_seq, best_score

# Main entry point

def propose_candidate(seed: int = 42, budget_s: float = 1000, **kwargs) -> List[float]:
    """
    Search for a step‑function (non‑negative vector) that minimises
    ``evaluate_sequence`` within the time budget.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # Load the supplied construction (if any), otherwise fall back to uniform.
    try:
        base = np.array(height_sequence_1, dtype=float)  # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)            # normalisation used throughout
    Nfft = 1 << ((2 * n - 1).bit_length())     # next power‑of‑two ≥ 2·n‑1

    # normalise the base vector to the required L1‑norm
    base = base * (target_sum / base.sum())
    p0 = base / target_sum                      # probability vector (sum = 1)
    eps_log = 1e-12
    y0 = np.log(p0 + eps_log)                  # log‑probabilities for optimisation

    best_seq = base.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Assemble a modest collection of diverse starting points
    starts = [
        y0,                                    # known construction
        np.zeros(n),                          # uniform (logits = 0)
        _random_logit_start(n),               # dense Dirichlet sample
    ]

    # sparse starts with various support sizes
    for sp in (30, 80, 200):
        starts.append(_random_logit_start(n, sparsity=sp))

    # jittered version of the known solution
    starts.append(y0 + np.random.randn(n) * 0.05)

    # ------------------------------------------------------------------
    # Run L‑BFGS from each start while respecting the time limit
    for idx, y_start in enumerate(starts):
        if time.time() >= deadline:
            break
        print(f"[lbfgs] start {idx + 1}/{len(starts)}")
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Additional random restarts (more aggressive exploration)
    extra_starts = 30
    for i in range(extra_starts):
        if time.time() >= deadline:
            break
        # random sparsity between 20 and 400 non‑zero entries
        spars = np.random.randint(20, 401)
        y_start = _random_logit_start(n, sparsity=spars)
        if np.random.rand() < 0.3:
            y_start += np.random.randn(n) * 0.02
        p_start = np.exp(y_start - np.max(y_start))
        p_start /= p_start.sum()
        conv0 = _conv_full(p_start, Nfft)
        init_tau = max(conv0.max() * 2.0, 1e-12)

        best_seq, best_score = _run_l_bfgs(
            y0=y_start,
            init_tau=init_tau,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # Final mirror‑descent refinement on the best solution so far
    if time.time() < deadline - 5.0:
        print("[refine] mirror‑descent")
        p_best = best_seq / target_sum
        best_seq, best_score = _mirror_descent(
            p0=p_best,
            target_sum=target_sum,
            Nfft=Nfft,
            deadline=deadline,
            best_seq=best_seq,
            best_score=best_score,
        )

    # ------------------------------------------------------------------
    # One final sanity check with the exact evaluator
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, length = {best_seq.shape[0]}")
    # Return as a plain Python list (as required by the evaluator)
    return best_seq.tolist()
