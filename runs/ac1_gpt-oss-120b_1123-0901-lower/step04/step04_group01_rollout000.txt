def cross_correlation(w, a, fft_len):
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start+n]

"""
Hybrid smooth‑max Adam optimizer for the step‑function problem.

The algorithm builds a probability vector a via a softmax over free logits x,
so that a_i ≥ 0 and Σ a_i = 1 (the objective is homogeneous, this loss of
generality is harmless).  At each iteration we compute the linear self‑
convolution c = a*a.  Instead of the non‑differentiable max we use a smooth
approximation
        f_β(c) = Σ_i w_i c_i ,   w_i ∝ exp(β·c_i) ,
where the temperature β is slowly increased – a classic “soft‑max
annealing”.  The gradient of f_β w.r.t. a is 2·(w ★ a).  After back‑propagating
through the softmax we update the logits with Adam.  Occasionally we shake
the logits to escape stagnation and after the main phase we perform a cheap
greedy local search (pairwise mass transfers).  For larger problem sizes we
warm‑start from the best solution found at the previous size (zero‑padding).

The routine respects the 1000 s budget, prints progress, and returns the
best sequence it discovered as a plain Python list.
"""

import time
import numpy as np


def full_convolution(a: np.ndarray, fft_len: int = None) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    n = a.shape[0]
    if fft_len is None:
        fft_len = 1 << int(np.ceil(np.log2(2 * n - 1)))
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    return conv


def cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Returns vector of length n where
        (w ★ a)[i] = Σ_k w[k]·a[i + k - (n-1)] .
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def propose_candidate(seed: int = 0, budget_s: float = 1000.0, **kwargs) -> list:
    """
    Search for a non‑negative step‑function that minimises
        F(a) = 2·n·max(a*a) / (sum(a)²)

    The optimisation proceeds over increasing lengths, uses a smooth‑max
    Adam descent on the simplex (softmax parametrisation), temperature
    annealing, occasional random shaking, and a final greedy pairwise
    refinement.  Returns the best sequence found (as a list of floats).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    best_seq = None          # NumPy array containing the best vector found
    best_val = float("inf")  # corresponding objective value

    # Problem sizes – we start small and grow, re‑using the previous solution.
    lengths = [512, 1024, 2048, 4096, 8192, 16384]

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n=== Optimising length n={n} ===")

        # FFT lengths reused many times (powers of two for speed)
        fft_len_conv = 1 << int(np.ceil(np.log2(2 * n - 1)))
        fft_len_corr = 1 << int(np.ceil(np.log2(3 * n)))  # ≥ (2n‑1)+n‑1

        # Number of restarts – modest because we already have good warm‑starts
        restarts = 3 if n <= 4096 else 2

        for r in range(restarts):
            if time.time() > deadline:
                break

            # ---- initialise logits ------------------------------------------------
            if best_seq is not None and len(best_seq) == n:
                # perfect warm start
                init_a = best_seq.copy()
                print(f"[n={n} restart={r}] warm start (same length)")
            elif best_seq is not None:
                # zero‑pad previous best to the new length
                old_n = len(best_seq)
                pad_total = n - old_n
                pad_left = pad_total // 2
                pad_right = pad_total - pad_left
                init_a = np.pad(best_seq, (pad_left, pad_right), mode='constant')
                init_a = np.maximum(init_a, 0.0)
                init_a_sum = init_a.sum()
                if init_a_sum < 0.01:                     # degenerate padding
                    init_a = np.random.dirichlet(np.ones(n))
                else:
                    init_a /= init_a_sum
                print(f"[n={n} restart={r}] warm start (padded from {old_n})")
            elif 'height_sequence_1' in globals():
                cand = np.array(height_sequence_1, dtype=np.float64)
                if cand.shape[0] >= n:
                    cand = cand[:n]
                else:
                    cand = np.pad(cand, (0, n - cand.shape[0]), mode='constant')
                cand = np.maximum(cand, 0.0)
                if cand.sum() < 0.01:
                    cand = np.random.dirichlet(np.ones(n))
                else:
                    cand /= cand.sum()
                init_a = cand
                print(f"[n={n} restart={r}] start from height_sequence_1")
            else:
                init_a = np.random.dirichlet(np.ones(n))
                print(f"[n={n} restart={r}] random start")

            # logits for softmax parametrisation
            x = np.log(init_a + 1e-15)

            # ---- Adam state -------------------------------------------------------
            m = np.zeros(n, dtype=np.float64)
            v = np.zeros(n, dtype=np.float64)
            beta1, beta2, eps = 0.9, 0.999, 1e-8
            lr = 0.2                     # initial learning rate
            decay_every = 5000           # learning‑rate decay interval

            # smooth‑max temperature schedule
            beta_temp = 0.5              # start very smooth
            beta_factor = 1.02           # increase each decay step

            # iteration budget scaling with size
            if n <= 1024:
                max_iters = 40000
            elif n <= 4096:
                max_iters = 30000
            elif n <= 8192:
                max_iters = 20000
            else:
                max_iters = 15000

            t = 0
            last_improve = time.time()

            # ---- main optimisation loop -----------------------------------------
            while t < max_iters and time.time() < deadline:
                # soft‑max → probability vector (∑a = 1)
                max_x = np.max(x)
                exp_x = np.exp(x - max_x)
                a = exp_x / exp_x.sum()

                # linear self‑convolution
                conv = full_convolution(a, fft_len=fft_len_conv)

                # smooth max weights w_i ∝ exp(β·conv_i)
                scaled = beta_temp * conv
                max_scaled = np.max(scaled)
                exp_scaled = np.exp(scaled - max_scaled)       # stabilise
                w = exp_scaled / exp_scaled.sum()

                # gradient of smooth objective w.r.t. a
                grad_a = 2.0 * cross_correlation(w, a, fft_len=fft_len_corr)

                # back‑prop through softmax
                inner = np.dot(a, grad_a)
                grad_x = a * (grad_a - inner)

                # Adam update
                t += 1
                m = beta1 * m + (1.0 - beta1) * grad_x
                v = beta2 * v + (1.0 - beta2) * (grad_x * grad_x)
                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)
                x = x - lr * m_hat / (np.sqrt(v_hat) + eps)

                # learning‑rate decay & temperature annealing
                if t % decay_every == 0:
                    lr *= 0.7
                    beta_temp = min(beta_temp * beta_factor, 50.0)   # cap β

                # periodic true‑objective check (every 200 iters)
                if t % 200 == 0:
                    val = 2.0 * n * conv.max()   # exact objective (since Σa=1)
                    if val < best_val - 1e-12:
                        best_val = val
                        best_seq = a.copy()
                        last_improve = time.time()
                        print(f"[n={n} r={r} iter={t:05d}] new best {val:.6f}")

                # stagnation handling – small random shake
                if time.time() - last_improve > 30.0:
                    x += (np.random.rand(n) - 0.5) * 1e-2
                    last_improve = time.time()
                    print(f"[n={n} r={r} iter={t}] shake due to stagnation")

            # ---- greedy pairwise refinement ---------------------------------------
            if best_seq is not None and len(best_seq) == n:
                a_local = best_seq.copy()
                # number of cheap local moves – tuned to remaining time
                local_moves = 3000
                for it in range(local_moves):
                    if time.time() > deadline:
                        break
                    i, j = np.random.choice(n, 2, replace=False)
                    # attempt to move a small fraction of mass from i to j
                    move_frac = np.random.rand() * 0.5 + 0.5   # between 0.5‑1.0
                    eps_move = min(a_local[i], 0.02) * move_frac
                    if eps_move <= 0:
                        continue
                    a_new = a_local.copy()
                    a_new[i] -= eps_move
                    a_new[j] += eps_move
                    # compute new objective quickly (Σa=1)
                    conv_new = full_convolution(a_new, fft_len=fft_len_conv)
                    val_new = 2.0 * n * conv_new.max()
                    if val_new < best_val - 1e-12:
                        best_val = val_new
                        best_seq = a_new.copy()
                        a_local = a_new
                        print(f"[local refine] improved to {val_new:.6f}")

    # -------------------------------------------------------------------------
    # Fallback – uniform vector (objective = 2) – should never be used.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)

    print(f"\nFinal best value: {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
