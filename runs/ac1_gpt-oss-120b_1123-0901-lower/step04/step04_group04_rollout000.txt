    best_score = evaluate_sequence(best_seq.tolist())

    # Helper for objective
    def objective_of(p):
        # p sums to 1
        maxconv = np.max(np.convolve(p, p))
        return (target_sum**2) * maxconv

    # Main loop
    iter_count = 0
    while time.time() < deadline:
        iter_count += 1
        # compute current M
        p = f / target_sum  # probability vector
        conv = np.convolve(p, p)
        M = conv.max()
        # build constraint matrix A
        rows = 2 * n - 1

"""
Hybrid Frank‑Wolfe + smooth L‑BFGS optimiser for the max‑convolution
objective used by ``evaluate_sequence``.

The algorithm works with a fixed ℓ₁‑norm (∑a = √(2·n)).  Starting from the
known construction (``height_sequence_1``) or a uniform vector it repeats
the following steps until the time budget is exhausted:

1. Compute M = max_k (f∗f)_k  (full convolution of the current
   sequence ``f`` with itself).
2. Solve the linear programme

        maximise    Σ b_j
        subject to  (f ∗ b)_k ≤ M   for all k = 0 … 2·n‑2
                    b_j ≥ 0

   The LP is linear because ``f`` is fixed; it is solved with
   ``scipy.optimize.linprog`` (HiGHS interior‑point method).
3. Rescale the LP solution ``b`` to satisfy the required ℓ₁‑norm.
4. Perform a cheap line‑search over a few mixing coefficients
   ``t ∈ [0,1]`` and replace ``f`` by the convex combination
   ``(1‑t)·f + t·b`` that yields the smallest exact objective
   (computed via ``evaluate_sequence``).
5. If no improvement is found, inject a small random jitter to
   ``f`` and re‑normalise – this helps escape plateaus.

When the LP step cannot be solved (numerical failure or remaining
time too small) the algorithm falls back to a short continuation of the
smooth‑max L‑BFGS routine from the previous state‑of‑the‑art code.
All operations respect the 1000 s budget; the best sequence found is
returned as a plain ``list[float]``.
"""

import time
import math
import numpy as np
from typing import List

# SciPy is required for the LP solver.  If it is not available we simply
# skip the LP phase and rely on the L‑BFGS fallback.
try:
    from scipy.optimize import linprog, minimize
    _SCIPY_AVAILABLE = True
except Exception:  # pragma: no cover
    _SCIPY_AVAILABLE = False


def _full_conv(v: np.ndarray) -> np.ndarray:
    """Return the full convolution v∗v (length 2·len(v)-1)."""
    return np.convolve(v, v)


def _lp_step(f: np.ndarray, M: float, target_sum: float,
             time_left: float) -> np.ndarray | None:
    """
    Solve the LP   maximise Σ b_j   s.t.   (f∗b)_k ≤ M,   b ≥ 0.
    ``f`` is a one‑dimensional array, ``M`` is a scalar.
    Returns the optimal vector ``b`` scaled to have ℓ₁‑norm = target_sum,
    or ``None`` if the LP could not be solved within the time budget.
    """
    n = f.shape[0]
    rows = 2 * n - 1

    # Build the dense constraint matrix A where
    #   (A @ b)_k = (f ∗ b)_k .
    # Column i of A contains f[i] on rows i … i+n‑1.
    A = np.zeros((rows, n), dtype=float)
    for i in range(n):
        A[i:i + n, i] = f[i]

    # Linear programme: minimise -sum(b)  (i.e. maximise sum(b))
    c = -np.ones(n, dtype=float)

    # Upper‑bound for each convolution entry.
    b_ub = M * np.ones(rows, dtype=float)

    # Set a per‑solve time limit (linear‑solver respects “time_limit”
    # option for HiGHS; we give it the remaining budget).
    options = {"presolve": True, "time_limit": max(0.0, time_left)}

    res = linprog(
        c,
        A_ub=A,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options=options,
    )
    if not res.success:
        return None

    b = res.x
    s = b.sum()
    if s == 0.0:
        return None
    # Rescale to the required ℓ₁‑norm.
    b = b * (target_sum / s)
    return b


def _line_search_mix(f: np.ndarray, g: np.ndarray,
                     target_sum: float, best_score: float,
                     time_deadline: float) -> (np.ndarray, float):
    """
    Search a few mixing coefficients t ∈ (0,1] and return the best
    (f_new, score).  ``best_score`` is the current best objective; if
    none of the candidates improve it, the original ``f`` is returned.
    """
    # Mixing candidates – we include a fine grid near 0 and a few larger steps.
    t_candidates = np.concatenate(
        (np.linspace(0.0, 0.5, 11)[1:], np.linspace(0.5, 1.0, 6)[1:])
    )
    # Ensure we never exceed the global deadline.
    max_calls = len(t_candidates)
    # Evaluate each candidate.
    improved = False
    best_local_f = f
    best_local_score = best_score

    for t in t_candidates:
        if time.time() >= time_deadline:
            break
        f_new = (1.0 - t) * f + t * g
        # Numerical safety – enforce non‑negativity and correct ℓ₁‑norm.
        f_new = np.maximum(f_new, 0.0)
        s = f_new.sum()
        if s == 0.0:
            continue
        f_new = f_new * (target_sum / s)

        cur_score = evaluate_sequence(f_new.tolist())
        if cur_score < best_local_score:
            best_local_score = cur_score
            best_local_f = f_new
            improved = True

    return best_local_f, best_local_score

# Optional L‑BFGS continuation (the procedure used in the previous
# state‑of‑the‑art code).  It is kept as a fallback when the LP approach
# cannot progress.
# ----------------------------------------------------------------------
def _softmax_weights(conv: np.ndarray, tau: float):
    """Stable soft‑max of convolution values; returns (weights, smooth max)."""
    maxc = conv.max()
    if tau <= 0.0:
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        return w, maxc
    ex = np.exp((conv - maxc) / tau)
    sum_ex = ex.sum()
    w = ex / sum_ex
    Ltau = maxc + tau * math.log(sum_ex)
    return w, Ltau


def _weighted_rev(a: np.ndarray, w: np.ndarray, Nfft: int):
    """
    Compute rev_i = Σ_k w_k a_{k‑i} using FFTs.
    """
    A_fft = np.fft.rfft(a, n=Nfft)
    W_rev_fft = np.fft.rfft(w[::-1], n=Nfft)
    conv_rev = np.fft.irfft(W_rev_fft * A_fft, n=Nfft)
    conv_rev = conv_rev[: len(w) + a.shape[0] - 1]
    idx_start = 2 * a.shape[0] - 2
    rev = conv_rev[idx_start - np.arange(a.shape[0])]
    return rev


def _conv_full(p: np.ndarray, Nfft: int):
    """Full convolution p∗p using FFT (length 2·n−1)."""
    A_fft = np.fft.rfft(p, n=Nfft)
    conv = np.fft.irfft(A_fft * A_fft, n=Nfft)[: 2 * p.shape[0] - 1]
    return conv


def _objective_and_grad(y: np.ndarray, tau: float, Nfft: int):
    """
    Returns (smooth max Lτ, gradient w.r.t. y) for the softmax temperature τ.
    """
    y_max = np.max(y)
    exp_y = np.exp(y - y_max)            # stable soft‑max of logits
    p = exp_y / exp_y.sum()

    conv = _conv_full(p, Nfft)
    w, Ltau = _softmax_weights(conv, tau)

    # ∇_p Lτ = 2·rev, where rev is the weighted reverse convolution.
    rev = _weighted_rev(p, w, Nfft)
    grad_p = 2.0 * rev

    # Chain rule: ∂p/∂y = diag(p) - p pᵀ
    dot = np.dot(p, grad_p)
    grad_y = p * (grad_p - dot)
    return Ltau, grad_y


def _run_l_bfgs(y0: np.ndarray, init_tau: float,
                target_sum: float, Nfft: int,
                deadline: float, best_seq: np.ndarray,
                best_score: float) -> (np.ndarray, float):
    """
    Continuation of the smooth‑max L‑BFGS optimisation.  The routine
    reduces the temperature τ geometrically and keeps the best exact
    score encountered.
    """
    y = y0.copy()
    tau = max(init_tau, 1e-12)
    tau_decay = 0.65
    min_tau = 1e-12

    while time.time() < deadline and tau > min_tau:
        def fun(y_vec):
            L, g = _objective_and_grad(y_vec, tau, Nfft)
            return L, g

        # moderate budget for each inner optimisation
        opts = {'maxfun': 250, 'maxiter': 250, 'ftol': 1e-12, 'gtol': 1e-12}
        res = minimize(fun, y, method='L-BFGS-B', jac=True, options=opts)
        y = res.x

        # exact objective
        y_max = np.max(y)
        exp_y = np.exp(y - y_max)
        p = exp_y / exp_y.sum()
        conv = _conv_full(p, Nfft)
        cur_score = (target_sum ** 2) * conv.max()

        if cur_score < best_score:
            best_score = cur_score
            best_seq = (p * target_sum).copy()
            print(f"[improve‑lbfgs] τ={tau:.2e}  score={best_score:.6f}")

        tau *= tau_decay
        if np.random.rand() < 0.15:          # occasional jitter
            y += np.random.randn(y.shape[0]) * 0.02

    return best_seq, best_score

def propose_candidate(seed: int = 42,
                     budget_s: float = 1000,
                     **kwargs) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises
    ``evaluate_sequence`` within the given time budget.

    The routine mixes a Frank‑Wolfe / LP phase with a short L‑BFGS
    continuation (as a fallback).  It returns the best sequence found as
    a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 5.0      # small safety margin

    # ------------------------------------------------------------------
    # Load a starting construction.  ``height_sequence_1`` is provided
    # by the outer harness; fall back to a uniform vector.
    # ------------------------------------------------------------------
    try:
        base = np.array(height_sequence_1, dtype=float)   # type: ignore
    except Exception:
        base = np.ones(2100, dtype=float)

    n = base.shape[0]
    target_sum = math.sqrt(2.0 * n)

    # Normalise to the required ℓ₁‑norm (∑a = √(2·n)).
    base = base * (target_sum / base.sum())
    f = base.copy()

    best_seq = f.copy()
    best_score = evaluate_sequence(best_seq.tolist())
    print(f"[init] n={n}, initial score={best_score:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation loop: Frank‑Wolfe style LP updates.
    # ------------------------------------------------------------------
    iteration = 0
    while time.time() < deadline:
        iteration += 1
        # 1) current maximal auto‑convolution
        conv_f = _full_conv(f)
        M = conv_f.max()
        time_left = deadline - time.time()

        # 2) Solve the LP (if SciPy is available and we have time)
        if _SCIPY_AVAILABLE and time_left > 1.0:
            b = _lp_step(f, M, target_sum, time_left)
        else:
            b = None

        if b is None:
            # LP failed or not available – switch to L‑BFGS continuation.
            print("[fallback] switching to L‑BFGS")
            # Prepare the log‑probability vector for the L‑BFGS routine.
            p0 = f / target_sum
            eps = 1e-12
            y0 = np.log(p0 + eps)
            # Rough temperature estimate.
            init_tau = max(_full_conv(p0).max() * 2.0, 1e-12)
            Nfft = 1 << ((2 * n - 1).bit_length())
            best_seq, best_score = _run_l_bfgs(
                y0=y0,
                init_tau=init_tau,
                target_sum=target_sum,
                Nfft=Nfft,
                deadline=deadline,
                best_seq=best_seq,
                best_score=best_score,
            )
            break

        # 3) Mixing line‑search
        f_new, new_score = _line_search_mix(f, b, target_sum,
                                            best_score, deadline)

        if new_score < best_score - 1e-12:
            f = f_new
            best_seq = f_new.copy()
            best_score = new_score
            print(f"[improve] iter {iteration:4d}  score={best_score:.6f}")
        else:
            # No improvement – inject a tiny random perturbation and
            # re‑normalise.  This often helps escape shallow plateaus.
            f = f + 0.02 * np.random.randn(n)
            f = np.maximum(f, 0.0)
            s = f.sum()
            if s == 0.0:
                f = np.ones(n) * (target_sum / n)
            else:
                f = f * (target_sum / s)

    # ------------------------------------------------------------------
    # Final sanity check (in case the best sequence changed after the loop)
    # ------------------------------------------------------------------
    final_score = evaluate_sequence(best_seq.tolist())
    if final_score < best_score:
        best_score = final_score
        best_seq = best_seq.copy()

    print(f"[done] best score = {best_score:.6f}, iterations = {iteration}, length = {n}")
    return best_seq.tolist()
