"""
propose_candidate
-----------------

A multi‑stage optimizer that searches for a non‑negative step‑function
(sequence) minimising

        F(a) = 2·n·max(a * a) / (sum(a)²)

where ``a * a`` is the linear convolution of ``a`` with itself.
The optimiser works on the probability simplex (sum(a)=1) and combines:

*   Several cheap initialisations (uniform, Dirichlet, exponential).
*   A projected Adam optimiser that first uses a smooth‑max surrogate
    and later switches to the exact sub‑gradient of the max‑convolution.
*   A final hill‑climbing phase that swaps tiny amounts of mass between
    two coordinates.

All operations respect a global time budget (default 1000 s).  The
best sequence found is returned as a plain Python list – exactly the
format expected by the external ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Compute (w ★ a)[i] = Σ_k w[k]·a[i+k-(n-1)], i = 0 … n‑1.
    Used for the smooth‑max surrogate (gradient of Σ w·(a*a)).
    """
    n = a.shape[0]
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = n - 1
    return corr[start:start + n]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be 1‑D")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _grad_of_max_conv(a: np.ndarray, k_max: int) -> np.ndarray:
    """
    Exact gradient of the max‑convolution term.
    For a given shift k_max,
        grad_j = 2 * a[k_max - j]   if 0 ≤ k_max-j < n
                0 otherwise.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if k_max < n:
        # overlap = a[0..k_max] with reversed a[0..k_max]
        g[: k_max + 1] = 2.0 * a[: k_max + 1][::-1]
    else:
        p = k_max - n + 1               # first index that participates
        # a[p..n-1] overlaps with reversed part
        g[p:] = 2.0 * a[p:][::-1]
    return g


def _exact_obj(a: np.ndarray, n: int) -> float:
    """Exact objective (2·n·max(conv)).  Assumes sum(a)=1."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000, **kwargs):
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects ``budget_s`` seconds and prints progress.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    best_seq = None          # np.ndarray, sum == 1
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to create exponential candidates.
    def _exp_candidate(n, c):
        """a_i ∝ exp(-c*i/n), normalised to sum==1."""
        alpha = c / n
        # exact normalisation for a finite walk
        numerator = 1.0 - np.exp(-alpha)
        denominator = 1.0 - np.exp(-alpha * n)
        K = numerator / denominator
        a = K * np.exp(-alpha * np.arange(n))
        # tiny numerical drift – renormalise explicitly
        a /= a.sum()
        return a

    # ------------------------------------------------------------------
    # Stage 1 – generate a diverse set of starting points.
    lengths = [256, 512, 1024, 2048, 4096]          # moderate sizes keep runtime low
    init_candidates = []   # list of (a, n)

    for n in lengths:
        if time.time() > deadline:
            break
        print(f"\n--- Initialising length n={n}")

        # uniform
        a_uni = np.full(n, 1.0 / n, dtype=np.float64)
        init_candidates.append((a_uni, n))

        # Dirichlet random
        a_dir = np.random.dirichlet(np.ones(n))
        init_candidates.append((a_dir, n))

        # exponential family with several c‑values
        for c in (0.5, 1.0, 1.5, 2.0, 3.0, 5.0):
            a_exp = _exp_candidate(n, c)
            init_candidates.append((a_exp, n))

    # evaluate them once and keep the best few
    init_candidates.sort(key=lambda tup: _exact_obj(tup[0], tup[1]))
    init_candidates = init_candidates[:20]          # keep a manageable pool

    # ------------------------------------------------------------------
    # Stage 2 – Adam optimisation with smooth‑max → exact max
    for idx, (a0, n) in enumerate(init_candidates):
        if time.time() > deadline:
            break
        print(f"\n=== Refinement #{idx+1} (n={n}) ===")

        fft_len_conv = _next_pow2(2 * n - 1)
        fft_len_corr = _next_pow2(3 * n)   # generous size for cross‑correlation

        a = a0.copy()
        m = np.zeros(n, dtype=np.float64)
        v = np.zeros(n, dtype=np.float64)
        beta1, beta2, eps = 0.9, 0.999, 1e-8
        lr = 0.12                     # a bit gentler than the previous code

        # smooth‑max temperature schedule
        lam = 0.5
        lam_factor = 1.15
        max_iters = 30000
        last_improve = time.time()
        t = 0

        while time.time() < deadline and t < max_iters:
            # ----- convolution -----
            conv = _full_convolution(a, fft_len_conv)
            max_c = conv.max()

            # ----- gradient -----
            if lam < 1e4:                         # smooth surrogate
                w = np.exp(lam * (conv - max_c))
                w_sum = w.sum()
                if w_sum == 0.0:
                    w = np.full_like(conv, 1.0 / conv.size)
                else:
                    w /= w_sum
                grad = 2.0 * _cross_correlation(w, a, fft_len_corr)
            else:                                 # exact sub‑gradient
                k_max = int(np.argmax(conv))
                grad = _grad_of_max_conv(a, k_max)

            # ----- Adam step -----
            t += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)
            step = lr * m_hat / (np.sqrt(v_hat) + eps)
            a_new = a - step
            a = _project_simplex(a_new)

            # ----- temperature schedule -----
            lam = min(lam * lam_factor, 5e5)

            # ----- monitor exact objective -----
            val = _exact_obj(a, n)

            if val < best_val:
                best_val = val
                best_seq = a.copy()
                last_improve = time.time()
                print(f"[iter {t:05d}] new best: {val:.6f}")

                if best_val < 1.45:                # early stop if we already beat the target comfortably
                    print("Strong target achieved – halting early.")
                    deadline = time.time()
                    break

            # ----- stagnation handling -----
            if time.time() - last_improve > 5.0:
                # add a tiny random perturbation before re‑projecting
                noise = (np.random.rand(n) - 0.5) * 1e-3
                a = _project_simplex(a + noise)
                last_improve = time.time()
                print(f"[iter {t}] shake (stagnation)")

    # ------------------------------------------------------------------
    # Stage 3 – random local refinement on the current best solution
    if best_seq is not None and time.time() < deadline:
        a = best_seq.copy()
        n = a.shape[0]
        fft_len_conv = _next_pow2(2 * n - 1)
        print("\n--- Final random hill‑climbing refinement ---")
        while time.time() < deadline:
            i, j = np.random.choice(n, 2, replace=False)
            # move a modest fraction of mass from i to j
            delta = 1e-3 * a[i] * np.random.rand()
            if delta == 0.0:
                continue
            a_prop = a.copy()
            a_prop[i] -= delta
            a_prop[j] += delta
            a_prop = _project_simplex(a_prop)
            val = _exact_obj(a_prop, n)
            if val < best_val:
                best_val = val
                best_seq = a_prop.copy()
                a = a_prop
                print(f"[refine] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform vector if nothing succeeded.
    if best_seq is None:
        n_fallback = 1024
        best_seq = np.full(n_fallback, 1.0 / n_fallback, dtype=np.float64)
        best_val = _exact_obj(best_seq, n_fallback)
        print(f"Fallback uniform sequence, value = {best_val:.6f}")

    print(f"\nBest value discovered: {best_val:.6f} with length {best_seq.shape[0]}")
    return best_seq.tolist()
