"""
search for a step‑function that minimises
        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The implementation uses three phases:

1. **Diverse pool generation** – deterministic families (uniform,
   sinusoid, edge‑exponential/power‑law, beta‑pdf, double‑Gaussian,
   arcsine‑like) together with many Dirichlet draws and a few cyclic
   shifts of a possible warm‑start.  Every vector is normalised to
   sum = 1 and its exact objective is evaluated with an FFT‑based
   convolution.  The best `top_k` are retained.

2. **Projected gradient descent** – each elite candidate is refined
   with a simple projected gradient method that works on the true
   max‑convolution (no smooth surrogate).  The step size is
   back‑tracked and the iterate is projected onto the probability
   simplex after every step.  More time is given to the most promising
   candidates.

3. **Long‑run random hill‑climber** – the current best vector is
   further improved by many cheap mass‑transfer moves; a move is kept
   only when it reduces the exact objective.

All phases respect the global 1000 s wall‑clock limit; the best
sequence found is returned as a plain Python list (compatible with the
provided `evaluate_sequence` function).
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Both `w` and `a` are real, `w` length = 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    # extract the n entries that correspond to the gradient
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Deterministic families (all vectors are returned with sum = 1) ------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    x = np.maximum(x, 0.0)
    x /= x.sum()
    return x


def _arcsine(n: int) -> np.ndarray:
    """Discrete analogue of the arcsine pdf – heavy at the ends."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = 1.0 / np.sqrt(i * (n + 0.5 - i))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Return a few cyclic shifts of an optional warm‑start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


# ----------------------------------------------------------------------
# Projected gradient descent --------------------------------------------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact objective.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum(a)==1  → denominator = 1

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val

# Cheap mass‑transfer hill climber --------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: transfer a random fraction (up to 30 %) of the
    mass from one index to another and keep the move if the exact
    objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Returns the best step‑function found within the
    given wall‑clock budget.
    """
    # ------------------------------------------------------------------
    # initialise RNG and time limits
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # a safety margin

    # ------------------------------------------------------------------
    # problem size (feel free to increase – the code adapts automatically)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a very diverse initial pool
    pool = _warm_and_shifts(warm, n)

    # deterministic families – a fairly dense grid of parameters
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    pool.append(_arcsine(n))

    for gamma in np.logspace(-2, 2, 30):
        pool.append(_edge_exponential(n, float(gamma)))

    for alpha in np.linspace(-5.0, -0.1, 30):
        pool.append(_edge_power_law(n, float(alpha)))

    a_vals = np.logspace(-1, 1, 12)
    b_vals = np.logspace(-1, 1, 12)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    for sigma in np.linspace(0.01, 0.6, 30):
        pool.append(_double_gaussian(n, float(sigma)))

    # random Dirichlet draws with a wide variety of concentrations
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(4):
            pool.append(_random_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    # quick evaluation of the whole pool, keep a modest elite
    top_k = 30
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()          # sum(a)==1 → denominator = 1
        scored.append((val, a))

        # leave a comfortable fraction of time for refinement
        if time.time() >= deadline * 0.25:
            break

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # Phase 2 – refined gradient descent on the elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 10:
            break

        # a decreasing slice of the remaining budget
        remaining = deadline - time.time()
        # geometric allocation: earlier candidates get more time
        total = sum(range(1, top_k + 1))
        share = remaining * (top_k - rank) / total
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=300_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank:02d}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap local moves (use any remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=2_000_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
