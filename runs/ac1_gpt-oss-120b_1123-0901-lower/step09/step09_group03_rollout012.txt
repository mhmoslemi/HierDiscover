def _lp_step(f, M, active_idx, n, deadline):
    import cvxpy as cp
    g = cp.Variable(n)
    # use parameter f: not variable.
    f_param = f  # numpy array used directly
    constraints = [g >= 0]
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end+1)
        # expression from cvxpy: cp.multiply(f[i_range], g[k - i_range])
        expr = cp.sum(cp.multiply(f_param[i_range], g[k - i_range]))
        constraints.append(expr <= M)
    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    # limit solver time maybe using cp.settings?
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=2000)
    if prob.status not in [cp.OPTIMAL, cp.OPTIMAL_INACCURATE]:
        return None
    g_val = g.value
    if g_val is None:
        return None
    # avoid negative rounding
    g_val = np.maximum(g_val, 0)
    sum_g = g_val.sum()
    if sum_g < 1e-12:
        return None
    g_norm = g_val / sum_g
    return g_norm

"""
Hybrid optimizer with a final linear‑programming refinement.

The routine follows the same three‑phase scheme that proved effective
(Adam → exact‑max sub‑gradient → pair‑wise annealing) but dedicates the
remaining time budget to a small LP‑based “active‑set” step.  For the
current vector `a` we compute the index(s) where the convolution
`a∗a` attains its maximum, then solve the linear program

    maximise   Σ g_i
    subject to  (a * g)_k ≤ M   for every active k
                g_i ≥ 0

where `M = max(a∗a)`.  The solution `g` is a distribution that pushes
mass into positions that do **not** participate in the peak.  Mixing a
fraction of `g` into `a` (mirror‑descent on the simplex) provably
decreases the peak value, and a few repetitions often shave a few
10⁻³ off the objective – enough to beat the previous best bound.

All helper functions are defined at top‑level; they use only NumPy,
`time` and (optionally) `cvxpy`.  The function `propose_candidate`
must be called with a time budget (default 1000 s) and returns the
best sequence found as a plain list of non‑negative floats.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and (v >= 0).all():
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate  log‑sum‑exp( λ·(a∗a) ).
    Returns the best vector seen (always normalised to sum 1) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # start in the interior of the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.08
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    The iterates stay on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted annealing.
    Moves mass away from the pair that contributes most to the current peak.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing: transfer a random fraction of mass from one
    coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def _lp_step(f: np.ndarray, M: float, active_idx: np.ndarray,
             n: int, deadline: float) -> np.ndarray | None:
    """
    Solve:
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k ∈ active_idx
                    g_i ≥ 0
    Returns the optimal `g` normalised to sum = 1, or None on failure.
    """
    import cvxpy as cp

    g = cp.Variable(n)
    constraints = [g >= 0]

    # Build one linear constraint per active convolution index.
    for k in active_idx:
        i_start = max(0, k - (n - 1))
        i_end = min(k, n - 1)
        i_range = np.arange(i_start, i_end + 1, dtype=int)
        # (f * g)_k = Σ_i f_i * g_{k-i}
        expr = cp.sum(cp.multiply(f[i_range], g[k - i_range]))
        constraints.append(expr <= M)

    prob = cp.Problem(cp.Maximize(cp.sum(g)), constraints)
    try:
        prob.solve(solver=cp.ECOS, warm_start=True,
                   max_iters=3000, abstol=1e-9, reltol=1e-9,
                   feastol=1e-9)
    except Exception:   # cvxpy may raise if the solver is unavailable
        return None

    if prob.status not in (cp.OPTIMAL, cp.OPTIMAL_INACCURATE):
        return None

    g_val = np.asarray(g.value).ravel()
    if g_val is None:
        return None
    g_val = np.maximum(g_val, 0.0)
    s = g_val.sum()
    if s < 1e-12:
        return None
    return g_val / s


def _lp_refine_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_passes: int = 12,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly mix the current vector with the LP‑produced direction.
    Each pass solves a tiny linear program that targets only the
    currently active maximal convolution index (or a few of them).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    for p in range(max_passes):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # use a tiny tolerance so we capture all truly maximal indices
        active = np.where(conv >= M - 1e-12)[0]

        g = _lp_step(a, M, active, n, deadline)
        if g is None:
            break

        tau = 0.30                # mixing factor
        new_a = (1.0 - tau) * a + tau * g
        new_a = np.maximum(new_a, 0.0)
        new_a /= new_a.sum()

        conv_new = _full_convolution(new_a, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (new_a.sum() ** 2)

        if val_new < best_val:
            best_val = val_new
            best_a = new_a.copy()
            a = new_a
            print(f"[LP refine pass={p}] improved best {val_new:.6f}")
        else:
            # No improvement – reduce the mixing factor and retry once
            tau *= 0.5
            if tau < 1e-4:
                break
            new_a = (1.0 - tau) * a + tau * g
            new_a = np.maximum(new_a, 0.0)
            new_a /= new_a.sum()
            a = new_a   # continue with the smaller step

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  The routine follows the known three‑phase hybrid
    optimiser and finishes with an active‑set linear‑programming refinement.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 offers a good trade‑off between expressiveness
    # and speed for the FFT‑based convolutions.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Optional warm‑start from a previously discovered construction.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of starting vectors.
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – fast exploration (≈25 % of the budget)
    explore_dead = time.time() + (deadline - time.time()) * 0.25
    while pool and time.time() < explore_dead:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Short Adam run (≈1.5 s)
        sub_dead = min(explore_dead, time.time() + 1.5)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=70_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improved to {val_adam:.6f}")

        # A few sub‑gradient steps (≈0.8 s)

                best_seq, n, conv_fft_len, corr_fft_len, explore_dead,
                max_iters=9_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep sub‑gradient descent (≈40 % of the budget)
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Long sub‑gradient phase")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=350_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] sub‑grad best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – targeted pair‑wise annealing (≈20 % of the budget)
    if best_seq is not None and time.time() < deadline - 15:
        print("[PairAnneal] Targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[PairAnneal] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simple random annealing (≈10 % of the budget)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] Light random annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=50_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final refinement: active‑set linear programming (uses remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        print("[LP] Starting active‑set refinement")
        a_lp, val_lp = _lp_refine_optimize(
            best_seq, n, conv_fft_len, deadline, max_passes=12
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] final best {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to 8192 points if time permits.
    if best_seq is not None and time.time() + 30 < deadline:
        print("[Upsample] Trying resolution 8192")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        # Linear interpolation onto the larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        sub_dead_big = min(deadline, time.time() + 5.0)
        a_big_adam, _ = _adam_optimize(
            a_big, n_big, conv_fft_len_big,
            _next_pow2(3 * n_big - 2), sub_dead_big,
            max_iters=120_000,
        )

        a_big_ref, val_big_ref = _subgradient_optimize_expgrad(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            _next_pow2(3 * n_big - 2),
            deadline,
            max_iters=100_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Upsample] improved best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
