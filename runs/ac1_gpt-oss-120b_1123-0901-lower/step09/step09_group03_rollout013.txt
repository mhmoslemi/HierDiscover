"""
Hybrid multi‑restart optimizer for the step‑function objective

    F(a) = 2·n·max(a∗a) / (∑a)²  (∑a > 0, a_i ≥ 0)

The algorithm works in four stages within the 1000 s budget:

1.  **Exploratory restarts** – many random Dirichlet starts are fed to a
    very short Adam optimizer (smooth‑max surrogate).  This quickly
    discovers promising basins that the deterministic phases might miss.
2.  **Long projected sub‑gradient descent** – a Euclidean‐projected
    sub‑gradient method (exact max) refines the best candidate.
3.  **Pair‑wise targeted annealing** – deterministic moves that
    repeatedly take mass away from the pair (i,j) that contributes most
    to the current maximal convolution entry.
4.  **Final Adam fine‑tuning** – a longer Adam run on the final vector
    to polish the solution.

All iterates stay on the probability simplex (∑a = 1) via either a
softmax (Adam) or Euclidean projection.  The code is self‑contained,
uses only NumPy and the FFT for fast convolutions, and respects the
time limit with a safety margin.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (zero‑padded) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx = 1}."""
    if v.sum() == 1.0 and (v >= 0).all():
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr: float = 0.05,
    lam_start: float = 0.04,
    lam_factor: float = 1.018,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate  log‑sum‑exp( λ·(a∗a) ).
    Returns the best vector seen (always normalised to sum 1) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior point of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start
    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (fast, no extra FFTs)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smoothed surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule for smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional small shake near the end
        if it % 3000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max objective.
    The iterate stays on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad it={it:06d}] best {val:.6f}")

        # active set (all indices attaining the max, up to numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalisation – protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _euclid_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the exact max objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[EuclidSGD it={it:06d}] best {val:.6f}")

        # active indexes of the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a - eta * grad
        a = _project_simplex(a)                       # stay on the simplex

        it += 1

def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Targeted annealing: repeatedly remove a small fraction from the (i,j)
    pair that most contributes to the current maximal convolution entry.
    """
    a = a0.copy()
    val = 2.0 * n * _full_convolution(a, conv_fft_len).max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.07
    T_end = 5e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = max(T_end, T_start * (1.0 - move / float(max_moves)))

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # indices i that can pair to kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        # adaptive move size: larger when the pair is heavy
        pair_mass = min(a2[i_src], a2[j_src]) if i_src != j_src else a2[i_src]
        delta = pair_mass * 0.08                     # 8 % of the pair mass
        if delta < 1e-14:
            move += 1
            continue

        if i_src == j_src:
            # single‑index case – move mass to two random other places
            t1, t2 = np.random.choice(n, size=2, replace=False)
            a2[i_src] -= delta
            a2[t1] += delta / 2
            a2[t2] += delta / 2
        else:
            # remove from both ends, redistribute to two random slots
            t1, t2 = np.random.choice(n, size=2, replace=False)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        # numerical safety
        a2 = np.maximum(a2, 0.0)
        a2 = _project_simplex(a2)

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    # warm‑start (if any) and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 8, n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.01, 0.04, 0.1, 0.3, 0.8, 1.5, 3.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edge families (decay towards the centre)
    for gamma in (0.15, 0.4, 0.9, 1.8):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families (heavy tails)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-1.8, -0.6, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.07, 0.18, 0.35):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid (used in the original pool)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Search for a non‑negative step‑function that minimizes the evaluation
    score.  The routine respects the given time budget and always returns
    the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    safety = 0.5                      # reserve a little slack
    deadline = time.time() + budget_s - safety

    # problem size – a larger n gives more freedom at modest cost
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase I – many short random restarts with Adam (≈20 % of budget)
    pool = _initial_candidates(n, warm)
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    print("[Phase I] Random restarts with short Adam")
    while pool and time.time() < phase1_dead:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(phase1_dead, time.time() + 2.5)   # ≤2.5 s per run

            a0,

            lam_factor=1.02,
        )
        if val_adam < best_val:

            best_seq = a_adam.copy()
            print(f"[Phase I] New best {val_adam:.6f}")

        # one quick sub‑gradient refinement (≈1 s)
        if time.time() < phase1_dead:
            sub_dead2 = min(phase1_dead, time.time() + 1.0)
            a_sub, val_sub = _subgradient_optimize_expgrad(
                a_adam,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_dead2,
                max_iters=10_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase I] Refined best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long Euclidean projected sub‑gradient descent (≈40 % budget)
    if best_seq is not None:
        phase2_dead = time.time() + (deadline - time.time()) * 0.40
        print("[Phase II] Euclidean projected sub‑gradient")
        a_euc, val_euc = _euclid_subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase2_dead,
            max_iters=400_000,
            eta0=0.85,
        )
        if val_euc < best_val:
            best_val = val_euc
            best_seq = a_euc.copy()
            print(f"[Phase II] Improved best {val_euc:.6f}")

    # ------------------------------------------------------------------
    # Phase III – pair‑wise targeted annealing (≈30 % budget)
    if best_seq is not None and time.time() < deadline - 10:
        phase3_dead = time.time() + (deadline - time.time()) * 0.30
        print("[Phase III] Pair‑wise targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            phase3_dead,
            max_moves=250_000,
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[Phase III] Improved best {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final Adam fine‑tuning (remaining time)
    if best_seq is not None and time.time() < deadline - 1:
        phase4_dead = deadline
        print("[Phase IV] Final Adam fine‑tuning")
        a_final, val_final = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase4_dead,
            max_iters=200_000,
            lr=0.04,
            lam_start=0.08,
            lam_factor=1.015,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Phase IV] New best {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return (uniform if everything failed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the bounds accepted by evaluate_sequence (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished. Best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
