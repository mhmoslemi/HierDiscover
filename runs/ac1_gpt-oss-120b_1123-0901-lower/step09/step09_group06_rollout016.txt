"""
Minimize  F(a) = 2·n·max(a∗a) / (sum(a)^2)  for a non‑negative step function.
The implementation below builds a diverse initial pool (including a warm‑start
if one is available), then runs three optimisation stages that fit into the
1000 s budget:

1. **Direct search on a low‑dimensional parametrisation** – we describe a
   candidate by K piece‑wise‑constant heights.  A cheap differential‑evolution
   (DE) optimiser searches the K‑dimensional space; each fitness evaluation
   uses an FFT‑based convolution (O(n log n)).

2. **Projected sub‑gradient descent** on the full‑resolution vector.
   The exact objective is non‑smooth (max of a convolution) but its
   sub‑gradient can be written in closed form: for the index k* where the
   convolution attains its maximum,
       g_i = 2·a_{k*‑i} (zero when the index is out of range)
   and the gradient of the whole objective adds a constant term that disappears
   after projection onto the simplex.  After each step we Euclidean‑project
   back onto the probability simplex.

3. **Cheap random hill‑climber** – random mass transfers between two positions
   that are kept only when they improve the true objective.

All steps are timed and stop gracefully before the deadline.  The best
sequence found is returned as a plain Python list; it is clipped to the range
[0, 1000] required by ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize, differential_evolution

# Basic linear‑algebra utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    """Monotone decreasing exponential (only left side, not mirrored)."""
    a = np.exp(-gamma * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _right_exponential(n: int, gamma: float) -> np.ndarray:
    """Monotone increasing exponential (mirror of left_exponential)."""
    a = np.exp(-gamma * np.arange(n - 1, -1, -1, dtype=np.float64))
    a /= a.sum()
    return a


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)

    shape /= shape.sum()

        return a
    pos = rng.choice(n, size=k, replace=False)

# Low‑dimensional optimisation (piecewise‑constant parametrisation)
def _expand_piecewise_constant(heights: np.ndarray, n: int) -> np.ndarray:
    """Expand a K‑vector of heights to a full length‑n vector (piecewise constant)."""
    K = heights.size
    base = n // K
    rem = n % K
    blocks = []
    for i, h in enumerate(heights):
        length = base + (1 if i < rem else 0)
        blocks.append(np.full(length, h, dtype=np.float64))
    return np.concatenate(blocks)


def _run_de_on_piecewise(n: int,
                         conv_fft_len: int,
                         deadline: float,
                         rng: np.random.Generator,
                         max_evals: int = 5000) -> np.ndarray | None:
    """
    Differential‑evolution on a K‑segment piecewise‑constant representation.
    Returns the best full‑resolution vector (probability simplex) or None
    if the time budget is exhausted.
    """
    # pick a divisor of n that gives a moderate K
    possible_K = [8, 16, 32, 64, 128]
    K = next(k for k in possible_K if n % k == 0 and k <= n // 4)
    bounds = [(0.0, 5.0) for _ in range(K)]

    def de_obj(heights):
        a = _expand_piecewise_constant(np.array(heights), n)
        total = a.sum()
        if total == 0.0:
            return np.inf
        a = a / total                    # project onto simplex
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # sum(a)=1 → objective = 2·n·max_c
        return 2.0 * n * max_c

    # adapt max_iter to remaining time (roughly 0.001 s per evaluation)
    remaining = deadline - time.time()
    if remaining < 5.0:
        return None
    # rough budget: 1 µs per float op → allow at most 4000 evaluations safely
    max_iter = max(1, min(200, int(max_evals / (len(bounds) * 10))))
    # run DE
    result = differential_evolution(
        de_obj,
        bounds,
        maxiter=max_iter,
        popsize=15,
        seed=int(rng.integers(1 << 30)),
        polish=False,
        updating='deferred',
        workers=1,
        disp=False,
        tol=1e-6,
    )
    if not result.success:
        return None
    best_heights = np.array(result.x)
    a_best = _expand_piecewise_constant(best_heights, n)
    a_best /= a_best.sum()
    return a_best

# Projected sub‑gradient descent on the exact non‑smooth objective
def _subgradient_descent(a0: np.ndarray,
                         conv_fft_len: int,
                         deadline: float,
                         max_iters: int = 30_000) -> np.ndarray:
    """
    Simple sub‑gradient descent with Euclidean projection onto the simplex.
    Very cheap per iteration (O(n) gradient) – suitable for a final polishing
    phase after a good starting point is known.
    """
    a = a0.copy()
    n = a.shape[0]
    best_a = a.copy()
    # exact objective (sum = 1 after every projection)
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        k_star = int(conv.argmax())

        # gradient of max_c w.r.t a (sub‑gradient of the max)
        grad = np.zeros_like(a)
        i_start = max(0, k_star - (n - 1))
        i_end = min(k_star, n - 1)
        # indices that actually appear in the max convolution term
        idx = k_star - np.arange(i_start, i_end + 1)
        grad[i_start:i_end + 1] = a[idx]

        # full objective gradient = (4·n)·(grad - max_c·1)  (the constant part disappears after projection)
        grad = grad - max_c  # broadcasting scalar; after projection the constant shift is irrelevant
        step = 0.4 / np.sqrt(it + 1)            # diminishing stepsize
        a_new = a - step * grad
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_onto_simplex(a_new)

        # evaluate
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val - 1e-12:
            best_val = val_new
            best_a = a_new.copy()
            a = a_new
            if it % 2000 == 0:
                print(f"[SubGrad it={it:05d}] new best {best_val:.6f}")
        else:
            # occasional tiny random shake to escape flat regions
            if it % 1000 == 0:
                a_new = a + (rng.random(n) - 0.5) * 1e-4
                a_new = np.maximum(a_new, 0.0)
                a_new = _project_onto_simplex(a_new)
                a = a_new
        it += 1

    return best_a

# Cheap hill‑climber (random mass moves)
def _local_random_search(a0: np.ndarray,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 80_000) -> np.ndarray:
    """Randomly transfer a small amount of mass between two indices if it improves the true objective."""
    a = a0.copy()
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.25)   # move up to 25 % of mass from i
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[HillClimb it={it:06d}] improved to {best_val:.6f}")
        it += 1
    return best_a

# Main entry point required by the competition harness
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function minimizing the evaluation score.
    The routine respects the time budget (default 1000 s) and always returns
    the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # Length of the vector (use warm start length if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # truncate / pad to a power‑of‑two length for convenience
        n = warm.shape[0]
    else:
        n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of deterministic starters
    pool = []

    # warm start and simple shifts (if it exists)
    if warm is not None:
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
            pool.append(warm.copy())
            for shift in (n // 4, n // 2, 3 * n // 4):
                pool.append(np.roll(warm, shift))

    # deterministic families (many of the classic shapes)
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))

    for gamma in np.logspace(-4, -0.5, 10):
        pool.append(_left_exponential(n, gamma))

    for gamma in np.logspace(-4, -0.5, 10):
        pool.append(_right_exponential(n, gamma))

    for alpha in np.linspace(-8, -0.2, 12):
        pool.append(_edge_power_law(n, alpha))

    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0), (5.0, 0.2)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))
    pool.append(_linear_decrease(n))
    pool.append(_linear_increase(n))

    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # 2) Quick scan of the pool (FFT‑based evaluation, cheap)
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a /= a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")
        # leave a minute for deeper optimisation
        if time.time() > deadline - 60:
            break

    # ------------------------------------------------------------------
    # 3) Low‑dimensional DE optimisation (if we still have time)
    if best_seq is not None and time.time() < deadline - 30:
        a_de = _run_de_on_piecewise(n, conv_fft_len, deadline, rng, max_evals=8000)
        if a_de is not None:
            conv = _full_convolution(a_de, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a_de.copy()
                print(f"[DE] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected sub‑gradient descent for fine tuning
    if best_seq is not None and time.time() < deadline - 20:
        a_sub = _subgradient_descent(best_seq, conv_fft_len, deadline, max_iters=25000)
        conv = _full_convolution(a_sub, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a_sub.copy()
            print(f"[Sub‑grad] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Final cheap random hill‑climber (last few seconds)
    if best_seq is not None and time.time() < deadline - 3:
        a_final = _local_random_search(best_seq, conv_fft_len, deadline, max_moves=120_000)
        conv = _full_convolution(a_final, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = a_final.copy()
            print(f"[HillClimb final] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the evaluator's allowed range and return as list
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
