def _prune(f, K):
    if K >= len(f):
        return f
    # find threshold value
    idx = np.argpartition(f, -K)[-K:]
    mask = np.zeros_like(f, dtype=bool)
    mask[idx] = True
    f2 = np.where(mask, f, 0.0)
    if f2.sum() == 0:
        return np.full_like(f, 1.0 / len(f))
    return f2 / f2.sum()
```

"""
LP‑Annealing optimizer for the step‑function objective

The algorithm alternates two phases:

1.  Linear‑programming (LP) phase.
    For the current distribution f we compute M = max(f∗f).
    We then solve the LP
        maximise   ∑ g_i
        subject to g_i ≥ 0
                   f∗g ≤ M  (pointwise)
    The solution g is a feasible direction that can carry more mass
    without increasing the peak of the convolution.
    We move a fraction t towards the normalised g:
        f ← (1‑t)·f + t·(g/∑g) .
    After each move we prune the smallest entries to keep the vector
    sparse (≈200 non‑zero entries) – this makes the next LP cheap.

2.  Sub‑gradient refinement.
    When the LP budget is exhausted we finish with a few
    exponentiated‑gradient sub‑gradient steps on the true non‑smooth
    objective (the maximum convolution entry).  This tightens the
    solution further.

Both phases stay on the probability simplex, thus the homogeneity of
the evaluation function guarantees that the scale of the vector is
irrelevant.  The whole routine respects the time budget (default 1000 s)
and always returns the best sequence found.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact evaluation of F = 2·n·max(a∗a)/(∑a)² .  Assumes a ≥ 0."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s * s)


def _build_A_ub(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the (2n‑1) × n constraint matrix for  f∗g ≤ M .
    Only non‑zero entries of f are used – the matrix is very sparse.
    """
    n = f.shape[0]
    nz_idx = np.where(f > 1e-14)[0]
    nz_val = f[nz_idx]
    k = nz_idx.size
    if k == 0:
        # degenerate case – return a zero matrix
        return sp.csr_matrix((2 * n - 1, n), dtype=np.float64)

    # rows[i,j] = nz_idx[i] + j
    row_idx = np.add.outer(nz_idx, np.arange(n)).ravel()
    col_idx = np.tile(np.arange(n), k)
    data = np.repeat(nz_val, n)
    A = sp.csr_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n))
    return A


def _prune_keep_top(a: np.ndarray, keep: int) -> np.ndarray:
    """Keep only the `keep` largest entries (by mass) and renormalise."""
    if keep >= a.size:
        return a
    # argpartition for speed
    idx = np.argpartition(a, -keep)[-keep:]
    mask = np.zeros_like(a, dtype=bool)
    mask[idx] = True
    a2 = np.where(mask, a, 0.0)
    total = a2.sum()
    if total <= 0.0:
        # fallback to uniform
        return np.full_like(a, 1.0 / a.size)
    return a2 / total


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The method combines a sparse LP phase with a final
    sub‑gradient refinement.  It returns the best sequence found as a
    plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------
    # problem size – we keep the vector relatively sparse (≈200 nnz)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------
    # Initial candidate – a random sparse Dirichlet on ~sqrt(4n) positions
    m0 = int(np.sqrt(4 * n))          # ≈128 for n=4096
    a = np.zeros(n, dtype=np.float64)
    pos = np.random.choice(n, size=m0, replace=False)
    a[pos] = np.random.dirichlet(np.ones(m0))
    a /= a.sum()                       # stay on the simplex

    # use a warm start if the notebook already produced one
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.size != n:
            if w.size > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.size))
        w = np.maximum(w, 0.0)
        if w.sum() > 0.01:
            w /= w.sum()
            # blend a little with the warm start
            a = 0.5 * a + 0.5 * w
            a /= a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------
    # ---------- LP‑based descent (≈70 % of the budget) ------------
    # Keep a small active set for speed
    keep_top = 256
    t = 0.25                         # convex‑combination step size

    lp_deadline = time.time() + (deadline - time.time()) * 0.70
    lp_iter = 0

    while time.time() < lp_deadline:
        # 1) evaluate current max convolution
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # 2) build sparse constraint matrix (only non‑zero entries of a)
        A_ub = _build_A_ub(a)
        b_ub = np.full(2 * n - 1, M)

        # 3) solve the LP: maximise ∑ g_i  ⇒ minimise -∑ g_i
        c = -np.ones(n, dtype=np.float64)
        res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method='highs',
                          options={'presolve': True, 'time_limit': 5.0})
        if not res.success:
            # LP failed – fall back to a tiny random jitter and continue
            jitter = np.random.rand(n)
            jitter /= jitter.sum()
            a = 0.9 * a + 0.1 * jitter
            a = _prune_keep_top(a, keep_top)
            a /= a.sum()
            continue

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            # should not happen, but guard against division by zero
            g = np.full(n, 1.0 / n)
        else:
            g = g / g_sum                     # bring g onto the simplex

        # 4) convex combination and pruning
        a = (1.0 - t) * a + t * g
        a = _prune_keep_top(a, keep_top)     # maintain sparsity
        a /= a.sum()                         # renormalise (numerical safety)

        # 5) evaluate
        val = _objective(a, conv_fft_len)
        lp_iter += 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={lp_iter:04d}] new best {best_val:.6f}")

        # early stop if we have already reached the target
        if best_val <= 1.503:
            print("[LP] target reached, exiting LP phase")
            break

        # adapt step size gently (optional)
        if lp_iter % 50 == 0 and t > 0.05:
            t *= 0.9

    # ------------------------------------------------------------
    # ------- Sub‑gradient refinement (≈rest of budget) ----------
    sub_deadline = deadline
    if time.time() < sub_deadline - 2:
        print("[SubGrad] start refinement")
        a = best_a.copy()
        it = 0
        while time.time() < sub_deadline:
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum() ** 2)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[SubGrad it={it:06d}] new best {best_val:.6f}")

            # active‑set weight (smooth surrogate of the max)
            active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()

            # gradient of the true max (see paper)
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # exponentiated‑gradient update (mirror descent on the simplex)
            eta = 1.0 / np.sqrt(it + 1.0)
            a = a * np.exp(-eta * grad)

            # protect against numerical drift
            s = a.sum()
            if s <= 0 or np.isnan(s):
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            it += 1
            if it % 200_000 == 0:
                # occasional aggressive pruning to keep size moderate
                a = _prune_keep_top(a, keep_top)

    # ------------------------------------------------------------
    # Final sanity – clip and return a plain list
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
