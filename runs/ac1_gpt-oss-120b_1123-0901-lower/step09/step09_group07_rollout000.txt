"""
Propose a low‑score non‑negative step function.

Algorithm Overview
------------------
The search is performed in three time‑aware phases.

1. **Fourier‑basis random sampling** – a low‑dimensional family
   `a_i = softmax( B @ c )`, where `B` contains a constant term and a few
   cosine/sine Fourier basis functions, is sampled aggressively.  Each
   candidate is evaluated with an FFT‑based convolution and the best
   (smallest objective) is retained.

2. **Adam optimisation in the coefficient space** – starting from the best
   coefficient vector `c` a standard Adam optimiser is run on a smooth‑max
   surrogate of the max‑convolution objective.  The optimisation works on
   the logits (soft‑max) representation, thus staying on the probability
   simplex automatically.  The temperature for the smooth‑max grows
   gradually, improving the gradient‑signal.

3. **Fine‑grained stochastic hill‑climbing** – a cheap local search that
   repeatedly transfers a random fraction of probability mass between two
   positions and accepts improvements immediately.

The routine respects the 1000 s budget, uses only NumPy (no external
solvers), and returns a list of non‑negative heights ready for
`evaluate_sequence`.  If a previous solution is stored in the global
variable `height_sequence_1` it is used as a warm start.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities (identical to the reference implementation)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a) / (sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _softmax(logits: np.ndarray) -> np.ndarray:
    mx = logits.max()
    ex = np.exp(logits - mx)
    return ex / ex.sum()


# ----------------------------------------------------------------------
# Fourier basis construction
def _fourier_basis(n: int, num_freqs: int = 8) -> np.ndarray:
    """
    Return an (n, 1+2*num_freqs) matrix whose columns are:
        1,
        cos(2π k i / n), sin(2π k i / n)  for k=1..num_freqs.
    """
    i = np.arange(n, dtype=np.float64)
    cols = [np.ones_like(i)]
    for k in range(1, num_freqs + 1):
        angle = 2.0 * np.pi * k * i / n
        cols.append(np.cos(angle))
        cols.append(np.sin(angle))
    return np.column_stack(cols)


# ----------------------------------------------------------------------
# Adam optimiser working on the low‑dimensional coefficient vector `c`
def _adam_optimize_c(
    c0: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.06,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the coefficient vector `c` that defines the
    probability vector via a softmax over the Fourier basis `B`.
    Returns the best probability vector and its objective value.
    """
    eps = 1e-15
    c = c0.astype(np.float64).copy()
    m = np.zeros_like(c)
    v = np.zeros_like(c)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.1          # initial temperature for smooth‑max
    lam_factor = 1.03  # slowly increase

    best_c = c.copy()
    best_val = np.inf

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # probability vector on the simplex
        logits = B @ c
        a = _softmax(logits)

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_c = c.copy()
            if it % 20000 == 0:
                print(f"[Adam on c it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # gradient w.r.t. coefficients c
        grad_c = B.T @ grad_log

        # Adam update on c
        m = beta1 * m + (1.0 - beta1) * grad_c
        v = beta2 * v + (1.0 - beta2) * (grad_c ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        c -= step

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random perturbation near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            c += (np.random.rand(*c.shape) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.94

    best_a = _softmax(B @ best_c)
    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing on the probability vector
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Move a random fraction of mass from one index to another.
    Accept only improvements.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    for it in range(max_moves):
        if time.time() >= deadline:
            break
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        # evaluate
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

    return best_a, best_val

            print(f"[Warm‑start] improved to {val0:.6f}")

        # use warm coefficients as an additional candidate for phase 2
        warm_c = c0.copy()
    else:
        warm_c = None

    # ------------------------------------------------------------
    # Phase 2 – Adam optimisation in coefficient space
    if best_seq is not None and time.time() < deadline - 15:
        # initialise Adam with the best coefficient vector known so far
        if warm_c is not None:
            init_c = warm_c
        else:
            # obtain a coefficient vector that reproduces the current best a
            eps = 1e-12
            log_best = np.log(best_seq + eps)
            init_c, *_ = np.linalg.lstsq(B, log_best, rcond=None)

        print("Phase 2 – Adam optimisation on Fourier coefficients")
        a_adam, val_adam = _adam_optimize_c(
            init_c,
            B,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            lr_start=0.07,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing
    if best_seq is not None and time.time() < deadline - 4:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------
    # fallback – uniform distribution if nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to obey `evaluate_sequence` limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # expose the result globally for warm‑start in a possible next run
    globals()["height_sequence_1"] = best_seq.tolist()
    return best_seq.tolist()
