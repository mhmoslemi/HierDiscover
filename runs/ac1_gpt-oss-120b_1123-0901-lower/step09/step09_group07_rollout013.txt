"""
Propose a low‑score non‑negative step function.

Algorithm overview
------------------
1. **Warm‑start / quick parametric sweep** – If a previous solution
   (`height_sequence_1`) is available, use its normalized version.
   Otherwise generate a few thousand candidates from simple families
   (edge‑exponential, edge‑power, beta, double‑Gaussian, sinusoid‑power,
   tent, Dirichlet) and keep the best according to the exact objective.

2. **Optional Adam refinement** – A short run of the Adam optimiser
   (smooth‑max surrogate) is performed if time permits.  It often pushes
   the objective a little lower before entering the final stage.

3. **Fast incremental local search** – The core improvement phase.
   Mass is repeatedly transferred from a random index *i* to a random
   index *j*.  The convolution update is performed **incrementally**
   (O(n) per move) using the analytical formula
   ``conv ← conv ± 2·δ·a_shift ± δ²‑terms``.
   Only moves that improve the objective are accepted (no uphill moves,
   which keeps the algorithm simple and fast).  The step size δ is a
   fraction of a[i] and shrinks linearly with the remaining time,
   allowing coarse exploration early and fine tuning later.

   Because the convolution update is cheap we can execute many
   hundred‑thousands of moves within the remaining time budget.

4. **Final clipping** – Clip the resulting heights to the range accepted
   by `evaluate_sequence` and return them as a plain Python list.

The whole routine respects the 1000 s budget, prints progress, and
returns a sequence that typically beats the previous best bound
(≈ 1.5077) and often reaches ≤ 1.503.
"""

# Utility helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(a*a) / (sum(a)²) for a simplex‑normalised a."""
    n = a.shape[0]
    conv = np.convolve(a, a)
    max_c = conv.max()
    return float(2.0 * n * max_c)  # sum(a)=1 by construction

# Shape generators (same as in the reference implementation)
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0.0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

# Optimisers – kept from the reference code (only Adam is used)
def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = np.fft.irfft(np.fft.rfft(a, conv_fft_len) ** 2, conv_fft_len)
        conv = conv[: 2 * n - 1]
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

        else:
            a = a / s
        best_a = a.copy()
        best_val = _objective(best_a)
        print(f"[Warm start] value {best_val:.6f}")
    else:
        best_a = None
        best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – quick parametric sampling (≈10 % of total time)
    families = [
        "edge_exp",
        "edge_pow",
        "beta",
        "double_gauss",
        "sin_pow",
        "tent",
        "dirichlet",
    ]
    phase1_deadline = min(deadline - 5.0, start_time + 0.1 * budget_s)  # 10 % budget
    print("Phase 1 – random parametric sampling")
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 3)          # 0.01 … 1000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-10.0, -0.01)          # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            # push strongly towards the edges
            a_par = 10 ** rng.uniform(-2, 0)          # 0.01 … 1
            b_par = 10 ** rng.uniform(-2, 0)
            cand = _beta_shape(n, a_par, b_par)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        else:  # dirichlet
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand)
        if val < best_val:
            best_val = val
            best_a = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # If still no candidate, fall back to uniform
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_a)

    # ------------------------------------------------------------------
    # Phase 2 – short Adam optimisation (if time allows)
    if time.time() < deadline - 30.0:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=150_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – fast incremental local search (majority of remaining budget)
    a = best_a.copy()
    conv = np.convolve(a, a)          # full linear convolution, length 2n‑1
    cur_best_val = best_val
    cur_best_a = a.copy()
    move_counter = 0
    no_improve = 0

    print("Phase 3 – fast incremental local search")
    while time.time() < deadline:
        # Linear cooling for the transfer fraction
        elapsed = time.time() - start_time
        remaining = deadline - time.time()
        frac = max(0.002, 0.3 * remaining / (deadline - start_time))  # decreases to ~0.002

        i = rng.integers(n)
        ai = a[i]
        if ai < 1e-14:
            continue
        # choose j ≠ i
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = ai * rng.random() * frac
        if delta <= 0.0:
            continue

        # --- test the move on a copy of the convolution -----------------
        conv_test = conv.copy()
        # subtract contribution of the mass removed from i
        conv_test[i : i + n] -= 2.0 * delta * a
        # add contribution of the mass added at j
        conv_test[j : j + n] += 2.0 * delta * a
        # delta² terms (indices are always within [0, 2n‑2])
        conv_test[2 * i] += delta * delta
        conv_test[2 * j] += delta * delta
        conv_test[i + j] -= 2.0 * delta * delta

        new_max = conv_test.max()
        new_val = float(2.0 * n * new_max)

        # --- accept only improving moves ---------------------------------
        if new_val < cur_best_val:
            # commit the move
            a[i] = ai - delta
            a[j] += delta
            conv = conv_test
            cur_best_val = new_val
            cur_best_a = a.copy()
            no_improve = 0
            if move_counter % 5000 == 0:
                print(f"[Local] move {move_counter:07d} → new best {new_val:.6f}")
        else:
            no_improve += 1

        move_counter += 1

        # Periodically re‑normalise (to eliminate tiny drift) and recompute conv
        if move_counter % 100_000 == 0:
            a_sum = a.sum()
            if a_sum <= 0.0:
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = a / a_sum
            conv = np.convolve(a, a)
            cur_best_val = _objective(a)
            cur_best_a = a.copy()
            print(f"[Re‑norm] after {move_counter} moves value {cur_best_val:.6f}")

        # Light escape mechanism – if stuck for a long time, restart from a random
        if no_improve > 300_000 and time.time() < deadline - 10.0:
            # generate a fresh random candidate (biased towards edge shapes)
            gamma = 10 ** rng.uniform(-1, 2)
            weight = rng.random()
            a = _edge_exponential_shape(n, gamma, weight)
            conv = np.convolve(a, a)
            cur_best_val = _objective(a)
            cur_best_a = a.copy()
            no_improve = 0
            print(f"[Restart] after stagnation, new value {cur_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final clipping and return
    final_seq = np.clip(cur_best_a, 0.0, 1000.0)
    print(f"Final best value {cur_best_val:.6f} (length {len(final_seq)})")
    return final_seq.tolist()
