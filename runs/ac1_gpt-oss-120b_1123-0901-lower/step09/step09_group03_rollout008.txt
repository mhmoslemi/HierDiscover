def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters, eta0, method):
    a = np.maximum(a0, 0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c   # sum =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute active set
        active = (conv >= max_c - 1e-14)   # tolerance
        # Note: use these indices for w
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)
        if method == 'exp':
            a = a * np.exp(-eta * grad)
        elif method == 'proj':
            a = a - eta * grad
            a = np.maximum(a, 0)
        # Project to simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def _greedy_smoothing_optimize(a0, n, conv_fft_len, deadline, max_moves=50000, epsilon=0.01):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        k = int(conv.argmax())
        # compute contributions for pairs that sum to k
        i_start = max(0, k-(n-1))
        i_end = min(n-1, k)
        idx = np.arange(i_start, i_end+1)
        prod = a[idx] * a[k-idx]
        # choose the largest product
        i_src = idx[np.argmax(prod)]
        j_src = k - i_src
        # compute delta as small fraction epsilon of min(a_i,a_j)
        delta = min(a[i_src], a[j_src]) * epsilon
        if delta < 1e-14:
            it += 1
            continue
        # pick random destination indices different from i_src/j_src
        t1 = np.random.randint(n)
        while t1 == i_src or t1 == j_src:
            t1 = np.random.randint(n)
        t2 = np.random.randint(n)
        while t2 == i_src or t2 == j_src or t2 == t1:
            t2 = np.random.randint(n)
        # move delta from both src to destinations
        a[i_src] -= delta
        a[j_src] -= delta
        a[t1] += delta/2
        a[t2] += delta/2
        # ensure nonnegative
        a = np.maximum(a, 0.0)
        # renorm
        s = a.sum()
        if s <= 0:
            a = np.full(n, 1.0/n)
        else:
            a /= s
        # evaluate new value
        conv_new = _full_convolution(a, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val:
            best_val = val_new
            best_a = a.copy()
            # optionally accept move
        else:
            # revert? We keep new a anyway maybe risk.
            # We'll accept only if improvement else revert.
            a = best_a.copy()
        it += 1
    return best_a, best_val

   longer Adam run is executed, then two long sub‑gradient phases
   (exponentiated‑gradient and a projected‑gradient variant) are
   performed.  Finally a pair‑wise targeted annealing step removes the
   remaining peaks.

4. **Cleanup.**  The best vector is clipped to the range accepted by
   ``evaluate_sequence`` and returned as a plain Python list.

All steps respect the global time budget (default 1000 s) and always
return a feasible sequence, so the runner never receives ``-inf``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for sub‑gradient:
    returns vector g where g_i = ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate
        L_λ(a) = log‑sum‑exp( λ·(a∗a) )
    Returns the best vector seen (normalised to sum = 1) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                               # learning‑rate
    lam = 0.08                               # initial temperature
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → probability vector on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # ∑a = 1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) update
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase temperature (sharpen the max)
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when we are close to the deadline
        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on the simplex) sub‑gradient
    optimiser for the true non‑smooth objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_optimize_projgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Projected Euclidean gradient descent on the simplex (instead of
    exponentiated‑gradient).  Sometimes this direction explores a
    different part of the landscape.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30000 == 0:
                print(f"[ProjGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted annealing.
    At each step the index k where (a∗a) is maximal is located.
    Among the pairs (i, j) with i+j=k we pick the pair with greatest product
    a_i·a_j and move a small fraction of mass away from that pair.
    Acceptance follows a temperature schedule.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        # Linear temperature decay
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # Find the pair (i,j) on the active diagonal with largest product
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # single‑index case: move a small fraction elsewhere
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05   # 5 % move
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # pair case: remove a small amount from both ends simultaneously
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        # Numerical safety
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors for the coarse stage."""
    cand = []

    # Warm start (if available)
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Multi‑resolution optimiser.  Returns a list of non‑negative heights
    (length 4096) that gives a low value of ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse stage (n=256)
    n0 = 256
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # warm start from a previous solution, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] > n0:
            warm_arr = warm_arr[:n0]
        elif warm_arr.shape[0] < n0:
            warm_arr = np.pad(warm_arr, (0, n0 - warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() >= 0.01:
            warm_arr = warm_arr / warm_arr.sum()
        else:
            warm_arr = None
    else:
        warm_arr = None

    pool = _initial_candidates(n0, warm_arr) + _spike_candidates(n0)

    best_a0 = None
    best_val0 = float("inf")
    stage1_deadline = start_time + (deadline - start_time) * 0.20

    # Run cheap sub‑gradient runs on many starts
    while time.time() < stage1_deadline and pool:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n0))
        a0 = a0 / a0.sum()

        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0, n0, conv_fft_len0, corr_fft_len0,
            stage1_deadline,
            max_iters=30_000,
            eta0=1.0,
        )
        if val_opt < best_val0:
            best_val0 = val_opt
            best_a0 = a_opt.copy()
            print(f"[Coarse] improved to {val_opt:.6f}")

        # occasional short Adam sweep
        if time.time() < stage1_deadline:
            a_adam, val_adam = _adam_optimize(
                best_a0, n0, conv_fft_len0, corr_fft_len0,
                min(stage1_deadline, time.time() + 2.0),
                max_iters=50_000,
            )
            if val_adam < best_val0:
                best_val0 = val_adam
                best_a0 = a_adam.copy()
                print(f"[Coarse‑Adam] improved to {val_adam:.6f}")

    # fallback if nothing succeeded
    if best_a0 is None:
        best_a0 = np.full(n0, 1.0 / n0, dtype=np.float64)
        best_val0 = 2.0 * n0 * np.max(_full_convolution(best_a0, conv_fft_len0))

    # ------------------------------------------------------------------
    # 2. Medium resolution (n=1024)
    n1 = 1024
    conv_fft_len1 = _next_pow2(2 * n1 - 1)
    corr_fft_len1 = _next_pow2(3 * n1 - 2)

    # Linear interpolation from coarse optimum
    x0 = np.linspace(0, 1, n0, endpoint=False)
    x1 = np.linspace(0, 1, n1, endpoint=False)
    a1 = np.interp(x1, x0, best_a0)
    a1 = np.maximum(a1, 0.0)
    a1 = a1 / a1.sum()

    stage2_deadline = start_time + (deadline - start_time) * 0.55

    # Short Adam refinement
    if time.time() < stage2_deadline:
        a1_adam, val1_adam = _adam_optimize(
            a1, n1, conv_fft_len1, corr_fft_len1,
            min(stage2_deadline, time.time() + 5.0),
            max_iters=200_000,
        )
        if val1_adam < best_val0:
            best_val0 = val1_adam
            a1 = a1_adam.copy()
            print(f"[Medium‑Adam] improved to {val1_adam:.6f}")

    # Longer exponentiated‑gradient sub‑gradient
    if time.time() < stage2_deadline:
        a1_sub, val1_sub = _subgradient_optimize_expgrad(
            a1, n1, conv_fft_len1, corr_fft_len1,
            stage2_deadline,
            max_iters=300_000,
            eta0=0.9,
        )
        if val1_sub < best_val0:
            best_val0 = val1_sub
            a1 = a1_sub.copy()
            print(f"[Medium‑Sub] improved to {val1_sub:.6f}")

    # ------------------------------------------------------------------
    # 3. Fine resolution (n=4096)
    n2 = 4096
    conv_fft_len2 = _next_pow2(2 * n2 - 1)
    corr_fft_len2 = _next_pow2(3 * n2 - 2)

    # Upsample
    x2 = np.linspace(0, 1, n2, endpoint=False)
    a2 = np.interp(x2, x1, a1)
    a2 = np.maximum(a2, 0.0)
    a2 = a2 / a2.sum()

    stage3_deadline = deadline - 30.0   # keep ~30 s for final polishing

    # Adam on full grid
    if time.time() < stage3_deadline:
        a2_adam, val2_adam = _adam_optimize(
            a2, n2, conv_fft_len2, corr_fft_len2,
            min(stage3_deadline, time.time() + 8.0),
            max_iters=300_000,
        )
        if val2_adam < best_val0:
            best_val0 = val2_adam
            a2 = a2_adam.copy()
            print(f"[Fine‑Adam] improved to {val2_adam:.6f}")

    # Exponentiated‑gradient sub‑gradient (long)
    if time.time() < stage3_deadline:
        a2_subexp, val2_subexp = _subgradient_optimize_expgrad(
            a2, n2, conv_fft_len2, corr_fft_len2,
            stage3_deadline,
            max_iters=400_000,
            eta0=0.85,
        )
        if val2_subexp < best_val0:
            best_val0 = val2_subexp
            a2 = a2_subexp.copy()
            print(f"[Fine‑Exp] improved to {val2_subexp:.6f}")

    # Projected‑gradient sub‑gradient (additional)
    if time.time() < stage3_deadline:
        a2_subproj, val2_subproj = _subgradient_optimize_projgrad(
            a2, n2, conv_fft_len2, corr_fft_len2,
            stage3_deadline,
            max_iters=250_000,
            eta0=0.7,
        )
        if val2_subproj < best_val0:
            best_val0 = val2_subproj
            a2 = a2_subproj.copy()
            print(f"[Fine‑Proj] improved to {val2_subproj:.6f}")

    # Pair‑wise targeted annealing (to knock down the remaining peak)
    if time.time() < deadline - 10:
        a2_pair, val2_pair = _pair_targeted_anneal_optimize(
            a2, n2, conv_fft_len2, deadline - 10,
            max_moves=150_000,
        )
        if val2_pair < best_val0:
            best_val0 = val2_pair
            a2 = a2_pair.copy()
            print(f"[Fine‑PairAnneal] improved to {val2_pair:.6f}")

    # ------------------------------------------------------------------
    # 4. Final sanity clip & return
    best_seq = np.clip(a2, 0.0, 1000.0)

    print(f"Final best value {best_val0:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
