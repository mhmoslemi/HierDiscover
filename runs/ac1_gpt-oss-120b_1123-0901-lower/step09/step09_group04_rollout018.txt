    # function to normalize vector to target_sum
    def _normalize(v):
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            v = np.full_like(v, 1.0 / v.shape[0])
            s = v.sum()
        return v * (target_sum / s)

    best_val = float('inf')
    best_seq = None

    # initial deterministic families
    families = []

"""
Propose a low‑value step‑function for the given evaluation.

The search works in four stages:

1. **Broad warm‑start** – a large pool of deterministic families
   (uniform, sinusoid, edge‑exponential / power‑law, beta, double‑Gaussian)
   plus many random monotone / Dirichlet draws.  All candidates are
   normalised to the target L1‑norm √(2 n) and the best becomes the seed.

2. **Smooth‑max Adam optimisation** – the exact max‑convolution is
   approximated by a log‑sum‑exp surrogate and Adam is run for a
   time‑aware fraction of the remaining budget (larger learning‑rate
   and longer horizon than the reference).

3. **Accelerated projected‑gradient descent (PGD)** – a sub‑gradient of
   the true max‑convolution is used together with Nesterov momentum.
   The step‑size is larger (η₀≈1.2) and the method runs until the
   global deadline.

4. **Final cheap hill‑climber** – many random mass‑transfer moves are
   performed; improving moves are kept.  The routine always remembers
   the best vector seen and returns it as a plain Python list.

All convolutions are performed by FFT on a power‑of‑two length for speed.
The returned sequence complies with the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:      # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------

    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _sorted_dirichlet(n: int, alpha: float) -> np.ndarray:
    v = np.random.dirichlet(np.full(n, alpha))
    v.sort()
    return v[::-1]   # decreasing


def _random_monotone(n: int) -> np.ndarray:
    v = np.sort(np.random.rand(n))[::-1]   # decreasing
    if v.sum() < 1e-12:
        v = np.full(n, 1.0 / n)
    else:
        v = v / v.sum()
    return v


def _decreasing_exponential(n: int, gamma: float) -> np.ndarray:
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _stretched_exponential(n: int, gamma: float, p: float) -> np.ndarray:
    t = (np.arange(n, dtype=np.float64) / (n - 1)) ** p
    shape = np.exp(-gamma * t)
    shape /= shape.sum()
    return shape


def _power_law_decreasing(n: int, p: float) -> np.ndarray:
    shape = 1.0 / ((np.arange(1, n + 1, dtype=np.float64)) ** p)
    shape /= shape.sum()
    return shape


def _random_sparse(n: int, k: int, alpha: float = 1.0) -> np.ndarray:
    """k non‑zero entries placed uniformly at random."""
    idx = np.random.choice(n, size=k, replace=False)
    vals = np.random.dirichlet(np.full(k, alpha))
    res = np.zeros(n, dtype=np.float64)
    res[idx] = vals
    return res


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse deterministic pool."""
    pool = []

    # Dirichlet draws (different concentrations)
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool


# ----------------------------------------------------------------------
#  Fast exact objective (same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

#  Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns best vector (∑=√(2 n)) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # temperature (inverse)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max gives a point on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the very end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Accelerated projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum() * np.sqrt(2.0 * n)      # enforce the target sum

    best_a = a.copy()
    best_val = float("inf")
    v = np.zeros_like(a)   # momentum term
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Nesterov momentum (FISTA‑style)
        mu = it / (it + 3.0)
        y = a + mu * v

        # gradient step from the extrapolated point
        eta = eta0 / np.sqrt(it + 1)
        a_next = y - eta * grad
        a_next = _project_onto_simplex(a_next) * np.sqrt(2.0 * n)

        v = a_next - a
        a = a_next

        # occasional jitter when time is low
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum() * np.sqrt(2.0 * n)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_onto_simplex(a2) * np.sqrt(2.0 * n)

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem size – a larger n gives more flexibility
    n = 8192
    target_sum = np.sqrt(2.0 * n)

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (optional external variable)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum() * target_sum
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build an initial candidate pool
    # ------------------------------------------------------------------
    pool = []

    # deterministic families (including a few extra shapes)
    pool.extend(_warm_and_shifts(warm, n) if warm is not None else [])
    pool.extend(_initial_pool(n, rng))

    # additional deterministic shapes
    for gamma in (0.01, 0.02, 0.05, 0.1, 0.2):
        pool.append(_decreasing_exponential(n, gamma))
    for gamma in (0.01, 0.05, 0.1):
        for p in (0.5, 1.0, 2.0, 3.0):
            pool.append(_stretched_exponential(n, gamma, p))
    for p in (0.5, 1.0, 1.5, 2.0, 3.0):
        pool.append(_power_law_decreasing(n, p))

    # random Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, alpha))
            pool.append(_sorted_dirichlet(n, alpha))

    # random monotone vectors
    for _ in range(30):
        pool.append(_random_monotone(n))

    # random sparse vectors
    for k in (8, 16, 32):
        for _ in range(6):
            pool.append(_random_sparse(n, k, alpha=1.0))

    # ------------------------------------------------------------------
    # Scan the pool (fast exact evaluation)
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a * (target_sum / s)        # normalise to √(2 n)

        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(0.35 * remaining, 250.0)   # up‑to 35 % of the rest
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Accelerated PGD on the exact max‑convolution
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=1_200_000,
            eta0=1.4,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final cheap hill‑climber
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, target_sum / n, dtype=np.float64)

    # Clip to the limits accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out
