def _anneal_search(a, n, conv_fft_len, deadline):
    conv = _full_convolution(a, conv_fft_len)
    best_obj = 2 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    T0 = 1.0   # initial temperature
    T_min = 1e-6
    decay_rate = 0.9999 (makes 500k steps reach near min)
    step = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-15: continue
        j = np.random.randint(n)
        if i == j: continue
        delta = a[i] * np.random.rand() * 0.5  # up to 50% of i
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        obj2 = 2 * n * conv2.max() / (a2.sum() ** 2)
        if obj2 < best_obj or np.random.rand() < np.exp(-(obj2 - best_obj)/T):
            a = a2
            conv = conv2
            if obj2 < best_obj:
                best_obj = obj2
                best_a = a2.copy()
        # update temperature as geometric schedule:
        T = max(T_min, T * decay_rate)
        step += 1
    return best_a, best_obj

Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

Algorithm
---------

   dense sweep over their parameters and many Dirichlet draws.
   All vectors are normalised to sum = 1.  The best `top_k` vectors are
   kept.

   a projected gradient method on the *exact* max‑convolution objective.
   The sub‑gradient is obtained from the active set of shifts attaining
   the current maximum.  A simple back‑tracking line‑search together
   with a small momentum term is used.  The best‑found vector is
   allocated the largest iteration budget.

   runs on the current best vector.  Moves that improve the objective
   are always accepted; worsening moves are accepted with a
   temperature‑controlled probability, allowing the search to escape
   shallow local minima.

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum(a)^2).  a is assumed non‑negative."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max() / (a.sum() ** 2)

# Simple Euclidean projection onto the probability simplex (sum=1, ≥0)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the simplex {x ≥ 0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Deterministic families (all vectors are returned with sum == 1)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    v = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    v = np.maximum(v, 0.0)
    v /= v.sum()
    return v


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape

    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0, 10.0):
        for _ in range(4):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential – fairly dense grid
    for gamma in np.logspace(-2, 2, 20):      # 0.01 … 100
        pool.append(_edge_exponential(n, float(gamma)))

    # Edge‑power‑law – extended range
    for alpha in np.linspace(-8.0, 4.0, 25):
        pool.append(_edge_power_law(n, float(alpha)))

    # Beta pdf – grid in log‑space
    a_vals = np.logspace(-1, 1, 12)   # 0.1 … 10
    b_vals = np.logspace(-1, 1, 12)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    # Double‑Gaussian – fine sweep
    for sigma in np.linspace(0.02, 0.5, 20):
        pool.append(_double_gaussian(n, float(sigma)))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start (if any)."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out

# Projected sub‑gradient descent with a tiny momentum term
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Refine a candidate via projected sub‑gradient descent."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    step = 0.5          # initial step size
    momentum = np.zeros_like(a)   # simple momentum accumulator

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum within a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0.0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # momentum update (exponential moving average)
        momentum = 0.9 * momentum + 0.1 * grad

        # ----- back‑tracking line‑search -----
        step_try = step
        while step_try > 1e-12:
            a_new = a - step_try * momentum
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if val_new <= val:
                a = a_new
                # modestly increase step after a successful move
                step = min(1.0, step_try * 1.2)
                break
            step_try *= 0.5
        else:
            # could not improve – shrink the step size
            step *= 0.5

        it += 1

    return best_a, best_val

# Simulated‑annealing local optimiser
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber with a temperature schedule.
    Randomly moves a fraction of mass from one position to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    T = 1.0                # initial temperature
    T_min = 1e-6
    decay = 0.99995        # geometric decay per move
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (up to 30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        improve = val2 < best_val
        if improve or (np.random.rand() < np.exp(-(val2 - best_val) / max(T, 1e-12))):
            a = a2
            conv = conv2
            if improve:
                best_val = val2
                best_a = a2.copy()
                if it % 5000 == 0:
                    print(f"[Anneal it={it:06d}] improved to {val2:.6f}")

        T = max(T_min, T * decay)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point called by the judge.
    Returns the best sequence discovered within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well, but any power‑of‑two ≤ 4096 is allowed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Optional warm start (if the previous run stored a variable)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a very diverse pool of start vectors
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # ------------------------------------------------------------------
    # Quick scan of the whole pool – keep the best few candidates
    top_k = 30
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        scored.append((val, a))
        if time.time() >= deadline * 0.2:        # keep enough time for refinement
            break

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # Phase 2 – refined projected sub‑gradient descent
    best_seq = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        # allocate a larger time slice for the leading candidates
        remaining = deadline - time.time()
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        max_iters = 250_000 if rank == 0 else 120_000
        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=max_iters,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing polishing
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Anneal] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform if nothing was found (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits imposed by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
