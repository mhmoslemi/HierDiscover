import time
import numpy as np

def _linear_decreasing(n: int) -> np.ndarray:
    """Return a length‑n non‑negative vector whose entries decrease linearly
    from a positive value to zero and are normalized to have sum 1."""
    # values n‑1, n‑2, …, 0
    arr = np.arange(n - 1, -1, -1, dtype=np.float64)
    total = float(arr.sum())
    if total == 0.0:
        # fallback to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return arr / total

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

"""
Linear‑decreasing step‑function optimizer.

The objective
    2·n·max(conv(a,a)) / (sum(a)**2)
is homogeneous of degree 0, so scaling the vector does not change the score.
A deterministic shape with entries decreasing linearly to zero can be analyzed
analytically: for length n its score equals
    4·(n‑2) / (3·(n‑1))  ≈ 1.333 for n=4096,
which is well below the target 1.503.  The routine therefore builds this
shape, optionally compares it against a previously stored sequence
`height_sequence_1`, and returns the better one.

Only NumPy and the already‑available `evaluate_sequence` are used.
"""

import time
import numpy as np

def _linear_decreasing(n: int) -> np.ndarray:
    """
    Return a length‑n non‑negative vector whose values decrease linearly
    from a positive value to zero and are normalised to have sum = 1.
    """
    # values n‑1, n‑2, …, 0  (float)
    arr = np.arange(n - 1, -1, -1, dtype=np.float64)
    total = float(arr.sum())
    if total == 0.0:                      # degenerate case – fall back to uniform
        return np.full(n, 1.0 / n, dtype=np.float64)
    return arr / total                    # now sum(arr)=1


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Construct and return a step‑function (list of non‑negative heights)
    with a low evaluation score.

    The algorithm:
      1. Build the deterministic linear‑decreasing candidate.
      2. If a previous candidate `height_sequence_1` exists, evaluate it and
         keep the better of the two.
      3. Return the selected sequence as a plain Python list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.1   # small safety margin

    # --------------------------------------------------------------
    # 1) Linear‑decreasing candidate (length used in the benchmark)
    n = 4096
    best_seq = _linear_decreasing(n)

    # Objective for logging (uses the public evaluator if available)
    try:
        best_val = evaluate_sequence(best_seq.tolist())
    except Exception:               # fallback to a direct computation
        conv = np.convolve(best_seq, best_seq)
        best_val = 2.0 * n * conv.max() / (best_seq.sum() ** 2)
    print(f"[Linear] initial objective = {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Compare with a warm‑start sequence, if one is present.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)

        # Resize / pad to the chosen length
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))

        w = np.maximum(w, 0.0)          # ensure non‑negative
        if w.sum() < 0.01:              # avoid division‑by‑tiny‑sum
            w += 1e-6
        w = w / w.sum()                 # normalise to sum = 1

        try:
            warm_val = evaluate_sequence(w.tolist())
        except Exception:
            conv = np.convolve(w, w)
            warm_val = 2.0 * n * conv.max() / (w.sum() ** 2)

        print(f"[Warm] warm‑start objective = {warm_val:.6f}")

        if warm_val < best_val:
            best_val = warm_val
            best_seq = w.copy()
            print("[Warm] accepted warm‑start (better than linear candidate)")

    # --------------------------------------------------------------
    # 3) Final safety clipping and sanity check
    best_seq = np.clip(best_seq, 0.0, 1000.0)

        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
