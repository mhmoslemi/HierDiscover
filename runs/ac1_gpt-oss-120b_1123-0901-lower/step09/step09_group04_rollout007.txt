def _lp_step(a, M, n):
    # Build matrix C as sparse diag
    offsets = np.arange(n)
    data = [np.full(n, a[k], dtype=np.float64) for k in range(n)]
    C = scipy.sparse.diags(data, offsets, shape=(2*n-1, n), format='csc')
    # Some solvers require csc/ csr; hi-High may prefer csr. Use csr.
    C = C.tocsr()
    # objective: maximize sum(g) i.e., minimize -sum(g)
    c = -np.ones(n)
    b_ub = M * np.ones(2*n-1)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

                print(f"[LP it={lp_iter} adopt g] new best {val_new:.6f}")

        lp_iter += 1

    # At this point best_small found
    # Upsample to big n
    pos_small = (np.arange(n_small) + 0.5) / n_small
    pos_big = (np.arange(n_big) + 0.5) / n_big
    a_big = np.interp(pos_big, pos_small, best_small)
    a_big = np.maximum(a_big, 0.0)
    if a_big.sum() < 0.01:
        a_big = np.full(n_big, 1.0/n_big)
    else:
        a_big = a_big / a_big.sum()

    # Also evaluate other candidates for big n (e.g., warm_big, uniform) and keep best
    best_seq = a_big
    best_val = _objective(best_seq)
    # Check warm_big if exists
    if warm_big is not None:
        val_warm = _objective(warm_big)
        if val_warm < best_val:
            best_val = val_warm
            best_seq = warm_big.copy()
            print(f"[Warm] improved best {val_warm:.6f}")

    # Maybe also evaluate simple uniform for big n
    uniform_big = np.full(n_big, 1.0 / n_big)
    val_uni = _objective(uniform_big)
    if val_uni < best_val:
        best_val = val_uni
        best_seq = uniform_big.copy()
        print(f"[Uniform] improved best {val_uni:.6f}")

    # Now run PGD for remainder time
    if time.time() < deadline - 5:
        pgd_deadline = deadline - 5
        a_pgd, val_pgd = _pgd_optimize(best_seq, n_big,
                                      _next_pow2(2*n_big - 1),
                                      _next_pow2(3*n_big - 2),
                                      pgd_deadline,
                                      max_iters=500_000,
                                      eta0=0.9)
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd
            print(f"[PGD] new best {val_pgd:.6f}")

    # Final cheap hill-climber
    if time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(best_seq, n_big,
                                              _next_pow2(2*n_big - 1),
                                              deadline,
                                              max_moves=80_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc
            print(f"[Local] final best {val_loc:.6f}")

    # Ensure compliance
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a low‑value step‑function.

Algorithm (≈100 s of LP, the rest of the budget for gradient fine‑tuning).

1. Warm‑start & diverse deterministic families (Dirichlet, edge‑exponential,
   power‑law, beta, double‑Gaussian, sinusoid, uniform) on a modest size
   n_small = 1024.  The best vector from this pool is taken as the seed.

2. Linear‑programming refinement on the small size:
   * For the current a compute M = max(a*a).
   * Solve   maximise Σ g_j   subject to   a * g ≤ M   and   g ≥ 0.
     The convolution a*g is linear in g, therefore a standard LP.
   * Normalise the solution g, blend a←(1‑t)a + t g with a few trial
     step‑sizes t∈{1,0.5,0.2} and keep the best improvement.
   * Repeat the LP step a dozen times (or until the time budget for this
     phase runs out).  The LP uses a sparse Toeplitz matrix and the HiGHS
     solver (SciPy).

3. Upsample the improved small‑size vector to the target length
   n_big = 4096 (linear interpolation, renormalise).

4. Projected‑gradient descent (PGD) on the full size.  The exact max‑convolution
   gradient is computed via FFT‑based cross‑correlation, and Euclidean
   projection onto the probability simplex is applied each step.  The step‑size
   follows η₀/√t with η₀=0.9.

5. Final cheap hill‑climber: random mass‑transfer moves that are kept if they
   improve the exact objective.

All stages keep the best vector seen so far and return it as a plain Python
list.  The sequence is non‑negative, finite and respects the limits required
by ``evaluate_sequence``.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Objective (exact)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Candidate generators
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a deterministic starter pool."""
    pool = []
    # Dirichlet draws with several alphas
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))
    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))
    # Power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    # Beta pdfs
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))
    # Double Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    # Sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    return pool


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start exists, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


# ----------------------------------------------------------------------
#  Linear‑programming refinement
# ----------------------------------------------------------------------


def _lp_step(a: np.ndarray, M: float, n: int) -> np.ndarray | None:
    """
    Solve   max Σ g_j   s.t.   a * g ≤ M   and   g ≥ 0.
    Returns the optimal g (as a dense vector) or None on failure.
    """
    # Build the (2n‑1)×n convolution matrix of a (constant diagonals)
    offsets = np.arange(n, dtype=int)            # 0 … n‑1
    data = [np.full(n, a[k], dtype=np.float64) for k in range(n)]
    C = sp.diags(data, offsets, shape=(2 * n - 1, n), format='csc')

    c = -np.ones(n, dtype=np.float64)            # minimise -Σg   → maximise Σg
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    bounds = [(0, None)] * n

    try:
        res = linprog(c, A_ub=C, b_ub=b_ub, bounds=bounds, method='highs')
    except Exception as e:
        print("[LP] exception:", e)
        return None

    if not res.success:
        # Occasionally the LP can be infeasible due to numerical errors.
        # In that case just return None.
        return None
    return res.x


# ----------------------------------------------------------------------
#  Optimiser (Projected Gradient Descent)
# ----------------------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent directly on the exact max‑convolution.
    Returns the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        # occasional jitter near the end
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Cheap local hill‑climber
# ----------------------------------------------------------------------


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer moves; keep the move if it improves the exact objective.
    """
    a = a0.copy()
    best_val = _objective(a)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric used by ``evaluate_sequence``.
    The routine respects the time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    #  Problem sizes
    # ------------------------------------------------------------------
    n_big = 4096          # final resolution
    n_small = 1024        # LP stage works on a smaller vector

    # ------------------------------------------------------------------
    #  Warm start handling
    # ------------------------------------------------------------------
    warm_big = globals().get("height_sequence_1")
    if warm_big is not None:
        warm_big = np.asarray(warm_big, dtype=np.float64)
        if warm_big.shape[0] > n_big:
            warm_big = warm_big[:n_big]
        elif warm_big.shape[0] < n_big:
            warm_big = np.pad(warm_big, (0, n_big - warm_big.shape[0]))
        warm_big = np.maximum(warm_big, 0.0)
        if warm_big.sum() >= 0.01:
            warm_big = warm_big / warm_big.sum()
        else:
            warm_big = None
    else:
        warm_big = None

    # Down‑sample warm start for the small problem (if any)
    if warm_big is not None:
        pos_big = (np.arange(n_big) + 0.5) / n_big
        pos_small = (np.arange(n_small) + 0.5) / n_small
        warm_small = np.interp(pos_small, pos_big, warm_big)
        warm_small = np.maximum(warm_small, 0.0)
        if warm_small.sum() >= 0.01:
            warm_small = warm_small / warm_small.sum()
        else:
            warm_small = None
    else:
        warm_small = None

    # ------------------------------------------------------------------
    #  Stage 0 – initial pool (small size)
    # ------------------------------------------------------------------
    pool_small = _warm_and_shifts(warm_small, n_small) + _initial_pool(n_small, rng)

    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0):
        for _ in range(4):
            pool_small.append(_random_dirichlet(n_small, alpha))

    best_small = None
    best_val_small = float("inf")
    for cid, cand in enumerate(pool_small):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n_small))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val_small:
            best_val_small = val
            best_small = a.copy()
            print(f"[Init small {cid}] best {val:.6f}")

    if best_small is None:
        best_small = np.full(n_small, 1.0 / n_small)
        best_val_small = _objective(best_small)

    # ------------------------------------------------------------------
    #  Stage 1 – LP refinement on the small problem
    # ------------------------------------------------------------------
    lp_deadline = deadline - 30          # keep ~30 s for later phases
    max_lp_iters = 12
    lp_iter = 0
    while time.time() < lp_deadline and lp_iter < max_lp_iters:
        conv_small = _full_convolution(best_small, _next_pow2(2 * n_small - 1))
        M = conv_small.max()
        g = _lp_step(best_small, M, n_small)
        if g is None:
            break
        # normalise the LP solution
        g = g / g.sum()

        improved = False
        for t in (1.0, 0.5, 0.2):
            a_new = (1.0 - t) * best_small + t * g
            a_new = a_new / a_new.sum()
            val_new = _objective(a_new)
            if val_new < best_val_small:
                best_val_small = val_new
                best_small = a_new
                improved = True
                print(f"[LP it={lp_iter} t={t}] new best {val_new:.6f}")
                break
        if not improved:
            # fall back to pure LP solution if it is better on its own
            val_g = _objective(g)
            if val_g < best_val_small:
                best_val_small = val_g
                best_small = g
                print(f"[LP it={lp_iter} adopt g] new best {val_g:.6f}")

        lp_iter += 1

    # ------------------------------------------------------------------
    #  Stage 2 – up‑sample to the final size
    # ------------------------------------------------------------------
    pos_small = (np.arange(n_small) + 0.5) / n_small
    pos_big = (np.arange(n_big) + 0.5) / n_big
    a_big = np.interp(pos_big, pos_small, best_small)
    a_big = np.maximum(a_big, 0.0)
    if a_big.sum() < 0.01:
        a_big = np.full(n_big, 1.0 / n_big)
    else:
        a_big = a_big / a_big.sum()

    best_seq = a_big
    best_val = _objective(best_seq)

    # compare with the (optional) original warm start
    if warm_big is not None:
        val_warm = _objective(warm_big)
        if val_warm < best_val:
            best_val = val_warm
            best_seq = warm_big.copy()
            print(f"[Warm] improved best {val_warm:.6f}")

    # ------------------------------------------------------------------
    #  Stage 3 – Projected‑gradient descent on the full size
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        pgd_deadline = deadline - 5
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            pgd_deadline,
            max_iters=500_000,
            eta0=0.9,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd
            print(f"[PGD] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    #  Stage 4 – final cheap hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_big,
            conv_fft_len_big,
            deadline,
            max_moves=80_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    #  Safety & final output
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
