"""
Hybrid optimizer – random‑parameter seed + deep continuous refinement.

* Phase 0 – warm start (if a previous solution exists).
* Phase 1 – very fast random search in several low‑dimensional families
  (asymmetric exponentials, power‑law, piece‑wise constant, Dirichlet).
  Each candidate is evaluated by a cheap FFT convolution, keeping the best
  few vectors.
* Phase 2 – a long Adam run on a smooth‑max surrogate, starting from the
  best random seed.
* Phase 3 – exponentiated‑gradient sub‑gradient descent on the exact max‑
  convolution (the `max` is non‑smooth, we use an active‑set sub‑gradient).
* Phase 4 – projected gradient descent on the exact objective.
* Phase 5 – simulated‑annealing “local search’’ for final polishing.
* Optional Phase 6 – up‑sample to 8192 points and a short Adam pass
  (if time permits).

All iterates are kept on the probability simplex (∑a = 1) because the
evaluation function is homogeneous; only non‑negative values are used.
The routine respects a 1000 s budget and always returns the best
sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility routines
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection onto the probability simplex {x≥0, Σx=1}.
    """
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation primitives (Adam, sub‑gradient, GD, anneal)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max re‑parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2                # initial temperature for smooth‑max
    lam_factor = 1.018

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # Soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact max‑convolution value (used for logging & objective)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the smooth‑max (exact for the surrogate)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 60_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # Active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Uses a diminishing step size and keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _make_asym_exponential(n, gamma_l, gamma_r, w_u, w_l_frac):
    """
    a_i = wL * exp(-γL * i) + wR * exp(-γR * (n-1-i)) + wU * 1
    where wL = (1-wU) * w_l_frac, wR = (1-wU) * (1-w_l_frac)
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma_l * i)
    right = np.exp(-gamma_r * (n - 1 - i))
    wL = (1.0 - w_u) * w_l_frac
    wR = (1.0 - w_u) * (1.0 - w_l_frac)
    a = wL * left + wR * right + w_u * np.ones_like(left)
    return a


def _make_powerlaw_symmetric(n, p):
    """
    a_i = (i+1)^{-p} + (n-i)^{-p}
    """
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** (-p) + (n - i) ** (-p)
    return a


def _make_piecewise_constant(n, blocks):
    """
    Random block masses, uniform within each block.
    """
    block_len = n // blocks
    a = np.zeros(n, dtype=np.float64)
    masses = np.random.dirichlet(np.ones(blocks))
    for b in range(blocks):
        a[b * block_len:(b + 1) * block_len] = masses[b] / block_len
    # If n not divisible by blocks we leave the tail zero – negligible effect.
    return a


def _random_dirichlet(n, alpha):
    """Simple Dirichlet draw."""
    return np.random.dirichlet(alpha * np.ones(n))

        else:
            # pure Dirichlet draw with random concentration
            alpha = np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])
            cand = _random_dirichlet(n, alpha)

        # Normalise to the simplex (avoids divide‑by‑zero later)
        s = cand.sum()
        if s < 1e-12:
            continue
        cand = cand / s

        # Fast evaluation (exact, but cheap via FFT)
        conv = _full_convolution(cand, conv_fft_len)
        val = 2.0 * n * conv.max()

        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Random it={it:07d}] new best {val:.6f}")

        it += 1

    print(f"[Random] Completed {it} samples, best value {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Adam on smooth‑max surrogate (long run)
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq is not None:
        adam_dead = time.time() + remaining * 0.40   # ~40 % of what remains
        print("[Adam] Starting long Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_dead,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – exact sub‑gradient (exponentiated‑gradient)
    remaining = deadline - time.time()
    if remaining > 20.0 and best_seq is not None:
        sub_dead = time.time() + remaining * 0.45   # roughly half of what’s left
        print("[Subgrad] Starting exponentiated‑gradient phase")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=900_000, eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – projected gradient descent on exact max‑conv
    remaining = deadline - time.time()
    if remaining > 15.0 and best_seq is not None:
        gd_dead = time.time() + remaining * 0.30
        print("[GD] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, gd_dead,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase 5 – short annealing refinement
    remaining = deadline - time.time()
    if remaining > 10.0 and best_seq is not None:
        ann_dead = time.time() + remaining * 0.60   # use most of the rest
        print("[Anneal] Simulated‑annealing local search")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, ann_dead,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase 6 – optional up‑sampling to a finer grid
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq is not None:
        try:
            print("[Upsample] Interpolating to 8192 points")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            big_dead = time.time() + min(remaining - 5.0, 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                big_dead, max_iters=400_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best value {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check (optional – can be removed if time‑critical)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print("Final evaluation failed – returning best known sequence.")
    return best_seq.tolist()
