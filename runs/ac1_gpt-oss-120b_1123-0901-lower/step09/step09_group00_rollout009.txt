"""
Hybrid optimizer with extended search.

The routine builds a richer pool of initial step‑functions (uniform,
Dirichlet draws, exponential, power‑law, sinusoid, triangular,
double‑Gaussian, cosine, random sparse spikes, piece‑wise blocks and an optional
warm‑start).  Each candidate is refined by a long Adam run on a smooth‑max
surrogate, followed by an exponentiated‑gradient sub‑gradient phase.
The current best vector is then repeatedly *jittered* and re‑optimised,
interleaved with a plain projected gradient descent on the exact max‑convolution
and a simulated‑annealing local search.  Finally, if time permits,
the best solution is up‑sampled to a finer grid (8192 points) and a short
Adam pass is applied.  The algorithm keeps all vectors on the probability
simplex (∑a_i=1) and enforces mirror symmetry a_i = a_{n‑1‑i}.  It respects the
global time budget (≈ 1000 s) and always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _symmetrize(v: np.ndarray) -> np.ndarray:
    """Force mirror symmetry: v[i] = v[-i-1]."""
    return (v + v[::-1]) * 0.5


def _normalize(v: np.ndarray) -> np.ndarray:
    """Make non‑negative and project onto the simplex."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s < 1e-12:
        v = np.full_like(v, 1.0 / len(v))
    else:
        v = v / s
    return v

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2                # temperature for smooth‑max
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 60_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    best_a = _symmetrize(best_a)
    best_a = _normalize(best_a)
    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    best_a = _symmetrize(best_a)
    best_a = _normalize(best_a)
    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    best_a = _symmetrize(best_a)
    best_a = _normalize(best_a)
    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.5
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    best_a = _symmetrize(best_a)
    best_a = _normalize(best_a)
    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families (product i·(n‑i+1))
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.12, 0.2, 0.35):
        shift = int(0.25 * n)
        g1 = np.exp(-0.5 * ((i - shift) / (sigma * n)) ** 2)
        g2 = np.exp(-0.5 * ((i - (n - 1 - shift)) / (sigma * n)) ** 2)
        shape = g1 + g2
        shape /= shape.sum()
        cand.append(shape)

    # Raised cosine (half‑wave)
    shape = 0.5 * (1 - np.cos(2 * np.pi * np.arange(n) / (n - 1)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Half‑wave sinusoid (already present in original code)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (8, 16, 32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (12, 20, 30):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _jitter(a: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian noise and re‑project onto the simplex."""
    b = a + np.random.normal(scale=scale, size=a.shape)
    b = np.maximum(b, 0.0)
    b = _normalize(b)
    b = _symmetrize(b)
    return b

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with minimal evaluation score.

    The optimizer respects the given time budget (≈ 1000 s) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    safety = 0.5
    deadline = time.time() + max(0.0, budget_s - safety)

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start if a previous solution exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = _normalize(warm)
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a diverse initial pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # add a few random shuffled Dirichlet variants and piece‑wise blocks
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        a = _normalize(a)
        pool.append(a)

    # --------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = min(12, len(pool))

    # --------------------------------------------------------------
    # Phase I – deep optimisation of a few promising starts
    for idx, a0 in enumerate(pool):
        if time.time() >= deadline or idx >= max_candidates:
            break

        a0 = _normalize(a0)

        # allocate a slice of the remaining budget (roughly 30 s per start)
        sub_dead = min(deadline, time.time() + 30.0)

        # 1) Adam on smooth‑max
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=600_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient phase
        if time.time() < deadline - 5.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=800_000, eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase II – iterative jitter‑and‑re‑optimise loop
    while best_seq is not None and time.time() < deadline - 40.0:
        # each loop consumes ~15 s (Adam)
        jittered = _jitter(best_seq, scale=0.015)
        sub_dead = min(deadline, time.time() + 15.0)
        a_jit, val_jit = _adam_optimize(
            jittered, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000,
        )
        if val_jit < best_val:
            best_val = val_jit
            best_seq = a_jit.copy()
            print(f"[Jitter] improved to {val_jit:.6f}")

    # --------------------------------------------------------------
    # Phase III – fine‑tune the current best with projected gradient descent
    if best_seq is not None and time.time() < deadline - 15.0:
        print("[Fine] Projected gradient descent on exact max‑conv")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase IV – stochastic local search
    if best_seq is not None and time.time() < deadline - 10.0:
        print("[Anneal] Final simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=180_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase V – optional up‑sampling to a finer grid
    if best_seq is not None and time.time() + 30.0 < deadline:
        try:
            print("[Upsample] Attempting finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation of the current best
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = _normalize(a_big)
            a_big = _symmetrize(a_big)

            sub_dead_big = min(deadline, time.time() + 20.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=400_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")

            # a short gradient‑descent refinement at the high resolution
            if time.time() < deadline - 5.0:
                a_big, val_big = _max_conv_descent(
                    best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                    deadline, max_iters=150_000,
                )
                if val_big < best_val:
                    best_val = val_big
                    best_seq = a_big.copy()
                    print(f"[Upsample] GD refined best {val_big:.6f}")

        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # --------------------------------------------------------------
    # Ensure a non‑empty return and clip to evaluator limits
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final logging (evaluate once for a clean printout)
    try:
        final_val = evaluate_sequence(best_seq.tolist())
        print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    except Exception:
        print(f"Final best (un‑evaluated) value {best_val:.6f}")

    return best_seq.tolist()
