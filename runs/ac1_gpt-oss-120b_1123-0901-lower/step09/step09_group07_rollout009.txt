def _lp_refine(a, n, conv_fft_len, deadline, max_steps=10, tol=1e-12):
    for step in range(max_steps):
        if time.time() > deadline: break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # identify active indices
        active_idx = np.where(conv >= max_c - 1e-12)[0]
        if len(active_idx) == 0:
            break
        # Build constraint matrix
        S = len(active_idx)
        A = np.zeros((S, n))
        for row_idx, k in enumerate(active_idx):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            # j runs from start to end
            # g_j coefficient = a_{k - j}
            # Compute slice
            ks = k - np.arange(start, end+1)
            A[row_idx, start:end+1] = a[ks]
        # constraints: M = max_c
        b = np.full(S, max_c)
        # set objective: minimize L2 distance from a maybe? We'll do to maximize sum of weights times something.
        # Use objective -c^T g with c = a (to move away from current a)
        c = -a  # maximize inner product with -a? Actually maximize -a·g means minimize a·g -> move mass to positions where a is small.
        # Solve LP
        from scipy.optimize import linprog
        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            break
        g = res.x
        s = g.sum()
        if s < 1e-12:
            break
        g = g / s  # normalize to sum 1
        # mixing parameter t decreasing with step
        t = 0.2  # could be decreasing
        a = (1 - t) * a + t * g
        # re-normalize to sum 1 (should remain)
        a = a / a.sum()
    return a

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Rich random sampling** – generate candidates from many parametric families,
   extending the original set with inverse‑index, logarithmic, and extreme beta
   shapes (including a,b<1).  Every candidate is normalised to the probability
   simplex (sum≈1) and evaluated with an FFT‑based objective.  The best is kept.
2. **Linear‑program refinement** – using the current best vector `a`,
   build the set of “active’’ convolution constraints (those attaining the
   maximal value).  Solve a linear programme that tries to move probability
   mass away from the large entries of `a` while respecting the active
   constraints.  The solution is mixed back into `a` (mirroring the LP‑step
   described in the problem statement).  A few LP iterations are performed.
3. **Gradient optimisation** – run three first‑order methods on the exact
   smooth‑max surrogate of the objective: Adam, exponentiated‑gradient
   sub‑gradient, and Euclidean projected gradient descent.  They use FFT for
   fast convolution and run until the remaining time is exhausted.
4. **Final stochastic hill‑climbing** – a cheap mass‑transfer local search
   (randomly move a fraction of probability mass between two indices) for the
   last seconds, accepting only improvements.

All phases share a common `n = 4096` (fast enough for the 1000 s budget) but
the code can be adapted to larger `n`.  The final sequence is clipped to the
range accepted by `evaluate_sequence` and returned as a plain list.
"""

import time
import numpy as np
from scipy.optimize import linprog


# ----------------------------------------------------------------------
# FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact (clamped) objective used for short‑circuit evaluation.
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) – identical to evaluate_sequence up to clamping."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Parametric shape generators (all return a normalised probability vector)
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    return s / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _inv_index_shape(n: int) -> np.ndarray:
    shape = 1.0 / (np.arange(1, n + 1, dtype=np.float64))
    ss = shape.sum()
    return shape / ss


def _logarithmic_shape(n: int) -> np.ndarray:
    # avoid log(1)=0 by starting at 2
    shape = 1.0 / np.log(np.arange(2, n + 2, dtype=np.float64))
    ss = shape.sum()
    return shape / ss


def _arc_beta_shape(n: int, a: float, b: float) -> np.ndarray:
    # same as _beta_shape but with a,b<1 (arc‑sine‑like)
    return _beta_shape(n, a, b)


def _mixture_two_shapes(shape1: np.ndarray, shape2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination of two normalised shapes."""
    mix = w * shape1 + (1.0 - w) * shape2
    ss = mix.sum()
    return mix / ss if ss > 0 else np.full_like(mix, 1.0 / mix.shape[0])

# Optimisers (identical to the reference implementation)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.02   # slower rise – more stable early on

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random perturbation near deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step‑size
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Linear‑program refinement (LP step from the reference description)
def _lp_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_steps: int = 8,
) -> np.ndarray:
    """
    Perform a few LP‑based refinement steps.
    For each step we identify the active convolution constraints (those attaining the
    current maximal value) and solve a linear programme that pushes probability mass
    away from the large entries of `a` while respecting those constraints.
    The resulting vector is mixed back into `a`.
    """
    for step in range(max_steps):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active_idx = np.where(conv >= max_c - 1e-12)[0]
        if active_idx.size == 0:
            break

        S = active_idx.size
        # Build constraint matrix: each row k corresponds to (a * g)_k ≤ max_c
        A = np.zeros((S, n), dtype=np.float64)
        for row, k in enumerate(active_idx):
            start = max(0, k - (n - 1))
            end = min(k, n - 1)
            # positions j in [start, end] have coefficient a_{k‑j}
            ks = k - np.arange(start, end + 1, dtype=np.int64)
            A[row, start : end + 1] = a[ks]

        b = np.full(S, max_c, dtype=np.float64)

        # Objective: minimise dot(a, g) – i.e. move mass to indices where a is small.
        c = a.astype(np.float64)  # linprog minimises cᵀx, so we use +a.
        # We actually want to *minimise* a·g, therefore we keep c = a.
        res = linprog(c, A_ub=A, b_ub=b,
                      bounds=[(0, None)] * n,
                      method='highs',
                      options={'presolve': True})
        if not res.success:
            # If LP fails (numerical issues) we abort further LP steps.
            break

        g = res.x
        total = g.sum()
        if total < 1e-12:
            break
        g = g / total  # renormalise to a probability vector

        # Mix the new direction with a modest step size.
        t = 0.25 if step == 0 else 0.15
        a = (1 - t) * a + t * g
        a = a / a.sum()

    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (previous best stored globally)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Phase 1 – extensive random‑parameter sweep
    best_seq = None
    best_val = float("inf")

    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "sin_pow", "tent", "dirichlet", "inv_index",
        "logarithmic", "arc_beta"
    ]
    # time allocated to Phase 1: ~25 % of the remaining budget
    phase1_deadline = min(deadline - 5.0, time.time() + 0.25 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-3, 3)           # 1e‑3 … 1e3
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-20.0, -0.05)          # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 2)              # 0.01 … 100
            b = 10 ** rng.uniform(-2, 2)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 8.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        elif fam == "inv_index":
            cand = _inv_index_shape(n)
        elif fam == "logarithmic":
            cand = _logarithmic_shape(n)
        elif fam == "arc_beta":
            a = 10 ** rng.uniform(-2, -0.1)   # 0.01 … 0.8
            b = 10 ** rng.uniform(-2, -0.1)
            cand = _arc_beta_shape(n, a, b)
        else:
            continue

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

        # Occasionally try a random convex combination of the current best and a fresh candidate
        if best_seq is not None and rng.random() < 0.03:
            w = rng.random()
            combo = _mixture_two_shapes(best_seq, cand, w)
            valc = _objective(combo, conv_fft_len)
            if valc < best_val:
                best_val = valc
                best_seq = combo.copy()
                print(f"[Phase 1] combo improved to {valc:.6f}")

    # Try a few circular shifts of a warm‑start vector (if any)
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – LP‑based refinement (if we have a decent candidate)
    if best_seq is not None and time.time() < deadline - 30:
        print("Phase 2 – LP refinement")
        a_lp = _lp_refine(best_seq, n, conv_fft_len, deadline - 20, max_steps=8)
        val_lp = _objective(a_lp, conv_fft_len)
        if val_lp < best_val:
            best_val, best_seq = val_lp, a_lp.copy()
            print(f"[Phase 2] LP improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – deep gradient refinement
    if best_seq is not None and time.time() < deadline - 15:
        print("Phase 3 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 3] Adam improved to {val_adam:.6f}")

        if time.time() < deadline - 10:
            print("Phase 3 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=250_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Phase 3] Sub‑grad improved to {val_sub:.6f}")

        if time.time() < deadline - 5:
            print("Phase 3 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.35,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[Phase 3] PGD improved to {val_pgd:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climbing (final seconds)
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 4 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 4] local search improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety clipping (matches evaluate_sequence's limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    # Return as plain Python list
    return best_seq.tolist()
