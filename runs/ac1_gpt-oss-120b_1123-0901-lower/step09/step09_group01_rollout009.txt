**Implementation for simplex projection**:

```
def _project_simplex(v):
    """Projects v onto the probability simplex {x >=0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

def _subgradient_optimize_euclid(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=10_000_000):
    a = a0.copy()
    a = np.clip(a, 1e-16, None)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a,n)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c/(a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active set
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0/active.sum()
        grad_a = 2.0*_cross_correlation(w, a, corr_fft_len)
        eta = 1.0/np.sqrt(it+1)
        a = a - eta * grad_a
        # project onto simplex
        a = _project_simplex(a)
        # avoid extreme zeros: if any a < 1e-12 set to zero? projection will handle.
        it += 1
    return best_a, best_val
```

"""
Propose a low‑score step‑function.

The optimizer works in four stages:

1. **Broad parametric sweep** – many analytic families (exponential,
   power‑law, stretched, logistic, cosine‑hump, triangular, plateau,
   shifted‑exponential, two‑exponential mixtures, random spikes, …) are
   sampled on a dense grid and also with random parameters.  The best
   shape is kept.

2. **Smooth‑max Adam** – starting from the incumbent, Adam is run on the
   log‑sum‑exp (smooth‑max) surrogate of the convolution peak.
   The temperature is increased until the surrogate becomes the true
   maximum.

3. **Exact‑max sub‑gradient** – two complementary sub‑gradient methods
   are applied (exponentiated‑gradient and Euclidean‑projected gradient)
   on the real objective 2·n·max(conv)/(∑a)².  Both run until the time
   budget is almost exhausted and the best vector of each is kept.

4. **Simulated‑annealing fine‑tuning** – a final local search transfers
   a random fraction of mass between two coordinates and accepts
   worsening moves with a slowly decreasing temperature.

All stages respect a 1000 s time budget; the routine always returns a
valid non‑negative list of coefficients.  The implementation is fully
self‑contained and uses only NumPy and SciPy (for the LP‑based warm‑start
we fall back to a simple random restart, keeping the code lightweight).

Author: ChatGPT – inequality‑constrained optimisation specialist.
"""

# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (full length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for sub‑gradient computation.
    (w is a weight vector of the same length as the full convolution.)
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

# Objective helpers -----------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x ≥ 0, Σx = 1}.
    """
    if v.ndim != 1:
        raise ValueError("simplex projection expects a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    a /= a.sum()

def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_logistic(alpha: float, mu: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 / (1.0 + np.exp(alpha * (t - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_cosine(p: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.cos(np.pi * (t - 0.5))
    a = np.maximum(a, 0.0) ** p
    a /= a.sum()
    return a


def _candidate_triangle(p: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = 1.0 - np.abs(2 * t - 1)
    a = np.maximum(a, 0.0) ** p
    a /= a.sum()
    return a


def _candidate_plateau(K: int, n: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if K > 0:
        a[:K] = 1.0
    a /= a.sum()

def _candidate_shifted_exponential(alpha: float, shift: int, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * np.maximum(i - shift, 0.0))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exp_power(alpha: float, p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) * ((i + 1) ** p)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_mixture_expo(alpha1: float, alpha2: float, w1: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = w1 * np.exp(-alpha1 * i) + (1.0 - w1) * np.exp(-alpha2 * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (simplex‑feasible) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.1          # temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # light learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient using exponentiated‑gradient updates.
    Step size ηₜ = η₀/√(t+1).  Returns best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Sub‑grad EG it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_optimize_euclid(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient using Euclidean projected gradient descent.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Sub‑grad Euclid it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.2
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem dimension – the original code used 4096
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and possibly update the incumbent
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Deterministic parametric sweep ---------------------------------
    # symmetric power‑law
    for p in np.linspace(-5.0, -0.05, 35):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # symmetric exponential
    for alpha in np.logspace(-5, 0.9, 35):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # stretched exponential
    alphas = np.logspace(-4, 0.6, 12)
    betas = np.linspace(0.2, 3.5, 14)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # beta distribution
    a_pars = [0.15, 0.3, 0.6, 1.0, 2.0, 4.0]
    b_pars = [0.15, 0.3, 0.6, 1.0, 2.0, 4.0]
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # double Gaussian
    for sig in np.logspace(-3, -0.2, 22):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 28):
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # monotone left‑power
    for p in np.linspace(-4.5, 1.5, 30):
        a = _candidate_left_power(p, n)
        _maybe_update(a)

    # logistic families
    for alpha in np.linspace(0.2, 15.0, 18):
        for mu in np.linspace(0.0, 1.0, 6):
            a = _candidate_logistic(alpha, mu, n)
            _maybe_update(a)

    # cosine hump
    for p in np.linspace(0.5, 5.0, 14):
        a = _candidate_cosine(p, n)
        _maybe_update(a)

    # triangular hump
    for p in np.linspace(0.5, 5.0, 14):
        a = _candidate_triangle(p, n)
        _maybe_update(a)

    # plateau shapes
    for K_frac in np.linspace(0.01, 0.5, 12):
        K = max(1, int(K_frac * n))
        a = _candidate_plateau(K, n)
        _maybe_update(a)

    # shifted exponential
    for _ in range(30):
        alpha = np.exp(np.random.uniform(-5, 2))
        shift = int(np.random.uniform(0, n // 2))
        a = _candidate_shifted_exponential(alpha, shift, n)
        _maybe_update(a)

    # exp × power mixtures
    for _ in range(40):
        alpha = np.exp(np.random.uniform(-5, 2))
        p = np.random.uniform(-5, 5)
        a = _candidate_exp_power(alpha, p, n)
        _maybe_update(a)

    # two‑exponential mixtures
    for _ in range(30):
        alpha1 = np.exp(np.random.uniform(-5, 2))
        alpha2 = np.exp(np.random.uniform(-5, 2))
        w1 = np.random.rand()
        a = _candidate_mixture_expo(alpha1, alpha2, w1, n)
        _maybe_update(a)

    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            a = _candidate_spike(k, n, random=True)
            _maybe_update(a)

    # ------------------------------------------------------------------
    # 1b) Randomised search over the above families --------------------
    # Run a moderate number of completely random parameter draws.
    family_funcs = [
        lambda: _candidate_exponential(np.exp(np.random.uniform(-5, 2)), n),
        lambda: _candidate_left_exponential(np.exp(np.random.uniform(-5, 2)), n),
        lambda: _candidate_left_power(np.random.uniform(-6, 2), n),
        lambda: _candidate_logistic(np.random.uniform(0.2, 20.0),
                                    np.random.rand(), n),
        lambda: _candidate_cosine(np.random.uniform(0.5, 6.0), n),
        lambda: _candidate_triangle(np.random.uniform(0.5, 6.0), n),
        lambda: _candidate_plateau(int(np.random.uniform(1, n // 2)), n),
        lambda: _candidate_shifted_exponential(
            np.exp(np.random.uniform(-5, 2)),
            int(np.random.uniform(0, n // 2)),
            n,
        ),
        lambda: _candidate_exp_power(
            np.exp(np.random.uniform(-5, 2)),
            np.random.uniform(-6, 6),
            n,
        ),
        lambda: _candidate_mixture_expo(
            np.exp(np.random.uniform(-5, 2)),
            np.exp(np.random.uniform(-5, 2)),
            np.random.rand(),
            n,
        ),
        lambda: _candidate_spike(int(np.random.uniform(1, 800)), n, random=True),
    ]

    random_trials = 300
    for _ in range(random_trials):
        try:
            a = np.random.choice(family_funcs)()
            _maybe_update(a)
        except Exception:
            continue  # ignore occasional numerical hiccups

    # ------------------------------------------------------------------
    # Warm start from a previously saved sequence (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam optimisation (smooth‑max) ---------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] start")
        # give Adam roughly 20 % of the remaining budget, but at most 30 s
        adam_time = min(30.0, (deadline - time.time()) * 0.2)
        adam_deadline = time.time() + adam_time
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient (exponentiated‑gradient) -----------------
    if best_seq is not None and time.time() < deadline - 10:
        remaining = deadline - time.time()
        # allocate ~60 % of the remaining time to EG, the rest to Euclidean
        sub_deadline_eg = time.time() + remaining * 0.6
        a_sub_eg, val_sub_eg = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline_eg,
            max_iters=3_000_000,
            eta0=0.8,
        )
        if val_sub_eg < best_val:
            best_val = val_sub_eg
            best_seq = a_sub_eg.copy()
            print(f"[Phase‑Subgrad EG] improved to {val_sub_eg:.6f}")

        # Euclidean projected gradient for the remaining budget
        if time.time() < deadline - 5:
            sub_deadline_eu = deadline - 2.0
            a_sub_eu, val_sub_eu = _subgradient_optimize_euclid(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline_eu,
                max_iters=3_000_000,
                eta0=0.9,
            )
            if val_sub_eu < best_val:
                best_val = val_sub_eu
                best_seq = a_sub_eu.copy()
                print(f"[Phase‑Subgrad Euclid] improved to {val_sub_eu:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning ---------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase‑Anneal] start")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=400_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity handling
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Optional final verification using the public evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
