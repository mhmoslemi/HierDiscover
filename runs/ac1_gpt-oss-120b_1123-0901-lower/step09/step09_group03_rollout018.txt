    pool = _initial_candidates(n, warm) + _spike_candidates(n)
    # quick evaluate each candidate limited by time
    for a0 in pool:

            break
        a0 = a0 / a0.sum()
        val = _objective(a0, n)  # compute directly
        if val < best_val:
            best_val = val
            best_seq = a0.copy()
    # main optimization on best_seq
    if best_seq is not None:
        # Phase A: Adam
        sub_dead = min(deadline, time.time() + 30.0) # allocate 30 sec
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000)

                best_seq = a_sub.copy()

        if time.time() < deadline - 5:
            a_pair, val_pair = _pair_targeted_anneal_optimize(best_seq, n, conv_fft_len, deadline, max_moves=300_000)

    if best_seq is not None and time.time() + 40 < deadline:

"""
Hybrid multi‑scale optimizer for the step‑function objective

F(a) = 2·n·max(a∗a) / (∑a)²   (a_i ≥ 0, n = len(a))

The search works in two resolution levels:

1.  Coarse optimisation at n = 4096.
    •  A diversified pool of seeds (Dirichlet draws, exponential / power‑law
       shapes, sparse spikes, asymmetric mixtures, …) is evaluated.
    •  The best seed is refined by three deterministic phases:
        – Adam on a log‑sum‑exp (smooth‑max) surrogate.
        – Exact‑max exponentiated‑gradient sub‑gradient descent.
        – Pair‑wise targeted simulated annealing.
2.  Upsampling to n = 16384 (if time permits) and a repeat of the three
    deterministic phases plus a final light random‑annealing stage.

All phases respect the simplex constraint (non‑negative, sum = 1) by working
with soft‑max logits or Euclidean simplex projection.  FFT‑based convolution
(and cross‑correlation) makes each iteration cheap even for the large grid.
The routine always returns a valid list; values are finally clipped to the
range accepted by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length = 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for (sub‑)gradient.
    Returns a vector of length len(a) = n giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact value of the evaluation function (no clipping)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float('inf')
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate  log‑sum‑exp( λ·(a∗a) ).
    Returns (best_a, best_val) where best_val = exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # interior point of the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.04                       # start with a modest temperature
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (gradient wrt logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule for smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when time is about to run out
        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    Keeps all iterates on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of maximal indices (sub‑gradient of max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent)
        a = a * np.exp(-eta * grad_a)

        # renormalise (protect against numerical drift)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted annealing.
    At each step the index k where (a∗a) is maximal is located.
    Among the admissible pairs (i, j) with i+j=k we pick the one with the
    largest product a_i·a_j and move a small fraction away from that pair.
    The move is accepted with Metropolis probability.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a, best_val = a.copy(), val

    T_start, T_end = 0.05, 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # find the pair (i,j) with i+j=kmax that maximises a_i*a_j
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_src = int(idx[np.argmax(prod)])
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # single‑index case – move a small fraction elsewhere
            if a2[i_src] < 1e-14:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # remove a little from both ends simultaneously
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a, a, val = val2, a2.copy(), a2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2

        move += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing with random mass transfers.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a, best_val = a.copy(), val

    T_start, T_end = 0.08, 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a, a, val = val2, a2.copy(), a2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Diversified pool of starting vectors (size ≈ 30)."""
    cand = []

    # warm start + cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        sh = left + right
        sh /= sh.sum()
        cand.append(sh)

    # Asymmetric exponential families (skewed)
    for gl, gr in ((0.5, 2.0), (1.0, 3.0), (0.2, 1.5)):
        left = np.exp(-gl * np.arange(n, dtype=np.float64))
        right = np.exp(-gr * np.arange(n - 1, -1, -1, dtype=np.float64))
        sh = left + right
        sh /= sh.sum()
        cand.append(sh)

    # Power‑law edge families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        sh = np.where(prod > 0, prod ** alpha, 0.0)
        if sh.sum() > 0:
            sh /= sh.sum()
            cand.append(sh)

    # Gaussian mixtures (3 bumps, random centres)
    xs = np.arange(n, dtype=np.float64)
    for _ in range(3):
        centres = np.random.choice(n, size=3, replace=False)
        sigma = np.random.rand() * 0.05 * n + 0.01 * n
        sh = np.zeros(n)
        for c in centres:
            sh += np.exp(-0.5 * ((xs - c) / sigma) ** 2)
        sh = np.maximum(sh, 0.0)
        sh /= sh.sum()
        cand.append(sh)

    # Double‑Gaussian (edges)
    for sigma in (0.1, 0.3, 0.6):
        sh = np.exp(-0.5 * ((xs) / (sigma * n)) ** 2) \
           + np.exp(-0.5 * ((xs - (n - 1)) / (sigma * n)) ** 2)
        sh /= sh.sum()
        cand.append(sh)

    # Half‑wave sinusoid
    sh = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sh = np.maximum(sh, 0.0)
    sh /= sh.sum()
    cand.append(sh)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    # random‑position spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    Returns the best non‑negative step‑function (list of floats) found within
    the given time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # 1. Coarse level (n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start from a previous solution, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # build start pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # quick scan of the pool (time‑bounded)
    for a0 in pool:
        if time.time() > deadline:
            break
        a0 = a0 / a0.sum()
        val = _objective(a0, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a0.copy()
            print(f"[Init] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2. Deterministic refinement at the coarse resolution
    if best_seq is not None and time.time() < deadline - 20:
        # Adam (smooth‑max)
        sub_dead = min(deadline, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(

        if val_adam < best_val:

            if val_sub < best_val:

                print(f"[Refine‑Sub] best {val_sub:.6f}")

        # Pair‑wise targeted annealing
        if time.time() < deadline - 5:
            a_pair, val_pair = _pair_targeted_anneal_optimize(
                best_seq, n, conv_fft_len, deadline, max_moves=300_000
            )
            if val_pair < best_val:
                best_val, best_seq = val_pair, a_pair.copy()
                print(f"[Refine‑Pair] best {val_pair:.6f}")

    # ------------------------------------------------------------------
    # 3. Upsample to a finer grid (n = 16384) if time permits
    if best_seq is not None and time.time() + 40 < deadline:
        print("[Upsample] moving to n = 16384")
        n_big = 16384
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # Adam on the large grid
        sub_dead_big = min(deadline, time.time() + 30.0)
        a_big_adam, val_big_adam = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big, sub_dead_big,
            max_iters=250_000
        )
        if val_big_adam < best_val:
            best_val, best_seq, n = val_big_adam, a_big_adam.copy(), n_big
            conv_fft_len, corr_fft_len = conv_fft_len_big, corr_fft_len_big
            print(f"[Upsample‑Adam] best {val_big_adam:.6f}")

        # Sub‑gradient on large grid
        if time.time() < deadline - 10:
            a_big_sub, val_big_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=400_000, eta0=0.8
            )
            if val_big_sub < best_val:
                best_val, best_seq = val_big_sub, a_big_sub.copy()
                print(f"[Upsample‑Sub] best {val_big_sub:.6f}")

        # Pair‑wise annealing on large grid
        if time.time() < deadline - 5:
            a_big_pair, val_big_pair = _pair_targeted_anneal_optimize(
                best_seq, n, conv_fft_len, deadline, max_moves=200_000
            )
            if val_big_pair < best_val:
                best_val, best_seq = val_big_pair, a_big_pair.copy()
                print(f"[Upsample‑Pair] best {val_big_pair:.6f}")

    # ------------------------------------------------------------------
    # 4. Final light random annealing (any remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Final Anneal] random annealing")
        a_final, val_final = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=100_000
        )
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[Final Anneal] best {val_final:.6f}")

    # ------------------------------------------------------------------
    # 5. Fallback – guarantee a feasible return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)
        n = 4096

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
