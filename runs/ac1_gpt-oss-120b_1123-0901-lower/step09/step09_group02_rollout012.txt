* Build a diverse pool of analytic families (uniform, half‑sinusoid,
  symmetric beta, triangular, Gaussian, Dirichlet draws, and the
  previous warm‑start if present).
* Pick the best few candidates (by a fast FFT‑based evaluation) and
  run a long exponentiated‑gradient sub‑gradient descent that minimises
  the true `max` (the active‑set is detected each iteration).
* Finish with a short simulated‑annealing phase (mass‑transfer moves
  plus occasional swaps) to fine‑tune the permutation of the entries.
* A tiny “power‑flattening” post‑process (a⁽ᵖ⁾, p<1) is tried to see
  whether a slight flattening of the distribution reduces the peak
  further.
* The whole method respects the user supplied budget (default 1000 s)
  and always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using real FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the exact sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (no clipping, no safety checks)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimisation (exact max)
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex with the exact max‑convolution
    sub‑gradient.  a0 is assumed non‑negative (it is re‑normalised internally).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # true objective (for bookkeeping)
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set = indices where convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of the true max (see paper)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step size η_t = η0 / sqrt(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise back to the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (mass transfer + swaps)
# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Simple annealing with two move types."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate family generation
# ----------------------------------------------------------------------
def _candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Build a modestly diverse set of start vectors."""
    pool = []

    # Warm start (if any) – also an up‑sampled version for the larger n
    if warm is not None:
        if warm.shape[0] != n:
            # simple repeat‑until‑fit up‑sampling
            factor = n // warm.shape[0]
            remainder = n % warm.shape[0]
            warm_up = np.repeat(warm, factor)
            if remainder:
                warm_up = np.concatenate([warm_up, warm[:remainder]])
            warm = warm_up[:n]
        pool.append(warm.copy())

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Half‑sinusoid (positive half wave)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    pool.append(s)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    pool.append(tri)

    # Symmetric Beta‑type (α = β)
    x = np.arange(1, n + 1, dtype=np.float64) / n
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        beta = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        if beta.sum() > 0:
            beta /= beta.sum()
            pool.append(beta)

    # Central Gaussian (single peak)
    xs = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    for sigma in (0.08, 0.12, 0.16, 0.20):
        g = np.exp(-0.5 * ((xs - mu) / (sigma * n)) ** 2)
        if g.sum() > 0:
            g /= g.sum()
            pool.append(g)

    # Random Dirichlet draws (various concentrations)
    rng = np.random.default_rng()
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(rng.dirichlet(alpha * np.ones(n)))

    # Interleaved permutation of every candidate (helps spread mass)
    interleaved = []
    perm = (np.arange(n) * 2) % n
    for a in pool:
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    pool.extend(interleaved)

    return pool


# ----------------------------------------------------------------------
# Main optimisation driver
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The implementation respects the given time budget and always
    returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size – a larger n allows a lower theoretical bound.
    # 8192 is still cheap for FFT‑based convolutions.
    # --------------------------------------------------------------
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (previous best, if available)
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            # up‑sample by simple repetition (keeps non‑negativity)
            factor = n // warm.shape[0]
            remainder = n % warm.shape[0]
            warm_up = np.repeat(warm, factor)
            if remainder:
                warm_up = np.concatenate([warm_up, warm[:remainder]])
            warm = warm_up[:n]
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build initial pool and pick a handful of promising candidates.
    # --------------------------------------------------------------
    pool = _candidate_pool(n, warm)

    best_seq = None
    best_val = float("inf")

    # quick screening (no heavy logging)
    for a in pool:
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    print(f"[Init] Best of pool value {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase I – long exponentiated‑gradient sub‑gradient optimisation.
    # Allocate about 70 % of the remaining budget.
    # --------------------------------------------------------------
    subgrad_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.70)
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        subgrad_deadline,
        max_iters=800_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Phase I] Sub‑gradient improvement → {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase II – a short simulated‑annealing polishing run (~20 % budget).
    # --------------------------------------------------------------
    if time.time() < deadline - 40:
        anneal_deadline = deadline - 30
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            anneal_deadline,
            max_moves=120_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase II] Annealing improvement → {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase III – optional power‑flattening (p < 1) to see if a tiny
    #                reshaping yields a lower peak.
    # --------------------------------------------------------------
    if time.time() < deadline - 5:
        for p in (0.8, 0.85, 0.9, 0.95):
            a_pow = np.power(best_seq, p)
            a_pow /= a_pow.sum()
            val_pow = _evaluate_objective(a_pow, n, conv_fft_len)
            if val_pow < best_val:
                best_val = val_pow
                best_seq = a_pow.copy()
                print(f"[Power p={p}] improvement → {val_pow:.6f}")

    # --------------------------------------------------------------
    # Final sanity: clip and return as plain Python list.
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
