"""
stepfunction_optimizer_v3.py

A time‑bounded optimizer for

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

The routine works in four phases:

1. **Quick Adam on a log‑sum‑exp surrogate** – cheap non‑smooth
   approximation of the max‑convolution.  A few seconds are enough to
   push a random start far away from the uniform plateau.

2. **Long exponentiated‑gradient sub‑gradient descent** – exact max
   is used, the update lives on the probability simplex via the
   mirror‑descent (exponential) map.

3. **Fine additive projected‑gradient descent** – for the last part of
   the budget a simple Euclidean projection step is used, which often
   reduces the objective a little further after the mirror‑descent
   stagnates.

4. **Very short simulated‑annealing (mass‑transfer + swaps)** – a
   local stochastic search that only accepts improvements (tiny uphill
   probability) to escape flat plateaus.

All phases respect a global deadline; the best sequence seen so far is
always remembered and finally returned (clipped to the allowed range).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase temperature of the smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.90

        it += 1

    return best_a, best_val


def _subgradient_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient optimisation (exact max)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not np.any(active):
            it += 1
            continue
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_additive(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Additive projected‑gradient sub‑gradient descent (exact max)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[AddGrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not np.any(active):
            it += 1
            continue
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # simple diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad_a
        a_new = np.maximum(a_new, 0.0)
        a = _project_simplex(a_new)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
    swap_prob: float = 0.10,
) -> tuple[np.ndarray, float]:
    """Very short simulated‑annealing with mass‑transfer and swaps."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            i = np.random.randint(n)
            if a[i] < 1e-15:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.35
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Initial candidate generators
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Diverse pool of starter vectors (uniform, Dirichlet, a few analytic shapes)."""
    cand = []

    # warm start
    if warm is not None:
        cand.append(warm.copy())

    # uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws with varying concentration – these give many different
    # sparsity levels without too much overhead.
    for alpha in (0.02, 0.1, 0.4, 0.8, 2.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # simple half‑sine (positive part of sin)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # central Gaussian
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    for sigma in (0.03, 0.07, 0.12):
        g = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        g /= g.sum()
        cand.append(g)

    # interleaved version of each candidate (helps to spread heavy entries)
    interleaved = []
    perm = (np.arange(n) * 2) % n
    for a in cand:
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # problem size – 4096 has been a good trade‑off in earlier experiments
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (previous best, if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a small but diverse pool of starter vectors
    pool = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – fast Adam on the smooth‑max surrogate (≈10 % of budget)
    phase1_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.1)
    for a0 in pool:
        if time.time() >= phase1_deadline:
            break
        a = a0 / a0.sum()
        sub_dead = min(phase1_deadline, time.time() + 3.0)
        a_adam, val_adam = _adam_optimize(
            a, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improvement → {val_adam:.6f}")

    # If no candidate succeeded (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _evaluate_objective(best_seq, n, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient sub‑gradient optimisation
    # Allocate roughly 70 % of the remaining budget.
    phase2_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.70)
    print("[Phase II] Starting exponentiated‑gradient sub‑gradient optimisation")
    a_exp, val_exp = _subgradient_expgrad(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        phase2_deadline,
        max_iters=1_500_000,
        eta0=1.0,
    )
    if val_exp < best_val:
        best_val = val_exp
        best_seq = a_exp.copy()
        print(f"[Phase II] new best {val_exp:.6f}")

    # ------------------------------------------------------------------
    # Phase III – additive projected‑gradient fine tuning (≈15 % budget)
    phase3_deadline = min(deadline, time.time() + (deadline - time.time()) * 0.15)
    print("[Phase III] Starting additive projected‑gradient refinement")
    a_add, val_add = _subgradient_additive(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        phase3_deadline,
        max_iters=800_000,
        eta0=0.5,
    )
    if val_add < best_val:
        best_val = val_add
        best_seq = a_add.copy()
        print(f"[Phase III] new best {val_add:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – short simulated‑annealing (≈5 % budget)
    if time.time() < deadline - 5:
        phase4_deadline = deadline
        print("[Phase IV] Short simulated‑annealing moves")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            phase4_deadline,
            max_moves=60_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] new best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – clip to allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
