"""
Propose a low‑value step‑function.

The optimizer works in three stages:

1. **Diverse warm‑starts** – a big pool of deterministic families
   (uniform, sinusoid, edge‑exponential, power‑law, beta, double‑Gaussian)
   plus many random Dirichlet draws.  The best candidate (by the exact
   objective) is kept as a seed.

2. **Smooth‑max projected gradient descent** – a mirror‑descent
   (exponentiated‑gradient) method that works directly on the exact
   max‑convolution.  The gradient is taken of a smooth approximation of
   the max (soft‑max with a temperature that is increased slowly).  The
   update is multiplicative, stays on the probability simplex and uses
   an adaptive stepsize η_t = η0 / √t.  This phase is run repeatedly
   from several perturbed restarts until almost the whole time budget is
   exhausted.

3. **Cheap hill‑climber** – a few random mass‑transfer moves are tried
   in the last seconds to squeeze any remaining improvement.

All operations use FFT‑based convolutions (power‑of‑two lengths) for
speed.  The routine always returns the best sequence seen so far and
conforms to the restrictions of ``evaluate_sequence`` (non‑negative,
finite, values ≤ 1000)."""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Simple utilities
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Candidate generators
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    return shape / shape.sum()


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    return shape / shape.sum()


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = idx ** alpha + (n - idx + 1) ** alpha
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (i / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    return shape / shape.sum()


def _random_dirichlet(rng: np.random.Generator, n: int, alpha: float) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (deterministic families)."""
    pool: list[np.ndarray] = []

    # Dirichlet draws (different concentrations)
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(rng, n, alpha))

    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool

def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

#  Smooth‑max projected gradient descent (exponentiated‑gradient)

def _smooth_pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.6,
    beta_final: float = 6.0,
    max_iters: int = 10_000_000,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact max‑convolution using a smooth‑max
    surrogate.  The weight function w is a soft‑max over the convolution
    values with a temperature that is increased linearly from 0 to
    ``beta_final``.  The update is multiplicative (exponentiated‑gradient)
    and the iterate is projected onto the simplex by normalisation.
    Returns the best vector (sum==1) and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    beta = 0.0
    beta_inc = beta_final / max_iters  # linear schedule

    while time.time() < deadline and it < max_iters:
        # convolution and objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c  # sum(a) == 1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[SmoothPGD it={it:06d}] new best {best_val:.6f}")

        # smooth max weights
        if beta > 0:
            # numerical stability: shift by max_c (negative exponent)
            w = np.exp(beta * (conv - max_c))
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                w = np.full_like(conv, 1.0 / conv.size)
            else:
                w /= w_sum
        else:
            w = np.full_like(conv, 1.0 / conv.size)

        # gradient of the smooth surrogate
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # stepsize (decreasing)
        eta = eta0 / np.sqrt(it + 1)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s == 0:
            a = np.full(n, 1.0 / n)
        else:
            a /= s

        # increase temperature
        beta = min(beta + beta_inc, beta_final)

        # tiny random jitter when we are close to the deadline (helps escape plateaus)
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val

#  Cheap hill‑climber (random mass transfers)
# ----------------------------------------------------------------------


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)  # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

#  Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with low evaluation value.
    The routine respects the time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a power‑of‑two FFT size works best
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (optional)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build initial pool of candidates
    # ------------------------------------------------------------------
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # add a few extra Dirichlet draws with various concentrations
    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0):
        for _ in range(8):
            pool.append(_random_dirichlet(rng, n, alpha))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Quick scan of the pool
    # ------------------------------------------------------------------
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = rng.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many short smooth‑PGD restarts
    # ------------------------------------------------------------------
    # We keep spawning short optimisation runs until the remaining time
    # becomes too small for a meaningful run.
    while best_seq is not None and time.time() < deadline - 30:
        remaining = deadline - time.time()
        # allocate a slice of the remaining budget (≈20 % of what is left)
        run_time = min(remaining * 0.2, 120.0)  # at most 2 min per restart
        sub_deadline = time.time() + run_time

        # initialise from the current best with a small random perturbation
        a0 = best_seq.copy()
        jitter = (np.random.rand(n) - 0.5) * 0.1   # ±5 % jitter
        a0 = a0 * (1.0 + jitter)
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum < 1e-12:
            a0 = rng.dirichlet(np.ones(n))
        else:
            a0 = a0 / a0_sum

        a_opt, val_opt = _smooth_pgd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            eta0=0.6,
            beta_final=6.0,
            max_iters=5_000_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Restart] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climber
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local moves] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety fall‑back
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
