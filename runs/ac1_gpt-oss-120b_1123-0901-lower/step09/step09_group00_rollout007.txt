def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # choose n
    n = 8192
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2) # maybe not needed now

    # Warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        # interpolate to new length
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # Use linear interpolation
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm_up = np.interp(x_new, x_old, warm)
        else:
            warm_up = warm.copy()
        warm_up = np.clip(warm_up, 0, 1000)
        if warm_up.sum() < 0.01:
            warm_up = None
        else:
            warm_up /= warm_up.sum()
    else:
        warm_up = None

    # Build initial population
    pop_size = 30
    pop = []
    if warm_up is not None:
        pop.append(warm_up)
    # Add rotated copies
    if warm_up is not None:
        for shift in (n//8, n//4, 3*n//8):
            rot = np.roll(warm_up, shift)
            pop.append(rot)
    # fill rest with random Dirichlet
    while len(pop) < pop_size:
        a = np.random.dirichlet(np.ones(n))
        pop.append(a)

    # Ensure each on simplex
    pop = [p / p.sum() if p.sum() > 0 else np.random.dirichlet(np.ones(n)) for p in pop]

    # Helper: fitness
    def fitness(a):
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        return 2.0 * n * max_c

    # Compute fitness for initial.
    pop_fitness = np.array([fitness(ind) for ind in pop])
    best_idx = int(pop_fitness.argmin())
    best_seq = pop[best_idx].copy()
    best_val = float(pop_fitness[best_idx])
    print(f"[Init] best value {best_val:.6f}")

    # Differential Evolution loop
    # parameters
    F = 0.8
    CR = 0.9
    iter = 0
    while time.time() < deadline - 60:  # keep 60s for fine phases
        for i in range(pop_size):
            # select a,b,c
            idxs = np.random.choice([j for j in range(pop_size) if j != i], 3, replace=False)
            a_vec = pop[idxs[0]]
            b_vec = pop[idxs[1]]
            c_vec = pop[idxs[2]]
            mutant = a_vec + F * (b_vec - c_vec)
            # clip and project
            mutant = np.clip(mutant, 0, None)
            mutant = _project_simplex(mutant)
            # crossover
            cross_mask = np.random.rand(n) < CR
            if not np.any(cross_mask):
                cross_mask[np.random.randint(n)] = True
            trial = np.where(cross_mask, mutant, pop[i])
            # evaluate
            val = fitness(trial)
            if val < pop_fitness[i]:
                pop[i] = trial
                pop_fitness[i] = val
                if val < best_val:
                    best_val = val
                    best_seq = trial.copy()
                    print(f"[DE it={iter} ind={i}] new best {best_val:.6f}")
        iter += 1
        # maybe adapt F and CR
        if iter % 10 == 0:
            # slight random perturb of F,CR
            F = 0.5 + np.random.rand() * 0.5
            CR = 0.7 + np.random.rand() * 0.3

    # Now run Adam optimizer on best_seq (smooth-max)
    # Reuse _adam_optimize defined earlier but adapt to new n
    # Use time remaining

    # Compute conv_fft_len and corr_fft_len already defined, reuse.
    # Since _adam_optimize expects a0 in simplex (sum=1)
    time_remaining = deadline - time.time()
    if time_remaining > 5:
        sub_dead = time.time() + min(time_remaining*0.6, 30.0)  # allocate up to 30 sec
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=400_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam post] improved to {best_val:.6f}")

    # Fine gradient descent

    if time_remaining > 5:

            print(f"[GD post] improved to {best_val:.6f}")

    # Anneal final local search

exponential families, power‑laws, sinusoid, warm‑start) and runs a classic
Differential‑Evolution loop on the simplex (projection after mutation).  The
best individual is then refined with the smooth‑max Adam optimiser, a
projected gradient descent on the exact max‑convolution, and a short simulated
annealing pass.  All stages respect the global time budget (≈ 1000 s) and the
algorithm always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient phases."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns the best vector (on the simplex) encountered and its exact
    objective value 2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Returns the best vector and its exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # keep a small safety margin for the host environment
    total_budget = max(10.0, budget_s - 2.0)
    deadline = time.time() + total_budget

    # ------------------------------------------------------------------
    # problem size – a power of two makes FFT cheap
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to the larger grid
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Helper to evaluate the exact objective (sum of a is assumed to be 1)
    def _objective(a: np.ndarray) -> float:
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # Build a small but diverse initial population (size ≤ 20)
    pop = []

    if warm is not None:
        pop.append(warm.copy())
        # a few rotated copies to diversify the start
        for shift in (n // 8, n // 4, 3 * n // 8):
            pop.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pop.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edge families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        pop.append(shape)

    # Power‑law families (already used in the reference implementation)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pop.append(shape)

    # Half‑wave sinusoid
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    sinusoid /= sinusoid.sum()
    pop.append(sinusoid)

    # Uniform
    pop.append(np.full(n, 1.0 / n))

    # If we still have fewer than 20 individuals, fill with random Dirichlet draws
    while len(pop) < 20:
        pop.append(np.random.dirichlet(np.ones(n)))

    # Ensure every individual lies on the simplex
    pop = [p / max(p.sum(), 1e-12) for p in pop]

    pop = [np.clip(p, 0.0, None) for p in pop]
    pop = [_project_simplex(p) for p in pop]

    pop_fitness = np.array([_objective(p) for p in pop])
    best_idx = int(pop_fitness.argmin())
    best_seq = pop[best_idx].copy()
    best_val = float(pop_fitness[best_idx])
    print(f"[Init] best objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Differential Evolution phase (≈ 50 % of the budget)
    de_deadline = min(deadline, time.time() + total_budget * 0.5)
    pop_size = len(pop)
    F = 0.8
    CR = 0.9
    de_iter = 0

    while time.time() < de_deadline:
        for i in range(pop_size):
            # choose three distinct indices different from i
            idxs = np.random.choice([j for j in range(pop_size) if j != i], 3, replace=False)
            a_vec, b_vec, c_vec = pop[idxs[0]], pop[idxs[1]], pop[idxs[2]]

            mutant = a_vec + F * (b_vec - c_vec)
            mutant = np.clip(mutant, 0.0, None)
            mutant = _project_simplex(mutant)

            # binomial crossover
            mask = np.random.rand(n) < CR
            if not np.any(mask):
                mask[np.random.randint(n)] = True
            trial = np.where(mask, mutant, pop[i])

            val = _objective(trial)
            if val < pop_fitness[i]:
                pop[i] = trial
                pop_fitness[i] = val
                if val < best_val:
                    best_val = val
                    best_seq = trial.copy()
                    print(f"[DE it={de_iter} ind={i}] new best {best_val:.6f}")

        de_iter += 1
        # occasional modest randomisation of F / CR to keep search lively
        if de_iter % 10 == 0:
            F = 0.5 + np.random.rand() * 0.5
            CR = 0.7 + np.random.rand() * 0.3

    # ------------------------------------------------------------------
    # Adam optimisation on the smooth‑max surrogate (≈ 30 % of the budget)
    time_left = deadline - time.time()
    if time_left > 10.0:
        sub_dead = time.time() + min(time_left * 0.6, 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Projected gradient descent on the exact max‑convolution (≈ 10 % of budget)
    time_left = deadline - time.time()
    if time_left > 5.0:
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final short annealing (use any remaining time)
    time_left = deadline - time.time()
    if time_left > 2.0:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=100_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to the evaluator’s allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
