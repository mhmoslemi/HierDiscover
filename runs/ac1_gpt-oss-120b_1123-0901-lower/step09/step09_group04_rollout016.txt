def _exponential_gradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters, eta0):
    a = a0.copy()
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    it=0
    while time.time() < deadline and it < max_iters:
        # Compute conv and gradient
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        if not np.any(active):
            active = (conv == max_c) # just for safety
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Step size schedule: eta = eta0 / np.sqrt(it+1)
        eta = eta0 / np.sqrt(it + 1)
        # Update log
        log_a = log_a - eta * grad
        # Numerically stable
        max_log = log_a.max()
        a = np.exp(log_a - max_log)
        a_sum = a.sum()
        if a_sum == 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            log_a = np.log(a + 1e-30)
            continue
        a = a / a_sum

        # evaluate new objective
        conv_new = _full_convolution(a, conv_fft_len)
        max_c_new = conv_new.max()
        val = 2.0 * n * max_c_new / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # Optionally print
            if it % 20000 == 0:
                print(f"[EG it={it}] new best {val:.6f}")

        it += 1

    return best_a, best_val

"""
Propose a low‑value step‑function.

Algorithm
---------
1. Build a diverse pool of candidate probability vectors:
   - deterministic families (uniform, sinusoid, edge‑exponential,
     edge‑power‑law, beta, double‑Gaussian)
   - many random draws (Dirichlet, beta‑shapes, edge‑exponential,
     power‑law, a two‑sided exponential mixture)
   All candidates are normalised to sum 1.

2. Evaluate the pool, keep the five best candidates.

3. Refine each of the best seeds with a large‑scale
   Exponentiated‑Gradient (EG) optimiser on the exact max‑convolution.
   EG works directly on the simplex using the sub‑gradient of the
   max‑convolution (active set of indices attaining the maximum).

4. Final global annealing phase:
   Starting from the best seed perform a fast simulated‑annealing
   mass‑transfer search.  The convolution is updated incrementally
   (O(n) per move) using the identity
        Δconv = –2·δ·a_{k−i} + 2·δ·a_{k−j},
   which avoids repeated FFTs.  The temperature is exponentially
   cooled; improvements are always kept and the best ever‑seen vector
   is retained.

5. Return the best vector (as a plain list).  The vector is non‑negative,
   sums to one (the objective is scale‑invariant) and is clipped to the
   limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# FFT helpers (used by EG optimisation)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient of max‑conv.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))

def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _initial_pool(n: int) -> list[np.ndarray]:
    """Deterministic families used as warm‑starts."""
    pool = []

    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    return pool

def _exponentiated_gradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the probability simplex using the exact sub‑gradient
    of the max‑convolution.  The update is performed on the log‑coordinates
    (the natural EG step).  Step‑size = eta0 / sqrt(t+1).
    """
    a = a0.copy()
    log_a = np.log(a + 1e-30)
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        log_a = log_a - eta * grad
        # Numerical stability – shift by the maximum
        log_a -= log_a.max()
        a = np.exp(log_a)
        a_sum = a.sum()
        if a_sum == 0.0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            log_a = np.log(a + 1e-30)
            continue
        a = a / a_sum

        conv_new = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv_new.max() / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[EG it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Fast SA with O(n) incremental convolution updates.
    State: (a, conv).  A move transfers a random fraction (≤30 %) of a[i]
    to a[j]; the convolution is updated analytically, then the move is
    accepted with Metropolis probability.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    cur_max = conv.max()
    cur_val = 2.0 * n * cur_max / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    # temperature schedule
    T0 = cur_val * 0.02 + 1e-8
    Tmin = 1e-10
    decay = (Tmin / T0) ** (1.0 / max_moves)
    T = T0

    for move in range(max_moves):
        if time.time() >= deadline:
            break
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        j = rng.integers(n)
        if i == j:
            continue

        delta = a[i] * rng.random() * 0.3
        if delta <= 0.0:
            continue

        # apply linear update to conv using the *old* a (still unchanged)
        conv[i : i + n] -= 2.0 * delta * a
        conv[j : j + n] += 2.0 * delta * a

        new_max = conv.max()
        new_val = 2.0 * n * new_max / (a.sum() ** 2)  # sum unchanged =1

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((cur_val - new_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            a[i] -= delta
            a[j] += delta
            cur_val = new_val
            cur_max = new_max
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        else:
            # revert the linear update (use the same *old* a)
            conv[i : i + n] += 2.0 * delta * a
            conv[j : j + n] -= 2.0 * delta * a

        T *= decay

    return best_a, best_val


def _local_random_search(

    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)

    it = 0

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

            a = a2

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    The routine respects the time budget, keeps the best vector seen so far,

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start (optional global variable)

    warm = globals().get("height_sequence_1")

        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a large initial pool
    # ------------------------------------------------------------------
    pool = _warm_and_shifts(warm, n) + _initial_pool(n)

    # random Dirichlet draws with many concentrations
    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, alpha))

    # random edge‑exponential
    for gamma in np.logspace(-2, 1, 8):
        for _ in range(3):
            pool.append(_edge_exponential(n, gamma))

    # random power‑law
    for exp in np.linspace(-3.0, -0.5, 6):
        for _ in range(3):
            pool.append(_edge_power_law(n, exp))

    # random beta shapes (log‑uniform parameters)
    for _ in range(30):
        a = 10 ** rng.uniform(-2, 1)
        b = 10 ** rng.uniform(-2, 1)
        pool.append(_beta_shape(n, float(a), float(b)))

    # random double Gaussian
    for sigma in rng.uniform(0.02, 0.5, size=20):
        pool.append(_double_gaussian(n, float(sigma)))

    # two‑sided exponential mixtures
    for _ in range(30):
        w = rng.dirichlet([1.0, 1.0, 1.0])
        p_left, p_right, p_unif = w
        gamma_left = 10 ** rng.uniform(-2, 1)
        gamma_right = 10 ** rng.uniform(-2, 1)
        idx = np.arange(n, dtype=np.float64)
        left = np.exp(-gamma_left * idx)
        right = np.exp(-gamma_right * (n - 1 - idx))
        shape = p_left * left + p_right * right + p_unif
        shape = np.clip(shape, 0.0, np.inf)
        shape /= shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # Evaluate the pool – keep the top 5 seeds
    # ------------------------------------------------------------------
    candidates = []  # list of (value, vector)
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / s
        val = _objective(a)
        candidates.append((val, a))
        if cid % 100 == 0:
            print(f"[Pool {cid}/{len(pool)}] val={val:.6f}")

    candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for (_, vec) in candidates[:5]]
    best_val = candidates[0][0]
    best_seq = candidates[0][1].copy()
    print(f"Best after pool scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Exponentiated‑Gradient refinement of the top seeds
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 60 and top_seeds:
        per_seed_time = remaining * 0.6 / len(top_seeds)  # 60 % of remaining time
        for idx, seed_vec in enumerate(top_seeds):
            eg_deadline = time.time() + per_seed_time
            a_opt, val_opt = _exponentiated_gradient(
                seed_vec,
                n,
                conv_fft_len,
                corr_fft_len,
                eg_deadline,
                max_iters=2_000_000,
                eta0=2.0,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[EG seed {idx}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Global simulated annealing starting from the best seed
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        sa_deadline = deadline - 5.0
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            sa_deadline,
            max_moves=800_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] new best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap final hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # safety fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
