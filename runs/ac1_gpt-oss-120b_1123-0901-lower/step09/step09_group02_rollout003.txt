"""
stepfunction_optimizer_v3.py

A time‑bounded multi‑stage optimizer that pushes the bound
2·n·max(conv) / (∑a)² below the previous record (≈1.5077) and
aims for the target 1.503.  The strategy is:

1. Initialise from a warm start (if present) or from a few
   analytically‑motivated shapes (central Gaussian, symmetric
   Beta, triangular).  All vectors are normalised to sum‑to‑1
   (the objective is scale invariant).

2. Adam on a smooth‑max (log‑sum‑exp) surrogate with an
   aggressively increasing temperature λ.  This quickly
   reduces the large‑peak region of the convolution.

3. Exact‑max exponentiated‑gradient sub‑gradient descent
   (mirror descent on the simplex).  The step‑size follows an
   AdaGrad‑like schedule for robustness.

4. Simulated‑annealing with two move types (mass transfer and
   swaps).  The temperature decays slowly (geometric schedule)
   and we allow occasional large transfers to escape plateaus.

5. Pure permutation hill‑climbing (swap‑only) to fine‑tune the
   ordering of the entries.

The routine respects a user‑supplied time budget (default 1000 s)
and always returns the best sequence discovered before the deadline.
All helper functions are top‑level, use only NumPy and the standard
library, and avoid closures or lambdas.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers (re‑used throughout)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective evaluation (exact, no clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (∑a)²  (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:            # safeguard – should never happen after normalisation
        return np.inf
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best (normalised) vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # start in log‑soft‑max space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                                 # a touch larger than before
    lam = 0.05                                 # initial temperature for log‑sum‑exp
    lam_mul = 1.06                             # grow λ fairly quickly

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max projection (a stays on simplex) -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate true objective for bookkeeping
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        conv = _full_convolution(a, conv_fft_len)
        cmax = conv.max()
        w = np.exp(lam * (conv - cmax))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, conv_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – make λ sharper over time
        lam = min(lam * lam_mul, 5e6)

        # tiny jitter near the deadline to shake out of plateaus
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    return best_a, best_val


def _subgrad_expgrad(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_iters: int = 800_000,
                     eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror‑descent on simplex) using the exact‑max sub‑gradient.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0

    # AdaGrad‑style accumulator for per‑coordinate step‑size adaptation
    G2 = np.zeros_like(a)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # same as _objective(a,…)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)

        # AdaGrad accumulator
        G2 += grad ** 2
        adapt = np.sqrt(G2) + 1e-8
        # classic exponentiated‑gradient step with per‑coord adapt
        a = a * np.exp(-eta0 * grad / adapt)

        # renormalise onto the simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 250_000,
                     swap_prob: float = 0.14) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with mass‑transfer and swap moves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    # start temperature – large enough to accept many uphill moves
    T = 0.12
    decay = 0.99993  # geometric decay per move

    it = 0
    while time.time() < deadline and it < max_moves:
        # annealing temperature schedule
        T = max(1e-5, T * decay)

        if np.random.rand() < swap_prob:
            # ---- swap move ----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ---- mass‑transfer move ----
            i = np.random.randint(n)
            if a[i] < 1e-16:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            # fraction in (0, 0.5] – allow occasional larger jumps
            frac = np.random.rand() * 0.5
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(a0: np.ndarray,
                        n: int,
                        conv_fft_len: int,
                        deadline: float,
                        max_swaps: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing with swaps only; accepts only improving moves
    (with a tiny uphill probability) to polish the ordering.
    """
    a = a0.copy()
    val = _objective(a, n, conv_fft_len)
    best_val = val
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance to escape flat plateaus
            if np.random.rand() < 0.008:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]   # revert

        it += 1

    return best_a, best_val

# Candidate factories

def _central_gaussian(n: int, sigma: float) -> np.ndarray:
    """Single central Gaussian, normalised to sum 1."""
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _symmetric_beta(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the discrete grid."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    s = shape.sum()
    if s == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= s
    return shape


def _triangular(n: int) -> np.ndarray:
    """Simple symmetric triangular shape."""
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    return tri

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a small but diverse pool of starts
    # ------------------------------------------------------------------
    start_pool = []

    # 1) warm start (if any) and a few cyclic shifts
    if warm is not None:
        start_pool.append(warm.copy())
        for shift in (n // 5, n // 2, 3 * n // 5):
            start_pool.append(np.roll(warm, shift))

    # 2) analytical families
    for sigma in (0.06, 0.10, 0.14, 0.20):
        start_pool.append(_central_gaussian(n, sigma))
    for alpha in (1.5, 2.2, 3.0, 4.0):
        start_pool.append(_symmetric_beta(n, alpha))
    start_pool.append(_triangular(n))

    # 3) a couple of random Dirichlet draws (moderate concentration)
    for concentration in (0.3, 0.8, 1.5):
        start_pool.append(np.random.dirichlet(concentration * np.ones(n)))

    # ------------------------------------------------------------------
    # Global bookkeeping
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Primary optimisation loop – iterate over the start pool
    # ------------------------------------------------------------------
    while start_pool and time.time() < deadline - 30:   # keep ~30 s for final polishing
        a0 = start_pool.pop(0)

        # ensure proper normalisation (sum = 1)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ---- Phase A: Adam on smooth‑max ----
        adam_deadline = min(deadline - 20, time.time() + 6.0)   # a few seconds per candidate
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, adam_deadline, max_iters=200_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Pool] Adam improved → {val_adam:.6f}")

        # ---- Phase B: Sub‑gradient (exp‑grad) ----
        sub_deadline = min(deadline - 10, time.time() + 12.0)
        a_sub, val_sub = _subgrad_expgrad(
            best_seq, n, conv_fft_len, sub_deadline,
            max_iters=400_000, eta0=0.85
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Pool] Sub‑grad improved → {val_sub:.6f}")

    # ------------------------------------------------------------------
    # If nothing was produced (unlikely) fall back to a uniform vector
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase C: Long simulated‑annealing refinement (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 12:
        print("[Anneal] Long annealing started")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=200_000, swap_prob=0.13
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] New best after anneal → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase D: Permutation‑only hill climb (final polishing)
    # ------------------------------------------------------------------
    if time.time() < deadline - 4:
        print("[Permute] Final hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq, n, conv_fft_len, deadline,
            max_swaps=100_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] New best after permute → {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Clip to allowed range and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
