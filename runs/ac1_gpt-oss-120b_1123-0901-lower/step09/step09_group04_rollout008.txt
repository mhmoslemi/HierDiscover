"""
Multi‑restart heuristic for the step‑function problem.

The algorithm repeatedly (while time permits)

1. draws a fresh random start on the probability simplex,
   occasionally mixing in the current best solution,
2. runs a smooth‑max Adam optimiser (log‑sum‑exp surrogate) to
   quickly move towards low‑max regions,
3. refines with a sub‑gradient projected‑gradient descent that works
   on the exact max‑convolution,
4. finishes with a cheap hill‑climber that moves random fractions of
   mass between coordinates.

All three phases respect the global deadline and the best vector ever
seen (in terms of ``evaluate_sequence``) is kept and finally returned.
The code works for any length; we use ``n = 4096`` (a good trade‑off
between quality and speed).  The routine always returns a non‑negative,
finite list that complies with the limits enforced by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Simplex projection (Euclidean)
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:            # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best vector (centre‑of‑simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # initialise on the simplex via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.04   # faster increase – stabilises earlier

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 15000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (soft‑max) weights over convolution entries
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – become sharper quickly
        lam = min(lam * lam_factor, 1e7)

        # tiny random jitter in the last seconds
        if it % 5000 == 0 and time.time() > deadline - 20:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 4000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size (decreasing as 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        # random jitter near the end to escape tiny plateaus
        if it % 15000 == 0 and time.time() > deadline - 15:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: move a random fraction of mass from one index
    to another and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine runs several independent optimisation “runs” (Adam → PGD →
    local search) from random starts, keeps the best vector seen and returns it.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # see whether a previous construction is supplied --------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # a few Dirichlet concentration choices for random draws ------------
    dirichlet_alphas = [0.05, 0.1, 0.2, 0.3, 0.6, 1.0, 2.0, 5.0]

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    runs_done = 0

    # ------------------------------------------------------------------
    #  Main multi‑restart loop
    # ------------------------------------------------------------------
    min_run_time = 30.0  # we need at least this much for a meaningful run
    while time.time() < deadline - min_run_time:
        remaining = deadline - time.time()
        run_time = min(140.0, remaining * 0.75)   # a single run gets ~140 s
        run_deadline = time.time() + run_time

        # ----- pick an initial vector -----------------------------------
        if runs_done == 0 and warm is not None:
            a0 = warm.copy()
        else:
            alpha = float(rng.choice(dirichlet_alphas))
            a0 = np.random.dirichlet(np.full(n, alpha))

        a0 = a0 / a0.sum()

        # Occasionally blend with the current global best to explore neighbourhoods
        if best_seq is not None and rng.random() < 0.25:
            mix = rng.random()
            a0 = mix * best_seq + (1.0 - mix) * a0
            a0 = a0 / a0.sum()

        # ----- Adam phase (smooth‑max) ---------------------------------
        adam_deadline = time.time() + 0.3 * run_time
        a_adam, val_adam = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=250_000,
            lr_start=0.04,
        )

        # ----- PGD phase (exact max) ------------------------------------
        pgd_deadline = time.time() + 0.55 * run_time
        a_pgd, val_pgd = _pgd_optimize(
            a_adam,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=500_000,
            eta0=0.85,
        )

        # ----- Final cheap hill‑climber ---------------------------------
        local_deadline = run_deadline
        a_loc, val_loc = _local_random_search(
            a_pgd,
            n,
            conv_fft_len,
            local_deadline,
            max_moves=120_000,
        )

        # ----- Update global best ----------------------------------------
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Run {runs_done}] new global best {best_val:.6f}")

        runs_done += 1

    # safety fallback ----------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(
        f"Final best value {best_val:.6f} after {runs_done} runs "
        f"(length {len(best_seq)})"
    )
    return best_seq.tolist()
