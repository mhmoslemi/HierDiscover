"""
A multi‑restart, multi‑family stochastic search for a non‑negative step function
minimising  F(a)=2·n·max(a∗a)/(sum(a)²).

The algorithm proceeds in three time‑aware phases:

1. **Diversified sampling** – many candidate vectors are drawn from several
   parametric families (exponential, power‑law, beta, double‑Gaussian,
   sinusoid‑power, triangular, two‑Gaussian, Dirichlet).  For each candidate
   we also try two deterministic reorderings that spread large entries
   across the domain (alternating high/low).  The best K≈8 elite vectors are
   kept.

2. **Short Adam refinements** – every elite vector is refined by a few seconds
   of Adam optimisation on a smooth‑max surrogate of the max‑convolution.
   The best refined vector replaces its parent in the elite set.

3. **Final polish** – the current overall best undergoes a longer Adam run,
   followed by a sub‑gradient and a projected‑gradient refinement, and finally
   a cheap stochastic local‑move hill‑climber.

All operations respect the 1000 s budget; the routine returns a list of
non‑negative heights that can be passed directly to ``evaluate_sequence``.
"""

import time
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple projection ------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a)/(sum(a)²) – fast version of evaluate_sequence."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Reordering heuristics -------------------------------------------------
def _alternating_reorder(v: np.ndarray) -> np.ndarray:
    """Place the biggest values alternately from left and right."""
    n = v.shape[0]
    order = np.argsort(-v)                     # descending
    out = np.empty_like(v)
    left, right = 0, n - 1
    toggle = True
    for idx in order:
        if toggle:
            out[left] = v[idx]
            left += 1
        else:
            out[right] = v[idx]
            right -= 1
        toggle = not toggle
    return out


def _spread_reorder(v: np.ndarray) -> np.ndarray:
    """Scatter large values uniformly: 0, n/2, 1, n/2+1, …."""
    n = v.shape[0]
    order = np.argsort(-v)
    out = np.empty_like(v)
    step = max(1, n // len(order))
    pos = 0
    for idx in order:
        out[pos % n] = v[idx]
        pos += step
    return out

# Parametric shape generators -------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    return s / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _two_gaussian_shape(n: int, sigma: float, d: float) -> np.ndarray:
    """Two equal Gaussians placed symmetrically at distance d·(n‑1)."""
    i = np.arange(n, dtype=np.float64)
    centre_left = (n - 1) * (0.5 - d / 2)
    centre_right = (n - 1) * (0.5 + d / 2)
    left = np.exp(-0.5 * ((i - centre_left) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - centre_right) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.02   # slower rise

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for early‑stop monitoring
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random perturbation near the end of the run
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # decay learning‑rate
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (rng.random() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4096 gives a good balance of resolution vs speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if a previous run stored a vector in the globals)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm /= warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – diversified random sampling + deterministic reordering
    elite_vals = []          # will hold up to K best objective values
    elite_vecs = []          # corresponding vectors
    K = 8                    # size of elite pool
    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "sin_pow", "tent", "two_gauss", "dirichlet"
    ]

    phase1_deadline = min(deadline - 5.0,
                          time.time() + 0.25 * (deadline - time.time()))
    print("Phase 1 – random sampling and reordering (elite size =", K, ")")

    def _maybe_update(val, vec):
        """Insert (val,vec) into the elite pool if it is among the K best."""
        if len(elite_vals) < K:
            elite_vals.append(val)
            elite_vecs.append(vec.copy())
        else:
            worst_idx = int(np.argmax(elite_vals))
            if val < elite_vals[worst_idx]:
                elite_vals[worst_idx] = val
                elite_vecs[worst_idx] = vec.copy()

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-3, 3)          # 0.001 … 1000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-30.0, -0.1)          # stronger edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 1)             # 0.01 … 10
            b = 10 ** rng.uniform(-2, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "two_gauss":
            sigma = rng.uniform(0.005, 0.4)
            d = rng.uniform(0.05, 0.45)          # distance as fraction of the range
            cand = _two_gaussian_shape(n, sigma, d)
        else:   # dirichlet
            alpha = rng.uniform(0.02, 3.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        # plain candidate
        val = _objective(cand, conv_fft_len)
        _maybe_update(val, cand)

        # two deterministic reorderings – they often improve the same multiset
        cand_alt1 = _alternating_reorder(cand)
        val1 = _objective(cand_alt1, conv_fft_len)
        _maybe_update(val1, cand_alt1)

        cand_alt2 = _spread_reorder(cand)
        val2 = _objective(cand_alt2, conv_fft_len)
        _maybe_update(val2, cand_alt2)

    # include warm‑start and a few circular shifts as extra elite entries
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            _maybe_update(val, cand)

    # retain the absolute best found so far
    best_idx = int(np.argmin(elite_vals))
    best_seq = elite_vecs[best_idx].copy()
    best_val = elite_vals[best_idx]
    print(f"[Phase 1] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – short Adam refinements for each elite member
    remaining = deadline - time.time()
    if remaining > 25.0:
        per_elite_time = max(5.0, remaining / (len(elite_vecs) + 1))
        print("Phase 2 – short Adam refinements (≈", round(per_elite_time, 1), "s each)")

        for i, vec in enumerate(elite_vecs):
            cur_dead = time.time() + per_elite_time
            if cur_dead > deadline - 2.0:
                break
            refined, val_ref = _adam_optimize(
                vec, n, conv_fft_len, corr_fft_len, cur_dead,
                max_iters=200_000, lr_start=0.05
            )
            if val_ref < best_val:
                best_val, best_seq = val_ref, refined.copy()
                print(f"[Elite {i}] Adam improvement to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final polished optimisation on the current best
    if time.time() < deadline - 8.0:
        print("Phase 3 – final Adam → Sub‑grad → PGD → local search")
        # Adam (longer)
        best_seq, best_val = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline - 6.0,
            max_iters=400_000, lr_start=0.05
        )
        # Sub‑gradient
        best_seq, best_val = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline - 3.0,
            max_iters=200_000, eta0=0.85
        )
        # Projected‑gradient
        best_seq, best_val = _pgd_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline - 2.0,
            max_iters=150_000, eta0=0.35
        )
        # Cheap stochastic hill‑climbing for the very last seconds
        best_seq, best_val = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000
        )
        print(f"[Phase 3] final value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to respect evaluate_sequence limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Returning sequence of length {len(best_seq)} with objective {best_val:.6f}")
    return best_seq.tolist()
