def _adam_lowdim(w0, B, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=100000):
    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)
    beta1, beta2 = 0.9, 0.999
    lr = 0.05
    lam = 0.02
    lam_factor = 1.02

    best_val = float('inf')
    best_seq = None
    it = 0
    while time.time() < sub_dead and it < max_iters:
        logits = B @ w
        # Softmax stable
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
        # smooth-max
        w_weights = np.exp(lam * (conv - max_c))
        w_sum = w_weights.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w_weights = np.full_like(conv, 1.0 / conv.size)
        else:
            w_weights /= w_sum
        grad_a = 2.0 * _cross_correlation(w_weights, a, corr_fft_len)
        # grad wrt logits:
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)
        # chain to w
        grad_w = B.T @ grad_log
        # Adam update
        m = beta1*m + (1-beta1)*grad_w
        v = beta2*v + (1-beta2)*(grad_w**2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        w -= step
        # update lam (smoothness)
        lam = min(lam * lam_factor, 1e6)
        # lighten
        if it % 10000 == 0:
            # optional noise
            w += np.random.randn(*w.shape) * 1e-3
        it += 1
    return best_seq, best_val

"""
Hybrid optimizer for the step‑function bound.

The algorithm consists of four phases:

1. **Exploratory parametric sweep** – a cheap scan over several
   parametric families (exponential tails, double Gaussian, power‑law,
   sinusoid, Dirichlet draws, spike patterns).  The best candidate from
   this sweep seeds the later phases.

2. **Low‑dimensional Adam** – the vector is represented as a soft‑max
   over a low‑dimensional sinusoidal basis (≈20 basis functions).  This
   reduces the optimisation problem to a handful of parameters while
   still being able to represent highly non‑trivial shapes.  Adam on a
   smooth‑max surrogate quickly moves the solution away from the uniform
   stationary point.

3. **Full‑dimensional exponentiated‑gradient sub‑gradient descent** –
   starting from the low‑dimensional solution we run many sub‑gradient
   steps (mirror‑descent on the probability simplex) on the *exact*
   max‑convolution objective.

4. **Targeted pair‑wise annealing** – a simulated‑annealing phase that
   repeatedly identifies the peak index of the convolution and moves a
   small fraction of mass away from the most contributing pair.  A final
   light random‑annealing round cleans up any remaining noise.

The routine respects the 1000 s budget, always returns a feasible vector,
and prints intermediate bounds so the progress is visible.
"""

import time
import numpy as np

# FFT helpers ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Full‑dimensional optimisation primitives ---------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior point
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.08
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near the deadline
        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 70_000 == 0 and it > 0:
            lr *= 0.93
        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated annealing (random mass transfers)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Targeted annealing that attacks the worst pair for the current peak."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generators ----------------------------------------------------

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    for sigma in (0.1, 0.3, 0.6):
        i = np.arange(n, dtype=np.float64)
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))
    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


# ----------------------------------------------------------------------
# Low‑dimensional Adam (basis‑parameterised) ---------------------------

def _adam_lowdim(
    w0: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    sub_dead: float,
    max_iters: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate where a = softmax(B·w).
    Returns the best full‑dimensional vector and its exact objective.
    """
    w = w0.copy()
    m = np.zeros_like(w)
    v = np.zeros_like(w)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.04
    lam_factor = 1.018

    best_a = None
    best_val = float("inf")
    it = 0

    while time.time() < sub_dead and it < max_iters:
        logits = B @ w
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Low‑dim Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        wgt = np.exp(lam * (conv - max_c))
        wgt_sum = wgt.sum()
        if wgt_sum == 0 or np.isnan(wgt_sum):
            wgt = np.full_like(conv, 1.0 / conv.size)
        else:
            wgt /= wgt_sum

        grad_a = 2.0 * _cross_correlation(wgt, a, corr_fft_len)

        # gradient wrt logits (softmax Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # chain rule to low‑dim parameters
        grad_w = B.T @ grad_log

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_w
        v = beta2 * v + (1.0 - beta2) * (grad_w ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        w -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 15_000 == 0 and time.time() > sub_dead - 30:
            w += np.random.randn(*w.shape) * 1e-3

        it += 1

    if best_a is None:      # fallback (should never happen)
        best_a = np.full(n, 1.0 / n)
        best_val = 2.0
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Searches for a non‑negative step‑function that
    minimises ``evaluate_sequence`` within the given time budget.
    Returns the best sequence as a plain Python list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (moderate – enough FFT speed, enough resolution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep ----------------------------------
    print("[Phase 1] parametric sweep")
    best_seq = None
    best_val = float("inf")

    cand = _initial_candidates(n, warm) + _spike_candidates(n)

    for a in cand:
        # a is already normalised to sum 1 in the generators
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – low‑dimensional Adam on a sinusoidal basis -------------
    # Build a modest Fourier‑type basis (constant + cosines + sines)
    d_cos = 10
    d_sin = 9
    i = np.arange(n, dtype=np.float64)
    x = i / (n - 1)
    cos_basis = np.column_stack([np.cos(np.pi * k * x) for k in range(d_cos)])
    sin_basis = np.column_stack([np.sin(np.pi * k * x) for k in range(1, d_sin + 1)])
    B = np.hstack([cos_basis, sin_basis])   # shape (n, d_cos+d_sin)
    p = B.shape[1]

    # Allocate roughly 30 % of the remaining budget to this phase
    phase2_dead = time.time() + (deadline - time.time()) * 0.30
    restarts = 0
    while time.time() < phase2_dead:
        # random low‑dim init; occasionally seed from the current best
        if best_seq is not None and np.random.rand() < 0.2:
            logit_guess = np.log(np.maximum(best_seq, 1e-12))
            w0 = np.linalg.lstsq(B, logit_guess, rcond=None)[0]
        else:
            w0 = np.random.randn(p) * 0.2
        a_opt, val_opt = _adam_lowdim(
            w0,
            B,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_dead=phase2_dead,
            max_iters=60_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Low‑dim] improved best {val_opt:.6f}")
        restarts += 1

    # ------------------------------------------------------------------
    # Phase 3 – full‑dimensional sub‑gradient descent -------------------
    if best_seq is not None and time.time() < deadline - 25:
        print("[Phase 3] sub‑gradient descent")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad] improved best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – targeted pairwise annealing ----------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase 4] targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[PairAnneal] improved best {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – light random annealing (use any remaining time) --------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase 5] final random annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=40_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback to uniform if, for any reason, nothing was found
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0

    # Clip to evaluator limits and final report
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
