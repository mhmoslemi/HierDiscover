def _simulated_annealing(a, n, conv_len, deadline, max_moves):
    a = a.copy()
    best_val = compute_value(a)
    best = a.copy()
    temperature = 0.1
    move = 0
    while time.time() < deadline and move < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-12: continue
        j = rng.integers(n)
        if i == j: continue
        delta = a[i] * rng.random() * 0.5
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = compute_value(a2)
        if val2 < best_val or rng.random() < np.exp((best_val - val2)/temperature):
            a = a2
            if val2 < best_val:
                best_val = val2
                best = a2.copy()
        temperature *= 0.9999
        move += 1
    return best, best_val

"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of three phases:

1. **Broad sampling** – Generate a very diverse set of candidate
   distributions (uniform, Dirichlet, edge‑exponential, edge‑power‑law,
   mixtures of those with a uniform component, inverted Hann/Kaiser
   windows, etc.).  All candidates are normalised to have sum = 1 and
   their exact objective values are evaluated with an FFT‑based
   convolution.  The best `top_k` vectors are retained.

2. **Refinement** – For each retained vector a projected‑gradient descent
   is performed on the true objective (the max‑convolution is
   differentiable almost everywhere via its active set).  The routine
   uses a back‑tracking line‑search and projects every iterate onto the
   probability simplex.  More time is given to the most promising
   candidates.

3. **Polishing** – A cheap simulated‑annealing hill‑climber explores
   random mass‑transfer moves starting from the best refined vector;
   occasional uphill moves are accepted according to a temperature schedule.
   The final best vector is returned (clipped to the [0, 1000] range).

The whole search respects a 1000 s wall‑clock budget and is deliberately
written without closures, lambdas or file I/O so that it can be used as a
drop‑in `propose_candidate` implementation.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT‑based convolution helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes (all returned *already normalised*)

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_exponential_mixed(n: int, gamma: float, w: float) -> np.ndarray:
    """Convex mixture of an edge‑exponential with the uniform distribution."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    base = left + left[::-1]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    shape = w * base + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _edge_power_law_mixed(n: int, alpha: float, w: float) -> np.ndarray:
    """Mixture of a power‑law edge shape with the uniform distribution."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    base = (idx ** alpha) + ((n - idx + 1) ** alpha)
    base = np.maximum(base, 0.0)
    base_sum = base.sum()
    if base_sum == 0:
        base = np.full(n, 1.0 / n)
    else:
        base /= base_sum
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    shape = w * base + (1.0 - w) * uniform
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _invert_hann(n: int) -> np.ndarray:
    """1 - Hann window (high at the edges, low in the centre)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * x)
    shape = 1.0 - hann
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _invert_kaiser(n: int, beta: float) -> np.ndarray:
    """Return a normalised 1 - Kaiser window (beta may be zero)."""
    # The function is optional – missing scipy yields a skip.
    try:
        from scipy.signal import kaiser
    except Exception:  # pragma: no cover
        return _uniform(n)
    w = kaiser(n, beta)
    shape = 1.0 - w / w.max()
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Objective evaluation helpers (pure, no security checks)

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(conv) / (sum(a)^2) for a (assumed non‑negative)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Projected gradient descent on the exact objective

def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on 2·n·max(conv)/(sum)**2."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    step0 = 0.5
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # back‑tracking line search
        step = step0
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climber with a simple simulated‑annealing schedule

def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    rng: np.random.Generator,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Random mass‑transfer moves, occasional uphill acceptance."""
    a = a0.copy()
    best_val = _objective(a, n, fft_len)
    best_a = a.copy()
    temperature = 0.2  # starting temperature
    it = 0

    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        # move a fraction (≤30 %) of the source mass
        delta = a[i] * rng.random() * 0.3
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_onto_simplex(a2)  # keep exact sum = 1

        val2 = _objective(a2, n, fft_len)

        # accept if better, otherwise accept with probability exp(-(Δ)/T)
        if val2 < best_val or rng.random() < math.exp((best_val - val2) / max(temperature, 1e-12)):
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 5_000 == 0:
                    print(f"[SA it={it:06d}] improved to {val2:.6f}")

        # exponential cooling
        temperature *= 0.99995
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that minimises the objective.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1. Broad generation of candidates
    candidates = []

    # Warm‑start (if supplied by the judge)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            candidates.append(w)
            for shift in (n // 4, n // 2, 3 * n // 4):
                candidates.append(np.roll(w, shift))

    # Uniform
    candidates.append(_uniform(n))

    # Random Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            candidates.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential family (dense grid + mixtures)
    for gamma in np.logspace(-2, 2, 40):               # 0.01 … 100
        candidates.append(_edge_exponential(n, float(gamma)))
        for wmix in np.linspace(0.0, 1.0, 5):
            candidates.append(_edge_exponential_mixed(n, float(gamma), float(wmix)))

    # Edge‑power‑law family (dense grid + mixtures)
    for alpha in np.linspace(-6.0, 2.0, 40):
        candidates.append(_edge_power_law(n, float(alpha)))
        for wmix in np.linspace(0.0, 1.0, 5):
            candidates.append(_edge_power_law_mixed(n, float(alpha), float(wmix)))

    # Beta‑pdf shapes (log‑grid)
    a_vals = np.logspace(-2, 2, 20)   # 0.01 … 100
    b_vals = np.logspace(-2, 2, 20)
    for a in a_vals:
        for b in b_vals:
            candidates.append(_beta_shape(n, float(a), float(b)))

    # Inverted Hann window
    candidates.append(_invert_hann(n))

    # Inverted Kaiser windows (if scipy is present)
    for beta in np.linspace(0.0, 10.0, 21):
        candidates.append(_invert_kaiser(n, float(beta)))

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool – keep the best few
    top_k = 30
    scored = []
    for idx, cand in enumerate(candidates):
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if time.time() >= start + 0.2 * budget_s:      # reserve time for refinement
            break

    scored.sort(key=lambda x: x[0])
    elite = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # 2. Refinement via projected gradient descent
    best_seq = None
    best_val = float("inf")

    for rank, a0 in enumerate(elite):
        if time.time() >= deadline - 5.0:
            break
        # Allocate a decreasing slice of the remaining budget
        remaining = deadline - time.time()
        # The first (best) candidates get a larger share
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        sub_deadline = time.time() + share

        a_opt, v_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=200_000,
        )
        if v_opt < best_val:
            best_val = v_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {v_opt:.6f}")

    # ------------------------------------------------------------------
    # 3. Final polishing with simulated annealing (cheap random moves)
    if best_seq is not None and time.time() < deadline - 1.0:
        a_loc, v_loc = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            rng,
            max_moves=300_000,
        )
        if v_loc < best_val:
            best_val = v_loc
            best_seq = a_loc.copy()
            print(f"[SA] final best {v_loc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Ensure everything complies with the evaluator's guards
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
