"""
Propose a low‑value step‑function sequence.

Algorithm
---------

* **Phase 1 – coarse random exploration (n≈512).**  
  Generate a large, diverse set of candidates (Dirichlet draws,
  beta‑pdf, edge‑exponential, edge‑power‑law, sinusoid, uniform,
  double‑Gaussian).  The parameters are sampled on a log‑scale so that
  extreme shapes are also inspected.  The best candidate is kept.

* **Phase 2 – coarse refinement.**  
  Starting from the best coarse candidate run a short projected‑gradient
  descent (PGD) on the exact max‑convolution and a quick Adam optimiser
  on a smooth‑max surrogate.  The result is still a vector of length
  512.

* **Phase 3 – up‑sample & fine optimisation (n=4096).**  
  Interpolate the coarse solution to the target length, renormalise and
  run a long PGD (and a final cheap hill‑climber) until the overall
  deadline is reached.

* **Safety fallback.**  
  If no candidate is ever found a uniform distribution of the required
  length is returned.

All vectors are kept on the probability simplex (sum = 1); the
objective `2·n·max(conv)/(sum²)` is scale‑invariant, so the final
sequence can be passed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Simplex projection
# ----------------------------------------------------------------------


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Candidate families
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta pdf on a discrete grid (mid‑point rule).  Handles a,b<1."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # Use log‑space for stability when a,b are far from 1
    with np.errstate(divide="ignore", invalid="ignore"):
        log_shape = (a - 1) * np.log(x) + (b - 1) * np.log1p(-x)
    shape = np.exp(log_shape)
    # Replace NaNs / infinities that may appear for extreme parameters
    shape = np.nan_to_num(shape, nan=0.0, posinf=0.0, neginf=0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
#  Objective (exact, scale‑invariant)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²).  a is assumed non‑negative."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Optimisers (same as the reference implementation)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max projection onto simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent style through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Upscaling helper
# ----------------------------------------------------------------------


def _upscale(a_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from a_small (size m) to size n_large."""
    m = a_small.shape[0]
    if m == n_large:
        return a_small.copy()
    x_small = np.linspace(0, 1, m, endpoint=False)
    x_large = np.linspace(0, 1, n_large, endpoint=False)
    a_large = np.interp(x_large, x_small, a_small)
    a_large = np.maximum(a_large, 0.0)
    s = a_large.sum()
    if s < 1e-12:
        a_large = np.full(n_large, 1.0 / n_large)
    else:
        a_large /= s
    return a_large


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget and always returns the best
    sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.2  # safety margin

    # problem sizes
    n_fine = 4096
    n_coarse = 512

    conv_len_fine = _next_pow2(2 * n_fine - 1)
    corr_len_fine = _next_pow2(3 * n_fine - 2)
    conv_len_coarse = _next_pow2(2 * n_coarse - 1)
    corr_len_coarse = _next_pow2(3 * n_coarse - 2)

    # ------------------------------------------------------------------
    # Phase 1 – random search on the coarse grid
    # ------------------------------------------------------------------
    phase1_deadline = start_time + 0.4 * budget_s
    best_a_coarse = None
    best_val_coarse = float("inf")
    samples = 0

    while time.time() < phase1_deadline:
        choice = rng.integers(0, 7)  # 0..6

        if choice == 0:  # Dirichlet
            alpha = rng.choice([0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 1.5, 2.0, 3.0, 5.0])
            a = _random_dirichlet(n_coarse, float(alpha))
        elif choice == 1:  # Beta pdf
            a_param = 10 ** rng.uniform(-1, 1)   # 0.1 – 10
            b_param = 10 ** rng.uniform(-1, 1)
            a = _beta_shape(n_coarse, a_param, b_param)
        elif choice == 2:  # Edge‑exponential
            gamma = 10 ** rng.uniform(-2, 1)    # 0.01 – 10
            a = _edge_exponential(n_coarse, gamma)
        elif choice == 3:  # Edge‑power‑law
            alpha_pow = rng.uniform(-4.0, -0.2)
            a = _edge_power_law(n_coarse, alpha_pow)
        elif choice == 4:  # Half‑wave sinusoid (central mass – kept for diversity)
            a = _sinusoid(n_coarse)
        elif choice == 5:  # Uniform
            a = _uniform(n_coarse)
        else:               # Double Gaussian at the ends
            sigma = rng.uniform(0.02, 0.3)
            a = _double_gaussian(n_coarse, sigma)

        # normalise to the simplex
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s

        val = _objective(a)
        if val < best_val_coarse:
            best_val_coarse = val
            best_a_coarse = a.copy()
            print(f"[Coarse rnd] samples={samples} val={val:.6f}")

        samples += 1

    print(f"Coarse random search finished – best {best_val_coarse:.6f} after {samples} samples")

    # ------------------------------------------------------------------
    # Phase 2 – coarse refinement (PGD + short Adam)
    # ------------------------------------------------------------------
    if best_a_coarse is not None and time.time() < deadline - 250:
        # PGD for a while
        pgd_deadline = time.time() + 0.15 * budget_s
        a_c, v_c = _pgd_optimize(
            best_a_coarse,
            n_coarse,
            conv_len_coarse,
            corr_len_coarse,
            pgd_deadline,
            max_iters=500_000,
            eta0=1.0,
        )
        if v_c < best_val_coarse:
            best_val_coarse = v_c
            best_a_coarse = a_c.copy()
            print(f"[Coarse PGD] improved to {v_c:.6f}")

        # A few Adam iterations
        if time.time() < deadline - 200:
            adam_deadline = time.time() + 0.1 * budget_s
            a_c2, v_c2 = _adam_optimize(
                best_a_coarse,
                n_coarse,
                conv_len_coarse,
                corr_len_coarse,
                adam_deadline,
                max_iters=200_000,
                lr_start=0.03,
            )
            if v_c2 < best_val_coarse:
                best_val_coarse = v_c2
                best_a_coarse = a_c2.copy()
                print(f"[Coarse Adam] improved to {v_c2:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – up‑scale to the full size and fine optimisation
    # ------------------------------------------------------------------
    if best_a_coarse is None:
        best_a_coarse = _uniform(n_coarse)

    a_fine = _upscale(best_a_coarse, n_fine)
    best_val_fine = _objective(a_fine)
    print(f"Upscaled to n={n_fine}, initial fine value {best_val_fine:.6f}")

    # Fine PGD for the remaining budget (≈80 % of the rest)
    remaining = deadline - time.time()
    if remaining > 30:
        pgd_fine_deadline = time.time() + 0.8 * remaining
        a_f, v_f = _pgd_optimize(
            a_fine,
            n_fine,
            conv_len_fine,
            corr_len_fine,
            pgd_fine_deadline,
            max_iters=2_000_000,
            eta0=0.8,
        )
        if v_f < best_val_fine:
            best_val_fine = v_f
            a_fine = a_f.copy()
            print(f"[Fine PGD] improved to {v_f:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climber (last seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        local_deadline = deadline - 1.0
        a_f, v_f = _local_random_search(
            a_fine,
            n_fine,
            conv_len_fine,
            local_deadline,
            max_moves=150_000,
        )
        if v_f < best_val_fine:
            best_val_fine = v_f
            a_fine = a_f.copy()
            print(f"[Final local] improved to {v_f:.6f}")

    # ------------------------------------------------------------------
    # Safeguard and return
    # ------------------------------------------------------------------
    best_seq = np.clip(a_fine, 0.0, 1000.0).tolist()
    print(f"Final best value {best_val_fine:.6f} (length {len(best_seq)})")
    return best_seq
