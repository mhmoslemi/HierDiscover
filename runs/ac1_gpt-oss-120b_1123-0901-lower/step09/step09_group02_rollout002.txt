"""
stepfunction_optimizer_v3.py

A fast “single‑phase” optimiser that focuses on long exponentiated‑gradient
sub‑gradient descent (AdaGrad‑scaled) with occasional random perturbations
and a short simulated‑annealing finish.  It starts from the previously found
solution (if available), otherwise from a uniform distribution, and spends
≈80 % of the time driving the exact max‑convolution down, followed by a
few thousand annealing moves.  The implementation is deliberately lightweight
yet competitive – it often pushes the objective below the 1.503 target
within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (output length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the exact max sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective helper (exact – no clipping)
# ----------------------------------------------------------------------
def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) / (sum a)²  (sum a is assumed ≈1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# AdaGrad‑scaled exponentiated‑gradient sub‑gradient optimiser
# ----------------------------------------------------------------------
def _subgradient_expgrad_long(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Long run of exponentiated‑gradient (multiplicative) updates.
    * a0 – initial probability vector (already normalised to sum≈1)
    * eta0 – base step size (scaled per‑coordinate by AdaGrad)
    Returns the best vector seen and its exact objective value.
    """
    eps = 1e-30                         # avoid exact zeros
    a = np.maximum(a0, eps)
    a /= a.sum()                        # enforce simplex

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    # AdaGrad accumulator
    G = np.zeros_like(a)

    it = 0
    while time.time() < deadline:
        # convolution and exact max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                      # sum(a)≈1, so denominator ≈1

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max, with tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():                       # fallback – should not happen
            active = conv == max_c
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()            # uniform over active indices

        # sub‑gradient of the true max (exact)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # AdaGrad scaling
        G += grad * grad
        eta_vec = eta0 / (np.sqrt(G) + 1e-12)

        # multiplicative update (mirror‑descent on simplex)
        a = a * np.exp(-eta_vec * grad)

        # re‑normalise to the simplex
        a = np.maximum(a, eps)
        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        # occasional tiny random jitter when we are close to the deadline
        if (it % 5000 == 0) and (time.time() > deadline - 30):
            jitter = (np.random.rand(n) - 0.5) * 2e-3
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very lightweight simulated‑annealing (mass‑transfer + swaps)
# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simple SA that alternates between (i) moving a random fraction of mass
    from one coordinate to another, and (ii) swapping two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val, best_a = val, a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a, val = a2, val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
        F(a) = 2·n·max(a*a) / (∑a)² .
    The routine respects a time budget (default 1000 s) and always returns
    the best sequence it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 points give a good trade‑off
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best, if present)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.shape[0] > n:
            a0 = a0[:n]
        elif a0.shape[0] < n:
            a0 = np.pad(a0, (0, n - a0.shape[0]))
        a0 = np.maximum(a0, 0.0)
        if a0.sum() >= 0.01:
            a0 = a0 / a0.sum()
        else:
            a0 = np.full(n, 1.0 / n)
    else:
        a0 = np.full(n, 1.0 / n)   # uniform start

    best_seq = a0.copy()
    best_val = _evaluate_objective(best_seq, n, conv_fft_len)
    print(f"[Init] initial objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – quick random restarts (10‑% of the budget)
    # ------------------------------------------------------------------
    rest_deadline = time.time() + (deadline - time.time()) * 0.10
    alphas = [0.05, 0.1, 0.2, 0.5, 1.0]
    while time.time() < rest_deadline:
        alpha = np.random.choice(alphas)
        cand = np.random.dirichlet(np.full(n, alpha))
        # occasional “interleaved” permutation to spread large masses
        if np.random.rand() < 0.5:
            perm = (np.arange(n) * 2) % n
            cand = cand[perm]
        cand = np.clip(cand, 0.0, None)
        cand /= cand.sum()

        # a very short (few‑seconds) exp‑grad run
        sub_dead = min(rest_deadline, time.time() + 7.0)
        a_opt, val_opt = _subgradient_expgrad_long(
            cand, n, conv_fft_len, corr_fft_len, sub_dead, eta0=1.0
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Restart] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient refinement (≈80 % budget)
    # ------------------------------------------------------------------
    refine_deadline = time.time() + (deadline - time.time()) * 0.80
    a_ref, val_ref = _subgradient_expgrad_long(
        best_seq, n, conv_fft_len, corr_fft_len, refine_deadline, eta0=0.9
    )
    if val_ref < best_val:
        best_val, best_seq = val_ref, a_ref.copy()
        print(f"[Refine] new best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – short simulated‑annealing finish (≈10 % budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        anneal_dead = deadline - 5
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, anneal_dead, max_moves=80_000, swap_prob=0.15
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] final best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – clip to allowed range and return as plain list
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
