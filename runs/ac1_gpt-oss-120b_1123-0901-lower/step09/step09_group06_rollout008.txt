def _adam_exact_optimize(a0, n, conv_fft_len, deadline, max_iters=300000, lr_start=0.02):
    eps = 1e-12
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9; beta2 = 0.999; eps_adam = 1e-8
    lr = lr_start
    best_a = a0.copy()
    best_val = 2 * n * np.max(_full_convolution(a0, conv_fft_len)) / (np.sum(a0)**2)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute gradient for max:
        k_max = np.argmax(conv)
        # gradient of max w.r.t a:
        grad_max = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >=0) & (idx < n)
        grad_max[mask] = 2.0 * a[idx[mask]]
        # gradient of objective w.r.t a (since sum=1, we ignore denominator):
        grad_a = 2.0 * n * grad_max  # because F = 2n * max, derivative 2n * grad_max
        # Actually derivative of max conv only: grad_conv = grad_max, derivative of F = 2*n*grad_max.
        # We'll compute grad = 2 * n * grad_max.
        grad_F = 2.0 * n * grad_max
        # Chain rule: grad wrt logits
        avg = np.dot(a, grad_F)
        grad_log = a * (grad_F - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1**(it+1))
        v_hat = v / (1 - beta2**(it+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        # maybe decay lr with iteration.
        if it % 50000 == 0 and it > 0:
            lr *= 0.8
        # add occasional noise near deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-3
        it += 1
    return best_a, best_val

"""
Optimisation of the step–function objective
-------------------------------------------

The goal is to minimise

    F(a) = 2·n·max(a∗a) / (∑a)²

over non‑negative vectors a of length *n* (the sum can be fixed to 1
without loss of generality).  The code follows four phases:

1. **Broad initialisation** – a very diverse set of deterministic families
   (edge‑exponential, edge‑power‑law, β‑pdf, double‑Gaussian, raised‑cosine,
   semicircle, sinusoid, uniform, sparse spikes) together with many
   random Dirichlet draws.  All candidates are normalised to unit sum and the
   best few are retained.

2. **Exact‑gradient Adam** – the true objective (max‑convolution) is
   differentiable everywhere except where the location of the maximum
   changes.  Using the analytic gradient of the active convolution entry
   we run Adam on the soft‑max (log‑its → simplex) parametrisation.
   This brings the solution close to a (local) optimum of the exact objective.

3. **Continuation L‑BFGS** – a smooth‑max surrogate (log‑sum‑exp) is
   optimised at a schedule of increasing temperatures α.  This stage can
   escape shallow plateaus left by the Adam phase.

4. **Cheap hill‑climber** – random “mass‑moves’’ between indices are
   accepted only when they improve the true objective.  The last seconds
   of the budget are devoted to this cheap local search.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns the best sequence found as a plain Python list.
All helper functions are defined at top‑level; no closures or lambdas are used.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the smooth‑max gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families of candidate shapes
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Exact‑gradient Adam optimiser (max‑convolution)
def _adam_exact_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the *exact* objective  F = 2·n·max(a∗a).
    The iterate lives on the simplex via a soft‑max (logits → probabilities).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start

    best_a = a0.copy()
    conv = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a0.sum() ** 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current objective
        conv = _full_convolution(a, conv_fft_len)
        k_max = int(np.argmax(conv))
        max_c = float(conv[k_max])
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[AdamExact it={it:06d}] new best {val:.6f}")

        # gradient of the active convolution entry
        grad_max = np.zeros_like(a)
        idx = k_max - np.arange(n)
        mask = (idx >= 0) & (idx < n)
        grad_max[mask] = 2.0 * a[idx[mask]]

        # gradient of the true objective (sum a = 1, so denominator fixed)
        grad_a = 2.0 * n * grad_max

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.8

        # tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-3

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max L‑BFGS continuation (identical to the original but with a larger schedule)
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Smooth‑max objective + gradient for a given temperature α."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c  # ≤ 0
    exp_alpha_v = np.exp(alpha * v)
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    w = exp_alpha_v / sum_exp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    L‑BFGS on a ladder of increasing smooth‑max temperatures.
    Returns the best exact sequence and its true objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    best_logits = logits.copy()
    best_val = float("inf")
    alphas = [5.0, 20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len, corr_fft_len, alpha)

        remaining = deadline - time.time()
        max_iter = 120 if remaining > 30 else 40

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = 2.0 * n * np.max(_full_convolution(a, conv_fft_len)) / (a.sum() ** 2)
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val

# Cheap mass‑move hill‑climber
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Random local moves that are accepted only when they improve the true objective."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[LocalMove {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

        if time.time() > deadline - 120:  # reserve ~2 min for deep optimisation
            break

    # store the overall best of the scan
    current_best_idx = int(np.argmin(best_vals))
    best_seq = best_candidates[current_best_idx].copy()
    best_val = best_vals[current_best_idx]
    print(f"[Initial] best value after scan {best_val:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – Exact‑gradient Adam optimisation on each of the top candidates
    remaining = deadline - time.time()
    if remaining > 30:
        per_candidate_time = remaining / len(best_candidates)
        for i, cand in enumerate(best_candidates):
            sub_deadline = time.time() + per_candidate_time - 0.5
            a_opt, val_opt = _adam_exact_optimize(
                cand,
                n,
                conv_fft_len,
                sub_deadline,
                max_iters=200_000,
                lr_start=0.04,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[AdamExact cand {i}] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – Continuation L‑BFGS refinement (if we still have time)
    if time.time() < deadline - 30:
        a_opt, val_opt = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[L‑BFGS] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – Cheap random hill‑climber for the final few seconds
    if time.time() < deadline - 3:
        a_opt, val_opt = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Local final] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
