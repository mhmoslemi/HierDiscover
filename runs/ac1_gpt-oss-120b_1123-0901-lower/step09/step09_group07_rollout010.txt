"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Problem size** – we use a larger support (n≈8192) which gives a finer
   discretisation of the continuous formulation and therefore a better
   lower bound.

2. **Diverse parametric sweep** – a fast “band‑it” style loop draws millions
   of candidates from several families (edge‑exponential, edge‑power‑law,
   beta, double‑gaussian, sinusoid‑power, sigmoid, cauchy, monotone linear,
   Dirichlet).  For each candidate the cheap FFT‑based convolution is evaluated
   and the best *K* candidates are retained.

3. **Multi‑start gradient refinement** – each of the *K* retained vectors,
   together with a few fresh random Dirichlet seeds, is fed to an Adam optimiser
   that minimises the smooth‑max surrogate of the max‑convolution.  The Adam
   runs are independent and are allocated a fair share of the remaining
   budget; the best result of all runs is kept.

4. **Sub‑gradient / Projected‑gradient polishing** – the best Adam result is
   refined by one pass of exponentiated‑gradient sub‑gradient descent and then
   by Euclidean projected gradient descent (both using the exact max‑convolution
   objective).

5. **Final stochastic hill‑climbing** – a cheap mass‑transfer local search
   (move a random fraction of probability mass from one index to another)
   runs until the deadline, accepting any improvement.

All vectors are kept on the probability simplex (non‑negative, sum≈1), which
makes the objective scale‑invariant.  The final vector is clipped to the range
accepted by `evaluate_sequence` and returned as a plain Python list.

The routine is deliberately time‑aware; it never exceeds the 1000 s budget.
"""

import time
import numpy as np
import math

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²).  Fast evaluation using FFT."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Shape generators -------------------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    """γ‑exponential decay from each edge; weight mixes left/right."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sigmoid_shape(n: int, slope: float, mid: float) -> np.ndarray:
    """Logistic (sigmoid) shape on [0, n‑1]."""
    x = np.arange(n, dtype=np.float64)
    shape = 1.0 / (1.0 + np.exp(-slope * (x - mid)))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _cauchy_shape(n: int, scale: float) -> np.ndarray:
    """Heavy‑tailed Cauchy‑like shape centred at the middle."""
    x = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    shape = 1.0 / (1.0 + (x / scale) ** 2)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _linear_monotone_shape(n: int, slope: float) -> np.ndarray:
    """Linear monotone increasing (slope>0) or decreasing (slope<0)."""
    x = np.arange(n, dtype=np.float64)
    shape = 1.0 + slope * (x - (n - 1) / 2.0)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # temperature (inverse)
    lam_factor = 1.04    # a little faster rise than in the reference code

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max surrogate weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (log‑space) update
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional tiny jitter when the deadline is near
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.90

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (rng.random() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – a slightly larger n gives a finer approximation.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous run stored a vector)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – aggressive parametric sampling
    best_val = float("inf")
    best_seq = None
    # keep the top‑K vectors for later refinement
    K = 12
    top_vals = [float("inf")] * K
    top_seqs = [None] * K

    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "sin_pow", "tent", "sigmoid", "cauchy", "linear_mono", "dirichlet"
    ]
    phase1_deadline = min(deadline - 30.0, time.time() + 0.20 * (deadline - time.time()))
    print("Phase 1 – random parametric sweep (n={})".format(n))

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-3, 4)          # 0.001 … 10000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-12.0, -0.05)        # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 2)
            b = 10 ** rng.uniform(-2, 2)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.2, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "sigmoid":
            slope = rng.uniform(-10.0, 10.0)
            mid = rng.uniform(0, n - 1)
            cand = _sigmoid_shape(n, slope, mid)
        elif fam == "cauchy":
            scale = rng.uniform(0.01 * n, 0.5 * n)
            cand = _cauchy_shape(n, scale)
        elif fam == "linear_mono":
            slope = rng.uniform(-5.0 / n, 5.0 / n)   # small slope to keep numbers modest
            cand = _linear_monotone_shape(n, slope)
        else:   # dirichlet
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Phase 1] new global best {val:.6f} via {fam}")

        # keep top‑K
        if val < max(top_vals):
            idx = top_vals.index(max(top_vals))
            top_vals[idx] = val
            top_seqs[idx] = cand.copy()

    # Include warm‑start shifted versions as extra candidates
    if warm is not None:
        for shift in (0, n // 8, n // 4, n // 2):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")
            if val < max(top_vals):
                idx = top_vals.index(max(top_vals))
                top_vals[idx] = val
                top_seqs[idx] = cand.copy()

    # ------------------------------------------------------------------
    # Phase 2 – multi‑start Adam optimisation
    # We now have up to K+warm candidates.  Run Adam on each of them, each
    # getting an equal slice of the remaining budget.
    remaining = deadline - time.time()
    if remaining > 10.0:
        # reserve a small slice for final polishing
        opt_deadline = time.time() + remaining - 8.0
        candidates = [seq for seq in top_seqs if seq is not None]
        if best_seq is not None and best_seq not in candidates:
            candidates.append(best_seq)

        per_run = max(2, opt_deadline - time.time()) / (len(candidates) + 1e-9)
        print(f"Phase 2 – Adam optimisation on {len(candidates)} starts")
        for idx, cand in enumerate(candidates):
            run_deadline = time.time() + per_run
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                run_deadline,
                max_iters=400_000,
                lr_start=0.05,
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Adam #{idx}] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – sub‑gradient then projected‑gradient polishing
    if best_seq is not None and time.time() < deadline - 6.0:
        print("Phase 3 – Sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline - 4.0,
            max_iters=250_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Sub‑grad] improved to {val_sub:.6f}")

        if time.time() < deadline - 2.0:
            print("Phase 3 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline - 1.0,
                max_iters=200_000,
                eta0=0.35,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climbing for the final seconds
    if best_seq is not None and time.time() < deadline - 0.5:
        print("Phase 4 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
