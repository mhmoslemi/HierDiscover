"""
Propose a non‑negative step function (vector) that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

Algorithm
---------

We use three phases tuned for the 1000 s wall‑clock budget.

1. **Diverse seeding** – generate a modest pool of candidates
   (deterministic families, Dirichlet draws, and a warm start if
   available).  All vectors are normalised to sum to 1 and the exact
   objective is evaluated with an FFT‑based convolution.  The best
   `top_k` candidates survive.

2. **Smoothed gradient descent (annealed soft‑max)** – for every elite
   candidate we run a projected gradient descent on a differentiable
   surrogate of the max‑convolution:
        Lτ(a) = (1/τ)·log Σk exp( τ·(a∗a)[k] ).
   The weight vector (soft‑max) provides a dense gradient everywhere,
   allowing the optimiser to escape shallow local minima.  τ is
   increased slowly (annealing) so that the surrogate converges to the
   true max.  The projection onto the probability simplex enforces
   non‑negativity and unit sum.

3. **Active‑set refinement & cheap local moves** – after the smooth
   phase we switch to the exact sub‑gradient (active‑set) method from
   the reference implementation for a short final polish, followed by
   a guided hill‑climber that transfers a small fraction of mass from
   indices with large gradient to those with small gradient.

All phases respect the global 1000 s deadline and the best
vector found is returned as a plain Python list.
"""

import time
import numpy as np

# Helper utilities (FFT, projection, families, etc.)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------- deterministic families (already normalised) -------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse seed pool (deterministic families + random draws)."""
    pool = []

    # Dirichlet draws with several concentrations
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(2):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf – a few symmetric & skewed cases
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Sinusoid and uniform (always useful)
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool

# Smoothed (soft‑max) gradient descent with annealing
def _smooth_gd(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    init_step: float = 0.2,
    init_tau: float = 5.0,
    max_tau: float = 2000.0,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the log‑sum‑exp surrogate of max."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()          # true objective (sum = 1)

    step = init_step
    tau = init_tau

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c

        # update annealing temperature (slow exponential growth)
        tau = min(max_tau, tau * 1.001)

        # soft‑max weights (stable version)
        exp_term = np.exp(tau * (conv - max_c))   # max gets exp(0)=1
        weight = exp_term / exp_term.sum()

        # gradient of the surrogate (same form as sub‑gradient)
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # simple back‑tracking line‑search (projected)
        cur_step = step
        improved = False
        while cur_step > 1e-12:
            a_new = a - cur_step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()

            if val_new <= cur_val:
                a = a_new
                cur_val = val_new
                improved = True
                # keep the best ever seen
                if val_new < best_val:
                    best_val = val_new
                    best_a = a.copy()
                    print(f"[SmoothGD it={it:06d}] new best {best_val/(2*n):.6f}")
                break
            cur_step *= 0.5

        # adapt step size for next iteration
        if improved:
            step = min(step * 1.1, 1.0)
        else:
            step *= 0.5
        # safety break
        if step < 1e-12:
            break

        it += 1

    return best_a, best_val / (2 * n)   # return the *objective* value (not 2n*max)


# ----------------------------------------------------------------------
# Exact active‑set gradient descent (fast variant of the reference)
def _active_set_gd(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent using the true max (active‑set) sub‑gradient."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    step = 0.5

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[ActiveGD it={it:06d}] new best {best_val/(2*n):.6f}")

        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # back‑tracking line‑search (projected)
        cur_step = step
        while cur_step > 1e-12:
            a_new = a - cur_step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()

            if val_new <= val:
                a = a_new
                break
            cur_step *= 0.5
        step = cur_step   # keep the last successful step length
        it += 1

    return best_a, best_val / (2 * n)


# ----------------------------------------------------------------------
# Guided hill‑climber (deterministic mass transfers)
def _guided_hillclimb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Transfer mass from high‑gradient to low‑gradient positions."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max()
    best_val = cur_val
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        # recompute active‑set based gradient
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        src = np.argmax(grad)          # decreasing this reduces the max
        dst = np.argmin(grad)          # increasing this helps
        if a[src] < 1e-14 or src == dst:
            # fallback to a tiny random move
            src = np.random.randint(n)
            dst = np.random.randint(n)
            if src == dst or a[src] < 1e-14:
                it += 1
                continue

        delta = min(a[src] * 0.2, 0.01)     # move up to 20 % of source or 0.01
        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            if it % 5000 == 0:
                print(f"[HillClimb it={it:06d}] improved to {best_val/(2*n):.6f}")

        it += 1

    return best_a, best_val / (2 * n)


# ----------------------------------------------------------------------
# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a near‑optimal step‑function within the given time budget."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # problem size – we deliberately use a larger n than the previous run
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start handling (if a previous vector is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            repeats = n // warm.shape[0] + 1
            warm = np.tile(warm, repeats)[:n]
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – generate a moderate, diverse initial pool
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # quick evaluation of the whole pool (stop early if we run out of time)
    top_k = 12
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        scored.append((val, a))
        if time.time() > deadline * 0.2:   # reserve ~80 % of time for optimisation
            break

    scored.sort(key=lambda x: x[0])
    elite = [a for _, a in scored[:top_k]]
    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 2 – smoothed gradient descent (annealed soft‑max) on elites
    for idx, a0 in enumerate(elite):
        if time.time() >= deadline - 20:
            break
        # allocate a decreasing share of the remaining budget
        remaining = deadline - time.time()
        share = remaining * (top_k - idx) / max(1, (top_k - idx) * (top_k - idx + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _smooth_gd(
            a0, n, conv_fft_len, corr_fft_len, local_deadline,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[SmoothGD #{idx}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – exact active‑set refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        a_ref, val_ref = _active_set_gd(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=150_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[ActiveRefine] improved global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap guided hill‑climber for final polishing
    if best_seq is not None and time.time() < deadline - 2:
        a_fin, val_fin = _guided_hillclimb(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_moves=200_000,
        )
        if val_fin < best_val:
            best_val = val_fin
            best_seq = a_fin.copy()
            print(f"[HillClimb] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the bounds imposed by evaluate_sequence (harmless safety)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
