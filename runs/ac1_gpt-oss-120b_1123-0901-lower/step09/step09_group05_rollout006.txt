"""
A three‑stage optimizer for the step‑function objective

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The implementation improves on the reference code by

* exploring a much richer set of parametric families (asymmetric exponentials,
  powered sinusoids, extra Dirichlet draws, etc.);
* keeping a larger elite set (top‑k = 30) and allocating a geometric
  budget of the remaining time to them;
* running a longer projected‑gradient descent (sub‑gradient on the exact
  max‑convolution) with back‑tracking line‑search;
* finishing with an aggressive hill‑climbing phase that adapts the moved
  mass and performs up to a few million cheap random transfers.

All operations respect the 1000 s wall‑clock limit and the final sequence
is returned as a plain Python list compatible with the provided
`evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)−1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Deterministic families (all return vectors with sum == 1) --------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _sinusoid_power(n: int, p: float) -> np.ndarray:
    base = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(base, 0.0) ** p
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_exponential_asym(n: int, gamma: float) -> np.ndarray:
    """Single‑sided exponential decay, then normalised."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    out = []
    if warm is not None:
        out.append(warm.copy())
        # three quarter‑turn shifts
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


# ----------------------------------------------------------------------
# Candidate pool creation -------------------------------------------------

def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Coarse deterministic families + a few Dirichlet draws."""
    pool = []

    # Dirichlet draws with a range of concentration parameters
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(2):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (symmetric)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑exponential asymmetric (single side)
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        pool.append(_edge_exponential_asym(n, gamma))

    # Power‑law
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf (symmetric and skewed)
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool


def _grid_candidates(n: int) -> list[np.ndarray]:
    """Finer sweep over the main parametric families."""
    out = []

    # Edge‑exponential – log‑spaced (symmetric)
    for gamma in np.logspace(-2, 1.5, 30):
        out.append(_edge_exponential(n, float(gamma)))

    # Edge‑exponential – asymmetric
    for gamma in np.logspace(-2, 1.5, 20):
        out.append(_edge_exponential_asym(n, float(gamma)))

    # Edge‑power‑law – linear sweep
    for alpha in np.linspace(-4.0, 2.0, 30):
        out.append(_edge_power_law(n, float(alpha)))

    # Beta – grid in log‑space
    a_vals = np.logspace(-1, 1, 12)
    b_vals = np.logspace(-1, 1, 12)
    for a in a_vals:
        for b in b_vals:
            out.append(_beta_shape(n, float(a), float(b)))

    # Double‑Gaussian – linear sweep
    for sigma in np.linspace(0.02, 0.5, 30):
        out.append(_double_gaussian(n, float(sigma)))

    # Powered sinusoid
    for p in np.linspace(0.5, 4.0, 15):
        out.append(_sinusoid_power(n, float(p)))

    return out


# ----------------------------------------------------------------------
# Exact projected gradient descent ---------------------------------------

def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  F(a)=2·n·max(a*a)/sum(a)^2.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        # --- objective and active set ---------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2500 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # Active‑set weights (uniform over all indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        weight = active.astype(float)
        wsum = weight.sum()
        if wsum > 0:
            weight /= wsum

        # Sub‑gradient of the max‑convolution term
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # ----- back‑tracking line‑search (projected) -----------------
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climber -----------------------------------------------------

def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap stochastic hill‑climber: transfer a random fraction of the
    mass from one index to another and keep the move if it improves the exact
    objective.  The fraction of moved mass decays linearly from 0.5 to 0.01.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # linearly decreasing move size
        frac = 0.5 - 0.49 * (it / max_moves)   # from 0.5 down to 0.01
        delta = a[i] * np.random.rand() * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Return a non‑negative step function (as a Python list) that approximately
    minimizes 2·n·max(a∗a)/(sum(a)^2) within the allotted time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # keep a small safety margin for interpreter overhead
    deadline = start_time + budget_s - 0.5

    # --------------------------------------------------------------
    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # optional warm start -------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a rich and diverse pool of starter vectors -----------------
    pool = _warm_and_shifts(warm, n)
    pool += _initial_pool(n, rng)
    pool += _grid_candidates(n)

    # Evaluate the whole pool (or as many as time permits) ---------------
    top_k = 30
    scored = []
    pool_evaluation_deadline = start_time + budget_s * 0.15  # 15 % of total budget
    for idx, cand in enumerate(pool):
        if time.time() >= pool_evaluation_deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        scored.append((val, a))

    if not scored:
        # fallback – uniform distribution
        return _uniform(n).tolist()

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # --------------------------------------------------------------
    # Phase 2 – refined projected gradient descent on the elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    total_weight = top_k * (top_k + 1) // 2  # sum_{i=1..top_k} i
    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 5:
            break
        remaining = deadline - time.time()
        # allocate a linearly decreasing share of the remaining budget
        share = remaining * (top_k - rank) / total_weight
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=500_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase 2 #{rank}] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final cheap local moves (uses any remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=2_000_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase 3] final best {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback guard – should never happen, but keeps the contract safe
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clamp to the allowed numeric range (the evaluator does this again)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
