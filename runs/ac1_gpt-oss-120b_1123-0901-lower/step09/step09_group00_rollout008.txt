def propose_candidate(seed=0, budget_s=1000):
    """
    Hybrid optimizer with parameter sweep and extended local refinement.
    The algorithm:
    1. Generate a set of parametric distributions (uniform, arcsine,
       Beta(α,β) for a grid of α,β) and select the best according to the
       evaluation function.
    2. Use this as a warm start for a long Adam optimisation on the
       smooth‑max surrogate (log‑sum‑exp) with a slowly increasing temperature.
    3. Refine with exponentiated‑gradient sub‑gradient descent.
    4. Perform projected gradient descent on the exact max‑convolution.
    5. Finish with a long simulated‑annealing local search on the simplex.
    6. Return the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # resize by interpolation
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() < 0.01:
            warm = None
        else:
            warm = warm / warm.sum()
    else:
        warm = None

    # Phase A: parametric sweep (Beta distributions and arcsine blend)
    best_seq = None
    best_val = float('inf')
    # define candidate generation functions
    def beta_shape(alpha, beta):
        x = (np.arange(n) + 0.5) / n
        # Guard against alpha,beta < 1 causing infinities at edges, but midpoint ensures finite.
        shape = np.power(x, alpha - 1) * np.power(1.0 - x, beta - 1)
        shape = np.where(np.isfinite(shape), shape, 0.0)
        if shape.sum() == 0:
            shape = np.full_like(shape, 1.0 / n)
        else:
            shape /= shape.sum()
        return shape

    # candidates from grid
    alphas = [0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 2.0, 5.0]
    for a in alphas:
        for b in alphas:
            cand = beta_shape(a, b)
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Sweep] Beta({a},{b}) val {val:.6f}")

    # also test arcsine (0.5,0.5) + uniform mix
    uniform = np.full(n, 1.0 / n)
    arcsine = beta_shape(0.5, 0.5)
    for lam in np.linspace(0, 1, 11):

        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Sweep] mix λ={lam:.2f} val {val:.6f}")

    # if we have warm start with better objective, use it
    if warm is not None:
        val_warm = evaluate_sequence(warm.tolist())
        if val_warm < best_val:
            best_val = val_warm
            best_seq = warm.copy()
            print(f"[Warm] warm start val {val_warm:.6f}")

    # Phase B: Adam optimisation
    if best_seq is not None and time.time() < deadline - 10:
        sub_dead = min(deadline, time.time() + 120.0)  # allocate up to 2 minutes
        a_adam, val_adam = _adam_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                                          max_iters=500_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # Phase D: projected gradient descent on exact max conv
    if best_seq is not None and time.time() < deadline - 15:
        a_gd, val_gd = _max_conv_descent(

        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")

2. **Adam on a smooth‑max surrogate** – optimises the log‑sum‑exp
   approximation of the max‑convolution, slowly raising the temperature.

3. **Exponentiated‑gradient sub‑gradient phase** – optimises the exact
   max‑convolution using the sub‑gradient obtained from the active set.

4. **Projected gradient descent** on the exact max‑convolution.

5. **Extended simulated‑annealing** – local moves on the simplex with a
   long cooling schedule to escape shallow local minima.

All vectors are kept on the probability simplex (sum = 1), because the
objective is homogeneous of degree 0.  The routine respects the global
time budget (≈ 1000 s) and always returns the best sequence encountered.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns (best_vector, best_objective).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns (best_vector, best_objective).  The iterate always stays on
    the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Plain projected gradient descent on the exact max‑convolution.
    Returns (best_vector, best_objective).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Returns (best_vector, best_objective).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-6
    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of the mass from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """
    Discretised Beta(α,β) density on the mid‑points of n cells,
    normalised to sum to 1.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    if shape.sum() == 0.0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # interpolate to the required length
            x_old = np.linspace(0, 1, warm.shape[0], endpoint=False)
            x_new = np.linspace(0, 1, n, endpoint=False)
            warm = np.interp(x_new, x_old, warm)
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase A – cheap parametric sweep (Beta families + uniform/arc‑mix)
    best_seq = None
    best_val = float("inf")

    # grid of Beta parameters (covers U‑shaped to bell‑shaped densities)
    alphas = [0.2, 0.3, 0.4, 0.5, 0.7, 1.0, 2.0, 5.0]
    for a in alphas:
        for b in alphas:
            cand = _beta_shape(n, a, b)
            val = evaluate_sequence(cand.tolist())
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Sweep] Beta({a},{b}) → {val:.6f}")

    # uniform + arcsine mixtures (arcsine = Beta(0.5,0.5))
    uniform = np.full(n, 1.0 / n)
    arcsine = _beta_shape(n, 0.5, 0.5)
    for lam in np.linspace(0.0, 1.0, 11):
        cand = (1.0 - lam) * uniform + lam * arcsine
        cand /= cand.sum()
        val = evaluate_sequence(cand.tolist())
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Sweep] mix λ={lam:.2f} → {val:.6f}")

    # consider the warm start if it is better
    if warm is not None:
        val_w = evaluate_sequence(warm.tolist())
        if val_w < best_val:
            best_val = val_w
            best_seq = warm.copy()
            print(f"[Warm] warm start → {val_w:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Adam optimisation on smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 20:
        sub_dead = min(deadline, time.time() + 120.0)  # up to 2 min for Adam
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase C – exponentiated‑gradient sub‑gradient optimisation
    if best_seq is not None and time.time() < deadline - 20:
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=600_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase D – projected gradient descent on the exact max‑convolution
    if best_seq is not None and time.time() < deadline - 15:
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase E – long simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
