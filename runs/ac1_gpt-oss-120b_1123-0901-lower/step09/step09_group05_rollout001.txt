def lp_step(f, M):
    n = len(f)
    b = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)
    conv_expr = cp.conv(f_const, b)  # length 2n-1
    constraints = [conv_expr <= M]
    obj = cp.Maximize(cp.sum(b))
    prob = cp.Problem(obj, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)   # or 'GLPK' if available

    if prob.status not in ["optimal", "optimal_inaccurate"]:
        return None

    b_opt = b.value
    return b_opt

"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

Algorithm
---------

1. **Initial pool** – generate a modest but diverse set of candidate vectors
   (uniform, sinusoid, edge‑exponential, edge‑power‑law, beta‑pdf, double‑Gaussian,
   a few Dirichlet draws, and possible warm‑start shifts).  All vectors are
   normalised to sum = 1 and the best one is kept.

2. **LP refinement** – starting from the best candidate, repeatedly solve
   a linear program

        maximise   Σ b_i
        subject to b_i ≥ 0,
                   (f ⊛ b)_k ≤ M   ∀k ,

   where `f` is the current sequence, `M = max(f⊛f)` and `⊛` denotes linear
   convolution.  The LP is solved with CVXPY (ECOS).  The optimal `b` is
   normalised to a probability vector `g = b / Σb`.  We try to replace `f`
   by `g`, otherwise we mix them with a back‑tracking line‑search on the
   mixing coefficient `t`.  Each successful step strictly reduces the
   objective, and the loop stops when no improvement can be found or the
   time budget is exhausted.

3. **Polishing** – a short projected‑gradient descent (exact max‑convolution
   gradient) followed by a cheap random‑move hill‑climber are applied to the
   current best vector to extract any remaining benefit.

All phases respect the global 1000 s wall‑clock limit.  The returned sequence
is a plain Python list compatible with the provided `evaluate_sequence`
function.
"""

import time
import numpy as np
import cvxpy as cp

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of deterministic shapes (all sum to 1) ----------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Objective evaluation --------------------------------------------------
def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact evaluation of F(a) (assumes a is non‑negative)."""
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 0.01:                     # follow the security guard in evaluate_sequence
        return np.inf
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    n = a.shape[0]
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Projected gradient descent (same as in the reference code) ---------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the true (non‑smooth) objective."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = np.inf
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val

# Simple random local search --------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very cheap hill‑climber moving random mass fractions."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Linear‑programming refinement -----------------------------------------
def _lp_step(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve  max Σ b_i   s.t.  b_i ≥ 0,  (f ⊛ b)_k ≤ M  ∀k.
    Returns the optimal b (numpy array) or None on failure.
    """
    n = f.shape[0]

    b = cp.Variable(n, nonneg=True)
    f_const = cp.Constant(f)                     # treated as a constant vector
    conv_expr = cp.conv(f_const, b)              # length = 2·n‑1

    # a tiny tolerance helps numerical stability
    constraints = [conv_expr <= M + 1e-12]

    prob = cp.Problem(cp.Maximize(cp.sum(b)), constraints)

    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000, verbose=False)
    except Exception as exc:
        print("LP solve exception:", exc)
        return None

    if prob.status not in ("optimal", "optimal_inaccurate"):
        # In some corner cases the problem can be infeasible due to numerical noise.
        return None

    return b.value

            print("[LP] trivial solution – aborting")
            break

        g = b_opt / sum_b                     # normalize to sum = 1

        # try a full replacement first
        val_g = _objective(g, conv_fft_len)
        if val_g < best_val - 1e-12:
            best_val = val_g
            best_seq = g.copy()
            print(f"[LP it={lp_iter}] full replace -> {val_g:.6f}")
            lp_iter += 1
            continue

        # otherwise mix with back‑tracking line‑search on t∈(0,1]
        t = 0.5
        improved = False
        while t > 1e-3 and time.time() < deadline - 1:
            f_new = (1.0 - t) * f + t * g
            val_new = _objective(f_new, conv_fft_len)
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_seq = f_new.copy()
                improved = True
                print(f"[LP it={lp_iter}] mix t={t:.4f} -> {val_new:.6f}")
                break
            t *= 0.5
        if not improved:
            print(f"[LP it={lp_iter}] no improvement – terminating LP phase")
            break
        lp_iter += 1

    # ------------------------------------------------------------------
    # Phase 2 – Short projected gradient descent for polishing
    if best_seq is not None and time.time() < deadline - 2:
        gd_deadline = time.time() + min(30.0, deadline - time.time() - 2.0)
        a_opt, val_opt = _gd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            gd_deadline,
            max_iters=100_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[GD] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Final cheap random‑move hill‑climber
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safeguard final output
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
