def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    fft_len = _next_pow2(2*n-1)

    def compute_val(a):
        conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
        max_c = conv.max()
        return 2.0 * n * max_c / (a.sum() ** 2)

    # initial best: uniform
    best_a = np.full(n, 1.0/n)
    best_val = compute_val(best_a)
    # Optionally evaluate via evaluate_sequence for final.
    
    # define phases
    # Phase 1: random param candidates
    while time.time() < deadline - 20: # keep some time for refinement
        # random choice of family
        family = np.random.choice(['gaussian','double','beta','raised','tri','dirichlet'])
        if family == 'gaussian':
            sigma_frac = np.random.uniform(0.01, 0.3)
            a = np.exp(-0.5 * ((np.arange(n)- (n-1)/2)/(sigma_frac*n))**2)
        elif family == 'double':
            sigma_frac = np.random.uniform(0.01,0.3)
            d_frac = np.random.uniform(0.1, 0.9)
            mu1 = (n-1)/2 - d_frac*n/2
            mu2 = (n-1)/2 + d_frac*n/2
            x = np.arange(n)
            a = np.exp(-0.5 * ((x-mu1)/(sigma_frac*n))**2) + np.exp(-0.5 * ((x-mu2)/(sigma_frac*n))**2)
        elif family == 'beta':
            alpha = np.random.uniform(0.5, 10.0)
            x = (np.arange(1,n+1)/n)
            a = (x**(alpha-1))*((1-x)**(alpha-1))
        elif family == 'raised':
            width_frac = np.random.uniform(0.2, 0.9)
            half = width_frac * n / 2.0
            mu = (n-1)/2
            x = np.arange(n)
            dist = np.abs(x - mu)
            a = np.where(dist <= half, 0.5*(1+np.cos(np.pi*dist/half)), 0.0)
        elif family == 'tri':
            a = np.minimum(np.arange(n)+1, n-np.arange(n))
        elif family == 'dirichlet':
            conc = np.random.uniform(0.05,5.0)
            a = np.random.dirichlet(np.full(n, conc))
        # some families may produce all zeros; skip
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = compute_val(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # maybe print
            print(f"[Random] New best {best_val:.6f} family {family}")
    
    # Phase 2: subgradient refinement (exponential gradient)
    # Use function similar to _subgradient_optimize_expgrad but tailored.
    # We'll run while time left.
    
    def subgrad_refine(a_init):
        a = a_init.copy()
        # ensure nonnegative, sum=1
        for i in range(2000000): # iter but break on deadline
            if time.time() >= deadline - 5:
                break
            # compute conv
            conv = np.fft.irfft(np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len)[:2*n-1]
            max_c = conv.max()
            val = 2.0 * n * max_c / (a.sum()**2)
            # update best
            nonlocal best_val, best_a
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Subgrad] new best {best_val:.6f}")
            # active set
            active = np.isclose(conv, max_c, atol=1e-12)
            if active.any():
                w = np.zeros_like(conv)
                w[active] = 1.0 / active.sum()
            else:
                w = np.zeros_like(conv)
            # compute gradient via cross-correlation
            # cross-corr w with a reversed
            rev_a = a[::-1]
            grad = 2.0 * np.fft.irfft(np.fft.rfft(w, fft_len) * np.fft.rfft(rev_a, fft_len), fft_len)[:n]
            # exponentiated gradient update

            # renorm

            if s <= 0:

                s = a.sum()
            a = a / s

def _next_pow2(x: int) -> int:

def _conv_max(a: np.ndarray, fft_len: int) -> float:
    """Return max value of the linear convolution a*a using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * a.shape[0] - 1]
    return conv.max()

    """Exact objective 2·n·max(conv) for a normalised (sum‑1) sequence."""
    return 2.0 * n * _conv_max(a, fft_len)

    d = d_frac * n
    mu1 = mu - d / 2.0
    mu2 = mu + d / 2.0
    g1 = np.exp(-0.5 * ((x - mu1) / (sigma_frac * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - mu2) / (sigma_frac * n)) ** 2)
    return g1 + g2


def _shape_beta(x, n, alpha, beta):
    t = (np.arange(1, n + 1, dtype=np.float64) / n)
    return (t ** (alpha - 1)) * ((1.0 - t) ** (beta - 1))


def _shape_monotone_exponential(x, n, gamma):
    return np.exp(-gamma * x)


def _shape_power_law(x, n, p):
    return 1.0 / np.power(x + 1.0, p)


def _shape_raised_cosine(x, n, width_frac):
    half = width_frac * n / 2.0
    mu = (n - 1) / 2.0
    dist = np.abs(x - mu)
    a = np.where(dist <= half, 0.5 * (1.0 + np.cos(np.pi * dist / half)), 0.0)
    return a


def _shape_triangular(x, n):
    return np.minimum(x + 1, n - x).astype(np.float64)


def _interleave(a):
    """Spread large entries by the “even‑odd” interleaving permutation."""
    n = a.shape[0]
    perm = (np.arange(n) * 2) % n
    b = np.empty_like(a)
    b[perm] = a
    return b


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimisation
# ----------------------------------------------------------------------
def _subgradient_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta0: float = 1.0,
) -> np.ndarray:
    """
    Optimise the exact max‑convolution objective on the simplex.
    Returns the best vector found (the caller evaluates the exact objective).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best = a.copy()
    best_val = _objective(best, n, fft_len)

    it = 0
    while time.time() < deadline:
        conv = np.fft.irfft(
            np.fft.rfft(a, fft_len) * np.fft.rfft(a, fft_len), fft_len
        )[: 2 * n - 1]
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[SubGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of max‑convolution (smooth at active set)
        rev_a = a[::-1]
        grad = 2.0 * np.fft.irfft(
            np.fft.rfft(w, fft_len) * np.fft.rfft(rev_a, fft_len), fft_len
        )[:n]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s <= 0.0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best


# ----------------------------------------------------------------------
# Permutation‑only hill‑climb (very cheap, optional)
# ----------------------------------------------------------------------
def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_swaps: int = 15000,
) -> np.ndarray:
    a = a0.copy()
    best = a.copy()
    best_val = _objective(best, n, fft_len)

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            continue
        # perform swap
        a[i], a[j] = a[j], a[i]
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best = a.copy()
        else:
            # revert with small probability to escape plateaus
            if np.random.rand() < 0.01:
                best_val = val
            else:
                a[i], a[j] = a[j], a[i]
        it += 1
    return best

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function that yields a low value of
    `evaluate_sequence`.  The routine respects the given time budget and
    always returns the best sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # tiny safety margin

    # ------------------------------------------------------------------
    # Problem size (use the warm‑start length if available)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        n = warm.shape[0]
        if n == 0:
            n = 4096
    else:
        n = 4096

    fft_len = _next_pow2(2 * n - 1)   # length for FFT‑based convolution
    x = np.arange(n, dtype=np.float64)  # shared index array

    # ------------------------------------------------------------------
    # Initialise best candidate (uniform)
    # ------------------------------------------------------------------
    best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_val = _objective(best_a, n, fft_len)
    print(f"[Init] uniform objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Warm start (if provided)
    # ------------------------------------------------------------------
    if warm is not None and warm.size == n:
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            val_warm = _objective(warm, n, fft_len)
            if val_warm < best_val:
                best_val = val_warm
                best_a = warm.copy()
                print(f"[Warm] warm start objective {val_warm:.6f}")

    # ------------------------------------------------------------------
    # Phase I – randomised pool exploration (≈ 30 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + (deadline - time.time()) * 0.30
    while time.time() < explore_deadline:
        # --------------------------------------------------------------
        # pick a random parametric family
        # --------------------------------------------------------------
        r = np.random.rand()
        if r < 0.12:                                    # centred Gaussian
            sigma = np.random.uniform(0.015, 0.25)
            a = _shape_gaussian(x, n, sigma)

        elif r < 0.24:                                  # double Gaussian
            sigma = np.random.uniform(0.015, 0.25)
            d = np.random.uniform(0.15, 0.85)
            a = _shape_double_gaussian(x, n, sigma, d)

        elif r < 0.36:                                  # symmetric Beta (α=β)
            alpha = np.random.uniform(1.2, 5.0)
            a = _shape_beta(x, n, alpha, alpha)

        elif r < 0.48:                                  # skewed Beta
            alpha = np.random.uniform(0.5, 4.5)
            beta = np.random.uniform(0.5, 4.5)
            a = _shape_beta(x, n, alpha, beta)

        elif r < 0.58:                                  # monotone exponential
            gamma = np.random.uniform(0.5, 5.0)
            a = _shape_monotone_exponential(x, n, gamma)

        elif r < 0.68:                                  # power‑law decay
            p = np.random.uniform(0.5, 3.0)
            a = _shape_power_law(x, n, p)

        elif r < 0.78:                                  # raised cosine
            width = np.random.uniform(0.25, 0.85)
            a = _shape_raised_cosine(x, n, width)

        elif r < 0.86:                                  # triangular
            a = _shape_triangular(x, n)

        else:                                            # Dirichlet draw
            conc = np.random.uniform(0.05, 4.5)
            a = np.random.dirichlet(np.full(n, conc))

        # --------------------------------------------------------------
        # occasional extra tricks
        # --------------------------------------------------------------
        # interleaved permutation (spreads large entries)
        if np.random.rand() < 0.18:
            a = _interleave(a)

        # small random jitter – keeps the search stochastic
        if np.random.rand() < 0.25:
            jitter = np.random.normal(scale=0.02, size=n)
            a = np.maximum(a + jitter, 0.0)

        # normalise to sum = 1 (scale invariance)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s

        # --------------------------------------------------------------
        # fast evaluation
        # --------------------------------------------------------------
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Random] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – intensive sub‑gradient refinement (≈ 55 % of budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        print("[Refine] Starting exponentiated‑gradient phase")
        best_a = _subgradient_expgrad(
            best_a,
            n,
            fft_len,
            deadline - 15,          # leave a few seconds for final tweaks
            eta0=0.9,
        )
        best_val = _objective(best_a, n, fft_len)
        print(f"[Refine] best after sub‑grad {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase III – cheap permutation hill‑climb (remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        print("[Permute] Final permutation hill‑climb")
        best_a = _permute_hill_climb(
            best_a,
            n,
            fft_len,
            deadline,
            max_swaps=20000,
        )
        best_val = _objective(best_a, n, fft_len)
        print(f"[Permute] final objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Return – ensure the sequence obeys evaluate_sequence's limits
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)

    # Verify using the official evaluator (optional but useful for logging)
    try:
        from __main__ import evaluate_sequence  # the evaluation function is injected
    except Exception:
        evaluate_sequence = None
    if evaluate_sequence is not None:
        verified = evaluate_sequence(best_a.tolist())
        print(f"[Verify] evaluate_sequence returns {verified:.6f}")

    print(f"Finished – best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
