"""
Propose a low‑value step‑function.

The strategy consists of four stages:

1. **Fast diverse warm‑starts** – generate thousands of candidates from
   deterministic families (uniform, sinusoid, edge‑exponential,
   power‑law, beta, double‑Gaussian, ramp, triangular) and from many
   Dirichlet draws with different concentration parameters.
   Every candidate is normalised to a probability distribution and its
   exact objective (2·n·max(conv)) is evaluated.  The best few candidates
   become seeds for local refinement.

2. **Short projected‑gradient refinements** – each seed is improved by a
   few thousand steps of Projected Gradient Descent (PGD) on the exact
   max‑convolution.  This cheap phase explores many basins quickly.

3. **Long PGD on the current global best** – the best seed is further
   optimised with a line‑search PGD that adapts its step size
   (≈ η₀/√t) and projects onto the probability simplex at each step.
   This stage consumes the bulk of the remaining time budget.

4. **Final cheap hill‑climber** – random mass‑transfer moves are applied
   for the last seconds to squeeze out any remaining improvement.

All FFT‑based convolutions use a power‑of‑two length for speed.
The algorithm works for any sequence length; we use n = 2048 (fast
yet large enough to approach the asymptotic bound).  The returned list
contains non‑negative numbers and respects the limits of ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    Returns the real part of the result (exact for real input).
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (all return a probability vector, sum = 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    return s / s.sum()


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    return shape / shape.sum()


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    return shape / s if s > 0 else np.full(n, 1.0 / n)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    return shape / s if s > 0 else np.full(n, 1.0 / n)


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    return shape / shape.sum()


def _ramp(n: int) -> np.ndarray:
    shape = np.linspace(1.0, 0.0, n, dtype=np.float64)
    return shape / shape.sum()


def _triangular(n: int) -> np.ndarray:
    # symmetric triangle peaking in the middle
    half = (n + 1) // 2
    left = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        right = left[::-1]
    else:
        right = left[:-1][::-1]
    shape = np.concatenate([left, right])
    return shape / shape.sum()


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Objective and gradient utilities
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(conv) for a probability vector."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


def _conv_and_max(a: np.ndarray, conv_fft_len: int):
    """Return (full convolution, its maximum)."""
    conv = _full_convolution(a, conv_fft_len)
    return conv, conv.max()


def _subgradient_max(a: np.ndarray, conv: np.ndarray, conv_fft_len: int,
                     corr_fft_len: int) -> np.ndarray:
    """
    Sub‑gradient of max(conv) w.r.t. a.
    The active set consists of all shifts attaining the maximum.
    """
    max_c = conv.max()
    active = np.isclose(conv, max_c, atol=1e-14)
    # uniform weight over the active shifts
    w = np.zeros_like(conv)
    if active.any():
        w[active] = 1.0 / active.sum()
    # gradient of the smooth linear functional w·conv is
    # 2 * (w ★ a)
    return 2.0 * _cross_correlation(w, a, corr_fft_len)


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------


def _pgd_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                  corr_fft_len: int, deadline: float,
                  max_iters: int = 500_000,
                  eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent on the exact max‑convolution.
    Returns the best vector encountered and its objective value.
    """
    a = np.clip(a0, 0.0, 1.0)
    a = a / a.sum()

    best_a = a.copy()
    conv, max_c = _conv_and_max(a, conv_fft_len)
    best_val = 2.0 * n * max_c

    it = 0
    while time.time() < deadline and it < max_iters:
        # gradient of the (non‑smooth) objective at the current a
        grad = _subgradient_max(a, conv, conv_fft_len, corr_fft_len)

        # step‑size schedule (decreasing like 1/√t)
        eta = eta0 / np.sqrt(it + 1)

        # simple back‑tracking line search (max 3 reductions)
        for _ in range(3):
            a_new = a - eta * grad
            a_new = _project_onto_simplex(a_new)
            conv_new, max_c_new = _conv_and_max(a_new, conv_fft_len)
            val_new = 2.0 * n * max_c_new
            if val_new <= best_val or val_new < 2.0 * n * max_c:
                # accept the move
                a, conv, max_c = a_new, conv_new, max_c_new
                if val_new < best_val:
                    best_val = val_new
                    best_a = a.copy()
                break
            else:
                eta *= 0.5  # shrink step and retry

        # occasional jitter near the end of the budget
        if it % 20_000 == 0 and time.time() > deadline - 8:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()
            conv, max_c = _conv_and_max(a, conv_fft_len)

        it += 1

    return best_a, best_val


def _short_pgd(a0: np.ndarray, n: int, conv_fft_len: int,
               corr_fft_len: int, max_iters: int = 10_000) -> tuple[np.ndarray, float]:
    """
    A short PGD run used during the exploratory phase.
    It consumes negligible time and returns the (possibly) improved vector.
    """
    deadline = time.time() + 1.0  # at most one second per short run
    return _pgd_optimize(a0, n, conv_fft_len, corr_fft_len,
                         deadline, max_iters=max_iters, eta0=0.8)


def _local_random_search(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 100_000) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber applied in the final seconds.
    Moves a random fraction of mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {best_val:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size (2048 gives a good trade‑off between speed and quality)
    # ------------------------------------------------------------------
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)     # length for a*a
    corr_fft_len = _next_pow2(3 * n - 2)     # length for (w ★ a)

    # ------------------------------------------------------------------
    # Warm‑start from a previously submitted sequence, if available
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build an extensive initial pool of candidates
    # ------------------------------------------------------------------
    pool = []

    # deterministic families (including a few extra shapes)
    if warm is not None:
        pool.append(warm.copy())
    pool.extend([
        _uniform(n),
        _sinusoid(n),
        _ramp(n),
        _triangular(n),
    ])

    for gamma in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        pool.append(_edge_exponential(n, gamma))

    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.04, 0.06, 0.08, 0.12, 0.2):
        pool.append(_double_gaussian(n, sigma))

    # a large batch of random Dirichlet draws with diverse concentrations
    alphas = (0.05, 0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0)
    for alpha in alphas:
        for _ in range(30):          # 30 draws per alpha → 240 candidates
            pool.append(_random_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    # Quick scan of the pool – keep the best K seeds
    # ------------------------------------------------------------------
    best_val = float("inf")
    best_seq = None
    top_seeds = []        # list of (value, vector)
    K = 8                 # we will keep the best K for deeper refinement

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init {idx:04d}] new global best {val:.6f}")

        # maintain a small heap of the best K seeds
        if len(top_seeds) < K:
            top_seeds.append((val, a.copy()))
        else:
            # replace the worst if current is better
            worst_val, _ = max(top_seeds, key=lambda t: t[0])
            if val < worst_val:
                # remove the worst and insert current
                worst_idx = max(range(len(top_seeds)),
                               key=lambda i: top_seeds[i][0])
                top_seeds.pop(worst_idx)
                top_seeds.append((val, a.copy()))

    # sort the seeds by quality (best first)
    top_seeds.sort(key=lambda t: t[0])

    # ------------------------------------------------------------------
    # Phase 1 – many short PGD runs on random seeds (exploration)
    # ------------------------------------------------------------------
    while time.time() < deadline - 30:
        # draw a fresh random Dirichlet vector
        a0 = _random_dirichlet(n, np.exp(rng.uniform(-3, 1)))
        a_opt, val_opt = _short_pgd(a0, n, conv_fft_len, corr_fft_len,
                                    max_iters=5_000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Exploratory] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – longer PGD refinements on the top K seeds
    # ------------------------------------------------------------------
    for seed_idx, (seed_val, seed_vec) in enumerate(top_seeds):
        remaining = deadline - time.time()
        # allocate a modest slice of the remaining budget for each seed
        if remaining < 5:
            break
        slice_deadline = time.time() + min(remaining * 0.15, 12.0)

        a_ref, val_ref = _pgd_optimize(seed_vec, n, conv_fft_len,
                                       corr_fft_len, slice_deadline,
                                       max_iters=300_000, eta0=0.9)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine seed {seed_idx}] new best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – intensive PGD on the current global best
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 8:
        a_final, val_final = _pgd_optimize(best_seq, n, conv_fft_len,
                                           corr_fft_len, deadline,
                                           max_iters=1_500_000, eta0=0.9)
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Final PGD] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap local hill‑climber in the last few seconds
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(best_seq, n, conv_fft_len,
                                              deadline, max_moves=150_000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback (should never be needed)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} with length {len(best_seq)}")
    return best_seq.tolist()
