def project_simplex(v):
    """Euclidean projection of v onto simplex {x: x>=0, sum(x)=1}."""
    # Works for v any real numbers.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(len(v)) + 1) > 0)[0]
    if len(rho) == 0:
        # all values are zero after projection
        return np.full_like(v, 1.0 / len(v))
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function optimizer (search for low evaluation score).

Algorithm
---------
1. Broad parametric sweep:
   - Generate many symmetric / monotone shapes (power‑law,
     exponential, stretched‑exponential, beta, double‑Gaussian,
     sine, cosine, triangular, quadratic, left‑exponential,
     left‑power).
   - Normalise each candidate to sum = 1 and evaluate the exact
     objective using an FFT‑based convolution.
   - Keep the best candidate and remember the parameters that
     produced it for later refinement.

2. Local refinement of the most promising families:
   - For 1‑D families (power, exponential, sine, cosine,
     double‑Gaussian) run a bounded scalar optimisation
     (SciPy’s minimise_scalar if available) in a neighbourhood
     of the best parameter.
   - For 2‑D families (stretched‑exponential, beta) perform a
     fine grid search around the best parameters.
   - Update the incumbent if a better value is found.

3. Sub‑gradient descent on the exact maximal‐convolution objective.
   - The sub‑gradient is the convex combination of the gradients of
     the active shifts (those attaining the current maximum).
   - Perform projected gradient descent on the probability simplex
     (Euclidean projection).  The step size follows ηₜ = η₀/√(t+1).
   - Several short restarts are executed: the incumbent is slightly
     perturbed by Gaussian noise and a fresh descent run is started.
   - The best sequence discovered over all restarts is retained.

4. Final sanity check with the public ``evaluate_sequence`` routine
   and return the best sequence as a plain list of floats.

The routine respects the given time budget (default 1000 s) and always
returns a valid non‑negative list.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ------------------------------------------------------------
# FFT utilities
# ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for sub‑gradient.
    Returns a vector of length len(a) = n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ------------------------------------------------------------
# Objective (exact) – no clipping, normalise to sum=1 internally.
# ------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (scale‑invariant)."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    a_norm = a / s
    conv = _full_convolution(a_norm, _next_pow2(2 * n - 1))
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ------------------------------------------------------------
# Simplex projection (Euclidean)
# ------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto the probability simplex {x≥0,∑x=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u - (cssv - 1) / (np.arange(n) + 1) > 0)[0]
    if rho.size == 0:
        # all zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ------------------------------------------------------------
# Candidate generators (all families are normalised to sum=1)
# ------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1].  a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sinusoidal shape a_i ∝ sin(π·(i+0.5)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_cosine(power: float, n: int) -> np.ndarray:
    """Half‑cosine shape a_i ∝ (1+cos(π·i/(n‑1)))^power."""
    i = np.arange(n, dtype=np.float64)
    a = (1.0 + np.cos(np.pi * i / (n - 1))) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_triangle(n: int) -> np.ndarray:
    """Triangular shape a_i ∝ min(i+1, n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_quadratic(power: float, n: int) -> np.ndarray:
    """Quadratic‑like shape a_i ∝ ((i+1)*(n‑i))^power."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ------------------------------------------------------------
# Sub‑gradient optimiser on the exact max‑convolution (projected)
# ------------------------------------------------------------
def _pgd_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.5,
) -> Tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    Returns the best vector seen and its objective.
    """
    a = np.copy(a0)
    # ensure simplex
    a = _project_simplex(a)

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # linear convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)   # sum(a)=1 after projection

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional logging
            if it % 200_000 == 0:
                print(f"[PGD it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max‑conv
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule
        eta = eta0 / math.sqrt(it + 1.0)

        # projected gradient step (Euclidean projection onto simplex)
        a_tmp = a - eta * grad
        a = _project_simplex(a_tmp)

        it += 1

    return best_a, best_val

                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-4},
            )
            if res.success:
                a = _candidate_exponential(math.exp(res.x), n)
                _update_if_better(a, "exp", math.exp(res.x))

        # ---- double‑Gaussian refinement -------------------------------
        if "double_gauss" in best_params and minimize_scalar is not None:
            sig0 = best_params["double_gauss"]
            log_sig0 = math.log(sig0)
            lo, hi = log_sig0 - 2.0, log_sig0 + 2.0
            res = minimize_scalar(
                lambda t: _objective(_candidate_double_gaussian(math.exp(t), n), n),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-4},
            )
            if res.success:
                a = _candidate_double_gaussian(math.exp(res.x), n)
                _update_if_better(a, "double_gauss", math.exp(res.x))

        # ---- sine refinement -----------------------------------------
        if "sine" in best_params and minimize_scalar is not None:
            pow0 = best_params["sine"]
            lo, hi = max(0.1, pow0 - 2.0), pow0 + 2.0
            res = minimize_scalar(
                lambda p: _objective(_candidate_sine(p, n), n),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-4},
            )
            if res.success:
                a = _candidate_sine(res.x, n)
                _update_if_better(a, "sine", res.x)

        # ---- cosine refinement ---------------------------------------
        if "cosine" in best_params and minimize_scalar is not None:
            pow0 = best_params["cosine"]
            lo, hi = max(0.1, pow0 - 2.0), pow0 + 2.0
            res = minimize_scalar(
                lambda p: _objective(_candidate_cosine(p, n), n),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-4},
            )
            if res.success:
                a = _candidate_cosine(res.x, n)
                _update_if_better(a, "cosine", res.x)

        # ---- stretched exponential grid refinement -------------------
        if "stretched" in best_params:
            a0, b0 = best_params["stretched"]
            # create modest log‑spaced neighbourhoods
            alphas = np.logspace(
                math.log10(max(1e-6, a0 / 3.0)), math.log10(a0 * 3.0), 9
            )
            betas = np.linspace(max(0.1, b0 - 1.5), b0 + 1.5, 9)
            for alpha in alphas:
                for beta in betas:
                    a = _candidate_stretched(alpha, beta, n)
                    _update_if_better(a, "stretched", (alpha, beta))

        # ---- beta distribution grid refinement -----------------------
        if "beta" in best_params:
            a0, b0 = best_params["beta"]
            a_vals = np.linspace(max(0.1, a0 * 0.5), a0 * 1.5, 9)
            b_vals = np.linspace(max(0.1, b0 * 0.5), b0 * 1.5, 9)
            for a_par in a_vals:
                for b_par in b_vals:
                    a = _candidate_beta(a_par, b_par, n)
                    _update_if_better(a, "beta", (a_par, b_par))

        # ---- quadratic power refinement -------------------------------
        if "quadratic" in best_params and minimize_scalar is not None:
            p0 = best_params["quadratic"]
            lo, hi = max(0.01, p0 - 2.0), p0 + 2.0
            res = minimize_scalar(
                lambda p: _objective(_candidate_quadratic(p, n), n),
                bounds=(lo, hi),
                method="bounded",
                options={"xatol": 1e-4},
            )
            if res.success:
                a = _candidate_quadratic(res.x, n)
                _update_if_better(a, "quadratic", res.x)

        print(f"[Refine] best after local refinement: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (multiple short restarts)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase‑PGD] start projected sub‑gradient optimisation")
        # initial run from the current best
        sub_deadline = min(deadline, time.time() + 30.0)  # at most 30 s for this run
        a_sub, val_sub = _pgd_subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=5_000_000,
            eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑PGD] improved to {val_sub:.6f}")

        # additional random‑restart runs while time permits
        while time.time() < deadline - 2.0:
            # small perturbation of the incumbent
            noise = np.random.normal(scale=0.005, size=n)
            start = np.maximum(best_seq + noise, 0.0)
            if start.sum() < 1e-12:
                start = np.random.dirichlet(np.ones(n))
            else:
                start = start / start.sum()

            sub_deadline = min(deadline, time.time() + 15.0)  # 15 s per restart
            a_sub, val_sub = _pgd_subgradient_optimize(
                start,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=3_000_000,
                eta0=0.6,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase‑PGD‑restart] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity check and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator's admissible range (0 … 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final logging with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        # If evaluate_sequence is not importable in the test harness we ignore.
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
