"""
Propose a low‑score non‑negative step function.

Algorithm (≈1000 s):
1. **Random‑parametric sweep** – sample many shapes from simple families,
   keep the best (fast FFT convolution).
2. **Linear‑program improvement** – for the current best `a` solve
   ``max Σ a_i g_i``  subject to ``(a*g)_k ≤ M`` (M = max (a∗a)).
   The LP returns a vector `g` with total mass S>1; after normalising
   ``a←g/S`` the maximal convolution drops to ``M/S²``.  Iterate while
   this yields an improvement.
3. **Gradient polishing** – a few passes of Adam, sub‑gradient and
   projected‑gradient descent on the exact max‑convolution objective.
4. **Cheap hill‑climbing** – final stochastic mass‑transfer moves.

All phases respect the global deadline; the best sequence found is
returned (clipped to the limits of ``evaluate_sequence``) and stored in
the global variable ``height_sequence_1`` for warm‑starts in later runs.
"""

import time
import numpy as np
from scipy.linalg import toeplitz
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Fast objective (identical to evaluate_sequence up to clipping)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (i / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    return s / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    return shape / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _cosine_shape(n: int, power: float = 1.0) -> np.ndarray:
    """A smooth bell‑shaped cosine (raised to `power`)."""
    x = np.arange(n, dtype=np.float64) / (n - 1)
    s = (np.cos(np.pi * (x - 0.5))) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    return s / ss if ss > 0 else np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int, corr_fft_len: int,
                   deadline: float, max_iters: int = 300_000,
                   lr_start: float = 0.04) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float,
                          max_iters: int = 300_000, eta0: float = 0.8) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                  corr_fft_len: int, deadline: float,
                  max_iters: int = 300_000, eta0: float = 0.4) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑program helper --------------------------------------------------
def _conv_matrix(a: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix with entries (a∗g)_k = Σ_i a_i g_{k‑i}
    written as A·g, i.e. each column j contains `a` shifted down by j.
    Dense because the LP expects a dense matrix.
    """
    n = a.shape[0]
    c = np.concatenate([a, np.zeros(n - 1, dtype=np.float64)])
    r = np.concatenate([np.array([a[0]], dtype=np.float64), np.zeros(n - 1, dtype=np.float64)])
    return toeplitz(c, r)  # shape (2n‑1, n)


def _lp_expand(a: np.ndarray, M: float, deadline: float) -> tuple[np.ndarray, bool]:
    """
    Solve the LP:
        maximise   Σ g_i
        subject to (a∗g)_k ≤ M   for all k
                    g_i ≥ 0
    If the optimum mass S exceeds 1, return the normalised vector g / S.
    Returns (new_a, improved_flag).
    """
    n = a.shape[0]
    # Build constraint matrix once (dense)
    A = _conv_matrix(a)
    b = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)          # maximise Σ g_i  → minimise –Σ g_i
    bounds = [(0, None)] * n

    # linprog may take a few seconds – guard against overrunning the deadline
    time_left = deadline - time.time()
    if time_left < 1.0:
        return a, False

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True, 'time_limit': time_left})
    if not res.success:
        print("[LP] solver failed:", res.message)
        return a, False

    g = res.x
    S = g.sum()
    if S <= 1.0 + 1e-8:
        return a, False

    new_a = g / S
    return new_a, True

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a step‑function that minimises evaluate_sequence()."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start ---------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – diverse parametric sampling
    best_seq = None
    best_val = float("inf")
    families = ["edge_exp", "edge_pow", "beta", "double_gauss",
                "sin_pow", "tent", "cosine", "dirichlet"]
    phase1_deadline = min(deadline - 5.0,
                          time.time() + 0.2 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 2)
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-10.0, -0.1)
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-1, 1)
            b = 10 ** rng.uniform(-1, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.4)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 5.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "cosine":
            power = rng.uniform(0.5, 4.0)
            cand = _cosine_shape(n, power)
        else:   # dirichlet
            alpha = rng.uniform(0.05, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # Include warm‑start rotations as extra candidates
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – LP‑based expansion (mass‑increase under current max)
    if best_seq is not None and time.time() < deadline - 30:
        print("Phase 2 – linear‑program expansion")
        while time.time() < deadline - 20:
            conv = _full_convolution(best_seq, conv_fft_len)
            M = conv.max()
            new_a, improved = _lp_expand(best_seq, M, deadline)
            if not improved:
                break
            # recompute objective for the new vector
            new_val = _objective(new_a, conv_fft_len)
            if new_val < best_val:
                best_val, best_seq = new_val, new_a.copy()
                print(f"[LP] improved to {new_val:.6f}")
            else:
                # even if the objective didn't improve (numerical), accept the vector
                best_seq = new_a.copy()
                best_val = new_val
            # continue with the updated sequence

    # ------------------------------------------------------------------
    # Phase 3 – deep gradient refinement (Adam → sub‑gradient → PGD)
    if best_seq is not None and time.time() < deadline - 12:
        print("Phase 3 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=250_000, lr_start=0.05)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 3] Adam improved to {val_adam:.6f}")

    if best_seq is not None and time.time() < deadline - 8:
        print("Phase 3 – Sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=200_000, eta0=0.85)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase 3] Sub‑grad improved to {val_sub:.6f}")

    if best_seq is not None and time.time() < deadline - 4:
        print("Phase 3 – Projected‑gradient optimisation")
        a_pgd, val_pgd = _pgd_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=150_000, eta0=0.35)
        if val_pgd < best_val:
            best_val, best_seq = val_pgd, a_pgd.copy()
            print(f"[Phase 3] PGD improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap stochastic hill‑climbing for the final seconds
    if best_seq is not None and time.time() < deadline - 1:
        print("Phase 4 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 4] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely) – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence's limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # store for future runs
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
