def _fast_objective(a, n, conv_fft_len):
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[:2*n-1]
    max_c = conv.max()
    return 2.0 * n * max_c

    for p in np.linspace(0.2, 0.8, 7):

"""
Hybrid optimizer for the step‑function objective

F(a) = 2·n·max(a∗a) / (∑a)²  (the sum cancels – we may enforce ∑a=1)

The algorithm proceeds in five stages:

1.  **Rich parametric seeding** – a large pool of analytically motivated
    shapes (power‑law, exponential, beta‑type, sinusoidal, uniform,
    random Dirichlet, sparse spikes) is generated and the best one (according
    to a fast FFT‑based objective) is kept.  This gives a far better starting
    point than a plain random vector.

2.  **Few‑step Adam on a smooth‑max surrogate** – a short Adam run on the
    log‑sum‑exp surrogate quickly removes obvious “spikes”.

3.  **Long exponentiated‑gradient sub‑gradient descent** – exact‑max
    sub‑gradients are used while staying on the simplex; this phase drives the
    value down close to a stationary point.

4.  **Targeted pair‑wise simulated annealing** – the index `k` where the
    convolution is maximal is located; the pair `(i,j)` that contributes most
    to this peak is softened and a small amount of mass is moved elsewhere.
    A temperature schedule permits occasional uphill moves.

5.  **Final random annealing + optional up‑sampling** – a light simulated
    annealing sweep cleans the remaining “roughness”.  If time permits the
    whole solution is interpolated to a finer grid (n=8192) and a short
    refinement is performed there as well.

All phases respect the global time budget (`budget_s` seconds) and the
best vector seen is always returned as a plain Python list of non‑negative
floats (clipped to the limits accepted by `evaluate_sequence`).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate log‑sum‑exp( λ·(a∗a) )."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.08
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Classical simulated‑annealing with random mass transfers."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Targeted annealing that attacks the most offending pair."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _fast_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Fast evaluation of 2·n·max(conv) (∑a is 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _parametric_candidates(n: int) -> list[np.ndarray]:
    """Create a rich pool of analytically‑motivated starting points."""
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Power‑law a_i ∝ (i+1)^‑p  (p≈0.5 is the classic smoothing shape)
    for p in np.linspace(0.2, 0.8, 7):
        a = (np.arange(1, n + 1, dtype=np.float64) ** (-p))
        a = a / a.sum()
        cand.append(a)

        # symmetric version (mirror around centre)
        a_sym = np.concatenate([a[: n // 2], a[: n // 2][::-1]])
        if a_sym.shape[0] == n:
            cand.append(a_sym / a_sym.sum())

    # Exponential decay (both one‑sided and symmetric)
    for lam in (0.001, 0.003, 0.005, 0.01, 0.02, 0.05):
        a = np.exp(-lam * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        cand.append(a)

        a_sym = a + np.exp(-lam * np.arange(n - 1, -1, -1, dtype=np.float64))
        cand.append(a_sym / a_sym.sum())

    # Beta‑type: a_i ∝ (i+1)^α (n-i)^α  (α<0 concentrates mass at ends)
    for alpha in np.linspace(-0.9, 2.0, 7):
        i = np.arange(1, n + 1, dtype=np.float64)
        shape = (i ** alpha) * ((n - i + 1) ** alpha)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # Sinusoidal powers (half‑wave)
    for gamma in (0.5, 1.0, 2.0, 3.0):
        i = np.arange(n, dtype=np.float64)
        shape = np.sin(np.pi * (i + 0.5) / n) ** gamma
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            cand.append(shape)

    # Random Dirichlet draws with many concentration levels
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Sparse spikes
    cand.extend(_spike_candidates(n))

    # Mixtures with uniform (helps to move away from extreme corners)
    uniform = np.full(n, 1.0 / n)
    for shape in list(cand):
        for w in (0.2, 0.5, 0.8):
            mix = (1 - w) * uniform + w * shape
            mix = mix / mix.sum()
            cand.append(mix)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function that minimizes the evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem size – start with a moderate length that still executes many iterations
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previously‑found solution (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 0 – massive parametric random search (≈15 % of the budget)
    phase0_deadline = time.time() + (deadline - time.time()) * 0.15
    candidate_pool = _parametric_candidates(n)
    if warm is not None:
        candidate_pool.append(warm.copy())

    best_a = None
    best_val = float("inf")
    evals = 0

    while candidate_pool and time.time() < phase0_deadline:
        a0 = candidate_pool.pop(np.random.randint(len(candidate_pool)))
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        val = _fast_objective(a0, n, conv_fft_len)
        evals += 1
        if val < best_val:
            best_val = val
            best_a = a0.copy()
            print(f"[Phase0] new best {val:.6f} after {evals} evals")

    # If parametric stage produced nothing (unlikely) fall back to a random Dirichlet
    if best_a is None:
        best_a = np.random.dirichlet(np.ones(n))
        best_val = _fast_objective(best_a, n, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase I – short Adam run to quickly lower the loss
    phaseI_dead = min(deadline, time.time() + 5.0)   # 5 s (adjustable)
    a_adam, val_adam = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, phaseI_dead, max_iters=120_000
    )
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient sub‑gradient descent (≈45 % of remaining time)
    if time.time() < deadline - 30:
        print("[Phase II] Long sub‑gradient descent")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Phase II] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – targeted pair‑wise annealing (≈25 % of remaining time)
    if time.time() < deadline - 15:
        print("[Phase III] Pair‑wise targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_a,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_pair < best_val:
            best_val, best_a = val_pair, a_pair.copy()
            print(f"[Phase III] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – light random annealing with whatever time is left
    if time.time() < deadline - 2:
        print("[Phase IV] Final random annealing")
        a_last, val_last = _anneal_optimize(
            best_a,
            n,
            conv_fft_len,
            deadline,
            max_moves=40_000,
        )
        if val_last < best_val:
            best_val, best_a = val_last, a_last.copy()
            print(f"[Phase IV] final best {val_last:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to n=8192 if we still have a comfortable margin
    if time.time() + 30 < deadline:
        print("[Upsample] Trying resolution 8192")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_a)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        sub_dead_big = min(deadline, time.time() + 6.0)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=150_000,
        )

        # longer sub‑gradient refinement on the big grid
        a_big_ref, val_big_ref = _subgradient_optimize_expgrad(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=120_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val, best_a = val_big_ref, a_big_ref.copy()
            print(f"[Upsample] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a valid return
    if best_a is None:
        best_a = np.full(4096, 1.0 / 4096, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)

    print(f"Finished. Best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
