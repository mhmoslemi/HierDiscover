def _fista_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    y = a.copy()
    a_prev = a.copy()
    t = 1.0
    best_a = a.copy()
    best_val = float('inf')
    itr = 0
    init_step = 0.5
    step = init_step

    while time.time() < deadline and itr < max_iters:
        conv = _full_convolution(y, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)   # a.sum()=1
        # record best
        if val < best_val:
            best_val = val
            best_a = y.copy()
            # optionally print
            if itr % 10000 == 0:
                print(f"[FISTA it={itr:06d}] new best {val:.6f}")

        # active weight
        active_mask = np.isclose(conv, max_c, atol=1e-12)
        weight = active_mask.astype(float)
        sumw = weight.sum()
        if sumw > 0:
            weight /= sumw
        else:
            weight = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(weight, y, corr_fft_len)

        # backtracking line search
        step = init_step
        while step > 1e-12:
            a_new = y - step * grad
            a_new = _project_onto_simplex(a_new)
            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum()**2)
            if val_new <= val:
                break
            step *= 0.5

        a_prev = a
        a = a_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0*t*t)) / 2.0
        y = a + ((t-1)/t_new)*(a - a_prev)
        t = t_new
        itr += 1

    # final best evaluation
    return best_a, best_val
```

def _initial_pool(n, rng, warm=None):
    pool = []

    # Dirichlet draws
    for alpha in (0.05,0.1,0.2,0.5,1.0,2.0,5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge exponential
    for gamma in (0.1,0.5,1.0,2.0,5.0,10.0):
        pool.append(_edge_exponential(n,gamma))

    # Edge power law
    for alpha in (-3.0,-2.0,-1.0,-0.5):
        pool.append(_edge_power_law(n,alpha))

    # Beta shapes
    for a,b in [(0.5,0.5),(0.5,2.0),(2.0,0.5),(2.0,2.0),(5.0,1.0),(1.0,5.0)]:
        pool.append(_beta_shape(n,a,b))

    # Double gaussian
    for sigma in (0.05,0.1,0.2,0.4):
        pool.append(_double_gaussian(n,sigma))

    # Sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # Warm start and shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    return pool
```

def _prepare_warm(warm, n):
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s >= 0.01:
        warm = warm / s
    else:
        warm = None
    return warm
```

def propose_candidate(seed=0, budget_s=1000):

    rng = np.random.default_rng(seed)
    start_time = time.time()

    # parameters
    n_big = 4096
    n_small = 1024

    # precompute FFT lengths

    # retrieve warm start if any

"""
Propose a non‑negative step function that (approximately) minimises

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm works in three phases and uses the full objective (no smooth
surrogate).

1. **Coarse search (n_small ≈ 1024).**
   A diverse pool of deterministic families and Dirichlet draws is built,
   evaluated, and the best few are refined with a fast projected‑gradient
   (FISTA) optimiser.  The best coarse vector is up‑sampled to the target
   size.

2. **Fine optimisation (n_big = 4096).**
   The up‑sampled vector, a possible warm‑start supplied by the judge and a
   second pool of candidates are evaluated.  The elite vectors are refined
   with the same accelerated optimiser; the top‑ranked candidate receives
   the largest share of the remaining budget.

3. **Final polishing.**
   A cheap hill‑climber (random mass transfers) runs until the time limit
   is reached.

All steps respect the 1000 s wall‑clock budget, print progress messages,
and finally return the best sequence as a plain Python list (compatible
with the provided `evaluate_sequence` function).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Shape families (all return vectors with sum == 1) --------------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration ‘alpha’."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Helper for warm‑starts ------------------------------------------------

def _prepare_warm(warm, n: int) -> np.ndarray | None:
    """Clip, pad/truncate and normalise a warm start to length n."""
    if warm is None:
        return None
    warm = np.asarray(warm, dtype=np.float64)
    if warm.shape[0] > n:
        warm = warm[:n]
    elif warm.shape[0] < n:
        warm = np.pad(warm, (0, n - warm.shape[0]))
    warm = np.maximum(warm, 0.0)
    s = warm.sum()
    if s >= 0.01:
        return warm / s
    return None


def _upsample(a_small: np.ndarray, n_big: int) -> np.ndarray:
    """Linear interpolation from length len(a_small) to n_big."""
    x_small = np.linspace(0.0, 1.0, len(a_small), endpoint=False)
    x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
    return np.interp(x_big, x_small, a_small)


# ----------------------------------------------------------------------
# Pool generation -------------------------------------------------------

def _initial_pool(n: int, rng: np.random.Generator, warm: np.ndarray | None = None) -> list[np.ndarray]:
    """Create a diverse seed pool (deterministic families + Dirichlet draws)."""
    pool = []

    # Dirichlet draws – several concentration levels
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (coarse γ‑grid)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law (negative α gives a “U” shape)
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf – symmetric & skewed
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian (peaks at both ends)
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # Warm start and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    return pool


# ----------------------------------------------------------------------
# Objective / gradient helpers -------------------------------------------

def _objective_val(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a∗a)/(sum(a)^2)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max() / (a.sum() ** 2)


def _fista_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Accelerated projected gradient (FISTA) on the exact objective.
    Returns the best vector seen and its value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    y = a.copy()          # extrapolated point
    a_prev = a.copy()
    t = 1.0                # momentum coefficient

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        # evaluate at extrapolated point y
        conv = _full_convolution(y, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (y.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = y.copy()
            if it % 5000 == 0:
                print(f"[FISTA it={it:06d}] new best {val:.6f}")

        # active set weight (sub‑gradient of the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        weight = active.astype(float)
        wsum = weight.sum()
        if wsum > 0:
            weight /= wsum
        else:
            weight = np.zeros_like(conv)

        # gradient of max‑convolution (up to a constant factor)
        grad = 2.0 * _cross_correlation(weight, y, corr_fft_len)

        # back‑tracking line‑search (projected)
        step = init_step
        while True:
            a_candidate = y - step * grad
            a_candidate = _project_onto_simplex(a_candidate)

            conv_cand = _full_convolution(a_candidate, conv_fft_len)
            max_c_cand = conv_cand.max()
            val_cand = 2.0 * n * max_c_cand / (a_candidate.sum() ** 2)

            if val_cand <= val or step < 1e-12:
                a_new = a_candidate
                break
            step *= 0.5

        # FISTA extrapolation
        a_prev, a = a, a_new
        t_new = (1.0 + np.sqrt(1.0 + 4.0 * t * t)) / 2.0
        y = a + ((t - 1.0) / t_new) * (a - a_prev)
        t = t_new
        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: transfer a random fraction of mass from one
    position to another and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val - 1e-12:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for the step‑function that minimises the given evaluation function.
    Returns the best sequence found within the allocated time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5            # small safety margin

    # ------------------------------------------------------------------
    # Problem sizes
    n_big = 4096
    n_small = 1024

    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    conv_fft_big = _next_pow2(2 * n_big - 1)
    corr_fft_big = _next_pow2(3 * n_big - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if the judge supplied one)
    warm_big = globals().get("height_sequence_1")
    warm_big = _prepare_warm(warm_big, n_big)
    warm_small = None
    if warm_big is not None:
        warm_small = _upsample(warm_big, n_small)
        warm_small = _prepare_warm(warm_small, n_small)

    # ------------------------------------------------------------------
    # Phase 1 – coarse optimisation on a smaller vector
    pool_small = _initial_pool(n_small, rng, warm_small)

    # Quick evaluation of the pool
    scored_small = []
    for cand in pool_small:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _objective_val(a, n_small, conv_fft_small)
        scored_small.append((val, a))
    scored_small.sort(key=lambda x: x[0])
    top_small = [a for _, a in scored_small[:10]]

    # Refine the best few with accelerated GD
    best_small = None
    best_val_small = float("inf")
    # allocate ~20 % of the total budget for this stage
    phase1_deadline = start_time + budget_s * 0.20
    for a0 in top_small:
        if time.time() >= phase1_deadline:
            break
        a_opt, val_opt = _fista_optimize(
            a0,
            n_small,
            conv_fft_small,
            corr_fft_small,
            min(phase1_deadline, deadline),
            max_iters=80_000,
        )
        if val_opt < best_val_small:
            best_val_small = val_opt
            best_small = a_opt.copy()
            print(f"[Coarse refine] new best {val_opt:.6f}")

    if best_small is None:
        # fallback
        best_small = top_small[0].copy()

    # ------------------------------------------------------------------
    # Upsample to the target size
    best_big = _upsample(best_small, n_big)
    best_big = np.maximum(best_big, 0.0)
    sb = best_big.sum()
    if sb > 0.0:
        best_big = best_big / sb

    # ------------------------------------------------------------------
    # Phase 2 – fine optimisation on the full size
    # Assemble a richer pool (deterministic families + warm start + perturbations)
    pool_big = [best_big]
    if warm_big is not None:
        pool_big.append(warm_big)

    # deterministic families (no warm‑start here)
    pool_big.extend(_initial_pool(n_big, rng, None))

    # a few random perturbations of the current best
    for _ in range(5):
        noise = rng.normal(scale=0.01, size=n_big)
        pert = best_big + noise
        pert = np.maximum(pert, 0.0)
        sp = pert.sum()
        if sp > 1e-12:
            pert = pert / sp
            pool_big.append(pert)

    # Quick evaluation of the big pool
    scored_big = []
    for cand in pool_big:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _objective_val(a, n_big, conv_fft_big)
        scored_big.append((val, a))
    scored_big.sort(key=lambda x: x[0])
    top_k_big = 15
    top_big = [a for _, a in scored_big[:top_k_big]]

    best_seq = None
    best_val = float("inf")

    # Remaining time after phase 1
    remaining_time = deadline - time.time()
    # triangular weighting: better candidates get larger time shares
    total_weight = sum(range(top_k_big, 0, -1))

    for rank, a0 in enumerate(top_big):
        now = time.time()
        if now >= deadline - 2:
            break
        weight = top_k_big - rank
        share = remaining_time * weight / total_weight
        local_deadline = now + share

        a_opt, val_opt = _fista_optimize(
            a0,
            n_big,
            conv_fft_big,
            corr_fft_big,
            local_deadline,
            max_iters=300_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Fine refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap local moves
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_big,
            conv_fft_big,
            deadline,
            max_moves=400_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] new best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    # Clip to the limits imposed by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
