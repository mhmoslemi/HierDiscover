"""
Propose a low‑evaluation‑score step function.

The search proceeds in three stages:

1. **Extensive 1‑D family sweep** – we generate many one‑sided
   exponential and power‑law sequences (a_i ∝ exp(‑γ i) or a_i ∝ (i+1)⁻ᵖ),
   normalise them to unit L¹ norm and evaluate every cyclic shift
   (0, n/4, n/2, 3n/4).  The best few candidates are kept.

2. **Fine‑grained Adam optimisation** – the best candidate from stage 1
   is refined with Adam on the smooth‑max surrogate (log‑sum‑exp) while
   staying on the probability simplex (logits → soft‑max).  The optimiser
   runs until the time budget is almost exhausted.

3. **Cheap local hill‑climber** – random mass transfers between indices are
   performed for the last few seconds; only moves that improve the true
   objective are accepted.

All helper routines are defined at top‑level (no closures, no lambdas) and
the routine respects the 1000 s deadline, always returning the best
sequence discovered.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Basic utilities -------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple families -------------------------------------------------------
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i), one‑sided (no mirroring)."""
    a = np.exp(-gamma * np.arange(n, dtype=np.float64))
    a /= a.sum()
    return a


def _left_power_law(n: int, p: float) -> np.ndarray:
    """a_i ∝ (i+1)^(-p), one‑sided."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    a = idx ** (-p)
    a /= a.sum()
    return a


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """symmetrised exponential (mass on both edges)."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # inverse temperature (smooth‑max weight)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (exact)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – grow slowly
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # occasional learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Smooth‑max objective (log‑sum‑exp) and its gradient for L‑BFGS."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c          # ≤ 0
    exp_alpha_v = np.exp(alpha * v)
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    w = exp_alpha_v / sum_exp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Continuation L‑BFGS on increasing temperatures α."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 2:
            break
        # minimise (obj,grad) with extra args – no closure needed
        args = (n, conv_fft_len, corr_fft_len, alpha)
        res = minimize(_l_bfgs_obj_grad, logits, args=args,
                       method="L-BFGS-B", jac=True,
                       options={"maxiter": 150, "disp": False})
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber moving a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation score.
    The routine respects a time budget (≈1000 s) and always returns the best
    sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # length of the sequence – keep the same as the reference implementation
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Helper to compute the (exact) objective quickly
    def _objective(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec, conv_fft_len)
        max_c = conv.max()
        s = a_vec.sum()
        if s < 1e-12:
            return np.inf
        return 2.0 * n * max_c / (s * s)

    # ------------------------------------------------------------------
    # Build a diverse pool of starters
    pool: List[np.ndarray] = []

    # Warm start and its shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Deterministic families already used in the reference code
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    pool.append(_raised_cosine(n))
    pool.append(_edge_exponential(n, 0.1))
    pool.append(_edge_power_law(n, -2.0))

    # ------------------------------------------------------------------
    # New 1‑D families: left‑hand‑only exponential & power‑law with many params
    gamma_vals = np.logspace(-5, -1, 120)          # 1e‑5 … 0.1
    shifts = (0, n // 4, n // 2, 3 * n // 4)
    for gamma in gamma_vals:
        base = _left_exponential(n, gamma)
        for shift in shifts:
            pool.append(np.roll(base, shift))

    p_vals = np.linspace(0.2, 8.0, 80)           # power‑law exponent
    for p in p_vals:
        base = _left_power_law(n, p)
        for shift in shifts:
            pool.append(np.roll(base, shift))

    # Random Dirichlet vectors (wide range of concentration)
    for alpha in np.logspace(-2, 0, 30):
        pool.append(_random_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    # Quick scan of all candidates – keep the top few
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    top_k = 20
    top_candidates: List[tuple[float, np.ndarray]] = []

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] new best {val:.6f}")

        # keep a small sorted list of the best candidates
        if len(top_candidates) < top_k:
            top_candidates.append((val, a.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif val < top_candidates[-1][0]:
            top_candidates[-1] = (val, a.copy())
            top_candidates.sort(key=lambda x: x[0])

        if time.time() > deadline - 60:
            break

    # ------------------------------------------------------------------
    # Phase 2 – deep Adam optimisation starting from the current best
    if best_seq is not None and time.time() < deadline - 40:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam final] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – optional L‑BFGS continuation (only if time permits)
    if best_seq is not None and time.time() < deadline - 15:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap random hill‑climber
    if best_seq is not None and time.time() < deadline - 3:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
