#!/usr/bin/env python
"""
Propose a low‑value step‑function.

The optimizer works in three stages:

1. **Initial pool** – a diverse set of shapes (uniform, Gaussian,
   Beta, triangular, exponential, and interleaved permutations).  The
   best candidate from the pool is taken as a warm start.  If a previous
   solution (`height_sequence_1`) exists it is added to the pool.

2. **Gradient‑based refinement** – a short Adam optimisation on a
   smooth‑max (log‑sum‑exp) surrogate followed by an exact‑max
   exponentiated‑gradient sub‑gradient phase.  This quickly drives the
   objective down and produces a solid baseline.

3. **Targeted simulated annealing** – a heavyweight local‑search that
   mixes three move types:
      * random mass transfer,
      * pair swap,
      * a *targeted* move which reduces the current maximum convolution
        by moving mass away from the indices that contribute to that
        maximum.
   The temperature decays linearly; uphill moves are accepted with the
   Metropolis probability.  The best vector seen during the run is
   kept.

The whole routine respects the user‑supplied budget (default 1000 s) and
always returns the best sequence found before termination.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)²."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (∑=1, v≥0)."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(best_a, n, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase λ)
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient sub‑gradient phase."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_targeted(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with three move types:

        * random mass‑transfer,
        * swap of two coordinates,
        * *targeted* mass‑transfer aimed at the current maximal
          convolution index.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    p_target = 0.35   # probability of a targeted move
    p_swap = 0.12     # probability of a swap move

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        r = np.random.rand()
        a2 = a.copy()

        if r < p_target:
            # ----- targeted move -----
            k_max = conv.argmax()
            # try to pick a pair (i, j) that contributes to k_max
            found = False
            for _ in range(5):
                i = np.random.randint(n)
                j = k_max - i
                if 0 <= j < n and a[i] > 1e-14 and a[j] > 1e-14:
                    found = True
                    break
            if not found:
                # fallback to a normal mass transfer
                i = np.random.randint(n)
                if a[i] < 1e-14:
                    it += 1
                    continue
                j = np.random.randint(n)
                if i == j:
                    it += 1
                    continue
                frac = np.random.rand() * 0.4
                delta = a[i] * frac
                a2[i] -= delta
                a2[j] += delta
            else:
                # move mass away from the larger of the two contributors
                donor = i if a[i] > a[j] else j
                receiver = np.random.randint(n)
                while receiver == donor:
                    receiver = np.random.randint(n)
                frac = np.random.rand() * 0.4
                delta = a[donor] * frac
                a2[donor] -= delta
                a2[receiver] += delta

        elif r < p_target + p_swap:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            # ----- generic mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            print(f"[Anneal it={it:06d}] new best {val2:.6f}")

        # Metropolis acceptance
        if val2 < val or np.random.rand() < np.exp(-(val2 - val) / T):
            a = a2
            conv = conv2
            val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a rich pool of starter vectors."""
    cand = []

    # warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric exponentials (both ends)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Double‑Gaussian (peaks at the ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Central Gaussian
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        shape_sum = shape.sum()
        if shape_sum > 0:
            shape /= shape_sum
            cand.append(shape)

    # Symmetric Beta‑type shapes (α = β)
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Interleaved permutations (helps spread large entries)
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns (equally spaced or random)."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a plain Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for the time budget
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build the initial pool
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick sift through the pool (≈20 % of the budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + (deadline - time.time()) * 0.20
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # quick Adam refinement
        sub_dead = min(explore_deadline, time.time() + 4.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improved → {val_adam:.6f}")

        # short sub‑gradient polishing (if time permits)
        if time.time() < explore_deadline:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, explore_deadline,
                max_iters=15_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Sub‑grad improved → {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep sub‑gradient refinement of the incumbent (≈30 %)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 40:
        print("[Refine] Long sub‑gradient optimisation started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] New best after sub‑grad → {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – heavy targeted simulated‑annealing (≈40 % budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Anneal] Starting targeted simulated annealing")
        a_ann, val_ann = _anneal_targeted(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] Improved best → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – occasional random restarts to escape possible traps
    # ------------------------------------------------------------------
    while time.time() < deadline - 5:
        # jitter the current best slightly and anneal again (shorter run)
        jitter = np.random.rand(n) * 1e-3
        a0 = best_seq * (1.0 + jitter)
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()
        sub_dead = time.time() + 12.0  # short budget for this mini‑run
        a_mid, _ = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=30_000, eta0=0.9
        )
        a_ann, val_ann = _anneal_targeted(
            a_mid,
            n,
            conv_fft_len,
            deadline,
            max_moves=50_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Restart] New best after mini‑anneal → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback when nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
