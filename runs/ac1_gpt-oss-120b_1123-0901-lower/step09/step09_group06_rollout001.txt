def _adam_max_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200_000, lr_start=0.1):
   eps = 1e-15
   logits = np.log(a0 + eps)
   m = np.zeros_like(logits)
   v = np.zeros_like(logits)

   beta1=0.9; beta2=0.999; eps_adam=1e-8
   lr = lr_start
   best_a = a0.copy()
   best_val = evaluate_sequence(a0.tolist())

   it=0
   while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # argument of max
        max_idx = conv.argmax()
        # gradient w.r.t a using argmax
        w = np.zeros_like(conv)
        w[max_idx] = 1.0
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # chain rule through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**(it+1))
        v_hat = v/(1-beta2**(it+1))
        step = lr*m_hat/(np.sqrt(v_hat) + eps_adam)
        logits -= step

        # every 5000 iterations, add tiny random jitter to avoid stagnation
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n)-0.5)*2e-2

        # evaluate objective
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[AdamMax it={it}] new best {val:.6f}")

        # decay learning rate
        if it % 50000 == 0 and it>0:
            lr *= 0.95

        it+=1

   return best_a, best_val

Propose a low‑value step function.

Algorithm
---------

1. **Diverse start‑pool** – deterministic families (edge exponential,
   edge power‑law, β‑pdf, double‑Gaussian, sinusoid, raised‑cosine,
   semi‑circle, uniform, alternating spikes, block spikes, random Dirichlet)
   together with the optional warm‑start ``height_sequence_1``.  All candidates
   are normalised to a probability simplex and the best few are kept.

2. **Exact‑max Adam** – the objective
        F(a)=2·n·max(a∗a)/(∑a)²
   is scale‑invariant, so we work on the simplex (soft‑max of logits).
   The gradient of the *hard* max convolution is computed analytically
   (a one‑hot weight at the arg‑max of the convolution).  Adam is run on
   the logits with this exact sub‑gradient until the time‑budget is almost
   exhausted.

3. **Cheap hill‑climber** – a final short random‑move phase that
   randomly transfers a fraction of mass between two positions and
   keeps the move only if the true objective improves.

All helper functions are top‑level, there are no closures or lambdas,
and the routine always returns the best sequence found before the 1000 s
deadline.

import time

# Low‑level utilities
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the exact gradient.
    Returns an array of length len(a) containing
    sum_j w[j] * a[j‑i] (zeros outside the range).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:

def _alternating_spikes(n: int) -> np.ndarray:

    a = np.zeros(n, dtype=np.float64)
    a[::2] = 1.0
    a_sum = a.sum()
    if a_sum > 0:
        a /= a_sum
    return a


def _block_spike(n: int, k: int) -> np.ndarray:
    """Uniform block of length k at the left edge."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        a[:k] = 1.0 / k
    return a

# Optimiser – Adam on the *exact* max convolution
def _adam_max_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser that uses the *hard* max (a one‑hot weight) for the
    gradient.  Operates on logits → soft‑max so the iterate always stays
    on the simplex.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start

    best_a = a0.copy()
    # the exact objective can be computed from the convolution max
    conv0 = _full_convolution(a0, conv_fft_len)
    best_val = 2.0 * n * conv0.max() / (a0.sum() ** 2)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # convolution and arg‑max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        max_idx = int(conv.argmax())

        # one‑hot weight for the max index
        w = np.zeros_like(conv)
        w[max_idx] = 1.0

        # exact gradient w.r.t. a (sub‑gradient of the max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # occasional tiny random shake near the deadline (helps escape plateaus)
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # evaluate true objective (cheap because we already have conv)
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam‑max it={it:06d}] new best {val:.6f}")

        # learning‑rate decay (slow)
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber (same as in the reference implementation)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Simple mass‑movement hill‑climber, only accepts improving moves."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of the source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimizes the evaluation
    function.  The routine respects the time budget, prints progress and
    always returns the best sequence (plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – can be tweaked, 4096 gives a good trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of starters
    pool: List[np.ndarray] = []

    if warm is not None:
        pool.append(warm.copy())
        for sh in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, sh))

    # deterministic families
    for a in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, a))

    for gamma in np.logspace(-2, 1.2, 15):
        pool.append(_edge_exponential(n, gamma))

    for p in np.linspace(-8, -0.2, 15):
        pool.append(_edge_power_law(n, p))

    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0),
        (5.0, 0.2), (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))
    pool.append(_alternating_spikes(n))

    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    for k in (64, 128, 256, 512, 1024):
        pool.append(_block_spike(n, k))

    # ------------------------------------------------------------------
    # Quick scan of the pool
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx:03d}] value {val:.6f}")

        if time.time() > deadline - 120:   # keep ~2 min for deeper phases
            break

    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation with the exact max‑gradient
    if time.time() < deadline - 30:
        a_adam, val_adam = _adam_max_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam‑max final] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap final hill‑climber
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final safety clipping (evaluate_sequence also clips, but we keep it)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity print
    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

# Helper for sparse spike generation (used in the pool)
def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k equally weighted spikes placed at random positions."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a
