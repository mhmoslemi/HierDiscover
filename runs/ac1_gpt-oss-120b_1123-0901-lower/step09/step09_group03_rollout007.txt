   * **Exact‑max exponentiated‑gradient** – sub‑gradient descent on the
     true non‑smooth objective; the update is mirror‑descent w.r.t.
     the KL‑divergence (a ← a·exp(‑η·g)) and the step size η decays as
     1/√t.
   * **Targeted simulated annealing** – repeatedly locate the index k
     where the convolution attains its maximum, find the pair (i,j)
     with i+j=k that contributes the most (largest a_i·a_j) and move a
     small fraction of mass away from that pair.  Moves are accepted
     with a temperature that decays linearly.
4. If time permits, up‑sample the current best vector to 8192 points,
   fine‑tune it with a short Adam run and a longer exponentiated‑gradient
   phase.
5. Return the best sequence (clipped to the limits of `evaluate_sequence`).

The optimiser keeps a *running best* and stops early as soon as the
objective drops below the target 1.503.  All operations are vectorised
and use FFT‑based convolutions, which makes tens of millions of
iterations feasible within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (output length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (exact, but cheap thanks to FFT) -----------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return F(a)=2·n·max(a∗a) / (∑a)²  (a is assumed non‑negative)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Optimisation phases ----------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # stay interior of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.045
    lam = 0.05
    lam_factor = 1.025               # speed‑up of the temperature rise

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (track best)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule for smooth‑max
        lam = min(lam * lam_factor, 1e8)

        # tiny shake near the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # occasional LR decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        # early exit if we already beat the target
        if best_val < 1.503:
            break

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices that attain the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against numerical drift
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # early stop when target reached
        if best_val < 1.503:
            break

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Targeted simulated‑annealing moves that disturb the most
    responsible pair at the current peak of the convolution."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.07
    T_end = 5e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        # linear temperature decay
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # indices that can form the pair i+j=kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # single‑index case – shift a small fraction elsewhere
            if a2[i_src] < 1e-14:
                move += 1
                continue
            delta = a2[i_src] * 0.04          # 4 % of the mass
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # remove a tiny chunk from the dominant pair
            delta = min(a2[i_src], a2[j_src]) * 0.04
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        # occasional early break
        if best_val < 1.503:
            break

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation ----------------------------------------------------
def _window_candidates(n: int) -> list[np.ndarray]:
    """Classic DSP windows (Hann, Hamming, Blackman, Kaiser)."""
    i = np.arange(n, dtype=np.float64)
    cand = []

    # Hann
    hann = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    cand.append(hann / hann.sum())

    # Hamming
    hamming = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    cand.append(hamming / hamming.sum())

    # Blackman
    black = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    cand.append(black / black.sum())

    # Kaiser with a few betas
    for beta in (2.0, 5.0, 9.0, 12.0):
        a = 2.0 * i / (n - 1) - 1.0
        kaiser = np.i0(beta * np.sqrt(1.0 - a ** 2)) / np.i0(beta)
        if np.all(np.isfinite(kaiser)):
            cand.append(kaiser / kaiser.sum())

    return cand


def _u_shape_candidates(n: int) -> list[np.ndarray]:
    """U‑shaped families a_i ∝ ((i+1)*(n‑i))^α   (α can be negative)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)          # (i+1)*(n-i)
    cand = []
    for alpha in np.linspace(-2.5, 2.5, 31):
        shape = prod ** alpha
        if shape.sum() == 0 or np.isnan(shape).any():
            continue
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        cand.append(shape)
    return cand


def _exponential_candidates(n: int) -> list[np.ndarray]:
    """Simple left‑to‑right exponential decays (and their reverses)."""
    cand = []
    for gamma in (0.2, 0.5, 1.0, 1.5, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        for shape in (left, right, left + right):
            shape = np.maximum(shape, 0.0)
            s = shape.sum()
            if s > 0:
                cand.append(shape / s)
    return cand


def _random_dirichlet_candidates(n: int) -> list[np.ndarray]:
    """A handful of Dirichlet draws with diverse concentration parameters."""
    cand = []
    for alpha in (0.01, 0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        cand.append(a)
    return cand


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified start pool."""
    cand = []

    # warm start (and a few cyclic shifts)
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 6, n // 3, n // 2):
            cand.append(np.roll(warm, shift))

    cand.extend(_window_candidates(n))
    cand.extend(_u_shape_candidates(n))
    cand.extend(_exponential_candidates(n))
    cand.extend(_random_dirichlet_candidates(n))

    # A simple flat baseline – always useful
    cand.append(np.full(n, 1.0 / n))

    return cand


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing the evaluation
    score.  The routine respects the given time budget and always returns
    the best sequence it found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – keep it moderate for fast iterations
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diversified pool and pick the best raw candidate
    pool = _initial_candidates(n, warm)

    best_seq = None
    best_val = float("inf")
    print("[Pool] evaluating %d initial candidates" % len(pool))

    for idx, a in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool] candidate {idx:03d} gives new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – smooth‑max Adam (≈30 % of the remaining budget)
    if best_seq is not None and time.time() < deadline - 20:
        phase_dead = time.time() + (deadline - time.time()) * 0.30
        print("[Phase I] Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            phase_dead,
            max_iters=150_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient sub‑gradient descent
    if best_seq is not None and time.time() < deadline - 15:
        print("[Phase II] Sub‑gradient (exp‑grad) optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – targeted annealing (≈20 % of remaining time)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Pair‑wise targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[Phase III] annealing improved to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – optional up‑sampling if time permits (larger grid)
    if best_seq is not None and time.time() + 30 < deadline:
        print("[Phase IV] Upsample to 8192 and refine")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation to the finer grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        sub_dead_big = min(deadline, time.time() + 8.0)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=180_000,
        )

        # longer sub‑gradient refinement on the large grid
        a_big_ref, val_big = _subgradient_optimize_expgrad(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Phase IV] up‑sampled best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to the limits imposed by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
