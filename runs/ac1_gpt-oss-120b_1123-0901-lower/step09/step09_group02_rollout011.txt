"""
stepfunction_optimizer_lp.py

A two‑stage optimizer:

* Stage 1 – fast first‑order descent (exponentiated‑gradient on the true
  max‑convolution).  This quickly reaches the level obtained by the
  previous submission (~1.508).

* Stage 2 – LP‑based improvement (the “LP‑step” from the original paper).
  For the current iterate f we solve

        max  ∑ g_i
        s.t. (f * g)_k ≤ M   (for all k),   g_i ≥ 0 ,

  where M = max_k (f * f)_k.  The solution g is then projected back to the
  probability simplex and mixed with f:  f ← (1‑t)·f + t·g.  Repeating this
  step (with a slowly decreasing step size t) pushes the maximal
  convolution down further, typically beating the earlier best bound.

All vectors are kept on the probability simplex (∑a_i = 1, a_i ≥ 0); the
objective 2·n·max(conv)/(∑a)^2 then equals the maximal convolution itself,
so we only need to minimise  max(conv).

The code respects a user‑supplied time budget (default 1000 s) and always
returns the best sequence found before the deadline.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x ≥ 0, ∑x = 1}."""
    if np.abs(v.sum() - 1.0) < 1e-12 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no clipping) – useful for bookkeeping."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated‑gradient on the true max)
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set of positions attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑based refinement
# ----------------------------------------------------------------------
def _lp_iteration(a: np.ndarray, t: float, n: int, conv_fft_len: int, deadline: float):
    """
    Perform a single LP‑step:
       maximise ∑ g_i   s.t.   (a * g)_k ≤ M  ∀k,   g_i ≥ 0
    then mix the result back into a with step size t.
    Returns the new a and the updated max‑convolution.
    """
    # compute current bound M
    conv = _full_convolution(a, conv_fft_len)
    M = conv.max()

    rows = 2 * n - 1
    # Build the dense‑ish Toeplitz matrix via sparse diagonals.
    # Each diagonal i carries the constant a[i].
    offsets = np.arange(n, dtype=np.int64)
    C = sp.diags(a, offsets=offsets, shape=(rows, n), format='csc')

    b_ub = np.full(rows, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximise sum(g)  → minimise -sum(g)

    # Linear programme
    res = linprog(
        c,
        A_ub=C,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, deadline - time.time())},
    )

    if not res.success:
        # If the LP fails, fall back to the current a.
        return a, M

    g = res.x
    sum_g = g.sum()
    if sum_g <= 0:
        return a, M

    # Scale g back to the simplex (sum = 1)
    g = g / sum_g

    # Convex combination and projection
    a_new = (1.0 - t) * a + t * g
    a_new = _project_simplex(a_new)

    # New max convolution
    conv_new = _full_convolution(a_new, conv_fft_len)
    M_new = conv_new.max()
    return a_new, M_new


def _lp_refine(a: np.ndarray, n: int, conv_fft_len: int, deadline: float):
    """Run a few LP‑steps with a decreasing step size."""
    t = 0.2  # initial mixing weight
    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < 30:
        a, M = _lp_iteration(best_a, t, n, conv_fft_len, deadline)
        val = 2.0 * n * M   # sum(a)=1 ⇒ denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP it={it:02d}] improved value {val:.6f}")
        else:
            # no improvement – shrink step size
            t *= 0.5
            if t < 0.01:
                break
        it += 1
    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a combination of exponentiated‑gradient sub‑gradient optimisation
    followed by a dedicated LP‑based refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – we keep the previously successful length
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)   # = 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # = 16384

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        warm_sum = warm.sum()
        if warm_sum >= 0.01:
            warm = warm / warm_sum
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a simple initial candidate if no warm start is present
    if warm is None:
        # a centred Gaussian – a shape that is already pretty good
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        sigma = 0.15
        warm = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        warm = np.maximum(warm, 0.0)
        warm /= warm.sum()

    # ------------------------------------------------------------------
    # Phase I – exponentiated‑gradient sub‑gradient optimisation
    sub_deadline = time.time() + (deadline - time.time()) * 0.55
    a_sub, val_sub = _subgradient_optimize_expgrad(
        warm,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_deadline,
        max_iters=400_000,
        eta0=0.9,
    )
    print(f"[Phase I] sub‑gradient best value {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – LP‑based refinement (the novel part)
    lp_deadline = time.time() + (deadline - time.time()) * 0.40
    a_lp, val_lp = _lp_refine(a_sub, n, conv_fft_len, lp_deadline)
    print(f"[Phase II] LP refinement best value {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase III – a quick simulated‑annealing hill‑climb to polish
    # (optional but cheap)
    if time.time() < deadline - 5:
        # simple mass‑transfer annealing
        a = a_lp.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val = val
        best_a = a.copy()

        T_start = 0.07
        T_end = 1e-4
        moves = 0
        max_moves = 30_000
        while time.time() < deadline and moves < max_moves:
            T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
            i = np.random.randint(n)
            if a[i] < 1e-14:
                moves += 1
                continue
            j = np.random.randint(n)
            if i == j:
                moves += 1
                continue
            frac = np.random.rand() * 0.35
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                a = a2
                val = val2
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
            moves += 1
        a_lp = best_a
        val_lp = best_val
        print(f"[Phase III] annealing polished value {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Final sanity: clip, normalise and return
    a_lp = np.clip(a_lp, 0.0, 1000.0)
    a_lp = _project_simplex(a_lp)   # ensure strict sum‑to‑1 (avoids tiny drift)

    final_val = _evaluate_objective(a_lp, n, conv_fft_len)
    print(f"Final best value {final_val:.6f} (length {len(a_lp)})")
    return a_lp.tolist()
