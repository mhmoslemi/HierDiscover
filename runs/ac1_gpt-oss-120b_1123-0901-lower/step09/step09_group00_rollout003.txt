                # but keep val as current state's value
                val = val2

"""
Hybrid optimiser for the max‑auto‑convolution objective.

The search works in three stages:
1.  A fast random‑sampling stage builds a diverse pool of start vectors
    (Dirichlet draws, exponential, double‑Gaussian, half‑sine, warm‑start
    if available) and selects the best one according to the exact objective.
2.  The selected vector is refined by a long Adam run on a smooth‑max surrogate,
    followed by an exponentiated‑gradient sub‑gradient phase, a Polyak‑step
    projected gradient descent and finally a simulated‑annealing local search.
    Each phase runs as long as the global time budget permits.
3.  The best vector encountered is returned as a plain Python list; it is
    already projected onto the feasible set (non‑negative, sum>0).

The implementation relies on FFT‑based convolutions (O(n log n)) and a
Euclidean projection onto the probability simplex.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions (all top‑level – required by the rules)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << ((x - 1).bit_length())


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum(a)²).  Assumes a≥0."""
    s = a.sum()
    if s < 0.01:
        return np.inf
    # keep the scale (the objective is homogeneous, we keep the current scale)
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max() / (s * s))


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑softmax parametrisation)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.001
    lam_max = 1e6

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- soft‑max → probability vector ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- objective (exact, for book‑keeping) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)          # sum(a)=1 ⇒ denominator =1
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {cur_val:.6f}")

        # ---- smooth‑max weights ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smooth‑max (w.r.t a) ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule ----
        lam = min(lam * lam_factor, lam_max)

        # ---- learning‑rate decay (very gentle) ----
        if (it + 1) % 300_000 == 0:
            lr *= 0.7

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 3_000_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {cur_val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- exponentiated‑gradient step (mirror descent on simplex) ----
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # keep positivity
        # re‑project onto the simplex
        a_sum = a.sum()
        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / a_sum

        it += 1

    return best_a, best_val


def _gd_optimize(a0: np.ndarray,
                 n: int,
                 conv_fft_len: int,
                 corr_fft_len: int,
                 deadline: float,
                 max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent with a Polyak step size."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = float(2.0 * n * max_c)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[GD it={it:07d}] new best {cur_val:.6f}")

        # ---- sub‑gradient of the exact max (active‑set) ----
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- Polyak step size (requires best value known) ----
        diff = cur_val - best_val
        if diff > 0:
            step = diff / (np.linalg.norm(grad) ** 2 + 1e-12)
            step = min(step, 0.2)        # stability cap
        else:
            step = 0.0

        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 250_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    cur_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = cur_val

    T_start = 0.08
    T_end = 1e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        # linearly cooled temperature
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤40 %) of mass from i to j
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            cur_val = val2
        else:
            # metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                a = a2
                cur_val = val2

        it += 1

    return best_a, best_val

# Candidate generators

def _deterministic_shapes(n: int) -> list[np.ndarray]:
    """A handful of hand‑crafted shapes (all normalised to sum = 1)."""
    shapes = []

    # symmetric exponential decay
    beta = 4.0 / n
    idx = np.arange(n, dtype=np.float64)
    exp_shape = np.exp(-beta * np.abs(idx - (n - 1) / 2.0))
    exp_shape /= exp_shape.sum()
    shapes.append(exp_shape)

    # double Gaussian (as in the reference code)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.15, 0.25):
        g = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + \
            np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        g /= g.sum()
        shapes.append(g)

    # half‑sine (non‑negative)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    shapes.append(s)

    # power‑law (triangular‑like)
    k = np.arange(1, n + 1, dtype=np.float64)
    triangular = k * (n - k + 1)
    triangular = triangular.astype(float) ** -0.7
    triangular /= triangular.sum()
    shapes.append(triangular)

    # uniform
    shapes.append(np.full(n, 1.0 / n))

    return shapes


def _random_dirichlet_pool(n: int, pool_size: int = 160) -> list[np.ndarray]:
    """Generate a pool of Dirichlet draws with varying concentration."""
    pool = []
    for _ in range(pool_size):
        # log‑uniform concentration in [1e‑2, 1e2]
        concentration = 10 ** np.random.uniform(-2, 2)
        a = np.random.dirichlet(concentration * np.ones(n))
        pool.append(a)
    return pool

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The routine respects the given time budget (≈ 1000 s)
    and always returns the best sequence found (as a Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # small safety margin

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if supplied by the surrounding harness)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a large, cheap pool of start vectors and keep the best few
    pool = _deterministic_shapes(n)
    pool.extend(_random_dirichlet_pool(n, pool_size=120))

    if warm is not None:
        pool.append(warm)

    best_val = float("inf")
    best_seq = None

    # quick evaluation of every candidate (exact objective, but cheap with FFT)
    for a in pool:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()

    print(f"[Init] best initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation (smooth max)
    sub_dead = min(deadline, time.time() + 350.0)
    best_seq, best_val = _adam_optimize(best_seq, n,
                                        conv_fft_len, corr_fft_len,
                                        sub_dead, max_iters=2_000_000)
    print(f"[Phase 1] after Adam best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient sub‑gradient optimisation
    sub_dead = min(deadline, time.time() + 250.0)
    best_seq, best_val = _subgradient_optimize(best_seq, n,
                                               conv_fft_len, corr_fft_len,
                                               sub_dead, max_iters=3_000_000,
                                               eta0=0.9)
    print(f"[Phase 2] after sub‑grad best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Plain projected gradient descent (Polyak step size)
    sub_dead = min(deadline, time.time() + 150.0)
    best_seq, best_val = _gd_optimize(best_seq, n,
                                      conv_fft_len, corr_fft_len,
                                      sub_dead, max_iters=600_000)
    print(f"[Phase 3] after GD best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing fine‑tuning
    sub_dead = min(deadline, time.time() + 120.0)
    best_seq, best_val = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          sub_dead, max_moves=250_000)
    print(f"[Phase 4] after annealing best = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling (if we still have time)
    if time.time() + 30.0 < deadline:
        try:
            print("[Upsample] Trying a finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation to the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big /= a_big.sum()

            sub_dead = min(deadline, time.time() + 20.0)
            a_big, val_big = _adam_optimize(a_big, n_big,
                                            conv_fft_len_big, corr_fft_len_big,
                                            sub_dead, max_iters=1_000_000)
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] Adam refinement improved to {val_big:.6f}")

            # a quick final projected‑gradient pass on the finer grid
            sub_dead = min(deadline, time.time() + 10.0)
            a_big, val_big = _gd_optimize(a_big, n_big,
                                          conv_fft_len_big, corr_fft_len_big,
                                          sub_dead, max_iters=200_000)
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] GD refinement improved to {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # Ensure a feasible return (fallback to uniform if everything failed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's allowed range and recast to list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
