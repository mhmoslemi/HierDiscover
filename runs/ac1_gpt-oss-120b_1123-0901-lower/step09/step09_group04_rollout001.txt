"""
Propose a low‑value step‑function.

The algorithm works in four phases:

1. **Diverse warm‑starts** – generate a large pool of deterministic families
   (uniform, sinusoid, edge‑exponential, edge‑power‑law, beta, double‑Gaussian)
   plus many random Dirichlet draws.  Each candidate is normalized to sum to 1
   and evaluated; the best becomes the seed.

2. **Fine‑grained param sweep** – the families above are scanned over dense
   parameter grids (e.g. 200 values for the exponential decay) to discover
   a shape that already beats the initial best.

3. **Adaptive sub‑gradient descent** – starting from the best seed we minimise
   the exact objective 2·n·max(conv(a,a))/(∑a)².
   The gradient of the max‑convolution (a sub‑gradient) is obtained from the
   active set of indices attaining the maximum.  At each iteration a few
   candidate step‑sizes are tried; the move is projected onto the probability
   simplex.  The step‑size is increased when progress is made and halved
   otherwise, giving a fast, robust descent without needing a hand‑tuned schedule.

4. **Cheap hill‑climber** – in the last seconds a handful of random mass‑transfer
   moves (moving a fraction of the mass from one index to another) are performed
   to squeeze any remaining improvement.

All FFT‑based convolutions are performed at a power‑of‑two length for speed.
The routine always returns the best vector found within the allotted wall‑time
(≤ 1000 s).  The returned list satisfies the validation performed by
``evaluate_sequence`` (non‑negative, finite, and each entry ≤ 1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho_arr = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho_arr.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho_arr[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes (all return a probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α (scalar)."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence's formula)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²).  a is assumed non‑negative."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adaptive sub‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact L∞‑convolution objective.
    Returns the best vector seen and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    step = 1.0  # initial step length in the simplex tangent space
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        act_cnt = int(active.sum())
        if act_cnt == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / act_cnt

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # try a handful of step sizes
        improved = False
        best_candidate = a
        best_cand_val = best_val
        for factor in (1.0, 0.5, 0.25, 0.125):
            step_sz = step * factor
            a_new = _project_onto_simplex(a - step_sz * grad)
            val_new = _objective(a_new)
            if val_new < best_cand_val - 1e-12:
                best_cand_val = val_new
                best_candidate = a_new
                improved = True

        if improved:
            a = best_candidate
            best_val = best_cand_val
            best_a = a.copy()
            step = min(step * 1.02, 5.0)  # gently increase step when we are improving
            if it % 2000 == 0:
                print(f"[Subgrad it={it:06d}] new best {best_val:.6f}  step={step:.4f}")
        else:
            # no improvement – shrink step
            step = max(step * 0.5, 1e-6)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climber (mass‑transfer moves)
# ----------------------------------------------------------------------


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Randomly transfer a fraction of mass from one index to another,
    keep the move if the objective improves.
    """
    a = a0.copy()
    best_val = _objective(a)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget (≤ budget_s seconds) and always
    returns the best sequence (as a plain Python list) found so far.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # small safety margin

    # ------------------------------------------------------------------
    # Problem size (can be tuned – 4096 works well with the time budget)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from the previous best if available
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build an initial pool of diverse candidates
    # ------------------------------------------------------------------
    pool = []

    # Add deterministic families with a few default parameters
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))

    # Edge‑exponential (coarse grid – later we do a fine sweep)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law (coarse)
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf (some classic shapes)
    for a, b in [(0.5, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2):
        pool.append(_double_gaussian(n, sigma))

    # Add the warm‑start (with a few cyclic shifts)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Add a few random Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(8):
            pool.append(_random_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    # Quick scan of the pool
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = rng.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    # ------------------------------------------------------------------
    # Fine‑grained parametric sweep (still cheap)
    # ------------------------------------------------------------------
    # edge‑exponential --------------------------------------------------
    if time.time() < deadline - 5:
        for gamma in np.logspace(-2, 2, 200):
            shape = _edge_exponential(n, gamma)
            val = _objective(shape)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Sweep exp gamma={gamma:.4f}] new best {val:.6f}")

    # edge‑power‑law ----------------------------------------------------
    if time.time() < deadline - 5:
        for alpha in np.linspace(-4.0, -0.2, 200):
            shape = _edge_power_law(n, alpha)
            val = _objective(shape)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Sweep pow alpha={alpha:.4f}] new best {val:.6f}")

    # beta‑distribution -------------------------------------------------
    if time.time() < deadline - 5:
        for _ in range(120):
            a_param = 10 ** np.random.uniform(-1, 1)  # roughly 0.1–10
            b_param = 10 ** np.random.uniform(-1, 1)
            shape = _beta_shape(n, a_param, b_param)
            val = _objective(shape)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Sweep beta a={a_param:.3f} b={b_param:.3f}] new best {val:.6f}")

    # double‑Gaussian ---------------------------------------------------
    if time.time() < deadline - 5:
        for sigma in np.linspace(0.01, 0.4, 120):
            shape = _double_gaussian(n, sigma)
            val = _objective(shape)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Sweep dg sigma={sigma:.4f}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Adaptive sub‑gradient descent (core optimiser)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        sub_deadline = deadline - 8  # leave a few seconds for final polishing
        a_sgd, val_sgd = _subgradient_descent(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
        )
        if val_sgd < best_val:
            best_val = val_sgd
            best_seq = a_sgd.copy()
            print(f"[Subgradient descent] new best {val_sgd:.6f}")

    # ------------------------------------------------------------------
    # Final cheap hill‑climber (last few seconds)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local moves] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback & final formatting
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
