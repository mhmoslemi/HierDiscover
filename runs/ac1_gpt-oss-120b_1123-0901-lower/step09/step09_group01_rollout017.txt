"""
Step‑function optimizer – extended search.

The optimizer proceeds in four phases:

1. **Broad parametric sweep** – many more families are sampled
   (power‑law, exponential, stretched‑exponential, linear decay,
   two‑exponential mixtures, piecewise‑constant, random spikes, …).
   The ten best vectors (smallest objective) are retained.

2. **Smooth‑max Adam** – each of the retained candidates is refined
   with a short Adam run on the log‑sum‑exp surrogate of the max‑convolution.
   The best result among them is kept.

3. **Exact‑max sub‑gradient (exponentiated‑gradient)** – a long mirror‑descent
   run from the current best vector drives the solution towards a (near) local
   optimum of the original non‑smooth objective.

4. **Fine‑tuning** – a cheap convex combination with the uniform vector and a
   brief simulated‑annealing sweep are performed if time permits.

All steps respect the total 1000 s budget and the routine always returns a
valid non‑negative list of coefficients.
"""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for the (sub)gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(a*a)/(sum a)² (scaling‑invariant)."""
    if a.sum() < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p, p can be negative."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_linear_decay(ratio: float, n: int) -> np.ndarray:
    """
    Linear decay from a peak to zero.
    `ratio` ∈ (0,1] defines the support length L = ceil(ratio·n).
    a_i = L‑i for i<L, else 0.
    """
    L = max(1, int(math.ceil(ratio * n)))
    a = np.arange(L, 0, -1, dtype=np.float64)   # L, L‑1, … ,1
    seq = np.zeros(n, dtype=np.float64)
    seq[:L] = a
    seq /= seq.sum()
    return seq


def _candidate_two_exponential(alpha: float, beta: float, w: float, n: int) -> np.ndarray:
    """Mixture w·e^{‑αi} + (1‑w)·e^{‑βi}."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-alpha * i) + (1.0 - w) * np.exp(-beta * i)
    a /= a.sum()
    return a


def _candidate_piecewise_constant(k: int, n: int) -> np.ndarray:
    """k blocks of equal length, each block gets a random Dirichlet share."""
    if k >= n:
        k = n
    block_len = n // k
    rest = n - block_len * k
    # Dirichlet on k blocks
    weights = np.random.dirichlet(np.ones(k))
    a = np.empty(n, dtype=np.float64)
    for j in range(k):
        start = j * block_len
        end = start + block_len
        a[start:end] = weights[j] / block_len
    if rest > 0:
        # distribute remaining positions uniformly with a tiny amount
        a[-rest:] = (weights[-1] / block_len) / rest
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.5          # start temperature
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max → a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Chain rule through soft‑max (mirrored step)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine tuner."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.15
    T_end = 5e-4
    for it in range(max_moves):
        if time.time() >= deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # transfer a random fraction (≤30 %) of the source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

    return best_a, best_val

# Helper to keep the K best candidates ----------------------------------

def _push_top(heap, entry, k):
    """
    `heap` holds at most k entries as a max‑heap based on the first element (value).
    `entry` is a (value, array) tuple.
    """
    if len(heap) < k:
        heapq.heappush(heap, (-entry[0], entry[1]))
    else:
        # heap element with largest value is at top (most negative)
        if -heap[0][0] > entry[0]:
            heapq.heapreplace(heap, (-entry[0], entry[1]))

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.
    The routine respects the time budget and always returns a valid
    non‑negative list of coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # problem dimension – keep the original 4096 (good balance between quality and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep the 10 best candidates ----------
    top_k = 10
    top_heap = []          # max‑heap of (‑value, vector) pairs

    def _maybe_update(a: np.ndarray):
        nonlocal top_heap
        val = _objective(a, n)
        if val == np.inf:
            return
        # keep global best for logging
        if not top_heap or val < -top_heap[0][0]:
            print(f"[Sweep] new incumbent {val:.6f}")
        _push_top(top_heap, (val, a.copy()), top_k)

    # -- monotone power‑law -------------------------------------------------
    for p in np.linspace(-10.0, -0.05, 40):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # -- monotone exponential ------------------------------------------------
    for alpha in np.logspace(-5, 1.2, 35):   # up to ~15
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # -- stretched exponential -----------------------------------------------
    alphas = np.logspace(-4, 0.4, 16)
    betas = np.linspace(0.2, 2.5, 14)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # -- linear decay (support length ratios) --------------------------------
    for ratio in np.linspace(0.1, 1.0, 10):
        a = _candidate_linear_decay(ratio, n)
        _maybe_update(a)

    # -- two‑exponential mixture ---------------------------------------------
    for _ in range(30):
        alpha = 10 ** np.random.uniform(-5, 0.8)
        beta = 10 ** np.random.uniform(-5, 0.8)
        w = np.random.rand()
        a = _candidate_two_exponential(alpha, beta, w, n)
        _maybe_update(a)

    # -- piecewise constant blocks -------------------------------------------
    for k in [8, 16, 32, 64, 128]:
        a = _candidate_piecewise_constant(k, n)
        _maybe_update(a)

    # -- beta distributions --------------------------------------------------
    a_pars = [0.1, 0.3, 0.6, 1.0, 2.0, 4.0, 8.0]
    b_pars = [0.1, 0.3, 0.6, 1.0, 2.0, 4.0, 8.0]
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # -- double Gaussian ------------------------------------------------------
    for sig in np.logspace(-2.5, -0.2, 24):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # -- random spike patterns ------------------------------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            a = _candidate_spike(k, n, random=True)
            _maybe_update(a)

    # -- warm start from previous run ----------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    # extract the best vector from the heap
    best_seq, best_val = None, float("inf")
    if top_heap:
        best_entry = min((-val, vec) for (val, vec) in top_heap)
        best_val = best_entry[0]
        best_seq = best_entry[1].copy()
    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement for the top few candidates -----------------------
    # reserve some budget for the long sub‑gradient phase
    remaining = deadline - time.time()
    if remaining > 30.0:
        # split remaining time equally among the best 3 candidates (if we have them)
        candidates = sorted([(-val, vec) for (val, vec) in top_heap])[:3]
        per_candidate = max(5.0, (remaining - 30.0) / max(1, len(candidates)))
        print("[Phase‑Adam] refining top candidates")
        for idx, (_, cand) in enumerate(candidates):
            if time.time() >= deadline - 10.0:
                break
            adam_deadline = min(deadline - 5.0, time.time() + per_candidate)
            a_adam, val_adam = _adam_optimize(
                cand, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=500_000
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] candidate {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long sub‑gradient optimisation (exact max) -----------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Phase‑Subgrad] long exponentiated‑gradient descent")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.0,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Convex combination with the uniform vector -----------------------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[Phase‑Mix] convex combination with uniform")
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        for gamma in np.linspace(0.0, 1.0, 21):
            combo = (1.0 - gamma) * best_seq + gamma * uniform
            combo /= combo.sum()
            val = _objective(combo, n)
            if val < best_val:
                best_val = val
                best_seq = combo.copy()
                print(f"[Phase‑Mix] gamma={gamma:.2f} → {val:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning -------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        print("[Phase‑Anneal] final simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety – ensure a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final evaluated score (evaluator missing).")

    return best_seq.tolist()
