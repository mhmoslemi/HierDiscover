"""
Propose a non‑negative step function that (approximately) minimises

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

The optimiser has three phases:

1. **Broad sampling** – a large pool of candidates is built from deterministic
   families (uniform, sinusoid, edge‑exponential, edge‑power‑law,
   beta‑pdf, double‑Gaussian) plus many Dirichlet draws with a wide range of
   concentration parameters.  All candidates are normalised to sum‑to‑one and the
   two‑fold convolution is evaluated with an FFT.  The best `top_k` vectors are
   retained.

2. **Projected Adam optimisation** – each elite candidate is refined with a
   projected Adam optimiser that works on the *exact* max‑convolution (the
   active set of indices attaining the current maximum provides a sub‑gradient).
   After every gradient step the iterate is projected onto the probability
   simplex, guaranteeing feasibility.  The number of optimisation steps is
   allocated dynamically from the remaining time budget.

3. **Heavy‑weight hill climbing** – the current best vector is subjected to a
   cheap but massive random‑move hill‑climber (mass transfers between two
   positions).  The move is accepted only if it strictly improves the true
   objective; the loop continues until the wall‑clock deadline is reached.

All phases respect the global time budget (default 1000 s).  The final
sequence is returned as a plain Python list; it can be fed directly into the
provided `evaluate_sequence` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Deterministic families (all return vectors with sum == 1) ----------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a seed pool consisting of deterministic families and a modest
    number of Dirichlet draws."""
    pool = []

    # deterministic families ------------------------------------------------
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
                 (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # a few Dirichlet draws (coarse) --------------------------------------
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    return pool


def _grid_candidates(n: int) -> list[np.ndarray]:
    """Finer sweep over families – many more gamma/alpha values."""
    out = []

    # edge‑exponential – log‑spaced
    for gamma in np.logspace(-2, 1.5, 30):
        out.append(_edge_exponential(n, float(gamma)))

    # edge‑power‑law – linear sweep
    for alpha in np.linspace(-4.0, 2.0, 30):
        out.append(_edge_power_law(n, float(alpha)))

    # beta pdf – grid in log‑space
    a_vals = np.logspace(-1, 1, 15)
    b_vals = np.logspace(-1, 1, 15)
    for a in a_vals:
        for b in b_vals:
            out.append(_beta_shape(n, float(a), float(b)))

    # double‑Gaussian – linear sweep
    for sigma in np.linspace(0.02, 0.5, 30):
        out.append(_double_gaussian(n, float(sigma)))

    return out


# ----------------------------------------------------------------------
# Helper: evaluate the true objective of a *probability* vector (sum == 1)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a∗a) for a already normalised (sum=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Projected Adam optimiser (works on the exact max‑convolution) -------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    init_lr: float = 0.5,
    beta1: float = 0.9,
    beta2: float = 0.999,
    eps: float = 1e-8,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam on the true objective.
    Returns (best_vector, best_value).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full_like(a, 1.0 / n)
    a = a / a.sum()                     # stay on the simplex

    m = np.zeros_like(a)
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 1

    while time.time() < deadline and it <= max_iters:
        # active‑set weight (sub‑gradient of the max)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14).astype(float)
        if active.sum() > 0:
            active /= active.sum()
        else:
            active = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(active, a, corr_fft_len)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)
        step = init_lr / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new)

        # evaluate
        val_new = _objective(a_new, n, conv_fft_len)

        # simple acceptance rule – keep move if it improves
        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()

        # keep the better of the two for the next iteration
        if val_new < _objective(a, n, conv_fft_len):
            a = a_new
        # else we stay with the old a (Adam may still move)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber (mass‑transfer moves) -----------------------
def _random_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climber: transfer a random fraction (up to 30 %) of the mass
    from one position to another and keep the move if it improves the exact
    objective.
    """
    a = a0.copy()
    best_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[HillClimb it={it:06d}] new best {best_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Returns the best sequence found within the given
    time budget.
    """
    # ------------------------------------------------------------------
    # basic setup
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a big pool of start vectors ---------------------------------
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng) + _grid_candidates(n)

    # add a large batch of Dirichlet samples (this dominates the exploration)
    # we stop early if we run out of time
    dirichlet_alphas = [0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0]
    dirichlet_per_alpha = 150          # ~1350 additional candidates
    for alpha in dirichlet_alphas:
        for _ in range(dirichlet_per_alpha):
            if time.time() >= deadline * 0.2:       # reserve 20 % for refinement
                break
            pool.append(_random_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    # quick evaluation of the whole pool, keep the best few -------------
    top_k = 30
    scored = []                        # list of (value, vector)
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        if time.time() >= deadline * 0.2:
            break

    scored.sort(key=lambda x: x[0])
    elite = [a for _, a in scored[:top_k]]
    print(f"[Init] evaluated {len(scored)} candidates, best so far {scored[0][0]:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement on elite candidates
    best_seq = None
    best_val = float("inf")

    for rank, a0 in enumerate(elite):
        if time.time() >= deadline - 10:
            break
        # allocate a slice of the remaining budget (more for higher rank)
        remaining = deadline - time.time()
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=100_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final intensive hill‑climber on the current best ----------
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _random_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=500_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[HillClimb] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the bounds that evaluate_sequence applies
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
