assistantfinal**Solution Overview**

The scoring function  

\[
F(a)=\frac{2\,n\;\max (a*a)}{(\sum a)^2}
\]

is minimized when the *peak* of the linear convolution `a*a` is as small as possible
while the vector stays on the simplex (non‑negative, sum = 1).  
The global lower bound is  

\[
F(a)\ge\frac{2n}{2n-1}\approx 1
\]

so values a little above 1 are reachable for a sufficiently flat convolution.

The previous implementation already uses a good pipeline (diverse initialisation,
smooth‑max Adam, continuation L‑BFGS and a simple hill‑climber).  
To push the bound from 1.5077 to the target ≤ 1.503 we add two new ideas
without breaking the existing workflow:

1. **Exact max‑gradient optimisation** – instead of the log‑sum‑exp
   surrogate we optimise the *real* max using a sub‑gradient
   (the gradient of the active convolution indices).  Adam applied to this
   sub‑gradient gives a much more focussed descent once the search is already
   close to a good region.

2. **Simulated annealing fine‑tuner** – a temperature‑controlled random walk that
   occasionally accepts non‑improving moves.  This helps to escape the tiny
   plateaus that a pure hill‑climber cannot leave.

Both stages are cheap (they reuse the FFT‑based convolution already present)
and fit comfortably into the 1000 s budget.  The code keeps the best sequence
found at any time and returns it as a plain Python list.

```python
"""
Search for a non‑negative step‑function that minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of five stages:

1.  **Diverse initialisation** – a large pool of deterministic families
    (edge‑exponential, edge‑power‑law, β‑pdf, double‑Gaussian,
    logistic, raised‑cosine, semicircle) together with random Dirichlet
    and sparse‑spike vectors.  The best few candidates are kept.

2.  **Smooth‑max Adam optimisation** – the log‑sum‑exp surrogate is
    minimised on the simplex (logits → softmax) with Adam, as in the
    reference implementation.

3.  **Exact‑max Adam optimisation** – Adam is applied again, this time
    using the true sub‑gradient of the max (weights are 1 on the active
    convolution indices).  This gives a sharper descent very close to a
    local optimum.

4.  **Continuation L‑BFGS** – a temperature continuation (α‑schedule) refines
    the solution further, exactly as before.

5.  **Simulated‑annealing hill‑climber** – random mass moves are accepted
    according to a temperature schedule, allowing the search to escape
    tiny plateaus and improve the final value.

All stages respect the 1000 s time budget, print intermediate progress,
and always return the best sequence encountered.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------

def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _logistic_shape(n: int, gamma: float) -> np.ndarray:
    """Symmetric logistic decay from both edges."""
    x = np.arange(n, dtype=np.float64)
    left = 1.0 / (1.0 + np.exp(gamma * (x - n / 2.0)))
    right = 1.0 / (1.0 + np.exp(-gamma * (x - n / 2.0)))
    shape = left + right
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape

def _adam_optimize_smoothmax(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # inverse temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam‑smooth it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _adam_optimize_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser that directly targets the true max (sub‑gradient)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam‑max it={it:06d}] new best {val:.6f}")

        # indices attaining the max (allow ties)
        max_inds = np.flatnonzero(conv == max_c)
        w = np.zeros_like(conv)
        w[max_inds] = 1.0 / max_inds.size

        # true sub‑gradient of max
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # simple learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Continuation L‑BFGS on the smooth‑max surrogate (same as reference)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len,
                                    corr_fft_len, alpha)

        remaining = deadline - time.time()
        max_iter = 150 if remaining > 30 else 50

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val


def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Objective and gradient for the smooth‑max surrogate (used by L‑BFGS)."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c
    exp_alpha_v = np.exp(alpha * v)
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    w = exp_alpha_v / sum_exp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Temperature‑controlled random walk that accepts uphill moves."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    start = time.time()
    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling: T falls from 1e-2 to 1e-6
        frac = (deadline - time.time()) / (deadline - start)
        T = 1e-2 * (frac ** 1.5) + 1e-6

        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a fraction of the source mass; the fraction shrinks with time
        max_frac = 0.3 * frac + 0.01
        delta = a[i] * (np.random.rand() * max_frac)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < cur_val or np.random.rand() < np.exp(-(val2 - cur_val) / T):
            a = a2
            cur_val = val2

            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 5000 == 0:
                    print(f"[SA move {it:06d}] new best {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------- problem size -------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------- warm‑start handling ---------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------- build candidate pool -------------
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # random Dirichlet (various concentrations)
    for alpha in (0.05, 0.1, 0.3, 0.7, 1.5):
        pool.append(_random_dirichlet(n, alpha))

    # edge exponential – fine‑grained gamma sweep
    for gamma in np.logspace(-3, 2, 30):
        pool.append(_edge_exponential(n, gamma))

    # edge power‑law – extended exponent range
    for alpha in np.linspace(-12, -0.2, 30):
        pool.append(_edge_power_law(n, alpha))

    # beta PDFs – diverse shapes
    beta_params = [
        (0.5, 0.5), (0.5, 3.0), (3.0, 0.5), (2.0, 5.0),
        (5.0, 2.0), (0.2, 0.2), (0.2, 6.0), (6.0, 0.2), (0.1, 0.1)
    ]
    for a_, b_ in beta_params:
        pool.append(_beta_shape(n, a_, b_))

    # double Gaussian
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # logistic decays
    for gamma in (0.01, 0.05, 0.2, 0.8, 2.0):
        pool.append(_logistic_shape(n, gamma))

    # sinusoid, uniform, raised‑cosine, semicircle
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # sparse spikes – many sparsities
    for k in (32, 64, 128, 256, 512, 1024):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick scan of the pool
    best_seq = None
    best_val = float("inf")
    rng.shuffle(pool)

    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-8:
            continue
        a /= s
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        if time.time() > deadline - 70:   # reserve ~1 min for deep phases
            break

    # ------------------------------------------------------------------
    # Stage 2 – smooth‑max Adam optimisation (bulk of the budget)
    if best_seq is not None and time.time() < deadline - 50:
        a_adam, val_adam = _adam_optimize_smoothmax(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=350_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Smooth‑max Adam] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – exact‑max Adam optimisation
    if best_seq is not None and time.time() < deadline - 30:
        a_maxadam, val_maxadam = _adam_optimize_max(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
            lr_start=0.04,
        )
        if val_maxadam < best_val:
            best_val = val_maxadam
            best_seq = a_maxadam.copy()
            print(f"[Exact‑max Adam] new best {val_maxadam:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – continuation L‑BFGS refinement
    if best_seq is not None and time.time() < deadline - 15:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – simulated‑annealing fine tuner (remaining seconds)
    if best_seq is not None and time.time() < deadline - 3:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[Simulated‑Annealing] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never be needed)
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to evaluator limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
