"""
Search for a non‑negative step sequence minimizing
    F(a) = 2·n·max(a∗a) / (sum(a)^2)

We use a smooth approximation of the max (soft‑max) and a
momentum‑based projected gradient descent on the simplex.
The algorithm starts from a warm start (if provided) and a few
hand‑crafted families, then refines each candidate with an
annealing schedule on the soft‑max temperature.
A cheap random‑move hill‑climber is run at the end.
"""

import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_onto_simplex(v):
    # Fast Euclidean projection onto the probability simplex.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Simple parametric families (all returned already normalised to sum 1)
def _uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _sinusoid(n):
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape

def _edge_exponential(n, gamma):
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape

def _edge_power_law(n, alpha):
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _beta_shape(n, a, b):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _double_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape

def _random_dirichlet(n, alpha, rng):
    return rng.dirichlet(np.full(n, alpha))

# ----------------------------------------------------------------------
def _smooth_gd_optimize(a0, n, conv_fft_len, corr_fft_len, deadline):
    """
    Momentum based projected gradient descent on a smooth approximation
    of the max‑convolution.
    Returns best vector encountered and its (exact) objective value.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = _uniform(n)
    else:
        a = a / s

    best_a = a.copy()
    best_val = float("inf")
    v = np.zeros_like(a)

    beta = 1.0                # temperature for the soft‑max
    step = 0.5                # base step size
    mu = 0.9                  # momentum coefficient
    it = 0

    while time.time() < deadline:
        # ---- exact objective (used only for bookkeeping) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[smooth‑GD β={beta:.1f} it={it:06d}] new best {val:.6f}")

        # ---- smooth weights (soft‑max) ----
        # protect against overflow: shift by max
        shifted = conv - conv.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # ---- gradient of the smooth objective ----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- momentum & projection onto the simplex ----
        v = mu * v - step * grad
        a = a + v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _uniform(n)

        # ---- schedule updates ----
        it += 1
        # increase temperature slowly; after every few thousand steps double it
        if it % 3000 == 0:
            beta = min(beta * 2.0, 256.0)
            step = max(step * 0.8, 1e-4)   # slowly shrink step as landscape becomes sharper

    return best_a, best_val

def _local_random_search(a0, n, conv_fft_len, deadline, max_moves=200_000):
    """
    Very cheap hill‑climber that transfers a random fraction of the mass
    from one index to another and keeps the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[local it={it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  A small population of analytically‑motivated
    seeds (uniform, sinusoid, edge‑exponential, power‑law, beta,
    double‑Gaussian and a warm start if available) is constructed.
    The best few seeds are refined with a smooth‑max gradient descent
    that uses momentum and an annealing schedule on the soft‑max
    temperature.  After the global time budget is almost exhausted,
    a lightweight random‑move hill‑climber performs a final polish.
    The returned list satisfies all checks performed by
    ``evaluate_sequence``.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----- optional warm start -------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ----- build a modest pool of initial candidates ---------------------------
    pool = []

    # deterministic families (few representative parameters)
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    for gamma in (0.2, 1.0, 5.0):
        pool.append(_edge_exponential(n, gamma))
    for alpha in (-2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    for a, b in [(0.5, 0.5), (2.0, 2.0), (5.0, 1.0)]:
        pool.append(_beta_shape(n, float(a), float(b)))
    for sigma in (0.08, 0.15, 0.3):
        pool.append(_double_gaussian(n, sigma))

    # a few random Dirichlet draws
    for alpha in (0.3, 0.7, 1.5):
        for _ in range(2):
            pool.append(_random_dirichlet(n, alpha, rng))

    # warm start and a few cyclic shifts
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):

    # ----- quick evaluation of the pool, keep the most promising -------------

    scored.sort(key=lambda x: x[0])

    best_val = float("inf")

        local_deadline = time.time() + share

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Search for a non‑negative step sequence minimizing

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

Approach
--------
* Build a modest pool of analytically motivated candidates
  (uniform, sinusoid, edge‑exponential, edge‑power‑law, beta‑pdf,
  double‑Gaussian, a few Dirichlet draws, and a warm start if available).
* Keep the best few seeds and refine each of them with a **smooth‑max**
  gradient descent:
      - the max‑convolution is approximated by a soft‑max with temperature β.
  - we use a momentum update and project back onto the probability simplex.
  - β is increased geometrically (annealing) while the step size is
    reduced, allowing the optimiser to first flow in a smooth landscape
    and later focus on the true peak.
* When the wall‑clock budget is almost exhausted a cheap random‑move
  hill‑climber finalises the solution.
The returned list satisfies all checks performed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Simple families (all returned already normalised to sum 1) ----------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Smooth‑max gradient descent -------------------------------------------
def _smooth_gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Momentum based projected gradient descent on a smooth approximation
    of max(a*a).  Returns the best vector seen and its *exact* objective.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = _uniform(n)
    else:
        a = a / s

    best_a = a.copy()
    best_val = float("inf")
    v = np.zeros_like(a)

    beta = 1.0          # soft‑max temperature
    step = 0.5          # base step size
    mu = 0.9            # momentum coefficient
    it = 0

    while time.time() < deadline:
        # exact objective (used only for logging / best‑track)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c            # sum(a)=1 → denominator omitted
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[smooth‑GD β={beta:.1f} it={it:06d}] new best {val:.6f}")

        # soft‑max weights (stable formulation)
        shifted = conv - conv.max()
        w = np.exp(beta * shifted)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = w / w_sum

        # gradient of the smooth objective
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum step + projection onto simplex (clip‑and‑renorm is cheap)
        v = mu * v - step * grad
        a = a + v
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = _uniform(n)

        it += 1

        # anneal temperature and shrink step size periodically
        if it % 3000 == 0:
            beta = min(beta * 2.0, 256.0)
            step = max(step * 0.8, 1e-4)

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap local random moves -----------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of the mass from one position to another.
    Keep the move if it improves the *exact* objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)   # move ≤30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[local it={it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point required by the judge.  Produces a step sequence of length
    4096 that (approximately) minimises the evaluation function.
    """
    # ---- reproducibility --------------------------------------------------
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # ---- time management --------------------------------------------------
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ---- problem dimensions -----------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- optional warm start ------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ----------------------------------------------------------------------
    # 1. Build a modest pool of seeds ---------------------------------------
    pool = []

    # deterministic families (few representative parameters)
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    for gamma in (0.2, 1.0, 5.0):
        pool.append(_edge_exponential(n, gamma))
    for alpha in (-2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    for a, b in [(0.5, 0.5), (2.0, 2.0), (5.0, 1.0)]:
        pool.append(_beta_shape(n, float(a), float(b)))
    for sigma in (0.08, 0.15, 0.3):
        pool.append(_double_gaussian(n, sigma))

    # a few random Dirichlet draws (different concentrations)
    for alpha in (0.3, 0.7, 1.5):
        for _ in range(2):
            pool.append(_random_dirichlet(n, alpha, rng))

    # warm start and a few cyclic shifts
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ----------------------------------------------------------------------
    # 2. Quick evaluation, keep the elite candidates ------------------------
    top_k = 10
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        scored.append((val, a))
        # leave enough time for the optimisation phases
        if time.time() > deadline * 0.2:
            break
    scored.sort(key=lambda x: x[0])
    elite = [a for _, a in scored[:top_k]]

    best_seq = None
    best_val = float("inf")

    # ----------------------------------------------------------------------
    # 3. Phase 1 – smooth‑max gradient descent on the elite candidates -------
    for rank, a0 in enumerate(elite):
        if time.time() >= deadline - 5.0:
            break
        # allocate a decreasing share of the remaining budget
        remaining = deadline - time.time()
        share = remaining * (top_k - rank) / max(
            1, (top_k - rank) * (top_k - rank + 1) / 2
        )
        local_deadline = time.time() + share

        a_opt, val_opt = _smooth_gd_optimize(
            a0, n, conv_fft_len, corr_fft_len, local_deadline
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[refine #{rank}] new global best {val_opt:.6f}")

    # ----------------------------------------------------------------------
    # 4. Phase 2 – final cheap hill‑climber ----------------------------------
    if best_seq is not None and time.time() < deadline - 2.0:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[final local] best {val_loc:.6f}")

    # ----------------------------------------------------------------------
    # 5. Fallback ------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
