```
            elif warm_arr.shape[0] < n:
                warm_arr = np.pad(warm_arr, (0,n-warm_arr.shape[0]))
            warm_arr = np.maximum(warm_arr,0.0)
            if warm_arr.sum()>0:
                warm_arr = warm_arr / warm_arr.sum()
                pool.append(warm_arr.copy())
        # deterministic families:
        for gamma in (0.01,0.05,0.1,0.3,0.5,1.0,2.0):
            pool.append(_exp_decay_left(n,gamma))
            pool.append(_exp_decay_right(n,gamma))
        for alpha in (-8,-4,-2,-1,-0.5,-0.2):
            pool.append(_power_law_left(n,alpha))
            pool.append(_power_law_right(n,alpha))
        for a,b in ((0.2,1.0),(0.5,1.5),(2.0,0.5),(0.1,5.0),(5.0,0.1)):
            pool.append(_beta_shape(n,a,b))
        for k in (32,64,128,256):

        # uniform and sinusoid as well
        pool.append(_uniform(n))
        pool.append(_sinusoid(n))

            if s<1e-8: continue
            a /= s
            conv = _full_convolution(a, conv_len)
            val = 2.0 * n * conv.max()

                best_seq = a.copy()
                best_n = n
                best_conv_len = conv_len
                best_corr_len = _next_pow2(3*n-2)
```

The program searches for a non‑negative step‑function (a vector `a`) that minimises  

\[
F(a)=\frac{2\,n\;\max(a\ast a)}{\bigl(\sum a\bigr)^2}
\]

where `a∗a` is the full linear convolution of `a` with itself.  
The approach is a multi‑stage “global‑then‑local’’ optimisation:

1. **Diverse initialisation** – many deterministic families (exponential,
   power‑law, β‑pdf, sinusoid, raised‑cosine, …) together with random Dirichlet
   and sparse‑spike vectors are generated for several problem sizes
   (`n = 1024, 2048, 4096`).  The best few candidates are kept.

2. **Adam on a smooth‑max surrogate** – the non‑smooth `max` is replaced by a
   temperature‑controlled log‑sum‑exp.  The surrogate is minimised on the
   probability simplex by Adam while the temperature is increased
   exponentially.  Several Adam runs (different starts) are executed until the
   time budget is almost exhausted.

3. **Projected sub‑gradient descent on the exact max** – after the Adam
   phase a hard‑max sub‑gradient (`w` is a one‑hot at the maximal convolution
   index) is computed and the iterate is projected onto the simplex.
   This often yields a small but consistent improvement.

4. **Cheap random hill‑climber** – in the last seconds a very cheap
   mass‑move hill‑climber explores the neighbourhood of the current best
   solution.

All helper routines (`_full_convolution`, `_cross_correlation`,
`_project_onto_simplex`, the various shape generators, the optimiser
implementations) are defined at top level – no closures or lambdas are used.
The routine respects the 1000 s deadline, always returns the best sequence it
found, and prints progress information.

```python
"""
Multi‑stage optimiser for the step‑function score

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

It combines a large pool of deterministic families, several Adam runs on a
smooth‑max surrogate, a projected sub‑gradient descent on the exact max, and a
final cheap random hill‑climber.  The code works for any length `n` (default
4096) and respects the 1000 s time budget.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Shape families --------------------------------------------------------
def _exp_decay_left(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·i)  (mass concentrated on the left)."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _exp_decay_right(n: int, gamma: float) -> np.ndarray:
    """a_i ∝ exp(-γ·(n‑1‑i))  (mass concentrated on the right)."""
    shape = np.exp(-gamma * np.arange(n - 1, -1, -1, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _power_law_left(n: int, alpha: float) -> np.ndarray:
    """a_i ∝ (i+1)^α  (α<0 → decreasing)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = idx ** alpha
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _power_law_right(n: int, alpha: float) -> np.ndarray:
    """Mirror of _power_law_left."""
    shape = _power_law_left(n, alpha)[::-1]
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑pdf shape on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _sinusoid(n: int) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    return s


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    s = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    s = np.maximum(s, 0.0)
    s /= s.sum()
    return s


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration α."""
    return rng.dirichlet(np.full(n, alpha))


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k spikes of equal height, the rest zero."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Objective (fast) ------------------------------------------------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(conv) for a normalized (sum=1) vector."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max surrogate --------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.1          # inverse temperature (smooth‑max sharpness)
    lam_factor = 1.25  # increase temperature quickly

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        val = _objective(a, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (chain rule already inside _cross_correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when near deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent on the *exact* max ----------------------
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Hard‑max sub‑gradient descent with simplex projection."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    step0 = 0.5
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_val = conv.max()
        # one‑hot (or uniform over ties) weight vector
        w = np.zeros_like(conv)
        idxs = np.where(conv >= max_val - 1e-14)[0]
        w[idxs] = 1.0 / len(idxs)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = step0 / np.sqrt(it + 1)
        a = a - step * grad_a
        a = _project_onto_simplex(a)

        val = 2.0 * n * max_val
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad it={it}] improved to {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber ---------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Random mass moves accepted only if they improve the exact objective."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)  # move up to 30 % of mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget, prints progress and always returns
    the best sequence (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Choose several problem sizes – smaller n is cheaper but may give a
    # better discrete optimum.
    n_list = [1024, 2048, 4096]
    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Stage 1 – build a diverse pool of starters and keep the best few.
    for n in n_list:
        if time.time() > deadline - 120:   # keep ~2 min for deep work
            break

        conv_fft_len = _next_pow2(2 * n - 1)
        # pool of candidates for this n
        pool = []

        # warm start (if provided)
        warm = globals().get("height_sequence_1")
        if warm is not None:
            warm_arr = np.asarray(warm, dtype=np.float64)
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            elif warm_arr.shape[0] < n:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
            warm_arr = np.maximum(warm_arr, 0.0)
            if warm_arr.sum() > 0:
                warm_arr = warm_arr / warm_arr.sum()
                pool.append(warm_arr.copy())

        # deterministic families (both left‑ and right‑skewed)
        for gamma in (0.01, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0):
            pool.append(_exp_decay_left(n, gamma))
            pool.append(_exp_decay_right(n, gamma))

        for alpha in (-8.0, -4.0, -2.0, -1.0, -0.5, -0.2):
            pool.append(_power_law_left(n, alpha))
            pool.append(_power_law_right(n, alpha))

        beta_params = [
            (0.2, 5.0), (5.0, 0.2), (0.5, 2.0), (2.0, 0.5),
            (0.5, 0.5), (2.0, 2.0), (0.1, 0.1), (0.1, 5.0), (5.0, 0.1)
        ]
        for a, b in beta_params:
            pool.append(_beta_shape(n, a, b))

        # smooth primitives
        pool.append(_sinusoid(n))
        pool.append(_raised_cosine(n))
        pool.append(_uniform(n))

        # sparse random spikes
        for k in (32, 64, 128, 256):
            pool.append(_sparse_spikes(n, k, rng))

        # random Dirichlet draws
        for a in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
            pool.append(_random_dirichlet(n, a, rng))

        # ----------------------------------------------------------------
        # Quick scan of the pool
        rng.shuffle(pool)
        local_best_seq = None
        local_best_val = float("inf")
        for cand in pool:
            a = np.maximum(cand, 0.0)
            s = a.sum()
            if s < 1e-12:
                continue
            a /= s
            val = _objective(a, conv_fft_len)
            if val < local_best_val:
                local_best_val = val
                local_best_seq = a.copy()
                print(f"[Init n={n}] candidate value {val:.6f}")

            if time.time() > deadline - 120:
                break

        # keep the best for this size
        if local_best_seq is not None and local_best_val < global_best_val:
            global_best_val = local_best_val
            global_best_seq = local_best_seq.copy()
            best_n = n
            best_conv_len = conv_fft_len
            best_corr_len = _next_pow2(3 * n - 2)
            print(f"[Init n={n}] new global best {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – deep Adam optimisation on the current global best.
    if global_best_seq is not None and time.time() < deadline - 60:
        n = best_n
        a_adam, val_adam = _adam_optimize(
            global_best_seq,
            n,
            best_conv_len,
            best_corr_len,
            deadline,
            max_iters=300_000,
            lr_start=0.08,
        )
        if val_adam < global_best_val:
            global_best_val = val_adam
            global_best_seq = a_adam.copy()
            print(f"[Adam] improved global best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – projected sub‑gradient descent on the exact max.
    if global_best_seq is not None and time.time() < deadline - 30:
        n = best_n
        a_sub, val_sub = _subgradient_descent(
            global_best_seq,
            n,
            best_conv_len,
            best_corr_len,
            deadline,
            max_iters=200_000,
        )
        if val_sub < global_best_val:
            global_best_val = val_sub
            global_best_seq = a_sub.copy()
            print(f"[Subgradient] improved global best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – cheap random hill‑climber for the very last seconds.
    if global_best_seq is not None and time.time() < deadline - 3:
        n = best_n
        a_final, val_final = _local_random_search(
            global_best_seq,
            n,
            best_conv_len,
            deadline,
            max_moves=150_000,
        )
        if val_final < global_best_val:
            global_best_val = val_final
            global_best_seq = a_final.copy()
            print(f"[Local final] improved global best to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if global_best_seq is None:
        global_best_seq = _uniform(1024)

    # Clamp to the range accepted by the evaluator.
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)

    print(f"Finished – best value {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
```
