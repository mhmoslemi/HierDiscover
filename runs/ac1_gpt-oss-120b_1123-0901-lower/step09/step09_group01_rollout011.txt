Define proj_simplex:

def proj_simplex(v):
    """Project v onto probability simplex {x>=0, sum x = 1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function optimizer (improved multi‑phase search).

The routine executes the following stages within the given time budget:

1️⃣  Broad parametric sweep – many analytic families (power‑law,
    exponential, stretched‑exponential, beta, double‑Gaussian,
    monotone, spikes, …).  Each shape is normalised to sum = 1 and
    evaluated with an FFT‑based exact convolution.  The best vector
    found so far is kept.

2️⃣  Smooth‑max Adam – starting from the incumbent a soft‑max
    parametrisation is refined on the log‑sum‑exp surrogate of the
    max‑convolution.  The temperature of the surrogate is increased
    slowly, the Adam step‑size schedule is prolonged and the final
    vector is projected onto the probability simplex.

3️⃣  Euclidean mirror‑descent (projected sub‑gradient) – the true
    max‑convolution is used to build a sub‑gradient (active‑set
    weights).  The iterate is updated by a classic projected‑gradient
    step (gradient descent on the probability simplex).  This often
    beats exponentiated‑gradient on the same problem.

4️⃣  Random‑restart cheap sub‑gradient passes – a handful of random
    Dirichlet initialisations are run for a few thousand iterations;
    any improvement replaces the incumbent.

5️⃣  Simulated‑annealing fine‑tuning – local mass‑transfer moves are
    accepted according to a temperature schedule.

All phases stop as soon as the overall deadline (≈ 1000 s) is reached.
The function always returns a non‑negative list (the best sequence
encountered).  The implementation relies only on NumPy and SciPy
(available in the evaluation environment) and respects the “no
closures” rule.
"""

import time
import math
import numpy as np
from scipy.optimize import minimize_scalar

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a of length (2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (needed for sub‑gradients)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(a*a)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------

def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    # Implementation of the algorithm from:
    # Wang, Weiran, and Miguel A. Carreira‑Perpiñán. “Projection onto the probability simplex:
    # an efficient algorithm with a simple proof, and an application.” (2013)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non‑positive -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1  # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution on [0,1]."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes of equal mass (or Dirichlet)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False) if random else \
            np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_linear(n: int) -> np.ndarray:
    """Linearly decreasing from left to right (sum = 1)."""
    i = np.arange(n, dtype=np.float64)
    a = (n - i)
    a /= a.sum()
    return a


def _candidate_sine(n: int) -> np.ndarray:
    """Edge‑peaked sine (sin(π·i/(n‑1)) is zero at ends → invert)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(math.pi * (i + 1) / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -------------------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5          # start temperature
    lam_factor = 1.01  # slower heating → more accurate gradients

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max on logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent step (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (approaches max)
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_expgrad(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         deadline: float,
                         max_iters: int = 1_000_000,
                         eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent (mirror‑descent)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_euclidean(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 800_000,
                           eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution (Euclidean simplex)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Euclid it={it:06d}] new best {val:.6f}")

        # Active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a
        a = _proj_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing local search."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _refine_left_exponential(alpha_start: float, n: int) -> tuple[np.ndarray, float]:
    """Fine‑tune a left‑exponential shape using a 1‑D scalar optimiser."""
    def obj(alpha):
        a = _candidate_left_exponential(alpha, n)
        return _objective(a, n)
    res = minimize_scalar(obj, bounds=(alpha_start * 0.2, alpha_start * 5.0),
                          method='bounded', options={'xatol': 1e-5, 'maxiter': 50})
    a_opt = _candidate_left_exponential(res.x, n)
    return a_opt, _objective(a_opt, n)

# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  The algorithm follows the multi‑phase scheme
    described in the module docstring.  It always returns a list of
    floats that respect the evaluator's checks.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – keep the original setting (4096) for compatibility
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---------------------------------------------------------------
    # 1) Broad parametric sweep
    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray, tag: str = "Sweep"):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # power‑law (negative exponents)
    for p in np.linspace(-4.0, -0.1, 35):
        _maybe_update(_candidate_power(p, n), "Power")

    # symmetric exponential
    for alpha in np.logspace(-4, 0.8, 30):
        _maybe_update(_candidate_exponential(alpha, n), "ExpSym")

    # stretched exponential
    for alpha in np.logspace(-3, 0.5, 12):
        for beta in np.linspace(0.3, 3.0, 10):
            _maybe_update(_candidate_stretched(alpha, beta, n), "Stretched")

    # beta distributions
    for a_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for b_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
            _maybe_update(_candidate_beta(a_par, b_par, n), "Beta")

    # double Gaussian
    for sig in np.logspace(-2.5, -0.2, 18):
        _maybe_update(_candidate_double_gaussian(sig, n), "DGaus")

    # monotone left‑exponential (important family)
    left_exp_seq = None
    left_exp_val = None
    for alpha in np.logspace(-4, 1.0, 25):
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a, "LeftExp")
        # keep the best left‑exponential for fine tuning later
        if left_exp_val is None or _objective(a, n) < left_exp_val:
            left_exp_val = _objective(a, n)
            left_exp_seq = a

    # monotone left‑power
    for p in np.linspace(-3.0, 1.0, 30):
        _maybe_update(_candidate_left_power(p, n), "LeftPow")

    # linear decreasing
    _maybe_update(_candidate_linear(n), "Linear")

    # sine‑based edge shape
    _maybe_update(_candidate_sine(n), "Sine")

    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            _maybe_update(_candidate_spike(k, n, random=True), "Spike")

    # Warm‑start from a previous global variable if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "Warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ---------------------------------------------------------------
    # 1b) Optional local refinement of the best left‑exponential
    if left_exp_seq is not None and time.time() < deadline - 10:
        print("[Refine] fine‑tune left‑exponential via scalar optimisation")
        a_ref, val_ref = _refine_left_exponential(alpha_start=0.5, n=n)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] improved to {val_ref:.6f}")

    # ---------------------------------------------------------------
    # 2) Smooth‑max Adam (longer than the original)
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] extended Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 30.0)  # give ~30 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ---------------------------------------------------------------
    # 3) Euclidean projected sub‑gradient descent (core phase)
    if best_seq is not None and time.time() < deadline - 150:
        print("[Phase‑Euclid] projected sub‑gradient descent")
        euclid_deadline = min(deadline - 5, time.time() + 120.0)  # about 2 min
        a_eu, val_eu = _subgradient_euclidean(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            euclid_deadline,
            max_iters=3_000_000,
            eta0=0.9,
        )
        if val_eu < best_val:
            best_val = val_eu
            best_seq = a_eu.copy()
            print(f"[Phase‑Euclid] improved to {val_eu:.6f}")

    # ---------------------------------------------------------------
    # 4) Random‑restart cheap sub‑gradient passes (exploration)
    rest_deadline = deadline - 30.0
    while time.time() < rest_deadline:
        # Random Dirichlet initialisation with a random concentration
        conc = np.exp(np.random.uniform(np.log(0.05), np.log(5.0)))
        a0 = np.random.dirichlet(np.full(n, conc))
        # Run a very short Euclidean sub‑gradient (≈ 2000 iters)
        short_dead = time.time() + 4.0
        a_tmp, val_tmp = _subgradient_euclidean(
            a0, n, conv_fft_len, corr_fft_len, short_dead,
            max_iters=2000, eta0=0.9
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Restart] new best {val_tmp:.6f}")

    # ---------------------------------------------------------------
    # 5) Optional exponentiated‑gradient fine‑tuning (if time left)
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑ExpGrad] final exponentiated‑gradient pass")
        exp_deadline = deadline - 5.0
        a_exp, val_exp = _subgradient_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            exp_deadline,
            max_iters=2_000_000,
            eta0=0.9,
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Phase‑ExpGrad] improved to {val_exp:.6f}")

    # ---------------------------------------------------------------
    # 6) Simulated‑annealing fine‑tuning (last few seconds)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase‑Anneal] final simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ---------------------------------------------------------------
    # Final sanity check and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
