# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (len = 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for (sub‑)gradient of Σ w·(a∗a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate  log‑sum‑exp( λ·(a∗a) ).
    Returns the best vector (normalized to sum 1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior of the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.06
    lam_factor = 1.017

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule for smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny shake when we approach the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 800_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true objective.
    The iterates stay on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(a0: np.ndarray,
                                   n: int,
                                   conv_fft_len: int,
                                   deadline: float,
                                   max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Targeted annealing: identify the most offending (i,j) pair that
    contributes to the peak of the convolution and move a small fraction
    of its mass elsewhere.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.07
    T_end = 5e-6
    move = 0

    while time.time() < deadline and move < max_moves:
        # linear temperature decay
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # Determine the pair (i,j) with i+j=kmax that maximises a_i*a_j
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        # we move a small fraction (5 %) of the weaker of the two entries
        if i_src == j_src:
            # degenerate case – move from that index to a random other one
            if a2[i_src] < 1e-14:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        # ensure feasibility
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of starting vectors."""
    cand = []

    # warm start and cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws of varying concentration
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edge profiles
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # U‑shaped (large mass at both ends)
    eps = 1e-4
    shape = 1.0 / (np.arange(n) + eps) + 1.0 / (np.arange(n, 0, -1) + eps)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes the evaluation
    score.  The function respects the given time budget and always returns
    the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    #  Primary problem size
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution is stored globally)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diversified pool of starting vectors
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – fast Adam exploration (≈30 % of budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + (deadline - time.time()) * 0.30
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Very short Adam run (≈1–2 s)
        sub_dead = min(explore_deadline,
                       time.time() + 1.8)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=70_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improved to {val_adam:.6f}")

        # Brief sub‑gradient refinement (≈0.8 s)
        if time.time() < explore_deadline:
            a_sub, val_sub = _subgradient_optimize(
                best_seq, n, conv_fft_len, corr_fft_len,
                explore_deadline,
                max_iters=9_000,
                eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep sub‑gradient descent (≈40 % of budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Long sub‑gradient phase")
        a_ref, val_ref = _subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Subgrad long best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – pair‑wise targeted annealing (≈20 % of budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[PairAnneal] Targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[PairAnneal] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – optional up‑sampling to a larger grid (if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() + 35 < deadline:
        print("[Upsample] Refining on n=8192")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # Linear interpolation to larger grid
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # Short Adam on larger grid
        sub_dead_big = min(deadline, time.time() + 5.0)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=200_000,
        )

        # Longer sub‑gradient refinement on larger grid
        a_big_ref, val_big_ref = _subgradient_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big
            print(f"[Upsample] refined best {val_big_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase V – random local perturbations (final polishing)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Perturb] Final random tweaks")
        for _ in range(8):
            # tiny Gaussian noise, re‑project onto simplex
            noise = np.random.normal(scale=1e-3, size=best_seq.shape)
            a_tmp = best_seq + noise
            a_tmp = np.maximum(a_tmp, 0.0)
            a_tmp = _project_simplex(a_tmp)
            tmp_val = evaluate_sequence(a_tmp.tolist())
            if tmp_val < best_val:
                best_val = tmp_val
                best_seq = a_tmp.copy()
                print(f"[Perturb] improved to {tmp_val:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(4096, 1.0 / 4096, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```

This implementation expands on the previous hybrid optimizer:

* **Larger search space** – an up‑sample to 8192 points is attempted when time permits.
* **More aggressive iteration budgets** – Adam, sub‑gradient and annealing phases run longer.
* **Additional diverse initial families**, notably a “U‑shaped’’ profile that concentrates mass at the ends.
* **Final random perturbations** to escape any remaining shallow local minima.

All operations stay inside the given time budget, respect the non‑negative constraint, and return the best sequence found.
