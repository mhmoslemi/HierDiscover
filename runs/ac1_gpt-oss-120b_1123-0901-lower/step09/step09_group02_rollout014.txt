def propose_candidate(seed=0, budget_s=1000):
    set random seed
    n=4096
    conv_len = _next_pow2(2*n-1)
    deadline = time.time()+budget_s-0.5
    best_seq = None; best_val = inf

    # generate candidate families
    cand = []
    # families: dirichlet, geometric decreasing, power law, random permutations
    for p in [0.5, 0.8, 1.0, 1.2, 1.5]:
        a = np.arange(1, n+1) ** (-p)
        a /= a.sum()
        cand.append(a)
    for gamma in [0.2, 0.5, 1.0, 2.0]:
        a = np.exp(-gamma * np.arange(n))
        a /= a.sum()
        cand.append(a)
    # uniform
    cand.append(np.full(n, 1.0/n))
    # random permutations of each
    for a in list(cand):
        perm = np.random.permutation(n)
        cand.append(a[perm])
    # random Dirichlet draws
    for alpha in [0.2, 0.5, 1.0, 2.0, 5.0]:
        cand.append(np.random.dirichlet(alpha*np.ones(n)))
    # include warm start if exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=float)
        if w.shape[0] != n:
            if w.shape[0] > n: w = w[:n]
            else: w = np.pad(w,(0,n-w.shape[0]))
        w = np.maximum(w,0)
        w /= w.sum()
        cand.append(w)

    # evaluate each candidate quickly using objective (full conv)
    for a in cand:
        val = _evaluate_objective(a, n, conv_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
    print(f"Initial best {best_val}")

    # main annealing loop: greedy random modifications
    a = best_seq.copy()
    conv = _full_convolution(a, conv_len)
    max_b = conv.max()
    best_val = 2.0*n*max_b/(a.sum()**2)
    best_seq = a.copy()
    # loop until deadline
    it = 0
    while time.time() < deadline:
        # temperature schedule based on progress
        # use simple linear decay from T_max to T_min over some iterations
        # we can compute fraction = (deadline - time.time()) / (deadline - start)
        # but easier: using decreasing schedule with iteration count.
        T = max(1e-5, 0.1 * (1 - it/1e6))
        # choose move type randomly
        if np.random.rand() < 0.1:
            # swap move
            i,j = np.random.randint(n, size=2)
            if i==j: continue
            a2 = a.copy()
            a2[i],a2[j] = a2[j],a2[i]
        else:
            # mass transfer move focusing on contribution reduction
            # pick index i weighted by a[i]*c_i where c_i is sum of a[j] for j such that i+j=idx_max?
            idx_max = conv.argmax()
            # compute contributions array w_i = a_i * a_{idx_max - i} if within bounds else 0
            i_candidates = np.arange(max(0, idx_max - (n-1)), min(n-1, idx_max)+1)
            contribs = a[i_candidates] * a[idx_max - i_candidates]
            if contribs.sum()==0:
                i = np.random.randint(n)
            else:
                # sample i proportionally to contribs
                probs = contribs / contribs.sum()
                i = np.random.choice(i_candidates, p=probs)
            # pick j uniformly random far away from i (maybe to reduce peak)
            j = np.random.randint(n)
            # ensure i!=j
            if i==j:
                continue
            # move fraction delta from i to j
            frac = np.random.rand()*0.5  # up to 50%
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        # evaluate candidate
        conv2 = _full_convolution(a2, conv_len)
        max_b2 = conv2.max()
        val2 = 2.0*n*max_b2/(a2.sum()**2)

        # accept if improvement or with probability exp(-(val2-val)/T)
        if val2 < best_val:
            best_val = val2
            best_seq = a2.copy()
            a = a2
            conv = conv2
        else:
            if np.random.rand() < np.exp(-(val2-best_val)/T):
                a = a2
                conv = conv2

        it+=1
        if it % 50000 == 0:
            print(f"[Iter {it}] best {best_val}")

    # final clip
    best_seq = np.clip(best_seq,0,1000)
    print(f"Final best {best_val}")
    return best_seq.tolist()

"""
stepfunction_optimizer_v3.py

A time‑bounded multi‑phase optimiser for

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0 .

Main ideas
---------

* Produce a *richer* initial pool – in addition to the families used before we
  add a pure left‑hand exponential and several power‑law decreasing shapes.
  (Monotone decreasing vectors create very little overlap when they are
  convolved with a reversed copy, which is the dominant term of the objective.)

* Phase I – quick filtering of the pool with a short Adam run on a
  log‑soft‑max surrogate (smooth‑max).  This leaves only the most promising
  candidates.

* Phase II – a long exponentiated‑gradient sub‑gradient optimisation on the
  exact max (no smoothing).  The step‑size schedule is deliberately slow so
  that the algorithm keeps moving even after thousands of iterations.

* Phase III – aggressive simulated‑annealing with two move types (mass
  transfer and swaps).  The temperature schedule is extended and the number
  of moves is substantially larger than in the previous version.

* Phase IV – pure permutation hill‑climbing (swap‑only) to polish the final
  vector.

The routine respects a user‑supplied time budget (default 1000 s) and always
returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (output length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w★a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns the best vector
    (always kept normalised to a simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # tiny jitter near the end of the budget
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max exponentiated‑gradient sub‑gradient optimisation.
    Operates on the simplex (∑a=1, a≥0).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set at the current max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # Exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # Re‑normalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search with mass‑transfer and swaps.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.06
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.45   # up to 45 % transfer
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            # uphill move with Metropolis probability
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        it += 1
        if it % 30_000 == 0:
            print(f"[Anneal it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Permutation‑only hill‑climbing (swap moves only).
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # tiny chance to accept uphill move (helps escape flat plateaus)
            if np.random.rand() < 0.009:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]

        it += 1
        if it % 20_000 == 0:
            print(f"[Permute it={it:06d}] best {best_val:.6f}")

    return best_a, best_val

# Candidate generation
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified starting pool (adds monotone decreasing families)."""
    cand = []

    # Warm start (if available)
    if warm is not None:
        cand.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Left‑only exponential families (no mirroring)
    for gamma in (0.15, 0.30, 0.45, 0.60):
        shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law decreasing shapes a_i ∝ (i+1)^(-p)
    for p in (0.5, 0.8, 1.0, 1.2, 1.5):
        idx = np.arange(1, n + 1, dtype=np.float64)
        shape = idx ** (-p)
        shape /= shape.sum()
        cand.append(shape)

    # Central Gaussian (single peak)
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.25):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Simple triangular shape (rise‑to‑centre, fall)
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Interleaved permutation of every candidate – spreads large entries
    # uniformly across the index range (helps to decorrelate a and rev(a)).
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------

    warm = globals().get("height_sequence_1")

        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build the initial pool
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm)

            a0 = np.random.dirichlet(np.ones(n))

        sub_dead = min(phase1_deadline, time.time() + 3.0)   # a few seconds per candidate
        a_adam, val_adam = _adam_optimize(

    # ------------------------------------------------------------------

        a_ann, val_ann = _anneal_optimize(
            best_seq,

            conv_fft_len,
