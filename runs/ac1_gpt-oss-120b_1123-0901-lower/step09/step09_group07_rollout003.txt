"""
Search for a low‑score non‑negative step function.

Phase 1 – Very wide random sampling from many parametric families
(edge exponential, edge power‑law, beta, double‑Gaussian, sinusoid power,
custom two‑power, convex combo of exponential+uniform, Dirichlet).
The best candidate is kept.

Phase 2 – Long Adam optimisation on a smooth‑max surrogate of the
max‑convolution.  The gradient of the surrogate is computed exactly by an
FFT‑based cross‑correlation, the variables are kept on the probability
simplex via a soft‑max (mirrored‑descent) parametrisation.

Phase 3 – A cheap stochastic hill‑climber (mass‑transfer moves) finishes the
last seconds.

The best vector is stored in the global name ``height_sequence_1`` for warm‑
starting future runs.  All numbers are clipped to the range accepted by
``evaluate_sequence`` (≥0, ≤1000) before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the Adam gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to the clamping step)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Parametric shape generators
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0.0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _custom_power_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = np.power(i, alpha) * np.power(n - i + 1.0, beta)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _combo_exp_uniform_shape(n: int, lam: float, gamma: float) -> np.ndarray:
    """lam∈[0,1] mixes a symmetric edge‑exponential with the uniform."""
    exp_shape = _edge_exponential_shape(n, gamma, weight=0.5)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    shape = lam * exp_shape + (1.0 - lam) * uniform
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02            # temperature (inverse)
    lam_factor = 1.015    # slower increase → more stable later

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → point on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max over convolution entries
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (2·(w ★ a))
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        # tiny random jitter when we are close to the deadline
        if it % 3000 == 0 and time.time() > deadline - 30.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple stochastic local search (mass transfer)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue
        # move a random fraction (up to 30 %) of mass from i to j
        delta = a[i] * (rng.random() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Find a non‑negative step function (list of heights) that minimises

        2·n·max(a∗a) / (sum(a)²)

    within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem size – 4096 gives a good compromise between flexibility and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run stored a vector)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – exhaustive parametric sweep (≈40 % of the budget)
    families = [
        "edge_exp",
        "edge_pow",
        "beta",
        "double_gauss",
        "sin_pow",
        "tent",
        "dirichlet",
        "custom_pow",
        "combo_exp_uni",
    ]

    phase1_deadline = min(deadline - 5.0, time.time() + 0.40 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10.0 ** rng.uniform(-3.0, 3.0)          # 1e‑3 … 1e3
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-15.0, -0.01)              # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10.0 ** rng.uniform(-3.0, 2.0)             # 0.001 … 100
            b = 10.0 ** rng.uniform(-3.0, 2.0)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.001, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.05, 10.0)                # allow <1
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.001, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        elif fam == "custom_pow":
            alpha = rng.uniform(-10.0, 10.0)
            beta = rng.uniform(-10.0, 10.0)
            cand = _custom_power_shape(n, alpha, beta)
        else:   # combo_exp_uni
            lam = rng.random()
            gamma = 10.0 ** rng.uniform(-3.0, 3.0)
            cand = _combo_exp_uniform_shape(n, lam, gamma)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # Include warm‑start shifts (they sometimes improve the bound)
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement (use the remaining time, leaving a small margin)
    if best_seq is not None and time.time() < deadline - 10.0:
        print("Phase 2 – Adam optimisation")
        # Give Adam the whole rest of the budget except the final few seconds
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline - 5.0,
            max_iters=800_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing for the final seconds
    if best_seq is not None and time.time() < deadline - 1.0:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")

    # store for warm‑starting the next invocation
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
