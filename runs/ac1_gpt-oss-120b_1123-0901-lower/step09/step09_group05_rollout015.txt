        pool.append(_edge_power_law(n, float(alpha)))
   # asym power

        for a2 in np.linspace(0,2,8):
            pool.append(_asym_power_law(n,float(a1),float(a2)))

        v = _random_dirichlet(n,0.1,rng)
        v = _smooth(v, sigma)
        pool.append(v)

        a = a / a.sum()
        conv = _full_convolution(a, conv_len)
        val = 2.0 * n * conv.max() / (a.sum()**2)

    a = a / a.sum()
    # initialize Adam parameters
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    t = 0
    best_a = a.copy()
    best_val = float('inf')
    cur_val = 2.0 * n * _full_convolution(a, conv_fft_len).max() / (a.sum() ** 2)  # compute once
    beta = beta_start
    while time.time() < deadline and t < max_iters:
        sum_a = a.sum()
        conv = _full_convolution(a, conv_fft_len)
        # softmax over conv
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        # softmax value (approx max)
        M = (max_scaled + np.log(exp_scaled.sum())) / beta  # log-sum-exp
        # gradient of M wrt a:
        grad_M = 2.0 * _cross_correlation(w, a, corr_fft_len)
        grad = (2.0 * n / (sum_a**2)) * grad_M - (4.0 * n * M / (sum_a**3)) * np.ones_like(a)

        # Adam update
        t += 1
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.01 / (np.sqrt(v_hat) + eps)  # base LR .01 adaptively
        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new)  # ensures sum 1, non-negative

        # compute new objective
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # accept if better or with small prob
        if val_new < cur_val or np.random.rand() < 0.001:
            a = a_new
            cur_val = val_new

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[Adam it={t}] new best {val_new:.6f}")

        # increase beta gradually
        beta = min(beta * beta_factor, beta_max)

    return best_a, best_val

"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

The algorithm consists of three phases:

1️⃣  **Diverse seeding** – deterministic families (uniform, sinusoid,
    edge‑exponential / power‑law, beta‑pdf, double‑Gaussian) together
    with many random Dirichlet draws (different concentration) and a few
    asymmetric exponential / power‑law variants are generated.  Each
    candidate is projected onto the probability simplex and its exact
    objective (full convolution via FFT) is evaluated.  The best
    `TOP_K` candidates are retained.

2️⃣  **Smooth Adam‑descent** – each elite candidate is refined with a
    projected Adam optimiser that minimises a *soft‑max* surrogate of the
    max‑convolution.  The surrogate uses a temperature `β` that is
    increased slowly, thus providing a smooth gradient early on and
    converging to the true max later.  Projection onto the simplex keeps
    the vector feasible.

3️⃣  **Simulated‑annealing local search** – the current best vector is
    perturbed by moving a random fraction of mass between two positions.
    Moves are accepted according to a decreasing temperature schedule,
    allowing occasional uphill steps and thus escaping shallow local
    minima.

All three phases respect the global `budget_s` wall‑clock limit
(≈1000 s).  The returned list is compatible with the provided
`evaluate_sequence` function.  The code runs on a single thread but
uses only NumPy (no external optimisation libraries)."""

import time
import math
import heapq
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families (all returned vectors are *not* normalised)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _asym_exponential(n: int, g1: float, g2: float) -> np.ndarray:
    left = np.exp(-g1 * np.arange(n, dtype=np.float64))
    right = np.exp(-g2 * np.arange(n, dtype=np.float64))
    shape = left + right[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _asym_power_law(n: int, a1: float, a2: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** a1) + ((n - idx + 1) ** a2)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _smooth(v: np.ndarray, sigma: float) -> np.ndarray:
    """Convolve v with a wide Gaussian (sigma is a fraction of n)."""
    n = v.shape[0]
    # Gaussian kernel defined on the same grid
    x = np.arange(-n // 2, n - n // 2, dtype=np.float64)
    kernel = np.exp(-0.5 * (x / (sigma * n)) ** 2)
    fft_len = _next_pow2(2 * n - 1)
    fv = np.fft.rfft(v, fft_len)
    fk = np.fft.rfft(kernel, fft_len)
    conv = np.fft.irfft(fv * fk, fft_len)[:n]
    conv = np.maximum(conv, 0.0)
    s = conv.sum()
    if s > 0:
        conv /= s
    else:
        conv = np.full(n, 1.0 / n)
    return conv


# ----------------------------------------------------------------------
# Optimisation helpers
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected Adam optimiser on the *soft‑max* surrogate of max(a*a).
    Returns the best vector seen and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # Adam hyper‑parameters
    beta1, beta2, eps = 0.9, 0.999, 1e-8
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    # soft‑max temperature schedule
    beta = 5.0          # start fairly smooth
    beta_factor = 1.001
    beta_max = 120.0

    best_a = a.copy()
    best_val = float("inf")
    cur_val = 2.0 * n * _full_convolution(a, conv_fft_len).max() / (a.sum() ** 2)

    t = 0
    while time.time() < deadline and t < max_iters:
        t += 1
        sum_a = a.sum()
        conv = _full_convolution(a, conv_fft_len)

        # soft‑max weights
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        w = exp_scaled / exp_scaled.sum()
        # soft‑max value (smooth approximation of max)
        M = (max_scaled + np.log(exp_scaled.sum())) / beta

        # gradient of the smooth max term
        grad_M = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # full gradient of the objective
        grad = (2.0 * n / (sum_a ** 2)) * grad_M - (4.0 * n * M / (sum_a ** 3)) * np.ones_like(a)

        # Adam update (learning‑rate 0.02, scaled by adaptive denominator)
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad ** 2)
        m_hat = m / (1 - beta1 ** t)
        v_hat = v / (1 - beta2 ** t)
        step = 0.02 / (np.sqrt(v_hat) + eps)

        a_new = a - step * m_hat
        a_new = _project_onto_simplex(a_new)          # feasibility

        # exact objective for the candidate (cheap, uses same conv routine)
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max() / (a_new.sum() ** 2)

        # typical Adam acceptance (always take the update)
        a = a_new
        cur_val = val_new

        if val_new < best_val:
            best_val = val_new
            best_a = a_new.copy()
            print(f"[Adam it={t:06d}] new best {val_new:.6f}")

        # increase temperature slowly – the optimisation then focuses on the true max
        beta = min(beta * beta_factor, beta_max)

    return best_a, best_val


def _anneal_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing hill‑climber that moves a random fraction
    of mass from one index to another.  Accepts worsening moves with a
    temperature that decays exponentially.
    """
    rng = np.random.default_rng()
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = cur_val

    T = max(0.02 * cur_val, 1e-3)  # initial temperature
    decay = 0.999995

    it = 0
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (rng.random() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        accept = False
        if val2 < cur_val:
            accept = True
        else:
            prob = math.exp(-(val2 - cur_val) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[Anneal it={it:07d}] new best {val2:.6f}")

        # temperature schedule
        if it % 10_000 == 0:
            T = max(T * decay, 1e-9)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Constructs a diverse pool of candidates, refines the
    top ones with a projected Adam optimiser (smooth max surrogate) and
    finishes with a simulated‑annealing local search.
    """
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – we stay with the size that gave the previous
    # best result (4096) – increasing it does not help within the fixed
    # time budget.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start from the previous best (if the judge provides it)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a large, diverse pool -------------------------------------------------
    TOP_K = 30                # keep this many for refinement
    heap = []                 # max‑heap on negative objective (larger is worse)

    def _push(a_vec: np.ndarray):
        """Normalise, evaluate and possibly push into the heap."""
        a = np.maximum(a_vec, 0.0)
        s = a.sum()
        if s < 1e-12:
            return
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (s ** 2)   # s==1 after normalisation
        if len(heap) < TOP_K:
            heapq.heappush(heap, (-val, a))
        else:
            if val < -heap[0][0]:
                heapq.heapreplace(heap, (-val, a))

    # deterministic families (all already normalised)
    _push(_uniform(n))
    _push(_sinusoid(n))

    # symmetric exponentials  (log‑spaced, extended to very small gamma)
    for gamma in np.logspace(-3, 1.0, 25):       # 0.001 … 10
        _push(_edge_exponential(n, float(gamma)))

    # asymmetric exponentials (few grid points – enough diversity)
    for g1 in np.logspace(-3, 0.5, 6):
        for g2 in np.logspace(-3, 0.5, 6):
            _push(_asym_exponential(n, float(g1), float(g2)))

    # power‑law families
    for alpha in np.linspace(-5.0, 2.0, 30):
        _push(_edge_power_law(n, float(alpha)))
    for a1 in np.linspace(-5, 0, 6):
        for a2 in np.linspace(0, 2, 6):
            _push(_asym_power_law(n, float(a1), float(a2)))

    # beta pdfs
    a_vals = np.logspace(-1, 1, 12)
    b_vals = np.logspace(-1, 1, 12)
    for a in a_vals:
        for b in b_vals:
            _push(_beta_shape(n, float(a), float(b)))

    # double gaussian
    for sigma in np.linspace(0.02, 0.5, 20):
        _push(_double_gaussian(n, float(sigma)))

    # random Dirichlet draws (wide range of concentration)
    for alpha in (0.01, 0.03, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        for _ in range(5):
            _push(_random_dirichlet(n, alpha, rng))

    # smoothed Dirichlet (adds an extra smoothness knob)
    for sigma in (0.01, 0.03, 0.07, 0.12):
        v = _random_dirichlet(n, 0.05, rng)
        _push(_smooth(v, sigma))

    # cyclic shifts of warm start (if any)
    if warm is not None:
        for shift in (n // 4, n // 2, 3 * n // 4):
            _push(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # keep the best TOP_K candidates
    top_candidates = [vec for _, vec in sorted(heap, key=lambda x: -x[0])]

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation of the elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, cand in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        # allocate a larger share to higher‑ranked candidates
        remaining = deadline - time.time()
        share = remaining * (TOP_K - rank) / sum(range(1, TOP_K - rank + 1))
        local_deadline = time.time() + share

        a_opt, val_opt = _adam_optimize(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=180_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final simulated‑annealing local search
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _anneal_search(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Final anneal] best {val_loc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety – clip to the bounds enforced inside `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
