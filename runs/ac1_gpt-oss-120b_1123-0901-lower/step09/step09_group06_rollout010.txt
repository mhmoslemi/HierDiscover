def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

"""
Propose a non‑negative step‑function (list of heights) that minimises

        F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in four stages:

1.  **Diverse initialisation** – many deterministic families (exponential,
    power‑law, β‑pdf, double‑Gaussian, sinusoid, raised‑cosine, semicircle,
    linear‑decrease) together with random Dirichlet vectors and sparse spikes.
    All candidates are normalised to sum = 1 and the best few are kept.

2.  **Exact sub‑gradient descent** – for each of the top candidates we minimise
    the true objective (no log‑sum‑exp smoothing).  At each iteration the
    convolution is computed, the index of the maximal entry is identified and
    the sub‑gradient of that entry is taken:
        grad = 2·(w★a)  with w a one‑hot at the maximal index.
    After a step the vector is projected onto the probability simplex.
    A simple learning‑rate schedule and early‑stopping are used.

3.  **Continuation L‑BFGS on a smooth‑max surrogate** – the best sequence from
    stage 2 is refined with the same L‑BFGS continuation used in the reference
    code (higher temperatures, a few hundred iterations each).

4.  **Cheap random hill‑climber** – a final local search moves a random
    fraction of mass between two positions whenever the exact objective
    improves.

All computation respects the 1000 s wall‑time budget and the routine always
returns the best sequence found (as a plain Python list).  No closures or
lambda‑expressions are used – every helper is defined at module top‑level.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Basic linear‑algebra / FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (gradient of a∗a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Families of candidate shapes
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _linear_decrease(n: int) -> np.ndarray:
    """Deterministic shape linear decreasing from the left edge to zero."""
    shape = np.arange(n, 0, -1, dtype=np.float64)
    shape = shape / shape.sum()
    return shape


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


# ----------------------------------------------------------------------
# Objective helper (exact, no clipping)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a∗a) / (sum(a)²).  a is assumed non‑negative."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

# Exact sub‑gradient descent (no smoothing)
def _subgradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 20000,
    init_lr: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the true objective.
    Returns the best encountered sequence and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() == 0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    lr = init_lr
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_idx = int(conv.argmax())

        # one‑hot weight vector at the maximal index
        w = np.zeros_like(conv)
        w[max_idx] = 1.0

        # sub‑gradient of the maximal convolution entry
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient step + simplex projection
        a = a - lr * grad
        a = _project_onto_simplex(a)

        # evaluate periodically
        if it % 250 == 0:
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Subgrad it={it:05d}] new best {val:.6f}")
            # simple learning‑rate adaptation
            if val > best_val:
                lr *= 0.9
            else:
                lr *= 1.01
            lr = max(min(lr, 2.0), 1e-5)

        it += 1

    return best_a, best_val

# Continuation L‑BFGS (the same idea as the reference implementation)
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Smooth‑max (log‑sum‑exp) surrogate and its gradient."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c
    exp_alpha_v = np.exp(alpha * v)
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    w = exp_alpha_v / sum_exp
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0, 80000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len, corr_fft_len, alpha)

        remaining = deadline - time.time()
        max_iter = 200 if remaining > 30 else 70

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val

# Cheap random hill‑climber (unchanged from reference)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns a plain Python list; the routine always respects the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (height_sequence_1, if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Build a diverse pool of initial candidates
    pool: List[np.ndarray] = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families
    for gamma in np.logspace(-3, 2, 30):
        pool.append(_edge_exponential(n, gamma))
    for alpha in np.linspace(-10, -0.1, 30):
        pool.append(_edge_power_law(n, alpha))
    for a_par, b_par in [(0.3, 0.3), (0.5, 0.5), (0.8, 0.8), (1.2, 1.2),
                         (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a_par, b_par))
    for sigma in (0.01, 0.02, 0.05, 0.1, 0.2):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))
    pool.append(_linear_decrease(n))

    for k in (64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    for alpha in np.logspace(-2, 1, 15):
        pool.append(_random_dirichlet(n, alpha))

    # a few pure random vectors
    for _ in range(10):
        pool.append(np.random.rand(n))

    # --------------------------------------------------------------
    # Quick evaluation of all candidates, keep the best few
    best_val = float("inf")
    best_seq: np.ndarray | None = None
    evaluated: List[tuple[float, np.ndarray]] = []

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        if time.time() > deadline - 120:  # reserve time for deep phases
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = evaluate_sequence(a.tolist())
        evaluated.append((val, a.copy()))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

    # --------------------------------------------------------------
    # Keep top K candidates for sub‑gradient refinement
    evaluated.sort(key=lambda x: x[0])
    top_k = 20
    top_candidates = [a for _, a in evaluated[:top_k]]

    # --------------------------------------------------------------
    # Stage 2 – exact sub‑gradient descent on the true objective
    for idx, cand in enumerate(top_candidates):
        if time.time() > deadline - 60:
            break
        a_opt, val_opt = _subgradient_descent(
            cand,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=25000,
            init_lr=0.5,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Subgrad refine {idx}] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – continuation L‑BFGS on smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 20:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS final] new best {val_lbgfs:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – cheap random hill‑climber for the final seconds
    if best_seq is not None and time.time() < deadline - 3:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
