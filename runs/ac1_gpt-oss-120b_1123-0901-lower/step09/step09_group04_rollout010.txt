def _pgd_momentum(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=500000, eta0=1.5, beta=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time()<deadline and it<max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2*n*max_c/(a.sum()**2)
        if val < best_val:
            best_val=val
            best_a = a.copy()
        # compute subgradient
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # momentum
        v = beta*v + (1-beta)*grad  # or classic: v = beta*v + grad
        # step size schedule
        eta = eta0 / (1 + 0.00001*it) # slowly decreasing
        a = a - eta * v
        # projection onto simplex
        a = _project_onto_simplex(a)
        # maybe jitter near end
        if time.time()>deadline-20 and it%5000==0:
            a += (np.random.rand(n)-0.5) * 1e-3
            a = np.maximum(a,0)
            a /= a.sum()
        it+=1
    return best_a, best_val

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

"""
Propose a low‑value step‑function for the given evaluation.

Algorithm
---------
1. **Diverse warm‑starts** – generate a large pool of deterministic families
   (edge‑exponential, edge‑power‑law, Beta, double‑Gaussian, sinusoid,
   uniform) with a wide range of parameters plus many random Dirichlet draws.
   Each candidate is normalised to sum‑one and the best (lowest exact
   objective) is kept as a seed.

2. **Adam on a smooth‑max surrogate** – starting from the seed we run a
   log‑sum‑exp (smooth‑max) optimiser for a short, time‑aware interval.
   The optimiser works on the log‑simplex (soft‑max) representation and
   uses Adam with a slowly increasing temperature.

3. **Projected‑gradient descent with momentum** – we then perform many
   iterations of sub‑gradient descent on the exact max‑convolution.
   Momentum (heavy‑ball) and a slowly decaying step size accelerate
   convergence.  Projection onto the probability simplex guarantees feasibility.

4. **Simulated‑annealing hill‑climber** – finally a cheap SA loop makes
   random mass‑transfer moves (source→target) and accepts them with a
   temperature schedule.  This often squeezes a few extra digits.

All phases respect the global time budget (`budget_s`).  The routine always
keeps the best vector seen and returns it as a plain Python list.  The vector
contains non‑negative numbers and complies with the limits imposed by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Candidate families
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (i+1)^α + (n-i)^α."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
#  Exact objective (same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best vector (sum==1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # start on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # soft‑max temperature (inverse)
    lam_factor = 1.02   # temperature schedule

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a ∈ simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake when close to deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_momentum(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.2,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with heavy‑ball momentum on the exact
    max‑convolution objective.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = _random_dirichlet(n, 1.0)
    a = a / a.sum()
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD‑M it={it:06d}] new best {val:.6f}")

        # active set for the max
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum update
        v = beta * v + (1.0 - beta) * grad

        # slowly decaying step size
        eta = eta0 / (1.0 + it * 1e-4)

        a = a - eta * v
        a = _project_onto_simplex(a)

        # occasional jitter near the end to escape tiny plateaus
        if it % 10000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a = a / a.sum()

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap SA hill‑climber: random mass‑transfer moves are accepted
    according to a simple temperature schedule.
    """
    a = a0.copy()
    val = _objective(a)
    best_a = a.copy()
    best_val = val

    it = 0
    T0 = 0.02  # initial temperature

    while time.time() < deadline and it < max_moves:
        # linear cooling
        T = T0 * max(0.0, 1 - it / max_moves)

        i = np.random.randint(n)
        if a[i] < 1e-12:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * np.random.rand() * 0.3

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val_new = _objective(a_new)
        diff = val_new - val

        # Metropolis acceptance
        if diff < 0 or np.random.rand() < np.exp(-diff / max(T, 1e-12)):
            a = a_new
            val = val_new
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                if it % 5000 == 0:
                    print(f"[SA move {it:06d}] improved to {val_new:.6f}")

        it += 1

    return best_a, best_val

        pool.append(_edge_power_law(n, alpha))

    for a, b in [(0.3, 0.3), (0.5, 0.5), (0.7, 0.7), (1.0, 1.0),
                 (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # random Dirichlet draws with several concentrations
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0, 10.0):
        for _ in range(10):
            pool.append(_random_dirichlet(n, alpha))

    # incorporate the optional warm start and its cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # Quick scan of the pool
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = _random_dirichlet(n, 1.0)
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max (short burst)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.20 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Projected‑gradient descent with momentum (main work)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        a_pgd, val_pgd = _pgd_momentum(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=1_000_000,
            eta0=1.2,
            beta=0.9,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD‑M] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Simulated‑annealing final hill‑climber
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback safety
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the range enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
