def propose_candidate(seed=0,budget_s=1000):
    np.random.seed(seed)
    deadline = time.time()+budget_s-0.5
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    # build basis
    basis = _build_basis(n)
    
    best_seq = None
    best_val = float('inf')
    
    # Phase 0: random mixtures
    phase_end = time.time() + (deadline - time.time()) * 0.3   # 30% time for random search
    while time.time() < phase_end:
        # choose random number of components
        k = np.random.randint(2,6)  # 2-5 components
        idxs = np.random.choice(len(basis), size=k, replace=False)
        # Dirichlet weights
        weights = np.random.dirichlet(np.ones(k))
        a = np.zeros(n)
        for w, i in zip(weights, idxs):
            a += w * basis[i]
        # Normalize to sum 1
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random mix] new best {val:.6f}")
    
    # Phase 1: subgradient refinement
    if best_seq is not None:
        ref_end = deadline - 200  # keep time for anneal maybe 200 sec left for final phases
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, _next_pow2(3*n-2), ref_end,
            max_iters=800_000, eta0=0.9)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] improved to {val_ref:.6f}")
    
    # Phase 2: annealing mass transfer and swaps (long)
    if time.time() < deadline - 10:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=120_000, swap_prob=0.15)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")
    
    # Phase 3: final permutation hill climbing
    if time.time() < deadline - 5:
        a_perm, val_perm = _permute_hill_climb(
            best_seq, n, conv_fft_len, deadline,
            max_swaps=50_000)
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] improved to {val_perm:.6f}")
    
    # final clip and return list
    if best_seq is None:
        best_seq = np.full(n, 1.0/n)
    
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best {best_val:.6f} length {len(best_seq)}")
    return best_seq.tolist()

"""
Step‑function optimizer (≈1000 s budget)

The algorithm builds a rich dictionary of elementary non‑negative shapes
(Uniform, Gaussians, Beta‑type, triangular, sinusoid, sparse spikes, …).
A random convex combination of a few of those shapes is sampled many
times; the best mixture is kept as a warm‑start.  Afterwards the
candidate is refined by three deterministic phases:

1. exponentiated‑gradient sub‑gradient descent on the true max‑convolution
2. simulated‑annealing with mass‑transfer and swaps
3. permutation‑only hill climbing

All phases respect the global deadline and the routine always returns
the best sequence discovered (as a plain ``list`` of floats).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution helpers (fast and exact)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using real FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the exact‑max sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a)/(∑a)² (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Shape dictionary
# ----------------------------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _central_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
    shape_sum = shape.sum()
    return shape / shape_sum


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (x / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((x - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape_sum = shape.sum()
    return shape / shape_sum


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on the grid (mid‑point rule avoids zeros)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
    s = shape.sum()
    if s <= 0 or np.isnan(s):
        return None
    return shape / s


def _triangular(n: int) -> np.ndarray:
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n)).astype(np.float64)
    return tri / tri.sum()


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    return shape / shape.sum()


def _spike_pattern(n: int, k: int) -> np.ndarray:
    """k equally spaced spikes, each receiving 1/k of the total mass."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    a[pos] = 1.0 / k
    return a


def _random_sparse_spike(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    pos = np.random.choice(n, size=k, replace=False)
    masses = np.random.dirichlet(np.ones(k))
    a[pos] = masses
    return a


def _build_basis(n: int) -> list[np.ndarray]:
    """Create a list with a wide variety of non‑negative shapes."""
    basis = []

    # 1. Uniform
    basis.append(_uniform(n))

    # 2. Central Gaussians (different σ)
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        basis.append(_central_gaussian(n, sigma))

    # 3. Double‑Gaussians (peaks at both ends)
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        basis.append(_double_gaussian(n, sigma))

    # 4. Symmetric Beta‑type (α from very peaked to very flat, also U‑shaped)
    for alpha in (0.3, 0.5, 0.7, 0.9, 1.0, 1.2, 1.5, 2.0, 3.0, 4.0, 5.0):
        b = _beta_shape(n, alpha)
        if b is not None:
            basis.append(b)

    # 5. Simple triangular shape
    basis.append(_triangular(n))

    # 6. Positive half‑wave sinusoid
    basis.append(_sinusoid(n))

    # 7. Sparse spike families (different sparsities)
    for k in (16, 32, 64, 128, 256, 512):
        basis.append(_spike_pattern(n, k))
    for k in (32, 64, 128):
        for _ in range(2):
            basis.append(_random_sparse_spike(n, k))

    # 8. Interleaved permutation of each primitive (helps to spread mass)
    interleaved = []
    for a in basis:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    basis.extend(interleaved)

    # Normalise all once (avoid accidental zero‑sum shapes)
    normed = []
    for a in basis:
        s = a.sum()
        if s > 0:
            normed.append(a / s)
    return normed


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the exact‑max surrogate.
    Operates on the probability simplex (∑a = 1, a ≥ 0).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # a little progress report every 5·10⁴ iters
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # --- active set (indices attaining the max) ----------
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()          # uniform weight on active entries

        # gradient of the true max (exact sub‑gradient)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative EG update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise to the simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search.
    Two move types: (i) mass transfer, (ii) swap of two entries.
    Temperature decays linearly.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ---- swap move ----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ---- mass‑transfer move ----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # uphill acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation hill‑climbing (swap moves only). Accept only
    improving swaps, but allow a tiny probability of uphill moves
    to escape flat plateaus.
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            if np.random.rand() < 0.01:   # occasional uphill acceptance
                val = val2
            else:
                a[i], a[j] = a[j], a[i]   # revert

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimizes
    ``evaluate_sequence``.  The strategy is:

    1. Create a dictionary of primitive shapes.
    2. Randomly blend a few shapes many times and keep the best mixture.
    3. Refine the best mixture with exponentiated‑gradient sub‑gradient descent.
    4. Further improve with simulated‑annealing (mass transfer + swaps).
    5. Finish with a permutation‑only hill‑climb.
    The routine obeys the supplied time budget and always returns the best
    sequence found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 points gives a good trade‑off between quality
    # and speed (the constant in the objective makes the absolute size
    # largely irrelevant).
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build the library of elementary shapes
    # ------------------------------------------------------------------
    basis = _build_basis(n)

    # ------------------------------------------------------------------
    # Warm‑start from a previously returned solution (if any)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] != n:
            if warm_arr.shape[0] > n:
                warm_arr = warm_arr[:n]
            else:
                warm_arr = np.pad(warm_arr, (0, n - warm_arr.shape[0]))
        warm_arr = np.clip(warm_arr, 0.0, 1000.0)
        s = warm_arr.sum()
        if s >= 0.01:
            warm_arr = warm_arr / s
            basis.append(warm_arr)          # give the warm start a place in the pool

    # ------------------------------------------------------------------
    # Phase A – random convex combinations of a few primitives
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    phase_a_deadline = time.time() + (deadline - time.time()) * 0.30  # ~30 % of budget

    while time.time() < phase_a_deadline:
        # randomly pick 2‑5 different primitives
        k = np.random.randint(2, 6)
        idxs = np.random.choice(len(basis), size=k, replace=False)
        weights = np.random.dirichlet(np.ones(k))   # convex coefficients
        a = np.zeros(n, dtype=np.float64)
        for w, idx in zip(weights, idxs):
            a += w * basis[idx]

        s = a.sum()
        if s < 1e-12:
            continue
        a /= s                                    # enforce sum = 1

        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random mix] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – exponentiated‑gradient sub‑gradient refinement
    # ------------------------------------------------------------------
    if best_seq is not None:
        # keep some slack for the remaining phases
        refine_deadline = deadline - 250.0
        if refine_deadline > time.time():
            a_ref, val_ref = _subgradient_optimize_expgrad(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                refine_deadline,
                max_iters=1_500_000,
                eta0=0.9,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Refine] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase C – simulated annealing (mass transfer + swaps)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 70:
        print("[Anneal] starting simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
            swap_prob=0.14,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase D – permutation‑only hill climbing
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Permute] starting permutation hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=100_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] improved to {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep it safe
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Final sanitisation – the evaluator will clip again, but we do it
    # early to stay within its limits.
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
