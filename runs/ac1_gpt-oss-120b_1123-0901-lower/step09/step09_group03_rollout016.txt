"""
Propose a step‑function sequence that minimises
    F(a) = 2·n·max(a∗a) / (∑a)²
for the given evaluator.

Approach
--------
1.  Build a small, carefully chosen family of analytic shapes:
        – uniform
        – ``((i+1)(n‑i))^{-p}``  (U‑shaped, p∈[0.1,2])
        – `exp(-γ·i)+exp(-γ·(n‑1‑i))` (edge‑exponential, γ∈[1e‑3,0.8])
   Each shape is mixed with the uniform distribution:
        a = α·uniform + (1‑α)·shape      (α∈[0,1]).

   A dense grid over (p,α) and (γ,α) is evaluated (≈ 4 000 candidates)
   using a fast FFT based convolution; the best candidate is kept.

2.  Starting from that best candidate, run a long
    exponentiated‑gradient (mirror‑descent) sub‑gradient optimisation
    on the exact non‑smooth objective.  The iteration count is bounded
    only by the remaining time budget.

3.  Return the best vector seen (as a plain list).  All entries are
    non‑negative and automatically respect the evaluator’s clipping
    limits.

The whole routine respects the 1000 s wall‑time limit and never
returns ``None`` – a uniform vector is used as a final fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (result length = 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) / (∑a)².
    The vector `a` is assumed to be non‑negative; it is normalised
    only for numerical safety.
    """
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact objective.
    The iterate stays on the probability simplex.  The function returns
    the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set: indices where convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of the (exact) max‑conv term
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule η = η₀ / √t
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine performs a dense grid search over a few
    analytically‑motivated families and then refines the best candidate
    with a long sub‑gradient optimisation.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – a moderate size that still gives a tight bound
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1.  Analytic families + uniform mixture
    uniform = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = None
    best_val = float("inf")

    # optional warm start from a previous solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]), "constant")
        w = np.maximum(w, 0.0)
        if w.sum() > 0.0:
            w = w / w.sum()
            val = _objective(w, n, conv_fft_len)
            best_seq, best_val = w, val
            print(f"[Warm] initial value {val:.6f}")

    # parameter grids
    alphas = np.linspace(0.0, 1.0, 41)               # α (weight of uniform)
    p_vals = np.linspace(0.10, 2.00, 40)            # power‑law exponent
    gamma_vals = np.logspace(-3, -0.1, 40)          # exponential decay

    idx = np.arange(n, dtype=np.float64)

    # ----- power‑law (U‑shaped) family -----
    for p in p_vals:
        # shape proportional to ((i+1)*(n-i))^{-p}
        w = ((idx + 1.0) * (n - idx)) ** (-p)
        w_sum = w.sum()
        if w_sum == 0.0:
            continue
        w_norm = w / w_sum
        for alpha in alphas:
            a = alpha * uniform + (1.0 - alpha) * w_norm
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, a.copy()
        # occasional time‑check to avoid overrunning the budget
        if time.time() > deadline:
            break

    # ----- exponential edge family -----
    for gamma in gamma_vals:
        w = np.exp(-gamma * idx) + np.exp(-gamma * (n - 1 - idx))
        w_norm = w / w.sum()
        for alpha in alphas:
            a = alpha * uniform + (1.0 - alpha) * w_norm
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, a.copy()
        if time.time() > deadline:
            break

    # ----- mixed families (optional extra diversity) -----
    # (a few extra random Dirichlet draws just in case)
    for _ in range(12):
        a = np.random.dirichlet(np.ones(n))
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a.copy()
        if time.time() > deadline:
            break

    print(f"[Grid] best value after analytic search {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2.  Long sub‑gradient refinement (exponentiated gradient)
    if best_seq is not None and time.time() < deadline - 2.0:
        # Ensure strictly positive entries for the exp‑gradient updates
        best_seq = np.maximum(best_seq, 1e-12)
        best_seq = best_seq / best_seq.sum()

        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=1.0,
        )
        if val_ref < best_val:
            best_val, best_seq = val_ref, a_ref.copy()
            print(f"[Subgrad] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final safety fallback
    if best_seq is None:
        best_seq = uniform.copy()
        best_val = _objective(best_seq, n, conv_fft_len)

    # Clip to the evaluator’s accepted range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
