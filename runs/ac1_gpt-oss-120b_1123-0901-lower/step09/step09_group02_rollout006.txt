def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 8192   # possibly double
    # compute FFT lengths
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # adjust size
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            # pad by periodic tiling or zeros
            pad = n - warm.shape[0]
            warm = np.concatenate([warm, np.zeros(pad)])
        warm = np.maximum(warm, 0.0)
        if warm.sum() < 0.01:
            warm = None
        else:
            warm = warm / warm.sum()
    else:
        warm = None

    # generate initial candidates
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq = None
    best_val = float('inf')
    # Phase I: quick eval of candidates with a few Adam steps
    # allocate maybe 0.1 of time
    explore_deadline = time.time() + (deadline - time.time()) * 0.15
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        sub_dead = min(explore_deadline, time.time() + 2.0)
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=50000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore Adam] new best {val_adam:.6f}")
    # If pool empty early, we can generate more random candidates on the fly
    if best_seq is None:
        best_seq = np.random.dirichlet(np.ones(n))
        best_seq = best_seq / best_seq.sum()

    # Phase II: long subgradient on incumbent
    if time.time() < deadline - 30:
        print("[Refine] Subgradient starting")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] new best {val_ref:.6f}")

    # Phase III: simulated annealing (mass+swap) with many moves
    if time.time() < deadline - 15:
        print("[Anneal] Simulated annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200000,
            swap_prob=0.15,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] new best {val_ann:.6f}")

    # Phase IV: permutation hill climb
    if time.time() < deadline - 5:
        print("[Permute] Hill climbing")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=60000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] new best {val_perm:.6f}")

    # Final adjustments: normalize? Not needed; evaluate_sequence is scale invariant.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} on length {n}")
    return best_seq.tolist()

"""
stepfunction_optimizer_v3.py

A time‑bounded multi‑phase optimizer that aims to lower

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

The main new ideas compared with the previous version are:

* **Bigger problem size** – we work with n = 8192 (instead of 4096).  The larger
  search space gives more freedom to flatten the self‑convolution and lets the
  constant in front of 1/n drop well below the 0.75 that was achieved before.

* **More aggressive refinement** – the sub‑gradient (exponentiated‑gradient)
  phase is run for many more iterations (up to a few‑hundred‑thousand) and
  the learning‑rate schedule is slightly slower, which allows a finer approach
  to the true optimum.

* **Longer simulated‑annealing** – the annealing phase now performs up to
  200 k moves (mass transfer + swaps) with a slower temperature decay, giving
  the optimizer more time to escape plateaus.

* **Additional random starts** – beyond the sophisticated families that were
  used before, a large pool of Dirichlet draws with many concentration levels
  is generated on‑the‑fly, increasing the chance of discovering a better basin.

* **Robust handling of a warm start** – if the variable `height_sequence_1`
  exists it is resized to the current length and used as one of the initial
  candidates.

All helpers are defined at top level, there are no closures or lambdas, and the
routine respects the user‑supplied time budget (default 1000 s).  The function
`propose_candidate` returns the best sequence found as a plain Python list.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Core FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective value (no clipping).  Used only for bookkeeping.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector seen (normalised to sum≈1) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # objective
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny jitter near deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase with exponentiated‑gradient updates.
    Works on the probability simplex (a normalised to sum=1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # sum is 1 ⇒ 2n*max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
    swap_prob: float = 0.15,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with two move types:
        * mass transfer (as before)
        * swap of two coordinates (probability = swap_prob)
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.35
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation hill‑climbing: repeatedly swap two entries,
    keep the swap only if it does not increase the objective (tiny chance
    to accept uphill moves).
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            if np.random.rand() < 0.01:   # occasional uphill acceptance
                val = val2
            else:
                a[i], a[j] = a[j], a[i]   # revert swap

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of start vectors."""
    cand = []

    # warm start and its rotations
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with varied concentration
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Double‑Gaussian (peaks at both ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.2, 0.4):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Central Gaussian
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric Beta‑type shapes (α = β > 1)
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n)).astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Positive half‑wave sinusoid
    sin_shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sin_shape = np.maximum(sin_shape, 0.0)
    sin_shape /= sin_shape.sum()
    cand.append(sin_shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Interleaved permutations (helps to spread large entries)
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns (equally spaced or random)."""
    out = []
    for k in (16, 32, 48, 64, 96, 128, 192, 256, 384, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    # random‑position sparse spikes
    for k in (64, 128, 192, 256):
        for _ in range(4):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget and always returns the best
    sequence it found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – larger n gives more freedom to flatten the convolution
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            # pad with zeros to reach length n
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build an initial pool of candidate vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick exploration of many candidates (≈15 % of budget)
    explore_deadline = time.time() + (deadline - time.time()) * 0.15
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)

        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # a few short Adam runs (fast surrogate optimisation)
        sub_dead = min(explore_deadline, time.time() + 2.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore Adam] new best {val_adam:.6f}")

        # occasional random Dirichlet start to keep diversity
        if time.time() < explore_deadline and np.random.rand() < 0.07:
            a_rand = np.random.dirichlet(np.random.rand(n) + 0.01)
            a_rand = a_rand / a_rand.sum()
            a_rand, val_rand = _adam_optimize(
                a_rand, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=40_000
            )
            if val_rand < best_val:
                best_val = val_rand
                best_seq = a_rand.copy()
                print(f"[Explore Rand] new best {val_rand:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient sub‑gradient optimisation
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Starting long sub‑gradient optimisation")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] new best after sub‑grad → {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – simulated‑annealing with mass‑transfer & swaps
    if best_seq is not None and time.time() < deadline - 15:
        print("[Anneal] Simulated‑annealing phase")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
            swap_prob=0.15,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – permutation‑only hill‑climbing
    if best_seq is not None and time.time() < deadline - 5:
        print("[Permute] Permutation hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=80_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] new best after permutation → {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but keep a safe default
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the provided evaluator (optional logging)
    try:
        final_obj = evaluate_sequence(best_seq.tolist())
        print(f"Final reported value by evaluate_sequence: {final_obj:.6f}")
    except Exception as e:
        print("Evaluation failed:", e)

    print(f"Final best internal value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
