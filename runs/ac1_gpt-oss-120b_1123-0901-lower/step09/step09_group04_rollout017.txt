def _targeted_sa(a0, n, conv_fft_len, deadline, max_moves=200_000, move_frac=0.3):

    best_a = a.copy()
    best_val = _objective(a)

    T0 = 1e-2  # starting temperature for acceptance of worse moves
    # choose schedule decaying to near zero at end
    total_time = deadline - time.time()
    start = time.time()
    it = 0
    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # get shift(s) where max occurs
        ks = np.where(np.isclose(conv, max_c, atol=1e-15))[0]
        k = ks[np.random.randint(len(ks))]  # pick one
        # compute contributions
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        idx = np.arange(start_i, end_i + 1)
        contrib = a[idx] * a[k - idx]
        total_contrib = contrib.sum()
        if total_contrib == 0:
            # choose random move
            i = np.random.randint(n)
        else:
            prob = contrib / total_contrib
            i = np.random.choice(idx, p=prob)
        # choose partner
        j = k - i
        # decide source index (larger of i,j)
        if a[i] >= a[j]:
            src = i
        else:
            src = j
        # choose destination index far from src (optionally random)
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)
        # move a fraction of src mass
        delta = a[src] * (np.random.rand() * move_frac)
        if delta <= 0:
            it += 1
            continue
        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta
        # compute objective
        val = _objective(a_new)
        # temperature schedule linear decreasing
        elapsed = time.time() - start

        if accept:
            a = a_new
            if val < best_val:
                best_val = val
                best_a = a.copy()
        it += 1

"""
Simple yet effective search for a low‑value step‑function.

The algorithm follows three stages
    1.   Build a large, diverse pool of deterministic families
         (uniform, sinusoid, left‑exponential, left‑power‑law, beta)
         plus many Dirichlet draws, evaluate them and keep the best.
    2.   Refine the best seed with a few hundred thousand projected‑gradient
         steps (exact max‑convolution, projection onto the probability simplex).
    3.   Targeted simulated‑annealing / hill‑climbing:
         repeatedly identify the shift that attains the current maximum
         convolution, move a small amount of mass from a highly‑contributing
         pair to a random location and accept the move if it improves the
         objective (or with a temperature‑controlled probability).

All heavy work uses FFT‑based convolutions on a power‑of‑two length.
The routine always returns the best sequence found before the time budget
expires and respects the non‑negative, bounded requirements of
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (mostly deterministic families)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with scalar concentration α."""
    return np.random.dirichlet(np.full(n, alpha))


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    """Monotone decreasing exponential (mass concentrated on the left)."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _left_power_law(n: int, power: float) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^(-p)."""
    idx = np.arange(1, n + 1, dtype=np.float64)
    shape = idx ** (-power)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact evaluation 2·n·max(conv) / (sum(a)^2)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Projected‑gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max (within a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad
        a = _project_onto_simplex(a)

        # occasional jitter towards the end of the budget
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Targeted simulated‑annealing / hill‑climber
# ----------------------------------------------------------------------


def _targeted_sa(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
    move_frac: float = 0.25,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climber that repeatedly attacks the currently active shift
    (the one that gives the maximal convolution value).  A small
    amount of mass is moved from a highly contributing index towards a
    random location; worse moves are accepted with a temperature that
    linearly cools to zero.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    start = time.time()
    total = deadline - start
    T0 = 1e-2                     # initial temperature for uphill moves
    it = 0

    while time.time() < deadline and it < max_moves:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # pick a shift attaining the max (if many, choose randomly)
        ks = np.where(np.isclose(conv, max_c, atol=1e-15))[0]
        k = ks[np.random.randint(len(ks))]

        # compute contributions a[i] * a[k-i] for the active shift
        start_i = max(0, k - (n - 1))
        end_i = min(n - 1, k)
        idx = np.arange(start_i, end_i + 1)
        contrib = a[idx] * a[k - idx]
        total_contrib = contrib.sum()

        if total_contrib == 0.0:
            # fall back to a completely random move
            src = np.random.randint(n)
        else:
            prob = contrib / total_contrib
            src = np.random.choice(idx, p=prob)

        # partner index involved in the same product
        partner = k - src
        # move mass from the larger of the two
        src = src if a[src] >= a[partner] else partner

        # pick a destination far enough (any other index works)
        dst = np.random.randint(n)
        while dst == src:
            dst = np.random.randint(n)

        delta = a[src] * (np.random.rand() * move_frac)
        if delta <= 0.0:
            it += 1
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        new_val = _objective(a_new)

        # simple temperature schedule (linear decay)
        elapsed = time.time() - start
        T = T0 * max(0.0, 1.0 - elapsed / total)

        accept = False
        if new_val < best_val:
            accept = True
        elif T > 0.0 and np.random.rand() < np.exp(-(new_val - best_val) / T):
            accept = True

        if accept:
            a = a_new
            if new_val < best_val:
                best_val = new_val
                best_a = a.copy()
                print(f"[SA move {it:06d}] new best {new_val:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size (feel free to change, larger n gives finer resolution)
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start – if a previous solution is present in the globals
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a rich initial pool of candidates
    # ------------------------------------------------------------------
    pool = []

    # deterministic families
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0):
        pool.append(_left_exponential(n, gamma))
    for p in (0.3, 0.6, 1.0, 2.0):
        pool.append(_left_power_law(n, p))
    for a, b in [(0.3, 1.0), (0.5, 2.0), (2.0, 0.5), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # a bunch of Dirichlet draws with varied concentrations
    for alpha in (0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        for _ in range(12):
            pool.append(_random_dirichlet(n, alpha))

    # incorporate the optional warm start and a few cyclic shifts of it
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            shifted = np.roll(warm, shift)
            pool.append(shifted)

    # ------------------------------------------------------------------
    # Quick scan of the pool
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid:03d}] best {val:.6f}")

    # If something went terribly wrong, fall back to uniform
    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------
    # Phase 2 – short projected‑gradient descent (≈20% of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        pgd_deadline = time.time() + 0.20 * (deadline - time.time())
        a_pg, v_pg = _pgd_optimize(
            best_seq,
            n,
            fft_len,
            pgd_deadline,
            max_iters=600_000,
            eta0=0.9,
        )
        if v_pg < best_val:
            best_val, best_seq = v_pg, a_pg.copy()
            print(f"[PGD phase] improved to {v_pg:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – targeted simulated‑annealing / hill‑climber (rest of time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        a_sa, v_sa = _targeted_sa(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=250_000,
            move_frac=0.20,
        )
        if v_sa < best_val:
            best_val, best_seq = v_sa, a_sa.copy()
            print(f"[SA phase] improved to {v_sa:.6f}")

    # ------------------------------------------------------------------
    # Final cheap random jitter (last couple of seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        jitter = (np.random.rand(n) - 0.5) * 1e-4
        best_seq = np.maximum(best_seq + jitter, 0.0)
        best_seq /= best_seq.sum()

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
