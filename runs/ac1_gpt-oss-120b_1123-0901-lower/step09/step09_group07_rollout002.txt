"""
Propose a non‑negative step function with a low value of
    F(a)=2·n·max(a∗a)/(sum(a)²)

The algorithm improves on the reference implementation by:

* 1️⃣  A richer, time‑aware random‑parameter sweep (many families, extreme
     weights, pure monotone shapes) that keeps the *top‑k* candidates.
* 2️⃣  Light gradient refinements of every top‑k candidate (short Adam runs)
     to escape poor local minima.
* 3️⃣  A full‑scale refinement of the current best (Adam → sub‑gradient → PGD)
     with a slightly more aggressive learning‑rate schedule.
* 4️⃣  A final simulated‑annealing hill‑climber that is allowed to accept
     occasional uphill moves, helping to get past the last local plateau.
* 5️⃣  Support for an optional warm‑start via the global variable
     ``height_sequence_1`` and multiple independent restarts if time permits.

All heavy work uses FFT‑based convolutions (O(n log n)).  The whole search
fits into the 1000 s budget; intermediate progress is printed so the
runner can see improvements.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence up to clamping) ------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Parametric shape generators ------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _monotone_step_shape(n: int, mass: float = 0.5) -> np.ndarray:
    """A pure monotone decreasing step: first floor(mass·n) entries equal."""
    k = max(1, int(np.round(mass * n)))
    shape = np.concatenate([np.full(k, 1.0 / k), np.zeros(n - k)], dtype=np.float64)
    return shape


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.015  # smoother rise than the reference

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (exact)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (soft‑max Jacobian)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random jitter near the end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # LR decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (accepts uphill moves) -------------------------
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
    init_frac: float = 0.02,
    cooling: float = 0.9998,
) -> tuple[np.ndarray, float]:
    """Simple SA that moves a random fraction of mass from i → j."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = a.copy()
    best_val = cur_val

    # temperature proportional to current value
    T = init_frac * cur_val

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 40 %) of the source mass
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        dval = val2 - cur_val
        if dval < 0 or np.random.rand() < np.exp(-dval / T):
            a = a2
            cur_val = val2
            if cur_val < best_val:
                best_val = cur_val
                best_a = a.copy()
        # geometric cooling
        T *= cooling
        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimizes
    `evaluate_sequence`.  The routine proceeds through
        1. aggressive random‑parameter sampling,
        2. brief Adam refinements of the K best candidates,
        3. a full‑scale Adam → sub‑gradient → PGD refinement,
        4. a simulated‑annealing hill‑climber.
    Multiple independent restarts are performed if time permits.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 works well for the time budget
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # try to fetch a warm start from the previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase‑0: keep the global best across possible restarts
    global_best_seq = None
    global_best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper: evaluate a candidate and update the global best
    def _maybe_update(seq: np.ndarray, val: float):
        nonlocal global_best_seq, global_best_val
        if val < global_best_val:
            global_best_val = val
            global_best_seq = seq.copy()
            print(f"[Global] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Main restart loop – each restart consumes a slice of the budget
    while time.time() < deadline - 20.0:   # leave ~20 s for the final SA
        restart_start = time.time()
        # ---- Phase 1: random parametric sweep ---------------------------------
        top_k = 6
        best_cands = []   # list of (val, seq) sorted ascending

        families = [
            "edge_exp", "edge_pow", "beta", "double_gauss",
            "sin_pow", "tent", "dirichlet", "monotone_step",
        ]

        # allocate about 15 % of the remaining time to this sweep
        phase1_deadline = min(deadline - 5.0,
                              time.time() + 0.15 * (deadline - time.time()))
        print("Phase 1 – random parametric sampling")
        while time.time() < phase1_deadline:
            fam = rng.choice(families)
            if fam == "edge_exp":
                gamma = 10 ** rng.uniform(-2, 3)   # broadened range
                weight = rng.choice([0.0, 1.0, rng.random()])  # allow pure monotone
                cand = _edge_exponential_shape(n, gamma, weight)
            elif fam == "edge_pow":
                alpha = rng.uniform(-12.0, -0.05)  # stronger negative tail
                weight = rng.choice([0.0, 1.0, rng.random()])
                cand = _edge_power_law_shape(n, alpha, weight)
            elif fam == "beta":
                a = 10 ** rng.uniform(-2, 1)
                b = 10 ** rng.uniform(-2, 1)
                cand = _beta_shape(n, a, b)
            elif fam == "double_gauss":
                sigma = rng.uniform(0.005, 0.6)
                cand = _double_gaussian_shape(n, sigma)
            elif fam == "sin_pow":
                power = rng.uniform(0.2, 8.0)
                cand = _sinusoid_power_shape(n, power)
            elif fam == "tent":
                cand = _tent_shape(n)
            elif fam == "dirichlet":
                alpha = rng.uniform(0.02, 8.0)
                cand = _random_dirichlet_shape(n, alpha, rng)
            else:  # monotone_step
                mass = rng.uniform(0.01, 0.8)
                cand = _monotone_step_shape(n, mass)

            # warm‑start shifts are handled later
            val = _objective(cand, conv_fft_len)
            if len(best_cands) < top_k:
                best_cands.append((val, cand.copy()))
                best_cands.sort(key=lambda x: x[0])
            elif val < best_cands[-1][0]:
                best_cands[-1] = (val, cand.copy())
                best_cands.sort(key=lambda x: x[0])

        # incorporate optional warm‑start (and a few cyclic shifts)
        if warm is not None:
            for shift in (0, n // 4, n // 2, 3 * n // 4):
                cand = np.roll(warm, shift)
                val = _objective(cand, conv_fft_len)
                if len(best_cands) < top_k:
                    best_cands.append((val, cand.copy()))
                    best_cands.sort(key=lambda x: x[0])
                elif val < best_cands[-1][0]:
                    best_cands[-1] = (val, cand.copy())
                    best_cands.sort(key=lambda x: x[0])

        # ---- Phase 2a: cheap Adam refinements of the top‑k candidates ---------
        # each gets a short budget (≈30 k iters) to move away from poor basins
        print("Phase 2a – brief Adam refinements of top‑k")
        for val, cand in best_cands:
            sub_deadline = deadline - 10.0
            if time.time() > sub_deadline:
                break
            a_ref, val_ref = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=120_000,
                lr_start=0.08,
            )
            _maybe_update(a_ref, val_ref)

        # ---- Phase 2b: deep refinement of the current global best -------------
        if global_best_seq is not None and time.time() < deadline - 8.0:
            print("Phase 2b – deep Adam → sub‑gradient → PGD")
            # Adam
            a_adam, val_adam = _adam_optimize(
                global_best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=260_000,
                lr_start=0.06,
            )
            _maybe_update(a_adam, val_adam)

            # Sub‑gradient (if time permits)
            if time.time() < deadline - 5.0:
                a_sub, val_sub = _subgradient_optimize(
                    global_best_seq,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=180_000,
                    eta0=0.9,
                )
                _maybe_update(a_sub, val_sub)

            # PGD (if still time)
            if time.time() < deadline - 3.0:
                a_pgd, val_pgd = _pgd_optimize(
                    global_best_seq,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    deadline,
                    max_iters=130_000,
                    eta0=0.38,
                )
                _maybe_update(a_pgd, val_pgd)

        # ------------------------------------------------------------------
        # Phase 3 – simulated‑annealing hill‑climber (only the final restart)
        # we let the *last* restart run a longer SA if time is left
        if time.time() > deadline - 20.0:
            break   # exit the restart loop; we will do a final SA afterwards

        # otherwise start a new restart
        print(f"[Restart] completed in {time.time() - restart_start:.1f}s, "
              f"global best {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # Final Phase 3 – SA on the best sequence found so far
    if global_best_seq is not None and time.time() < deadline - 2.0:
        print("Final phase – simulated annealing")
        a_final, val_final = _simulated_annealing(
            global_best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
            init_frac=0.03,
            cooling=0.9997,
        )
        _maybe_update(a_final, val_final)

    # ------------------------------------------------------------------
    # Fallback to a simple uniform distribution if nothing was found
    if global_best_seq is None:
        global_best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        global_best_val = _objective(global_best_seq, conv_fft_len)

    # final safety clipping (matches evaluate_sequence limits)
    global_best_seq = np.clip(global_best_seq, 0.0, 1000.0)

    print(f"Returned best value {global_best_val:.6f} (length {len(global_best_seq)})")
    return global_best_seq.tolist()
