"""
propose_candidate
-----------------

A multi‑phase, heavily diversified optimisation for the step‑function problem:

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

The code works with a fixed length `n = 4096`.  It proceeds as follows:

1. **Rich initial pool** – deterministic families (uniform, Gaussian, Beta,
   triangular, sinusoid, double‑Gaussian) together with all pairwise convex
   mixtures of those families (weights on a 0…1 grid).  Each family is also
   offered in an “interleaved” permutation which spreads large entries over
   the vector.  A warm start (`height_sequence_1`) is also injected.

2. **Fast enumeration** – every candidate is evaluated exactly (FFT based
   convolution).  The best shape so far is retained.

3. **Long sub‑gradient refinement** – exponentiated‑gradient updates on the
   probability simplex using the exact sub‑gradient of the max‑convolution.
   The loop runs until the time budget is exhausted (or a safety limit).

4. **Simulated annealing** – mass‑transfer and swap moves with a geometric
   temperature schedule to escape shallow local minima.

5. **Permutation‑only hill climbing** – many random swaps, accepting only
   non‑worsening moves (tiny uphill probability) to fine‑tune the ordering.

6. **Post‑processing** – clipping to the evaluator’s allowed range
   `[0,1000]` and conversion to a plain Python list.

The routine respects the user‑supplied time budget (default 1000 s) and
always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation required for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Shape generators (all return a *probability* vector that sums to 1)
# ----------------------------------------------------------------------
def _uniform_shape(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _central_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
    if shape.sum() == 0:
        shape = np.full_like(shape, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _beta_shape(n: int, alpha: float) -> np.ndarray:
    """Symmetric Beta(α,α) on {1,…,n}."""
    x = np.arange(1, n + 1, dtype=np.float64) / n
    shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    if shape.sum() == 0:
        shape = np.full_like(shape, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _triangular_shape(n: int) -> np.ndarray:
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    shape = tri.astype(np.float64)
    shape /= shape.sum()
    return shape


def _sinusoid_shape(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Two symmetric peaks at both ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (i / (sigma * n)) ** 2)
    right = left[::-1]
    shape = left + right
    shape /= shape.sum()
    return shape


def _interleaved_perm(a: np.ndarray) -> np.ndarray:
    """Spread large coefficients by an even‑odd interleaving."""
    n = a.shape[0]
    perm = (np.arange(n) * 2) % n
    b = np.empty_like(a)
    b[perm] = a
    return b


def _mix_shapes(a: np.ndarray, b: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w*a + (1-w)*b, re‑normalised to sum‑1."""
    mix = w * a + (1.0 - w) * b
    s = mix.sum()
    if s <= 0:
        return a.copy()
    return mix / s


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient on the true max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional log
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing with mass‑transfer and swap moves."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # swap move
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # mass‑transfer move
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Swap‑only hill‑climbing with tiny uphill acceptance."""
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # 1 % uphill acceptance
            if np.random.rand() < 0.01:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]  # revert

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate pool construction
# ----------------------------------------------------------------------
def _initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse set of shape vectors."""
    pool = []

    # Warm start and its cyclic shifts
    if warm is not None:
        warm = warm.copy()
        pool.append(warm)
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Deterministic families
    pool.append(_uniform_shape(n))

    sigmas = (0.05, 0.08, 0.12, 0.15, 0.20, 0.30)
    for sigma in sigmas:
        pool.append(_central_gaussian_shape(n, sigma))

    alphas = (1.5, 2.0, 3.0, 4.0, 5.0)   # symmetric Beta, peaked
    for alpha in alphas:
        pool.append(_beta_shape(n, alpha))

    pool.append(_triangular_shape(n))
    pool.append(_sinusoid_shape(n))

    # double‑Gaussian (peaks at both ends)
    for sigma in (0.1, 0.2, 0.3):
        pool.append(_double_gaussian_shape(n, sigma))

    # Interleaved versions of all above (helps to spread mass)
    interleaved = [_interleaved_perm(a) for a in pool]
    pool.extend(interleaved)

    # Pairwise convex mixtures (coarse grid)
    base = pool.copy()
    grid_ws = np.linspace(0.0, 1.0, 11)  # 0,0.1,…,1
    for i in range(len(base)):
        for j in range(i + 1, len(base)):
            a_i, a_j = base[i], base[j]
            for w in grid_ws[1:-1]:  # skip extremes (already present)
                mix = _mix_shapes(a_i, a_j, w)
                pool.append(mix)
                pool.append(_interleaved_perm(mix))

    # Random Dirichlet draws – add some stochastic variety
    for _ in range(30):
        a = np.random.dirichlet(np.ones(n))
        pool.append(a)
        pool.append(_interleaved_perm(a))

    # Ensure all candidates are proper probability vectors
    cleaned = []
    for a in pool:
        a = np.asarray(a, dtype=np.float64)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        cleaned.append(a)
    return cleaned


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found before the time deadline.
    """
    np.random.seed(seed)
    # Small safety margin so that we always return before the hard timeout.
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size (chosen as a good trade‑off between quality and speed)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (if a previous solution is available)
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Phase 0 – generate a large, varied pool of candidates
    # --------------------------------------------------------------
    pool = _initial_pool(n, warm)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase I – fast enumeration of the pool (≈10 % of budget)
    # --------------------------------------------------------------
    enum_deadline = time.time() + (deadline - time.time()) * 0.10
    for a in pool:
        if time.time() >= enum_deadline:
            break
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Enum] new best {val:.6f}")

    # If nothing survived (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = _uniform_shape(n)
        best_val = _evaluate_objective(best_seq, n, conv_fft_len)

    # --------------------------------------------------------------
    # Phase II – long sub‑gradient refinement (≈55 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 30:
        print("[Subgrad] lengthy refinement started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.9,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Subgrad] improvement → {val_ref:.6f}")

    # --------------------------------------------------------------
    # Phase III – simulated annealing (≈30 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 15:
        print("[Anneal] simulated annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improvement → {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase IV – permutation hill‑climb (≈5 % of budget)
    # --------------------------------------------------------------
    if time.time() < deadline - 5:
        print("[Permute] hill‑climbing swaps")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=80_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] improvement → {val_perm:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and return
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respects evaluator limits
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
