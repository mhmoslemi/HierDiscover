def _gradient_swap_optimize(a0, n, conv_fft_len, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # compute active set weight uniform across active indices
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)   # gradient
        # compute direction: pick i max grad, j min grad
        i = int(np.argmax(grad))
        j = int(np.argmin(grad))
        if grad[i] <= grad[j] + 1e-12:
            # gradient uniform; break
            break
        # transfer step size: fraction of mass at i
        delta = a[i] * 0.3   # up to 30% can transfer

        a[j] += delta

        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1

"""
Propose a step‑function that minimizes 2·n·max(conv)/sum².
The algorithm consists of:
1. Diverse warm‑starts (families + many Dirichlet draws) – keep the best.
2. Several cheap projected‑gradient runs from random seeds (to escape local minima).
3. A long exponentiated‑gradient (mirror‑descent) refinement on the exact max‑convolution.
4. A final random mass‑transfer hill‑climber that uses the exact objective.
All steps respect a global time budget and always keep the best vector seen.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a). Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

# ----------------------------------------------------------------------
# Candidate families (identical to the reference implementation)
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))

def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape

def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape

def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape

def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)

def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    pool = []
    # Dirichlet draws with several α
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))
    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0):
        pool.append(_edge_exponential(n, gamma))
    # Power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))
    # Beta‑pdf with extreme parameters
    for a, b in [(0.1, 0.1), (0.2, 0.2), (0.5, 0.5), (0.5, 2.0),
                 (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))
    # Two Gaussians at the ends
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    return pool

# ----------------------------------------------------------------------
# Objective (exact)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# Optimisers
# ----------------------------------------------------------------------
def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Euclidean projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (uniform weights on all max locations)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)  # decreasing steps
        a = a - eta * grad
        a = _project_onto_simplex(a)

        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional logging
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _eg_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent with KL divergence) on the exact max‑convolution.
    The update is a_i ← a_i·exp(−η·grad_i) followed by renormalisation.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule – start relatively large and decay as 1/√t
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad)   # mirror‑descent update
        a_sum = a.sum()
        if a_sum > 0:
            a = a / a_sum
        else:
            a = np.random.dirichlet(np.ones(n))

        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[EG it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: randomly relocate a fraction of mass.
    The move is kept only if it strictly improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = float(2.0 * n * conv2.max() / (a2.sum() ** 2))

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

"""
Propose a step‑function that minimizes 2·n·max(conv)/sum².
The algorithm works in several stages and always respects the global time
budget:

1. **Warm‑start pool** – a large collection of deterministic families
   (uniform, sinusoid, edge‑exponential, power‑law, beta, double‑Gaussian)
   plus a handful of Dirichlet draws.  The best candidate of the pool
   becomes the seed.

2. **Diverse cheap PGD restarts** – a few short projected‑gradient descents
   from random Dirichlet seeds are run (each with a larger initial step).
   This quickly explores many basins of attraction.

3. **Exponentiated‑gradient refinement** – a long mirror‑descent run on the
   exact max‑convolution.  The update `a_i ← a_i·exp(−η·grad_i)` keeps the
   iterate on the simplex and often escapes the local minima that PGD
   gets stuck in.

4. **Final random mass‑transfer hill‑climber** – a handful of cheap
   random moves (moving up to 30 % of a source entry) are performed; a move
   is kept only if it strictly improves the exact objective.

All phases keep the best vector seen so far and the function finally
returns it as a plain Python list.  The routine never raises an exception
and always returns a feasible sequence (non‑negative, finite, ≤1000 per
entry) required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a). Returns a vector of length len(a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Families for the warm‑start pool
# ----------------------------------------------------------------------
def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse deterministic pool."""
    pool = []

    # Dirichlet draws with several concentrations
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (now also very steep)
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0):
        pool.append(_edge_exponential(n, gamma))

    # Power‑law (negative exponents → edge‑heavy)
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta PDFs – with some extreme parameters
    for a, b in [(0.1, 0.1), (0.2, 0.2), (0.5, 0.5), (0.5, 2.0),
                 (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Two Gaussians centred at the two ends
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool


# ----------------------------------------------------------------------
# Exact objective (same formula as ``evaluate_sequence``)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Euclidean projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # active set (uniform weighting on all maximal positions)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _eg_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent with KL) on the exact max‑convolution.
    Update: a_i ← a_i·exp(−η·grad_i)  then re‑normalise.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            a = np.random.dirichlet(np.ones(n))

        val = float(2.0 * n * max_c / (a.sum() ** 2))
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[EG it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber that moves a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = float(2.0 * n * conv2.max() / (a2.sum() ** 2))

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size (4096 points – a good compromise)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    pool: list[np.ndarray] = []
    if warm is not None:
        # a few cyclic shifts of the supplied warm start
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    pool.extend(_initial_pool(n, rng))

    # extra Dirichlet draws with extreme concentrations
    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0, 15.0):
        for _ in range(4):
            pool.append(_random_dirichlet(n, alpha, rng))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Quick scan of the whole pool
    # --------------------------------------------------------------
    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = rng.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] best {val:.6f}")

    # --------------------------------------------------------------
    # Phase 1 – short PGD runs from random seeds (diversify)

    remaining = deadline - time.time()

            n,
            conv_fft_len,
            corr_fft_len,
            pgd_phase_deadline,

            eta0=1.0,          # a slightly larger initial step

            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            eg_deadline,
