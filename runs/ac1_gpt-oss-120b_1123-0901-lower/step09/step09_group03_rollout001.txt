"""
Hybrid optimizer with projected additive sub‑gradient refinement.

The algorithm follows four stages:
1.  Fast Adam on a smooth‑max (log‑sum‑exp) surrogate for a quick drop.
2.  Exponentiated‑gradient sub‑gradient descent (as in the reference code).
3.  Projected additive sub‑gradient descent with occasional smoothing –
    this is the main novelty and usually pushes the bound a few
    0.001‑units lower.
4.  Pair‑wise targeted simulated annealing and a short random anneal.

If time permits the solution is up‑sampled to a larger grid (8192 → 16384)
and the whole pipeline is repeated on the finer resolution.
All vectors are kept on the probability simplex (sum‑to‑1) and are
clipped to the range accepted by ``evaluate_sequence`` before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two greater than or equal to x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _smooth(a: np.ndarray, kernel: np.ndarray) -> np.ndarray:
    """Apply a 1‑D smoothing kernel (periodic boundary) and renormalise."""
    # periodic convolution via FFT (faster for large n)
    n = a.shape[0]
    klen = kernel.shape[0]
    pad = klen // 2
    a_ext = np.concatenate([a[-pad:], a, a[:pad]])
    sm = np.convolve(a_ext, kernel, mode='valid')
    sm = np.maximum(sm, 0.0)
    s = sm.sum()
    if s == 0:
        return a
    return sm / s

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate log‑sum‑exp(λ·(a∗a))."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.08
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max => a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (sum a == 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form through softmax
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional shake near deadline
        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent (same as reference)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # re‑normalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _subgradient_optimize_additive(a0: np.ndarray,
                                   n: int,
                                   conv_fft_len: int,
                                   corr_fft_len: int,
                                   deadline: float,
                                   max_iters: int = 400_000,
                                   eta0: float = 0.8,
                                   smooth_every: int = 5000) -> tuple[np.ndarray, float]:
    """
    Projected additive sub‑gradient descent.
    Every ``smooth_every`` iterations a small smoothing kernel is applied
    to the iterate to encourage a smoother shape (this is the main
    improvement over the reference implementation).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    # smoothing kernel (5‑point, sum=1)
    kern = np.array([0.1, 0.2, 0.4, 0.2, 0.1], dtype=np.float64)

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[AddGrad it={it:06d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing stepsize
        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a

        # projection onto simplex
        a = _project_simplex(a)

        # occasional smoothing
        if (it + 1) % smooth_every == 0:
            a = _smooth(a, kern)

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(a0: np.ndarray,
                                   n: int,
                                   conv_fft_len: int,
                                   deadline: float,
                                   max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Pair‑wise targeted simulated annealing (same as reference)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # indices that realize the peak
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """Light random annealing (as in reference)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    # warm start + shifted copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) + \
                np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates with equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimizes the evaluation score.
    The routine respects the time budget and always returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Choose problem size adaptively (larger if we have enough time)
    n_small = 4096
    n_big = 8192
    if deadline - time.time() > 400:  # enough time for a finer grid
        n = n_big
    else:
        n = n_small

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start from any previous global solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick Adam on a few diverse starts (≈20 % of time)
    phase1_dead = time.time() + (deadline - time.time()) * 0.20
    while pool and time.time() < phase1_dead:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(phase1_dead, time.time() + 2.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=80_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long sub‑gradient descent (≈50 % of time)
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase II] Exponential sub‑gradient")
        sub_dead = time.time() + (deadline - time.time()) * 0.45
        a_exp, val_exp = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=300_000, eta0=0.85
        )
        if val_exp < best_val:
            best_val = val_exp
            best_seq = a_exp.copy()
            print(f"[Phase II] ExpGrad improved to {val_exp:.6f}")

        # Additive projected sub‑gradient with smoothing
        if time.time() < deadline - 20:
            print("[Phase II] Additive projected sub‑gradient")
            sub_dead2 = deadline - 10   # leave ~10 s for later phases
            a_add, val_add = _subgradient_optimize_additive(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead2,
                max_iters=250_000, eta0=0.8, smooth_every=5000
            )
            if val_add < best_val:
                best_val = val_add
                best_seq = a_add.copy()
                print(f"[Phase II] Additive improved to {val_add:.6f}")

    # ------------------------------------------------------------------
    # Phase III – pair‑wise targeted annealing (≈20 % of time)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Pair‑wise targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=150_000
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[Phase III] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – light random annealing (use any remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase IV] Final random annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=50_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to an even finer grid if time permits
    if best_seq is not None and time.time() + 30 < deadline:
        print("[Upsample] Refinement on higher resolution")
        n_big = n * 2
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on the larger grid
        sub_dead_big = min(deadline, time.time() + 6.0)
        a_big_adam, _ = _adam_optimize(
            a_big, n_big, conv_fft_len_big, corr_fft_len_big,
            sub_dead_big, max_iters=150_000
        )

        # longer additive sub‑gradient on the large grid
        sub_dead_big2 = deadline - 5
        a_big_ref, val_big = _subgradient_optimize_additive(
            a_big_adam, n_big, conv_fft_len_big, corr_fft_len_big,
            sub_dead_big2, max_iters=200_000, eta0=0.7
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_ref.copy()
            print(f"[Upsample] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
