def propose_candidate(seed=0,budget_s=1000):
    import time, numpy as np

    start = time.time()
    deadline = start + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    rng = np.random.default_rng(seed)

    best_seq = None
    best_val = float('inf')
    search_params = [(150,4),(180,4),(130,5),(100,5)]

    # Warm start optional? If height_sequence_1 exists may already be good; we will also keep it as candidate.

    # evaluate warm start candidate if any
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm_arr = np.asarray(warm, dtype=np.float64)
        if warm_arr.shape[0] > n:
            warm_arr = warm_arr[:n]
        elif warm_arr.shape[0] < n:
            warm_arr = np.pad(warm_arr, (0, n-warm_arr.shape[0]))
        warm_arr = np.maximum(warm_arr, 0.0)
        if warm_arr.sum() > 0.01:
            warm_arr /= warm_arr.sum()
            val = 2 * n * np.max(_full_convolution(warm_arr, conv_fft_len)) / (warm_arr.sum()**2)
            best_seq = warm_arr.copy()
            best_val = float(val)
            print(f"Warm start val {best_val}")

    for K, max_allowed in search_params:
        if time.time() >= deadline:
            break
        # number of attempts maybe dynamic: e.g., 1000 attempts or until time left small.
        max_trials = 5000
        # look for set by greedy
        S, maxN = generate_low_collision_set(n, K, max_allowed, rng, max_trials=max_trials)
        if S is None:
            continue
        # create vector
        a = np.zeros(n, dtype=np.float64)
        a[S] = 1.0 / K
        # evaluate quickly via sum_counts method
        # compute conv counts via convolution of binary indicator K-sum
        # use full convolution on indicator
        b = np.zeros(n, dtype=np.float64)
        b[S] = 1.0
        conv_counts = _full_convolution(b, conv_fft_len)  # this yields floats but integer values
        max_counts = conv_counts.max()
        val = 2 * n * max_counts / (K**2)  # because sum(a)=1
        print(f"Generated set K={K}, max_counts={int(max_counts)}, val approx {val:.6f}")
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"New best from set: val {best_val}")

    # if we have a best_seq, run refined gradient descent
    if best_seq is not None and time.time() < deadline - 5:
        # allocate half of remaining time for GD
        gd_deadline = time.time() + (deadline - time.time()) * 0.6
        a_opt, val_opt = _gd_optimize(best_seq, n, conv_fft_len, corr_fft_len, gd_deadline, max_iters=300000)
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"After GD best {best_val}")

    # final local random search for remaining time
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(best_seq, n, conv_fft_len, deadline, max_moves=500000)
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"After local search best {best_val}")

    if best_seq is None:
        best_seq = np.full(n, 1.0/n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f}")
    return best_seq.tolist()

"""
Propose a step function with very small max‑convolution.

The idea:
    * Build a sparse “low‑collision’’ set of indices S ⊂ {0,…,n‑1}.
      For every possible sum i+j the number of ordered pairs that produce it
      is bounded by a small constant C (here C=4).  With |S|=K we then have
      max(a∗a) = C / K² (because a_i = 1/K on S, 0 elsewhere).  The objective
      2·n·max(a∗a) ≈ 2·n·C/K² can be pushed well below 1.5 by taking K≈150.

    * The set S is obtained by a simple random greedy algorithm that respects
      the bound C.  Several (K,C) pairs are tried; the best candidate is kept.

    * The resulting uniform vector on S is used as a warm start for a
      projected‑gradient descent (exact line‑search) that directly minimizes
      the true max‑convolution.  Finally a cheap random‑move hill climber
      further polishes the result.

All phases respect the global 1000 s wall‑clock limit.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (used by the gradient descent)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Random low‑collision set construction
def _generate_low_collision_set(
    n: int,
    K: int,
    C: int,
    rng: np.random.Generator,
    max_attempts: int = 200_000,
):
    """
    Random greedy construction of a set S of size K such that for every sum
    s=i+j the ordered count of pairs (i,j) does not exceed C.
    Returns (list_of_indices, max_count) or (None, None) on failure.
    """
    used = np.zeros(n, dtype=bool)          # which positions are already taken
    S = []                                   # selected indices
    sum_counts = np.zeros(2 * n - 1, dtype=np.int16)  # ordered pair counts
    attempts = 0

    while len(S) < K and attempts < max_attempts:
        i = rng.integers(0, n)
        if used[i]:
            attempts += 1
            continue

        # self‑pair check (adds 1)
        if sum_counts[i + i] + 1 > C:
            attempts += 1
            continue

        # check all cross‑pairs (each adds 2)
        feasible = True
        for j in S:
            s = i + j
            if sum_counts[s] + 2 > C:
                feasible = False
                break

        if feasible:
            # accept i
            used[i] = True
            S.append(i)
            sum_counts[i + i] += 1                # self‑pair
            for j in S[:-1]:                      # cross‑pairs with earlier elements
                s = i + j
                sum_counts[s] += 2
            attempts = 0   # reset after a success
        else:
            attempts += 1

    if len(S) == K:
        return S, int(sum_counts.max())
    return None, None

# Projected gradient descent on the exact objective
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
):
    """
    Simple projected gradient descent on  F(a)=2·n·max(a*a) / (sum a)² .
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # back‑tracking line search (projected)
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val

# Very cheap hill‑climber (local random moves)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
):
    """
    Transfer a random fraction of the mass from one position to another,
    keeping the move if it improves the true objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # enforce the guards used by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
