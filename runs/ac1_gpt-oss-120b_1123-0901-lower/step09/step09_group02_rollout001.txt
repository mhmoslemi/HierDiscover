"""
stepfunction_optimizer_v3.py

A multi‑restart, multi‑phase optimizer for the objective

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

Key ideas beyond the previous version
-------------------------------------------------
1. **Very diverse random start pool** – we create thousands of random
   candidates from many analytic families (Gaussian, double‑Gaussian,
   Beta, power‑law, exponential, random Dirichlet, etc.) + their
   random permutations.  This gives a much richer set of basins to explore.

2. **Two‑stage refinement** – each start is first passed through a short
   Adam optimisation on a smooth‑max surrogate, then through a quick
   exponentiated‑gradient sub‑gradient phase.  The best few basins are
   refined for a long time.

3. **Noise injection** – during the long sub‑gradient phase we periodically
   inject a tiny uniform component (∼10⁻⁴) to prevent coordinates from
   becoming permanently zero, which greatly improves exploration.

4. **Longer simulated annealing** – the final SA uses many more moves,
   a slightly higher initial temperature and a “swap‑or‑transfer” move set.

5. **Final permutation hill‑climb** – as before, but with a larger
   budget.

All phases respect the global time budget (default 1000 s) and the
routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no clipping) – used for internal bookkeeping."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)

def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp (smooth‑max) surrogate.
    Returns the best vector seen (always on the simplex) and its exact value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.12
    lam = 5.0                 # initial smooth‑max temperature
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- soft‑max projection onto the simplex ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective for logging ----
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # stability: subtract max_c before exponentiation
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or not np.isfinite(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max (mirror‑descent form) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule (increase λ) ----
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near the deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _expgrad_subgradient(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
    noise_every: int = 20_000,
    noise_level: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max.
    Periodically injects a tiny uniform noise component to keep all
    coordinates alive.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # ---- active set (indices attaining the max) ----
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- step size (decreasing) ----
        eta = eta0 / np.sqrt(it + 1.0)

        # ---- exponentiated‑gradient update ----
        a = a * np.exp(-eta * grad_a)

        # ---- keep a strictly positive ----
        a = np.maximum(a, 1e-12)

        # ---- renormalise ----
        a /= a.sum()

        # ---- occasional uniform noise injection ----
        if (it + 1) % noise_every == 0:
            a = (1.0 - noise_level) * a + noise_level * (np.ones_like(a) / n)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing with two move types (mass transfer & swap) and a
    slower linear temperature decay.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ---- swap move ----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ---- mass‑transfer move ----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.45          # up to 45 % transfer
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # accept uphill move with Metropolis probability
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation hill‑climbing (swap‑only).  Accepts a swap only if it
    does not increase the objective, with a tiny probability (1 %) of
    accepting uphill moves to escape plateaus.
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]           # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance
            if np.random.rand() < 0.01:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]   # revert swap

        it += 1

    return best_a, best_val

def _random_gaussian_center(n: int) -> np.ndarray:
    """Single centred Gaussian (continuous, then discretised)."""
    sigma = np.random.uniform(0.03, 0.25)   # relative to n
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    g = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
    if g.sum() == 0:
        return None
    return g / g.sum()


def _random_gaussian_edges(n: int) -> np.ndarray:
    """Two symmetric Gaussians at the ends."""
    sigma = np.random.uniform(0.03, 0.25)
    x = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((x) / (sigma * n)) ** 2)
    g2 = np.exp(-0.5 * ((x - (n - 1)) / (sigma * n)) ** 2)
    g = g1 + g2
    if g.sum() == 0:
        return None
    return g / g.sum()


def _random_beta_symmetric(n: int) -> np.ndarray:
    """Beta(α,α) on [0,1] with random α in (0.2,5)."""
    alpha = np.random.uniform(0.2, 5.0)
    x = np.arange(1, n + 1, dtype=np.float64) / n
    shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
    if shape.sum() == 0:
        return None
    return shape / shape.sum()


def _random_powerlaw(n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i·(n‑i+1))^{-p} + edge term."""
    p = np.random.uniform(0.2, 2.0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = prod ** (-p)
    shape += 0.5 * (idx ** (-p) + (n - idx + 1) ** (-p))   # small edge bump
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        return None
    return shape / shape.sum()


def _random_exponential(n: int) -> np.ndarray:
    """Two‑sided exponential decay."""
    lam = np.random.uniform(0.1, 3.0)
    left = np.exp(-lam * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_spike_pattern(n: int) -> np.ndarray:
    """Sparse spike pattern with random positions and Dirichlet weights."""
    k = np.random.choice([32, 64, 128, 256])
    a = np.zeros(n, dtype=np.float64)
    pos = np.random.choice(n, size=k, replace=False)
    masses = np.random.dirichlet(np.ones(k))
    a[pos] = masses
    return a


def _random_triangular(n: int) -> np.ndarray:
    """Simple triangular shape (rise–fall)."""
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    if tri.sum() == 0:
        return None
    return tri / tri.sum()


def _generate_candidate_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Creates a large, heterogeneous pool of start vectors."""
    pool = []

    # 1. Warm start and a few cyclic shifts (if available)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 5, n // 2, 3 * n // 5):
            pool.append(np.roll(warm, shift))

    # 2. Analytic families with random parameters
    funs = [
        _random_gaussian_center,
        _random_gaussian_edges,
        _random_beta_symmetric,
        _random_powerlaw,
        _random_exponential,
        _random_triangular,
    ]
    for _ in range(80):          # 80 random draws from each family ⇒ 480 candidates
        for fn in funs:
            a = fn(n)
            if a is not None:
                pool.append(a)

    # 3. Random Dirichlet draws (different concentration)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        pool.append(a)

    # 4. Random spike patterns
    for _ in range(30):
        pool.append(_random_spike_pattern(n))

    # 5. Uniform baseline
    pool.append(np.full(n, 1.0 / n))

    # 6. Interleaved permutations of every candidate (helps to spread mass)
    interleaved = []
    for a in pool:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    pool.extend(interleaved)

    # Shuffle to avoid ordering bias
    np.random.shuffle(pool)
    return pool

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # Problem size – 4096 works well with FFT‑based convolutions
    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (previous solution, if any)
    warm = globals().get("height_sequence_1")

        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)

            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------

    pool = _generate_candidate_pool(n, warm)

    best_seq: np.ndarray | None = None

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
