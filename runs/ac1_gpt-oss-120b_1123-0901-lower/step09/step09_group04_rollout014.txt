def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·len(a)-1 elements (linear conv)
    return conv[: 2 * a.shape[0] - 1]

    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, v≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic families (all return a vector that already sums to 1)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    s = np.maximum(s, 0.0)
    return s / s.sum()


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _triangle(n: int) -> np.ndarray:
    """Symmetric triangle (peak in the middle)."""
    half = np.arange(1, n // 2 + 1, dtype=np.float64)
    if n % 2 == 0:
        shape = np.concatenate([half, half[::-1]])
    else:
        shape = np.concatenate([half, np.array([half[-1] + 1]), half[::-1]])
    shape /= shape.sum()
    return shape


def _arcsine(n: int) -> np.ndarray:
    """Discrete arcsine density (β(½,½)) – heavy at the edges."""
    eps = 1e-12
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = 1.0 / np.sqrt(i * (n + 0.5 - i) + eps)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Objective (exact same formula as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Return 2·n·max(convolution) / (sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max() / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (log‑sum‑exp)
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # initialise on the simplex (softmax later)

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.1          # low temperature to start, grows fast
    lam_factor = 1.2

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max (project onto simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for monitoring)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        w = w / w_sum if w_sum > 0 else np.full_like(w, 1.0 / w.size)

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent style gradient on the logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny perturbation when close to the deadline
        if it % 5_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # simple learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple projected sub‑gradient descent on the exact max‑conv
# ----------------------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    eta = eta0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max, with tolerance)
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            # no active index – shouldn't happen, skip update
            it += 1
            continue

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # adaptive stepsize: decrease slowly as 1/sqrt(t)
        eta = eta0 / np.sqrt(it + 1)

        a_new = a - eta * grad_a
        a_new = _project_onto_simplex(a_new)

        # accept if improvement, otherwise keep the old point (helps stability)
        if _objective(a_new, conv_fft_len) < val:
            a = a_new
        else:
            # occasionally shake a bit to escape tiny plateaus
            if it % 20_000 == 0 and time.time() > deadline - 20:
                jitter = (np.random.rand(n) - 0.5) * 1e-3
                a = _project_onto_simplex(np.maximum(a + jitter, 0.0))

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local mass‑transfer moves
# ----------------------------------------------------------------------


def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_fft_len)

    # initial temperature set to a small fraction of current value
    T0 = best_val * 0.02
    t = 0

    while time.time() < deadline and t < max_moves:
        # pick a source with non‑negligible mass
        i = np.random.randint(n)
        if a[i] < 1e-12:
            t += 1
            continue
        j = np.random.randint(n)
        if i == j:
            t += 1
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, conv_fft_len)

        # temperature linearly decays to zero
        T = T0 * (1.0 - t / max_moves)

        accept = False
        if val2 < best_val:
            accept = True
        else:
            # Metropolis criterion
            if T > 0 and np.log(np.random.rand()) < -(val2 - best_val) / T:
                accept = True

        if accept:
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if t % 5_000 == 0:
                    print(f"[SA move {t:06d}] improved to {val2:.6f}")

        t += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) initial random / deterministic pool
    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # deterministic families (some with a few parameter sweeps)
    families = [
        _uniform,
        _sinusoid,
        lambda nn: _edge_exponential(nn, 0.2),
        lambda nn: _edge_exponential(nn, 1.0),
        lambda nn: _edge_exponential(nn, 5.0),
        lambda nn: _edge_power_law(nn, -1.0),
        lambda nn: _edge_power_law(nn, -0.5),
        lambda nn: _beta_shape(nn, 0.5, 0.5),
        lambda nn: _beta_shape(nn, 2.0, 2.0),
        lambda nn: _double_gaussian(nn, 0.1),
        lambda nn: _double_gaussian(nn, 0.3),
        _triangle,
        _arcsine,
    ]

    # evaluate deterministic candidates
    for f in families:
        if time.time() >= deadline:
            break
        cand = f(n)
        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Deterministic] new best {val:.6f}")

    # random Dirichlet draws – we generate as many as time allows
    dirichlet_draws = 0
    while time.time() < deadline - 30 and dirichlet_draws < 30_000:
        # pick a concentration factor uniformly in log‑space
        log_alpha = rng.uniform(-2.0, 1.0)   # α ∈ [10⁻², 10¹]
        alpha = 10 ** log_alpha
        cand = rng.dirichlet(np.full(n, alpha))
        # ensure numerical safety (already sums to 1)
        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Dirichlet {dirichlet_draws}] new best {val:.6f}")
        dirichlet_draws += 1

    # If nothing succeeded (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq, conv_fft_len)

    # ------------------------------------------------------------------
    # 2) Adam refinement (≈25 % of remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.25 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=200_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refinement] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected‑gradient descent (long phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        pgd_deadline = deadline - 10  # reserve a few seconds for SA
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=600_000,
            eta0=1.2,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD refinement] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        a_sa, val_sa = _simulated_annealing(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[Simulated annealing] improved best {val_sa:.6f}")

    # ------------------------------------------------------------------
    # 5) final safety – clip to allowed range and return as list
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Finished. Best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
