"""
Hybrid optimizer with L‑BFGS smooth‑max refinement

The algorithm consists of three phases:

1.  **Warm‑start creation** – a diverse pool of simple analytic shapes
    (uniform, sinusoid, double‑Gaussian, power‑law, edge spikes, …) and,
    if available, the previous best solution ``height_sequence_1``.

2.  **Smooth‑max L‑BFGS** – the non‑smooth objective
       F(a)=2·n·max(a∗a)/(∑a)²
    is replaced by a log‑sum‑exp surrogate
       Sλ(a)=2·n·(1/λ)·log∑exp(λ·(a∗a)).
    For a schedule of increasing λ the surrogate becomes tighter.
    Each stage is minimised with L‑BFGS‑B on the log‑softmax
    parametrisation (logits → probabilities).  The gradient is obtained
    analytically via the convolution cross‑correlation, so each iteration
    only requires two FFTs.

3.  **Targeted pair‑wise annealing** – a short Metropolis annealing that
    repeatedly moves a small fraction of mass away from the pair (i,j)
    that contributes most to the current maximal convolution entry.
    This step exploits the exact objective and often yields a final
    improvement.

All phases respect the global time budget (default 1000 s) and the
returned sequence is always a plain Python list of non‑negative numbers.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
#  FFT‑based convolution utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns ∂/∂a_i Σ_k w_k (a∗a)_k.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
#  Smooth‑max objective for L‑BFGS
# ----------------------------------------------------------------------
def _smooth_max_func(
    logits: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
):
    """
    Returns (value, gradient) of the λ‑smoothed objective
    2·n·(1/λ)·log∑exp(λ·(a∗a)), where a = softmax(logits).
    """
    # Softmax → probability vector on the simplex
    mx = logits.max()
    exp_logits = np.exp(logits - mx)
    a = exp_logits / exp_logits.sum()

    # Convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # Smooth‑max via log‑sum‑exp (numerically stable)
    max_c = conv.max()
    scaled = np.exp(lam * (conv - max_c))
    sum_scaled = scaled.sum()
    if sum_scaled == 0 or np.isnan(sum_scaled):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = scaled / sum_scaled

    smooth_max = max_c + np.log(sum_scaled) / lam
    value = 2.0 * n * smooth_max

    # Gradient w.r.t. a: 2·cross_corr(w, a)   (derivative of each (a∗a)_k)
    grad_a = 4.0 * n * _cross_correlation(w, a, corr_fft_len)

    # Chain rule through soft‑max (Euclidean projection onto simplex)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    return float(value), grad_logits.astype(np.float64)


def _lbfgs_optimize(
    logits: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iter: int,
):
    """
    Perform a limited‑iteration L‑BFGS‑B optimisation of the
    λ‑smoothed surrogate, starting from ``logits``.
    """
    res = scipy.optimize.minimize(
        _smooth_max_func,
        logits,
        args=(lam, n, conv_fft_len, corr_fft_len),
        method="L-BFGS-B",
        jac=True,
        options=dict(maxiter=max_iter, disp=False),
    )
    return res.x, res.fun

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (different concentrations)
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edge families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(
            -0.5 * ((i - (n - 1)) / (sigma * n)) ** 2
        )
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  The routine respects the given time budget and
    always returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # problem size – moderate to keep each iteration cheap
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 1e-12:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a pool of initial candidates
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Main loop over candidates (stops when time runs out)
    # ------------------------------------------------------------------
    lambda_schedule = [0.1, 0.5, 1.0, 5.0, 20.0, 100.0, 500.0]
    for a0 in pool:
        if time.time() > deadline:
            break

        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # start from soft‑max logits (interior of the simplex)
        eps = 1e-12
        logits = np.log(a0 + eps)

        # ----------------------------------------------------------------
        # L‑BFGS refinement with increasing smoothness
        # ----------------------------------------------------------------
        for lam in lambda_schedule:
            if time.time() > deadline:
                break
            # number of L‑BFGS iterations – very cheap per step
            max_iters = 1200
            try:
                logits, _ = _lbfgs_optimize(
                    logits,
                    lam,
                    n,
                    conv_fft_len,
                    corr_fft_len,
                    max_iters,
                )
            except Exception as e:
                print(f"[L‑BFGS] exception {e} – aborting this candidate")
                break

            # Evaluate the *exact* objective (no smoothing)
            a = np.exp(logits - np.max(logits))
            a = a / a.sum()
            val_exact = evaluate_sequence(a.tolist())
            if val_exact < best_val:
                best_val = val_exact
                best_seq = a.copy()
                print(f"[L‑BFGS λ={lam:.2f}] new best {val_exact:.6f}")

        # ----------------------------------------------------------------
        # Short pair‑wise annealing to escape tiny plateaus
        # ----------------------------------------------------------------
        if best_seq is not None and time.time() < deadline - 5:
            a_pair, val_pair = _pair_targeted_anneal_opt(
                best_seq, n, conv_fft_len, deadline, max_moves=30_000
            )
            if val_pair < best_val:
                best_val = val_pair
                best_seq = a_pair.copy()
                print(f"[PairAnneal] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a valid return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to limits accepted by ``evaluate_sequence``
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
