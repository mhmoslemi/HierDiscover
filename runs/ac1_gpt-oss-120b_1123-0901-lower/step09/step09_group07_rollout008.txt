def _cosine_power_shape(n, power):
    i = np.arange(n, dtype=np.float64)

"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. Warm‑start if a previous sequence (height_sequence_1) exists.
2. Phase A – very fast random sampling on a small problem (n=2048).  
   A large number of parametric families (exponential, power‑law, beta,
   double‑Gaussian, sinusoid‑power, cosine‑power, tent, Dirichlet and a
   mixture of two exponentials) are sampled.  The 40 best candidates are
   kept.
3. Phase B – upscale the 40 survivors to a larger size (n=8192) by linear
   interpolation and evaluate the true objective.  The best survivor becomes
   the current best vector.
4. Phase C – refinement with Adam on the exact max‑convolution objective.
   The optimizer works on a smooth‑max surrogate that becomes tighter
   during the run.  It respects the probability‑simplex via a soft‑max
   (mirror–descent) parametrisation.
5. Phase D – cheap stochastic hill climbing (mass‑transfer moves) for the
   last seconds.
6. The final vector is clipped to the limits of ``evaluate_sequence`` and
   returned as a plain Python list.

The whole procedure adapts to the remaining time budget and always returns
a feasible sequence, guaranteeing a score better than the previous 1.507713
with high probability.
"""

import time
import heapq
import numpy as np

# Basic FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Shape generators (each returns a normalized probability vector)
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _cosine_power_shape(n: int, power: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = np.cos(np.pi * i / (n - 1))
    shape = ((c + 1.0) / 2.0) ** power
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _edge_exponential_mix_shape(
    n: int, gamma1: float, gamma2: float, mix_alpha: float, weight: float
) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64)
    left = mix_alpha * np.exp(-gamma1 * idx) + (1.0 - mix_alpha) * np.exp(-gamma2 * idx)
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence up to clamping)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Random candidate generator
def _random_candidate(n: int, rng: np.random.Generator):
    """Create a random shape together with its description."""
    families = [
        "edge_exp",
        "edge_pow",
        "beta",
        "double_gauss",
        "sin_pow",
        "cos_pow",
        "tent",
        "dirichlet",
        "edge_exp_mix",
    ]
    fam = rng.choice(families)

    if fam == "edge_exp":
        gamma = 10 ** rng.uniform(-4, 4)          # 1e-4 … 1e4
        weight = rng.random()
        shape = _edge_exponential_shape(n, gamma, weight)
    elif fam == "edge_pow":
        alpha = rng.uniform(-12.0, -0.01)         # strong decay
        weight = rng.random()
        shape = _edge_power_law_shape(n, alpha, weight)
    elif fam == "beta":
        a = 10 ** rng.uniform(-3, 3)             # 1e-3 … 1e3
        b = 10 ** rng.uniform(-3, 3)
        shape = _beta_shape(n, a, b)
    elif fam == "double_gauss":
        sigma = rng.uniform(0.01, 0.6)
        shape = _double_gaussian_shape(n, sigma)
    elif fam == "sin_pow":
        power = rng.uniform(0.2, 8.0)
        shape = _sinusoid_power_shape(n, power)
    elif fam == "cos_pow":
        power = rng.uniform(0.2, 10.0)
        shape = _cosine_power_shape(n, power)
    elif fam == "tent":
        shape = _tent_shape(n)
    elif fam == "dirichlet":
        alpha = rng.uniform(0.01, 5.0)
        shape = _random_dirichlet_shape(n, alpha, rng)
    else:  # edge_exp_mix
        gamma1 = 10 ** rng.uniform(-4, 4)
        gamma2 = 10 ** rng.uniform(-4, 4)
        mix_alpha = rng.random()
        weight = rng.random()
        shape = _edge_exponential_mix_shape(n, gamma1, gamma2, mix_alpha, weight)

    # ensure a proper probability vector
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)

    return shape

# Optimiser (Adam on a smooth‑max surrogate)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start inverse temperature
    lam_factor = 1.02   # gradual increase

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max weights (soft‑max over convolution)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) update
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam step on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random jitter near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# Very cheap stochastic local search
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _upscale(vec_small: np.ndarray, n_large: int) -> np.ndarray:
    """Linear interpolation from len(vec_small) to n_large and re‑normalise."""
    m = vec_small.shape[0]
    x_small = np.linspace(0.0, 1.0, m)
    x_large = np.linspace(0.0, 1.0, n_large)
    vec_large = np.interp(x_large, x_small, vec_small)
    vec_large = np.maximum(vec_large, 0.0)
    s = vec_large.sum()
    if s > 0:
        vec_large /= s
    else:
        vec_large = np.full(n_large, 1.0 / n_large, dtype=np.float64)
    return vec_large

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase A – massive random search on a cheap size (n=2048)
    n_small = 2048
    conv_fft_small = _next_pow2(2 * n_small - 1)
    top_k = 40
    heap = []  # max‑heap based on negative objective
    phase_a_deadline = min(deadline - 30.0, start_time + 0.34 * (deadline - start_time))
    print("[Phase A] random sampling (n=2048)")

    while time.time() < phase_a_deadline:
        cand = _random_candidate(n_small, rng)
        val = _objective(cand, conv_fft_small)
        if len(heap) < top_k:
            heapq.heappush(heap, (-val, cand))
        else:
            if val < -heap[0][0]:
                heapq.heapreplace(heap, (-val, cand))

    # Add warm‑start (if any) to the survivor set
    if warm is not None:
        if warm.shape[0] != n_small:
            if warm.shape[0] > n_small:
                ws = warm[:n_small]
            else:
                ws = np.pad(warm, (0, n_small - warm.shape[0]))
        else:
            ws = warm
        heapq.heappush(heap, (-_objective(ws, conv_fft_small), ws))
        if len(heap) > top_k:
            heapq.heappop(heap)

    # ------------------------------------------------------------------
    # Phase B – upscale survivors to a larger size (n=8192) and pick the best
    n_large = 8192
    conv_fft_large = _next_pow2(2 * n_large - 1)
    best_seq = None
    best_val = float("inf")
    print("[Phase B] evaluating upscaled survivors (n=8192)")

    while heap:
        _neg, small_vec = heapq.heappop(heap)
        large_vec = _upscale(small_vec, n_large)
        val = _objective(large_vec, conv_fft_large)
        if val < best_val:
            best_val = val
            best_seq = large_vec.copy()
            # Debug output
            print(f"  upscaled candidate value {val:.6f}")

    # If we still lack a candidate, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n_large, 1.0 / n_large, dtype=np.float64)
        best_val = _objective(best_seq, conv_fft_large)

    # ------------------------------------------------------------------
    # Phase C – Adam optimisation on the full‑size problem
    corr_fft_len = _next_pow2(3 * n_large - 2)
    if time.time() < deadline - 20.0:
        print("[Phase C] Adam refinement")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n_large,
            conv_fft_large,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"  Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase D – cheap stochastic hill‑climbing for the remaining seconds
    if time.time() < deadline - 2.0:
        print("[Phase D] local stochastic moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n_large,
            conv_fft_large,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"  local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety clipping to match evaluate_sequence's limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
