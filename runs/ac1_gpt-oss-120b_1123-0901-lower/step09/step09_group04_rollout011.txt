def _sgd_subgradient(a0, n, conv_fft_len, deadline):
    a = a0.copy()
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a)
    step = 1.0 # initial step
    min_step = 1e-6
    it = 0
    while time.time() < deadline:
        # compute conv and gradient
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2) # sum=1 => 2*n*max_c

        # gradient: compute active set weight uniform over indices achieving max
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, conv_fft_len)  # Should use corr_fft_len but we can reuse same len

        # determine step size via backtracking line search
        step_cur = step
        while True:
            a_candidate = _project_onto_simplex(a - step_cur * grad)
            # compute objective for candidate
            conv_c = _full_convolution(a_candidate, conv_fft_len)
            max_cand = conv_c.max()
            val_cand = 2.0 * n * max_cand / (a_candidate.sum() ** 2)
            if val_cand < val:
                break
            step_cur *= 0.5
            if step_cur < min_step:
                # cannot improve; break out
                break

        # if improvement found:
        if val_cand < best_val:
            best_val = val_cand
            best_a = a_candidate.copy()
            # maybe increase step (increase step for next iteration)
            step = step_cur * 1.1
            a = a_candidate
        else:
            # no improvement; reduce step
            step = step_cur * 0.5
            # optionally random perturbation to escape
            if np.random.rand() < 0.01:
                a = a + (np.random.rand(n) - 0.5) * 1e-3
                a = np.clip(a, 0, None)
                a = a / a.sum()
        it += 1
        if it % 2000 == 0:
            print(f"[Subgrad it={it}] best {best_val:.6f}")
    return best_a, best_val

def _exponential_gradient_descent(a0, ...):
    a = a0.copy()
    best = a.copy()
    best_val = _objective(a)
    eta = 1.0 (choose)
    t=0
    while time < deadline:
        conv, max_c = compute conv
        w = uniform over active indices (as before)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len) # gradient approx, positive values generally
        # exponentiated gradient step
        a = a * np.exp(-eta * grad)  # elementwise
        a = np.clip(a, 0, None)
        sum_a = a.sum()
        if sum_a <= 0:
            a = np.random.dirichlet(np.ones(len(a)))
        else:
            a = a / sum_a
        # optional step-size schedule: eta = eta0 / sqrt(t+1)
        val = _objective(a)
        if val < best_val:
            best_val = val
            best = a.copy()
        t+=1
    return best, best_val

"""
Propose a low‑value step‑function.

The search consists of four phases:

1. Diverse warm‑starts – a large pool of deterministic families (uniform,
   sinusoid, edge‑exponential, power‑law, beta, double‑Gaussian, parabolic)
   together with many Dirichlet draws.  The best candidate becomes the seed.

2. Long Adam optimisation – we minimise a smooth log‑sum‑exp surrogate of the
   max‑convolution.  The temperature λ is increased steadily so that the
   surrogate approaches the exact max.  Adam’s adaptive learning‑rate and a
   slowly decaying step‑size give a fast descent.

3. Projected gradient with momentum – an exact sub‑gradient descent on the
   true max‑convolution, using a Nesterov‑style momentum term and a
   1/√t step‑size schedule.  This refines the Adam result.

4. Final cheap hill‑climber – many random mass‑transfer moves are tried;
   any move that improves the objective is kept.  This squeezes out the last
   bits of improvement.

All phases respect the global time budget (default 1000 s) and the routine
always returns the best sequence found so far.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Candidate generators
# ----------------------------------------------------------------------


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start (if any)."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    """Dirichlet draw with concentration α (scalar)."""
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^α + (n-i)^α."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _parabolic(n: int) -> np.ndarray:
    """Symmetric parabola a_i ∝ (i+1)*(n-i)."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = i * (n - i + 1.0)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse deterministic pool."""
    pool = []

    # Dirichlet draws (different concentrations)
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Power‑law
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
                 (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple families
    pool.append(_sinusoid(n))
    pool.append(_parabolic(n))
    pool.append(_uniform(n))

    return pool


# ----------------------------------------------------------------------
#  Fast objective (same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

#  Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr_start: float = 0.05,
    lam_start: float = 0.2,
    lam_factor: float = 1.025,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (sum = 1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny shake near the end of the budget
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.2,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with a momentum term on the exact max‑convolution.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    v = np.zeros_like(a)  # momentum buffer
    best_a = a.copy()
    best_val = _objective(a)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD‑mom it={it:06d}] best {best_val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule
        eta = eta0 / np.sqrt(it + 1)

        # Nesterov‑style momentum
        v = beta * v - eta * grad
        a = a + v
        a = _project_onto_simplex(a)

        # small random jitter near the very end
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.35)      # up to 35 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – we use the same size as the reference solution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    #  Warm‑start handling (optional previous solution)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]), constant_values=0.0)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    #  Build a large and diverse initial pool
    # ------------------------------------------------------------------
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # extra random Dirichlet draws (various concentrations)
    alphas = [0.05, 0.1, 0.2, 0.3, 0.5, 0.8, 1.0, 1.5, 2.0, 5.0, 10.0]
    for alpha in alphas:
        for _ in range(12):
            pool.append(_random_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    #  Quick scan of the pool
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq)

    # ------------------------------------------------------------------
    #  Phase 1 – long Adam optimisation (≈70 % of remaining time)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:   # keep a few seconds for the final stages
        adam_deadline = time.time() + 0.70 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=600_000,
            lr_start=0.07,
            lam_start=0.2,
            lam_factor=1.025,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    #  Phase 2 – projected gradient with momentum (remainder – 5 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        pgd_deadline = deadline - 5.0
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=600_000,
            eta0=1.2,
            beta=0.9,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD‑mom] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    #  Phase 3 – final cheap hill‑climber (all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    #  Safety clipping and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
