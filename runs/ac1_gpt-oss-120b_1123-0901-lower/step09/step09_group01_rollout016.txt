"""
Step‑function optimizer (new version).

Main ideas
----------
1. **Broad parametric sweep** – as before, generate many analytic families
   (power‑law, exponential, stretched‑exponential, beta, double‑Gaussian,
   monotone, random spikes).  Keep the best few candidates.

2. **Random restarts + L‑BFGS on a smooth surrogate** – draw a Dirichlet‑type
   random vector, normalise it and run a limited L‑BFGS optimisation on the
   *smooth‑max* surrogate  
   `smoothmax(a)=log‑sum‑exp(λ·(a*a))/λ`.  The optimisation is performed on the
   logits `x` with `a = softmax(x)`, guaranteeing non‑negativity and sum‑to‑1.
   A schedule of increasing temperatures (`λ = 1, 10, 100, …`) drives the
   surrogate from a very smooth landscape to the true max‑convolution.

3. **Exact‑max sub‑gradient (exponentiated‑gradient)** – a long mirror‑descent
   phase refines the best vector found so far with the true non‑smooth
   objective.

4. **Optional simulated‑annealing fine‑tuning** – if time permits, a
   cheap hill‑climbing step that swaps small mass fractions between coordinates.

The whole routine respects the supplied time budget (default 1000 s) and
always returns a valid non‑negative list of coefficients.
"""

import time
import math
import numpy as np
from typing import List, Tuple

# ----------------------------------------------------------------------
# Low‑level FFT utilities ------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (needed for gradients)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) ----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² – the true quantity we want to minimise."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_random_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Simple Dirichlet draw with all concentration parameters = alpha."""
    # using Gamma sampling – NumPy does this for us.
    a = np.random.gamma(alpha, 1.0, size=n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Smooth‑max surrogate (used by L‑BFGS) ---------------------------------
def _smooth_obj_and_grad(
    x: np.ndarray,
    n: int,
    lam: float,
    conv_fft_len: int,
    corr_fft_len: int,
) -> Tuple[float, np.ndarray]:
    """
    Return (value, gradient) of 2·n·smoothmax(a) where a = softmax(x)
    and smoothmax = log‑sum‑exp(lam·(a*a))/lam.
    """
    # soft‑max (keeps sum = 1)
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # smooth‑max via log‑sum‑exp (numerically stable)
    shift = lam * conv.max()
    exp_shift = np.exp(lam * conv - shift)
    sum_exp = exp_shift.sum()
    smoothmax = (shift + np.log(sum_exp)) / lam

    f = 2.0 * n * smoothmax

    # weights for the gradient (soft‑max over the convolution entries)
    w = exp_shift / sum_exp

    # gradient wrt a, see derivation in the analysis section
    grad_a = (4.0 * n / lam) * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max (mirror‑descent form)
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return f, grad_x


# ----------------------------------------------------------------------
# L‑BFGS optimizer on the smooth surrogate -------------------------------
def _lbgfs_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    best_seq: np.ndarray,
    best_val: float,
    lam_start: float = 1.0,
    lam_factor: float = 10.0,
    stages: int = 5,
    maxit_per_stage: int = 200,
) -> Tuple[np.ndarray, float]:
    """
    Run a short L‑BFGS on the smooth‑max surrogate.
    The temperature λ is increased geometrically (lam_start·lam_factor**stage)
    to drive the surrogate towards the exact max.
    Returns the best vector found (and its exact objective).
    """
    import scipy.optimize

    # start from logits (log‑soft‑max parameters)
    x = np.log(a0 + 1e-15)

    for stage in range(stages):
        lam = lam_start * (lam_factor ** stage)

        # L‑BFGS – a few iterations are enough for a direction improvement
        res = scipy.optimize.minimize(
            _smooth_obj_and_grad,
            x,
            args=(n, lam, conv_fft_len, corr_fft_len),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": maxit_per_stage, "disp": False},
        )
        x = res.x

        # recover a from logits
        mx = x.max()
        a = np.exp(x - mx)
        a = a / a.sum()

        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS λ={lam:.1e}] new best {val:.6f}")

        # time check
        if time.time() > deadline - 0.5:
            break

    return best_seq, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) -----------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> Tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex with the true max‑convolution objective."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Simple simulated‑annealing polish ------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> Tuple[np.ndarray, float]:
    """
    Very cheap annealing: repeatedly transfer a random fraction of mass from
    one index to another and accept if it improves (or with a Metropolis
    probability).  Useful as a final fine‑tuning step.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem dimension (feel free to change – larger n gives more freedom)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ----------------------------------------
    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray, source: str):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{source}] new best {val:.6f}")

    # symmetric families -------------------------------------------------
    for p in np.linspace(-4.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n), "Power")

    for alpha in np.logspace(-4, 0.8, 30):
        _maybe_update(_candidate_exponential(alpha, n), "Exp")

    alphas = np.logspace(-3, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n), "Stretched")

    for a_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for b_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
            _maybe_update(_candidate_beta(a_par, b_par, n), "Beta")

    for sig in np.logspace(-2.5, -0.2, 18):
        _maybe_update(_candidate_double_gaussian(sig, n), "DoubleGauss")

    for alpha in np.logspace(-4, 1.0, 20):
        _maybe_update(_candidate_left_exponential(alpha, n), "LExp")

    for p in np.linspace(-3.0, 1.0, 25):
        _maybe_update(_candidate_left_power(p, n), "LPower")

    # random spikes ------------------------------------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True), "Spike")

    # optional warm start -------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "Warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Random Dirichlet restarts + L‑BFGS (smooth‑max) -----------------
    # Allocate about 30 % of the remaining budget to this phase.
    phase2_deadline = deadline - 200.0
    while time.time() < phase2_deadline:
        # draw a random Dirichlet; concentration α is sampled log‑uniformly
        log_alpha = np.random.uniform(-2.0, 2.0)   # α ∈ [0.01, 100]
        alpha = 10 ** log_alpha
        a_rand = _candidate_random_dirichlet(alpha, n)
        # quick check – keep only if it improves the incumbent
        val0 = _objective(a_rand, n)
        if val0 < best_val:
            best_val = val0
            best_seq = a_rand.copy()
            print(f"[RandDir] new best {val0:.6f}")

        # run a short L‑BFGS optimisation from this start
        best_seq, best_val = _lbgfs_optimize(
            a_rand,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            best_seq,
            best_val,
            lam_start=1.0,
            lam_factor=10.0,
            stages=4,
            maxit_per_stage=150,
        )

    # ------------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient descent (exponentiated‑gradient) -----
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning -----------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine‑tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity‑check and return ------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity logging using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
```

This implementation enriches the previous approach with multiple random restarts, a proper L‑BFGS optimisation on a smooth surrogate of the max‑convolution (which gives a stronger descent direction than simple Adam), and retains the long exact‑max sub‑gradient phase together with an optional annealing polish. It respects the 1000 s time budget, always returns a valid non‑negative sequence, and prints intermediate progress for transparency.
