"""
step‑function optimizer – extended smooth‑max Adam + richer parametric families.

The algorithm:
1️⃣  A fast parametric sweep over many asymmetric families (exponential,
    power‑law, mixed exponential + uniform, plateau + exponential, etc.).
    Each candidate is normalised to unit ℓ₁‑norm and evaluated with an
    FFT‑based convolution.  The best vector so far is kept.

2️⃣  A long “smooth‑max” Adam optimisation.
    * The step‑function is represented in log‑space (soft‑max) so the
      simplex constraint is satisfied automatically.
    * The objective  maxₖ (a★a)[k] is approximated by the smooth‑max
      (log‑sum‑exp) with a temperature λ that is increased gradually
      (continuation method).  Gradients are obtained analytically via a
      convolution with the weight vector w∝exp(λ·(conv‑max)).
    * Adam with standard β₁,β₂,eps is run for as many iterations as the
      remaining time budget allows (≈ 70 % of the total time).  The
      best exact objective encountered is stored.

3️⃣  A final exact‑max sub‑gradient phase (exponentiated‑gradient /
    mirror‑descent) for the remaining time (~30 %).  This refines the
    solution on the true non‑smooth objective.

The whole routine respects the 1000 s budget and always returns a
valid non‑negative list of coefficients.  Logging prints intermediate
best scores and timings.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT helpers ------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the relevant part
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of the smooth‑max.
    Both arguments are real; the result has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact evaluation of 2·n·max(a★a)/(∑a)²  (no clipping).
    The vector a is assumed non‑negative; the function is scale‑invariant,
    but we keep the denominator for safety.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ----------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_mix_exponential(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Mixture of a left‑exponential and a uniform baseline:
    a_i ∝ (1‑β)·exp(-α·i) + β·(1/n).
    """
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - beta) * np.exp(-alpha * i) + beta / n
    a /= a.sum()
    return a


def _candidate_plateau_exponential(p: float, m: int, alpha: float, n: int) -> np.ndarray:
    """
    Plateau + exponential tail.
    Fraction p of total mass is placed uniformly on the first m indices.
    The remaining (1‑p) is distributed as a left‑exponential starting at i=m.
    """
    if m <= 0 or m >= n:
        raise ValueError("invalid plateau size")
    a = np.empty(n, dtype=np.float64)
    # plateau part
    a[:m] = p / m
    # exponential tail
    tail_len = n - m
    i = np.arange(tail_len, dtype=np.float64)
    tail = np.exp(-alpha * i)
    tail_sum = tail.sum()
    if tail_sum == 0:
        tail = np.full(tail_len, 1.0 / tail_len)
    else:
        tail = tail / tail_sum
    a[m:] = (1.0 - p) * tail
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p  (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution density on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """k‑spike pattern – Dirichlet‑distributed masses on k random positions."""
    a = np.zeros(n, dtype=np.float64)
    if k == 0:
        a[:] = 1.0 / n
        return a
    pos = rng.choice(n, size=k, replace=False)
    masses = rng.dirichlet(np.ones(k))
    a[pos] = masses
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth‑max surrogate.
    Returns the best exact solution seen during the run.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08
    lam = 0.05          # starting temperature
    lam_factor = 1.015  # slow sharpening

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (only for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))   # max_c subtracted for stability
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max (w‑weighted sum of convolutions)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 5e6)

        # very occasional learning‑rate decay
        if it > 0 and it % 250_000 == 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using the exponentiated‑gradient update.
    Returns the best vector observed.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrored on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – safeguard against numerical loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size – 4096 works well for the reference evaluator
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1)  BROAD PARAMETRIC SWEEP
    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray, label: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {label}] new best {val:.6f}")

    # ----- monotone left‑exponential -------------------------------------------------
    for alpha in np.logspace(-4, 1.2, 25):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a, f"expα={alpha:.3g}")

    # ----- mixed exponential + uniform ----------------------------------------------
    for alpha in np.logspace(-4, 1.0, 20):
        for beta in np.linspace(0.0, 0.8, 9):
            a = _candidate_mix_exponential(alpha, beta, n)
            _maybe_update(a, f"mixα={alpha:.3g}β={beta:.2f}")

    # ----- plateau + exponential ----------------------------------------------------
    plateau_sizes = [64, 128, 256, 512, 1024]
    for m in plateau_sizes:
        for p in np.linspace(0.05, 0.4, 8):        # fraction of mass on plateau
            for alpha in np.logspace(-4, 0.8, 12):
                a = _candidate_plateau_exponential(p, m, alpha, n)
                _maybe_update(a, f"platm={m}p={p:.2f}α={alpha:.3g}")

    # ----- power‑law (negative exponents) --------------------------------------------
    for p in np.linspace(-4.0, -0.2, 30):
        a = _candidate_left_power(p, n)
        _maybe_update(a, f"powp={p:.2f}")

    # ----- beta‑distribution ---------------------------------------------------------
    a_pars = [0.2, 0.5, 1.0, 2.0, 5.0]
    b_pars = [0.2, 0.5, 1.0, 2.0, 5.0]
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a, f"beta a={a_par} b={b_par}")

    # ----- random spike patterns -----------------------------------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            a = _candidate_spike(k, n, rng)
            _maybe_update(a, f"spike k={k}")

    # ----- warm start from previous global variable ----------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "prev_global")

    print(f"[Sweep] best after parametric exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2)  EXTENDED SMOOTH‑MAX ADAM (majority of remaining budget)
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 30.0:
        # allocate roughly 70 % of remaining time to Adam
        adam_deadline = time.time() + 0.7 * time_left
        print("[Phase‑Adam] long smooth‑max Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=5_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3)  EXACT‑MAX SUB‑GRADIENT REFINEMENT (remaining time)
    time_left = deadline - time.time()
    if best_seq is not None and time_left > 5.0:
        print("[Phase‑Subgrad] exact‑max sub‑gradient refinement")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=5_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4)  Safety net – ensure we return a valid non‑negative list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (evaluate_sequence missing).")

    return best_seq.tolist()
