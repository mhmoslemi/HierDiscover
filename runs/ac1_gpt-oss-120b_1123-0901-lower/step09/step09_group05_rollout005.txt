"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

Algorithm
---------
1. **Generation** – a very diverse pool of candidates is built:
   * deterministic families (uniform, edge‑exponential, edge‑power‑law,
     beta‑pdf, double‑Gaussian, sinusoid);
   * many random Dirichlet draws with different concentration;
   * random convex mixtures of two already‑generated vectors.
   Every vector is normalised to have sum = 1.

2. **Screening** – the whole pool is evaluated with an FFT based
   convolution and the best ``top_k`` candidates are kept.

3. **Refinement** – each elite candidate is improved by
   projected (sub‑)gradient descent on the *exact* objective.
   The step‑size uses a simple back‑tracking line‑search and each
   iterate is projected onto the probability simplex.

4. **Polishing** – the current best vector is further optimised by a
   cheap hill‑climber that repeatedly moves a random fraction of the
   mass from one entry to another and keeps the move if it improves
   the objective.

All phases respect the global 1000 s wall‑clock budget; the function
returns the best sequence found (as a plain Python list).  The code
works for any length – we use ``n = 8192`` (larger than the previous
run) which gives the algorithm more degrees of freedom while keeping
the per‑iteration cost small.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Deterministic families (all vectors are normalised to sum == 1) -----
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(alpha * 1) and return a normalised vector."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Gradient descent on the exact (non‑smooth) objective -----------------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Projected (sub‑)gradient descent with back‑tracking line‑search."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # ----- back‑tracking line‑search (projected) -----
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climber (random mass transfers) --------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Randomly move a fraction of the mass from one index to another."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑value step‑function within the allotted wall‑clock time.
    Returns the best sequence (list of floats) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – a larger n gives more freedom without hurting speed
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Optional warm start from a previous solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 1️⃣  Build a rich pool of start vectors -------------------------
    pool = []

    # a) deterministic families (coarse grid)
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))

    for gamma in np.logspace(-3, 1.5, 40):
        pool.append(_edge_exponential(n, float(gamma)))

    for alpha in np.linspace(-5.0, 2.0, 36):
        pool.append(_edge_power_law(n, float(alpha)))

    a_vals = np.logspace(-1, 2, 12)
    b_vals = np.logspace(-1, 2, 12)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    for sigma in np.linspace(0.01, 0.5, 30):
        pool.append(_double_gaussian(n, float(sigma)))

    # b) random Dirichlet draws (various concentrations)
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0, 10.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # c) cyclic shifts of a warm start (if present)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # d) random convex mixtures of two already built vectors
    #    (helps creating hybrid shapes)
    existing = list(pool)   # snapshot
    for _ in range(200):
        i, j = rng.integers(0, len(existing), size=2)
        w = rng.random()
        mixed = w * existing[i] + (1.0 - w) * existing[j]
        mixed = mixed / mixed.sum()
        pool.append(mixed)

    # ------------------------------------------------------------------
    # 2️⃣  Quick screening – keep the best few candidates -------------
    top_k = 12
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        total = a.sum()
        if total < 1e-12:
            continue
        a = a / total
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        scored.append((val, a))
        # leave a bit of time for the refinement stage
        if time.time() > start_time + 0.20 * (budget_s - 0.5):
            break

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # 3️⃣  Refine elite candidates with projected gradient descent -------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 5.0:
            break
        # allocate a shrinking share of the remaining budget
        remaining = deadline - time.time()
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=250_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final cheap local optimisation (mass‑transfer hill‑climber) --
    if best_seq is not None and time.time() < deadline - 2.0:
        # number of moves scales with remaining time
        time_left = deadline - time.time()
        max_moves = int(250_000 * (time_left / (budget_s / 2)))
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=max_moves,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen) ------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits imposed by ``evaluate_sequence`` – they are far
    # above any values we ever generate, but the clipping is harmless.
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
