def _simplex_projection(v):
    """Projects v onto simplex {x >=0, sum x = 1}."""
    # expects v is numpy array.
    # Sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # If all v are negative, project to uniform distribution
        return np.full_like(v, 1.0/len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=int(1e9), eta0=1.0):
    a = a0.copy()
    a = np.maximum(a, 0)  # ensure non-negative
    a = a / a.sum() if a.sum()>0 else np.full(n, 1.0/n)
    best_a = a.copy()
    best_val = _objective(a, n)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute conv and max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Subgrad] it={it} new best {val:.6f}")
        # Compute subgradient
        active = np.isclose(conv, max_c, atol=1e-12)
        # If all entries zero maybe use uniform? but fine.
        if active.any():
            # uniform weight over active indices
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        # project onto simplex
        a = _simplex_projection(a)
    return best_a, best_val

import time, numpy as np

def _next_pow2(x): return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)
    return conv[:2*a.shape[0]-1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = a.shape[0]-1
    return corr[start:start+a.shape[0]]

def _objective(a,n):
    conv = _full_convolution(a,_next_pow2(2*n-1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return 2.0 * n * max_c / (s*s)

def _simplex_projection(v):
    """Project onto simplex sum=1, v>=0."""
    # Implementation from paper
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all zeros? Return uniform distribution
        return np.full_like(v, 1.0/n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
step‑function optimizer – multi‑phase search

The routine consists of:
1. Fast parametric sweep (power‑law, exponential, stretched‑exp,
   beta, double‑Gaussian, inverse‑sqrt, triangular, random Dirichlet,
   mixtures). The best shape is kept.
2. A high‑precision projected sub‑gradient descent on the exact
   max‑convolution objective (mirror‑descent on the simplex).
3. Many short random restarts of the same sub‑gradient method to
   escape local minima.
4. A final simulated‑annealing fine‑tuning run.

All phases respect the global time budget (default 1000 s) and the
function always returns a non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a required for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # v is all zeros – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution on [0,1] (a_par,b_par>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ √(i+1) + √(n‑i).  (smoothly higher at the ends)"""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1) + np.sqrt(n - i)
    a /= a.sum()
    return a


def _candidate_triangular(n: int) -> np.ndarray:
    """Symmetric triangular shape – peak in the centre."""
    i = np.arange(n, dtype=np.float64)
    a = np.minimum(i + 1, n - i)
    a[a < 0] = 0.0
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Optimisers -----------------------------------------------------------
def _projected_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient on the simplex (additive updates)."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Subgrad] it={it:06d} new best {val:.6f}")

        # active set → sub‑gradient
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # additive mirror‑descent step on the simplex
        a = a - eta * grad
        a = _simplex_projection(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (≤30 %) of mass from i to j
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase‑Subgrad‑Main] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Multiple short restarts (exploration) -------------------------
    restart = 0
    while time.time() < deadline - 8:   # keep a few seconds for final anneal
        sub_deadline = min(deadline, time.time() + 9.0)  # about 9 s per restart
        # generate start point
        if best_seq is not None and np.random.rand() < 0.6:
            # small perturbation of the current best
            eps = 0.01
            a0 = best_seq + eps * np.random.randn(n)
            a0 = np.maximum(a0, 0.0)
            a0 = a0 / a0.sum()
        else:
            # fresh random Dirichlet
            conc = np.random.uniform(0.05, 2.5)
            a0 = np.random.dirichlet(np.ones(n) * conc)

        a_opt, val_opt = _projected_subgradient_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            eta0=0.8,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Restart {restart}] new best {val_opt:.6f}")
        restart += 1

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing fine‑tuning -------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal‑Final] simulated‑annealing")
        a_opt, val_opt = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase‑Anneal‑Final] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:      # evaluator may not be present in the testing environment
        pass

    return best_seq.tolist()
