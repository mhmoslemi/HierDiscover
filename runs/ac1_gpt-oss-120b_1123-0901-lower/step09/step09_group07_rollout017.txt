"""
Fast search for a low‑score non‑negative step function.

Algorithm
---------
1. **Warm start** – use the global `height_sequence_1` if present.
2. **Broad random sampling** – generate many candidates from several
   parametric families (exponential edge, power‑law edge, beta,
   double‑Gaussian, sinusoid‑power, tent, Dirichlet) and keep the best.
3. **Gradient refinement** – run three first‑order optimisers
   (Adam, sub‑gradient, projected‑gradient) on the exact
   max‑convolution objective, each for as many iterations as time permits.
4. **Simulated‑annealing local search** – keep the current best vector,
   maintain its full linear convolution, and repeatedly move a random
   fraction of mass from one index to another.  Moves that improve the
   objective are always accepted; worsening moves are accepted with a
   probability decreasing with a temperature schedule.  Convolution updates
   are performed *incrementally* in O(n) using slice operations, which is
   much faster than recomputing an FFT each step.
5. Return the best sequence found, clipped to the limits of
   `evaluate_sequence`.

The whole routine respects the 1000 s budget (with a small safety margin)
and works for a fixed length `n = 4096`.  All helper functions are top‑level
and contain no closures, as required.
"""

import time
import numpy as np


# ----------------------------------------------------------------------
# Basic FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, fft_len: int) -> float:
    """Exact objective 2·n·max(a∗a)/(sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Shape generators -----------------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

# Optimisers (same as reference) ---------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        # cross‑correlation helper
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start : start + a.shape[0]]

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # cross‑correlation
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start : start + a.shape[0]]

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
        n_loc = v.shape[0]
        u = np.sort(v)[::-1]
        cssv = np.cumsum(u) - 1
        ind = np.arange(1, n_loc + 1)
        cond = u - cssv / ind > 0
        if not np.any(cond):
            return np.full_like(v, 1.0 / n_loc)
        rho = ind[cond][-1]
        theta = cssv[cond][-1] / rho
        return np.maximum(v - theta, 0.0)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start : start + a.shape[0]]

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed + 12345)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # -------------------------- problem size ----------------------------
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)
    corr_len = _next_pow2(3 * n - 2)

    # -------------------------- warm start -------------------------------
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    if warm is not None:
        val = _objective(warm, fft_len)
        best_seq = warm.copy()
        best_val = val
        print(f"[Init] warm start value {val:.6f}")

    # -------------------------- Phase 1 – random families ---------------
    families = ["edge_exp", "edge_pow", "beta", "double_gauss",
                "sin_pow", "tent", "dirichlet"]
    phase1_deadline = min(deadline - 30.0,
                          time.time() + 0.15 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 2)          # 0.01 … 100
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-10.0, -0.1)
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-1, 1)
            b = 10 ** rng.uniform(-1, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.4)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        else:   # dirichlet
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand, fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # -------------------------- Phase 2 – gradient refinement ------------
    if best_seq is not None and time.time() < deadline - 25.0:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(best_seq, n, fft_len, corr_len,
                                          deadline, max_iters=400_000,
                                          lr_start=0.06)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

        if time.time() < deadline - 15.0:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(best_seq, n, fft_len, corr_len,
                                                   deadline, max_iters=300_000,
                                                   eta0=0.9)
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Phase 2] Sub‑grad improved to {val_sub:.6f}")

        if time.time() < deadline - 5.0:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(best_seq, n, fft_len, corr_len,
                                           deadline, max_iters=250_000,
                                           eta0=0.35)
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[Phase 2] PGD improved to {val_pgd:.6f}")

    # Ensure we have a sequence to work with
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # -------------------------- Phase 3 – Simulated Annealing ------------
    # initialise current state = best_seq
    cur_seq = best_seq.copy()
    cur_conv = _full_convolution(cur_seq, fft_len)
    cur_val = best_val
    best_conv = cur_conv.copy()

    # SA hyper‑parameters
    max_frac = 0.5            # up to 50 % of source mass may be moved
    T = 0.001                 # initial temperature
    decay = 0.99995           # geometric decay per move
    step_vec = np.empty_like(cur_conv)

    print("Phase 3 – simulated‑annealing local search")
    while time.time() < deadline - 0.3:
        i = rng.integers(n)
        if cur_seq[i] < 1e-14:
            continue
        # choose a different target index
        j = rng.integers(n - 1)
        if j >= i:
            j += 1

        delta = cur_seq[i] * rng.random() * max_frac
        if delta <= 0.0:
            continue

        # build incremental change to convolution
        step_vec.fill(0.0)
        step_vec[i:i + n] -= 2.0 * delta * cur_seq
        step_vec[j:j + n] += 2.0 * delta * cur_seq
        step_vec[2 * i] += delta * delta
        step_vec[2 * j] += delta * delta
        step_vec[i + j] -= 2.0 * delta * delta

        new_max = (cur_conv + step_vec).max()
        new_val = 2.0 * n * new_max          # sum stays ≈1

        accept = False
        if new_val < cur_val:
            accept = True
        else:
            prob = np.exp(-(new_val - cur_val) / T)
            if rng.random() < prob:
                accept = True

        if accept:
            # apply the incremental update in‑place
            cur_conv[i:i + n] -= 2.0 * delta * cur_seq
            cur_conv[j:j + n] += 2.0 * delta * cur_seq
            cur_conv[2 * i] += delta * delta
            cur_conv[2 * j] += delta * delta
            cur_conv[i + j] -= 2.0 * delta * delta

            cur_seq[i] -= delta
            cur_seq[j] += delta

            cur_val = new_val

            if cur_val < best_val:
                best_val = cur_val
                best_seq = cur_seq.copy()
                best_conv = cur_conv.copy()
                print(f"[SA] new best {best_val:.6f}")

        # temperature schedule
        T = max(T * decay, 1e-12)

    # final safety clipping (matches evaluate_sequence limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # (optional) recompute exact objective for reporting
    final_val = _objective(best_seq, fft_len)
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
