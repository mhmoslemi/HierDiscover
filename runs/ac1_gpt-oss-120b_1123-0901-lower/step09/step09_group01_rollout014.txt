import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution_fft(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _objective_fft(a, n, fft_len):
    conv = _full_convolution_fft(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c)   # sum a = 1

def _grad_for_shift(a, k):
    n = a.shape[0]
    grad = np.zeros_like(a)
    if k < n:
        grad[:k+1] = 2.0 * a[k::-1]
    else:
        start = k - (n-1)
        grad[start:] = 2.0 * a[n-1:start-1:-1]
    return grad

def _subgradient_optimize_fast(a0, n, fft_len, deadline, eta0=0.9):
    # use log-parameters for mirror descent
    eps = 1e-16
    logs = np.log(a0 + eps)   # real logs
    best_a = None
    best_val = np.inf
    it = 0
    while time.time() < deadline:
        # compute current a from logs
        max_log = logs.max()
        a = np.exp(logs - max_log)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            logs = np.log(a + eps)
            s = 1.0
        a = a / s

        # objective
        conv = _full_convolution_fft(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # optionally print progress
            if it % 200000 == 0:
                print(f"[Subgrad] it={it} best={val:.6f}")

        # active shifts
        active = np.isclose(conv, max_c, atol=1e-12)
        active_idx = np.where(active)[0]
        if active_idx.size == 0:
            # should not happen
            grad = np.zeros_like(a)
        else:
            grad = np.zeros_like(a)
            for k in active_idx:
                grad += _grad_for_shift(a, k)
            grad /= float(active_idx.size)

        # step size
        eta = eta0 / np.sqrt(it + 1.0)

        # mirror descent update (log-space)
        logs -= eta * grad

        it += 1
    return best_a if best_a is not None else a0, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a larger problem dimension (n=8192) and a fast
    exponentiated‑gradient sub‑gradient descent that computes the
    gradient analytically (no extra FFT).  The routine respects the
    time budget, always returns a valid list, and prints intermediate
    best scores.

    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # problem size – larger gives a lower theoretical lower bound
    n = 8192
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (fast‑conv)
    best_a = None
    best_val = np.inf

    def _maybe_update(a):
        nonlocal best_a, best_val
        val = _objective_fft(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    for p in np.linspace(-5.0, -0.2, 20):
        i = np.arange(n, dtype=np.float64)
        a = ((i+1)*(n-i))**p
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum == 0:
            continue
        a /= a_sum
        _maybe_update(a)

    # ---- symmetric exponential (mirrored) -------------------------------
    for alpha in np.logspace(-5, 0.7, 20):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-alpha * i) + np.exp(-alpha * (n-1 - i))
        a /= a.sum()
        _maybe_update(a)

    # ---- stretched exponential -----------------------------------------
    alphas = np.logspace(-4, 0.4, 8)
    betas = np.linspace(0.3, 3.0, 8)
    for alpha in alphas:
        for beta in betas:
            i = np.arange(n, dtype=np.float64)
            t = i / (n-1)
            a = np.exp(-alpha * (t**beta)) + np.exp(-alpha * ((1 - t)**beta))
            a /= a.sum()
            _maybe_update(a)

    # ---- beta distribution ---------------------------------------------
    a_pars = [0.2, 0.5, 1.0, 2.0, 5.0]
    b_pars = [0.2, 0.5, 1.0, 2.0, 5.0]
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    for a_par in a_pars:
        for b_par in b_pars:
            a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
            a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
            a = np.maximum(a, 0.0)
            s = a.sum()
            if s == 0:
                continue
            a /= s
            _maybe_update(a)

    # ---- double‑Gaussian (mass at ends) ---------------------------------
    for sig in np.logspace(-2.5, -0.1, 12):
        t = np.arange(n, dtype=np.float64) / (n - 1)
        a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
        a /= a.sum()
        _maybe_update(a)

    # ---- monotone left‑exponential --------------------------------------
    for alpha in np.logspace(-5, 1.0, 12):
        i = np.arange(n, dtype=np.float64)
        a = np.exp(-alpha * i)
        a /= a.sum()
        _maybe_update(a)

    # ---- monotone left‑power --------------------------------------------
    for p in np.linspace(-3.0, 1.0, 12):
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) ** p
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        _maybe_update(a)

    # ---- random spike patterns ------------------------------------------
    spike_ks = [64, 128, 256, 512, 1024]
    for k in spike_ks:
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            a /= a.sum()
            _maybe_update(a)

    # ---- warm start from previous global variable -----------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Fast exponentiated‑gradient sub‑gradient phase
    if best_a is None:
        # fallback to uniform
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    if time.time() < deadline - 5:

        a_sub, val_sub = _subgradient_optimize_fast(
            best_a,
            n,
            fft_len,
            deadline,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_a = a_sub.copy()
            print(f"[Phase‑Subgrad‑Fast] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional brief simulated‑annealing fine tuning
    if time.time() < deadline - 1:
        print("[Phase‑Anneal‑Fine] simulated annealing")
        # simple anneal implementation (reuse earlier code)
        a = best_a.copy()
        conv = _full_convolution_fft(a, fft_len)
        val = 2.0 * n * conv.max()
        best_val2 = val
        best_a2 = a.copy()
        T_start = 0.2
        T_end = 1e-5
        moves = 0
        max_moves = 200_000
        while time.time() < deadline and moves < max_moves:
            T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
            i = np.random.randint(n)
            if a[i] < 1e-16:
                moves += 1
                continue
            j = np.random.randint(n)
            if i == j:
                moves += 1
                continue
            frac = np.random.rand() * 0.25
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution_fft(a2, fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_val2:
                best_val2 = val2
                best_a2 = a2.copy()
                a = a2
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
            moves += 1
        if best_val2 < best_val:
            best_val = best_val2
            best_a = best_a2.copy()
            print(f"[Phase‑Anneal‑Fine] improved to {best_val2:.6f}")

    # ------------------------------------------------------------------
    # Final sanity clip & return
    best_a = np.clip(best_a, 0.0, 1000.0)
    best_a = best_a / best_a.sum()
    # sanity evaluate using provided function
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence not available).")

    return best_a.tolist()

"""
Step‑function optimiser for the evaluation metric

    score = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm works in three phases:

1. **Broad parametric sweep** – generate a range of simple analytic
   families (mirrored power‑law, mirrored exponential, stretched‑exponential,
   beta‑distribution, double‑Gaussian, monotone exponential and power‑law,
   plus a U‑shaped power‑law).  All candidates are normalised to sum = 1 and
   evaluated with a fast FFT‑based convolution.  The best candidate is kept.

2. **Fast exponentiated‑gradient sub‑gradient descent** – the objective is
   convex in the maximum of the convolution, so a sub‑gradient is the
   (average) gradient of those active convolution entries.  The gradient can be
   computed analytically (no extra FFT) using simple slice operations.  The
   update is performed in log‑space (mirror‑descent with KL regulariser) which
   automatically preserves non‑negativity and the simplex constraint.
   The loop runs until the time budget is exhausted.

3. **Optional brief simulated‑annealing fine‑tuning** – random mass transfers
   are accepted according to a cooling schedule to escape any remaining local
   minima.

The problem size is chosen as n = 8192 – large enough to lower the theoretical
lower bound (≈ 1.0) while still allowing many optimisation iterations within the
1000 s budget.  The routine always returns a valid non‑negative list and prints
intermediate best scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using the FFT.
    Returns an array of length 2·len(a)‑1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective_fft(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) (sum(a) is assumed to be 1).
    """
    conv = _full_convolution_fft(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Gradient of a single active convolution shift -------------------------

def _grad_for_shift(a: np.ndarray, k: int) -> np.ndarray:
    """
    Sub‑gradient of conv_k = (a*a)[k] w.r.t. a (for a single active shift k).
    The result is a vector of length len(a).  The implementation uses only
    slicing – no extra FFTs.
    """
    n = a.shape[0]
    g = np.zeros_like(a)
    if k < n:
        # indices 0 … k receive a reversed prefix of a
        g[: k + 1] = 2.0 * a[k :: -1]
    else:
        # indices start … n‑1 receive a reversed suffix of a
        start = k - (n - 1)
        g[start:] = 2.0 * a[n - 1 : start - 1 : -1]
    return g


# ----------------------------------------------------------------------
# Fast exponentiated‑gradient sub‑gradient optimisation ------------------

def _subgradient_optimize_fast(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using an exponentiated‑gradient update.
    Sub‑gradient is computed analytically for the active convolution shifts.
    Returns the best vector seen (and its objective value).
    """
    eps = 1e-16
    # initialise log‑coordinates (mirror variables)
    logs = np.log(a0 + eps)

    best_a = None
    best_val = np.inf
    it = 0

    while time.time() < deadline:
        # recover a from log‑coordinates (stable soft‑max)
        max_log = logs.max()
        a = np.exp(logs - max_log)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            logs = np.log(a + eps)
            s = a.sum()
        a /= s

        # current objective
        conv = _full_convolution_fft(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad] it={it:07d} best={val:.6f}")

        # active shift(s)
        active = np.isclose(conv, max_c, atol=1e-12)
        idx = np.where(active)[0]
        if idx.size == 0:
            grad = np.zeros_like(a)
        else:
            grad = np.zeros_like(a)
            for k in idx:
                grad += _grad_for_shift(a, k)
            grad /= float(idx.size)

        # step size (decreasing)
        eta = eta0 / np.sqrt(it + 1.0)

        # mirror‑descent update in log‑space
        logs -= eta * grad

        it += 1

    # if the loop never improved (unlikely) fall back to the start point
    if best_a is None:
        best_a = a0.copy()
        best_val = _objective_fft(best_a, n, fft_len)

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine tuning ----------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simple annealing: move a random fraction of mass from one index to another.
    Accept moves that improve the objective or with a probability that decays
    with temperature.
    """
    a = a0.copy()
    conv = _full_convolution_fft(a, fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.2
    T_end = 1e-5
    moves = 0
    while time.time() < deadline and moves < max_moves:
        T = max(T_end, T_start * (1.0 - moves / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue
        frac = np.random.rand() * 0.25
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution_fft(a2, fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        moves += 1

    return best_a, best_val

        a /= a.sum()
        val = _objective_fft(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep LeftExp] new best {val:.6f}")

    # monotone left‑power (p can be negative)
    for p in np.linspace(-4.0, 1.0, 20):
        i = np.arange(n, dtype=np.float64)
        a = (i + 1) ** p
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s == 0:
            continue
        a /= s
        val = _objective_fft(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep LeftPower] new best {val:.6f}")

    # random spike patterns (few spikes)
    spike_ks = [64, 128, 256, 512, 1024]
    for k in spike_ks:
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            a /= a.sum()
            val = _objective_fft(a, n, fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sweep Spikes] new best {val:.6f}")

    # warm start from a previously stored sequence, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            val = _objective_fft(w, n, fft_len)
            if val < best_val:
                best_val = val
                best_seq = w.copy()
                print(f"[Sweep Warm] new best {val:.6f}")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Fast exponentiated‑gradient sub‑gradient optimisation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    if time.time() < deadline - 5.0:
        print("[Phase Subgrad‑Fast] starting mirror‑descent")
        a_opt, val_opt = _subgradient_optimize_fast(
            best_seq, n, fft_len, deadline, eta0=0.9
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase Subgrad‑Fast] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional annealing fine‑tuning
    if time.time() < deadline - 1.0:
        print("[Phase Anneal] fine‑tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, fft_len, deadline, max_moves=150_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return vector (non‑negative, normalised)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() == 0 or np.isnan(best_seq.sum()):
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / best_seq.sum()

    # sanity check with the public evaluator (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score (evaluate_sequence not available).")

    return best_seq.tolist()
