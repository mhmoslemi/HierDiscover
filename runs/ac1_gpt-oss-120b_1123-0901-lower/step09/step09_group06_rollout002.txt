def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Both vectors are real, w has length 2·len(a)-1, a has length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient wrt a (smooth‑max objective)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – become harder
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the deadline
        if it % 5_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# Projected sub‑gradient descent on the true max‑convolution
def _proj_subgradient_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 30_000,
    step_start: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient method on the exact max‑convolution.
    Returns best sequence found.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (all positions attaining the max)
        mask = (conv == max_c)
        active = np.flatnonzero(mask)
        if active.size == 0:
            break
        # uniform weights over the active set (sub‑gradient of max)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.size

        # gradient of max‑convolution w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule (diminishing)
        eta = step_start / np.sqrt(t + 1.0)

        a_new = a - eta * grad_a
        a_new = _project_onto_simplex(a_new)

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            # occasionally log progress
            if t % 5_000 == 0:
                print(f"[Subgrad step {t:06d}] improved to {val:.6f}")

        a = a_new

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects the 1000 s limit and always returns the best
    sequence discovered (as a plain Python list).
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # leave a safety margin

    # ------------------------------
    # problem size (chosen large enough to give a good bound)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------
    # gather a diverse pool of initial candidates
    pool = []

    # warm start, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s > 0.0:
            warm = warm / s
            pool.append(warm.copy())
            # a few cyclic shifts to diversify
            for shift in (n // 4, n // 2, 3 * n // 4):
                pool.append(np.roll(warm, shift))

    # deterministic families
    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))
    for p in np.linspace(-8, -0.2, 20):
        pool.append(_edge_power_law(n, p))
    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0), (5.0, 0.2),
        (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # random Dirichlet draws – a wide range of concentration parameters
    dirichlet_alphas = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    draws_per_alpha = 150   # total ≈ 1200 random vectors
    for alpha in dirichlet_alphas:
        for _ in range(draws_per_alpha):
            pool.append(_random_dirichlet(n, alpha, rng))

    # sparse spike candidates
    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick evaluation of the whole pool (using the fast FFT‑based objective)
    print(f"Evaluating {len(pool)} initial candidates...")
    heap: List[tuple[float, np.ndarray]] = []   # min‑heap of size K
    K = 12                                      # keep the best K candidates
    for cand in pool:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = _objective(a, n, conv_fft_len)
        if len(heap) < K:
            heapq.heappush(heap, (-val, a.copy()))   # store negative for max‑heap behaviour
        else:
            if -val > heap[0][0]:                     # better than current worst in heap
                heapq.heappushpop(heap, (-val, a.copy()))
        # early exit if time is running out
        if time.time() > deadline - 60:
            break

    # extract the top candidates (sorted by objective)
    top_candidates = [item[1] for item in heap]
    top_candidates.sort(key=lambda x: _objective(x, n, conv_fft_len))
    best_seq = top_candidates[0].copy()
    best_val = _objective(best_seq, n, conv_fft_len)
    print(f"[Init] best of pool {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam refinement for each top candidate (parallel in spirit)
    remaining_time = deadline - time.time()
    if remaining_time > 30:
        per_candidate_time = remaining_time / len(top_candidates)
        for idx, cand in enumerate(top_candidates):
            sub_deadline = time.time() + per_candidate_time - 2.0
            if sub_deadline <= time.time():
                break
            a_opt, val_opt = _adam_optimize(
                cand,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=200_000,
                lr_start=0.04,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Adam cand {idx}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – projected sub‑gradient descent on the exact max‑convolution
    if time.time() < deadline - 5:
        a_final, val_final = _proj_subgradient_descent(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=40_000,
            step_start=0.6,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Subgradient] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # Ensure compliance with the evaluator (clip, non‑negative, sum>0)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() == 0.0:
        best_seq = _uniform(n)

    # final sanity check using the official evaluator (optional but cheap)
    try:
        official = evaluate_sequence(best_seq.tolist())
        if official < best_val:
            best_val = official
            print(f"[Evaluator] confirmed best {official:.6f}")
    except Exception:   # noqa: BLE001 – defensive
        pass

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
