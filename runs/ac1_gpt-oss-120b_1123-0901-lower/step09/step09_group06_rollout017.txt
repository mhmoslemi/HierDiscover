"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)²

where “∗’’ is the linear convolution.  The algorithm works in several
phases inside the 1000 s time budget:

1. Generate a very diverse pool of deterministic families (exponential,
   power‑law, β‑pdf, double‑Gaussian, sinusoid, raised‑cosine, semicircle,
   edge‑spikes) together with many random Dirichlet vectors and a few
   sparse‑spike vectors.  The best few candidates are kept.

2. Run several (up to five) short Adam optimisations from different
   starts (the pool’s best, warm‑start if present and a few random
   Dirichlet draws).  The Adam optimiser works on a smooth‑max
   surrogate (soft‑max over the convolution values) and keeps the best
   exact objective found.

3. Refine the current best with a continuation L‑BFGS on the same
   surrogate, using a longer schedule of increasing temperature
   parameters (α = 10 → 2·10⁴).

4. Finish with a cheap stochastic hill‑climber that moves a random
   fraction of mass between two indices only when the true objective
   improves.

All heavy operations (convolution and cross‑correlation) are performed
with FFTs.  The routine always returns the best sequence found before
the deadline; the returned list obeys the evaluator’s clipping rules
(0 ≤ aᵢ ≤ 1000).  Progress is printed to stdout.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation of w with a (used for the gradient)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex utilities -----------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _edge_double_spike(n: int, gamma_l: float, gamma_r: float,
                       w_left: float = 0.5) -> np.ndarray:
    """Two exponential edges with independent decays and a weight for the left."""
    left = np.exp(-gamma_l * np.arange(n, dtype=np.float64))
    right = np.exp(-gamma_r * np.arange(n, dtype=np.float64))[::-1]
    shape = w_left * left + (1.0 - w_left) * right
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective used for comparison with evaluate_sequence."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate ----------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of max(a*a)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise in the log‑simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02               # initial inverse temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly tighten the surrogate
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when we are very close to the deadline
        if it % 5_000 == 0 and time.time() > deadline - 20:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS on the smooth‑max surrogate -----------------------
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Return (smooth‑max objective, gradient) for a given temperature α."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c                # ≤ 0
    exp_alpha_v = np.exp(alpha * v)  # safe, max = 1
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    # gradient weights
    w = exp_alpha_v / sum_exp

    # gradient wrt a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through soft‑max
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on a sequence of increasing temperatures α."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    # temperature schedule (more points than the previous code)
    alphas = [5.0, 20.0, 80.0, 300.0, 1200.0, 5000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 4:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len,
                                    corr_fft_len, alpha)

        remaining = deadline - time.time()
        max_iter = 180 if remaining > 30 else 70

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = _objective(a, n, conv_fft_len)
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climber ----------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Random hill‑climber (mass moves) that only accepts improvements."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (up to 30 %) of mass from i to j
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 10_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence (Python list) found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a rich pool of initial candidates
    pool = []

    # warm start and a few of its cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 8, n // 4, 3 * n // 8, n // 2):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with a wide range of concentration parameters
    for alpha in (0.01, 0.05, 0.1, 0.3, 0.8, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    # Edge‑exponential families
    for gamma in np.logspace(-3, 2, 30):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law families (very steep decays)
    for alpha in np.linspace(-12, -0.1, 30):
        pool.append(_edge_power_law(n, alpha))

    # Beta PDFs – include very sharp edge cases
    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (0.2, 0.2),
        (0.1, 0.1), (0.05, 0.05), (0.2, 5.0), (5.0, 0.2)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.01, 0.03, 0.07, 0.12, 0.2):
        pool.append(_double_gaussian(n, sigma))

    # Simple deterministic bases
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # Edge double‑spike (two independent exponentials)
    for gamma_l in (0.01, 0.1, 1.0):
        for gamma_r in (0.01, 0.1, 1.0):
            for w in (0.3, 0.5, 0.7):
                pool.append(_edge_double_spike(n, gamma_l, gamma_r, w))

    # Sparse random spikes
    for k in (64, 128, 256, 512, 1024):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick scan of the pool – keep the best few candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    top_candidates: List[tuple[np.ndarray, float]] = []

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        # keep a short list (size ≤ 5) of the current best candidates
        top_candidates.append((a.copy(), val))
        top_candidates.sort(key=lambda x: x[1])
        if len(top_candidates) > 5:
            top_candidates.pop()   # discard worst

        if time.time() > deadline - 70:     # reserve ~1 min for deep phases
            break

    # ------------------------------------------------------------------
    # Phase 2 – several short Adam runs from different starts
    # ------------------------------------------------------------------
    # Pre‑compute how many restarts we can afford
    remaining = deadline - time.time()
    # each Adam run will be given ~max 120 s (or less if we are close)
    max_restart_time = min(120.0, remaining / 4.0)
    restart_deadline = time.time() + max_restart_time

    # we will run Adam from the current best and from a few random Dirichlet starts
    restart_starts: List[np.ndarray] = [best_seq]
    # add a couple of random Dirichlet vectors (different α)
    for _ in range(3):
        alpha = 10 ** np.random.uniform(-2, 1)   # α ∈ [0.01, 10]
        restart_starts.append(_random_dirichlet(n, alpha))

    for idx, a0 in enumerate(restart_starts):
        if time.time() > deadline - 45:
            break
        # give each run a slice of the overall deadline
        sub_deadline = min(deadline - 5.0, time.time() + max_restart_time / len(restart_starts))
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len,
            sub_deadline, max_iters=250_000, lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam restart {idx}] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – continuation L‑BFGS refinement (if time permits)
    if best_seq is not None and time.time() < deadline - 20:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS final] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – final cheap hill‑climber (use the remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety net – should never trigger
    if best_seq is None:
        best_seq = _uniform(n)

    # Clip to the range accepted by the evaluator (0 ≤ aᵢ ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final verification using the provided evaluator (for logging only)
    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Finished – best reported value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
