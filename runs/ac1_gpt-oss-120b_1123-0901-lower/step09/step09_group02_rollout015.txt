def _solve_lp_for_b(a, M, active_idx):
    # Build sparse matrix A_active
    n = len(a)
    rows = []
    cols = []
    data = []
    for idx, i in enumerate(active_idx):
        j_start = max(0, i - (n - 1))
        j_end = min(n - 1, i)
        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        rows.extend([idx] * len(js))
        cols.extend(js.tolist())
        vals = a[i - js]
        data.extend(vals.tolist())
    A_active = csr_matrix((data, (rows, cols)), shape=(len(active_idx), n))
    c = -np.ones(n)  # maximize sum(b)
    b_ub = np.full(len(active_idx), M, dtype=np.float64)
    res = linprog(c, A_ub=A_active, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return current a
        return a.copy()
    return res.x
```

"""
Hybrid step‑function optimizer

The routine builds a diverse pool of starter vectors (Dirichlet draws,
exponential / Gaussian / Beta families, spikes, etc.) and runs a
short Adam optimisation on a smooth‑max surrogate.  The incumbent is
then refined with an exponentiated‑gradient sub‑gradient method.
After that we apply a *linear‑programming refinement*: fixing the
current vector a, we maximise the total mass of a new vector b under
the linear constraints (a * b)≤M, where M is the present maximal
convolution value.  By mixing a towards the LP solution we can often
push the maximal convolution a little lower.  Finally a short simulated
annealing / permutation hill‑climb is performed.  The whole process
respects the supplied time budget and always returns the best sequence
found.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# Utility helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) required for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)

# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 30_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            if np.random.rand() < 0.01:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement
# ----------------------------------------------------------------------
def _solve_lp_for_b(a: np.ndarray, M: float, active_idx: np.ndarray) -> np.ndarray:
    """
    Solve   maximize   1ᵀ b
    subject to  (a * b)_i ≤ M   for i in active_idx,
                b ≥ 0.
    Returns the optimal b (may be dense).
    """
    n = a.shape[0]
    rows = []
    cols = []
    data = []
    for row, i in enumerate(active_idx):
        j_start = max(0, i - (n - 1))
        j_end = min(n - 1, i)
        js = np.arange(j_start, j_end + 1, dtype=np.int64)
        rows.extend([row] * js.size)
        cols.extend(js.tolist())
        vals = a[i - js]
        data.extend(vals.tolist())

    A_active = csr_matrix((data, (rows, cols)), shape=(active_idx.size, n))
    c = -np.ones(n)                     # maximise sum(b)  → minimise -sum(b)
    b_ub = np.full(active_idx.size, M, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A_active,
        b_ub=b_ub,
        bounds=[(0, None)] * n,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fallback: return the current vector unchanged
        return a.copy()
    return res.x


def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Iteratively improve a by solving the LP described above and mixing
    towards the LP solution.  Stops when no progress is made or time
    expires.
    """
    a = a0.copy()
    best_val = _evaluate_objective(a, n, conv_fft_len)

    while time.time() < deadline - 1.0:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # active set (indices that are (almost) tight)
        tol = max(1e-12, M * 1e-6)
        active = np.where(conv >= M - tol)[0]

        if active.size == 0:
            break

        b = _solve_lp_for_b(a, M, active)
        if np.allclose(b, a, atol=1e-12):
            break

        b_norm = b / b.sum()
        improved = False
        for t in (0.5, 0.3, 0.2, 0.1, 0.05, 0.02, 0.01):
            a_new = (1.0 - t) * a + t * b_norm
            a_new = np.maximum(a_new, 0.0)
            a_new /= a_new.sum()
            val = _evaluate_objective(a_new, n, conv_fft_len)
            if val < best_val - 1e-9:
                best_val = val
                a = a_new
                improved = True
                print(f"[LP refine] improved to {val:.6f} with t={t}")
                break
        if not improved:
            # cannot improve with current active set; enlarge tolerance a bit
            if tol < M * 1e-3:
                # try a looser active set in the next iteration
                continue
            else:
                break

    return a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = (
            np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
            + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        )
        shape /= shape.sum()
        cand.append(shape)

    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    cand.append(np.full(n, 1.0 / n))

    # ---- new families ----------------------------------------------------
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    tri = np.minimum(np.arange(n) + 1, n - np.arange(n)).astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin

    # problem size – 4096 gives a good trade‑off
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # warm start (if previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # build start pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------- Phase I : quick Adam on many candidates ----------
    explore_deadline = time.time() + (deadline - time.time()) * 0.25
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(explore_deadline, time.time() + 4.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=120_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improvement → {val_adam:.6f}")

        if time.time() < explore_deadline:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                explore_deadline,
                max_iters=20_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Sub‑grad improvement → {val_sub:.6f}")

    # ------------------- Phase II : deep sub‑gradient refinement ----------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Refine] Long sub‑gradient optimisation started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=500_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] New best after sub‑grad → {val_ref:.6f}")

    # ------------------- Phase III : LP‑based refinement -------------------
    if best_seq is not None and time.time() < deadline - 20:
        print("[LP] Starting linear‑programming refinement")
        a_lp, val_lp = _lp_refine(
            best_seq,
            n,
            conv_fft_len,
            deadline - 20,  # reserve some time for later phases
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] Improvement → {val_lp:.6f}")

    # ------------------- Phase IV : simulated annealing -------------------
    if best_seq is not None and time.time() < deadline - 15:
        print("[Anneal] Starting simulated‑annealing moves")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=70_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] Improved best → {val_ann:.6f}")

    # ------------------- Phase V : permutation hill‑climbing ---------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Permute] Starting permutation hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=30_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] New best after permutation → {val_perm:.6f}")

    # fallback
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
