"""
propose_candidate – hybrid global‑parameter search + simple sub‑gradient refinement.

The algorithm works in three stages within the allotted time budget:

1.  **Diverse parametric families** – We generate a large set of candidate
    step‑functions using a handful of analytically‑defined families
    (exponential edges, power‑law edges, double‑Gaussian edges,
    raised sinusoids, random Dirichlet draws and sparse spikes).
    Each candidate is normalised to sum‑to‑one and its exact objective
    `2·n·max(a∗a)` (the denominator is 1) is evaluated via an FFT‑based
    convolution.  The best candidate seen is retained.

2.  **Sub‑gradient (exponentiated‑gradient) refinement** – Starting from the
    best candidate we run a classic exponentiated‑gradient sub‑gradient
    descent on the true non‑smooth objective (the max‑convolution).  This
    stays on the probability simplex and quickly squeezes the bound.

3.  **Targeted pair‑wise annealing** – A fast simulated‑annealing style
    move that looks at the index `k` where the convolution attains its
    maximum and transfers a small fraction of mass away from the most
    responsible pair `(i,j)`.  The move is accepted with a temperature‑
    dependent probability, allowing the algorithm to escape shallow
    plateaus.

All steps respect the security checks of `evaluate_sequence` (non‑negative,
finite entries, sum>0.01).  The final vector is clipped to `[0,1000]` and
returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _conv_max_fft(a: np.ndarray, fft_len: int) -> float:
    """Return max(a∗a) using FFT (linear convolution up to length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2·n‑1 entries (the true linear convolution)
    return float(conv[: 2 * a.shape[0] - 1].max())


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

                          deadline: float,
                          max_iters: int = 500_000,
                          eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    Returns the best vector (on the simplex) and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # convolution and objective
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # sum(a) is 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            grad = np.zeros_like(a)
        else:
            # weight for active convolutions (uniform over active indices)
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            # gradient of weighted sum: 2 * cross‑correlation(w, a)
            rev_a = a[::-1]
            fw = np.fft.rfft(w, fft_len)
            fr = np.fft.rfft(rev_a, fft_len)
            corr = np.fft.irfft(fw * fr, fft_len)
            start = n - 1
            grad = 2.0 * corr[start: start + n]

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)

        # projection back onto simplex (numerical safety)
        a = _project_simplex(a)

        it += 1
    return best_a, best_val


def _pair_targeted_anneal_optimize(a0: np.ndarray,
                                   n: int,
                                   fft_len: int,
                                   deadline: float,
                                   max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted simulated annealing.
    At each step the index k where (a∗a) is maximal is located.
    Among the admissible pairs (i,j) with i+j=k we pick the one with the
    largest product a_i·a_j and move a small fraction of mass away from that
    pair, accepting moves according to a temperature schedule.
    """
    a = a0.copy()
    # pre‑compute convolution once (will be recomputed each move)
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        # linear temperature decay
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        # recompute convolution & locate peak
        fa = np.fft.rfft(a, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        max_c = conv.max()
        kmax = int(conv.argmax())

        # determine the (i,j) pair with the largest product among those with i+j=kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # self‑pair case – move a small fraction elsewhere
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05   # 5 % move
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # pair case – remove a small amount from both ends
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        # protect against negativity and renormalise
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        # evaluate new state
        fa2 = np.fft.rfft(a2, fft_len)
        conv2 = np.fft.irfft(fa2 * fa2, fft_len)[: 2 * n - 1]
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing
    `F = 2·n·max(a∗a) / (∑a)²`.  The routine mixes a broad parametric
    random search with a short exponentiated‑gradient refinement and a final
    pair‑wise targeted annealing phase.  It always returns a valid sequence
    (non‑negative, finite, sum>0.01) before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    n = 4096
    fft_len = _next_pow2(2 * n - 1)          # length for convolution via FFT

    # helper: exact objective (sum of a is assumed to be 1)
    def _objective(a: np.ndarray) -> float:
        return 2.0 * n * _conv_max_fft(a, fft_len)

    # ------------------------------------------------------------------
    # initialise best with a possible warm‑start
    best_seq = None
    best_val = float("inf")
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]), mode="constant")
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            val = _objective(w)
            best_seq = w.copy()
            best_val = val
            print(f"[Warm] initial value {val:.6f}")

    # ------------------------------------------------------------------
    # parametric families for broad exploration
    i_arr = np.arange(n, dtype=np.float64)

    def gen_exp(alpha):
        arr = np.exp(-alpha * i_arr) + np.exp(-alpha * (n - 1 - i_arr))
        return arr / arr.sum()

    def gen_power(p):
        prod = (i_arr + 1) * (n - i_arr)
        arr = prod ** p
        arr = np.maximum(arr, 0)
        return arr / arr.sum()

    def gen_gauss(sigma):
        arr = np.exp(-0.5 * (i_arr / (sigma * n)) ** 2) + \
              np.exp(-0.5 * ((i_arr - (n - 1)) / (sigma * n)) ** 2)
        return arr / arr.sum()

    def gen_sin(power):
        arr = np.sin(np.pi * (i_arr + 0.5) / n)
        arr = np.maximum(arr, 0) ** power
        return arr / arr.sum()

    def gen_random_dir(alpha):
        return np.random.dirichlet(alpha * np.ones(n))

    def gen_spike(k):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        return a

    families = [
        ("exp",    lambda: gen_exp(10 ** np.random.uniform(-3, 2))),
        ("power",  lambda: gen_power(np.random.uniform(-2.5, -0.2))),
        ("gauss",  lambda: gen_gauss(np.random.uniform(0.01, 0.7))),
        ("sin",    lambda: gen_sin(np.random.uniform(0.5, 4.0))),
        ("rand1",  lambda: gen_random_dir(np.random.uniform(0.05, 0.5))),
        ("rand2",  lambda: gen_random_dir(np.random.uniform(0.5, 2.0))),
        ("spike",  lambda: gen_spike(np.random.choice([16, 32, 64, 128, 256, 512]))),
    ]

    # allocate ~60 % of the budget to the global random search
    search_deadline = time.time() + (deadline - time.time()) * 0.60
    evals = 0
    while time.time() < search_deadline:
        # pick a family uniformly at random
        name, sampler = families[np.random.randint(len(families))]
        a = sampler()
        # ensure strict positivity for the objective (tiny epsilon)
        a = np.maximum(a, 0.0)
        a_sum = a.sum()
        if a_sum < 1e-12:
            continue
        a /= a_sum
        val = _objective(a)
        evals += 1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Search {name}] new best {val:.6f} after {evals} evaluations")
        # occasional tiny jitter to escape potential ties
        if evals % 500 == 0 and time.time() > search_deadline - 5:
            a = best_seq * (1.0 + (np.random.rand(n) - 0.5) * 2e-3)
            a = np.maximum(a, 0.0)
            a /= a.sum()
            val = _objective(a)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Jitter] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Sub‑gradient refinement (≈30 % of remaining time)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Refine] sub‑gradient phase")
        sub_dead = deadline - 4  # keep a tiny buffer for final phase
        a_ref, val_ref = _subgradient_optimize(best_seq,
                                                n,
                                                fft_len,
                                                sub_dead,
                                                max_iters=500_000,
                                                eta0=0.9)
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Targeted pair‑wise annealing (use whatever time is left)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Anneal] pair‑wise targeted annealing")
        a_ann, val_ann = _pair_targeted_anneal_optimize(best_seq,
                                                        n,
                                                        fft_len,
                                                        deadline,
                                                        max_moves=200_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety‑net
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
