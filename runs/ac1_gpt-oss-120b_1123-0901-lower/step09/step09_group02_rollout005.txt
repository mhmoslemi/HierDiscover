    w = np.maximum(w, 0)
    w /= w.sum()
    cand.append(w)
    # Blackman
    w = 0.42 - 0.5 * np.cos(2 * np.pi * k / (n-1)) + 0.08 * np.cos(4 * np.pi * k / (n-1))

    cand.append(w)

    w = np.maximum(w, 0)
    w /= w.sum()
    cand.append(w)
    # Gaussian windows with sigma fractions:
    for sigma in (0.12, 0.15, 0.2, 0.25):
        x = np.arange(n)
        mu = (n-1)/2
        w = np.exp(-0.5*((x - mu)/(sigma*n))**2)

        w /= w.sum()

    # Raised cosine (Hann) already.
    # Sine power:
    for p in (0.5, 1.0, 2.0):
        w = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** p

        cand.append(w)
    # Inverted triangular:
    tri = np.minimum(np.arange(n)+1, n-np.arange(n))
    tri = tri.astype(float)
    tri = tri / tri.max()
    inv_tri = 1.0 - tri

"""
stepfunction_optimizer_v3.py

A multi‑restart, multi‑family optimizer for the objective

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

Key ideas added over v2
---------------------------------
1. **Broader starting pool** – besides the families already used we

   Blackman, Nuttall, Kaiser) and a family of U‑shaped Beta densities
   (α<1).  These shapes are known to have very flat autocorrelations and
   often beat the previous best values.

2. **Two‑stage refinement** – a cheap 5 s “quick‑Adam” pass is used to

   phase (up to several hundred k iterations).  The sub‑gradient phase
   drives the true max down, while the final Adam pass (high λ) polishes
   the result.

3. **Longer simulated‑annealing** – more moves and a slower temperature

4. **Adaptive time budgeting** – the total time budget (default 1000 s)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective value (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)

    cand.append(w)

    # Hamming
    w = 0.54 - 0.46 * np.cos(2.0 * np.pi * k / (n - 1))
    w /= w.sum()
    cand.append(w)

    # Blackman
    w = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * k / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * k / (n - 1))
    )
    w /= w.sum()
    cand.append(w)

    # Nuttall
    w = (
        0.355768
        - 0.487396 * np.cos(2.0 * np.pi * k / (n - 1))
        + 0.144232 * np.cos(4.0 * np.pi * k / (n - 1))
        - 0.012604 * np.cos(6.0 * np.pi * k / (n - 1))
    )
    w = np.maximum(w, 0.0)
    w /= w.sum()
    cand.append(w)

    # Kaiser windows with a few betas
    for beta in (0.5, 2.0, 5.0, 8.0):
        x = (2.0 * k) / (n - 1) - 1.0  # [-1,1]
        w = i0(beta * np.sqrt(1.0 - x ** 2)) / i0(beta)
        w = np.maximum(w, 0.0)
        w /= w.sum()
        cand.append(w)

    # Central Gaussian (already present in the old pool, but we keep a few more)
    for sigma in (0.08, 0.12, 0.18, 0.25):
        mu = (n - 1) / 2.0
        w = np.exp(-0.5 * ((k - mu) / (sigma * n)) ** 2)
        w /= w.sum()
        cand.append(w)

    # Raised‑cosine (same as Hann) already covered; we also add sine‑power families
    for p in (0.5, 1.0, 2.0, 3.0):
        w = np.sin(np.pi * (k + 0.5) / n) ** p
        w = np.maximum(w, 0.0)
        w /= w.sum()
        cand.append(w)

    # Inverted triangular (U‑shape)
    tri = np.minimum(k + 1, n - k).astype(np.float64)
    inv_tri = 1.0 - (tri / tri.max())
    inv_tri = np.clip(inv_tri, 0.0, None)
    if inv_tri.sum() > 0:
        inv_tri /= inv_tri.sum()
        cand.append(inv_tri)

    # Symmetric Beta‑type shapes with α<1 (U‑shaped)
    xs = np.arange(1, n + 1, dtype=np.float64) / n
    for alpha in (0.2, 0.3, 0.4, 0.5, 0.7):
        shape = (xs ** (alpha - 1.0)) * ((1.0 - xs) ** (alpha - 1.0))
        # avoid infinities at the boundaries
        shape[np.isinf(shape)] = 0.0
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    return cand


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified starter pool (includes the old families)."""
    cand = []

    # Warm start and its cyclic shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with varied concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Edge‑polynomial families (negative exponents)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian (peaks at both ends)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Central Gaussian (single peak) – extra sigmas
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric Beta‑type shapes (α = β > 1, centred)
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        s = shape.sum()
        if s > 0:
            shape /= s
            cand.append(shape)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Interleaved permutation of every candidate (helps spreading large mass)
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    # ------------------------------------------------------------------
    # NEW families (window based + U‑shaped Beta)
    cand.extend(_window_candidates(n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns (equally spaced or random)."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    # random‑position sparse spikes
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


# ----------------------------------------------------------------------
# Optimisation phases (slightly tuned hyper‑parameters)
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase λ)
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter when we are close to the deadline
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing with mass‑transfer and swaps."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # uphill acceptance
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 30_000,
) -> tuple[np.ndarray, float]:
    """Pure permutation hill‑climbing (mostly downhill)."""
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance (1 % chance)
            if np.random.rand() < 0.01:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]   # revert

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the user‑supplied time budget (default 1000 s)
    and always returns the best sequence it found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – 4096 is a good trade‑off, but we will switch to 8192
    # if enough budget remains (the algorithm automatically scales).
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a rich starter pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – rapid cheap refinement of many candidates (≈20 % of budget)
    explore_deadline = time.time() + (deadline - time.time()) * 0.20
    quick_iters = 25_000          # modest Adam run for each candidate
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(explore_deadline, time.time() + 4.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=quick_iters
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] improved to {val_adam:.6f}")

        # If we still have a couple of seconds, run a tiny sub‑gradient polish
        if time.time() < explore_deadline - 2.0:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, explore_deadline,
                max_iters=10_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore‑sub] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deep sub‑gradient optimisation of incumbent (≈45 % budget)
    if best_seq is not None and time.time() < deadline - 60:
        print("[Refine] long sub‑gradient optimisation started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] new best after sub‑grad → {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – final Adam polishing with very high λ (≈10 % budget)
    if best_seq is not None and time.time() < deadline - 15:
        print("[Polish] final high‑λ Adam")
        polish_dead = min(deadline, time.time() + 12.0)
        a_pol, val_pol = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            polish_dead,
            max_iters=120_000,
        )
        if val_pol < best_val:
            best_val = val_pol
            best_seq = a_pol.copy()
            print(f"[Polish] improved to {val_pol:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated‑annealing (≈15 % budget)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] starting simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=90_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] new best → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – permutation hill‑climbing (≈5 % budget)
    if best_seq is not None and time.time() < deadline:
        print("[Permute] hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=35_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] final best → {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Fallback – in the unlikely event no vector was produced
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
