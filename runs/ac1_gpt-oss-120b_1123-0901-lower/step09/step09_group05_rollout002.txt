    a = _project_onto_simplex(np.maximum(a0, 0))
    best_a = a.copy()
    best_val = float('inf')
    it=0
    step = 0.5
    alpha = init_alpha

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[SmoothGD it={it}] new best {val:.6f} alpha={alpha:.4f}")

        # compute weight via softmax
        # if alpha>0 else use active set
        scaled = conv * alpha
        max_scaled = scaled.max()
        exps = np.exp(scaled - max_scaled)
        w = exps / exps.sum()
        # gradient of weighted average
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # line search
        # adapt step size as before, maybe momentum too.

        # optional: increase alpha gradually to focus more on true max.
        if it % 1000 == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_factor, max_alpha)

        # momentum:

            vel = np.zeros_like(a)

            a = a_new

"""
propose_candidate
-----------------

Search for a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The algorithm consists of three phases:

1. **Diverse seeding** – a large pool of candidate vectors is built from
   many deterministic families (uniform, sinusoid, edge‑exponential,
   edge‑power‑law, arcsine, log‑shape, polynomial‑edge, beta‑pdf,
   double‑Gaussian) together with random Dirichlet draws and cyclic
   shifts of a possible warm‑start.  All candidates are normalised to
   unit ℓ₁‑norm and the exact objective is evaluated (via FFT).  The
   best ``top_k`` vectors are retained.

2. **Smoothed projected gradient descent** – each retained vector is
   refined with a gradient descent that minimises a *smooth* approximation
   of the max‑convolution.  The smooth weight is a soft‑max over the
   convolution entries, controlled by a temperature parameter ``alpha``
   that is increased during the run.  Gradients are obtained by a
   cross‑correlation (FFT‑based) and each step is projected back onto the
   probability simplex.  This tends to escape the tiny active‑set of the
   pure max‑subgradient and yields a lower final peak.

3. **Cheap random‑move hill‑climber** – the best vector from phase 2 is
   further polished by many local mass‑transfer moves that are kept only
   when they improve the true objective.

The whole procedure respects the global time budget (default 1000 s) and
returns a Python list ready for ``evaluate_sequence``.  The implementation
is fully self‑contained, uses only NumPy and the standard library, and
does not rely on external solvers.
"""

import time
import numpy as np

# Utility functions (FFT helpers, simplex projection, etc.)
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Deterministic families (all return vectors with ℓ₁‑norm == 1)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _arc_sine(n: int) -> np.ndarray:
    """Arcsine‑type shape, heavy at the boundaries."""
    i = np.arange(n, dtype=np.float64) + 0.5
    shape = 1.0 / np.sqrt(i * (n - i + 1.0))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _log_shape(n: int, c: float) -> np.ndarray:
    """Logarithmic shape: a_i ∝ max(0, -log(i/(n+1)) + c)."""
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = -np.log(i / (n + 1.0)) + c
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _poly_edge(n: int, p: float) -> np.ndarray:
    """Polynomial edge‑heavy shape: (i·(n‑i+1))**p, p<0."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = (i * (n - i + 1.0)) ** p
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _random_piecewise(n: int, rng: np.random.Generator) -> np.ndarray:
    """Random piecewise‑linear shape."""
    k = rng.integers(3, 7)                 # number of breakpoints
    pts = np.sort(rng.integers(0, n, size=k))
    vals = rng.random(k) + 0.05            # avoid zeros
    shape = np.interp(np.arange(n), pts, vals, left=vals[0], right=vals[-1])
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse seed pool (deterministic families + random Dirichlet)."""
    pool = []

    # Dirichlet
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(2):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (coarse)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law (coarse)
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf (few)
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple families
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # New families
    pool.append(_arc_sine(n))
    for c in (0.0, 0.1, 0.2):
        pool.append(_log_shape(n, c))
    for p in np.linspace(-5.0, -0.3, 12):
        pool.append(_poly_edge(n, p))

    # Random piecewise
    for _ in range(6):
        pool.append(_random_piecewise(n, rng))

    return pool


def _grid_candidates(n: int) -> list[np.ndarray]:

# Smoothed projected gradient descent (with soft‑max weight)
def _smooth_gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    init_alpha: float = 0.1,
    alpha_mult: float = 1.5,
    max_alpha: float = 200.0,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on a smooth approximation of the max‑convolution.
    Returns the best vector seen and its exact objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")

    # momentum parameters
    vel = np.zeros_like(a)
    beta = 0.9

    # learning‑rate controls
    init_step = 0.8
    step = init_step

    alpha = init_alpha
    it = 0

    while time.time() < deadline and it < max_iters:
        # ---- evaluate current objective ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 2000 == 0:
                print(f"[SmoothGD it={it:06d}] new best {val:.6f}  α={alpha:.3f}")

        # ---- compute smoothed weight (soft‑max) ----
        scaled = conv * alpha
        mx = scaled.max()
        exps = np.exp(scaled - mx)          # stable soft‑max
        w = exps / exps.sum()               # sum(w)=1, w ≥ 0

        # ---- gradient of weighted average ----
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- momentum update ----
        vel = beta * vel + (1.0 - beta) * grad

        # ---- back‑tracking line‑search ----
        step_try = step
        while step_try > 1e-12:
            a_new = a - step_try * vel
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step_try *= 0.5

        # adapt step size a bit (slow decay)
        step = max(step * 0.9995, 1e-12)

        # increase smoothing parameter gradually
        if it % 1000 == 0 and alpha < max_alpha:
            alpha = min(alpha * alpha_mult, max_alpha)

        it += 1

    return best_a, best_val

# Very cheap hill‑climber (random mass moves)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Random mass‑transfer hill‑climber that keeps any improvement of the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.35)   # up to 35 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point: generate a step‑function (list of non‑negative heights)
    that (approximately) minimises the evaluation function within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # problem size – larger n gives finer granularity but also larger FFTs
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # warm start (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a rich seed pool
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng) + _grid_candidates(n)

    # --------------------------------------------------------------
    # Quick evaluation of the pool and keep the best few
    top_k = 30
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max() / (a.sum() ** 2)
        scored.append((val, a))
        if time.time() >= deadline * 0.2:   # keep some time for refinement
            break

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # --------------------------------------------------------------
    # Phase 2 – smoothed projected gradient descent on elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 12:
            break
        # allocate a decreasing share of remaining budget
        remaining = deadline - time.time()
        # simple harmonic sharing: later ranks get less
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _smooth_gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=150_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final cheap local moves
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
