"""
Step‑function optimizer – 1000 s budget.

The routine works in three phases:

1️⃣  **Broad parametric sweep** – a large collection of
    monotone / symmetric families (exponential, power‑law,
    stretched‑exponential, beta, double‑Gaussian, random spikes …)
    is evaluated on a *large* grid (n = 8192).  Every vector is
    normalised to sum = 1 and the exact objective
    `2·n·max(conv)/sum²` (no smoothing) is computed with an FFT‑based
    convolution.  The best sequence together with its generating family
    and parameters is stored.

2️⃣  **Local refinement** – starting from the best family we perform a
    cheap stochastic line‑search on its parameters (log‑normal
    perturbations).  The search is stopped as soon as the global time
    budget is reached.

3️⃣  **Exact‑max sub‑gradient (exponentiated‑gradient)** – a long mirror‑descent
    run on the simplex drives the vector to a (near) optimum of the
    original non‑smooth objective.  The update is
        a ← a·exp(−η·∇maxconv)
    with a diminishing step size ηₜ = η₀/√(t+1).  The current best
    objective is reported whenever it improves.

The whole procedure always returns a valid, non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (needed for sub‑gradient)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of 2·n·max(a*a)/(sum a)² (scale‑invariant)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(−α·i) + exp(−α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution on [0,1] (continuous)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(−α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes with Dirichlet masses."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (np.random.choice(n, size=k, replace=False)
               if random else np.round(np.linspace(0, n - 1, k)).astype(int))
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Sub‑gradient optimiser (exponentiated‑gradient) -----------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent on the simplex."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # indices where the max is attained
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of the max‑convolution
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

    n = 8192

    best_seq = None

        nonlocal best_seq, best_val, best_fam, best_params
        val = _objective(a, n)
        if val < best_val:
            best_val = val

            _maybe_update(a, "spike", {"k": k})
    # end of sweep -----------------------------------------------------

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------

    #   (only if enough time is left)

        # a simple schedule: 3000 attempts or until time runs out
        attempts = 3000
        for _ in range(attempts):
            now = time.time()
            if now >= deadline - 5:
                break

            # perturb the stored parameters (log‑normal multiplicative noise)

                alpha = best_params["alpha"] * (10 ** np.random.normal(0, 0.07))
                alpha = max(alpha, 1e-8)
                a = _candidate_left_exponential(alpha, n) if best_fam == "left_exp" else _candidate_exponential(alpha, n)
                _maybe_update(a, best_fam, {"alpha": alpha})

            elif best_fam in ("left_pow", "power"):
                p = best_params["p"] * (10 ** np.random.normal(0, 0.07))
                a = _candidate_left_power(p, n) if best_fam == "left_pow" else _candidate_power(p, n)
                _maybe_update(a, best_fam, {"p": p})

            elif best_fam == "stretched":
                alpha = best_params["alpha"] * (10 ** np.random.normal(0, 0.07))
                beta = best_params["beta"] * (10 ** np.random.normal(0, 0.07))
                alpha = max(alpha, 1e-8)
                beta = max(beta, 0.1)
                a = _candidate_stretched(alpha, beta, n)
                _maybe_update(a, "stretched", {"alpha": alpha, "beta": beta})

            elif best_fam == "beta":
                a_par = best_params["a"] * (10 ** np.random.normal(0, 0.06))
                b_par = best_params["b"] * (10 ** np.random.normal(0, 0.06))
                a_par = max(a_par, 0.05)
                b_par = max(b_par, 0.05)
                a = _candidate_beta(a_par, b_par, n)
                _maybe_update(a, "beta", {"a": a_par, "b": b_par})

                sig = best_params["sig"] * (10 ** np.random.normal(0, 0.07))
                sig = max(sig, 1e-5)

                _maybe_update(a, "double_gauss", {"sig": sig})

            else:
                # fallback: random Dirichlet perturbation of the current best
                a = np.random.dirichlet(best_seq * 5 + 1e-3)
                a /= a.sum()
                _maybe_update(a, "dirichlet", {})

    # ------------------------------------------------------------------
    # 3) Long sub‑gradient descent (exact max) -------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Subgrad] long exact‑max optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity – ensure a valid output ---------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # optional sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
