"""
Propose a low‑score non‑negative step function.

Algorithm
---------
1. **Warm‑start** – if a previous best sequence is available in the global
   variable ``height_sequence_1`` it is taken, normalised and used as a seed.

2. **Exploratory parametric sweep** – a large number of candidate shapes are
   drawn from several low‑dimensional families (asymmetric exponential,
   edge‑power‑law, beta, double‑Gaussian, sinusoid‑power, tent, Dirichlet).
   The search space for the parameters is deliberately very wide
   (log‑uniform where sensible).  Each candidate is evaluated with a fast FFT‑
   based convolution; the best candidate is retained.

3. **Fine‑tune the parameters of the current best family** – a simple
   hill‑climber perturbs the parameters of the family that produced the best
   candidate (Gaussian steps on the log‑scale) and accepts improvements.

4. **Gradient based refinement** – three complementary optimisers
   (Adam on a smooth‑max surrogate, exponentiated‑gradient sub‑gradient,
   Euclidean projected‑gradient) are run sequentially on the best
   vector.  They operate directly on the exact max‑convolution objective
   and keep the best iterate seen.

5. **Stochastic local search** – in the last seconds a cheap mass‑transfer
   hill‑climber moves a random fraction (≤10 %) of probability mass from one
   coordinate to another and accepts any improvement.

All vectors are kept on the probability simplex (non‑negative and
normalised), which makes the objective scale‑invariant.  The final sequence
is clipped to the bounds accepted by ``evaluate_sequence`` and returned as a
plain list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (full length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the valid portion
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradient calculations.
    Returns a vector of length len(a) corresponding to the convolution
    of w with the reversed a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence up to clamping step) ---
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using an FFT based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Shape generators -------------------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    """Symmetric exponential drop‑off from both ends."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_asym_exponential_shape(
    n: int, gamma_l: float, gamma_r: float, weight_l: float
) -> np.ndarray:
    """
    Asymmetric exponential shape: left side uses gamma_l, right side uses
    gamma_r, combined with left weight `weight_l` (right weight = 1‑weight_l).
    """
    left = np.exp(-gamma_l * np.arange(n, dtype=np.float64))
    right = np.exp(-gamma_r * np.arange(n - 1, -1, -1, dtype=np.float64))
    shape = weight_l * left + (1.0 - weight_l) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    """Power‑law (inverse) fall‑off from both ends."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta distribution on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    """Sum of two Gaussian bells centred at the borders."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    """sin(π·x) raised to a power, symmetric."""
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0.0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    """Triangular (tent) shape."""
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0.0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet sample with concentration `alpha`."""
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

# Optimisers (same philosophy as the reference implementation) ---------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02            # start temperature (inverse)
    lam_factor = 1.01     # slower rise for stability

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (log‑soft‑max) step
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # tiny random perturbation when the deadline is close
        if it % 5000 == 0 and time.time() > deadline - 30.0:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climbing in the last seconds."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer at most 10 % of the source mass (fine‑grained)
        delta = a[i] * (np.random.rand() * 0.1)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function (list of heights) that minimises
    evaluate_sequence(.) under a 1000 s budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # small safety margin

    # Problem size – 4096 offers a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")
    best_family = None
    best_params = None

    # ------------------------------------------------------------------
    # Phase 1 – rich random parametric sampling
    families = [
        "edge_exp",
        "edge_asym_exp",
        "edge_pow",
        "beta",
        "double_gauss",
        "sin_pow",
        "tent",
        "dirichlet",
    ]
    # allocate roughly 0.3 × the total budget to this phase
    phase1_deadline = min(deadline - 5.0, time.time() + 0.3 * (deadline - time.time()))
    print("Phase 1 – random parametric sweep")

    while time.time() < phase1_deadline:
        fam = rng.choice(families)

        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 2)       # 0.01 … 100
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
            params = (gamma, weight)

        elif fam == "edge_asym_exp":
            gamma_l = 10 ** rng.uniform(-2, 2)
            gamma_r = 10 ** rng.uniform(-2, 2)
            weight_l = rng.random()
            cand = _edge_asym_exponential_shape(n, gamma_l, gamma_r, weight_l)
            params = (gamma_l, gamma_r, weight_l)

        elif fam == "edge_pow":
            alpha = rng.uniform(-20.0, -0.01)       # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
            params = (alpha, weight)

        elif fam == "beta":
            # broaden the range enormously (including U‑shaped extremes)
            a = 10 ** rng.uniform(-3, 1)            # 0.001 … 10
            b = 10 ** rng.uniform(-3, 1)
            cand = _beta_shape(n, a, b)
            params = (a, b)

        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.6)        # allow very narrow / wide bells
            cand = _double_gaussian_shape(n, sigma)
            params = (sigma,)

        elif fam == "sin_pow":
            power = rng.uniform(0.2, 8.0)
            cand = _sinusoid_power_shape(n, power)
            params = (power,)

        elif fam == "tent":
            cand = _tent_shape(n)
            params = None

        else:  # dirichlet
            alpha = rng.uniform(0.02, 8.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
            params = (alpha,)

        val = _objective(cand, conv_fft_len)

        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            best_family = fam
            best_params = params
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # ------------------------------------------------------------------
    # Phase 1‑b – local refinement of the parameters that produced the best shape
    if best_family is not None and time.time() < phase1_deadline:
        print("Phase 1‑b – fine‑grained parameter hill‑climb")
        # number of tiny perturbations we can afford
        param_iters = 2000
        for _ in range(param_iters):
            if time.time() > phase1_deadline:
                break
            # propose a Gaussian perturbation in log‑space where appropriate
            if best_family == "edge_exp":
                gamma, weight = best_params
                lg = np.log10(gamma) + rng.normal(scale=0.1)
                gamma1 = 10 ** np.clip(lg, -4, 4)
                weight1 = np.clip(weight + rng.normal(scale=0.05), 0.0, 1.0)
                cand = _edge_exponential_shape(n, gamma1, weight1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (gamma1, weight1)

            elif best_family == "edge_asym_exp":
                gL, gR, wL = best_params
                lgL = np.log10(gL) + rng.normal(scale=0.1)
                lgR = np.log10(gR) + rng.normal(scale=0.1)
                gammaL1 = 10 ** np.clip(lgL, -4, 4)
                gammaR1 = 10 ** np.clip(lgR, -4, 4)
                wL1 = np.clip(wL + rng.normal(scale=0.05), 0.0, 1.0)
                cand = _edge_asym_exponential_shape(n, gammaL1, gammaR1, wL1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (gammaL1, gammaR1, wL1)

            elif best_family == "edge_pow":
                alpha, weight = best_params
                alpha1 = alpha + rng.normal(scale=0.5)  # keep sign negative
                if alpha1 > -0.001:
                    alpha1 = -0.001
                weight1 = np.clip(weight + rng.normal(scale=0.05), 0.0, 1.0)
                cand = _edge_power_law_shape(n, alpha1, weight1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (alpha1, weight1)

            elif best_family == "beta":
                a, b = best_params
                la = np.log10(a) + rng.normal(scale=0.15)
                lb = np.log10(b) + rng.normal(scale=0.15)
                a1 = 10 ** np.clip(la, -5, 2)
                b1 = 10 ** np.clip(lb, -5, 2)
                cand = _beta_shape(n, a1, b1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (a1, b1)

            elif best_family == "double_gauss":
                sigma, = best_params
                sigma1 = np.clip(sigma + rng.normal(scale=0.02), 0.001, 1.0)
                cand = _double_gaussian_shape(n, sigma1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (sigma1,)

            elif best_family == "sin_pow":
                power, = best_params
                power1 = np.clip(power + rng.normal(scale=0.2), 0.1, 12.0)
                cand = _sinusoid_power_shape(n, power1)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (power1,)

            elif best_family == "tent":
                # tent shape has no parameters – try a tiny random perturbation of the vector itself
                cand = best_seq.copy()
                idx = rng.choice(n, size=2, replace=False)
                delta = cand[idx[0]] * rng.uniform(0.0, 0.05)
                cand[idx[0]] -= delta
                cand[idx[1]] += delta
                cand = _project_onto_simplex(cand)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()

            else:  # dirichlet
                alpha, = best_params
                alpha1 = np.clip(alpha + rng.normal(scale=0.2), 0.001, 10.0)
                cand = _random_dirichlet_shape(n, alpha1, rng)
                val = _objective(cand, conv_fft_len)
                if val < best_val:
                    best_val, best_seq = val, cand.copy()
                    best_params = (alpha1,)

    # ------------------------------------------------------------------
    # Phase 2 – deep gradient refinement (Adam → Sub‑gradient → PGD)
    if best_seq is not None and time.time() < deadline - 20:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

        if time.time() < deadline - 12:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=250_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Phase 2] Sub‑grad improved to {val_sub:.6f}")

        if time.time() < deadline - 6:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.45,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[Phase 2] PGD improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing for the final seconds
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – very unlikely, but keep a valid object
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety clipping (matches evaluate_sequence limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
