"""
Propose a non‑negative step function that (approximately) minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2) .

The search is performed in three stages:

1️⃣  **Broad generation** – a large pool of candidates is built from many
    parametric families (uniform, sinusoid, symmetric / one‑sided
    exponentials, power‑laws, beta‑pdfs, random Dirichlet draws and
    mixtures with the uniform vector).  The families are explored on a
    very fine grid for the decay parameters (γ down to 1e‑5) so that the
    sweet spot that lowers ``max(a∗a)`` below the uniform bound can be
    discovered.

2️⃣  **Projected gradient refinement** – each elite candidate is run
    through an exact projected‑gradient descent on the true objective.
    The gradient of the *max* convolution is obtained from the active
    set (indices attaining the current maximum) and the back‑tracking
    line‑search guarantees a monotone decrease.

3️⃣  **Heavy‑weight local search** – starting from the best refined
    vector we perform a cheap but massive hill‑climber that repeatedly
    transfers a random fraction of mass between two positions and
    accepts the move only if the objective improves.  This phase
    consumes all remaining time.

All phases respect the global 1000 s wall‑clock budget; the best
sequence found is returned as a plain ``list`` that can be passed to
``evaluate_sequence`` directly.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient of
    max(a∗a).  ``w`` is the (normalised) weight vector that indicates the
    active indices.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families (all return vectors *with* sum == 1) -------------

def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric “U‑shaped’’ exponential (both edges)."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_exponential_one_sided(n: int, gamma: float) -> np.ndarray:
    """Monotone decreasing exponential, normalised to sum = 1."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law (U‑shaped) a_i = (i+1)^α + (n-i)^α."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _edge_power_law_one_sided(n: int, alpha: float) -> np.ndarray:
    """Monotone power‑law a_i = (i+1)^α, normalised."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = np.maximum(idx ** alpha, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑pdf on the discrete grid, normalised."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Dirichlet draw with concentration ``alpha`` (returns a probability vector)."""
    return rng.dirichlet(np.full(n, alpha))


def _mix(shape1: np.ndarray, shape2: np.ndarray, lam: float) -> np.ndarray:
    """Convex mixture λ·shape1 + (1‑λ)·shape2, re‑normalised."""
    mixed = lam * shape1 + (1.0 - lam) * shape2
    mixed = np.maximum(mixed, 0.0)
    s = mixed.sum()
    if s > 0:
        mixed /= s
    else:
        mixed = np.full_like(mixed, 1.0 / mixed.shape[0])
    return mixed


# ----------------------------------------------------------------------
# Gradient descent on the exact objective --------------------------------

def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on 2·n·max(a∗a)/(sum a)^2."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    step0 = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set for the max element(s)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # back‑tracking line search (projected)
        step = step0
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)
            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple cheap hill‑climber ----------------------------------------------

def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Transfer a random fraction of mass from i → j and keep improvements."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main proposal function -------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Entry point for the judge.  Returns the best non‑negative step function
    found within the allotted wall‑clock budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    # leave a 0.5 s safety margin for the judge
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # problem size – using 4096 (the size that the reference solution used)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start ------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 1️⃣  broad generation of candidates
    pool = []

    # optional warm start and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families ------------------------------------------------
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))

    # fine grid for the exponential families (both symmetric and one‑sided)
    for gamma in np.logspace(-5, -0.5, 30):       # 1e‑5 … ≈0.3
        pool.append(_edge_exponential(n, float(gamma)))
        pool.append(_edge_exponential_one_sided(n, float(gamma)))

    # fine grid for power‑law families
    for alpha in np.linspace(-6.0, 0.0, 30):
        pool.append(_edge_power_law(n, float(alpha)))
        pool.append(_edge_power_law_one_sided(n, float(alpha)))

    # beta‑pdf grid (symmetric but very flexible)
    a_vals = np.logspace(-1, 1, 12)   # 0.1 … 10
    b_vals = np.logspace(-1, 1, 12)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    # random Dirichlet draws (various concentration levels)
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(4):
            pool.append(_random_dirichlet(n, alpha, rng))

    # mixtures with the uniform vector (adds an extra degree of freedom)
    uniform_vec = _uniform(n)
    for shape in list(pool):          # copy because we will extend pool
        lam = rng.uniform(0.0, 1.0)
        pool.append(_mix(shape, uniform_vec, lam))

    # ------------------------------------------------------------------
    # quick evaluation – keep the top‑k candidates
    top_k = 30
    scored = []
    for cand in pool:
        if time.time() >= deadline * 0.55:          # reserve time for refinement
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        scored.append((val, a))

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # 2️⃣  refined projected gradient descent on the elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 10:
            break
        # allocate a decreasing share of the remaining budget
        remaining = deadline - time.time()
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=400_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣  final cheap local search (use any remaining time)
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=800_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # fallback – never return an empty list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
