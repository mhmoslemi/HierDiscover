def hill_climb(a, conv, n, deadline, max_iters=None):
    best_a = a.copy()
    best_conv = conv.copy()
    best_val = 2.0*n*best_conv.max()
    it = 0
    while time.time() < deadline:
        # optionally break if max_iters reached
        if max_iters is not None and it >= max_iters: break

        if a[i] < 1e-15:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        # choose delta fraction of a[i]
        delta = a[i] * np.random.rand() * 0.5  # up to 50% of mass
        if delta <= 0:
            continue
        # compute new conv using efficient update
        conv2 = best_conv.copy()

        conv2[i:i+n] -= 2.0*delta * a
        conv2[j:j+n] += 2.0*delta * a

        idx2j = 2*j
        if idx2i < len(conv2):
            conv2[idx2i] += delta*delta

            conv2[idx2j] += delta*delta

        a2[j] += delta
        # compute new max

            best_a = a2.copy()
            best_conv = conv2.copy()

"""
Hill‑climbing + simulated‑annealing optimizer for the step‑function bound.

The algorithm works on the probability simplex (the sequence is kept non‑negative
and Σa = 1, which leaves the evaluation invariant).  It proceeds in three
phases:

1.  A diverse set of start vectors (uniform, Dirichlet draws, spike
    patterns, the previous solution if any) is built.
2.  Each start is refined by a fast O(n) hill‑climbing loop that updates the
    convolution incrementally when a small amount of mass is moved from one
    coordinate to another.  A simple simulated‑annealing schedule allows the
    walk to escape plateaus.
3.  The current global best undergoes a short “exact‑evaluation’’ polishing
    phase (full np.convolve) to capture the tiny improvements needed to beat
    the 1.503 target.

All heavy work uses NumPy FFT for the convolution so a few hundred thousand
updates fit comfortably into the 1000 s budget.  The routine always returns
the best sequence found before the deadline; the returned list consists of
non‑negative values (≤ 1000) and therefore passes the supplied
`evaluate_sequence` checks.
"""

import time
import numpy as np
from scipy.optimize import linprog  # not used directly but kept for potential extensions


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2*len(a)-1) using FFT.
    Returns only the valid part of length 2*n-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    n = a.shape[0]
    return conv[: 2 * n - 1]


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Convenient wrapper that chooses a suitable FFT length."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    return _full_convolution(a, fft_len)


# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _hill_climb(a0: np.ndarray,
                conv0: np.ndarray,
                n: int,
                deadline: float,
                max_iters: int | None = None) -> tuple[np.ndarray, float]:
    """
    Fast O(n) hill‑climbing / simulated‑annealing on the exact max‑convolution.
    Moves a random fraction of mass from one coordinate to another,
    updates the convolution incrementally, and accepts the move if it improves
    the objective or with a temperature‑dependent probability.
    Returns the best vector and its objective (2*n*max(conv)).
    """
    cur_a = a0.copy()
    cur_conv = conv0.copy()
    cur_val = 2.0 * n * cur_conv.max()
    best_a = cur_a.copy()
    best_conv = cur_conv.copy()
    best_val = cur_val

    # annealing schedule
    T_start = 0.05
    T_end = 1e-5
    it = 0

    while time.time() < deadline:
        if max_iters is not None and it >= max_iters:
            break

        # pick source i with non‑negligible mass and a different target j
        i = np.random.randint(n)
        if cur_a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a random fraction (up to 30 %) of the source mass
        delta = cur_a[i] * np.random.rand() * 0.30
        if delta <= 0.0:
            continue

        # ---- compute the updated convolution (incremental) ----
        # start from the current convolution
        new_conv = cur_conv.copy()

        # effect of moving δ from i to j (linear part)
        new_conv[i:i + n] -= 2.0 * delta * cur_a
        new_conv[j:j + n] += 2.0 * delta * cur_a

        # quadratic δ² contribution
        idx2i = 2 * i
        idx2j = 2 * j
        if idx2i < new_conv.size:
            new_conv[idx2i] += delta * delta
        if idx2j < new_conv.size:
            new_conv[idx2j] += delta * delta
        idxij = i + j
        if idxij < new_conv.size:
            new_conv[idxij] -= 2.0 * delta * delta

        # ---- build the new vector ----
        new_a = cur_a.copy()
        new_a[i] -= delta
        new_a[j] += delta

        # sanity: projection onto simplex to correct tiny round‑off drift
        new_a = _project_simplex(new_a)

        new_val = 2.0 * n * new_conv.max()

        # accept if improving, otherwise with a temperature‑dependent prob.
        accepted = False
        if new_val < best_val - 1e-12:           # strict improvement
            best_val = new_val
            best_a = new_a.copy()
            best_conv = new_conv.copy()
            accepted = True
        else:
            # annealing probability
            T = max(T_end,
                    T_start * (1.0 - it / (max_iters if max_iters else 200_000)))
            if np.random.rand() < np.exp(-(new_val - best_val) / T):
                accepted = True

        if accepted:
            cur_a = new_a
            cur_conv = new_conv

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """
    Build a modest but diverse pool of starting vectors.
    """
    cand = []

    # warm start (previous best) – also add a few cyclic shifts
    if warm is not None:
        warm = warm.copy()
        cand.append(warm)
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # uniformly random Dirichlet draws (different concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # shuffled Dirichlet variants
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        np.random.shuffle(a)
        cand.append(a)

    # spike patterns (equally spaced)
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # random sparse spikes
    for _ in range(4):
        k = np.random.choice([32, 64, 128])
        a = np.zeros(n)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        cand.append(a)

    # half‑wave sinusoid (already used in the reference)
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # uniform
    cand.append(np.full(n, 1.0 / n))

    return cand
