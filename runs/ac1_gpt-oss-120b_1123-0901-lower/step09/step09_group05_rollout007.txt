def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000):
    a = a0.copy()
    a = np.maximum(a, 0.)
    if a.sum() == 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05  # base learning rate
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    # schedule for beta parameter for softmax
    beta_start = 0.5
    beta_end = 20.0
    while time.time() < deadline and it < max_iters:
        # compute convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)  # sum=1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")
        # compute softmax weights
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)
        # avoid overflow: subtract max
        scaled = beta * conv
        max_scaled = scaled.max()
        exp_scaled = np.exp(scaled - max_scaled)
        weight = exp_scaled / exp_scaled.sum()
        # compute gradient
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        # project onto simplex
        a_new = _project_onto_simplex(a_new)
        a = a_new
        it += 1
    # compute final val for best
    return best_a, best_val

"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2) .

Algorithm
---------
1. **Warm start & diverse seed pool** – include the previous best
   (`height_sequence_1` if present), a few deterministic families
   (uniform, sinusoid, edge‑exponential, edge‑power‑law, beta‑pdf)
   and a handful of random Dirichlet draws.

2. **Quick sweep of parametric families** – dense grids for the
   edge‑exponential, edge‑power‑law and beta families are evaluated
   with a single FFT each; the best vector so far is retained.

3. **Smooth‑max Adam optimisation** – the non‑smooth max is replaced
   by a log‑sum‑exp (soft‑max) with a temperature β that is
   increased linearly during optimisation.  Adam updates are applied
   and after each step the vector is projected onto the probability
   simplex (ensuring non‑negativity and unit sum).

4. **Exact sub‑gradient descent** – using the true max‑objective a short
   projected sub‑gradient phase (identical to the reference implementation)
   further refines the best candidate.

5. **Cheap random hill‑climber** – finally a few hundred thousand
   mass‑transfer moves are tried; any improvement is kept.

All stages respect a global 1000 s wall‑clock budget.  The best
sequence found is returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers ------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection ------------------------------------------------------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex (sum=1, ≥0)."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families (all return vectors with sum == 1) ---------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Smooth‑max Adam optimiser -----------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation of the smooth approximation
        L_β(a) = (1/β)·log∑_k exp(β·(a∗a)_k)
    The schedule for β goes from 0.5 to 20.
    Returns the best vector seen and its *true* objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    a /= a.sum()

    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1, beta2 = 0.9, 0.999
    eps = 1e-8
    base_lr = 0.05

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    beta_start, beta_end = 0.5, 20.0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {cur_val:.6f}")

        # temperature schedule
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        # soft‑max weights
        scaled = beta * conv
        offset = scaled.max()
        exp_scaled = np.exp(scaled - offset)
        weight = exp_scaled / exp_scaled.sum()

        # gradient of the smooth loss
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad
        v = beta2 * v + (1 - beta2) * (grad * grad)

        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = base_lr * m_hat / (np.sqrt(v_hat) + eps)
        a_new = a - step
        a = _project_onto_simplex(a_new)

        it += 1

    # final evaluation of the best point
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (best_a.sum() ** 2)
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact projected sub‑gradient descent (as in the reference code) ------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* objective.
    Returns the best vector encountered and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap random hill‑climber -----------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Transfer a random fraction of mass from i→j and keep any improvement."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point called by the judge.
    Returns the best non‑negative step function found within the time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    corr_fft_len = _next_pow2(3 * n - 2)

    warm_seq = None
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)

            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            warm_seq = w / w.sum()

    # ----- build a modest initial pool -------------------------------
    pool = []
    if warm_seq is not None:

        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm_seq, shift))

    pool.extend([
        _uniform(n),
        _sinusoid(n),
        _edge_exponential(n, 0.01),
        _edge_exponential(n, 0.05),
        _edge_exponential(n, 0.1),
        _edge_exponential(n, 0.2),
        _edge_exponential(n, 0.5),
        _edge_exponential(n, 1.0),
        _edge_exponential(n, 2.0),
        _edge_exponential(n, 5.0),
        _edge_exponential(n, 10.0),
        _edge_power_law(n, -4),
        _edge_power_law(n, -2),
        _edge_power_law(n, -1),

        _edge_power_law(n, 0),

    # a few random Dirichlet draws

    # ------------------------------------------------------------------

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
