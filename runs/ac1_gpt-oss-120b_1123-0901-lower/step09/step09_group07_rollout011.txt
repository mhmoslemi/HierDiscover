def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5

    # Choose n based on time left: If budget > 800 -> n=8192 else 4096. Use 4096 if less.
    n = 4096
    if budget_s > 600:
        n = 8192

    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # Warm start from globals
    warm = globals().get("height_sequence_1")
    warm_seq = None
    if warm is not None:
        warm_seq = np.asarray(warm, dtype=np.float64)
        if warm_seq.shape[0] > n:
            warm_seq = warm_seq[:n]
        elif warm_seq.shape[0] < n:
            warm_seq = np.pad(warm_seq, (0, n-warm_seq.shape[0]))
        warm_seq = np.maximum(warm_seq, 0.0)
        if warm_seq.sum() >= 0.01:
            warm_seq = warm_seq / warm_seq.sum()

    # Phase 1: random parametric
    best_seq = None
    best_val = float('inf')
    families = ["edge_exp","edge_pow","beta","double_gauss","sin_pow","tent","dirichlet","logistic","piecewise","double_exp_general"]
    phase1_deadline = min(deadline - 5.0, time.time() + 0.4*(deadline - time.time()))  # allocate 40% to phase1

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10**rng.uniform(-3,3)   # broader
            weight = rng.random()
            cand = _edge_exponential_shape(n,gamma,weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-12.0,-0.1)
            weight = rng.random()
            cand = _edge_power_law_shape(n,alpha,weight)
        elif fam == "beta":
            a = 10**rng.uniform(-3,1)  # allow less than 0.1
            b = 10**rng.uniform(-3,1)
            cand = _beta_shape(n,a,b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005,0.5)
            cand = _double_gaussian_shape(n,sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.2,6.0)
            cand = _sinusoid_power_shape(n,power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.02,10.0)
            cand = _random_dirichlet_shape(n,alpha,rng)
        elif fam == "logistic":
            tau = rng.uniform(0.001,0.5) * n
            mid = (n-1)/2
            cand = _logistic_shape(n,mid,tau)
        elif fam == "piecewise":
            fraction = rng.uniform(0.01,0.5)   # fraction of edges
            k = max(1, int(fraction * n))
            ratio = 10**rng.uniform(-3,3)   # A/B
            cand = _piecewise_constant_shape(n,k,ratio)
        elif fam == "double_exp_general":
            gamma_l = 10**rng.uniform(-3,2)
            gamma_r = 10**rng.uniform(-3,2)
            weight_l = rng.random()
            weight_r = 1 - weight_l
            cand = _double_exponential_general_shape(n,gamma_l,gamma_r,weight_l,weight_r)
        else:
            continue

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase1] new best {val:.6f} via {fam}")

    # incorporate warm shifts
    if warm_seq is not None:
        for shift in (0, n//8, n//4, n//2):
            cand = np.roll(warm_seq, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Warm] improved to {val:.6f}")

    # Phase 2: gradient refinement: multiple restarts
    # Determine remaining time budget for phase 2.
    Now = time.time()
    time_left = deadline - now
    # We'll allocate at most 50% time for gradient
    phase2_deadline = now + min(time_left * 0.5, time_left - 10.0)

    # We'll run Adam, then Subgrad, then PGD sequentially, potentially multiple restarts if time permits.
    runs = []
    while time.time() < phase2_deadline:
        # Use best_seq as start
        a_start = best_seq if best_seq is not None else np.full(n,1.0/n)
        # Run Adam for up to some iterations/time
        sub_deadline = min(phase2_deadline, time.time() + 30.0) # 30s per Adam run
        a_adam, val_adam = _adam_optimize(a_start, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=200_000, lr_start=0.08)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase2] Adam improved to {val_adam:.6f}")
        # Subgrad
        sub_deadline = min(phase2_deadline, time.time() + 20.0)
        a_sub, val_sub = _subgradient_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=150_000, eta0=0.9)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase2] Subgrad improved to {val_sub:.6f}")
        # PGD
        sub_deadline = min(phase2_deadline, time.time() + 20.0)
        a_pgd, val_pgd = _pgd_optimize(best_seq, n, conv_fft_len, corr_fft_len, sub_deadline, max_iters=120_000, eta0=0.4)
        if val_pgd < best_val:
            best_val, best_seq = val_pgd, a_pgd.copy()
            print(f"[Phase2] PGD improved to {val_pgd:.6f}")

        # break if not much time left
        if time.time() > phase2_deadline - 5:
            break

    # Phase 3: local stochastic search with annealing
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase3 – annealing local search")
        sub_deadline = deadline
        a_loc, val_loc = _local_random_search(best_seq, n, conv_fft_len, sub_deadline, max_moves=200_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase3] local search improved to {val_loc:.6f}")

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0) # matches evaluate_sequence
    print(f"Final best value {best_val:.6f} (length {n})")
    return best_seq.tolist()

Algorithm
---------

* **Stage 0 – problem size**

  (non‑negative, sum ≈ 1).  Each optimiser runs until the global deadline is
  close.

* **Stage 3 – stochastic hill‑climbing (annealing)**  
  A cheap mass‑transfer local search with a simple simulated‑annealing

# Utility FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:

def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))

def _logistic_shape(n: int, mid: float, tau: float) -> np.ndarray:
    """Monotone logistic decay, normalised."""
    x = np.arange(n, dtype=np.float64)
    shape = 1.0 / (1.0 + np.exp((x - mid) / tau))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _piecewise_constant_shape(n: int, k: int, ratio: float) -> np.ndarray:
    """Two edge blocks of size k with height ratio*A/B, interior low."""
    k = max(1, min(k, n // 2 - 1))
    # ratio = A/B (edge height / interior height)
    B = 1.0 / ((2 * k) * ratio + (n - 2 * k))
    A = ratio * B
    shape = np.full(n, B, dtype=np.float64)
    shape[:k] = A
    shape[-k:] = A
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    return shape


def _double_exponential_general_shape(
    n: int, gamma_l: float, gamma_r: float, weight_l: float, weight_r: float
) -> np.ndarray:
    """Two independent edge exponentials."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma_l * i)
    right = np.exp(-gamma_r * (n - 1 - i))
    shape = weight_l * left + weight_r * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape

# Optimisers (mirrors the reference implementation but with slightly
# different default parameters)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.02   # slower rise – more stable early on

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny noise close to the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Very cheap stochastic hill‑climbing (accept only improvements)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

            best_seq /= s
        else:
            best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_val = _objective(best_seq, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase 2 – deep gradient refinement on the large grid
    if time.time() < deadline - 15:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            lr_start=0.08,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

        if time.time() < deadline - 10:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Phase 2] Sub‑grad improved to {val_sub:.6f}")

        if time.time() < deadline - 5:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=150_000,
                eta0=0.35,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[Phase 2] PGD improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing (last seconds)
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final clipping to obey evaluate_sequence's limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
