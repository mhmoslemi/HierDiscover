```
def _gd_submax(a, n, conv_fft_len, corr_fft_len, deadline, max_iters, step0=0.5):
    best_a = a.copy()
    best_val = eval_fast(a)
    a_prev = a.copy()
    momentum = np.zeros_like(a)
    beta = 0.9
    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # compute gradient using active set
        active = np.isclose(conv, max_c, atol=1e-12)   # tolerance
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # step size
        step = step0 / np.sqrt(it+1)
        # momentum
        momentum = beta * momentum + (1-beta) * grad
        a_new = a - step * momentum
        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        a_new = _project_simplex(a_new)
        a = a_new
    return best_a, best_val
```

def propose_candidate(seed=0, budget_s=1000):
    """
    Simple multistage optimizer.
    Phase 1: search on a small grid (n=1024) using many random starts and 
    projected sub‑gradient descent.
    Phase 2: up‑sample the best solution to a finer grid (n=4096 or 8192) 
    and refine with more iterations.
    Phase 3: a short simulated‑annealing pass for final polishing.
    Returns the best sequence found.
    """
    import time, numpy as np
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Phase 1: coarse
    n1 = 1024
    conv_len1 = _next_pow2(2*n1 - 1)
    corr_len1 = _next_pow2(3*n1 - 2)
    # Build candidate pool
    candidates = []
    # Uniform
    candidates.append(np.full(n1, 1.0/n1))
    # Dirichlet alphas
    for alpha in (0.05,0.1,0.2,0.5,1.0,2.0):
        candidates.append(np.random.dirichlet(alpha*np.ones(n1)))
    # Sinusoid powers
    i = np.arange(n1)
    for p in (0.2,0.5,1,2,3):
        shape = np.sin(np.pi*(i+0.5)/n1) ** p
        shape = np.maximum(shape,0)
        shape /= shape.sum()
        candidates.append(shape)
    # Exponential decays
    for gamma in (0.001,0.005,0.01,0.02,0.05):
        shape = np.exp(-gamma*i)
        shape /= shape.sum()
        candidates.append(shape)
    # Piecewise constant blocks
    for blocks in (4,8,16):
        block_len = n1//blocks
        a = np.zeros(n1)
        for b in range(blocks):
            mass = np.random.rand()
            a[b*block_len:(b+1)*block_len] = mass
        if a.sum()>0:
            a/=a.sum()
        candidates.append(a)
    # Spike pattern (few random positions)
    for k in (8,16,32):
        a = np.zeros(n1)
        pos = np.random.choice(n1,size=k,replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        candidates.append(a)

    # warm start if exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] >= n1:
            w = w[:n1]
        else:
            w = np.pad(w, (0, n1-w.shape[0]))
        w = np.maximum(w, 0.0)
        w = w / w.sum() if w.sum()>0 else np.full(n1,1.0/n1)
        candidates.append(w)

    best_seq = None
    best_val = float('inf')
    # allocate time per candidate as equal share of 30% of total budget
    phase1_time = (deadline - time.time()) * 0.35
    start_phase1 = time.time()
    num_cand = len(candidates)
    for idx, a0 in enumerate(candidates):
        if time.time() > deadline:
            break
        # time left for phase1
        time_left = deadline - time.time()
        # simple schedule: each candidate gets at most phase1_time/num_cand
        sub_dead = min(time.time() + phase1_time/num_cand, deadline)
        # ensure sum 1
        a = a0.copy()
        a = np.maximum(a,0)
        if a.sum()<1e-8:
            a = np.random.dirichlet(np.ones(n1))
        a = a / a.sum()
        a_opt, val_opt = _gd_submax(
            a, n1, conv_len1, corr_len1, sub_dead,
            max_iters=200_000, step0=0.5
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Coarse {idx}] new best {val_opt:.6f}")
    # Phase 2: upsample to finer grid
    if best_seq is not None and time.time() < deadline - 5:
        # Choose target n based on remaining budget
        remaining = deadline - time.time()
        # try n_fine = 4096 if enough time else stay n1.
        n_fine = 4096
        if remaining > 100 and (deadline - time.time()) > 20:
            # we attempt upsample
            # Interpolate
            x_coarse = np.linspace(0,1, n1, endpoint=False)
            x_fine = np.linspace(0,1, n_fine, endpoint=False)
            a_big = np.interp(x_fine, x_coarse, best_seq)
            # clip small negatives
            a_big = np.maximum(a_big, 0.0)
            a_big /= a_big.sum()
            conv_len_fine = _next_pow2(2*n_fine -1)
            corr_len_fine = _next_pow2(3*n_fine -2)
            sub_dead = deadline - 5
            a_fine_opt, val_fine_opt = _gd_submax(
                a_big, n_fine, conv_len_fine, corr_len_fine, sub_dead,
                max_iters=300_000, step0=0.5
            )
            if val_fine_opt < best_val:
                best_val = val_fine_opt
                best_seq = a_fine_opt.copy()
                print(f"[Fine] improved to {val_fine_opt:.6f}")
    # Phase 3: final annealing
    if best_seq is not None and time.time() < deadline - 2:
        n_curr = len(best_seq)
        conv_len_curr = _next_pow2(2*n_curr -1)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n_curr, conv_len_curr, deadline, max_moves=100_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] new best {val_ann:.6f}")
    # Ensure non-zero output
    if best_seq is None:
        best_seq = np.full(1024, 1.0/1024)
        best_val = 2*1024*(1/1024) # 2
    # Clip and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
multistage_step_optimizer.py

A lightweight optimizer for the “max‐convolution” benchmark.

The algorithm works in three stages:

1. **Coarse search (n≈1024).**  
   A diverse pool of candidate probability vectors (uniform, Dirichlet,
   exponential, sinusoid powers, beta‑shaped, piecewise‑constant and sparse
   spikes) is generated.  Each candidate is refined by a projected sub‑gradient
   descent on the exact objective  2·n·max(a∗a) .  The best coarse vector is kept.

2. **Upsampling & fine tuning (n≈4096).**  
   The coarse optimum is linearly interpolated to a finer grid, re‑normalised
   and further optimised with the same sub‑gradient descent (many more
   iterations).  A short “flattening” phase then tries to equalise the
   maximal and minimal convolution values by moving the vector in the
   direction *grad(max) – grad(min)*.

3. **Final stochastic polishing.**  
   A simple simulated‑annealing move set (transfer a random fraction of mass
   between two coordinates) is applied for a few 10⁵ moves.

All iterates stay on the probability simplex (projection onto the simplex)
so the evaluation function is scale‑invariant and the sum of the returned
coefficients is 1.  The routine respects the global 1000 s budget and always
returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _gd_submax(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    step0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution objective.
    Returns the best vector seen and its objective value 2·n·max(conv).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # sub‑gradient of the max (uniform weight over all active indices)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = step0 / np.sqrt(it + 1.0)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

    return best_a, best_val


def _flatten_balancer(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 100_000,
    step0: float = 0.4,
) -> tuple[np.ndarray, float]:
    """
    Additional pass that tries to push the maximal convolution down while
    raising the minimal one: direction = grad(max) – grad(min).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() > deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        min_c = conv.min()
        if max_c - min_c < 1e-14:
            break

        active_max = np.isclose(conv, max_c, atol=1e-12)
        active_min = np.isclose(conv, min_c, atol=1e-12)

        w_max = np.zeros_like(conv)
        w_min = np.zeros_like(conv)
        if active_max.any():
            w_max[active_max] = 1.0 / active_max.sum()
        if active_min.any():
            w_min[active_min] = 1.0 / active_min.sum()

        grad_max = 2.0 * _cross_correlation(w_max, a, corr_fft_len)
        grad_min = 2.0 * _cross_correlation(w_min, a, corr_fft_len)

        grad = grad_max - grad_min
        step = step0 / np.sqrt(it + 1.0)
        a -= step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Randomly transfers a fraction of the mass of one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget (~1000 s) and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Phase 0 – very small warm‑start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
    else:
        warm = None

    # ------------------------------------------------------------------
    # *********   Stage 1 – coarse optimisation (n≈1024)   *********
    n_coarse = 1024
    conv_len_c = _next_pow2(2 * n_coarse - 1)
    corr_len_c = _next_pow2(3 * n_coarse - 2)

    # Build a modest but diverse pool of start vectors (all on the simplex)
    pool = []

    # Uniform
    pool.append(np.full(n_coarse, 1.0 / n_coarse))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n_coarse)))

    # Power‑law / beta‑shaped families (U‑shaped for α<1)
    for alpha in (0.2, 0.4, 0.6, 0.8):
        i = np.arange(n_coarse)
        x = (i + 0.5) / n_coarse
        shape = (x ** (alpha - 1)) * ((1 - x) ** (alpha - 1))
        shape = np.maximum(shape, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Mixture of uniform and beta‑shapes
    for alpha in (0.2, 0.4, 0.6):
        beta = ( (np.arange(n_coarse) + 0.5) / n_coarse ) ** (alpha - 1) \
               * (1 - (np.arange(n_coarse) + 0.5) / n_coarse) ** (alpha - 1)
        beta = np.maximum(beta, 0.0)
        beta /= beta.sum()
        for p in (0.2, 0.5, 0.8):
            mix = p * beta + (1 - p) * (1.0 / n_coarse)
            mix /= mix.sum()
            pool.append(mix)

    # Exponential decays
    for gamma in (0.001, 0.005, 0.01, 0.02, 0.05):
        shape = np.exp(-gamma * np.arange(n_coarse))
        shape /= shape.sum()
        pool.append(shape)

    # Half‑wave sinusoid powers
    i = np.arange(n_coarse)
    for p in (0.2, 0.5, 1.0, 2.0):
        shape = np.sin(np.pi * (i + 0.5) / n_coarse) ** p
        shape = np.maximum(shape, 0.0)
        shape /= shape.sum()
        pool.append(shape)

    # Piece‑wise constant blocks
    for blocks in (4, 8, 16):
        blk_len = n_coarse // blocks
        a = np.zeros(n_coarse)
        for b in range(blocks):
            a[b * blk_len : (b + 1) * blk_len] = np.random.rand()
        if a.sum() > 0:
            a /= a.sum()
        pool.append(a)

    # Sparse spike patterns (few equally weighted spikes)
    for k in (8, 16, 32):
        a = np.zeros(n_coarse)
        pos = np.random.choice(n_coarse, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    # Warm‑start integration (if available)
    if warm is not None:
        if warm.shape[0] >= n_coarse:
            w = warm[:n_coarse]
        else:
            w = np.pad(warm, (0, n_coarse - warm.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() > 0:
            w /= w.sum()
        else:
            w = np.full(n_coarse, 1.0 / n_coarse)
        pool.append(w)

    best_seq = None
    best_val = float("inf")

    # Allocate roughly 30 % of the total budget to the coarse stage
    time_for_coarse = (deadline - time.time()) * 0.30
    per_candidate_budget = max(5.0, time_for_coarse / max(len(pool), 1))

    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break
        sub_dead = min(deadline, time.time() + per_candidate_budget)

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n_coarse))
        a0 = a0 / a0.sum()

        a_opt, val_opt = _gd_submax(
            a0,
            n_coarse,
            conv_len_c,
            corr_len_c,
            sub_dead,
            max_iters=300_000,
            step0=0.6,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Coarse {idx}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # *********   Stage 2 – up‑sample & fine optimisation (n≈4096)   *********
    if best_seq is not None and time.time() < deadline - 10:
        # decide target size (4096 if enough time, otherwise stay at 1024)
        n_fine = 4096
        if (deadline - time.time()) < 200:
            n_fine = n_coarse

        x_coarse = np.linspace(0, 1, n_coarse, endpoint=False)
        x_fine = np.linspace(0, 1, n_fine, endpoint=False)
        a_big = np.interp(x_fine, x_coarse, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        conv_len_f = _next_pow2(2 * n_fine - 1)
        corr_len_f = _next_pow2(3 * n_fine - 2)

        # long projected sub‑gradient descent
        sub_dead = deadline - 5
        a_fine, val_fine = _gd_submax(
            a_big,
            n_fine,
            conv_len_f,
            corr_len_f,
            sub_dead,
            max_iters=600_000,
            step0=0.5,
        )
        if val_fine < best_val:
            best_val = val_fine
            best_seq = a_fine.copy()
            print(f"[Fine] sub‑gradient improved to {val_fine:.6f}")

        # optional flattening pass (max‑min balancing)
        if time.time() < deadline - 5:
            bal_dead = deadline - 3
            a_bal, val_bal = _flatten_balancer(
                best_seq,
                n_fine,
                conv_len_f,
                corr_len_f,
                bal_dead,
                max_iters=120_000,
                step0=0.4,
            )
            if val_bal < best_val:
                best_val = val_bal
                best_seq = a_bal.copy()
                print(f"[Bal] improved to {val_bal:.6f}")

    # ------------------------------------------------------------------
    # *********   Stage 3 – final simulated‑annealing polish   *********
    if best_seq is not None and time.time() < deadline - 2:
        n_curr = len(best_seq)
        conv_len_curr = _next_pow2(2 * n_curr - 1)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n_curr, conv_len_curr, deadline, max_moves=100_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] final best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None:
        best_seq = np.full(1024, 1.0 / 1024)

    # Clip to the evaluator’s limits and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
