"""
Propose a step‑function sequence that (approximately) minimizes

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0 .

Approach
--------
1.  Work on the probability simplex (∑a = 1) and use the “soft‑max”
    (log‑sum‑exp) smooth approximation of the max.
2.  Parameterise a by logits: a = softmax(logits).  This guarantees
    non‑negativity and the sum‑constraint automatically.
3.  For a fixed smoothness λ the loss is
        Lλ(logits) = 2·n·log‑sum‑exp(λ·conv(a)) / λ .
    Its gradient w.r.t. the logits can be written in closed form
    using FFT‑based convolution / cross‑correlation.
4.  Optimize Lλ using L‑BFGS‑B (SciPy) – a second‑order method with a
    line‑search that converges much faster than plain Adam.
5.  Run several independent restarts with different Dirichlet
    initialisations and a schedule of increasing λ (coarse → sharp).
6.  Keep the best vector seen, then polish it with a short simulated‑
    annealing phase that uses mass‑transfer and swap moves.
7.  Finally clip to the evaluator’s allowed range and return the list.

The whole routine respects the supplied time budget (default 1000 s)
and always returns the best sequence discovered before the deadline.
"""

import time
import numpy as np
from scipy import optimize as _opt

# ----------------------------------------------------------------------
# FFT helpers (same as the reference implementation)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (no clipping)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Smooth‑max loss + gradient (log‑sum‑exp surrogate)
# ----------------------------------------------------------------------
def _smooth_max_loss_and_grad(logits: np.ndarray, lam: float,
                              n: int, conv_fft_len: int,
                              corr_fft_len: int):
    """
    Returns (loss, grad) for the smooth‑max surrogate.
    Parameters
    ----------
    logits : ndarray, shape (n,)
        Unconstrained parameters; a = softmax(logits).
    lam : float
        Smoothness parameter (λ). Larger → closer to true max.
    """
    # soft‑max → a on the simplex
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # log‑sum‑exp of λ·conv
    lc = lam * conv
    max_lc = lc.max()
    exp_lc = np.exp(lc - max_lc)          # safe exponentiation
    Z = exp_lc.sum()
    # smooth‑max = (log‑sum‑exp)/λ
    smooth_max = (max_lc + np.log(Z)) / lam
    loss = 2.0 * n * smooth_max

    # weights for the gradient (soft‑max over λ·conv)
    w = exp_lc / Z

    # gradient w.r.t. a (2·w ★ a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain‑rule through soft‑max (mirror‑descent form)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    return loss, grad_logits


# ----------------------------------------------------------------------
# Simple simulated‑annealing refinement (mass transfer + swaps)
# ----------------------------------------------------------------------
def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, deadline: float,
                    max_moves: int = 80_000,
                    swap_prob: float = 0.12) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val, best_a = val, a.copy()

    T_start, T_end = 0.08, 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a, val = a2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the given time budget and always returns the best sequence
    found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size (the original implementation used 4096, which gives a
    # good trade‑off between resolution and speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:

        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------

                   lam_schedule: tuple[float, ...],
                   maxiter_per_stage: int = 150) -> np.ndarray:
        logits = start_logits.copy()
        for lam in lam_schedule:
            if time.time() > deadline - 5:

            def fun(x, lam=lam):
                loss, grad = _smooth_max_loss_and_grad(
                    x, lam, n, conv_fft_len, corr_fft_len
                )
                return loss, grad

            # run a short L‑BFGS‑B optimisation
            res = _opt.minimize(fun, logits, jac=True, method='L-BFGS-B',
                                options={'maxiter': maxiter_per_stage,
                                         'disp': False})
            logits = res.x
        return logits

    # ------------------------------------------------------------------
    # Phase I – many random restarts with an increasing λ schedule
    lam_schedule = (0.5, 2.0, 10.0, 50.0, 200.0, 800.0, 3000.0)

    # optionally start from the warm vector first
    if warm is not None:
        logits0 = np.log(warm + 1e-12)
        logits_opt = _run_lbfgs(logits0, lam_schedule, maxiter_per_stage=250)
        a = np.exp(logits_opt - np.max(logits_opt))
        a = a / a.sum()
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Warm] new best {val:.6f}")

    # random Dirichlet restarts
    while time.time() < deadline - 15:
        # pick a random concentration – this yields a diverse family
        alpha = np.random.choice([0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0])
        a0 = np.random.dirichlet(alpha * np.ones(n))
        logits0 = np.log(a0 + 1e-12)

        # run L‑BFGS‑B on this start
        logits_opt = _run_lbfgs(logits0, lam_schedule,
                                maxiter_per_stage=120)

        # evaluate the exact objective
        a = np.exp(logits_opt - np.max(logits_opt))
        a = a / a.sum()
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Random] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – final simulated‑annealing refinement of the incumbent
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] polishing the best candidate")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=120_000, swap_prob=0.15

    best_seq = np.clip(best_seq, 0.0, 1000.0)
