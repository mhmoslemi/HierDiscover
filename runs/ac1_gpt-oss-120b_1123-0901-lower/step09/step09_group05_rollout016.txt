"""
Propose a non‑negative step function that (approximately) minimizes

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The search proceeds in three stages:

1. **Diverse pool generation** – deterministic families (uniform,
   sinusoid, exponential, power‑law, beta‑pdf, double‑Gaussian,
   triangular, left‑only exponential) together with many Dirichlet draws,
   random permutations of a base shape and a few cyclic shifts of an
   optional warm‑start are created.  Each candidate is projected onto the
   probability simplex (∑a=1) and its exact objective is evaluated with an
   FFT‑based convolution.  The best `top_k` vectors are retained.

2. **Projected gradient descent** – for every retained vector a sub‑gradient
   descent on the true non‑smooth objective is performed.  The sub‑gradient
   is obtained from the active set of indices attaining the current maximal
   convolution value; a back‑tracking line‑search with Euclidean projection
   onto the simplex guarantees feasibility.  More iterations and a larger
   share of the time budget are given to the most promising candidates.

3. **Extended hill‑climbing** – the current best vector is refined by a
   cheap random‑move search that repeatedly transfers a random fraction of
   mass from one position to another and keeps the move if it improves the
   exact objective.  The move budget is enlarged compared with the previous
   implementation.

All phases respect the global 1000 s wall‑clock budget; the best sequence
found is returned as a plain Python list, ready for `evaluate_sequence`.
"""

import time
import numpy as np

# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Deterministic families (all return vectors with sum == 1) ----------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = idx ** alpha + (n - idx + 1) ** alpha
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    shape = 1.0 - np.abs(i - mid) / mid
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Add a few cyclic shifts of a warm start."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse seed pool (deterministic families + random Dirichlet)."""
    pool = []

    # Dirichlet draws with several concentration levels
    for alpha in (0.05, 0.1, 0.2, 0.5, 0.8, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential (coarse grid – finer grid added later)
    for gamma in (0.1, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law
    for alpha in (-3.0, -2.0, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta pdf (symmetric and skewed)
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
                 (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double‑Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple families
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_triangular(n))

    return pool


def _grid_candidates(n: int) -> list[np.ndarray]:
    """Finer sweep over the main parametric families."""
    out = []

    # Edge‑exponential – log‑spaced
    for gamma in np.logspace(-2, 1.5, 30):
        out.append(_edge_exponential(n, float(gamma)))

    # Edge‑power‑law – linear sweep
    for alpha in np.linspace(-4.0, 2.0, 30):
        out.append(_edge_power_law(n, float(alpha)))

    # Beta – grid in log‑space
    a_vals = np.logspace(-1, 1, 15)
    b_vals = np.logspace(-1, 1, 15)
    for a in a_vals:
        for b in b_vals:
            out.append(_beta_shape(n, float(a), float(b)))

    # Double‑Gaussian – linear sweep
    for sigma in np.linspace(0.02, 0.5, 30):
        out.append(_double_gaussian(n, float(sigma)))

    return out

# Projected gradient descent on the exact objective --------------------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on  F(a)=2·n·max(a*a)/(sum a)^2.
    The vector is kept on the probability simplex.
    Returns the best vector found and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    init_step = 0.5

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # back‑tracking line‑search (projected)
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new / (a_new.sum() ** 2)

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val

# Cheap hill‑climber ----------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Random‑move hill‑climber.
    At each step a fraction of the mass is moved from one index to another;
    the move is kept if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Run the three‑stage search and return the best height sequence."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # optional warm start ------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # 1. build a rich pool of start vectors
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng) + _grid_candidates(n)

    # add a few random Dirichlet draws with extreme concentrations
    for alpha in (0.01, 0.05, 0.2, 0.7, 1.2, 3.0):
        for _ in range(4):
            pool.append(_random_dirichlet(n, alpha, rng))

    # add random permutations of a moderately peaked base shape
    base = _edge_exponential(n, gamma=2.0)
    for _ in range(10):
        perm = rng.permutation(n)
        pool.append(base[perm])

    # ------------------------------------------------------------------
    # quick evaluation of the whole pool, keep the best few
    top_k = 30
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()                # sum(a)=1 → denominator = 1
        scored.append((val, a))

        # stop early if we have already spent ~20 % of the budget
        if time.time() >= start_time + 0.2 * budget_s:
            break

    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]

    # ------------------------------------------------------------------
    # 2. projected gradient descent on the elite candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 12:
            break

        # simple heuristic to allocate more time to the leading candidates
        remaining = deadline - time.time()
        if rank < 5:
            share = remaining * 0.4
        else:
            share = remaining * 0.6 / (top_k - rank)
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=300_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine rank={rank}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3. final cheap local moves (uses any remaining time)
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=500_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
