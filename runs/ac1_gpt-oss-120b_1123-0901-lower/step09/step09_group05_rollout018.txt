"""
Improved search for a step‑function minimizing

    F(a) = 2·n·max(a∗a) / (sum(a)²)

The algorithm works in three phases:

1. **Diverse seeding** – a huge pool of candidates is built from many
   parametric families (uniform, asymmetric exponential, two‑Gaussian
   mixtures, linear tilts, Dirichlet draws, …).  Every candidate is
   normalised to sum‑to‑one and its exact objective is evaluated.
   The best `top_k` vectors are retained.

2. **Intensive refinement** – each elite vector is polished by a
   projected sub‑gradient descent on the *true* max‑convolution.
   A back‑tracking line‑search guarantees monotone improvement.
   The first few elites receive a larger share of the remaining time
   (up to several hundred thousand iterations).

3. **Heavy‑weight local search** – the current best vector is then
   subjected to a cheap but aggressive hill‑climber that repeatedly
   transfers a random fraction of mass between two positions and
   accepts any improvement.  The move budget is scaled to use all
   remaining wall‑clock time.

All phases respect the global 1000 s budget.
The function returns the best sequence found (as a plain Python list)
compatible with the supplied `evaluate_sequence` routine.
"""

import time
import numpy as np
from math import log, sqrt, exp, pi

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for sub‑gradients.
    w and a are real vectors, w is the weight (active‑set) vector.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _linear_tilt(n: int, lam: float) -> np.ndarray:
    """
    Convex combination of the uniform vector and a left‑skewed linear shape.
    lam=0 → uniform, lam=1 → linear decreasing (mass at the left side).
    """
    base = np.full(n, 1.0 / n, dtype=np.float64)
    lin = (n - np.arange(n, dtype=np.float64))
    lin = lin / lin.sum()
    out = (1.0 - lam) * base + lam * lin
    out = out / out.sum()
    return out


def _asym_exponential(n: int, gamma: float, alpha: float) -> np.ndarray:
    """
    Asymmetric sum of two exponentials.
    a_i ∝ α·exp(-γ·i) + (1-α)·exp(-γ·(n-1-i)).
    """
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = np.exp(-gamma * (n - 1 - i))
    shape = alpha * left + (1.0 - alpha) * right
    shape = shape / shape.sum()
    return shape


def _two_gaussian(n: int, sigma: float, c1: float, c2: float, w: float) -> np.ndarray:
    """
    Mixture of two Gaussians placed at c1 and c2 (indices), weight w on the first.
    """
    i = np.arange(n, dtype=np.float64)
    g1 = np.exp(-0.5 * ((i - c1) / (sigma * n)) ** 2)
    g2 = np.exp(-0.5 * ((i - c2) / (sigma * n)) ** 2)
    shape = w * g1 + (1.0 - w) * g2
    shape = shape / shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Objective utilities ----------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of F(a) for a normalised vector."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
# Projected sub‑gradient descent -----------------------------------------
def _gd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the exact objective.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    init_step = 0.8

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 5_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # active set – indices attaining the current maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        if weight.sum() > 0:
            weight /= weight.sum()
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # ----- back‑tracking line‑search (projected) -----
        step = init_step
        while step > 1e-12:
            a_new = a - step * grad
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = float(2.0 * n * max_c_new / (a_new.sum() ** 2))

            if not np.isnan(val_new) and val_new <= val:
                a = a_new
                break
            step *= 0.5

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap hill‑climber -----------------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber: transfer a random fraction of the mass from one
    position to another and keep the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = float(2.0 * n * conv2.max() / (a2.sum() ** 2))

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point -------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main routine called by the judge.
    Returns the best step‑function found within ``budget_s`` seconds.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # --------------------------------------------------------------
    n = 4096                       # keep the length used in the reference run
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---- optional warm start ----------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ---- build a massive pool of start vectors ----------------------
    pool = []

    # warm‑start and a few cyclic shifts (helps escape symmetry)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families
    pool.append(_uniform(n))
    pool.append(_linear_tilt(n, 0.1))

    # asymmetric exponential grid
    for gamma in np.logspace(-2, 1, 15):          # 0.01 … 10
        for alpha in np.linspace(0.0, 1.0, 9):   # weight on the left side
            pool.append(_asym_exponential(n, float(gamma), float(alpha)))

    # two‑Gaussian mixtures (centers spaced across the interval)
    for sigma in (0.05, 0.1, 0.2):
        for c1 in (n * 0.15, n * 0.35, n * 0.55):
            c2 = n - 1 - c1
            for w in (0.3, 0.5, 0.7):
                pool.append(_two_gaussian(n, sigma, float(c1), float(c2), float(w)))

    # linear tilts with finer sampling
    for lam in np.linspace(0.0, 0.2, 11):
        pool.append(_linear_tilt(n, float(lam)))

    # random Dirichlet draws (different concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, float(alpha), rng))

    # ----------------------------------------------------------------
    # Quick evaluation of the whole pool – retain the elite few
    top_k = 30
    scored = []
    for cand in pool:
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        scored.append((val, a))
        # early exit if we are spending too much time on the pool
        if time.time() >= deadline * 0.2:
            break

    scored.sort(key=lambda x: x[0])
    elite = [a for _, a in scored[:top_k]]
    print(f"Pool evaluated – best initial value {scored[0][0]:.6f}")

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the elite candidates
    best_seq = None
    best_val = float("inf")

    # allocate time roughly: first candidate gets ~40% of remaining,
    # the rest share the rest decreasingly.
    for rank, a0 in enumerate(elite):
        if time.time() >= deadline - 5:                # keep a few seconds buffer
            break
        remaining = deadline - time.time()
        # simple decreasing share
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=400_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – final cheap hill‑climber (uses whatever time is left)
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=800_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {val_loc:.6f}")

    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety – clip to the evaluator's range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
