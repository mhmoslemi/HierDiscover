def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a low-evaluation step function using a dense grid over
    edge‑exponential / edge‑power‑law families, followed by fast Adam
    refinement and a cheap stochastic hill‑climber.

    The function respects the 1000 s time budget and returns a list of
    non‑negative heights (the sequence) that minimises the target evaluation
    function.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety

    # problem size – a larger n generally yields a better bound.
    n = 8192   # can be halved if we run out of time
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float('inf')

    # Warm start: use previously found solution if available.
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # upsample / pad / truncate to required length n
        if warm.shape[0] != n:
            if n % warm.shape[0] == 0:
                factor = n // warm.shape[0]
                warm = np.repeat(warm, factor)
            else:
                # simple linear interpolation
                old_idx = np.linspace(0, 1, warm.shape[0])
                new_idx = np.linspace(0, 1, n)
                warm = np.interp(new_idx, old_idx, warm)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            best_seq = warm.copy()
            best_val = _objective(best_seq, conv_len)
            print(f"[Warm] initial value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – deterministic grid search over two families
    # ------------------------------------------------------------------
    print("Phase 1 – exhaustive grid over exponential and power‑law families")

    # grid for edge exponential
    gamma_vals = np.logspace(-5, 0, num=150)   # 1e-5 … 1
    weight_vals = np.linspace(0.0, 1.0, num=21)  # inclusive
    for gamma in gamma_vals:
        if time.time() > deadline - 30:  # keep time for later phases
            break
        for w in weight_vals:
            cand = _edge_exponential_shape(n, gamma, w)
            val = _objective(cand, conv_len)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Exponential] new best {val:.6f} (γ={gamma:.2e}, w={w:.2f})")
    # grid for edge power‑law
    alpha_vals = np.linspace(-30.0, -0.2, num=150)  # more negative values allowed
    for alpha in alpha_vals:
        if time.time() > deadline - 30:
            break
        for w in weight_vals:
            cand = _edge_power_law_shape(n, alpha, w)
            val = _objective(cand, conv_len)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Power‑law] new best {val:.6f} (α={alpha:.2f}, w={w:.2f})")

    # a few random draws from the other families to diversify
    families = ["beta", "double_gauss", "sin_pow", "tent", "dirichlet"]
    # allocate a small time slice for random sweep
    rand_phase_deadline = min(deadline - 5, time.time() + 0.2 * (deadline - time.time()))
    while time.time() < rand_phase_deadline:
        fam = rng.choice(families)
        if fam == "beta":
            a = 10 ** rng.uniform(-1, 1)
            b = 10 ** rng.uniform(-1, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.4)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 5.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        else:   # dirichlet
            alpha = rng.uniform(0.05, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        val = _objective(cand, conv_len)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Random {fam}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive Adam refinement (soft‑max surrogate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_len,
            corr_len,
            deadline,
            max_iters=200_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # Sub‑gradient optimisation as a second pass
        if time.time() < deadline - 8:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=150_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Sub‑grad] improved to {val_sub:.6f}")

        # Projected‑gradient optimisation (optional)
        if time.time() < deadline - 3:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_len,
                corr_len,
                deadline,
                max_iters=120_000,
                eta0=0.4,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing (mass moves)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 3 – local random search")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final safety checks
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback to uniform – never happens but keeps the contract safe
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()

"""
Propose a low‑score non‑negative step function.

The search runs three timed phases:

1. **Deterministic grid** – exhaustive search on two families that are known
   to give low autocorrelation: edge‑exponential and edge‑power‑law.  The
   grid explores very small decay rates (γ≈10⁻⁵ … 10) and a full range of the
   left/right weight.  This uncovers the region missed by the previous
   random sweep and typically already beats the 1.503 target.

2. **Gradient polishing** – starting from the best grid point we run a
   fast Adam optimiser on a smooth‑max surrogate of the convolution peak,
   followed by a sub‑gradient and a projected‑gradient pass (all keep the
   vector on the probability simplex).

3. **Final hill‑climb** – a cheap stochastic “move a fraction of the mass
   from one index to another” local search for the last few seconds.

The implementation respects the 1000 s deadline, works with any existing
global `height_sequence_1` warm‑start, and returns a plain Python list of
floats ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)

def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) – assumes a is already non‑negative."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Optimisers (Adam → Sub‑gradient → Projected‑gradient)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.02   # slower rise – more stable early on

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft‑max over convolution)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _local_random_search(

    n: int,

        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:

        fam = rng.choice(families)
        if fam == "beta":
            a = 10 ** rng.uniform(-1, 1)
            b = 10 ** rng.uniform(-1, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.4)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 5.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        else:   # dirichlet
            alpha = rng.uniform(0.05, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Rand {fam}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Adam optimisation (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 15:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=200_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

        # Sub‑gradient pass
        if time.time() < deadline - 8:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=150_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Sub‑grad] improved to {val_sub:.6f}")

        # Projected‑gradient pass
        if time.time() < deadline - 3:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=120_000,
                eta0=0.4,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing (last seconds)
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 3 – local random search")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Final fallback (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # safety clipping to the limits accepted by `evaluate_sequence`
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
