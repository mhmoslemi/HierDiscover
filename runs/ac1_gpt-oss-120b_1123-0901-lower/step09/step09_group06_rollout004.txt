"""
Propose a step‑function (non‑negative sequence) that lowers

    F(a) = 2·n·max(a∗a) / (sum(a)²)

within a 1000 s budget.  The search consists of:

1.  A large “population” of diverse candidates (Dirichlet, edge‑exponential,
    edge‑power‑law, β‑pdf, double‑Gaussian, sinusoid, raised‑cosine,
    semicircle, sparse spikes and a possible warm‑start).  The best few are
    kept.

2.  Several Adam runs on the smooth‑max surrogate, each started from a
    different elite candidate.  The temperature (inverse‑soft‑max
    parameter) is increased steadily, which acts as a simulated‑annealing
    schedule.

3.  Continuation L‑BFGS on a sequence of increasing temperatures α
    (10 → 20 000) to refine the best Adam result.

4.  A cheap simulated‑annealing hill‑climber that moves a random fraction
    of mass between two positions; moves that improve the exact objective
    are always kept, uphill moves are accepted with a decaying temperature.

All heavy work is done on the simplex (the vector is always re‑normalised),
and convolutions are evaluated with FFT for speed.  The routine respects
the 1000 s deadline and always returns the best sequence found.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


# ----------------------------------------------------------------------
# Convolution helpers (FFT based, fast for many evaluations)
def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for the Adam gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Projection (not used directly but kept for completeness)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families (all return a probability vector that sums to 1)
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Exact objective (fast version of evaluate_sequence)
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation of 2·n·max(a∗a)/(sum(a)²)."""
    a = np.maximum(a, 0.0)
    a = np.minimum(a, 1000.0)
    s = a.sum()
    if s < 0.01:
        return np.inf
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # inverse temperature (smooth‑max sharpness)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max (stay on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max object + gradient (used by L‑BFGS)
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Objective = 2·n·smooth_max(conv) and its gradient."""
    # soft‑max (logits → simplex)
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution via FFT
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    # smooth‑max = max + (1/α)·log Σexp(α·(c–max))
    max_c = conv.max()
    v = conv - max_c               # ≤ 0
    exp_alpha_v = np.exp(alpha * v)
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    # gradient weights
    w = exp_alpha_v / sum_exp

    # gradient w.r.t. a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max
    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


# ----------------------------------------------------------------------
# Continuation L‑BFGS (no closures – args are passed explicitly)
def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on a schedule of increasing temperatures α."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = np.inf

    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        # SciPy L‑BFGS with explicit args (no closure)
        res = minimize(
            _l_bfgs_obj_grad,
            logits,
            args=(n, conv_fft_len, corr_fft_len, alpha),
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": 150, "disp": False},
        )
        logits = res.x

        # exact evaluation of the resulting point
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = _objective(a, n, conv_fft_len)
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing hill‑climber (cheap local refinement)
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Random mass‑moves with a decaying temperature."""
    cur_a = a0.copy()
    cur_val = _objective(cur_a, n, conv_fft_len)

    best_a = cur_a.copy()
    best_val = cur_val

    rng = np.random.default_rng()
    temp = 0.025
    decay = 0.99996
    it = 0

    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        j = rng.integers(n)
        if i == j:
            it += 1
            continue
        if cur_a[i] <= 1e-16:
            it += 1
            continue
        # move a random fraction (up to 30 %) of mass from i to j
        frac = rng.random() * 0.3
        delta = cur_a[i] * frac
        a_new = cur_a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        new_val = _objective(a_new, n, conv_fft_len)
        # Metropolis acceptance
        if new_val < cur_val or rng.random() < np.exp(-(new_val - cur_val) / max(temp, 1e-9)):
            cur_a = a_new
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = a_new.copy()
                print(f"[SA it={it:06d}] new best {best_val:.6f}")

        temp = max(temp * decay, 1e-6)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The function respects a time budget (default 1000 s) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096                         # length of the vector
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous run stored `height_sequence_1`)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # 1) Build a diverse pool of starters
    pool: List[np.ndarray] = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families
    for a in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, a))
    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))
    for alpha in np.linspace(-8, -0.2, 20):
        pool.append(_edge_power_law(n, alpha))
    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0),
        (5.0, 0.2), (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))
    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick scan of the pool
    best_seq: np.ndarray | None = None
    best_val = np.inf

    rng.shuffle(pool)
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a /= a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        if time.time() > deadline - 120:   # keep ~2 min for deep phases
            break

    # ------------------------------------------------------------------
    # 2) Multiple Adam runs (different random perturbations of the elite)
    while time.time() < deadline - 45:
        # start from the current elite (or uniform if none)
        if best_seq is None:
            start = _uniform(n)
        else:
            start = best_seq.copy()
        # random jitter to escape possible plateaus
        jitter = rng.normal(scale=0.01, size=n)
        start = np.maximum(start + jitter, 0.0)
        if start.sum() < 1e-12:
            start = _uniform(n)
        start /= start.sum()

        a_adam, val_adam = _adam_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam run] new best {val_adam:.6f}")

        # one more quick Adam run with a larger learning‑rate
        a_adam2, val_adam2 = _adam_optimize(
            start,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=120_000,
            lr_start=0.08,
        )
        if val_adam2 < best_val:
            best_val = val_adam2
            best_seq = a_adam2.copy()
            print(f"[Adam fast] new best {val_adam2:.6f}")

    # ------------------------------------------------------------------
    # 3) Continuation L‑BFGS refinement
    if best_seq is not None and time.time() < deadline - 20:
        a_lbfgs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbfgs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # 4) Final cheap simulated‑annealing hill‑climber
    if best_seq is not None and time.time() < deadline - 5:
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA final] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Fallback – should never happen, but stay safe
    if best_seq is None:
        best_seq = _uniform(n)

    # Final clipping to obey the evaluator's safety limits
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```

This implementation expands on the previous state‑of‑the‑art pipeline by adding:

* a larger and more varied initial pool (including the optional warm‑start),
* several Adam restarts with different learning‑rates and random jitter,
* a longer L‑BFGS continuation (no closures, explicit ``args``),
* a simulated‑annealing hill‑climber for fine‑grained, stochastic improvement.

All helper functions are defined at the top level, no closures or lambda
expressions are used, and the routine respects the 1000 s deadline while
always returning the best sequence discovered.
