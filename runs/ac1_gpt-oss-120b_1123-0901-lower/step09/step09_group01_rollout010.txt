"""
Propose a low‑score step‑function.

The routine follows three stages:

1️⃣ Broad random search over several one‑parameter and two‑parameter
   families (exponential·power, logistic, stretched‑exponential) and
   mixture with a uniform baseline.  Every candidate is normalised to
   ℓ¹ = 1 and evaluated with a fast FFT‑based convolution.  The best
   few candidates are kept.

2️⃣ Multi‑start smoothed‑max optimisation.  Starting from each of the
   retained candidates (and a few random logits) we run Adam on the
   log‑sum‑exp surrogate of the max‑convolution (gradients obtained
   by a cross‑correlation).  The temperature of the log‑sum‑exp is
   increased quickly so the surrogate approaches the true max.

3️⃣ Long exact‑max sub‑gradient phase (exponentiated‑gradient mirror
   descent) drives the best vector to a (near) local optimum of the
   original non‑smooth objective.  An optional simulated‑annealing
   fine‑tuning is performed if time permits.

All steps constantly check the overall time budget (default 1000 s)
and always return a valid non‑negative list.  The final score is printed
using the public `evaluate_sequence` function (if available).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (the gradient of a*w‑convolution)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------
def _candidate_exp_power(alpha: float, p: float, n: int) -> np.ndarray:
    """Monotone decreasing shape   a_i ∝ exp(-α·i)·(i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) * ((i + 1) ** p)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total > 0.0:
        a /= total
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _candidate_logistic(mu: float, sigma: float, n: int) -> np.ndarray:
    """Logistic decay   a_i = 1/(1+exp((i‑μ)/σ))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - mu) / sigma))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _candidate_stretched_exp(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone stretched‑exponential   a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum > 0.0:
        a /= a_sum
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    return a


def _candidate_mix(a: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination with a uniform baseline."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    mixed = lam * a + (1.0 - lam) * uniform
    mixed_sum = mixed.sum()
    if mixed_sum > 0.0:
        mixed /= mixed_sum
    else:
        mixed = uniform
    return mixed


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5          # start temperature for smooth‑max
    lam_growth = 1.30  # speed‑up factor

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {cur_val:.6f}")

        # smooth‑max weights
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (softmax) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – make the surrogate sharper quickly
        lam = min(lam * lam_growth, 1e8)

        # occasionally shrink learning‑rate
        if it % 80_000 == 0 and it > 0:
            lr *= 0.85

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {cur_val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # decreasing step‑size ηₜ = η₀/√(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated‑gradient) update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against underflow
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuner."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val, best_a = val, a.copy()
    T_start, T_end = 0.12, 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤30 %) of mass from i to j
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a, conv, val = a2, conv2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, conv, val = a2, conv2, val2

        it += 1

    return best_a, best_val

    """
    Search for a step‑function with a low evaluation score.

    The routine respects the supplied time budget and always returns a
    valid non‑negative list (the best sequence found so far).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem dimension – keep the same size as the reference solution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # keep the top‑k candidates from the broad search
    top_k = 12
    top_seqs = []  # list of (value, array)

    def _register_candidate(a: np.ndarray):
        nonlocal best_seq, best_val, top_seqs
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

        # maintain a small leaderboard
        if len(top_seqs) < top_k:
            top_seqs.append((val, a.copy()))
            top_seqs.sort(key=lambda x: x[0])
        elif val < top_seqs[-1][0]:
            top_seqs[-1] = (val, a.copy())
            top_seqs.sort(key=lambda x: x[0])

    # --------------------------------------------------------------
    # 1️⃣ Broad random search
    # --------------------------------------------------------------
    # uniform baseline (useful as a mixing partner)
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    _register_candidate(uniform)

    # a) exponential·power family
    for _ in range(4000):
        if time.time() > deadline:
            break
        alpha = 10 ** np.random.uniform(-4.5, 2.0)          # 3e‑5 … 100
        p = np.random.uniform(-8.0, 3.0)                   # power exponent
        a = _candidate_exp_power(alpha, p, n)
        # mix with uniform (helps to moderate the extreme tail)
        lam = np.random.rand()
        a = _candidate_mix(a, lam)
        _register_candidate(a)

    # b) logistic decay
    for _ in range(2000):
        if time.time() > deadline:
            break
        mu = np.random.uniform(0.0, n * 0.3)               # shift of the knee
        sigma = 10 ** np.random.uniform(-4.0, 1.0)         # width
        a = _candidate_logistic(mu, sigma, n)
        lam = np.random.rand()
        a = _candidate_mix(a, lam)
        _register_candidate(a)

    # c) stretched exponential (monotone)
    for _ in range(2000):
        if time.time() > deadline:
            break
        alpha = 10 ** np.random.uniform(-4.5, 0.6)        # scale
        beta = np.random.uniform(0.2, 5.0)                # stretch factor
        a = _candidate_stretched_exp(alpha, beta, n)
        lam = np.random.rand()
        a = _candidate_mix(a, lam)
        _register_candidate(a)

    # d) random Dirichlet spikes (few spikes, non‑uniform masses)
    for k in (64, 128, 256, 512):
        for _ in range(6):
            if time.time() > deadline:
                break
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            a /= a.sum()
            lam = np.random.rand()
            a = _candidate_mix(a, lam)
            _register_candidate(a)

    # e) warm‑start from a previous run, if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _register_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2️⃣ Multi‑start Adam on smooth‑max
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15:
        print("[Phase‑Adam] multi‑start Adam optimisation")
        # we will run Adam from the top‑k candidates and from a few random starts
        candidates = [seq for (_, seq) in top_seqs[:max(3, len(top_seqs))]]
        # add a couple of purely random logits
        for _ in range(3):
            rand_logits = np.random.randn(n)
            a_rand = np.exp(rand_logits - np.max(rand_logits))
            a_rand /= a_rand.sum()
            candidates.append(a_rand)

        for idx, a0 in enumerate(candidates):
            if time.time() > deadline - 10:
                break
            print(f"   [Adam start #{idx+1}/{len(candidates)}]")
            a_opt, val_opt = _adam_optimize(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline - 5,
                max_iters=250_000,
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Phase‑Adam] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 3️⃣ Long exact‑max sub‑gradient descent
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 3:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=8_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4️⃣ Optional simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine‑tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=130_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Final clean‑up and return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if it exists)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
