"""
Propose a low‑value step‑function for the given evaluation.

Algorithm
---------
1. **Broad deterministic sweep** – generate a large and diverse
   family of candidate vectors (uniform, sinusoid, edge‑exponential,
   edge‑power‑law, Beta‑pdf, asymmetric exponentials, mixtures with the
   uniform vector, linear ramp, random Dirichlet draws).  All candidates
   are normalised to sum‑to‑one and evaluated exactly with the same FFT
   routine as ``evaluate_sequence``.  The best candidate becomes the seed.

2. **Projected‑gradient refinement** – starting from the best seed we run
   Euclidean projected‑gradient descent on the exact max‑convolution
   (the gradient is a cross‑correlation with the active set of binding
   constraints).  The step‑size decays as 1/√t and the iterates are always
   projected onto the probability simplex.

3. **Cheap hill‑climber** – in the last few seconds a handful of random
   mass‑transfer moves are performed; a move is kept when it strictly
   improves the objective.

The routine respects the global time budget (default 1000 s) and always
returns the best sequence found so far as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Candidate families
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_exponential_asym(n: int, gam_l: float, gam_r: float) -> np.ndarray:
    left = np.exp(-gam_l * np.arange(n, dtype=np.float64))
    right = np.exp(-gam_r * np.arange(n - 1, -1, -1, dtype=np.float64))
    shape = left + right
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _edge_power_law_asym(n: int, p_l: float, p_r: float) -> np.ndarray:
    left = (np.arange(n, dtype=np.float64) + 1.0) ** p_l
    right = (np.arange(n - 1, -1, -1, dtype=np.float64) + 1.0) ** p_r
    shape = left + right
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _linear_ramp(n: int) -> np.ndarray:
    """Strictly decreasing linear ramp (a_i ∝ n‑i)."""
    shape = np.arange(n, 0, -1, dtype=np.float64)
    shape /= shape.sum()
    return shape


def _mix(shape1: np.ndarray, shape2: np.ndarray, w: float) -> np.ndarray:
    """Convex combination w*shape1 + (1‑w)*shape2, normalised."""
    mix = w * shape1 + (1.0 - w) * shape2
    s = mix.sum()
    if s > 0:
        mix /= s
    else:
        mix = np.full_like(mix, 1.0 / mix.shape[0])
    return mix


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
#  Exact objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Optimisers
# ----------------------------------------------------------------------


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Euclidean projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c / (a.sum() ** 2))

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress print
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad
        a = _project_onto_simplex(a)

        # tiny random jitter near the deadline to escape plateaus
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber with random mass‑transfer moves."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = float(2.0 * n * conv.max() / (a.sum() ** 2))
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = float(2.0 * n * conv2.max() / (a2.sum() ** 2))

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[PGD refinement] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 3.  Final cheap hill‑climber
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=120000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local moves] improved best {val_loc:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
