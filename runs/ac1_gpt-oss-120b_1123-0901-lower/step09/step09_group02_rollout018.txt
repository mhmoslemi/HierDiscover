"""
stepfunction_optimizer_v3.py

A refined multi‑phase optimizer for

    F(a) = 2·n·max(a*a) / (∑a)² , a_i ≥ 0

Key ideas beyond the previous version

* a cheap linear‑programming “mass‑inflation’’ step:
  given the current f, maximise ∑b subject to f∗b ≤ max(f∗f).
  The LP adds a lot of extra mass while respecting the current max‑convolution
  bound, which drives the ratio F down.
* a short projected‑gradient descent on the exact objective (using the true
  sub‑gradient of the max‑convolution).  This deterministic fine‑tuning
  usually squeezes a few ‰ out of the best candidate.
* a slightly longer simulated‑annealing phase and slightly higher‑quality
  initial pool.  The whole routine fits into the 1000 s budget and always
  returns the best sequence seen.

The routine returns a plain ``list`` of non‑negative floats (the step‑function
heights).  All values are clipped to the evaluator’s allowed range
[0.0, 1000.0] before returning.
"""

import time
import numpy as np
import scipy.optimize
import scipy.sparse

# ----------------------------------------------------------------------
# Low‑level utilities (FFT‑based convolution, projections, etc.)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (used for gradient of a∗a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex (≥0, sum=1)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s == 0.0:
        return np.full_like(v, 1.0 / len(v))
    return v / s


def _evaluate_objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective value (no clipping)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Optimisation phases
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Adam on a log‑sum‑exp surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise via soft‑max
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny jitter near the deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle LR decay
        if it % 40_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 350_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient (mirror‑descent) phase."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)
        a = _project_to_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing with mass‑transfer and swaps."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 35_000,
) -> tuple[np.ndarray, float]:

    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance
            if np.random.rand() < 0.01:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val

# Linear‑programming refinement (adds mass without raising the max‑convolution)

def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the dense matrix H such that (H @ b) = f ∗ b  (linear convolution).
    Shape is (2·n‑1, n).  This is O(n²) memory (≈260 MiB for n=4096) but still
    fits comfortably on the execution platform.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    H = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        H[i:i + n, i] = f[i]
    return H


def _lp_refine(f: np.ndarray, n: int, conv_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:

    A_ub = H
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)

    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, 1000.0) for _ in range(n)]

    print("[LP] solving LP ...")
    res = scipy.optimize.linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
        method="highs",
        options={"presolve": True, "time_limit": max(0.0, deadline - time.time() - 2.0)},
    )
    if not res.success:
        print("[LP] LP failed:", res.message)
        return f, _evaluate_objective(f, n, conv_fft_len)

    b = res.x
    # normalise both vectors to sum=1 (objective is scale‑invariant)
    f_norm = _project_to_simplex(f)
    b_norm = _project_to_simplex(b)

    # convex combination (t=0.5 is a safe step size)
    t = 0.5
    new_a = (1.0 - t) * f_norm + t * b_norm
    new_a = _project_to_simplex(new_a)
    new_val = _evaluate_objective(new_a, n, conv_fft_len)
    print(f"[LP] mixed candidate value {new_val:.6f}")

    if new_val < _evaluate_objective(f_norm, n, conv_fft_len):
        return new_a, new_val
    else:
        return f_norm, _evaluate_objective(f_norm, n, conv_fft_len)


# ----------------------------------------------------------------------
# Deterministic projected‑gradient refinement on the exact objective
# ----------------------------------------------------------------------
def _gradient_refine(a0: np.ndarray, n: int, conv_fft_len: int,
                     corr_fft_len: int, deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected (onto simplex) gradient descent on the true objective.
    Uses the exact sub‑gradient of max(a∗a).
    """
    a = _project_to_simplex(a0.copy())
    best_a = a.copy()
    best_val = _evaluate_objective(a, n, conv_fft_len)

    it = 0
    alpha = 0.4  # initial step size
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        S = a.sum()                      # should be 1 after projection
        # active set weights (uniform on the maximisers)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient of max_c w.r.t a
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient of the whole objective (see analysis)
        grad_F = (2.0 * n / (S ** 2)) * (grad_max - (2.0 * max_c / S) * np.ones_like(a))

        improved = False
        step = alpha
        for _ in range(6):
            cand = a - step * grad_F
            cand = _project_to_simplex(cand)
            val = _evaluate_objective(cand, n, conv_fft_len)
            if val < best_val:
                a = cand
                best_a = cand
                best_val = val
                improved = True
                break
            step *= 0.5           # backtrack
        if not improved:
            # if no improvement, shrink step size globally
            alpha *= 0.8
            if alpha < 1e-6:
                break
        it += 1
        if it % 20_000 == 0:
            print(f"[GradRef it={it:06d}] best {best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation (diverse pool)
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet families
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge exponentials
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Edge polynomial (negative exponent)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Positive half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # NEW families -------------------------------------------------------
    # Central Gaussian
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        x = np.arange(n, dtype=np.float64)
        mu = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Symmetric Beta‑type
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        x = np.arange(1, n + 1, dtype=np.float64) / n
        shape = (x ** (alpha - 1)) * ((1.0 - x) ** (alpha - 1))
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Interleaved versions
    interleaved = []
    for a in cand:
        perm = (np.arange(n) * 2) % n
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse spike patterns."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence (as a Python list) it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size – 4096 points gives a good balance of quality / speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start from a previous solution if available
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diversified pool of starter vectors
    # ------------------------------------------------------------------
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – rapid exploration of many candidates (≈25 % budget)
    # ------------------------------------------------------------------
    explore_deadline = time.time() + (deadline - time.time()) * 0.25
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)

        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # short Adam refinement (bounded by a small sub‑deadline)
        sub_dead = min(explore_deadline, time.time() + 5.0)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improved → {val_adam:.6f}")

        # a few sub‑gradient steps for polishing (if time remains)
        if time.time() < explore_deadline and best_seq is not None:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                explore_deadline, max_iters=30_000, eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Sub‑grad improved → {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long sub‑gradient refinement (≈45 % budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 35:
        print("[Refine] Long sub‑gradient optimisation started")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=600_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] New best after sub‑grad → {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – linear‑programming mass‑inflation step (≈10 % budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        print("[LP] Starting LP‑based refinement")
        a_lp, val_lp = _lp_refine(best_seq, n, conv_fft_len, deadline)
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] Improved best → {val_lp:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated‑annealing with mass‑transfer & swaps (≈15 % budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Anneal] Starting simulated‑annealing moves")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=90_000,
            swap_prob=0.12,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] Improved best → {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – permutation‑only hill climbing (≈5 % budget)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Permute] Starting permutation hill‑climb")
        a_perm, val_perm = _permute_hill_climb(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_swaps=30_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Permute] New best after permutation → {val_perm:.6f}")

    # ------------------------------------------------------------------
    # Phase VI – final deterministic gradient refinement (remaining time)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[GradRef] Final projected‑gradient refinement")
        a_grad, val_grad = _gradient_refine(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000,
        )
        if val_grad < best_val:
            best_val = val_grad
            best_seq = a_grad.copy()
            print(f"[GradRef] New best after gradient step → {val_grad:.6f}")

    # ------------------------------------------------------------------
    # Fallback in the unlikely event that nothing was produced
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
