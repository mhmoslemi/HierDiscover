"""
Propose a low‑score non‑negative step function.

Strategy
========
1. **Fast multi‑start sampling** – generate many candidates from a rich set of
   parametric families (exponential, power‑law, beta, double‑Gaussian,
   sinusoid‑power, tent, Dirichlet and a new logistic family).  The ten best
   candidates (lowest objective) are kept in a min‑heap.

2. **Parallel refinement** – each of the top candidates is taken in turn and
   refined by a cascade of convex‑cone methods that directly optimise the
   exact non‑smooth objective:
        * Adam on a smooth‑max surrogate,
        * exponentiated‑gradient sub‑gradient,
        * Euclidean projected‑gradient descent.
   The optimisation is time‑aware: it stops as soon as the global deadline is
   reached.

3. **Final stochastic hill‑climbing** – a cheap random mass‑transfer local
   search is performed on the current best sequence for the last fraction of
   the budget.

All vectors are kept on the probability simplex (non‑negative, sum≈1);
the objective is scale‑invariant, therefore the final clip to
`[0,1000]` does not affect the score.  The implementation works for any
`n` (default 8192) and respects the 1000 s time limit.
"""

import time

# FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence up to the clamping step)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Shape generators
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _logistic_shape(n: int, mu: float, sigma: float) -> np.ndarray:
    """Monotonically decreasing sigmoid shape."""
    x = np.arange(n, dtype=np.float64)
    shape = 1.0 / (1.0 + np.exp((x - mu) / sigma))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape

# Optimisers (identical to reference, only tiny tuning)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # temperature inverse
    lam_factor = 1.01   # slower growth for stability

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.4,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30 % of source mass

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under a 1000 s budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size – 8192 gives more flexibility while still fast enough
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – diversified random sampling (≈40 % of budget)
    top_K = 5
    heap: list[tuple[float, np.ndarray]] = []   # min‑heap by objective value

    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "sin_pow", "tent", "dirichlet", "logistic"
    ]

    phase1_deadline = start_time + 0.40 * (deadline - start_time)

    print("Phase 1 – random parametric sampling (top‑K kept)")

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 2)            # 0.01 … 100
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-10.0, -0.1)
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-1, 1)
            b = 10 ** rng.uniform(-1, 1)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.01, 0.4)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.5, 5.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        else:   # logistic
            # monotone decreasing sigmoid – mu near the right side, sigma moderate
            mu = rng.uniform(0.5 * n, 0.95 * n)
            sigma = 10 ** rng.uniform(np.log10(0.001 * n), np.log10(0.2 * n))
            cand = _logistic_shape(n, mu, sigma)

        val = _objective(cand, conv_fft_len)

        # keep only the K best candidates
        if len(heap) < top_K:
            heapq.heappush(heap, (val, cand.copy()))
        else:
            if val < heap[0][0]:   # heap[0] is the worst of the best‑K
                heapq.heapreplace(heap, (val, cand.copy()))

    # add warm‑start shifted variants (if they exist)
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if len(heap) < top_K:
                heapq.heappush(heap, (val, cand.copy()))
            else:
                if val < heap[0][0]:
                    heapq.heapreplace(heap, (val, cand.copy()))

    # extract sorted list (best first)
    best_candidates = sorted(heap, key=lambda x: x[0])
    best_seq = best_candidates[0][1].copy()
    best_val = best_candidates[0][0]
    print(f"Best after Phase 1: {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – intensive refinement of the top‑K candidates
    # Allocate roughly half of the remaining time, split equally.
    remaining = deadline - time.time()
    if remaining > 10:
        per_candidate_budget = remaining * 0.55 / len(best_candidates)
        print("Phase 2 – refinement of top‑K candidates")
        for rank, (val0, cand0) in enumerate(best_candidates, 1):
            sub_deadline = time.time() + per_candidate_budget
            print(f"  Refining candidate #{rank} (initial {val0:.6f})")
            a_adam, val_adam = _adam_optimize(
                cand0,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=200_000,
                lr_start=0.05,
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"    [Adam] new global best {val_adam:.6f}")

            # continue only if we still have time left
            if time.time() >= sub_deadline:
                continue

            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=150_000,
                eta0=0.85,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"    [Sub‑grad] new global best {val_sub:.6f}")

            if time.time() >= sub_deadline:
                continue

            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=100_000,
                eta0=0.35,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"    [PGD] new global best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap stochastic hill‑climbing (≈5 % of total time)
    if time.time() < deadline - 1.0:
        print("Phase 3 – final local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Safety clipping (matches evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
