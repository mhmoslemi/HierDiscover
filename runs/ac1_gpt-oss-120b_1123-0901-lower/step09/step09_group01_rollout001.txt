"""
Propose a step‑function (non‑negative vector) with a low value of
    2·n·max(a*a)/(sum(a)^2)
where n = len(a).  The algorithm consists of:

1. Broad parametric sweep (including many symmetric / monotone families and
   a fine‑grained line‑search for the most promising families).

2. Random sparse exploration (few hundred non‑zero entries, Dirichlet
   masses).  The best candidate of all sweeps becomes the seed for the
   final optimisation.

3. Long Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
   The temperature (λ) is increased until the surrogate becomes almost a
   true max.

4. Projected‑gradient descent on the exact (non‑smooth) objective
   (sub‑gradient of max‑convolution).  Projection onto the probability
   simplex guarantees feasibility.

5. A short simulated‑annealing fine‑tuning pass.

All phases respect a global 1000 s budget (the deadline is checked
continuously).  The routine always returns a valid non‑negative list;
the final score is printed using the provided `evaluate_sequence`
function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helpers -----------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a (gradient of a*a w.r.t a).
    w must have length 2·len(a)‑1.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Simplex projection ----------------------------------------------------
def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex:
        {x | x_i >= 0, sum_i x_i = 1}.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1.0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero -> return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].sum() / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Exact objective (no clipping, sum is assumed to be 1) -----------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a*a) (sum(a) is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    if max_c < 0:
        max_c = 0.0
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone power‑law a_i ∝ (i+1)^p (p can be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_linear_decrease(n: int) -> np.ndarray:
    """Deterministic linear decrease a_i ∝ (n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = n - i
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_mix_exponential(alpha1: float, alpha2: float,
                               w: float, n: int) -> np.ndarray:
    """Mixture w·exp(-α1·i) + (1‑w)·exp(-α2·i)."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-alpha1 * i) + (1.0 - w) * np.exp(-alpha2 * i)
    a /= a.sum()
    return a


def _candidate_sparse_random(k: int, n: int) -> np.ndarray:
    """Random sparse vector with exactly k non‑zero entries."""
    idx = np.random.choice(n, size=k, replace=False)
    masses = np.random.dirichlet(np.ones(k))
    a = np.zeros(n, dtype=np.float64)
    a[idx] = masses
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # Soft‑max on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for tracking)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # Temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional decay of learning rate
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

    return best_a, best_val


def _projected_gradient_optimize(a0: np.ndarray, n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    """
    Projected (additive) gradient descent on the exact objective.
    Uses a diminishing step size η_t = 0.5 / sqrt(t+1).
    """
    a = np.clip(a0, 0.0, 1.0)
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (allow tiny numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0.0)
        if not active.any():
            grad = np.zeros_like(a)
        else:
            w = active.astype(np.float64)
            w /= w.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = 0.5 / np.sqrt(it + 1.0)
        a = a - eta * grad
        a = _simplex_projection(a)

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[ProjGrad it={it:06d}] new best {val:.6f}")

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing move‑mass‑between‑two‑indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = _objective(a, n, conv_fft_len)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4

    for it in range(max_moves):
        if time.time() >= deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

    return best_a, best_val

# Helper for fine‑grained line search -----------------------------------
def _refine_one_parameter(param_func, param_name: str,
                         param_range: np.ndarray,
                         n: int, conv_fft_len: int,
                         current_best_val: float,
                         deadline: float):
    """
    Simple grid‑refinement for a one‑parameter family.
    Returns (best_param, best_val, best_vector) possibly updating the global best.
    """
    best_param = None
    best_val = current_best_val
    best_vec = None

    # coarse grid
    for p in param_range:
        if time.time() >= deadline:
            break
        a = param_func(p, n)
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_param = p
            best_vec = a.copy()
            print(f"[Refine {param_name}] coarse p={p:.6g} => {val:.6f}")

    # if we have a candidate, perform a few local refinements
    if best_param is not None:
        for _ in range(4):
            lo = max(best_param * 0.5, 1e-8)
            hi = best_param * 2.0
            finer = np.logspace(np.log10(lo), np.log10(hi), 15)
            for p in finer:
                if time.time() >= deadline:
                    break
                a = param_func(p, n)
                val = _objective(a, n, conv_fft_len)
                if val < best_val:
                    best_val = val
                    best_param = p
                    best_vec = a.copy()
                    print(f"[Refine {param_name}] fine p={p:.6g} => {val:.6f}")

    return best_param, best_val, best_vec


def _refine_two_parameters(param_func,
                           p_range: np.ndarray, q_range: np.ndarray,
                           n: int, conv_fft_len: int,
                           current_best_val: float,
                           deadline: float):
    """Coarse‑to‑fine search for a two‑parameter family."""
    best_pair = None
    best_val = current_best_val
    best_vec = None

    for p in p_range:
        for q in q_range:
            if time.time() >= deadline:
                break
            a = param_func(p, q, n)
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_pair = (p, q)
                best_vec = a.copy()
                print(f"[Refine two‑param] p={p:.3g}, q={q:.3g} => {val:.6f}")

    return best_pair, best_val, best_vec

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.  Returns a list of non‑negative heights (float).
    The routine respects the given time budget and always returns a feasible
    vector (sum>0, entries≥0).  Intermediate progress is printed.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem size -------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)          # 8192
    corr_fft_len = _next_pow2(3 * n - 2)          # 12288 (sufficient)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep + fine refinements ----------------------
    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ----- basic families (same as before, but a bit denser) ------------
    for p in np.linspace(-4.0, -0.05, 40):
        _maybe_update(_candidate_power(p, n))

    for alpha in np.logspace(-4, 1.0, 35):
        _maybe_update(_candidate_exponential(alpha, n))

    for alpha in np.logspace(-3, 0.5, 20):
        for beta in np.linspace(0.3, 3.0, 15):
            _maybe_update(_candidate_stretched(alpha, beta, n))

    for a_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for b_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-2.5, -0.2, 25):
        _maybe_update(_candidate_double_gaussian(sig, n))

    for alpha in np.logspace(-4, 1.3, 30):
        _maybe_update(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-3.5, 1.5, 35):
        _maybe_update(_candidate_left_power(p, n))

    # deterministic linear decrease
    _maybe_update(_candidate_linear_decrease(n))

    # ----- random spike patterns -----------------------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # ----- warm start from previous run ----------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    # ----- fine‑grained refinement of the most promising families -------
    # we only run these if we still have a reasonable amount of time.
    if time.time() < deadline - 150.0:
        # left exponential
        a_range = np.logspace(-4, 1.5, 30)
        _, val, vec = _refine_one_parameter(_candidate_left_exponential,
                                            "left_exp",
                                            a_range,
                                            n, conv_fft_len,
                                            best_val,
                                            deadline)
        if vec is not None and val < best_val:
            best_val, best_seq = val, vec.copy()

        # symmetric exponential
        a_range = np.logspace(-4, 1.0, 30)
        _, val, vec = _refine_one_parameter(_candidate_exponential,
                                            "sym_exp",
                                            a_range,
                                            n, conv_fft_len,
                                            best_val,
                                            deadline)
        if vec is not None and val < best_val:
            best_val, best_seq = val, vec.copy()

        # power‑law (symmetric)
        p_range = np.linspace(-4.0, -0.05, 40)
        _, val, vec = _refine_one_parameter(_candidate_power,
                                            "sym_power",
                                            p_range,
                                            n, conv_fft_len,
                                            best_val,
                                            deadline)
        if vec is not None and val < best_val:
            best_val, best_seq = val, vec.copy()

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Random sparse exploration ---------------------------------------
    if time.time() < deadline - 120.0:
        print("[Sparse] random sparse search")
        while time.time() < deadline - 120.0:
            k = np.random.randint(30, 200)          # number of non‑zeros
            a = _candidate_sparse_random(k, n)
            val = _objective(a, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Sparse] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam optimisation (smooth‑max) ---------------------------------
    if best_seq is not None and time.time() < deadline - 60.0:
        print("[Phase‑Adam] long Adam optimisation")
        adam_deadline = min(deadline - 30.0, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          adam_deadline,
                                          max_iters=800_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected gradient descent on the exact max --------------------
    if best_seq is not None and time.time() < deadline - 30.0:
        print("[Phase‑ProjGrad] projected gradient descent")
        pg_deadline = min(deadline - 5.0, time.time() + 45.0)
        a_pg, val_pg = _projected_gradient_optimize(best_seq, n,
                                                    conv_fft_len, corr_fft_len,
                                                    pg_deadline,
                                                    max_iters=1_200_000)
        if val_pg < best_val:
            best_val = val_pg
            best_seq = a_pg.copy()
            print(f"[Phase‑ProjGrad] improved to {val_pg:.6f}")

    # ------------------------------------------------------------------
    # 5) Simulated‑annealing fine‑tuning (if any time left) -------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase‑Anneal] final simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          deadline,
                                          max_moves=200_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (non‑negative, sum>0, bounded)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        # evaluator may not be importable in the testing harness
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
