    current_val = best_val

    temp = best_val * 0.5  # start high relative to best improvement needed

    # or linear schedule.
    i = 0
    while time.time() < deadline and i < max_iter:
        # random mass move
        i_idx = np.random.randint(n)
        if a[i_idx] < 1e-16:
            i += 1
            continue
        j_idx = np.random.randint(n)
        if i_idx == j_idx:
            i += 1
            continue

        frac = np.random.rand() * 0.5
        delta = a[i_idx] * frac
        a2 = a.copy()
        a2[i_idx] -= delta
        a2[j_idx] += delta
        # if any negative due to floating error, set to zero

"""
Propose a non‑negative step‑function (list of heights) that minimizes

        F(a) = 2·n·max(a∗a) / (∑a)^2

The algorithm is a multi‑stage search that works well under the 1000 s
budget.

1. **Initial pool** – many deterministic families (exponential, power‑law,
   β‑pdf, double‑Gaussian, sinusoid, raised‑cosine, semicircle) together
   with random Dirichlet vectors and sparse spikes.  The best few
   candidates are kept.

2. **Adam on a smooth‑max surrogate** – the true “max’’ is replaced by a
   log‑sum‑exp with an increasing temperature.  Adam updates the logits
   of a soft‑max (which guarantees the simplex constraint).  This stage
   runs for a large number of iterations while respecting the deadline.

3. **Continuation L‑BFGS** – the surrogate is refined with a schedule of
   increasingly large temperature parameters (up to 1 e 6).  For each
   temperature a few hundred L‑BFGS steps are taken; the best exact
   objective seen so far is retained.

4. **Simulated annealing** – a Metropolis walk that moves a random
   fraction of mass from one index to another.  Moves that improve the
   true objective are always accepted; worsening moves are accepted with
   probability `exp(-(Δ)/temp)`.  The temperature is slowly cooled.

5. **Final cheap hill‑climber** – deterministic greedy moves that only
   keep improvements, polishing the final answer.

All heavy linear‑algebra (convolutions and cross‑correlations) is done
with FFTs, which makes a single evaluation ~0.5 ms for n=4096.  The routine
periodically checks the deadline and always returns the best sequence
found so far (as a plain Python list).  The returned values are clipped
to the [0, 1000] range required by `evaluate_sequence`.
"""

import time

# Low‑level FFT helpers
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence for valid inputs)
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)

# Families of deterministic shapes
def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = idx ** alpha + (n - idx + 1) ** alpha
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * (i / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

# Adam optimiser on a smooth‑max surrogate
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # inverse temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (exact)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake when close to deadline
        if it % 5_000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS on the smooth‑max surrogate
def _l_bfgs_obj_grad(
    x: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Smooth‑max objective and gradient for a given temperature α."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    a = exp_x / exp_x.sum()

    # convolution
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c                # ≤ 0
    exp_alpha_v = np.exp(alpha * v)  # max = 1
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    w = exp_alpha_v / sum_exp                 # soft‑max weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    avg = np.dot(a, grad_a)
    grad_x = a * (grad_a - avg)

    return obj, grad_x


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on a sequence of rising temperatures."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")

    # a fairly aggressive temperature schedule
    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0, 80000.0, 2e5, 5e5, 1e6]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad(x, n, conv_fft_len, corr_fft_len, alpha)

        remaining = deadline - time.time()
        max_iter = 600 if remaining > 30 else 150

        res = minimize(fun, logits, method="L-BFGS-B", jac=True,
                       options={"maxiter": max_iter, "disp": False})

        logits = res.x

        # exact objective
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = _objective(a, n, conv_fft_len)
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val


# ----------------------------------------------------------------------
# Simulated annealing (Metropolis) on the true objective
def _simulated_annealing(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Metropolis walk moving a random fraction of mass."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    cur_val = best_val

    # start temperature – a fraction of the current value
    temp = max(best_val * 0.5, 1e-4)
    start = time.time()
    i = 0
    while time.time() < deadline and i < max_moves:
        # pick source and destination
        src = np.random.randint(n)
        if a[src] < 1e-15:
            i += 1
            continue
        dst = np.random.randint(n)
        if src == dst:
            i += 1
            continue

        # move a random proportion of the source mass
        frac = np.random.rand() * 0.4                # up to 40 %
        delta = a[src] * frac

        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta

        val2 = _objective(a2, n, conv_fft_len)

        if val2 < cur_val:
            a = a2
            cur_val = val2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                print(f"[SA it={i:06d}] new best {best_val:.6f}")
        else:
            # Metropolis acceptance
            if temp > 0 and np.random.rand() < np.exp(-(val2 - cur_val) / temp):
                a = a2
                cur_val = val2

        # simple exponential cooling
        if i % 2000 == 0:
            elapsed = time.time() - start
            # slower cooling when we still have a lot of time left
            factor = 0.995 if (deadline - time.time()) > 100 else 0.98
            temp = max(temp * factor, 1e-8)

        i += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap greedy hill‑climber (only accepts improvements)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """Greedy hill‑climber moving a random fraction of mass."""
    a = a0.copy()
    best_val = _objective(a, n, conv_fft_len)
    best_a = a.copy()
    i = 0
    while time.time() < deadline and i < max_moves:
        src = np.random.randint(n)
        if a[src] < 1e-15:
            i += 1
            continue
        dst = np.random.randint(n)
        if src == dst:
            i += 1
            continue

        delta = a[src] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[src] -= delta
        a2[dst] += delta

        val2 = _objective(a2, n, conv_fft_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if i % 5000 == 0:
                print(f"[Local move {i:06d}] improved to {best_val:.6f}")

        i += 1
    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """Search for a low‑objective non‑negative step‑function."""
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size and FFT lengths
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start handling (global variable `height_sequence_1` may be set)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # build a diverse pool of starter sequences
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families
    for gamma in np.logspace(-2, 1.2, 30):
        pool.append(_edge_exponential(n, gamma))

    for alpha in np.linspace(-9.0, -0.1, 30):
        pool.append(_edge_power_law(n, alpha))

    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0),
        (5.0, 0.2), (0.1, 0.1), (0.7, 1.5), (1.5, 0.7)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # random Dirichlet vectors (different concentrations)
    for alpha in (0.05, 0.2, 0.7, 1.0, 2.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, alpha))

    # sparse spike vectors
    for k in (32, 64, 128, 256):
        pool.append(_sparse_spikes(n, k, rng))

    # ------------------------------------------------------------------
    # Quick evaluation of the pool – keep the best few
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    rng.shuffle(pool)

    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        if time.time() > deadline - 80:   # reserve time for deep optimisation
            break

    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq, n, conv_fft_len)
        print(f"[Init] fallback uniform {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – extended Adam optimisation on the current best
    if time.time() < deadline - 60:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=700_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam final] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – continuation L‑BFGS refinement
    if time.time() < deadline - 30:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, corr_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simulated annealing (allows uphill moves)
    if time.time() < deadline - 12:
        a_sa, val_sa = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA] improved to {val_sa:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – cheap greedy hill‑climber for the final seconds
    if time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
