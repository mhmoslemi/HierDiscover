"""
Propose a step‑function with a low evaluation value.

The search proceeds in three stages within the allocated time budget:

1. **Massive random sampling** – dense parametric families (exponential,
   power‑law, beta, double‑Gaussian, sinusoid‑power, tent) and also very
   sparse random supports are sampled.  Each candidate is evaluated with a
   fast FFT‑based convolution and the best is kept.

2. **Multiple short gradient refinements** – the current best vector is
   used as a warm‑start for several independent Adam optimisations
   (different random seeds, different learning‑rate schedules).  The best
   result of all runs is retained.

3. **Final stochastic hill‑climbing** – a cheap “move a random fraction
   of mass from one index to another’’ local search is run until the
   deadline.  The best sequence found is returned after clipping to the
   limits accepted by ``evaluate_sequence``.
"""
import time
import numpy as np

# ----------------------------------------------------------------------
# Simple FFT‑based convolution helpers -----------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to clamping) ------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) – the quantity we want to minimise."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Parametric shape generators -------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _random_sparse_shape(n: int, rng: np.random.Generator) -> np.ndarray:
    """Create a random sparse distribution (many zeros)."""
    # choose a support size – favour medium‑sized supports
    m = int(rng.integers(20, 120))
    inds = rng.choice(n, size=m, replace=False)
    # draw positive weights (exponential gives a skewed distribution)
    w = rng.exponential(scale=1.0, size=m)
    w /= w.sum()
    a = np.zeros(n, dtype=np.float64)
    a[inds] = w
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 300_000,
                   lr_start: float = 0.05) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate for the max‑convolution
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                  corr_fft_len: int, deadline: float,
                  max_iters: int = 300_000,
                  eta0: float = 0.35) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD  it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)
        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.4)   # up‑to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    evaluate_sequence().  The routine mixes large‑scale random sampling,
    many short Adam refinements (different seeds) and a final cheap hill‑climb.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # tiny safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – very broad random sampling (dense + sparse)
    best_seq = None
    best_val = float("inf")
    families = ["edge_exp", "edge_pow", "beta", "double_gauss",
                "sin_pow", "tent", "dirichlet", "sparse"]
    print("Phase 1 – massive random sampling")
    while time.time() < deadline - 200:   # reserve ~200 s for optimisation phases
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-3, 3)          # 0.001 … 1000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-12.0, -0.1)          # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-1.5, 1.5)
            b = 10 ** rng.uniform(-1.5, 1.5)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.02, 8.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        else:   # sparse
            cand = _random_sparse_shape(n, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # include warm‑start rotations
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – many short Adam refinements (different seeds)
    print("Phase 2 – Adam refinements with restarts")
    while time.time() < deadline - 30:          # keep ~30 s for final hill‑climb
        # small random perturbation around the current best
        perturb = best_seq * (1 + 0.02 * rng.standard_normal(n))
        perturb = np.maximum(perturb, 0.0)
        if perturb.sum() < 1e-12:
            perturb = np.random.dirichlet(np.ones(n))
        perturb = perturb / perturb.sum()

        # run a brief Adam optimisation (time‑bounded)
        sub_deadline = min(deadline - 5.0, time.time() + 12.0)   # ~12 s per run
        a_opt, val_opt = _adam_optimize(
            perturb,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=200_000,
            lr_start=0.06,
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Adam restart] improved to {val_opt:.6f}")

        # optional short PGD refinement (quick)
        pgd_deadline = min(deadline - 5.0, time.time() + 5.0)
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=80_000,
            eta0=0.4,
        )
        if val_pgd < best_val:
            best_val, best_seq = val_pgd, a_pgd.copy()
            print(f"[PGD after Adam] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing on the final seconds
    if best_seq is not None and time.time() < deadline - 1.0:
        print("Phase 3 – final local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to match evaluate_sequence's safety bounds
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
