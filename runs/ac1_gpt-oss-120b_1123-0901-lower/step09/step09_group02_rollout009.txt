"""
Propose a step‑function (non‑negative vector) that minimises
    F(a) = 2·n·max(a*a) / (∑a)²
using a purely numerical multi‑phase optimisation:

1.   Build a diverse pool of seed vectors (Dirichlet, Gaussians,
     Beta‑type, triangular, interleaved permutations, spikes).
2.   For every seed run a fast “smooth‑max” Adam optimisation
     (soft‑max parametrisation) – this quickly eliminates very bad
     candidates.
3.   The current best candidate is refined for the remainder of the
     budget by a long exponentiated‑gradient sub‑gradient descent on the
     exact max‑convolution.
4.   A generous simulated‑annealing phase (mass‑transfer + occasional
     swaps) is performed on the incumbent.
5.   Finally a short permutation‑only hill‑climb and a light smoothing
     post‑process attempt to shave off the last bits.

The routine respects the time budget (default 1000 s) and always returns
the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (full length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a required for the (sub‑)gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_len: int) -> float:
    """Exact value of 2·n·max(a*a) / (∑a)²."""
    conv = _full_convolution(a, conv_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _initial_pool(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a heterogeneous set of starting vectors."""
    candidates = []

    # warm start (if any) and cyclic shifts
    if warm is not None:
        candidates.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            candidates.append(np.roll(warm, shift))

    # Dirichlet draws – different concentration → different sparsity
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        candidates.append(np.random.dirichlet(alpha * np.ones(n)))

    # Symmetric centred Gaussian (single bump)
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    for sigma in (0.04, 0.08, 0.12, 0.20):
        g = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        g /= g.sum()
        candidates.append(g)

    # Symmetric Beta‑type (U‑shaped for α<1, peaked for α>1)
    xv = np.arange(1, n + 1, dtype=np.float64) / n
    for a in (0.7, 1.0, 1.5, 2.5, 4.0):
        b = (xv ** (a - 1)) * ((1.0 - xv) ** (a - 1))
        b /= b.sum()
        candidates.append(b)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n)).astype(np.float64)
    tri /= tri.sum()
    candidates.append(tri)

    # Exponential spike (left + right)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        spike = left + right
        spike /= spike.sum()
        candidates.append(spike)

    # Uniform
    candidates.append(np.full(n, 1.0 / n))

    # Interleaved permutation of every candidate – helps to decorrelate
    perm = (np.arange(n) * 2) % n
    interleaved = [c[perm] for c in candidates]
    candidates.extend(interleaved)

    # Sparse spikes (fixed number of non‑zeros)
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        candidates.append(a)

    return candidates

    best_val = _objective(best_a, n, conv_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max → a on simplex
        mx = logits.max()
        exp = np.exp(logits - mx)
        a = exp / exp.sum()

        # exact objective (for bookkeeping)
        val = _objective(a, n, conv_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_len)
        M = conv.max()
        w = np.exp(lam * (conv - M))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase smooth‑max temperature
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter close to deadline
        if it % 5000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # very gentle learning‑rate decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgrad_expgrad(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient sub‑gradient optimisation."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        M = conv.max()
        val = 2.0 * n * M / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[SubGrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, M, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_len)
        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad)

        # renormalise, guard against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 120_000,
    swap_prob: float = 0.13,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing with mass‑transfer and occasional swaps."""
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start, T_end = 0.09, 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            i, j = np.random.randint(n, size=2)
            if i != j:
                a2 = a.copy()
                a2[i], a2[j] = a2[j], a2[i]
            else:
                it += 1
                continue
        else:
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.45
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a, val = a2, val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a, val = a2, val2

        it += 1

    return best_a, best_val


def _permute_hill(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_swaps: int = 35_000,
) -> tuple[np.ndarray, float]:
    """Pure permutation hill‑climb (swap‑only)."""
    a = a0.copy()
    val = _objective(a, n, conv_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue
        a[i], a[j] = a[j], a[i]
        val2 = _objective(a, n, conv_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # tiny uphill acceptance to escape plateaus
            if np.random.rand() < 0.012:
                val = val2
            else:
                a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val


def _smooth_pass(a: np.ndarray) -> np.ndarray:
    """One light smoothing pass – 3‑point averaging, then re‑normalise."""
    b = np.empty_like(a)
    b[0] = (a[0] + a[1]) / 2.0
    b[-1] = (a[-2] + a[-1]) / 2.0
    b[1:-1] = (a[:-2] + a[1:-1] + a[2:]) / 3.0
    b = np.maximum(b, 0.0)
    if b.sum() > 0:
        b /= b.sum()
    return b

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.

    The function respects the supplied time budget and always returns the
    best sequence (as a plain Python list) discovered before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # Problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm‑start handling (global variable may contain a previous solution)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Build the candidate pool
    pool = _initial_pool(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase I – quick Adam refinement on many diverse seeds (≈15 % budget)
    phase_i_deadline = time.time() + (deadline - time.time()) * 0.15
    while pool and time.time() < phase_i_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:          # sanity, shouldn't happen
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(phase_i_deadline, time.time() + 3.0)  # short Adam run
        a_adam, val_adam = _adam_opt(
            a0, n, conv_len, corr_len, sub_dead, max_iters=100_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improvement → {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase II – intensive exponentiated‑gradient sub‑gradient optimisation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    print("[Phase II] Starting long sub‑gradient optimisation")
    a_sub, val_sub = _subgrad_expgrad(
        best_seq,
        n,
        conv_len,
        corr_len,
        deadline,
        max_iters=600_000,
        eta0=0.9,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Phase II] New best after sub‑grad → {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase III – simulated‑annealing (mass transfers + swaps)
    if time.time() < deadline - 30:
        print("[Phase III] Simulated‑annealing")
        a_ann, val_ann = _anneal(
            best_seq,
            n,
            conv_len,
            deadline,
            max_moves=100_000,
            swap_prob=0.13,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase III] Improved best → {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase IV – permutation‑only hill‑climbing
    if time.time() < deadline - 10:
        print("[Phase IV] Permutation hill‑climb")
        a_perm, val_perm = _permute_hill(
            best_seq,
            n,
            conv_len,
            deadline,
            max_swaps=40_000,
        )
        if val_perm < best_val:
            best_val = val_perm
            best_seq = a_perm.copy()
            print(f"[Phase IV] New best after permute → {val_perm:.6f}")

    # --------------------------------------------------------------
    # Phase V – tiny smoothing passes (often shave off a few 1e‑4)
    if time.time() < deadline - 2:
        print("[Phase V] Light smoothing")
        a_s = best_seq.copy()
        for _ in range(3):
            a_s = _smooth_pass(a_s)
        val_s = _objective(a_s, n, conv_len)
        if val_s < best_val:
            best_val = val_s
            best_seq = a_s
            print(f"[Phase V] Smoothing improved → {val_s:.6f}")

    # --------------------------------------------------------------
    # Final safety – enforce evaluator’s limits and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
