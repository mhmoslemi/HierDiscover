def _left_exponential(n, gamma):
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape.fill(1.0 / n)
    return shape

"""
Propose a step‑function (non‑negative vector) that minimizes

        F(a) = 2·n·max(a∗a) / (sum(a)^2)

The main insight is that a *one‑sided* exponential decay
a_i ∝ exp(−γ·i) (γ ≈ 2/n) yields a much smaller
max‑convolution than the symmetric families used before.
The search therefore consists of

1.  Building a large pool of deterministic families (including the new
    left‑exponential shape) together with random Dirichlet / sparse‑spike
    vectors.
2.  Evaluating every candidate (fast FFT‑based convolution) and keeping the
    best.
3.  A few hundred steps of projected (sub‑)gradient descent on the exact
    objective (gradient = 2·a shifted by the index of the maximal
    convolution entry).
4.  A cheap stochastic hill‑climber for the remaining time.

All helpers are defined at top level, the routine respects the 1000 s
deadline and always returns the best sequence found.
"""

import time
import numpy as np
from typing import List

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate families
def _random_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _left_exponential(n: int, gamma: float) -> np.ndarray:
    """One‑sided exponential (decreasing from the left, NOT symmetric)."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    s = shape.sum()
    shape = shape / s if s > 0 else np.full(n, 1.0 / n)
    return shape


# ----------------------------------------------------------------------
# Exact objective (fast, identical to evaluate_sequence for non‑negative vectors)
def _objective(a: np.ndarray, conv_len: int) -> float:
    """
    Returns 2·n·max(a*a)  (the denominator sum(a)**2 is 1 because a is
    normalised to sum==1).  The function works for any non‑negative a.
    """
    a = np.clip(a, 0.0, 1000.0)
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    a = a / s                     # normalise – makes the formula scale‑invariant
    fa = np.fft.rfft(a, conv_len)
    conv = np.fft.irfft(fa * fa, conv_len)[: 2 * a.shape[0] - 1]
    maxc = conv.max()
    return 2.0 * a.shape[0] * maxc


# ----------------------------------------------------------------------
# Projected (sub‑)gradient descent on the exact objective
def _projected_subgradient(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_iters: int = 1500,
) -> tuple[np.ndarray, float]:
    """
    Simple projected sub‑gradient descent.
    Gradient of max(conv) w.r.t. a is 2·a shifted by the index of the
    maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_len)

    lr = 0.2                     # initial step size
    it = 0
    while time.time() < deadline - 0.2 and it < max_iters:
        # convolution and index of the maximal entry
        fa = np.fft.rfft(a, conv_len)
        conv = np.fft.irfft(fa * fa, conv_len)[: 2 * n - 1]
        k = int(conv.argmax())

        # gradient of the *max* entry (sub‑gradient)
        grad = np.zeros_like(a)
        lo = max(0, k - (n - 1))
        hi = min(k, n - 1)
        idx = np.arange(lo, hi + 1, dtype=int)
        grad[idx] = 2.0 * a[k - idx]

        # projected gradient step (objective = 2·n·maxc, gradient scaled by 2·n)
        a -= lr * (2.0 * n) * grad
        a = np.maximum(a, 0.0)               # stay non‑negative
        a = _project_onto_simplex(a)          # enforce sum(a)=1

        val = _objective(a, conv_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[PGD it={it:04d}] new best {val:.6f}")

        # simple learning‑rate decay
        if it % 200 == 0 and lr > 1e-4:
            lr *= 0.8
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber (final polishing)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, conv_len)

    it = 0
    while time.time() < deadline - 0.1 and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a fraction of mass from i to j
        delta = a[i] * (np.random.rand() * 0.25)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2, conv_len)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_len = _next_pow2(2 * n - 1)

    # ---- warm start (if a previous solution is stored) ----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ---- build a diverse pool of starter vectors ----
    pool: List[np.ndarray] = []

    # warm start and a few cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 8, n // 4, 3 * n // 8):
            pool.append(np.roll(warm, shift))

    # random Dirichlet vectors
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha))

    # symmetric edge exponential (the previous state‑of‑the‑art family)
    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))

    # edge power‑law
    for alpha in np.linspace(-8, -0.2, 20):
        pool.append(_edge_power_law(n, alpha))

    # β‑pdf family
    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0),
        (5.0, 0.2), (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    # double Gaussian
    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # sinusoid, raised cosine, semicircle, uniform
    pool.append(_sinusoid(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))
    pool.append(_uniform(n))

    # NEW: one‑sided exponential (the key improvement)
    for gamma in np.logspace(-4, -2, 30):
        pool.append(_left_exponential(n, gamma))

    # sparse random spikes (different sparsities)
    for k in (32, 64, 128, 256, 512, 1024):
        pool.append(_sparse_spikes(n, k, rng))

    # --------------------------------------------------------------
    # Quick scan of the pool – keep the best candidate
    rng.shuffle(pool)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for idx, cand in enumerate(pool):
        if time.time() > deadline - 120:   # keep ~2 min for deeper phases
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 1e-12:
            continue
        a = a / a.sum()
        val = _objective(a, conv_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

    if best_seq is None:
        best_seq = _uniform(n)
        best_val = _objective(best_seq, conv_len)

    # --------------------------------------------------------------
    # Phase 2 – projected sub‑gradient descent on the exact objective
    if time.time() < deadline - 30:
        a_pg, val_pg = _projected_subgradient(
            best_seq, n, conv_len, deadline, max_iters=2000
        )
        if val_pg < best_val:
            best_val = val_pg
            best_seq = a_pg.copy()
            print(f"[PGD] new best {val_pg:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – cheap random hill‑climber for the final seconds
    if time.time() < deadline - 5:
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_len, deadline, max_moves=200_000
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Final safety – clip to the limits enforced by the evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
