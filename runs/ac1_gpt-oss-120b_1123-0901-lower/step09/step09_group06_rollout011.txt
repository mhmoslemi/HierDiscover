def get_active_indices(conv, M, top_k=10, rel_tol=1e-6):
    # indices where conv >= M - rel_tol * M
    tol = max(1e-12, M * rel_tol)
    idxs = np.where(conv >= M - tol)[0]
    if idxs.size < top_k:
        # add top_k highest indices

        idxs = np.unique(np.concatenate([idxs, top_idx]))
    return idxs

"""
propose_candidate – hybrid LP / gradient / hill‑climber optimizer

The routine builds a non‑negative step‑function that minimizes
    F(a) = 2·n·max(a∗a) / (sum(a)^2).

1. Warm start: reuse the previous best sequence if available,
   otherwise start from a Dirichlet sample.
2. LP phase: given the current shape f (sum = √(2n)), solve the linear
   program
        maximise   Σ b_i
        subject to conv(f,b) ≤ M   (M = max(conv(f,f)))
                b_i ≥ 0
   The optimum provides a new candidate g = (√(2n)/Σb)·b whose
   autocorrelation is ≤ M·√(2n)/Σb < M.  The new shape is obtained by
   mixing f and g (or by full replacement if g alone already improves).
   Repeating this step monotonically reduces the exact objective.
3. Adam phase: refine the best shape found by the LP using a smooth‑max
   surrogate (log‑sum‑exp) on the simplex.
4. Final hill‑climber: cheap random mass moves that accept only improvements.

The whole search respects the 1000 s budget and always returns the best
sequence discovered.
"""

import time
import numpy as np
import cvxpy as cp
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# Low‑level utilities ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate) ---------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on the log‑sum‑exp surrogate (simplex via soft‑max)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber ---------------------------------------------
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Move a random fraction of mass, keep only improving steps."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming improvement -----------------------------------------
def _lp_max_sum(basis: np.ndarray, M: float, target_sum: float) -> tuple[np.ndarray, float]:
    """
    Solve   maximise Σ b_i   subject to   conv(basis,b) ≤ M   and   b ≥ 0.
    Returns the optimal b vector and its sum.
    """
    n = basis.shape[0]
    # Parameter for the fixed basis (constant during solve)
    a_param = cp.Parameter(n, value=basis.astype(np.float64))
    b_var = cp.Variable(n, nonneg=True)

    # Linear convolution of variable b with constant a
    conv_expr = cp.conv(b_var, a_param)          # length 2·n‑1
    prob = cp.Problem(cp.Maximize(cp.sum(b_var)), [conv_expr <= M])
    try:
        prob.solve(solver=cp.ECOS, warm_start=True, max_iters=5000,
                   abstol=1e-9, reltol=1e-9, feastol=1e-9)
    except Exception as exc:                         # pragma: no cover
        print("LP solve exception:", exc)
        return None, None

    if b_var.value is None:
        return None, None

    b_opt = np.asarray(b_var.value, dtype=np.float64)
    return b_opt, float(b_opt.sum())


def _lp_improvement_phase(
    f: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    target_sum: float,
) -> tuple[np.ndarray, float]:
    """
    Repeated LP improvement (Section 2 of the description).
    Returns the best vector found and its exact objective value.
    """
    best_f = f.copy()
    conv = _full_convolution(best_f, conv_fft_len)
    best_M = conv.max()
    best_val = best_M                     # because sum = target_sum = √(2n)

    iteration = 0
    while time.time() < deadline and iteration < 200:
        conv = _full_convolution(f, conv_fft_len)
        M = conv.max()

        # Solve LP
        b_opt, sum_b = _lp_max_sum(f, M, target_sum)
        if b_opt is None:
            print("[LP] No feasible solution, terminating LP phase.")
            break
        if sum_b <= target_sum * (1.0 + 1e-9):
            # No slack – cannot improve further
            print(f"[LP it={iteration}] sum_b≈target ({sum_b:.6f}), stop.")
            break

        # Scale the LP solution to the required total mass
        scale = target_sum / sum_b
        g = scale * b_opt

        # Evaluate g alone
        conv_g = _full_convolution(g, conv_fft_len)
        M_g = conv_g.max()

        # Decide whether to replace or to mix
        if M_g < M:
            f_new = g
            new_M = M_g
            action = "replace"
        else:
            t = 0.5
            f_new = (1.0 - t) * f + t * g
            # renormalise to keep exact target sum (numerical drift)
            f_new = f_new * (target_sum / f_new.sum())
            conv_new = _full_convolution(f_new, conv_fft_len)
            new_M = conv_new.max()
            action = "mix"

        if new_M < best_val:
            best_val = new_M
            best_f = f_new.copy()
            print(f"[LP it={iteration}] new best M={best_val:.6f} (action={action})")

        f = f_new
        iteration += 1

    return best_f, best_val
