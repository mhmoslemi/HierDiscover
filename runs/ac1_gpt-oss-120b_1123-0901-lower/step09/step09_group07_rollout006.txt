Pseudo:

from scipy.sparse import csr_matrix, lil_matrix

def build_conv_matrix(f):
    n = len(f)
    rows = []
    cols = []
    data = []
    for i in range(n):
        row_start = i
        row_end = i + n  # exclusive
        # rows row_start to row_end-1 correspond to convolution shift indices
        # For each shift k in [i, i+n-1], coefficient is f[i] * g[k-i] where column = i.
        # So set A[k,i] = f[i] for k = i .. i+n-1
        rows.extend(range(row_start, row_end))
        cols.extend([i]*n)
        data.extend([f[i]]*n)
    from scipy.sparse import coo_matrix
    A = coo_matrix((data, (rows, cols)), shape=(2*n-1, n), dtype=np.float64).tocsr()
    return A

def _upsample_seq(seq, target_n):
    cur_n = len(seq)
    if cur_n == target_n:
        return seq.copy()
    # Simple interpolation: use np.interp over new indices
    x_old = np.linspace(0, 1, cur_n)
    x_new = np.linspace(0, 1, target_n)
    up = np.interp(x_new, x_old, seq)
    # Ensure nonnegatives
    up = np.maximum(up, 0.0)
    # Normalize sum to 1
    s = up.sum()
    if s > 0:
        up = up / s
    return up

def _next_pow2(x: int) -> int:

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient calculations."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (identical to evaluate_sequence up to clamping) --------------
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Parametric shape generators --------------------------------------------
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _inverse_sqrt_shape(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = 1.0 / np.sqrt(i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    shape /= ss
    return shape


def _log_shape(n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    shape = -np.log(i / n)          # decreasing, zero at i=n
    shape -= shape.min()            # shift to be non‑negative
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_sparse_shape(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k non‑zero entries placed uniformly at random."""
    if k >= n:
        return np.full(n, 1.0 / n, dtype=np.float64)
    pos = rng.choice(n, size=k, replace=False)
    vals = rng.exponential(scale=1.0, size=k)
    shape = np.zeros(n, dtype=np.float64)
    shape[pos] = vals
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _interpolate_to_n(old: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    if old.shape[0] == target_n:
        return old.copy()
    x_old = np.linspace(0.0, 1.0, old.shape[0])
    x_new = np.linspace(0.0, 1.0, target_n)
    up = np.interp(x_new, x_old, old)
    up = np.maximum(up, 0.0)
    s = up.sum()
    if s > 0:
        up /= s
    else:
        up = np.full(target_n, 1.0 / target_n, dtype=np.float64)
    return up

# Optimisers (identical to the reference implementation) -----------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # start temperature (inverse)
    lam_factor = 1.02   # slower rise – more stable early on

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max surrogate
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
    eta0: float = 0.35,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (rng.random() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the 1000 s time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – 8192 gives a good balance between quality and speed.
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)   # 16384
    corr_fft_len = _next_pow2(3 * n - 2)   # 32768

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        warm = np.maximum(warm, 0.0)
        warm = _interpolate_to_n(warm, n)
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – rich random‑parameter sweep
    best_seq = None
    best_val = float("inf")

    families = [
        "edge_exp", "edge_pow", "beta", "double_gauss",
        "sin_pow", "tent", "inv_sqrt", "log", "sparse", "dirichlet"
    ]

    phase1_deadline = min(deadline - 5.0,
                          time.time() + 0.35 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")

    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 3)          # 0.01 … 1000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-12.0, -0.05)         # stronger edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 2)              # 0.01 … 100
            b = 10 ** rng.uniform(-2, 2)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.45)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 6.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "inv_sqrt":
            cand = _inverse_sqrt_shape(n)
        elif fam == "log":
            cand = _log_shape(n)
        elif fam == "sparse":
            k = rng.integers(3, 30)                 # number of spikes
            cand = _random_sparse_shape(n, k, rng)
        else:   # dirichlet
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # include warm‑start variations (cyclic shifts)
    if warm is not None:
        for shift in (0, n // 8, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – deep gradient refinement (Adam → Sub‑gradient → PGD)
    if best_seq is not None and time.time() < deadline - 15:
        print("Phase 2 – Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            lr_start=0.06,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 2] Adam improved to {val_adam:.6f}")

        if time.time() < deadline - 10:
            print("Phase 2 – Sub‑gradient optimisation")
            a_sub, val_sub = _subgradient_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.9,
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Phase 2] Sub‑grad improved to {val_sub:.6f}")

        if time.time() < deadline - 5:
            print("Phase 2 – Projected‑gradient optimisation")
            a_pgd, val_pgd = _pgd_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=150_000,
                eta0=0.4,
            )
            if val_pgd < best_val:
                best_val, best_seq = val_pgd, a_pgd.copy()
                print(f"[Phase 2] PGD improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing for the final seconds
    if best_seq is not None and time.time() < deadline - 2:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Optional light smoothing (try a few smoothing radii)
    if best_seq is not None and time.time() < deadline - 1:
        print("Phase 4 – optional smoothing")
        smoothers = [3, 5, 7]  # odd kernel half‑widths
        for w in smoothers:
            kernel = np.concatenate(
                ([0.25], np.full(w - 1, 0.5 / (w - 1)), [0.25])
            )
            smoothed = np.convolve(best_seq, kernel, mode='same')
            smoothed = np.maximum(smoothed, 0.0)
            smoothed /= smoothed.sum()
            val = _objective(smoothed, conv_fft_len)
            if val < best_val:
                best_val, best_seq = val, smoothed.copy()
                print(f"[Smoothing width={w}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely) – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence's limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
