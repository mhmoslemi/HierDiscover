def _annealed_gd(a0, n, conv_fft_len, corr_fft_len, max_iters, beta_start, beta_end, lr_start, lr_end, mu, deadline):
    a = np.maximum(a0, 0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    a = a / a.sum()
    v = np.zeros_like(a)
    it = 0
    best_a = a.copy()
    best_val = float('inf')
    # Precompute beta schedule linear in log?
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        # compute beta and lr based on iteration ratio
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t
        conv = _full_convolution(a, conv_fft_len)
        # compute weights via softmax
        # to avoid overflow, subtract max
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        v = mu * v - lr * grad
        a = a + v
        # Projection onto simplex
        a = _project_onto_simplex(a)
        # evaluate objective occasionally
        if it % 100 == 0:
            # compute max conv (real)
            conv_val = conv.max()  # approximate
            val = 2.0 * n * conv_val / (a.sum() ** 2)
            if val < best_val:
                best_val = val
                best_a = a.copy()
    return best_a, best_val
```

"""
Optimise a non‑negative step function for the objective

    F(a) = 2·n·max(a*a) / (sum(a)^2)

The algorithm works in three phases, fitting into the 1000 s budget:

1. **Broad initial pool** – deterministic beta‑pdf shapes, a few Dirichlet
   draws (very spiky) and optional warm‑start are normalised to an L1‑sum
   of 1 and evaluated with a cheap FFT‑based convolution.  The best
   `top_k` candidates are kept.

2. **Annealed projected gradient descent** – instead of the hard
   sub‑gradient of the max‑convolution we use a smooth “soft‑max”
   surrogate.  The temperature β is increased from a small value to a
   large one while the learning rate decays.  After each gradient step the
   vector is projected onto the probability simplex by simple
   non‑negative clipping followed by renormalisation (much cheaper than an
   exact Euclidean projection).  Momentum (μ) accelerates convergence.
   Each elite candidate receives a time slice; the best result over all
   runs is kept.

3. **Cheap hill‑climber** – a random mass‑transfer local search (up to a
   few hundred thousand moves) finishes the remaining time budget,
   keeping any improvement.

All heavy lifting uses NumPy FFTs (O(n log n) per iteration) and runs
comfortably within the time limit.  The code works for any length; we
use `n = 8192` to have enough degrees of freedom to approach the
theoretical lower bound of 1.0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple projection onto the simplex (non‑negative + L1 = 1) ----------
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    # Fast O(n) method: clip negatives, renormalise; exactness is not needed
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s > 0:
        return v / s
    else:
        # fallback to uniform distribution
        n = v.shape[0]
        return np.full_like(v, 1.0 / n)


# ----------------------------------------------------------------------
# Parametric families ----------------------------------------------------

def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta(a,b) pdf evaluated on the centre of n equal sub‑intervals."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw from Dirichlet(alpha, …, alpha)."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Annealed gradient descent ---------------------------------------------

def _annealed_gd(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    max_iters: int,
    beta_start: float,
    beta_end: float,
    lr_start: float,
    lr_end: float,
    mu: float,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the *soft‑max* surrogate of the max‑convolution.
    Returns the best vector seen and its true objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    a = a / a.sum()

    v = np.zeros_like(a)               # momentum buffer
    best_a = a.copy()
    best_val = float("inf")

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        # schedules (geometric)
        t = it / max_iters
        beta = beta_start * (beta_end / beta_start) ** t
        lr = lr_start * (lr_end / lr_start) ** t

        # convolution of the current iterate
        conv = _full_convolution(a, conv_fft_len)

        # soft‑max weights (stable version)
        max_c = conv.max()
        exp_vals = np.exp(beta * (conv - max_c))
        w = exp_vals / exp_vals.sum()

        # gradient of the smooth objective
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # momentum update + projection onto simplex (cheap version)
        v = mu * v - lr * grad
        a = a + v
        a = _project_onto_simplex(a)

        # occasional exact evaluation
        if it % 200 == 0:
            max_c_exact = conv.max()
            val = 2.0 * n * max_c_exact / (a.sum() ** 2)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[GD it={it:06d}] new best {val:.6f}")

    # final exact evaluation (in case the loop ends without a recent check)
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    if val < best_val:
        best_val = val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
# Very cheap hill‑climber ----------------------------------------------

def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Transfer a random fraction of mass from one index to another.
    Keep the move if it improves the *exact* objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    rng = np.random.default_rng()

    for it in range(max_moves):
        if time.time() >= deadline:
            break
        i = rng.integers(n)
        if a[i] < 1e-14:
            continue
        j = rng.integers(n - 1)
        if j >= i:
            j += 1            # ensure i != j
        # move up to 30 % of the source mass
        delta = a[i] * (rng.random() * 0.3)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

    return best_a, best_val

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
