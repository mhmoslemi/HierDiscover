"""
Propose a low‑value step‑function for the given evaluation.

Strategy (time‑aware, ~1000 s total)

1️⃣ Warm‑starts – a rich pool of deterministic families (uniform, sinusoid,
   edge‑exponential, power‑law, beta, double‑Gaussian) together with many
   Dirichlet draws.  The best element is kept as seed.

2️⃣ Smooth‑max Adam – minimise a log‑sum‑exp surrogate of the max‑convolution.
   The logits are optimised on the simplex (soft‑max) and the temperature
   (inverse smoothness) is slowly increased, ending with an almost exact
   gradient of the true max.

3️⃣ Projected‑gradient descent (PGD) – Euclidean projected gradient descent
   on the exact max‑convolution using the active set (indices attaining the
   current maximum).  Step‑size decays as 1/√t.

4️⃣ Simulated‑annealing (SA) – a stochastic mass‑transfer walk that can accept
   temporary deteriorations.  Temperature decays linearly; large transfers
   (up to 70 % of the source mass) allow the search to escape plateaus.

5️⃣ Cheap hill‑climber – very local random moves in the last few seconds to
   squeeze any remaining improvement.

All steps keep the iterate on the probability simplex (non‑negative,
unit‑sum) and the best vector encountered is finally clipped to
[0, 1000] (the limits of ``evaluate_sequence``).  The routine always returns
the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:                     # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Objective (exact, same as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Exact value of 2·n·max(conv) / (sum(a)²).  a is assumed non‑negative."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Candidate generators (deterministic families + Dirichlet)
# ----------------------------------------------------------------------


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α (scalar)."""
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential mass concentrated at the two edges."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law shape: a_i ∝ (i+1)^α + (n‑i)^α."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑distribution pdf on the discrete grid (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1.0)) * ((1.0 - x) ** (b - 1.0))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0.0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    """Positive half‑wave sinusoid."""
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a diverse pool of start vectors (deterministic families)."""
    pool = []

    # Dirichlet draws with various concentrations
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    # Edge‑exponential
    for gamma in (0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0):
        pool.append(_edge_exponential(n, gamma))

    # Edge‑power‑law (negative exponents → strong edge emphasis)
    for alpha in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    # Beta‑pdf family
    for a, b in [(0.5, 0.5), (0.5, 2.0), (2.0, 0.5),
                 (2.0, 2.0), (5.0, 1.0), (1.0, 5.0)]:
        pool.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # Simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    return pool

#  Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate (log‑sum‑exp).  Returns best vector
    (sum==1) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # inverse temperature (soft‑max sharpness)
    lam_factor = 1.02   # increase temperature slowly

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- soft‑max (mirror‑descent) to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (chain rule through the soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on the logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (smooth‑max becomes sharper)
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the very end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent on the exact max‑convolution.
    Returns best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = _random_dirichlet(n, 1.0, np.random.default_rng())
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size  η = η₀ / √(t+1)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        # occasional jitter near the end to escape tiny plateaus
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _simulated_annealing_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing walk with mass‑transfer moves.
    The temperature decays linearly from 1 to ≈0.
    Returns the best vector seen.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    cur_val = best_val

    start = time.time()
    it = 0
    # parameters for the proposal
    max_frac = 0.7          # move up to 70 % of the source mass
    while time.time() < deadline and it < max_moves:
        # linear temperature schedule
        elapsed = time.time() - start
        total = deadline - start
        T = max(1e-9, (total - elapsed) / total)   # 1 → 0

        # pick source and destination indices
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # propose a transfer
        delta = a[i] * np.random.rand() * max_frac
        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new)

        # Metropolis acceptance
        if val < cur_val or np.random.rand() < np.exp(-(val - cur_val) / T):
            a = a_new
            cur_val = val
            if val < best_val:
                best_val = val
                best_a = a_new.copy()
                if it % 5000 == 0:
                    print(f"[SA it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Cheap hill‑climber: move a random fraction of mass from one index
    to another and keep the move if the objective improves.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation metric.  The routine respects the 1000 s budget and always
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem size (chosen large enough to expose the asymptotic regime)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)   # 8192
    corr_fft_len = _next_pow2(3 * n - 2)   # 16384

    # optional warm start supplied via a global variable
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    #  1️⃣ Warm‑start pool
    # ------------------------------------------------------------------
    pool = []  # list of ndarrays
    # add a few cyclic shifts of a supplied warm start (if any)
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    pool += _initial_pool(n, rng)

    # extra Dirichlet draws with a finer grid of α values
    for alpha in (0.05, 0.15, 0.4, 0.8, 1.5, 3.0, 7.0):
        for _ in range(8):
            pool.append(_random_dirichlet(n, alpha, rng))

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # quick scan of the pool (stop early if we run out of time)
    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = _random_dirichlet(n, 1.0, rng)
        a = a / a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {cid}] best {val:.6f}")

    # If no candidate survived (unlikely) fall back to uniform
    if best_seq is None:
        best_seq = _uniform(n)

    # ------------------------------------------------------------------
    #  2️⃣ Adam refinement (≈20 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.20 * remaining
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refinement] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    #  3️⃣ Long projected‑gradient descent (≈50 % of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        pgd_deadline = deadline - 10   # keep 10 s for later phases
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=800_000,
            eta0=0.8,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD refinement] new best {val_pgd:.6f}")

    # ------------------------------------------------------------------
    #  4️⃣ Simulated‑annealing (use whatever time is left, reserve ~5 s)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        sa_deadline = deadline - 5
        a_sa, val_sa = _simulated_annealing_optimize(
            best_seq,
            n,
            conv_fft_len,
            sa_deadline,
            max_moves=300_000,
        )
        if val_sa < best_val:
            best_val = val_sa
            best_seq = a_sa.copy()
            print(f"[SA refinement] new best {val_sa:.6f}")

    # ------------------------------------------------------------------
    #  5️⃣ Final cheap hill‑climber (last few seconds)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local moves] improved best {val_loc:.6f}")

    # safety clipping (evaluate_sequence caps values to [0,1000])
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
