def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

def _gd_optimize_momentum(a0,...):

    a = np.maximum(a0,0)

    # Normalize

    sum_a = a.sum()

    if sum_a < 1e-12: a = np.random.dirichlet(np.ones(n))

    a = a / a.sum()

    best_val = inf, best_a = a

    v = np.zeros_like(a)

    mu = 0.9   # momentum factor

    init_step = 0.5

    # while time and iteration limit:

    conv = _full_convolution(a, conv_fft_len)

    max_c = conv.max()

    val = 2.0 * n * max_c

    # set best

    while time.time()<deadline and it < max_iters:

        # recompute conv at each iteration

        conv = _full_convolution(a, conv_fft_len)

        max_c = conv.max()

        val = 2.0 * n * max_c

        if val < best_val:

            best_val = val

            best_a = a.copy()

            # optionally print

        # weight

        active = np.isclose(conv, max_c, atol=1e-14)

        weight = active.astype(float)

        wsum = weight.sum()

        if wsum > 0:

            weight /= wsum

        else:

            weight = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # update momentum

        v = mu * v + (1.0 - mu) * grad

        # line search

        step = init_step

        while step > 1e-12:

            a_new = a - step * v

            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)

            max_c_new = conv_new.max()

            val_new = 2.0 * n * max_c_new

            if not np.isnan(val_new) and val_new <= val:

                a = a_new

                break

            step *= 0.5

        it += 1

    return best_a, best_val

"""
Optimize a non‑negative step function (probability vector) to minimise

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

by a three‑stage search:

1. **Diverse seed pool** – many deterministic families (uniform,
   sinusoid, edge‑exponential, edge‑power‑law, central‑Gaussian,
   product‑power‑law, beta, double‑Gaussian) plus Dirichlet draws and
   cyclic shifts of an optional warm‑start.

2. **Accelerated projected gradient descent** – a momentum‑augmented
   sub‑gradient method (active‑set sub‑gradient of the max‑convolution)
   with back‑tracking line‑search and Euclidean projection onto the
   probability simplex.  The best of the elite seeds is refined within a
   time‑budget that favours higher‑rank candidates.

3. **Cheap hill‑climber** – random mass‑transfer moves that keep any
   improvement.  This final polishing extracts any remaining gain.

All computation uses FFT‑based convolution for speed.  The routine obeys
the 1000 s wall‑clock limit and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Projection onto the probability simplex (Euclidean)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Projection of v onto {x≥0, sum(x)=1}."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Deterministic families (all produce vectors with sum == 1)
def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    x = np.maximum(x, 0.0)
    return x / x.sum()


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    return shape / shape.sum()


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0:
        return _uniform(n)
    return shape / s


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0:
        return _uniform(n)
    return shape / s


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    return shape / shape.sum()


def _central_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    center = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - center) / (sigma * n)) ** 2)
    return shape / shape.sum()


def _product_power_law(n: int, beta: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = (i ** beta) * ((n - i + 1) ** beta)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0:
        return _uniform(n)
    return shape / s


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """Return warm start and a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


# ----------------------------------------------------------------------
# Gradient descent with momentum (sub‑gradient of max‑convolution)
def _gd_optimize_momentum(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent with Nesterov‑style momentum.
    Returns the best vector seen and its objective value.
    """
    # initialise on the simplex
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    # compute first objective
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    best_val = 2.0 * n * max_c

    v = np.zeros_like(a)            # momentum term
    mu = 0.9                        # momentum coefficient
    init_step = 0.5                 # base step size
    it = 0

    while time.time() < deadline and it < max_iters:
        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        weight = active.astype(float)
        wsum = weight.sum()
        if wsum > 0:
            weight /= wsum
        else:
            weight = np.zeros_like(conv)

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(weight, a, corr_fft_len)

        # Nesterov‑style momentum update
        v = mu * v + (1.0 - mu) * grad

        # back‑tracking line‑search (projected)
        step = init_step
        while step > 1e-12:
            a_new = a - step * v
            a_new = _project_onto_simplex(a_new)

            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new

            if not np.isnan(val_new) and val_new <= 2.0 * n * max_c:
                # accept the step
                a = a_new
                conv = conv_new
                max_c = max_c_new
                break
            step *= 0.5

        # record best
        if max_c * 2.0 * n < best_val:
            best_val = 2.0 * n * max_c
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[GD it={it:06d}] new best {best_val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple hill‑climber (random mass transfers)
def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climber that transfers a random fraction of mass
    from one position to another and keeps the move if it improves the exact objective.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        delta = a[i] * (np.random.rand() * 0.3)   # up to 30% of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local it={it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Main entry point.
    Returns a probability vector (list of non‑negative floats) that
    approximately minimises the evaluation function within the given time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – we keep the classic 4096 length
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a rich initial pool (deterministic families + random draws)
    pool = []
    pool.extend(_warm_and_shifts(warm, n))

    # deterministic families (coarse)
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))

    # edge exponential (log‑spaced)
    for gamma in np.logspace(-2, 1.5, 18):
        pool.append(_edge_exponential(n, float(gamma)))

    # edge power‑law (linear)
    for alpha in np.linspace(-4.0, 2.0, 18):
        pool.append(_edge_power_law(n, float(alpha)))

    # beta pdf (grid)
    a_vals = np.logspace(-1, 1, 9)
    b_vals = np.logspace(-1, 1, 9)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    # double Gaussian (linear sigma)
    for sigma in np.linspace(0.02, 0.5, 12):
        pool.append(_double_gaussian(n, float(sigma)))

    # central Gaussian (linear sigma)
    for sigma in np.linspace(0.02, 0.5, 12):
        pool.append(_central_gaussian(n, float(sigma)))

    # product power‑law (linear beta)
    for beta in np.linspace(-3.0, 3.0, 12):
        pool.append(_product_power_law(n, float(beta)))

    # random Dirichlet draws (various concentrations)
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(_random_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    # Quick evaluation of the pool, retaining the elite
    top_k = 28
    scored = []
    for cand in pool:
        if time.time() >= deadline * 0.3:   # reserve time for refinement
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        scored.append((val, a))
    scored.sort(key=lambda x: x[0])
    top_candidates = [a for _, a in scored[:top_k]]
    print(f"Selected {len(top_candidates)} elite seeds (best val {scored[0][0]:.6f})")

    # ------------------------------------------------------------------
    # Phase 2 – refined gradient descent on elite candidates
    best_seq = None
    best_val = float("inf")
    for rank, a0 in enumerate(top_candidates):
        if time.time() >= deadline - 8:
            break
        remaining = deadline - time.time()
        # allocate larger slices to higher‑rank candidates
        share = remaining * (top_k - rank) / max(1, (top_k - rank) * (top_k - rank + 1) / 2)
        local_deadline = time.time() + share

        a_opt, val_opt = _gd_optimize_momentum(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            local_deadline,
            max_iters=200_000,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine #{rank}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – final cheap hill‑climber on the current best
    if best_seq is not None and time.time() < deadline - 2:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=500_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _uniform(n)

    # Ensure feasibility (non‑negative, sum≈1, values ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Re‑normalise to avoid any drift due to clipping
    total = best_seq.sum()
    if total < 1e-12:
        best_seq = _uniform(n)
    else:
        best_seq = best_seq / total

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
