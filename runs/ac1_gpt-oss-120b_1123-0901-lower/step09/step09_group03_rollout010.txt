def _adam_combined_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=100_000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1=0.9; beta2=0.999; eps_adam=1e-8
    lr = 0.04
    best_a = a0.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()
        # compute conv & max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[AdamComb it={it}] new best {val:.6f}")
        # compute w for max (smooth)
        lam = 0.5 + it * 0.001  # gradually increase
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to active set
            active = (conv == max_c)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
        else:
            w /= w_sum
        # gradients
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # flattening gradient
        avg_c = 1.0 / (2 * n - 1)
        grad_flat = 2.0 * _cross_correlation(conv - avg_c, a, corr_fft_len)
        # combine
        alpha = max(0.2, 1.0 - it / max_iters)  # decrease from 1 to 0.2
        grad = alpha * grad_max + (1 - alpha) * grad_flat
        # mirror descent on simplex through logits
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)
        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step
        # learning rate decay maybe
        if it % 150_000 == 0 and it > 0:
            lr *= 0.7
        it += 1
    return best_a, best_val

"""
Hybrid optimizer with a flattening term.

The method works on the simplex (∑a=1, a_i≥0) using a log‑softmax
parameterisation a_i = exp(logits_i)/∑exp(logits).  Two gradient
components are mixed:

* a smooth‑max (soft‑max) approximation of the worst convolution term,
  providing a sub‑gradient of the true objective;
* an L2‑flattening term ½∑_k ( (a*a)_k – 1/(2n−1) )² whose gradient
  pushes the autocorrelation towards a constant shape (the theoretical
  lower bound).

The weight α of the max‑term is linearly annealed from 1 down to 0.2,
so the algorithm first attacks the peak and then refines the whole
profile.  Adam on the logits supplies an adaptive step‑size.
After a long Adam phase we perform a classic exponentiated‑gradient
sub‑gradient refinement and a pair‑wise targeted simulated‑annealing
move set.  If time permits the solution is up‑sampled to 8192 points
and refined again.

The routine respects the supplied time budget and always returns the
best sequence found (as a plain Python list).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation needed for gradients.
    Returns ∂/∂a_i Σ_k w_k (a∗a)_k  for i=0…n‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_combined_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a blend of the smooth‑max surrogate and an L2‑flattening term.
    Returns the best vector (always normalised to sum 1) and its exact
    objective value  F = 2·n·max(a∗a) .
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # interior of simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04                               # base learning rate

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    avg_target = 1.0 / (2 * n - 1)         # theoretical flat convolution level

    while time.time() < deadline and it < max_iters:
        # ---------- simplex point ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()        # a lives on the simplex

        # ---------- exact objective ----------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[AdamComb it={it:06d}] new best {val:.6f}")

        # ---------- smooth max weight ----------
        # temperature (λ) slowly grows, but stays far from overflow
        lam = 0.5 + it * 0.001                # ≈ up to ~200 after 200k iters
        w = np.exp(lam * (conv - max_c))     # max term gets weight 1
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fallback to exact sub‑gradient (uniform over active indices)
            active = np.isclose(conv, max_c, atol=1e-14)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
        else:
            w /= w_sum

        # ---------- gradients ----------
        # sub‑gradient of the max term
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient of the L2 flattening term ½∑(conv‑avg)²
        grad_flat = 2.0 * _cross_correlation(conv - avg_target, a, corr_fft_len)

        # anneal the relative weight of the two components
        alpha = max(0.2, 1.0 - it / max_iters)   # from 1 → 0.2
        grad = alpha * grad_max + (1.0 - alpha) * grad_flat

        # ---------- mirror‑descent (Adam) on logits ----------
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)            # ∇_θ log‑softmax

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.7

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true
    max‑convolution objective.  Keeps the iterate on the simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set for the current maximum
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)   # multiplicative EG step

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Targeted annealing that repeatedly attacks the pair (i,j) that
    contributes most to the current maximal convolution entry.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # indices i that can pair with j = kmax‑i
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # single‑index case – move a small fraction elsewhere
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # pair case – remove a little from both ends and redistribute
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1, t2 = np.random.randint(n), np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing moves that transfer a random fraction of mass
    from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified set of starting vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentration parameters
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential tails (symmetrised)
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law “edge” families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function that minimises the objective."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # moderate problem size – enough freedom but still fast per iteration
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 1e-8:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a pool of diverse seeds
    pool = _initial_candidates(n, warm)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick sanity runs (≈5 % of the budget)
    quick_dead = time.time() + (deadline - time.time()) * 0.05
    for a0 in pool:
        if time.time() >= quick_dead:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-8:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()
        # a very short Adam run just to get a sense of the landscape
        sub_dead = min(quick_dead, time.time() + 1.5)
        a_tmp, val_tmp = _adam_combined_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=12_000
        )
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Explore] improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # Fallback if nothing survived
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    # ------------------------------------------------------------------
    # Phase II – long combined‑Adam optimisation (≈50 % of the budget)

            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=300_000,
        )

            print(f"[Main] Adam improved to {val_long:.6f}")

    # ------------------------------------------------------------------
    # Phase III – exact‑max sub‑gradient refinement (≈30 % of the budget)

        a_sub, val_sub = _subgradient_optimize_expgrad(

            corr_fft_len,
            deadline,

        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Refine] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – targeted pair‑wise annealing (≈10 % of the budget)
    if time.time() < deadline - 8:
        print("[Anneal] Pair‑targeted")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,

        )
        if val_pair < best_val:

            print(f"[Anneal] Pair‑targeted improved to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Phase V – light random annealing (use any remaining time)

        print("[Anneal] Final random")

            best_seq,

            deadline,

        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short combined‑Adam on the larger grid

        a_big_opt, val_big_opt = _adam_combined_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            big_dead,
            max_iters=180_000,
        )

            print(f"[Upsample] improved to {val_big_opt:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a legal return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
