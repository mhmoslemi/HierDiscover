import numpy as np, scipy.sparse as sp, scipy.optimize as opt.

def build_conv_matrix(f):
    n = len(f)
    rows = 2*n - 1
    data = []
    offsets = []
    for d in range(n):
        offsets.append(d)
        data.append(np.full(n, f[d]))
    A = sp.diags(data, offsets, shape=(rows, n), format='csc')
    return A

"""
Propose a low‑score step‑function.

The algorithm consists of four phases:

1. **Exploratory sweep** – generate a large collection of
   analytically‑parameterised shapes (power‑law, exponential,
   stretched‑exponential, Beta, double‑Gaussian, monotone,
   random spikes).  Each shape is normalised to a probability
   vector (sum = 1) and evaluated with the exact objective; the best
   candidate is kept.

2. **Linear‑programming refinement** – starting from the best vector
   `a`, solve the LP  

        maximise   ∑ g_i
        subject to A(a)·g ≤ M ,   g ≥ 0

   where `M = max( a * a )` and `A(a)` is the convolution matrix
   (row k contains the coefficients `a_i` with offset `k`).  
   The solution `g` is a feasible distribution that pushes mass
   into “slack’’ positions of the autocorrelation.  A short line‑search
   mixes `a` with the normalised solution `ĝ` and keeps the mixture
   that gives the greatest reduction of the objective.  A few LP
   iterations dramatically lower the peak autocorrelation.

3. **Smooth‑max Adam** – a few seconds of Adam optimisation on a
   log‑sum‑exp surrogate of the max‑convolution further smoothes the
   shape while staying on the simplex.

4. **Exact‑max sub‑gradient (exponentiated‑gradient)** – the remaining
   time is spent on a mirror‑descent sub‑gradient method that works
   directly on the original non‑smooth objective.

All phases respect the overall 1000 s budget, always return a
non‑negative list and print the intermediate scores.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Fast convolution utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (needed for sub‑gradient)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping)
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a)/(sum a)² (sum a>0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Adam optimisation on smooth‑max surrogate
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

# Exponentiated‑gradient sub‑gradient optimisation (exact max)
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

# Simulated‑annealing fine‑tuning (optional)
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1
    return best_a, best_val

def _lp_refine(

                     g ≥ 0,
    where A(a) is the convolution matrix of `a` and M = max(a*a).
    The solution g is mixed with the current distribution by a short line
    search on the simplex.
    """
    try:
        import scipy.sparse as sp
        import scipy.optimize as opt
    except Exception as exc:      # SciPy not available – skip LP phase
        print("SciPy missing, skipping LP refinement:", exc)
        return a0, _objective(a0, n)

    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    for it in range(max_iter):
        if time.time() > deadline - 1.0:
            break

        # current bound M
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # Build convolution matrix A where each diagonal offset d contains a[d]
        data = [np.full(n, a[d]) for d in range(n)]
        offsets = np.arange(n)
        A = sp.diags(data, offsets, shape=(2 * n - 1, n), format="csc")
        b = np.full(2 * n - 1, M)

        # Linear program: maximise sum(g)  <=> minimise -sum(g)
        c = -np.ones(n)
        bounds = [(0, None)] * n

        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print(f"[LP it={it}] LP failed: {res.message}")
            break

        g = res.x
        if g.sum() <= 0:
            break
        g_norm = g / g.sum()

        # ----- line‑search mixing -----
        improved = False
        # try a handful of mixing ratios
        for tau in (0.05, 0.1, 0.2, 0.3, 0.5, 0.7, 0.9):
            a_new = (1.0 - tau) * a + tau * g_norm          # stays normalised
            val_new = _objective(a_new, n)
            if val_new < best_val - 1e-9:                    # strict improvement
                best_val = val_new
                best_a = a_new.copy()
                a = a_new.copy()
                improved = True
                print(f"[LP it={it}] new best {val_new:.6f} (tau={tau})")
                break
        if not improved:
            # no mixing improves – stop early
            break

    return best_a, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine follows the four phases described
    in the module docstring and always returns a valid sequence before
    the time budget expires.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # Problem dimension – 4096 gives a good trade‑off between quality
    # and computational cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep
    best_seq = None
    best_val = float("inf")

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-4, 0.8, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential
    alphas = np.logspace(-3, 0.5, 15)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta distributions
    for a_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
        for b_par in [0.2, 0.5, 1.0, 2.0, 5.0]:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-2.5, -0.2, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-4, 1.0, 20):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # monotone left‑power
    for p in np.linspace(-3.0, 1.0, 25):
        _maybe_update(_candidate_left_power(p, n))

    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(5):
            _maybe_update(_candidate_spike(k, n, random=True))

    # warm start from previously found sequence, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement (new phase)
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑LP] starting LP refinement")
        a_lp, val_lp = _lp_refine(
            np.asarray(best_seq, dtype=np.float64),
            n,
            conv_fft_len,
            deadline,
            max_iter=12,                 # a dozen LP solves are cheap
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[Phase‑LP] improved to {val_lp:.6f}")

    # ------------------------------------------------------------------
    # 3) Short Adam optimisation (smooth‑max)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑Adam] short Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 12.0)  # reserve time for later phases
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Long sub‑gradient descent (exact max)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning (if time permits)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (non‑negative, within clipping limits)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score – evaluator unavailable.")

    return best_seq.tolist()
