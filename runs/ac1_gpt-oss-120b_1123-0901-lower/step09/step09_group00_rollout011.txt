    Uses a fast random sampling phase, followed by a long Adam
    optimisation of the best candidate and a final exact projected
    gradient descent with backtracking line‑search and a short
    simulated‑annealing refinement.

    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

            warm = warm / warm.sum()

"""Propose a low‑value step‑function.

The algorithm works in three stages:

1. **Diverse sampling** – many random Dirichlet / spike / piece‑wise
   candidates are created, projected to the simplex and evaluated
   with a fast FFT‑based max‑convolution.  The best few (default 5)
   are retained.

2. **Long smooth‑max optimisation** – the current best candidate is
   refined with an Adam optimiser that minimises a log‑sum‑exp
   (smooth‑max) surrogate of the max‑convolution.  The optimiser runs
   until most of the time budget is exhausted.

3. **Exact projected gradient descent + annealing** – a classic
   projected gradient descent on the exact max‑convolution (with a
   cheap back‑tracking line‑search) is applied, followed by a short
   simulated‑annealing local search to escape possible tiny local
   minima.

All vectors are kept on the probability simplex (Σa=1, a≥0), because
the evaluation function is scale‑invariant.  The routine respects the
budget (≈1000 s) and always returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of the evaluation function for a simplex‑normalised a."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp) of the max‑conv."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # obtain current a from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (chain rule through soft‑max)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _max_conv_descent_line_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    init_step: float = 0.3,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Uses a simple back‑tracking line‑search to choose a step size that
    strictly decreases the current max value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    step = init_step

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if not np.any(active):
            # No active set – should not happen, but break to avoid waste.
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # line‑search – try to reduce the max value
        step_loc = step
        for _ in range(8):
            a_new = a - step_loc * grad
            a_new = _project_simplex(a_new)
            conv_new = _full_convolution(a_new, conv_fft_len)
            max_c_new = conv_new.max()
            val_new = 2.0 * n * max_c_new
            if val_new <= val * (1 - 1e-9):
                break
            step_loc *= 0.5

        if step_loc < 1e-12:
            # If line‑search fails, perform a tiny random perturbation
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-14:
                delta = min(a[i] * 0.02, step * 0.001)
                a[i] -= delta
                a[j] += delta
                a = _project_simplex(a)
            continue

        # accept the move
        a = a_new
        step = step_loc * 1.02          # modest increase after a successful step

        # slowly decay the step if we keep succeeding
        if it % 2000 == 0 and step > init_step * 0.1:
            step = max(step * 0.9, init_step * 0.1)

    return best_a, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Moves a random fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-5

    for it in range(max_moves):
        if time.time() >= deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_simplex(a2)

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:

        elif r < 0.90:                   # edge‑exponential families
