"""
Propose a low‑value step‑function for the evaluation metric

    F(a) = 2·n·max(a∗a) / (sum(a)^2)

The strategy is a *basis‑mix* approach:

1.  Build a diverse library of normalized “basis” shapes
    (edge‑exponential, edge‑power‑law, β‑pdf, double‑Gaussian,
    sinusoid, raised‑cosine, semicircle, triangular, Kaiser windows,
    and a possible warm‑start).

2.  Each basis alone is evaluated (fast FFT‑based surrogate) and the
    best few are retained.

3.  The final sequence is a convex combination of all bases.
    The combination weights live on the probability simplex; they are
    optimised with a three‑phase scheme:

       • Adam on a smooth‑max surrogate (log‑sum‑exp) with a slowly
         increasing temperature.
       • Continuation L‑BFGS on the same surrogate for several
         temperature levels.
       • A cheap random hill‑climber that moves a small amount of mass
         between two basis components.

   All steps respect the overall 1000 s budget and the best exact
   sequence found is returned.

The code uses only NumPy and SciPy, works for any length `n` (here
fixed to 4096) and always returns a plain Python list.
"""

import time
import numpy as np
from scipy.optimize import minimize
from typing import List

# ----------------------------------------------------------------------
# FFT‑based utilities (fast convolution / gradient)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    ``w`` has length 2·len(a)‑1, ``a`` length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


# ----------------------------------------------------------------------
# Basis generators (all return a *non‑negative* vector that sums to 1)

def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    # alpha is (negative) exponent; shape ∝ (i+1)^alpha + (n-i)^alpha
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = (i ** alpha) + ((n - i + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _triangular(n: int) -> np.ndarray:
    """Symmetric triangular (peak in the centre)."""
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    shape = np.kaiser(n, beta)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
# Fast surrogate objective (sum(a) = 1 → value = 2·n·max_conv)

def _fast_obj(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a∗a) assuming sum(a)=1 (no division needed)."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Adam optimisation on the weight logits (convex combination of bases)

def _adam_optimize_weights(
    logits0: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lr_start: float = 0.05,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate, optimizing the basis mixing weights."""
    eps = 1e-15
    logits = logits0.copy()
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr = lr_start
    lam = 0.02           # inverse temperature (smooth‑max sharpness)
    lam_factor = 1.015

    best_a = None
    best_val = float("inf")
    it = 0

    while time.time() < deadline:
        # ---- simplex projection via soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        w = exp_log / exp_log.sum()           # weight vector on simplex
        a = B @ w                              # current step‑function (sum=1)

        # ---- exact objective for bookkeeping ----
        val = _fast_obj(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max (log‑sum‑exp) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w_smooth = np.exp(lam * (conv - max_c))
        w_sum = w_smooth.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w_smooth = np.full_like(w_smooth, 1.0 / w_smooth.size)
        else:
            w_smooth /= w_sum

        # ---- gradient wrt a (exact for the smooth‑max) ----
        grad_a = 2.0 * _cross_correlation(w_smooth, a, corr_fft_len)

        # ---- chain rule through the linear combination ----
        grad_w = B.T @ grad_a                 # dF/dw (w are basis coefficients)

        # ---- chain rule through the soft‑max (logits → w) ----
        avg = np.dot(w, grad_w)
        grad_log = w * (grad_w - avg)

        # ---- Adam update ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # ---- schedule ----
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(logits.size) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Continuation L‑BFGS on the weight logits (same surrogate)

def _l_bfgs_obj_grad_weights(
    x: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    alpha: float,
) -> tuple[float, np.ndarray]:
    """Smooth‑max objective + gradient for a given temperature α."""
    mx = x.max()
    exp_x = np.exp(x - mx)
    w = exp_x / exp_x.sum()
    a = B @ w

    # convolution
    fa = np.fft.rfft(a, conv_fft_len)
    conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]

    max_c = conv.max()
    v = conv - max_c                     # ≤ 0
    exp_alpha_v = np.exp(alpha * v)      # safe, max = 1
    sum_exp = exp_alpha_v.sum()
    smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

    obj = 2.0 * n * smooth_max

    # gradient wrt a (exact for the smooth‑max)
    w_smooth = exp_alpha_v / sum_exp
    grad_a = 2.0 * _cross_correlation(w_smooth, a, corr_fft_len)

    # back‑prop to weight logits
    grad_w = B.T @ grad_a
    avg = np.dot(w, grad_w)
    grad_x = w * (grad_w - avg)

    return float(obj), grad_x


def _continuation_l_bfgs_weights(
    logits0: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """Run L‑BFGS on a schedule of increasing temperatures."""
    logits = logits0.copy()
    best_logits = logits.copy()
    best_val = float("inf")

    # temperature schedule (smooth ⇒ hard)
    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, B=B, conv_fft_len=conv_fft_len,
                corr_fft_len=corr_fft_len, alpha=alpha):
            return _l_bfgs_obj_grad_weights(x, B, n,
                                            conv_fft_len, corr_fft_len, alpha)

        remaining = deadline - time.time()
        # give a modest budget per stage (expensive steps)
        max_iter = 200 if remaining > 30 else 80

        res = minimize(
            fun,
            logits,
            method="L-BFGS-B",
            jac=True,
            options={"maxiter": max_iter, "disp": False},
        )
        logits = res.x

        # exact evaluation
        w = np.exp(logits - np.max(logits))
        w /= w.sum()
        a = B @ w
        exact_val = _fast_obj(a, n, conv_fft_len)   # fast surrogate (exact for sum=1)

        # also ask the official evaluator for the final contestants
        # (cheap now, we will call the true evaluator on the very best later)
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best (surrogate) {exact_val:.6f}")

    # final best on the simplex
    w = np.exp(best_logits - np.max(best_logits))
    w /= w.sum()
    a_best = B @ w
    return a_best, best_val


# ----------------------------------------------------------------------
# Cheap random hill‑climber on the weight simplex

def _local_random_search_weights(
    w0: np.ndarray,
    B: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Random mass moves between two basis components."""
    w = w0.copy()
    a = B @ w
    best_a = a.copy()
    best_val = _fast_obj(a, n, conv_fft_len)

    it = 0
    M = w.shape[0]

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(M)
        if w[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(M)
        if i == j:
            it += 1
            continue

        # move a random fraction of w[i] to w[j]
        delta = w[i] * (np.random.rand() * 0.3)

        w2 = w.copy()
        w2[i] -= delta
        w2[j] += delta

        a2 = B @ w2
        val2 = _fast_obj(a2, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            w = w2
            if it % 5000 == 0:
                print(f"[Local weight move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine builds a library of basis shapes, optimises a convex
    combination of them (Adam → L‑BFGS → random hill‑climber) and always
    returns the best sequence found within the time budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Collect basis shapes -------------------------------------------------

    basis: List[np.ndarray] = []

    # warm‑start (if provided by the environment)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        s = warm.sum()
        if s > 0:
            warm = warm / s
            basis.append(warm.copy())

    # Deterministic families (wide parameter sweeps)
    # Edge exponential
    for gamma in np.logspace(-2, 1.5, 30):
        basis.append(_edge_exponential(n, gamma))

    # Edge power‑law
    for alpha in np.linspace(-9.0, -0.1, 30):
        basis.append(_edge_power_law(n, alpha))

    # Beta PDFs – coarse grid
    beta_params = [
        (0.2, 0.2), (0.2, 0.5), (0.2, 2.0), (0.5, 0.2), (0.5, 0.5),
        (0.5, 2.0), (2.0, 0.5), (2.0, 2.0), (5.0, 1.0), (1.0, 5.0),
        (5.0, 5.0), (0.1, 0.1), (0.1, 5.0), (5.0, 0.1)
    ]
    for a, b in beta_params:
        basis.append(_beta_shape(n, a, b))

    # Double Gaussian
    for sigma in (0.01, 0.02, 0.04, 0.08, 0.15):
        basis.append(_double_gaussian(n, sigma))

    # Classical windows / shapes
    basis.append(_sinusoid(n))
    basis.append(_uniform(n))
    basis.append(_raised_cosine(n))
    basis.append(_semicircle(n))
    basis.append(_triangular(n))

    # Kaiser windows
    for beta in (3.0, 5.0, 7.0, 9.0, 12.0):
        basis.append(_kaiser_window(n, beta))

    # --------------------------------------------------------------
    # Quick scan of individual bases (fast surrogate evaluation)
    best_a = None
    best_val = float("inf")
    for idx, b in enumerate(basis):
        val = _fast_obj(b, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = b.copy()
            print(f"[Base {idx:03d}] surrogate value {val:.6f}")
        if time.time() > deadline - 60:   # reserve ~1 min for optimisation phases
            break

    # --------------------------------------------------------------
    # Assemble the basis matrix (columns = bases).  All bases are normalised.
    B = np.column_stack(basis)            # shape (n, M)
    M = B.shape[1]

    # --------------------------------------------------------------
    # Phase 2 – Adam optimisation of the mixing weights
    if time.time() < deadline - 40:
        # initialise logits as uniform (softmax → 1/M each)
        init_logits = np.log(np.full(M, 1.0 / M) + 1e-15)
        a_adam, val_adam = _adam_optimize_weights(
            init_logits,
            B,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            lr_start=0.05,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_a = a_adam.copy()
            print(f"[Adam final] new best {val_adam:.6f}")

    # --------------------------------------------------------------
    # Phase 3 – continuation L‑BFGS refinement
    if time.time() < deadline - 20:
        # initialise logits from the last Adam run (or uniform if Adam skipped)
        if best_a is not None:
            # recover the implicit weights: solve B @ w = best_a in least‑squares sense
            # because bases sum to 1, a simple projection works:
            # w = B.T @ best_a  (since columns of B are already normalised)
            w_est = B.T @ best_a
            w_est = np.maximum(w_est, 0.0)
            if w_est.sum() == 0:
                w_est = np.full(M, 1.0 / M)
            else:
                w_est /= w_est.sum()
            init_logits = np.log(w_est + 1e-15)
        else:
            init_logits = np.log(np.full(M, 1.0 / M) + 1e-15)

        a_lbgfs, val_lbgfs = _continuation_l_bfgs_weights(
            init_logits,
            B,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_a = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # --------------------------------------------------------------
    # Phase 4 – cheap random hill‑climber on the weight simplex
    if time.time() < deadline - 3:
        # obtain the current weight vector (softmax of last logits)
        w_cur = np.exp(init_logits - np.max(init_logits))
        w_cur /= w_cur.sum()
        a_loc, val_loc = _local_random_search_weights(
            w_cur,
            B,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_a = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Fallback to a safe uniform sequence if nothing was found
    if best_a is None:
        best_a = _uniform(n)

    # --------------------------------------------------------------
    # Final exact evaluation (uses the provided evaluator)
    exact_val = evaluate_sequence(best_a.tolist())
    print(f"Finished – exact value {exact_val:.6f} (length {len(best_a)})")
    # Clip to the allowed range before returning
    best_a = np.clip(best_a, 0.0, 1000.0)
    return best_a.tolist()
