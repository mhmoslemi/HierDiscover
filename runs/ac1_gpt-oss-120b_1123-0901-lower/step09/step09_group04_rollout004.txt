def _pgd_momentum(a0, n, conv_len, deadline, max_iters=500_000, eta0=1.0, beta=0.9):
    a = a0.copy()
    v = np.zeros_like(a)
    best_a = a.copy(); best_val = _objective(a)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0*n*max_c/(a.sum()**2)
        if val < best_val:
            best_val = val; best_a = a.copy()
        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2 * _cross_correlation(w, a, conv_len)
        eta = eta0 / np.sqrt(it+1)
        v = beta * v - eta * grad
        a = a + v
        a = _project_onto_simplex(a)
        it += 1
    return best_a, best_val
```

def _anneal(a0, n, conv_len, deadline):
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)
    temperature = 1e-2
    decay = 0.9995
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] < 1e-12:
            continue
        j = np.random.randint(n)
        if i == j:
            continue
        frac = np.random.rand() * 0.4
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2)
        delta_val = val2 - best_val
        if delta_val < 0 or np.random.rand() < np.exp(-delta_val / temperature):
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
        temperature *= decay
    return best_a, best_val

"""
Propose a low‑value step‑function for the given evaluation.

The algorithm works in several stages:

1. **Diverse start pool** – deterministic families (uniform, sinusoid,
   edge‑exponential, power‑law, beta, double‑gaussian, triangle, block)
   together with many random Dirichlet draws.  All candidates are
   projected onto the probability simplex (sum = 1).

2. **Screening** – the pool is evaluated quickly (FFT‑based exact objective)
   and the best *k* candidates are retained.

3. **Smooth‑max Adam** – the best candidate is refined with an Adam
   optimiser that minimises a log‑sum‑exp (smooth maximum) surrogate of the
   true objective.  The temperature is increased slowly so that the
   gradient stays informative even when the maximum is sharp.

4. **Projected‑gradient descent with momentum** – the Adam result is
   further improved by Euclidean PGD on the exact maximum‑convolution
   functional.  Momentum helps to escape shallow plateaus.

5. **Simulated‑annealing hill‑climber** – a final phase performs random
   mass‑transfer moves, accepting worsening moves with a decaying temperature
   to explore the neighbourhood of the current best solution.

All FFT‑based convolutions use power‑of‑two lengths for speed.
The routine always keeps the best vector seen so far and returns it as a
plain Python list.  The returned sequence is non‑negative,
finite and satisfies the limits imposed by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT helper utilities
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _triangle(n: int) -> np.ndarray:
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        dec = inc[::-1]
    else:
        dec = np.arange(half, 0, -1, dtype=np.float64)
    shape = np.concatenate([inc, dec])
    shape /= shape.sum()
    return shape


def _block(n: int, width_frac: float = 0.5) -> np.ndarray:
    """Uniform mass on a central block of relative width `width_frac`."""
    w = max(1, int(width_frac * n))
    start = (n - w) // 2
    shape = np.zeros(n, dtype=np.float64)
    shape[start : start + w] = 1.0 / w
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector with concentration α."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))

# Optimisers
# ----------------------------------------------------------------------


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 200_000,
    lr_start: float = 0.03,
    lam_start: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a log‑sum‑exp surrogate of the max‑convolution.
    Returns the best exact objective seen.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # stay on simplex via softmax
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = lam_start
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase smoothness)
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake near the end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_momentum(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 0.8,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Projected‑gradient descent with momentum on the exact max‑convolution.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    v = np.zeros_like(a)

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # step size = η₀ / sqrt(t)
        eta = eta0 / np.sqrt(it + 1)

        v = beta * v - eta * grad
        a = a + v
        a = _project_onto_simplex(a)

        # occasional jitter close to the deadline
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _annealing_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style random mass‑transfer moves.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a)

    temperature = 0.02
    decay = 0.9997
    it = 0

    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤30 %) of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2)
        delta_val = val2 - best_val

        # Metropolis acceptance
        if delta_val < 0 or np.random.rand() < np.exp(-delta_val / max(temperature, 1e-12)):
            a = a2
            if val2 < best_val:
                best_val = val2
                best_a = a2.copy()
                if it % 5000 == 0:
                    print(f"[Anneal move {it:06d}] new best {val2:.6f}")

        temperature *= decay
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size
    # ------------------------------------------------------------------
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # optional warm start from a previously found sequence
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build a large and diverse initial pool
    # ------------------------------------------------------------------
    pool = []

    # deterministic families
    pool.append(_uniform(n))
    pool.append(_sinusoid(n))
    pool.append(_triangle(n))
    pool.append(_block(n, width_frac=0.4))

    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_edge_exponential(n, gamma))

    for alpha in (-2.5, -2.0, -1.5, -1.0, -0.5):
        pool.append(_edge_power_law(n, alpha))

    for a, b in [(0.5, 0.5), (0.5, 1.0), (1.0, 0.5),
                 (1.0, 1.0), (1.5, 1.5), (2.0, 2.0)]:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    # dense grid of beta shapes (captures many U‑shaped / bell‑shaped families)
    a_vals = np.linspace(0.2, 2.0, 9)
    b_vals = np.linspace(0.2, 2.0, 9)
    for a in a_vals:
        for b in b_vals:
            pool.append(_beta_shape(n, float(a), float(b)))

    # random Dirichlet draws – many concentrations for variety
    for alpha in (0.05, 0.1, 0.2, 0.4, 0.8, 1.2, 2.0, 5.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, alpha, rng))

    # add a few cyclic shifts of the warm start if available
    if warm is not None:
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # quick evaluation of the pool – keep the best few vectors
    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    top_k = 12
    top = []    # list of (val, vector)

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{idx}] new global best {val:.6f}")

        # maintain a small list of the best candidates for multi‑restart
        if len(top) < top_k:
            top.append((val, a.copy()))
            top.sort(key=lambda x: x[0])
        else:
            if val < top[-1][0]:
                top[-1] = (val, a.copy())
                top.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # Phase 2 – Adam with smooth‑max (use the best candidate)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + 0.20 * remaining  # ~20 % of remaining time
        a_adam, val_adam = _adam_smooth_max(
            best_seq,
            n,
            conv_len,
            corr_len,
            adam_deadline,
            max_iters=250_000,
            lr_start=0.03,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refinement] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected gradient descent with momentum
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        pgd_deadline = deadline - 5   # leave a few seconds for final stage
        a_pgd, val_pgd = _pgd_momentum(
            best_seq,
            n,
            conv_len,
            corr_len,
            pgd_deadline,
            max_iters=800_000,
            eta0=0.9,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD+Momentum] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated‑annealing hill climber (fine‑tuning)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        anneal_deadline = deadline - 1
        a_ann, val_ann = _annealing_hill_climb(
            best_seq,
            n,
            conv_len,
            anneal_deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Annealing] new best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the limits enforced by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
