   Descent* with Nesterov momentum on the exact max‑convolution.  The
   two phases are interleaved and each runs until the global deadline,
   keeping the best vector ever seen.

3. **Cheap hill‑climber** – in the last seconds a few random
   mass‑transfer moves are attempted; a move is kept only if it improves
   the exact objective.

All convolutions are performed by FFT on a power‑of‑two length for
speed.  The routine never returns an invalid sequence – every element is
clipped to the range required by ``evaluate_sequence`` – and it always
returns the best vector found before the 1000 s budget expires.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)-1) via FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns vector of length len(a) (same shape as a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of vector v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
#  Exact objective (identical to evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    """Compute 2·n·max(conv) / (sum(a)²) for a (already normalised to sum=1)."""
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
#  Analytic families (all return a normalised probability vector)
# ----------------------------------------------------------------------


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _sinusoid(n: int) -> np.ndarray:
    x = np.sin(np.pi * (np.arange(n, dtype=np.float64) + 0.5) / n)
    x = np.maximum(x, 0.0)
    return x / x.sum()


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    """Symmetric exponential decay from both ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^{-α} + (n-i)^{-α}."""
    i = np.arange(n, dtype=np.float64) + 1.0
    shape = (i ** (-alpha)) + ((n - i + 1) ** (-alpha))
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    """Beta‑pdf on a regular grid (mid‑point rule)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s == 0:
        return _uniform(n)
    return shape / s


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    """Two Gaussians centred at the two ends."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    """Draw a Dirichlet vector (concentration alpha)."""
    return rng.dirichlet(np.full(n, alpha))


# ----------------------------------------------------------------------
#  Optimisers
# ----------------------------------------------------------------------


def _adam_smooth_max(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lr_start: float = 0.04,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (exactly normalised) encountered.
    """
    eps = 1e-15
    # initialise on the simplex via logits (log‑softmax)
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start

    # temperature (inverse of smoothing) – start small, grow quickly
    lam = 0.01
    lam_factor = 1.10

    best_a = a0.copy()
    best_val = _objective(a0)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # current (exact) objective – keep best
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w *= 1.0 / w_sum

        # gradient of the surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through the soft‑max (mirrored‑descent form)
        grad_log = a * (grad_a - np.dot(a, grad_a))

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase smoothing)
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.94

        # small random jitter near the deadline (helps escape plateaus)
        if time.time() > deadline - 30 and it % 5_000 == 0:
            logits += (np.random.rand(n) - 0.5) * 4e-2

        it += 1

    return best_a, best_val


def _pgd_nesterov(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 1.0,
    momentum: float = 0.9,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected‑gradient descent with Nesterov momentum on the exact
    max‑convolution objective.  Returns the best vector encountered.
    """
    a = a0.copy()
    # ensure feasibility
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a)

    v = np.zeros_like(a)          # momentum buffer
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5_000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step‑size schedule (η_t = η0 / sqrt(t+1))
        eta = eta0 / np.sqrt(it + 1)

        # Nesterov update
        v = momentum * v + eta * grad
        a_next = a - v
        a = _project_onto_simplex(a_next)

        # occasional jitter in the final moments
        if time.time() > deadline - 20 and it % 10_000 == 0:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_moves(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber: move a random fraction of mass from one index
    to another if the exact objective improves.
    """
    a = a0.copy()
    best_val = _objective(a)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move up to 30 % of the source mass
        delta = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        val2 = _objective(a2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5_000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
#  Candidate pool construction
# ----------------------------------------------------------------------


def _build_initial_pool(n: int, rng: np.random.Generator) -> list[np.ndarray]:
    """Create a large pool of deterministic families and random Dirichlet draws."""
    pool: list[np.ndarray] = []

    # Deterministic families with a dense grid of parameters
    # – edge exponential
    for gamma in np.geomspace(0.01, 20.0, num=30):
        pool.append(_edge_exponential(n, gamma))

    # – edge power‑law
    for alpha in np.linspace(0.1, 5.0, num=25):
        pool.append(_edge_power_law(n, alpha))

    # – beta shapes (symmetric a=b)
    for a in np.geomspace(0.05, 5.0, num=30):
        pool.append(_beta_shape(n, a, a))

    # – double Gaussian
    for sigma in np.linspace(0.02, 0.4, num=20):
        pool.append(_double_gaussian(n, sigma))

    # – simple sinusoid and uniform
    pool.append(_sinusoid(n))
    pool.append(_uniform(n))

    # Random Dirichlet draws with a variety of concentrations
    for alpha in (0.05, 0.12, 0.25, 0.5, 1.0, 2.0, 5.0, 10.0):
        for _ in range(6):
            pool.append(_random_dirichlet(n, alpha, rng))

    return pool


# ----------------------------------------------------------------------
#  Main entry point
# ----------------------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size – we use 8192 points (larger than the previous 4096)
    # ------------------------------------------------------------------
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start – optional previous best stored in global variable
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64).ravel()
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # build candidate pool (deterministic families + random draws)
    # ------------------------------------------------------------------
    pool = _build_initial_pool(n, rng)
    if warm is not None:
        # also add a few cyclic shifts of the warm start
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # ------------------------------------------------------------------
    # quick scan of the pool – keep the best K seeds
    # ------------------------------------------------------------------
    best_val = float("inf")
    best_seq = None
    K = 5                     # number of seeds that survive to local optimisation
    top_seeds: list[tuple[float, np.ndarray]] = []

    for idx, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = _random_dirichlet(n, 1.0, rng)
        a /= a.sum()
        val = _objective(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool cand {idx:04d}] new global best {val:.6f}")
        # maintain a tiny heap of the best K seeds
        if len(top_seeds) < K:
            top_seeds.append((val, a.copy()))
            top_seeds.sort(key=lambda x: x[0])
        elif val < top_seeds[-1][0]:
            top_seeds[-1] = (val, a.copy())
            top_seeds.sort(key=lambda x: x[0])

    print(f"After pool scan: best = {best_val:.6f}, keep {K} seeds for refinement")

    # ------------------------------------------------------------------
    # Stage 2 – local optimisation of each seed
    # ------------------------------------------------------------------
    for seed_idx, (seed_val, seed_vec) in enumerate(top_seeds):
        if time.time() >= deadline - 5:
            break
        print(f"--- Refinement of seed {seed_idx} (start val {seed_val:.6f}) ---")
        # allocate a fraction of the remaining time to this seed
        remaining = deadline - time.time()
        # first Adam (smooth max) – about 20 % of the remaining slice
        adam_deadline = time.time() + 0.20 * remaining
        a_adam, val_adam = _adam_smooth_max(
            seed_vec,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            lr_start=0.05,
        )
        # second PGD (exact max) – use the rest of the time for this seed
        pgd_deadline = deadline   # run until overall deadline
        a_pgd, val_pgd = _pgd_nesterov(
            a_adam,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            eta0=1.2,
        )
        # pick the better of the two runs
        if val_pgd < val_adam:
            cur_best_a = a_pgd
            cur_best_val = val_pgd
        else:
            cur_best_a = a_adam
            cur_best_val = val_adam

        # keep globally best
        if cur_best_val < best_val:
            best_val = cur_best_val
            best_seq = cur_best_a.copy()
            print(f"[Seed {seed_idx}] new global best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – final cheap hill‑climber (only if time permits)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Final local search] starting cheap hill‑climber")
        a_loc, val_loc = _local_random_moves(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local moves] improved global best to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # safety fall‑back
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = _uniform(n)

    # clip to limits required by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
