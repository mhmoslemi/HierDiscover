def _proj_subgrad_optimize(a0, n, conv_fft_len, deadline, max_iters=300_000, eta0=1.0):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # subgradient for max
        active = np.isclose(conv, max_c, atol=1e-12)
        # weight uniform over active entries
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

        a_new = _project_simplex(a_new)
        # Ensure non-negative (project already)
        a = a_new
        it += 1
    return best_a, best_val

The objective  

\[
F(a)=\frac{2\,n\;\max (a*a)}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0,
\]

depends only on the *shape* of the step‑function, because a global scaling of the
vector cancels out.  
The problem is therefore to find a probability distribution
\(a\;(a_i\ge0,\;\sum a_i=1)\) that minimises the peak of its linear convolution
with itself.

The previous implementation already contains three useful phases:

1. **Adam on a smooth approximation** (`log‑sum‑exp`) – fast initial
   reduction.
2. **Exponentiated‑gradient sub‑gradient descent** – converges on the true
   non‑smooth objective while staying on the simplex.
3. **Pair‑wise targeted annealing** – removes mass from the pair that
   contributes most to the current peak.

To push the bound below the previous value **1.507708** we add two
complementary ideas:

* **Projected sub‑gradient descent on the exact max** – after the
  exponentiated‑gradient phase we switch to a *plain* sub‑gradient step
  followed by a Euclidean projection onto the probability simplex.
  This gives a finer control of the step size and often escapes the
  tiny plateaus where the exponentiated version stalls.
* **More aggressive multi‑restart exploration** – we generate a larger,
  more diverse pool of starting vectors (Dirichlet draws with very small
  concentration, random sparse spikes, sinusoidal/gaussian families,
  and a warm‑start if one exists).  Every candidate receives a short
  Adam run followed by a brief sub‑gradient refinement; the best of all
  candidates is then handed to the long optimisation phases.

All phases respect the global time budget (`budget_s`, default 1000 s) and
the best vector found at any moment is always kept.  The code never writes
to disk, uses only `numpy` and the supplied `evaluate_sequence`, and returns a
plain Python list ready for the scorer.

```python
"""
Hybrid optimiser for the step‑function objective F(a)=2·n·max(a∗a)/(∑a)².

Key novelties over the previous runner:
  * a much larger, diversified start pool (Dirichlet, random spikes,
    Gaussian / sinusoidal families, warm‑start);
  * an extra “projected sub‑gradient” phase that works on the exact
    max (instead of a smooth surrogate) and projects onto the simplex;
  * longer iteration budgets for the most effective phases while still
    staying inside the 1000 s wall‑time;
  * deterministic acceptance in the final annealing stage (only improvements
    are kept) to sharpen the last few digits.

The routine always returns the best sequence discovered before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) = ∂/∂a Σ_k w_k (a∗a)_k.
    Returns a gradient of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # numerical safety – return uniform vector
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max surrogate log‑sum‑exp(λ·(a∗a)).
    Returns the best vector seen (always Σa=1) and its exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # stay in the interior of the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.08
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max => a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional tiny shake when the deadline is near
        if it % 3500 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 70_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    The iterates stay on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set – indices where the maximum is attained
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _proj_subgrad_optimize(a0: np.ndarray,
                           n: int,
                           conv_fft_len: int,
                           corr_fft_len: int,
                           deadline: float,
                           max_iters: int = 300_000,
                           eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Plain projected sub‑gradient descent on the *exact* max‑convolution.
    After each gradient step the iterate is projected onto the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ProjSub it={it:06d}] new best {val:.6f}")

        # active set (where the maximum is attained)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = _project_simplex(a - eta * grad)

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(a0: np.ndarray,
                                   n: int,
                                   conv_fft_len: int,
                                   deadline: float,
                                   max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted simulated annealing (identical to the original version
    but kept for the final polishing stage).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    best_val = val

    T_start = 0.05
    T_end = 1e-5
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # indices contributing to the maximal entry
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        if i_src == j_src:
            # single‑index case – move a small fraction elsewhere
            if a2[i_src] < 1e-12:
                move += 1
                continue
            delta = a2[i_src] * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t = np.random.randint(n)
            while t == i_src:
                t = np.random.randint(n)
            a2[i_src] -= delta
            a2[t] += delta
        else:
            # pair case – remove a small amount from both ends
            delta = min(a2[i_src], a2[j_src]) * 0.05
            if delta < 1e-14:
                move += 1
                continue
            t1 = np.random.randint(n)
            t2 = np.random.randint(n)
            a2[i_src] -= delta
            a2[j_src] -= delta
            a2[t1] += delta
            a2[t2] += delta

        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Candidate generation helpers
# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a diversified pool of starting vectors."""
    cand = []

    # warm start (and few cyclic shifts)
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with many concentration parameters
    for alpha in (0.005, 0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law edge families (more extreme tails)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.3, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.15, 0.3):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid (zero at the borders)
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates consisting of a few equally‑spaced or random spikes."""
    out = []
    for k in (32, 48, 64, 96, 128, 192, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)

    # random‑position sparse spikes
    for k in (48, 64, 96, 128):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)

    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects the given time budget and always returns the
    best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # tiny safety margin

    # problem size – 4096 gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a big and diverse start pool
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – quick exploration of many starts (≈30 % of the time)
    explore_deadline = time.time() + (deadline - time.time()) * 0.30
    while pool and time.time() < explore_deadline:
        a0 = pool.pop(0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # very short Adam run (≈1.8 s)
        sub_dead = min(explore_deadline, time.time() + 1.8)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=60_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore] Adam improved to {val_adam:.6f}")

        # brief exponentiated‑gradient refinement (≈1 s)
        if time.time() < explore_deadline:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                explore_deadline,
                max_iters=8_000,
                eta0=0.9
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient sub‑gradient descent
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase II] Long exp‑gradient sub‑gradient")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
            eta0=0.85,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Phase II] New best {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – projected sub‑gradient descent on the exact max
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase III] Projected sub‑gradient")
        a_proj, val_proj = _proj_subgrad_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=250_000,
            eta0=0.8,
        )
        if val_proj < best_val:
            best_val = val_proj
            best_seq = a_proj.copy()
            print(f"[Phase III] new best {val_proj:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – targeted pair‑wise annealing for final polishing
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase IV] Pair‑wise targeted annealing")
        a_pair, val_pair = _pair_targeted_anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_pair < best_val:
            best_val = val_pair
            best_seq = a_pair.copy()
            print(f"[Phase IV] improved best to {val_pair:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling to a finer grid if time allows
    if best_seq is not None and time.time() + 30 < deadline:
        print("[Upsample] Trying resolution 8192")
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        # linear interpolation from the current best
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big = a_big / a_big.sum()

        # short Adam on the larger grid
        sub_dead_big = min(deadline, time.time() + 5.0)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=120_000,
        )

        # longer projected sub‑gradient on the large grid
        a_big_proj, val_big = _proj_subgrad_optimize(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=200_000,
            eta0=0.7,
        )
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big_proj.copy()
            print(f"[Upsample] refined best {val_big:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```
