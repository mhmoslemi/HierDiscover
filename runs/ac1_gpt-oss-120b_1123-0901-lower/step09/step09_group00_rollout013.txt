"""
Hybrid optimizer for the step‑function “max‑convolution” bound.

The routine builds a tiny pool of diverse starters (warm start, uniform,
Dirichlet draws, simple exponentials).  It evaluates each starter with a
fast FFT‑based convolution and keeps the best one (objective = 2·n·max(conv) /
sum²).  The main work consists of a long Adam run on a smooth‑max surrogate,
followed by an exponentiated‑gradient sub‑gradient phase, a projected‑gradient
descent on the exact max‑convolution, and finally a simulated‑annealing
refinement.  While time permits the best solution is repeatedly perturbed
and re‑optimised with a short Adam run to escape possible local minima.
All iterates stay on the probability simplex (Σa=1) – the objective is
scale‑invariant, therefore this does not affect the score.
"""

import time
import numpy as np
from scipy.optimize import linprog  # optional, not used in the final pipeline


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective_value(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(conv) / (sum²).  a is assumed non‑negative."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    sum_a = a.sum()
    if sum_a < 0.01:  # matches the safety guard in evaluate_sequence
        return float("inf")
    return 2.0 * n * max_c / (sum_a ** 2)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5                # start temperature for smooth‑max
    lam_factor = 1.05

    best_a = a0.copy()
    best_val = _objective_value(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)   # a.sum() should be ~1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – push it up quickly
        lam = min(lam * lam_factor, 1e7)

        # modest learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector (still on the simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective_value(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Plain projected gradient descent on the exact max‑convolution.
    Uses a diminishing step size and keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective_value(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        it += 1

    return best_a, best_val


def _perturb(a: np.ndarray, scale: float = 0.02) -> np.ndarray:
    """Add small Gaussian noise, re‑project onto the simplex."""
    b = a + scale * np.random.randn(a.shape[0])
    b = np.maximum(b, 0.0)
    s = b.sum()
    if s < 1e-12:
        b = np.random.dirichlet(np.ones_like(a))
        s = b.sum()
    return b / s


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a little safety margin

    # problem size (feel free to adjust – larger n gives a better bound but
    # costs more per iteration)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous solution (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Small, diverse pool of starters
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # three circular shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws (different concentration)
    for alpha in (0.2, 0.5, 1.0, 2.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Simple exponential decay (monotone – helpful to break symmetry)
    for gamma in (0.3, 0.6, 1.0):
        a = np.exp(-gamma * np.arange(n, dtype=np.float64))
        a = a / a.sum()
        pool.append(a)

    # Double‑exponential (symmetric) – kept for completeness
    for gamma in (0.2, 0.5):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        a = left + right
        a = a / a.sum()
        pool.append(a)

    # Random shuffled Dirichlet
    a = np.random.dirichlet(np.ones(n))
    a = a[np.random.permutation(n)]
    pool.append(a)

    # ------------------------------------------------------------------
    # Evaluate starter pool, keep the best
    best_seq = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        cand = np.maximum(cand, 0.0)
        if cand.sum() < 0.01:
            cand = np.random.dirichlet(np.ones(n))
        cand = cand / cand.sum()
        val = _objective_value(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Init {idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – long Adam optimisation (≈ half of remaining budget)
    remaining = deadline - time.time()
    if remaining > 30.0:
        slice_adam = min(remaining * 0.55, 300.0)       # cap at ~5 min
        sub_dead = time.time() + slice_adam
        print("[Phase I] Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=2_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – exponentiated‑gradient sub‑gradient (≈ 20 % of budget)
    remaining = deadline - time.time()
    if remaining > 20.0:
        slice_sub = min(remaining * 0.25, 180.0)
        sub_dead = time.time() + slice_sub
        print("[Phase II] Sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000, eta0=1.0,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] Sub‑gradient improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase III – projected gradient descent on the exact max‑conv (≈10 %)
    remaining = deadline - time.time()
    if remaining > 10.0:
        slice_gd = min(remaining * 0.15, 150.0)
        sub_dead = time.time() + slice_gd
        print("[Phase III] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase III] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated‑annealing local search (use whatever time is left)
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("[Phase IV] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=int(remaining * 2000),   # about 2000 moves per second
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – stochastic perturb‑and‑re‑Adam loop (use any leftover seconds)
    while True:
        remaining = deadline - time.time()
        if remaining < 12.0:
            break
        # perturb, then short Adam run (~8 s)
        pert = _perturb(best_seq, scale=0.03)

        a_tmp, val_tmp = _adam_optimize(
            pert, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=600_000,
        )
        if val_tmp < best_val:
            best_val = val_tmp

    # ------------------------------------------------------------------
    # Ensure non‑empty return and clip to evaluator limits
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
