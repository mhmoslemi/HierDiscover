def _lbfgs_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, lambda_start=10.0, lambda_end=1e5, steps=5):
    start = a0
    lam = lambda_start
    best_a, best_val = start, evaluate_sequence(start.tolist())
    for s in range(steps):
        # run L-BFGS
        logits = np.log(start + 1e-12)
        def func(logits_flat):
            logits = logits_flat
            # softmax
            mx = logits.max()
            exp_log = np.exp(logits - mx)
            a = exp_log / exp_log.sum()
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            # smooth max

            w = w_exp / w_exp.sum()
            smooth = max_c + (1/lam) * np.log(w_exp.sum())

            # gradient

            avg = np.dot(a, grad_a)
            grad_log = a * (grad_a - avg)
            return float(val), grad_log.astype(np.float64)
        res = scipy.optimize.minimize(func, logits, method='L-BFGS-B', jac=True,
                                      options={'maxiter': 2000, 'disp': False})

    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # compute smooth_max using stabilized log-sum-exp
    c_max = np.max(conv)
    exp_term = np.exp(lam * (conv - c_max))
    sum_exp = np.sum(exp_term)
    smooth = c_max + (1.0/lam) * np.log(sum_exp)

    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return float(val), grad_log

2. **Smooth‑max L‑BFGS** – the current best vector is re‑parameterised by
   logits (soft‑max) and minimised with L‑BFGS on a log‑sum‑exp surrogate
   of the max–convolution.  The temperature λ is increased geometrically,
   allowing the optimisation to gradually focus on the true max.

3. **Exact‑max projected gradient descent** – after the surrogate stage a
   few thousand iterations of projected gradient descent on the exact
   max‑convolution are performed (sub‑gradient descent on the true
   objective).

4. **Local stochastic refinement** – a final simulated‑annealing style
   mass‑transfer move set explores tiny neighbourhoods and often squeezes
   out the last bits of improvement.

All work respects the global time budget (≈ 1000 s) and always returns the
best sequence found.  The evaluation function is invariant to scaling, so
the iterates are kept on the probability simplex (sum = 1).  No lambdas
or nested closures are used – every helper is defined at top level.
"""

import time
import numpy as np
import scipy.optimize


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w★a needed for (sub‑)gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x ≥ 0, Σx = 1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _lbfgs_obj(logits: np.ndarray,
               n: int,
               conv_fft_len: int,
               corr_fft_len: int,
               lam: float):
    """
    Smooth‑max objective and its gradient w.r.t. logits.
    """
    # soft‑max parametrisation (numerically stable)
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # smooth max = log‑sum‑exp / lam  (shifted by the max for stability)
    c_max = conv.max()
    exp_term = np.exp(lam * (conv - c_max))
    sum_exp = exp_term.sum()
    smooth = c_max + (1.0 / lam) * np.log(sum_exp)

    # objective (sum(a)==1 ⇒ denominator = 1)
    val = 2.0 * n * smooth

    # gradient w.r.t a (exact for the smooth‑max surrogate)
    w = exp_term / sum_exp                 # soft weights
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)

    return float(val), grad_log.astype(np.float64)


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    corr_fft_len: int,
                    lam: float,
                    deadline: float,
                    maxiter: int = 1500) -> tuple[np.ndarray, float]:
    """
    Run L‑BFGS on the smooth‑max surrogate starting from a0.
    Returns the (renormalised) vector and its exact evaluation.
    """
    eps = 1e-12
    logits0 = np.log(a0 + eps)

    # L‑BFGS – the optimiser stops early if the deadline is reached
    res = scipy.optimize.minimize(_lbfgs_obj,
                                  logits0,
                                  args=(n, conv_fft_len, corr_fft_len, lam),
                                  method='L-BFGS-B',
                                  jac=True,
                                  options={'maxiter': maxiter, 'disp': False})
    # recover probabilities
    logits = res.x
    mx = logits.max()
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    # safety – enforce the simplex (numerical drift)
    a = _project_simplex(a)

    val = evaluate_sequence(a.tolist())
    return a, float(val)


def _max_conv_descent(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the exact max‑convolution.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 80_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search moving a random fraction of mass.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        a2 = _project_simplex(a2)

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a low‑evaluation step‑function.
    The routine respects the given time budget (≈ 1000 s) and always
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5    # safety margin

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # build a modest but diverse pool of starters
    pool = []

    # warm start (previous best, if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # uniform
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # Dirichlet draws with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # symmetric power‑law (Beta‑like) families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.2, 9):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # a half‑wave sinusoid (already useful)
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus = sinus / sinus.sum()
    pool.append(sinus)

    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    # quick evaluation of the pool (few seconds)
    for a in pool:
        if time.time() > deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] new best {val:.6f}")

    # --------------------------------------------------------------
    # Phase I – smooth‑max L‑BFGS with increasing temperature
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    lam = 5.0
    while time.time() < deadline and lam < 1e5:
        a, val = _lbfgs_optimize(best_seq, n,
                                 conv_fft_len, corr_fft_len,
                                 lam, deadline,
                                 maxiter=1500)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[L‑BFGS λ={lam:.1f}] new best {val:.6f}")
        lam = min(lam * 10.0, 1e5)

    # --------------------------------------------------------------
    # Phase II – exact max‑convolution projected GD
    if time.time() < deadline - 10:
        a, val = _max_conv_descent(best_seq, n,
                                   conv_fft_len, corr_fft_len,
                                   deadline,
                                   max_iters=120_000)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[GD] improved best to {val:.6f}")

    # --------------------------------------------------------------
    # Phase III – stochastic local refinement (annealing)
    if time.time() < deadline - 5:
        a, val = _anneal_optimize(best_seq, n,
                                   conv_fft_len,
                                   deadline,
                                   max_moves=90_000)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Anneal] improved best to {val:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling to a finer grid if time permits
    if time.time() + 30 < deadline:
        try:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)
            # linear interpolation onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
            a_big = a_big / a_big.sum()
            # a short L‑BFGS refinement on the refined grid
            a_big, val_big = _lbfgs_optimize(a_big, n_big,
                                             conv_fft_len_big, corr_fft_len_big,
                                             lam=1e4, deadline=deadline,
                                             maxiter=800)
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # --------------------------------------------------------------
    # final safety clamp
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
