"""
Propose a low‑score non‑negative step function.

Algorithm (≈ 1000 s budget)

1. **Random parametric sweep** – generate many candidates from several
   families (edge‑exponential, edge‑power‑law, beta, double‑Gaussian,
   sinusoid‑power, triangular, Dirichlet, two‑edge blocks).  Keep the best.

2. **Smooth‑max L‑BFGS refinement** – optimise the soft‑max (log‑sum‑exp)
   surrogate of the convolution maximum.  The variables are logits of a
   soft‑max (so the sequence stays on the probability simplex).  A schedule
   of increasing smoothness parameters (lam) is used; each stage runs a
   limited‑memory BFGS (SciPy) on the surrogate.  After each stage the true
   objective (max‑convolution) is evaluated and the best vector is cached.

3. **Cheap stochastic hill‑climbing** – in the final seconds a simple
   random “mass‑transfer” local search moves a fraction of mass from one
   index to another, accepting any improvement.

The final vector is clipped to the range accepted by `evaluate_sequence`
and returned as a plain list.
"""

import time
import numpy as np
from numpy.fft import rfft, irfft

# ----------------------------------------------------------------------
# FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = rfft(a, fft_len)
    conv = irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = rfft(w, fft_len)
    fr = rfft(rev_a, fft_len)
    corr = irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Simple projection onto the probability simplex (used for fallback)
def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence up to clamping)
def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²)."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Parametric shape generators
def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _two_edge_block_shape(n: int, block_frac: float, rng: np.random.Generator) -> np.ndarray:
    """Two equal blocks at the edges, each carrying half of the total mass."""
    k = max(1, int(block_frac * n))
    shape = np.zeros(n, dtype=np.float64)
    shape[:k] = 0.5 / k
    shape[-k:] = 0.5 / k
    # if blocks overlap (very large block_frac) the centre may be double‑counted,
    # but the normalisation step fixes it.
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


# ----------------------------------------------------------------------
# Smooth‑max surrogate (log‑sum‑exp) for L‑BFGS
def _smoothmax_obj(u: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, lam: float):
    """
    Returns (value, gradient) of  (1/lam)·logsumexp(lam·(a*a))
    where a = softmax(u) and a*a is the linear convolution.
    """
    # soft‑max to stay on the simplex
    mx = u.max()
    exp_u = np.exp(u - mx)
    a = exp_u / exp_u.sum()

    # linear convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # log‑sum‑exp of lam·conv (stable)
    x = lam * conv
    x_max = x.max()
    exp_x = np.exp(x - x_max)
    sumexp = exp_x.sum()
    val = (np.log(sumexp) + x_max) / lam  # smooth approximation of max(conv)

    # weights for the gradient w.r.t. a
    w = exp_x / sumexp

    # d(conv)/da is the cross‑correlation with w
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max (mirror‑descent form)
    avg = np.dot(a, grad_a)
    grad_u = a * (grad_a - avg)

    return float(val), grad_u.astype(np.float64)


# ----------------------------------------------------------------------
# Cheap stochastic hill‑climbing (identical to the one in the reference)
def _local_random_search(a0: np.ndarray, n: int, conv_fft_len: int,
                         deadline: float, max_moves: int = 80_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        delta = a[i] * (np.random.rand() * 0.3)   # move up to 30 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")
        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    F(a)=2·n·max(a∗a)/(sum(a)²) under the given time budget.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # problem size – 4096 works well for the time budget
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous run stored a vector)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – broad parametric sweep (≈15 % of the budget)
    phase1_deadline = min(deadline - 5.0,
                         time.time() + 0.15 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")
    families = ["edge_exp", "edge_pow", "beta", "double_gauss",
                "sin_pow", "tent", "dirichlet", "block_edge"]
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-3, 4)          # 0.001 … 10 000
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-20.0, -0.05)          # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 2)              # 0.01 … 100
            b = 10 ** rng.uniform(-2, 2)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.5)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 8.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.02, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        else:   # block_edge
            block_frac = rng.uniform(0.001, 0.2)      # fraction of the array used for each edge block
            cand = _two_edge_block_shape(n, block_frac, rng)

        val = _objective(cand, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Phase 1] new best {val:.6f} via {fam}")

    # include a few shifted versions of a warm start (if any)
    if warm is not None:
        for shift in (0, n // 8, n // 4, n // 2):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Warm‑shift] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – smooth‑max L‑BFGS refinement (≈70 % of the budget)
    if best_seq is None:
        # fallback initialisation
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    a0 = best_seq.copy()
    # logits for the soft‑max parameterisation
    eps = 1e-15
    logits = np.log(a0 + eps)

    # schedule of smoothness parameters (lam).  Start small (smooth) and
    # increase gradually (sharper approximation of the true max).
    lam_schedule = [0.5, 2.0, 5.0, 12.0, 30.0, 80.0, 200.0, 600.0, 1800.0]
    max_bfgs_iters = 150   # per lam stage – small enough to respect the deadline

    # try to import scipy; if unavailable we skip this stage.
    try:
        from scipy.optimize import minimize
        have_scipy = True
    except Exception:  # pragma: no cover
        have_scipy = False
        print("SciPy not available – skipping L‑BFGS phase.")

    if have_scipy:
        print("Phase 2 – L‑BFGS smooth‑max optimisation")
        for lam in lam_schedule:
            if time.time() >= deadline - 10:
                break
            # minimise the smooth‑max surrogate
            res = minimize(_smoothmax_obj,
                           logits,
                           args=(n, conv_fft_len, corr_fft_len, lam),
                           method='L-BFGS-B',
                           jac=True,
                           options={'maxiter': max_bfgs_iters,
                                    'disp': False})
            # recompute the actual objective from the obtained logits
            u_opt = res.x
            a_opt = np.exp(u_opt - np.max(u_opt))
            a_opt /= a_opt.sum()
            val_actual = _objective(a_opt, conv_fft_len)
            if val_actual < best_val:
                best_val, best_seq = val_actual, a_opt.copy()
                print(f"[L‑BFGS lam={lam:.1f}] new best {val_actual:.6f}")

            # warm‑start the next lam‑stage from the current optimum,
            # add a tiny perturbation to escape possible plateaus.
            logits = u_opt + (np.random.rand(n) - 0.5) * 2e-3

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing (last few seconds)
    if time.time() < deadline - 2:
        print("Phase 3 – local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Phase 3] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (unlikely) – uniform distribution
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence's limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")

    # expose the result for the next iteration (optional)
    globals()["height_sequence_1"] = best_seq.tolist()

    return best_seq.tolist()
