"""
Propose a low‑score non‑negative step function.

The search consists of three time‑aware stages:

1. **Broad parametric sweep** – draw many candidates from a rich
   collection of families (symmetric / asymmetric edge exponentials,
   power‑law edges, beta, double‑Gaussian, sinusoid‑power, tent,
   Dirichlet and random convex combinations).  The best *K* candidates
   (by the fast FFT‑based objective) are retained.

2. **Multi‑restart gradient refinement** – each retained candidate is
   locally optimised by a short Adam run (smooth‑max surrogate),
   followed by a sub‑gradient and a projected‑gradient pass.  The
   optimizer respects the global deadline, so the total time is split
   evenly between the restarts.

3. **Intensive stochastic hill‑climbing** – the current best vector is
   further improved with a fast mass‑transfer local search that accepts
   any improvement.  The search runs until the very end of the budget.

All vectors are kept on the probability simplex (non‑negative and
normalised).  After the optimisation the sequence is clipped to the
range accepted by `evaluate_sequence` and returned as a plain list.
"""

import time
import math
import numpy as np

# FFT helpers (exactly the same as in the reference implementation)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a via FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Fast objective (identical to evaluate_sequence up to clamping)

def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """2·n·max(a∗a) / (sum(a)²) using FFT‑based convolution."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _edge_exponential_shape(n: int, gamma: float, weight: float = 0.5) -> np.ndarray:
    """Symmetric mixture of two edge exponentials with a common decay."""
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    right = left[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_exponential_asym(n: int,
                           gamma_left: float,
                           gamma_right: float,
                           weight: float = 0.5) -> np.ndarray:
    """Asymmetric mixture of two edge exponentials."""
    left = np.exp(-gamma_left * np.arange(n, dtype=np.float64))
    right = np.exp(-gamma_right * np.arange(n, dtype=np.float64))[::-1]
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_shape(n: int, alpha: float, weight: float = 0.5) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    left = idx ** alpha
    right = (n - idx + 1) ** alpha
    shape = weight * left + (1.0 - weight) * right
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _edge_power_law_asym(n: int,

def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _double_gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _sinusoid_power_shape(n: int, power: float) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n) ** power
    s = np.maximum(s, 0.0)
    ss = s.sum()
    if ss > 0:
        s /= ss
    else:
        s = np.full(n, 1.0 / n, dtype=np.float64)
    return s


def _tent_shape(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    shape = np.minimum(i + 1, n - i)
    shape = np.maximum(shape, 0.0)
    ss = shape.sum()
    if ss > 0:
        shape /= ss
    else:
        shape = np.full(n, 1.0 / n, dtype=np.float64)
    return shape


def _random_dirichlet_shape(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    v = rng.dirichlet(np.full(n, alpha))
    return v.astype(np.float64)


def _mix_two_shapes(a: np.ndarray, b: np.ndarray, lam: float) -> np.ndarray:
    """Convex combination of two simplex vectors, returning a simplex vector."""
    mix = lam * a + (1.0 - lam) * b
    ss = mix.sum()
    if ss > 0:
        mix /= ss
    else:
        mix = np.full_like(a, 1.0 / a.shape[0])
    return mix

# Optimisers (identical in spirit to the reference implementation)

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 200_000,
                   lr_start: float = 0.05) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain simplex vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent style gradient on logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 150_000,
                          eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pgd_optimize(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  corr_fft_len: int,
                  deadline: float,
                  max_iters: int = 120_000,
                  eta0: float = 0.4) -> tuple[np.ndarray, float]:
    """Projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        it += 1

    return best_a, best_val


def _local_random_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    """Simple stochastic mass‑transfer hill‑climbing."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()
    it = 0
    rng = np.random.default_rng()
    while time.time() < deadline and it < max_moves:
        i = rng.integers(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = rng.integers(n)
        if i == j:
            it += 1
            continue
        # move a random fraction (up to 40 %) of the source mass
        delta = a[i] * (rng.random() * 0.4)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    `evaluate_sequence`.  The routine respects a 1000 s wall‑clock budget.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # small safety margin

    # ------------------------------------------------------------------
    # problem size – 4096 is a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Phase 1 – rich parametric sampling, keep top‑K candidates
    K = 8
    best_cands = []   # list of (value, vector)

    def _maybe_push(val, vec):
        if len(best_cands) < K:
            best_cands.append((val, vec))
            best_cands.sort(key=lambda x: x[0])
        elif val < best_cands[-1][0]:
            best_cands[-1] = (val, vec)
            best_cands.sort(key=lambda x: x[0])

    families = [
        "edge_exp", "edge_exp_asym", "edge_pow", "edge_pow_asym",
        "beta", "double_gauss", "sin_pow", "tent", "dirichlet", "mix"
    ]

    phase1_deadline = min(deadline - 5.0,
                          time.time() + 0.25 * (deadline - time.time()))
    print("Phase 1 – random parametric sampling")
    while time.time() < phase1_deadline:
        fam = rng.choice(families)
        if fam == "edge_exp":
            gamma = 10 ** rng.uniform(-2, 2)          # 0.01 … 100
            weight = rng.random()
            cand = _edge_exponential_shape(n, gamma, weight)
        elif fam == "edge_exp_asym":
            gamma_l = 10 ** rng.uniform(-3, 3)
            gamma_r = 10 ** rng.uniform(-3, 3)
            weight = rng.random()
            cand = _edge_exponential_asym(n, gamma_l, gamma_r, weight)
        elif fam == "edge_pow":
            alpha = rng.uniform(-10.0, -0.1)          # strong edge bias
            weight = rng.random()
            cand = _edge_power_law_shape(n, alpha, weight)
        elif fam == "edge_pow_asym":
            alpha_l = rng.uniform(-15.0, -0.1)
            alpha_r = rng.uniform(-15.0, -0.1)
            weight = rng.random()
            cand = _edge_power_law_asym(n, alpha_l, alpha_r, weight)
        elif fam == "beta":
            a = 10 ** rng.uniform(-2, 2)             # 0.01 … 100
            b = 10 ** rng.uniform(-2, 2)
            cand = _beta_shape(n, a, b)
        elif fam == "double_gauss":
            sigma = rng.uniform(0.005, 0.6)
            cand = _double_gaussian_shape(n, sigma)
        elif fam == "sin_pow":
            power = rng.uniform(0.3, 8.0)
            cand = _sinusoid_power_shape(n, power)
        elif fam == "tent":
            cand = _tent_shape(n)
        elif fam == "dirichlet":
            alpha = rng.uniform(0.01, 5.0)
            cand = _random_dirichlet_shape(n, alpha, rng)
        else:  # mix two random families
            # draw two independent candidates and mix them
            s1 = rng.choice(families[:-1])   # avoid infinite recursion
            s2 = rng.choice(families[:-1])
            # generate base shapes (recursive style avoided by minor copy‑paste)
            # For simplicity we reuse the logic of the first two branches:
            if s1 == "edge_exp":
                g1 = 10 ** rng.uniform(-2, 2)
                w1 = rng.random()
                sh1 = _edge_exponential_shape(n, g1, w1)
            elif s1 == "edge_exp_asym":
                gl = 10 ** rng.uniform(-3, 3)
                gr = 10 ** rng.uniform(-3, 3)
                w1 = rng.random()
                sh1 = _edge_exponential_asym(n, gl, gr, w1)
            elif s1 == "edge_pow":
                al = rng.uniform(-10.0, -0.1)
                w1 = rng.random()
                sh1 = _edge_power_law_shape(n, al, w1)
            elif s1 == "edge_pow_asym":
                al = rng.uniform(-15.0, -0.1)
                ar = rng.uniform(-15.0, -0.1)
                w1 = rng.random()
                sh1 = _edge_power_law_asym(n, al, ar, w1)
            elif s1 == "beta":
                a1 = 10 ** rng.uniform(-2, 2)
                b1 = 10 ** rng.uniform(-2, 2)
                sh1 = _beta_shape(n, a1, b1)
            elif s1 == "double_gauss":
                sig1 = rng.uniform(0.005, 0.6)
                sh1 = _double_gaussian_shape(n, sig1)
            elif s1 == "sin_pow":
                pw1 = rng.uniform(0.3, 8.0)
                sh1 = _sinusoid_power_shape(n, pw1)
            elif s1 == "tent":
                sh1 = _tent_shape(n)
            else:
                sh1 = _random_dirichlet_shape(n, rng.uniform(0.01, 5.0), rng)

            # second shape
            if s2 == "edge_exp":
                g2 = 10 ** rng.uniform(-2, 2)
                w2 = rng.random()
                sh2 = _edge_exponential_shape(n, g2, w2)
            elif s2 == "edge_exp_asym":
                gl = 10 ** rng.uniform(-3, 3)
                gr = 10 ** rng.uniform(-3, 3)
                w2 = rng.random()
                sh2 = _edge_exponential_asym(n, gl, gr, w2)
            elif s2 == "edge_pow":
                al = rng.uniform(-10.0, -0.1)
                w2 = rng.random()
                sh2 = _edge_power_law_shape(n, al, w2)
            elif s2 == "edge_pow_asym":
                al = rng.uniform(-15.0, -0.1)
                ar = rng.uniform(-15.0, -0.1)
                w2 = rng.random()
                sh2 = _edge_power_law_asym(n, al, ar, w2)
            elif s2 == "beta":
                a2 = 10 ** rng.uniform(-2, 2)
                b2 = 10 ** rng.uniform(-2, 2)
                sh2 = _beta_shape(n, a2, b2)
            elif s2 == "double_gauss":
                sig2 = rng.uniform(0.005, 0.6)
                sh2 = _double_gaussian_shape(n, sig2)
            elif s2 == "sin_pow":
                pw2 = rng.uniform(0.3, 8.0)
                sh2 = _sinusoid_power_shape(n, pw2)
            elif s2 == "tent":
                sh2 = _tent_shape(n)
            else:
                sh2 = _random_dirichlet_shape(n, rng.uniform(0.01, 5.0), rng)

            lam_mix = rng.random()
            cand = _mix_two_shapes(sh1, sh2, lam_mix)

        val = _objective(cand, conv_fft_len)
        _maybe_push(val, cand)

    # incorporate warm‑start shifts as extra candidates
    if warm is not None:
        for shift in (0, n // 4, n // 2, 3 * n // 4):
            cand = np.roll(warm, shift)
            val = _objective(cand, conv_fft_len)
            _maybe_push(val, cand)

    # ------------------------------------------------------------------
    # Phase 2 – multi‑restart deep gradient refinement
    best_seq = None
    best_val = float("inf")
    # initialise from the current best among the pool
    if best_cands:
        best_val, best_seq = best_cands[0]

    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        print("Phase 2 – gradient refinement of top candidates")
        # allocate roughly equal time per candidate
        for idx, (init_val, init_seq) in enumerate(best_cands):
            # recompute remaining time before each start
            remaining = deadline - time.time()
            if remaining < 10:
                break
            sub_dead = time.time() + remaining / (len(best_cands) - idx)
            # Adam
            a_adam, val_adam = _adam_optimize(
                init_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=180_000, lr_start=0.05)
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Restart {idx}] Adam improved to {val_adam:.6f}")

            # Sub‑gradient (if time permits)
            if time.time() < sub_dead - 5:
                a_sub, val_sub = _subgradient_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                    max_iters=140_000, eta0=0.85)
                if val_sub < best_val:
                    best_val, best_seq = val_sub, a_sub.copy()
                    print(f"[Restart {idx}] Sub‑grad improved to {val_sub:.6f}")

            # PGD (if time permits)
            if time.time() < sub_dead - 2:
                a_pgd, val_pgd = _pgd_optimize(
                    best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                    max_iters=120_000, eta0=0.35)
                if val_pgd < best_val:
                    best_val, best_seq = val_pgd, a_pgd.copy()
                    print(f"[Restart {idx}] PGD improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap stochastic hill‑climbing (final seconds)
    if best_seq is not None and time.time() < deadline - 0.2:
        print("Phase 3 – final local random moves")
        a_loc, val_loc = _local_random_search(
            best_seq, n, conv_fft_len, deadline,
            max_moves=200_000)
        if val_loc < best_val:
            best_val, best_seq = val_loc, a_loc.copy()
            print(f"[Final] local search improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback – uniform distribution (should never be needed)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (matches evaluate_sequence's limits)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
