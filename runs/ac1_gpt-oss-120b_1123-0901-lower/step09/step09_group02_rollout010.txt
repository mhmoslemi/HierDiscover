def subgradient_opt(a, steps, deadline):
   a = np.maximum(a, 0)
   a = a / a.sum()
   best = a.copy()
   best_val = evaluate_objective(a)
   for it in range(steps):
       if time.time() > deadline: break
       conv = _full_convolution(a, conv_fft_len)
       max_c = conv.max()
       # active indices
       active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
       w = np.zeros_like(conv)
       w[active] = 1.0 / active.sum()
       grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
       eta = eta0 / np.sqrt(it + 1.0)
       a = a * np.exp(-eta * grad)
       a_sum = a.sum()
       if a_sum == 0: a = np.random.dirichlet(np.ones(n))
       else: a /= a_sum
       # evaluate etc.
   return best, best_val
```

"""
stepfunction_optimizer_v3.py

A faster, more exploratory optimizer for

    F(a) = 2·n·max(a*a) / (∑a)² ,   a_i ≥ 0

The routine works in four stages:

1. **Rich diverse start pool** – many analytic families (exponential,
   Gaussian, Beta, triangular, sinusoid) plus random Dirichlet /
   sparse spikes.  Every candidate is normalised to a simplex.

2. **Brief local refinements** – the K best start vectors are each
   polished by a short exponentiated‑gradient (sub‑gradient) run.
   This gives a cheap multi‑restart search.

3. **Long Adam run on the current best** – a continuation method that
   minimises a smooth log‑sum‑exp surrogate of the max.  The schedule
   slowly raises the smoothing parameter (λ) and uses Adam’s adaptive
   step sizes.  This is the main work and usually drives the objective
   below the previous record.

4. **Final polishing** – a short simulated‑annealing phase with
   mass‑transfer & swaps, followed by a pure‑swap hill‑climb.

All phases respect the user supplied time budget (default 1000 s) and
the routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (real‑to‑real)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a (needed for gradients)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective utilities
# ----------------------------------------------------------------------
def _evaluate_objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(conv) / (sum(a)²).
    No clipping – used only for internal bookkeeping.
    """
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return 2.0 * n * max_c / (s ** 2)


# ----------------------------------------------------------------------
# Candidate families
# ----------------------------------------------------------------------
def _generate_candidates(n: int) -> list[np.ndarray]:
    """Create a diverse pool of start vectors."""
    cand = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws with varied concentration
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(np.full(n, alpha)))

    # One‑sided exponentials (decay from the left)
    for gamma in (0.2, 0.5, 1.0, 2.0, 4.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        left /= left.sum()
        cand.append(left)
        cand.append(left[::-1])          # mirrored version (decay from the right)

    # Symmetric double‑exponential (both ends heavy)
    for gamma in (0.2, 0.5, 1.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Central Gaussian (single peak)
    x = np.arange(n, dtype=np.float64)
    mu = (n - 1) / 2.0
    for sigma in (0.05, 0.10, 0.15, 0.20, 0.30):
        shape = np.exp(-0.5 * ((x - mu) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Left‑only Gaussian (decay towards the right)
    for sigma in (0.05, 0.10, 0.15, 0.20):
        shape = np.exp(-0.5 * (x / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Symmetric Beta‑type (α=β>1 – centre peak)
    for alpha in (1.5, 2.0, 3.0, 4.0, 5.0):
        t = (np.arange(1, n + 1, dtype=np.float64) / n)
        shape = (t ** (alpha - 1)) * ((1.0 - t) ** (alpha - 1))
        shape /= shape.sum()
        cand.append(shape)

    # Beta‑type with α=β<1 – U‑shaped, mass at the edges
    for alpha in (0.2, 0.3, 0.5, 0.7):
        t = (np.arange(1, n + 1, dtype=np.float64) / n)
        shape = (t ** (alpha - 1)) * ((1.0 - t) ** (alpha - 1))
        shape /= shape.sum()
        cand.append(shape)

    # Simple triangular shape
    tri = np.minimum(np.arange(n) + 1, n - np.arange(n))
    tri = tri.astype(np.float64)
    tri /= tri.sum()
    cand.append(tri)

    # Positive half‑wave sinusoid
    sinus = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinus = np.maximum(sinus, 0.0)
    sinus /= sinus.sum()
    cand.append(sinus)

    # Random sparse spike patterns
    for k in (32, 64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        cand.append(a)

    # Interleaved permutation of every candidate (helps to spread mass)
    perm = (np.arange(n) * 2) % n
    interleaved = []
    for a in cand:
        b = np.empty_like(a)
        b[perm] = a
        interleaved.append(b)
    cand.extend(interleaved)

    # Ensure all candidates lie on the probability simplex
    out = []
    for a in cand:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        out.append(a)
    return out


# ----------------------------------------------------------------------
# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp).
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    # initialise logits for soft‑max parameterisation
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02          # increase λ slowly
    best_a = a0.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        val = _evaluate_objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the smooth‑max surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max  (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase smoothing parameter
        lam = min(lam * lam_factor, 1e7)

        # occasional jitter near the end of the budget
        if it % 4000 == 0 and time.time() > deadline - 60:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # gentle learning‑rate decay
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the exact max.
    Operates on the probability simplex (a normalised to sum = 1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _evaluate_objective(best_a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (strict equality to the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
    swap_prob: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search with mass‑transfer
    and occasional swaps.  Temperature decays linearly.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        if np.random.rand() < swap_prob:
            # ----- swap move -----
            i, j = np.random.randint(n, size=2)
            if i == j:
                it += 1
                continue
            a2 = a.copy()
            a2[i], a2[j] = a2[j], a2[i]
        else:
            # ----- mass‑transfer move -----
            i = np.random.randint(n)
            if a[i] < 1e-14:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * 0.4
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # accept uphill move with probability exp(-Δ/T)
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _permute_hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_swaps: int = 30_000,
) -> tuple[np.ndarray, float]:
    """
    Pure permutation hill‑climbing: repeatedly swap two entries,
    keeping the swap only if it does not increase the objective
    (with a tiny chance of accepting an uphill move).
    """
    a = a0.copy()
    val = _evaluate_objective(a, n, conv_fft_len)
    best_a = a.copy()
    best_val = val

    it = 0
    while time.time() < deadline and it < max_swaps:
        i, j = np.random.randint(n, size=2)
        if i == j:
            it += 1
            continue

        a[i], a[j] = a[j], a[i]          # perform swap
        val2 = _evaluate_objective(a, n, conv_fft_len)

        if val2 < best_val:
            best_val = val2
            best_a = a.copy()
            val = val2
        else:
            # occasional uphill acceptance to escape plateaus
            if np.random.rand() < 0.01:
                val = val2
            else:
                # revert swap
                a[i], a[j] = a[j], a[i]

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the given time budget and always returns the best sequence
    (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – 4096 points gives a good trade‑off between quality and speed
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1️⃣  Build a diversified pool of starter vectors
    # --------------------------------------------------------------
    pool = _generate_candidates(n)

    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 2️⃣  Quick evaluation of the raw pool and keep the K best
    # --------------------------------------------------------------
    evaluated = []
    for a in pool:
        val = _evaluate_objective(a, n, conv_fft_len)
        evaluated.append((val, a))
    evaluated.sort(key=lambda x: x[0])
    top_k = 10
    best_candidates = [a for _, a in evaluated[:top_k]]
    best_val, best_seq = evaluated[0][0], evaluated[0][1].copy()
    print(f"[Init] best of raw pool {best_val:.6f}")

    # --------------------------------------------------------------
    # 3️⃣  Short sub‑gradient (exponentiated‑gradient) polishing of the top K
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30 and best_seq is not None:
        per_candidate_dead = time.time() + remaining * 0.20 / len(best_candidates)
        for idx, a0 in enumerate(best_candidates):
            if time.time() >= deadline:
                break
            a_opt, val_opt = _subgradient_optimize_expgrad(
                a0, n, conv_fft_len, corr_fft_len, per_candidate_dead,
                max_iters=30_000, eta0=0.9
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[Short‑SG {idx}] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 4️⃣  Long Adam optimisation on the current incumbent
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Long‑Adam] start")
        a_opt, val_opt = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=1_200_000
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Long‑Adam] new best {val_opt:.6f}")

    # --------------------------------------------------------------
    # 5️⃣  Simulated‑annealing refinement
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 15:
        print("[Anneal] start")
        a_opt, val_opt = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=90_000, swap_prob=0.12
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Anneal] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 6️⃣  Final permutation‑only hill climbing
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Permute] start")
        a_opt, val_opt = _permute_hill_climb(
            best_seq, n, conv_fft_len, deadline, max_swaps=40_000
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[Permute] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 7️⃣  Safety fallback & final clipping
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
