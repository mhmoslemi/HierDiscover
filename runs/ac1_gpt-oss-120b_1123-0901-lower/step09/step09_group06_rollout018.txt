def _subgradient_descent(a0, n, conv_fft_len, deadline, max_iters=200000):
    a = a0.copy()
    best = a.copy()
    best_val = evaluate_sequence(a.tolist())
    step0 = 0.1  # initial step size
    for it in range(max_iters):
        if time.time() > deadline:
            break
        # compute convolution
        conv = _full_convolution(a, conv_fft_len)
        t_max = int(np.argmax(conv))
        # compute subgradient
        grad = np.zeros_like(a)
        # using vectorized: rev = np.zeros_like(a); valid indices:
        # for i where 0 <= t_max - i < n, grad[i] = 2*a[t_max - i]
        start = max(0, t_max - (n-1))
        end = min(n-1, t_max)
        # indices i from start..end correspond to j = t_max - i in [0,n-1]
        i_idx = np.arange(start, end+1, dtype=int)
        j_idx = t_max - i_idx
        grad[i_idx] = 2 * a[j_idx]
        # step size schedule
        eta = step0 / np.sqrt(it + 1)
        a = a - eta * grad
        # enforce non-negative and projection onto simplex
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s == 0:
            a = a0.copy()
        else:
            a /= s
        # evaluate
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best = a.copy()
        # optional early break if improvement < tolerance
    return best, best_val

        F(a) = 2·n·max(a * a) / (sum(a)²)

The implementation improves on the reference code in three ways:

import time

# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # degenerate case – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

# Families of deterministic candidates
def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _raised_cosine(n: int) -> np.ndarray:
    x = np.linspace(0, 1, n, dtype=np.float64)
    shape = 0.5 * (1 + np.cos(np.pi * (x - 0.5)))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _semicircle(n: int) -> np.ndarray:
    t = np.linspace(-1, 1, n, dtype=np.float64)
    shape = np.sqrt(np.maximum(0, 1 - t ** 2))
    shape /= shape.sum()
    return shape


def _linear_ramp(n: int, upward: bool = True) -> np.ndarray:
    """Monotone linear ramp (upward or downward)."""
    if upward:
        shape = np.arange(1, n + 1, dtype=np.float64)
    else:
        shape = np.arange(n, 0, -1, dtype=np.float64)
    shape /= shape.sum()
    return shape


def _two_level_step(n: int, k: int, ratio: float) -> np.ndarray:
    """
    First k positions have weight = ratio/k,
    remaining positions have weight = (1-ratio)/(n-k).
    """
    k = max(1, min(k, n - 1))
    ratio = np.clip(ratio, 0.0, 1.0)
    a = np.empty(n, dtype=np.float64)
    a[:k] = ratio / k
    if n - k > 0:
        a[k:] = (1.0 - ratio) / (n - k)
    else:
        a[k:] = 0.0
    return a


def _double_exponential(n: int, left_gamma: float, right_gamma: float) -> np.ndarray:
    """
    Two exponentials glued at the centre (left side uses left_gamma,
    right side uses right_gamma).  Both sides are normalised.
    """
    left = np.exp(-left_gamma * np.arange(n, dtype=np.float64))
    right = np.exp(-right_gamma * np.arange(n - 1, -1, -1, dtype=np.float64))
    shape = left + right
    shape /= shape.sum()
    return shape


def _sparse_spikes(n: int, k: int, rng: np.random.Generator) -> np.ndarray:
    """k spikes of equal weight."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        return a
    pos = rng.choice(n, size=k, replace=False)
    a[pos] = 1.0 / k
    return a

# Optimisers
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the log‑sum‑exp surrogate (identical to the
    reference implementation but with a slightly longer schedule)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02           # inverse temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for bookkeeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient wrt a (same as reference)
        n_corr = np.fft.rfft(a, conv_fft_len)
        rev_a = a[::-1]
        n_rev = np.fft.rfft(rev_a, conv_fft_len)
        cross = np.fft.irfft(n_corr * n_rev, conv_fft_len)[: n]
        grad_a = 2.0 * np.dot(w, cross)  # simplified, works because w is 1‑D

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # tiny random shake near the deadline
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Projected sub‑gradient descent on the *true* max‑objective.
    The gradient of max(a*a) w.r.t a is 2·a_rev shifted by the index
    of the maximal convolution entry.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    base_lr = 0.2

    while time.time() < deadline and it < max_iters:
        # convolution and index of the current worst entry
        conv = _full_convolution(a, conv_fft_len)
        t_max = int(conv.argmax())

        # sub‑gradient (2·a_{t_max‑i} where defined)
        grad = np.zeros_like(a)
        # i runs where 0 ≤ t_max‑i < n   →   i ∈ [t_max‑(n‑1) , t_max]
        lo = max(0, t_max - (n - 1))
        hi = min(n - 1, t_max)
        i_idx = np.arange(lo, hi + 1, dtype=int)
        j_idx = t_max - i_idx
        grad[i_idx] = 2.0 * a[j_idx]

        # step‑size schedule (classic 1/√t)
        eta = base_lr / np.sqrt(it + 1.0)
        a -= eta * grad

        # projection onto the simplex (non‑negativity + sum‑to‑1)
        a = np.clip(a, 0.0, None)
        s = a.sum()
        if s > 0:
            a /= s
        else:
            a = best_a.copy()   # fallback

        # evaluate
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # occasional small random perturbation (helps escaping plateaus)
        if it % 10000 == 0 and time.time() > deadline - 30:
            a += (np.random.rand(n) - 0.5) * 5e-3
            a = np.clip(a, 0.0, None)
            a_sum = a.sum()
            if a_sum > 0:
                a /= a_sum

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber moving a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        weight = a[i] * (np.random.rand() * 0.3)

        a2 = a.copy()
        a2[i] -= weight
        a2[j] += weight

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val


def _continuation_l_bfgs(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Continuation L‑BFGS on an increasing‑temperature log‑sum‑exp surrogate.
    (identical to the reference version – kept for robustness).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    best_logits = logits.copy()
    best_val = float("inf")
    alphas = [10.0, 100.0, 500.0, 2000.0, 8000.0, 20000.0]

    for alpha in alphas:
        if time.time() > deadline - 3:
            break

        def fun(x, n=n, conv_fft_len=conv_fft_len, alpha=alpha):
            mx = x.max()
            exp_x = np.exp(x - mx)
            a = exp_x / exp_x.sum()
            fa = np.fft.rfft(a, conv_fft_len)
            conv = np.fft.irfft(fa * fa, conv_fft_len)[: 2 * n - 1]
            max_c = conv.max()
            v = conv - max_c
            exp_alpha_v = np.exp(alpha * v)
            sum_exp = exp_alpha_v.sum()
            smooth_max = max_c + (1.0 / alpha) * np.log(sum_exp)

            # gradient (same as in reference)
            w = exp_alpha_v / sum_exp
            rev_a = a[::-1]
            fw = np.fft.rfft(w, conv_fft_len)
            fr = np.fft.rfft(rev_a, conv_fft_len)
            corr = np.fft.irfft(fw * fr, conv_fft_len)
            start = n - 1
            grad_a = 2.0 * corr[start:start + n]
            avg = np.dot(a, grad_a)
            grad_x = a * (grad_a - avg)

            obj = 2.0 * n * smooth_max
            return obj, grad_x

        remaining = deadline - time.time()
        max_iter = 150 if remaining > 30 else 50

        res = minimize(
            fun, logits, method="L-BFGS-B", jac=True, options={"maxiter": max_iter, "disp": False}
        )
        logits = res.x

        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        exact_val = evaluate_sequence(a.tolist())
        if exact_val < best_val:
            best_val = exact_val
            best_logits = logits.copy()
            print(f"[L‑BFGS α={alpha}] new best {exact_val:.6f}")

    a_best = np.exp(best_logits - np.max(best_logits))
    a_best /= a_best.sum()
    return a_best, best_val

# Main entry point
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with minimal evaluation score.
    The routine respects the time limit and always returns the best
    sequence it has discovered (as a plain Python list).
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size ----------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start handling (height_sequence_1 may be defined in the notebook)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() > 0:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of initial candidates
    pool = []

    # warm start and cyclic shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # deterministic families (same as reference)
    for alpha in (0.1, 0.3, 0.6, 1.0, 2.0, 5.0):
        pool.append(_random_dirichlet(n, alpha, rng))

    for gamma in np.logspace(-2, 1.2, 20):
        pool.append(_edge_exponential(n, gamma))

    for p in np.linspace(-8, -0.2, 20):
        pool.append(_edge_power_law(n, p))

    beta_params = [
        (0.5, 0.5), (0.5, 2.0), (2.0, 0.5), (2.0, 2.0),
        (5.0, 1.0), (1.0, 5.0), (0.2, 0.2), (0.2, 5.0), (5.0, 0.2),
        (0.1, 0.1)
    ]
    for a, b in beta_params:
        pool.append(_beta_shape(n, a, b))

    for sigma in (0.02, 0.05, 0.1, 0.2, 0.4):
        pool.append(_double_gaussian(n, sigma))

    pool.append(_sinusoid(n))
    pool.append(_uniform(n))
    pool.append(_raised_cosine(n))
    pool.append(_semicircle(n))

    # new families ----------------------------------------------------
    for k in (32, 64, 128, 256, 512):
        pool.append(_sparse_spikes(n, k, rng))

    # linear ramps
    pool.append(_linear_ramp(n, upward=False))
    pool.append(_linear_ramp(n, upward=True))

    # two‑level step families – a quick grid search
    for k in (64, 128, 256, 512, 1024):
        for r in (0.2, 0.35, 0.5, 0.65, 0.8):
            pool.append(_two_level_step(n, k, r))

    # double exponential
    for lg, rg in [(0.01, 3.0), (0.05, 1.0), (0.2, 0.2), (1.0, 0.01)]:
        pool.append(_double_exponential(n, lg, rg))

    # ------------------------------------------------------------------
    # Quick scan of the pool – keep the best few candidates
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    rng.shuffle(pool)

    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a /= s
        val = evaluate_sequence(a.tolist())
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] value {val:.6f}")

        if time.time() > deadline - 60:   # reserve ~1 min for deep phases
            break

    # ------------------------------------------------------------------
    # Phase 1 – aggressive sub‑gradient descent on the true max‑objective
    if best_seq is not None and time.time() < deadline - 300:
        # we will try a few restarts from the current best and from a fresh Dirichlet
        candidates = [best_seq]
        for _ in range(4):
            candidates.append(_random_dirichlet(n, 0.4, rng))
        for i, cand in enumerate(candidates):
            if time.time() > deadline - 250:
                break
            a_opt, val_opt = _subgradient_optimize(
                cand, n, conv_fft_len, deadline, max_iters=250_000
            )
            if val_opt < best_val:
                best_val = val_opt
                best_seq = a_opt.copy()
                print(f"[Subgrad restart {i}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – long Adam optimisation (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 150:
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_iters=400_000,
            lr_start=0.04,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam final] new best {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – continuation L‑BFGS refinement
    if best_seq is not None and time.time() < deadline - 60:
        a_lbgfs, val_lbgfs = _continuation_l_bfgs(
            best_seq, n, conv_fft_len, deadline
        )
        if val_lbgfs < best_val:
            best_val = val_lbgfs
            best_seq = a_lbgfs.copy()
            print(f"[L‑BFGS] new best {val_lbgfs:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – cheap random hill‑climber for the final seconds
    if best_seq is not None and time.time() < deadline - 3:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local final] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Fallback (should never happen)
    if best_seq is None:
        best_seq = _uniform(n)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Finished – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
