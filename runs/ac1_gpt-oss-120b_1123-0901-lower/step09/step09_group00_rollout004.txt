def _pgd_line_search(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=200000):
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2 * n * conv.max()
    for it in range(max_iters):
        if time.time() > deadline: break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # gradient as before
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Use backtracking line search to ensure objective decreases
        step = 1.0
        while step > 1e-8:
            a_new = a - step * grad
            a_new = np.maximum(a_new, 0.0)
            a_new = _project_simplex(a_new)
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2 * n * conv_new.max()
            if val_new <= val * (1 + 1e-6):
                a = a_new
                break
            step *= 0.5
        # if line search fails, break maybe
    return best_a, best_val
```

"""
Hybrid step‑function optimizer with deeper refinement.

* Warm‑starts from the previous solution (global variable `height_sequence_1`);
  otherwise falls back to the uniform distribution.
* Phase 1 – long Adam optimisation of the smooth‑max surrogate.
* Phase 2 – exponentiated‑gradient sub‑gradient phase on the exact max.
* Phase 3 – projected gradient descent on the exact max‑convolution.
* Phase 4 – optional up‑sampling to 8192 points and a second Adam pass.
* Phase 5 – simulated‑annealing local search.
* Phase 6 – cheap post‑processing: mixing with the uniform distribution and
  a light Gaussian blur.  All phases respect the global time budget (≈ 1000 s)
  and the best feasible sequence found is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector encountered and its exact surrogate objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector (still on the probability simplex) and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Uses a diminishing step size and keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _mix_with_uniform(a: np.ndarray, alpha: float) -> np.ndarray:
    """Convex combination a*alpha + (1‑alpha)*uniform, re‑normalised."""
    n = a.shape[0]
    uniform = np.full(n, 1.0 / n)
    mix = alpha * a + (1.0 - alpha) * uniform
    mix = np.maximum(mix, 0.0)
    s = mix.sum()
    if s < 1e-12:
        return a
    return mix / s


def _gaussian_blur(a: np.ndarray, sigma_frac: float) -> np.ndarray:
    """Light Gaussian blur (sigma as a fraction of the length)."""
    n = a.shape[0]
    sigma = sigma_frac * n
    if sigma < 1e-12:
        return a
    radius = int(3.0 * sigma)
    if radius < 1:
        return a
    x = np.arange(-radius, radius + 1)
    kernel = np.exp(-0.5 * (x / sigma) ** 2)
    kernel /= kernel.sum()
    blurred = np.convolve(a, kernel, mode='same')
    blurred = np.clip(blurred, 0.0, 1000.0)
    s = blurred.sum()
    if s < 1e-12:
        return a
    return blurred / s

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Primary grid size (initial optimisation)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous solution exists)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Starting point (warm if available, otherwise uniform)
    if warm is not None:
        a_start = warm.copy()
    else:
        a_start = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = a_start.copy()
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to update global best using the true evaluate_sequence()
    def _maybe_update(candidate: np.ndarray):
        nonlocal best_seq, best_val
        val = evaluate_sequence(candidate.tolist())
        if val < best_val:
            best_val = val
            best_seq = candidate.copy()
            print(f"[Update] new global best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max
    phase_dead = min(deadline, time.time() + 250.0)
    a_adam, _ = _adam_optimize(
        a_start, n, conv_fft_len, corr_fft_len, phase_dead, max_iters=800_000
    )
    _maybe_update(a_adam)

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient sub‑gradient
    if time.time() < deadline - 30:
        phase_dead = deadline - 30
        a_sub, _ = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, phase_dead,
            max_iters=600_000, eta0=0.9,
        )
        _maybe_update(a_sub)

    # ------------------------------------------------------------------
    # Phase 3 – Exact max‑convolution projected gradient descent
    if time.time() < deadline - 20:
        phase_dead = deadline - 20
        a_gd, _ = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, phase_dead,
            max_iters=400_000,
        )
        _maybe_update(a_gd)

    # ------------------------------------------------------------------
    # Phase 4 – Upsample to a finer grid (8192 points) if time permits
    if time.time() + 35 < deadline:
        try:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            phase_dead = min(deadline, time.time() + 30.0)
            a_big, _ = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                phase_dead, max_iters=600_000,
            )
            _maybe_update(a_big)

            # keep the up‑sampled version for the remaining stages
            best_seq = a_big
            n = n_big
            conv_fft_len = conv_fft_len_big
            corr_fft_len = corr_fft_len_big
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # Phase 5 – Simulated annealing for final local search
    if time.time() < deadline - 10:
        phase_dead = deadline - 10
        a_ann, _ = _anneal_optimize(
            best_seq, n, conv_fft_len, phase_dead, max_moves=120_000,
        )
        _maybe_update(a_ann)

    # ------------------------------------------------------------------
    # Phase 6 – Cheap post‑processing: mix with uniform & light blur
    if time.time() < deadline - 5:
        # mix with uniform
        for alpha in np.linspace(0.0, 1.0, 21):
            if time.time() >= deadline:
                break
            mixed = _mix_with_uniform(best_seq, alpha)
            _maybe_update(mixed)

        # Gaussian blur with a few bandwidths
        for sigma_frac in (0.005, 0.01, 0.02):
            if time.time() >= deadline:
                break
            blurred = _gaussian_blur(best_seq, sigma_frac)
            _maybe_update(blurred)

    # ------------------------------------------------------------------
    # Final safety clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
