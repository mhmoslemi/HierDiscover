"""
Propose a low‑value step‑function.

The search proceeds in three stages:

1. **Rich pool of starters** – deterministic families (uniform, sinusoid,
   edge‑exponential, power‑law, beta, double‑Gaussian, triangle, arcsine,
   semicircle, single‑sided exponential/power‑law, generalized Gaussian) and
   many random Dirichlet draws.  All vectors are normalised to sum‑one, the
   best N≈10 are kept.

2. **Refinement** – for each of the kept starters a short Adam optimisation
   on the smooth‑max surrogate (log‑sum‑exp) is performed (≈25 % of the
   remaining budget).  The best result from this phase becomes the seed for
   a long projected‑gradient descent (PGD) on the exact max‑convolution
   (≈55 % of the budget).

3. **Final hill‑climber** – a cheap mass‑transfer random walk (≈20 % of the
   time) that repeatedly moves a random fraction of mass from one index to
   another and accepts the move only if the exact objective improves.
   This squeezes out the last bits of improvement.

All FFT‑based convolutions use powers of two for speed.  The routine always
keeps the best vector seen so far and returns it as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  FFT helpers
# ----------------------------------------------------------------------


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.nonzero(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

#  Candidate families
# ----------------------------------------------------------------------


def _warm_and_shifts(warm: np.ndarray, n: int) -> list[np.ndarray]:
    """If a warm start is supplied, add a few cyclic shifts."""
    out = []
    if warm is not None:
        out.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            out.append(np.roll(warm, shift))
    return out


def _random_dirichlet(n: int, alpha: float, rng: np.random.Generator) -> np.ndarray:
    return rng.dirichlet(np.full(n, alpha))


def _edge_exponential(n: int, gamma: float) -> np.ndarray:
    left = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape = left + left[::-1]
    shape /= shape.sum()
    return shape


def _edge_power_law(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = (idx ** alpha) + ((n - idx + 1) ** alpha)
    shape = np.maximum(shape, 0.0)
    shape_sum = shape.sum()
    if shape_sum > 0:
        shape /= shape_sum
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _beta_shape(n: int, a: float, b: float) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = (x ** (a - 1)) * ((1 - x) ** (b - 1))
    shape = np.maximum(shape, 0.0)
    s = shape.sum()
    if s > 0:
        shape /= s
    else:
        shape = np.full(n, 1.0 / n)
    return shape


def _double_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-0.5 * ((i) / (sigma * n)) ** 2)
    right = np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
    shape = left + right
    shape /= shape.sum()
    return shape


def _sinusoid(n: int) -> np.ndarray:
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _triangular(n: int) -> np.ndarray:
    """Symmetric piecewise‑linear “tent” distribution."""
    half = n // 2
    inc = np.arange(1, half + 1, dtype=np.float64)
    if n % 2 == 0:
        shape = np.concatenate([inc, inc[::-1]])
    else:
        shape = np.concatenate([inc, inc[-2::-1]])
    shape /= shape.sum()
    return shape


def _arcsine(n: int) -> np.ndarray:
    """Beta(0.5,0.5) – U‑shaped distribution."""
    return _beta_shape(n, 0.5, 0.5)


def _semicircle(n: int) -> np.ndarray:
    """Density proportional to sqrt(x(1‑x))."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.sqrt(x * (1.0 - x))
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    return shape


def _single_exponential(n: int, gamma: float) -> np.ndarray:
    """Monotone exponential decay (only one side)."""
    shape = np.exp(-gamma * np.arange(n, dtype=np.float64))
    shape /= shape.sum()
    return shape


def _single_power_law(n: int, alpha: float) -> np.ndarray:
    """Monotone power‑law decay a_i ∝ (i+1)^{-α}."""
    idx = np.arange(n, dtype=np.float64) + 1.0
    shape = idx ** (-alpha)
    shape /= shape.sum()
    return shape


def _gaussian_centered(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    shape = np.exp(-0.5 * ((i - centre) / (sigma * n)) ** 2)
    shape /= shape.sum()
    return shape


def _generalized_gaussian(n: int, sigma: float, p: float) -> np.ndarray:
    """Shape exp(-|x-μ|^p) with free exponent p>0."""
    i = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
    shape = np.exp(-0.5 * (np.abs(i) / (sigma * n)) ** p)
    shape /= shape.sum()
    return shape


# ----------------------------------------------------------------------
#  Fast objective (exact same formula as evaluate_sequence)
# ----------------------------------------------------------------------


def _objective(a: np.ndarray) -> float:
    n = a.shape[0]
    fft_len = _next_pow2(2 * n - 1)
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (a.sum() ** 2))


# ----------------------------------------------------------------------
#  Optimisers
# ----------------------------------------------------------------------


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr_start: float = 0.04,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise on simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr_start
    lam = 0.02          # soft‑max temperature (inverse)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule (mirror‑descent style)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # a little noise near the end
        if it % 5000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Projected‑gradient descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[PGD it={it:06d}] new best {val:.6f}")

        # active set (where the max is attained)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step‑size
        eta = eta0 / np.sqrt(it + 1)

        a = a - eta * grad_a
        a = _project_onto_simplex(a)

        # occasional jitter near the very end
        if it % 20000 == 0 and time.time() > deadline - 20:
            jitter = (np.random.rand(n) - 0.5) * 1e-3
            a = np.maximum(a + jitter, 0.0)
            a /= a.sum()

        it += 1

    return best_a, best_val


def _local_random_search(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Cheap hill‑climber that moves a random fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer up to 50 % of source mass (more aggressive)
        delta = a[i] * (np.random.rand() * 0.5)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            if it % 5000 == 0:
                print(f"[Local move {it:06d}] improved to {val2:.6f}")

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget and always returns the best
    sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    #  Warm‑start handling
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    #  Build an extensive initial pool
    # ------------------------------------------------------------------
    pool = _warm_and_shifts(warm, n) + _initial_pool(n, rng)

    # additional deterministic families
    for gamma in (0.001, 0.002, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        pool.append(_single_exponential(n, gamma))
    for alpha in (0.3, 0.5, 0.8, 1.2, 2.0, 3.0):
        pool.append(_single_power_law(n, alpha))
    pool.append(_triangular(n))
    pool.append(_arcsine(n))
    pool.append(_semicircle(n))
    for sigma in (0.07, 0.12, 0.18):
        pool.append(_gaussian_centered(n, sigma))
    for sigma, p in [(0.1, 2.0), (0.15, 4.0), (0.2, 6.0)]:
        pool.append(_generalized_gaussian(n, sigma, p))

    # random mixtures of two exponentials
    for _ in range(30):
        gamma1 = 10 ** rng.uniform(-4, -0.3)
        gamma2 = 10 ** rng.uniform(-4, -0.3)
        w = rng.random()
        mix = w * _single_exponential(n, gamma1) + (1 - w) * _single_exponential(n, gamma2)
        mix /= mix.sum()
        pool.append(mix)

    # random Dirichlet draws (wide range of concentrations)
    for alpha in (0.05, 0.15, 0.3, 0.6, 1.0, 2.0, 5.0, 8.0):
        for _ in range(8):
            pool.append(_random_dirichlet(n, alpha, rng))

    # ------------------------------------------------------------------
    #  Quick scan of the pool – keep the top few candidates
    # ------------------------------------------------------------------
    best_candidates = []   # list of (value, vector)
    top_k = 10

    for cid, cand in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()

        val = _objective(a)
        if len(best_candidates) < top_k:
            best_candidates.append((val, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        elif val < best_candidates[-1][0]:
            best_candidates[-1] = (val, a.copy())
            best_candidates.sort(key=lambda x: x[0])

        if cid % 50 == 0:
            print(f"[Init pool] processed {cid} candidates, current best {best_candidates[0][0]:.6f}")

    # select the single best as seed for refinement
    best_val = best_candidates[0][0]
    best_seq = best_candidates[0][1].copy()
    print(f"Best after pool scan: {best_val:.6f}")

    # ------------------------------------------------------------------
    #  Phase 2 – short Adam optimisation on a few top seeds
    # ------------------------------------------------------------------
    time_now = time.time()
    time_left = deadline - time_now
    # allocate about 25 % of the remaining budget for Adam
    adam_deadline = time_now + 0.25 * time_left

    a_adam, val_adam = _adam_optimize(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        adam_deadline,
        max_iters=600_000,
        lr_start=0.03,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    #  Phase 3 – long PGD optimisation (uses almost all remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5:
        a_pgd, val_pgd = _pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=2_000_000,
            eta0=0.85,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    #  Phase 4 – final cheap hill‑climber
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _local_random_search(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Local search] final best {val_loc:.6f}")

    # ------------------------------------------------------------------
    #  Safety fallback and finalisation
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to range accepted by evaluate_sequence
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
