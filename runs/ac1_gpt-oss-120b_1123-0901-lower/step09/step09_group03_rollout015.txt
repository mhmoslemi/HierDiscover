"""
Hybrid stochastic–gradient optimizer for the step‑function objective

Goal: minimise  F(a)=2·n·max(a∗a)/(∑a)²  for a non‑negative vector a.
The implementation consists of three stages:

1.  **Rich initial pool** – a large collection of structured families
    (asymmetric double‑exponential, Beta‑shapes, powered sinusoid,
    random Dirichlet etc.).  All candidates are normalised to sum‑1 and
    the best few (by a cheap FFT‑based evaluation) are kept.

2.  **Gradient‑based refinement** – an Adam optimiser on a smooth
    log‑sum‑exp surrogate followed by an exact sub‑gradient method
    (exponentiated‑gradient / mirror descent) on the true non‑smooth
    objective.  The step‑size schedules are chosen to be aggressive
    early on and to decay slowly so that we escape shallow plateaus.

3.  **Targeted simulated annealing** – a pair‑wise move that always
    attacks the current peak of the convolution, with a classic
    Metropolis acceptance schedule.

The routine respects a user‑supplied time budget (default 1000 s) and
always returns the best sequence found as a plain Python list.
All entries are non‑negative and clipped to the range accepted by
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length = 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a) giving ∂/∂a_i Σ_k w_k (a∗a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _gen_double_exponential(n: int, gl: float, gr: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gl * i) + np.exp(-gr * (n - 1 - i))
    return a


def _gen_beta_shape(n: int, p: float, q: float) -> np.ndarray:
    x = (np.arange(n) + 0.5) / n
    a = np.power(x, p - 1.0) * np.power(1.0 - x, q - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    return a


def _gen_sine_power(n: int, alpha: float, eps: float = 1e-4) -> np.ndarray:
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0) + eps
    a = s ** alpha
    return a


def _initial_candidate_pool(n: int) -> list[np.ndarray]:
    """Create a diversified set of low‑dimensional candidates."""
    cand: list[np.ndarray] = []

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Double‑exponential families (asymmetric)
    for gl in (0.001, 0.01, 0.1, 0.5, 1.0):
        for gr in (0.001, 0.01, 0.1, 0.5, 1.0):
            cand.append(_gen_double_exponential(n, gl, gr))

    # Beta‑shapes (two‑parameter power law)
    for p in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
        for q in (0.5, 0.8, 1.0, 1.2, 1.5, 2.0, 3.0):
            cand.append(_gen_beta_shape(n, p, q))

    # Powered sinusoid (controls edge‑vs‑centre mass)
    for alpha in np.linspace(-2.0, 2.0, 9):
        cand.append(_gen_sine_power(n, alpha))

    # Random Dirichlet draws with various concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Shifted copies of a few base shapes to increase diversity
    base = cand[:6]  # first few are already diverse
    for shift in (n // 8, n // 4, n // 2):
        for b in base:
            cand.append(np.roll(b, shift))

    # Ensure no NaN / inf and normalise
    cleaned = []
    for a in cand:
        a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            continue
        a /= a.sum()
        cleaned.append(a)
    return cleaned

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max surrogate  log‑sum‑exp(λ·(a∗a)).
    Returns the best vector seen (normalised to sum = 1) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # interior of the simplex
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.02                # start temperature for smooth‑max
    lam_factor = 1.018

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # Softmax gives a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (fast because sum=1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the smooth surrogate (w.r.t a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent style gradient for logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # raise smooth‑max temperature
        lam = min(lam * lam_factor, 1e7)

        # occasional tiny random shake towards the end
        if it % 4000 == 0 and time.time() > deadline - 30:
            logits += (np.random.rand(n) - 0.5) * 2e-2

        # learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent using exponentiated‑gradient updates.
    Operates directly on the simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (all indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _pair_targeted_anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Pair‑wise targeted annealing.
    At each move we locate the convolution peak kmax, pick the pair
    (i,j) with i+j=kmax that contributes most, and move a tiny amount
    of mass away from that pair.  Metropolis acceptance with a linear
    temperature schedule is used.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    T_start = 0.06
    T_end = 5e-6
    move = 0

    while time.time() < deadline and move < max_moves:
        T = T_end + (T_start - T_end) * (1.0 - move / max_moves)

        # recompute convolution and peak
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        kmax = int(conv.argmax())

        # find the most damaging pair (i, j) with i+j=kmax
        i_start = max(0, kmax - (n - 1))
        i_end = min(n - 1, kmax)
        idx = np.arange(i_start, i_end + 1)
        prod = a[idx] * a[kmax - idx]
        i_local = int(np.argmax(prod))
        i_src = idx[i_local]
        j_src = kmax - i_src

        a2 = a.copy()
        delta = min(a2[i_src], a2[j_src]) * 0.07  # move 7 % of the smaller side
        if delta < 1e-14:
            move += 1
            continue

        # move the little chunk to two random fresh locations
        t1, t2 = np.random.randint(n), np.random.randint(n)
        while t1 == i_src or t1 == j_src:
            t1 = np.random.randint(n)
        while t2 == i_src or t2 == j_src or t2 == t1:
            t2 = np.random.randint(n)

        a2[i_src] -= delta
        a2[j_src] -= delta
        a2[t1] += delta
        a2[t2] += delta

        # project back to simplex (simple normalisation, zeros stay zero)
        a2 = np.maximum(a2, 0.0)
        s = a2.sum()
        if s == 0:
            a2 = np.full(n, 1.0 / n)
        else:
            a2 /= s

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        move += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _objective_on_simplex(a: np.ndarray, conv_fft_len: int, n: int) -> float:
    """Fast evaluation for a vector already normalised to sum = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  The routine respects the given time budget
    and always returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – a moderate length keeps the inner loops cheap
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Phase I – generate a rich pool and keep the very best few
    pool = _initial_candidate_pool(n)

    print(f"[Setup] generated {len(pool)} structured candidates.")
    best_seq = None
    best_val = float("inf")
    top_k = 5
    top_candidates: list[tuple[np.ndarray, float]] = []

    for a in pool:
        val = _objective_on_simplex(a, conv_fft_len, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool] new best {val:.6f}")

        # maintain a small heap of the best few
        if len(top_candidates) < top_k:
            top_candidates.append((a.copy(), val))
            top_candidates.sort(key=lambda kv: kv[1])
        elif val < top_candidates[-1][1]:
            top_candidates[-1] = (a.copy(), val)
            top_candidates.sort(key=lambda kv: kv[1])

    # --------------------------------------------------------------
    # Phase II – refine the best candidates with a two‑stage gradient
    # approach.  We allocate the majority of the remaining budget here.
    remaining = deadline - time.time()
    if remaining <= 0:
        remaining = 0.1
    # split the remaining time roughly 60 % Adam, 30 % sub‑gradient,
    # 10 % targeted annealing.
    time_adam = remaining * 0.55
    time_sub  = remaining * 0.30
    time_anneal = remaining * 0.15

    # take the single currently best candidate as starting point
    a_start = best_seq if best_seq is not None else np.full(n, 1.0 / n)

    # ---- Adam on smooth surrogate -------------------------------------------------
    ad_dead = time.time() + time_adam
    a_adam, val_adam = _adam_optimize(
        a_start,
        n,
        conv_fft_len,
        corr_fft_len,
        ad_dead,
        max_iters=300_000,
    )
    if val_adam < best_val:
        best_val = val_adam
        best_seq = a_adam.copy()
        print(f"[Refine‑Adam] improved to {val_adam:.6f}")

    # ---- Exact sub‑gradient (exponentiated‑gradient) ------------------------------
    sub_dead = time.time() + time_sub
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_seq,
        n,
        conv_fft_len,
        corr_fft_len,
        sub_dead,
        max_iters=600_000,
        eta0=0.85,
    )
    if val_sub < best_val:
        best_val = val_sub
        best_seq = a_sub.copy()
        print(f"[Refine‑Subgrad] improved to {val_sub:.6f}")

    # ---- Pair‑wise targeted annealing --------------------------------------------
    anneal_dead = time.time() + time_anneal
    a_pair, val_pair = _pair_targeted_anneal_optimize(
        best_seq,
        n,
        conv_fft_len,
        anneal_dead,
        max_moves=150_000,
    )
    if val_pair < best_val:
        best_val = val_pair
        best_seq = a_pair.copy()
        print(f"[Refine‑Anneal] improved to {val_pair:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling phase – if still time left, try a finer grid.
    if time.time() + 30 < deadline:
        n_big = 8192
        conv_fft_len_big = _next_pow2(2 * n_big - 1)
        corr_fft_len_big = _next_pow2(3 * n_big - 2)

        print("[Upsample] interpolating to larger grid")
        x_old = np.linspace(0, 1, n, endpoint=False)
        x_new = np.linspace(0, 1, n_big, endpoint=False)
        a_big = np.interp(x_new, x_old, best_seq)
        a_big = np.maximum(a_big, 0.0)
        a_big /= a_big.sum()

        # short Adam on bigger grid
        sub_dead_big = min(deadline, time.time() + 8.0)
        a_big_adam, _ = _adam_optimize(
            a_big,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            sub_dead_big,
            max_iters=180_000,
        )

        # longer sub‑gradient refinement on large grid
        a_big_ref, val_big_ref = _subgradient_optimize_expgrad(
            a_big_adam,
            n_big,
            conv_fft_len_big,
            corr_fft_len_big,
            deadline,
            max_iters=180_000,
            eta0=0.8,
        )
        if val_big_ref < best_val:
            best_val = val_big_ref
            best_seq = a_big_ref.copy()
            print(f"[Upsample‑Refine] improved best to {val_big_ref:.6f}")

    # --------------------------------------------------------------
    # safety: guarantee a valid return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final clipping to the range accepted by evaluate_sequence()
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
