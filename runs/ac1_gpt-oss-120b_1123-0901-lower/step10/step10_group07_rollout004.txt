def solve_lp_for_b(a, M, active_indices):
    n = len(a)
    rows = []
    for k in active_indices:
        # Build row vector
        row = np.zeros(n)
        # i range where a_{k-i} exists
        i_min = max(0, k - (n-1))
        i_max = min(n-1, k)
        # compute indices
        # for i in [i_min .. i_max], a_{k-i} exists
        # we can vectorize:
        idx = np.arange(i_min, i_max+1)
        row[idx] = a[k - idx]
        rows.append(row)
    A = np.vstack(rows) if rows else np.empty((0,n))
    # Linear program: minimize -sum b_i => maximize sum b_i
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=np.full(len(active_indices), M), bounds=bounds, method='highs')
    if res.success:
        b = res.x
    else:
        # fallback use original a
        b = a.copy()
    return b
```

def _solve_lp_active(a, M, active):
    n = a.shape[0]
    if active.size == 0:
        return None
    rows = []
    for k in active:
        row = np.zeros(n)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1)
        row[i] = a[k - i]
        rows.append(row)
    A = np.vstack(rows)  # shape (len(active), n)
    # objective maximize sum(b) -> minimize negative sum
    c = -np.ones(n)
    bounds = [(0, None)] * n
    # Use scipy's high-performance LP solver
    res = linprog(c, A_ub=A, b_ub=np.full(len(active), M), bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: return a (no change)
        return None
    return res.x
```

"""
Active‑set linear‑programming optimizer for the max–convolution objective.

The code starts from a feasible non‑negative step‑function (the previous
best `height_sequence_1` if available, otherwise a simple exponential
profile).  In each outer iteration it

1. computes the current peak `M = max(a*a)`,
2. builds a small linear program that maximises the total mass of a new
   vector `b` while respecting the *active* convolution constraints
   `(a ∗ b)_k ≤ M` for those indices `k` where the peak is (almost) attained,
3. rescales `b` to a probability vector and accepts it if it strictly
   reduces the objective `2·n·max(a*a)/(sum a)²`,
4. repeats the process until no further improvement is found or the time
   budget expires.

Finally a short exponentiated‑gradient (mirror‑descent on the simplex) run
polishes the result.  The routine always returns a valid list that can be
fed to `evaluate_sequence`.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the sub‑gradient of the max‑convolution.
    Result length = len(a); corresponds to the derivative of
    max_i (w_i·(a*a)_i) w.r.t. a when w is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective -------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (non‑negative a)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Linear‑programming step ----------------------------------------------

def _solve_lp_active(a: np.ndarray, M: float, active: np.ndarray) -> np.ndarray | None:
    """
    Solve  max  Σ b_i   s.t.  (a∗b)_k ≤ M  for all k∈active,  b ≥ 0.
    Returns the optimal b (may be dense) or None on failure.
    """
    n = a.shape[0]
    if active.size == 0:
        return None

    # Build the inequality matrix A (rows = active constraints)
    rows = []
    for k in active:
        row = np.zeros(n, dtype=np.float64)
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i = np.arange(i_start, i_end + 1, dtype=np.int64)
        # a_{k-i} is the coefficient of b_i in (a∗b)_k
        row[i] = a[k - i]
        rows.append(row)

    A = np.vstack(rows)                     # shape (|active|, n)
    c = -np.ones(n, dtype=np.float64)       # maximise Σb_i  → minimise -Σb_i
    bounds = [(0.0, None)] * n               # b_i ≥ 0

    # Solve with HiGHS (fast, sparse‑aware).  Small problems are dense anyway.
    res = linprog(c, A_ub=A, b_ub=np.full(active.shape[0], M),
                  bounds=bounds, method='highs', options={'presolve': True})

    if not res.success:
        return None
    return res.x


# ----------------------------------------------------------------------
# Exponentiated‑gradient (sub‑gradient) refinement ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex (exponentiated‑gradient) using the exact
    max‑convolution sub‑gradient.  Stops when the deadline is reached.
    Returns the best vector seen and its objective value.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set where the convolution attains its maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """

    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = a / a_sum                       # normalise to sum = 1

    best_a = a.copy()
    best_val = _objective(best_a, n)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # 1) Active‑set LP improvement loop
    #
    # Keep iterating as long as we see a strict improvement and we stay

    max_outer_iters = 50
    for outer in range(max_outer_iters):
        if time.time() >= deadline:

        M = conv.max()
        # indices where the convolution is (almost) maximal

        active = np.where(conv >= M - eps_active)[0]

        if active.size > max_active:

        if b is None:

            break

        # Normalise the LP solution to a probability vector
        b_sum = b.sum()
        if b_sum <= 0:

            break
        b_norm = b / b_sum

        # Evaluate the candidate
        cand_val = _objective(b_norm, n)

            best_val = cand_val
            best_a = b_norm
            print(f"[LP] improved objective to {best_val:.6f} (outer {outer})")
            # continue with the new point
        else:
            # No improvement – stop the LP refinement
            print("[LP] no further improvement, exiting LP loop")
            break

    # ------------------------------------------------------------------
    # 2) Short exponentiated‑gradient polishing (optional, time‑aware)
    if time.time() < deadline - 5.0:

        a_polished, val_polished = _subgradient_optimize_expgrad(

            n,
