def _subgradient_optimize_proj(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters):
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    for it in range(max_iters):
        if time.time() > deadline: break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # identify active indices
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # subgradient step
        step = 0.5 / np.sqrt(it + 1)  # step size
        a = a - step * grad
        a = np.clip(a, 0.0, None)
        a_sum = a.sum()
        if a_sum == 0: a = np.random.dirichlet(np.ones(n))
        else: a = a / a_sum
```

"""
Propose a step‑function (non‑negative sequence) with a low value of
evaluate_sequence.  The approach is a multi‑stage optimisation on the
probability simplex (∑a = 1).  It consists of:

1. Warm‑start from the previous best solution (height_sequence_1) or a
   smooth Gaussian shape.
2. An Adam optimiser on a smooth log‑sum‑exp approximation of the
   convolution maximum.
3. A long exponentiated‑gradient (sub‑gradient) phase that directly
   minimises the exact max‑convolution.
4. A few passes of projected gradient descent and simulated‑annealing
   for fine‑tuning.
5. If time permits, up‑sample the current best to a finer grid
   (8192 points) and repeat the Adam + sub‑gradient refinement.

All stages stop when the global deadline (≈ 1000 s) is reached.  The
best sequence encountered during the whole run is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions -----------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories ---------------------------------------------------

def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth log‑sum‑exp surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                                    # a bit larger than before
    lam = 0.05                                    # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 60_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 6_000_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found before the allotted time expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Base problem size
    n0 = 4096
    conv_fft_len0 = _next_pow2(2 * n0 - 1)
    corr_fft_len0 = _next_pow2(3 * n0 - 2)

    # ------------------------------------------------------------------
    # Warm start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n0:
            warm = warm[:n0]
        elif warm.shape[0] < n0:
            warm = np.pad(warm, (0, n0 - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    if warm is not None:
        a0 = warm.copy()
    else:
        # a smooth Gaussian works well as a generic start
        a0 = _make_gaussian(n0, sigma=0.22)

    best_seq = a0.copy()
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase 1 – Adam (smooth max)
    t_now = time.time()
    time_for_adam = min(180.0, deadline - t_now - 500.0)  # keep room for later phases
    if time_for_adam > 5.0:
        sub_dead = min(deadline, t_now + time_for_adam)
        a_adam, val_adam = _adam_optimize(
            a0, n0, conv_fft_len0, corr_fft_len0, sub_dead,
            max_iters=10_000_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase 1] Adam best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – Exponentiated‑gradient sub‑gradient (exact max)
    if time.time() < deadline - 10.0:
        sub_dead = deadline - 300.0  # reserve ~5 min for later fine passes
        sub_dead = min(deadline, sub_dead)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n0, conv_fft_len0, corr_fft_len0, sub_dead,
            max_iters=15_000_000, eta0=0.65,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase 2] Sub‑grad best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Projected gradient descent (fine‑tune)
    if time.time() < deadline - 8.0:
        sub_dead = deadline - 200.0
        sub_dead = min(deadline, sub_dead)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n0, conv_fft_len0, corr_fft_len0, sub_dead,
            max_iters=1_200_000,
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Phase 3] GD best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Simulated annealing (helps escape tiny plateaus)
    if time.time() < deadline - 5.0:
        sub_dead = deadline - 100.0
        sub_dead = min(deadline, sub_dead)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n0, conv_fft_len0, sub_dead,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase 4] Anneal best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – Upsample to a finer grid (8192) and repeat Adam+Subgrad
    if time.time() + 30 < deadline:
        try:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation to the finer grid
            x_small = np.linspace(0, 1, n0, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_small, best_seq)
            a_big = np.clip(a_big, 0.0, 1000.0)
            if a_big.sum() < 1e-12:
                a_big = _make_gaussian(n_big, sigma=0.18)
            else:
                a_big = a_big / a_big.sum()

            # Adam on the big grid
            t_big_start = time.time()
            time_big_adam = min(120.0, deadline - t_big_start - 300.0)
            if time_big_adam > 5.0:
                sub_dead_big = min(deadline, t_big_start + time_big_adam)
                a_big_adam, val_big_adam = _adam_optimize(
                    a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                    sub_dead_big, max_iters=12_000_000,
                )
                if val_big_adam < best_val:
                    best_val, best_seq = val_big_adam, a_big_adam.copy()
                    print(f"[Phase 5‑1] Big Adam best {best_val:.6f}")

            # Sub‑gradient on the big grid
            if time.time() < deadline - 5.0:
                sub_dead_big = deadline - 60.0
                sub_dead_big = min(deadline, sub_dead_big)
                a_big_sub, val_big_sub = _subgradient_optimize_expgrad(
                    best_seq, n_big, conv_fft_len_big, corr_fft_len_big,
                    sub_dead_big, max_iters=15_000_000, eta0=0.55,
                )
                if val_big_sub < best_val:
                    best_val, best_seq = val_big_sub, a_big_sub.copy()
                    print(f"[Phase 5‑2] Big Sub‑grad best {best_val:.6f}")

        except Exception as exc:
            print("[Upsample] exception:", exc)

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n0, 1.0 / n0, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
