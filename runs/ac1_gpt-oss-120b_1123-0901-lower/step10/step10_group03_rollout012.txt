def subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, iter_limit):
    a = np.maximum(a0, 0.0)
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    while time.time() < deadline and it < iter_limit:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # we could compute any early break if max_c < best_M?
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Subgrad restart it={it} new best {val:.6f}")
        # gradient for active indices:
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient: 2 * cross-correlation of w with a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)
        # multiply by exp(-eta * grad)
        a = a * np.exp(-eta * grad)
        # renorm
        suma = a.sum()
        if suma <= 0 or np.isnan(suma):
            # reset to random dirichlet
            a = np.random.dirichlet(np.ones(n))
            suma = a.sum()
        a /= suma
        it += 1
    return best_a, best_val
```

"""
Propose a step‑function (non‑negative sequence) with a minimal score.

Algorithm
---------
1. **Broad parametric sweep** – evaluate many analytic families
   (exponential, power‑law, stretched‑exponential, Gamma‑like,
   Beta, logistic, two–sided mixtures …).  The best candidate is kept.

2. **Fine‑tuning of the most promising families** – a cheap
   differential‑evolution (DE) optimisation is run on the
   low‑dimensional parameter space (e.g. α for an exponential,
   (α,β) for a stretched‑exponential, (α,p) for a power–exponential
   mixture).  This often discovers a slightly better shape.

3. **Exploratory multi‑restart exponentiated‑gradient (sub‑gradient)
   optimisation** – from the current best vector we generate several
   random starts (Dirichlet draws, perturbed best, random mixture
   families).  Each start runs a modest number of EG updates
   (≈2·10⁵ iterations) and the best result is retained.

4. **Intensive final EG run** – a long exponentiated‑gradient run from
   the best candidate obtained in step 3 uses the remaining time budget.

5. **Optional short annealing** – a cheap simulated‑annealing pass
   performs tiny mass transfers between coordinates.

All phases respect the given time budget (default 1000 s) and the routine
always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for the exact max‑convolution sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_two_sided(alpha_l: float, alpha_r: float, w: float, n: int) -> np.ndarray:
    left = _candidate_left_exponential(alpha_l, n)
    right = _candidate_left_exponential(alpha_r, n)[::-1]
    a = w * left + (1.0 - w) * right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent) on the exact max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # sparse logging to avoid overhead
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        # exponentiated‑gradient update with safe clipping
        update = np.exp(np.clip(-eta * grad, -700, 700))
        a = a * update

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing local search.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper for DE ---------------------------------------------------------

def _de_optimize(func, bounds, maxiter, popsize, seed):
    """Run differential evolution with a limited budget."""
    result = differential_evolution(
        func,
        bounds,
        maxiter=maxiter,
        popsize=popsize,
        seed=seed,
        polish=False,
        init='latinhypercube',
        disp=False,
    )
    return result.x, result.fun

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # Length – 4096 works well in the reference implementation.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Utility to update the current best
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (same families as the reference)
    for p in np.linspace(-6.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-4, 2.5, 35):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    for alpha in np.logspace(-4, 2.0, 30):
        _might = _candidate_left_exponential(alpha, n)
        _maybe_update(_might)
    for p in np.linspace(-6.0, 2.0, 30):
        _maybe_update(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 12):
        for alpha in np.logspace(-4, 1.5, 25):
            _maybe_update(_candidate_left_stretched(alpha, beta, n))

    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            _maybe_update(_candidate_gamma(shape, scale, n))

    a_pars = np.logspace(-2, 0.5, 12)
    b_pars = np.logspace(0, 1.0, 12)
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3, -0.2, 15):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # Additional random families for extra coverage
    for _ in range(200):
        # random left‑exponential
        alpha = 10 ** np.random.uniform(-4, 2)
        _maybe_update(_candidate_left_exponential(alpha, n))

        # random logistic
        beta = np.random.uniform(0.01, 5.0)
        mu = np.random.uniform(0, n - 1)
        _maybe_update(_candidate_logistic(beta, mu, n))

        # two‑sided mixture
        a1 = 10 ** np.random.uniform(-4, 2)
        a2 = 10 ** np.random.uniform(-4, 2)
        w = np.random.rand()
        _maybe_update(_candidate_two_sided(a1, a2, w, n))

        # random spike patterns
        k = np.random.choice([8, 16, 32, 64, 128, 256])
        _maybe_update(_candidate_spike(k, n, random=True))

    # Warm‑start from a previously‑found sequence if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        w = np.clip(w, 0.0, None)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after sweep: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) DE fine‑tuning of the most promising low‑dimensional families
    #    (run only if we still have enough time)
    def _de_wrapper_exponential(x):
        a = _candidate_left_exponential(x[0], n)
        return _objective(a, n)

    def _de_wrapper_stretched(x):
        a = _candidate_stretched(x[0], x[1], n)
        return _objective(a, n)

    def _de_wrapper_power_exponential(x):
        # a_i ∝ (i+1)^p * exp(-α·i)
        i = np.arange(n, dtype=np.float64)
        p, alpha = x[0], x[1]
        a = ((i + 1) ** p) * np.exp(-alpha * i)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return _objective(a, n)

    de_time_remaining = deadline - time.time() - 30.0
    if de_time_remaining > 5.0:
        # exponential (single parameter)
        bounds_exp = [(1e-6, 300.0)]
        x_opt, f_opt = _de_optimize(_de_wrapper_exponential,
                                    bounds_exp, maxiter=30, popsize=10,
                                    seed=seed + 1)
        _maybe_update(_candidate_left_exponential(x_opt[0], n))

        # stretched exponential (α,β)
        bounds_stretch = [(1e-6, 50.0), (0.1, 5.0)]
        x_opt, f_opt = _de_optimize(_de_wrapper_stretched,
                                    bounds_stretch, maxiter=30, popsize=10,
                                    seed=seed + 2)
        _maybe_update(_candidate_stretched(x_opt[0], x_opt[1], n))

        # power‑exponential mixture (p,α)
        bounds_powexp = [(-10.0, 10.0), (1e-6, 200.0)]
        x_opt, f_opt = _de_optimize(_de_wrapper_power_exponential,
                                    bounds_powexp, maxiter=30, popsize=10,
                                    seed=seed + 3)
        # build the candidate manually
        i = np.arange(n, dtype=np.float64)
        p, alpha = x_opt[0], x_opt[1]
        a = ((i + 1) ** p) * np.exp(-alpha * i)
        a = np.maximum(a, 0.0)
        a /= a.sum()
        _maybe_update(a)

    print(f"[DE] best after DE fine‑tuning: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Exploratory multi‑restart sub‑gradient optimisation
    #    (allocate roughly 30 % of the remaining budget)
    explore_deadline = deadline - 250.0  # keep ~250 s for the final long run
    restart_cnt = 0
    while time.time() < explore_deadline:
        restart_cnt += 1
        # pick a start point
        if best_seq is not None and np.random.rand() < 0.5:
            # perturb current best slightly
            noise = np.random.normal(scale=0.02, size=n)
            a0 = best_seq + noise
            a0 = np.maximum(a0, 0.0)
        else:
            # random Dirichlet with a random concentration
            conc = 10 ** np.random.uniform(-2.0, 0.5)  # 0.01 … 3.16
            a0 = np.random.dirichlet(np.full(n, conc))
        a0 = a0 / a0.sum()

        # number of iterations for this short run – aim for ~2e5 but respect time
        remaining = explore_deadline - time.time()
        it_limit = int(min(250_000, max(50_000, remaining * 3000)))
        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len,
            deadline=explore_deadline,
            max_iters=it_limit,
            eta0=1.0,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt
            print(f"[Exploration restart {restart_cnt}] new global best {val_opt:.6f}")

        # a tiny annealing pass for the found candidate (cheap, 2 % of remaining)
        if time.time() < explore_deadline - 5:
            a_ann, val_ann = _anneal_optimize(
                best_seq, n, conv_fft_len,
                deadline=explore_deadline,
                max_moves=30_000,
            )
            if val_ann < best_val:
                best_val = val_ann
                best_seq = a_ann
                print(f"[Exploration anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 4) Long final sub‑gradient optimisation from the best candidate
    if best_seq is not None and time.time() < deadline - 5:
        print("[Final] long sub‑gradient optimisation")
        a_final, val_final = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.2,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final
            print(f"[Final] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional short annealing fine‑tuner (if any time left)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Final] annealing fine‑tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann
            print(f"[Final] anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
