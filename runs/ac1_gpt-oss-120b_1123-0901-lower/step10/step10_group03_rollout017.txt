"""
Step‑function optimizer – extended search for a lower bound.
The algorithm improves on the previous baseline by:

* adding several new low‑dimensional families (Beta with parameters <1,
  power‑product, logarithmic, cosine‑based and 1/√‑law);
* performing a much finer parametric sweep (thousands of candidates);
* running a longer Adam refinement on the smooth‑max surrogate;
* running an exponentiated‑gradient sub‑gradient phase with a more
  aggressive schedule;
* adding a final simulated‑annealing phase with many more moves.

All heavy work respects the global time‑budget (default 1000 s) and the
routine always returns a valid non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_product_power(p: float, n: int) -> np.ndarray:
    """Symmetric product‑power a_i ∝ [(i+1)(n-i)]^p (p negative)."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) * (n - i)) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β)+exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta_general(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0 (allow <1)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_cosine(n: int) -> np.ndarray:
    """U‑shaped cosine: a_i ∝ 1 + cos(π i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 + np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_log_curve(n: int) -> np.ndarray:
    """Log‑type U shape a_i ∝ -log(t) - log(1‑t)   (t∈(0,1))."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = -np.log(t) - np.log(1.0 - t)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_harmonic(n: int) -> np.ndarray:
    """Harmonic U shape a_i ∝ 1/(i+1) + 1/(n‑i)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1.0) + 1.0 / (n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k <= 0:
        a[:] = 1.0 / n
        return a
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    masses = np.full(k, 1.0 / k)
    a[pos] = masses
    a /= a.sum()
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns best vector (on simplex) seen and its exact objective.
    """
    eps = 1e-16
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.05          # temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # cooling schedule
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay (gentle)
        if it % 200_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 15_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates
    (mirror descent on the simplex).  Step size ηₜ = η₀/√(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # renormalise safely
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.  Randomly moves a fraction
    of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper for recording best results --------------------------------------
def _maybe_update(a: np.ndarray, n: int, best):
    """Update best sequence/value if a is better."""
    val = _objective(a, n)
    if val < best['val']:
        best['val'] = val
        best['seq'] = a.copy()
        print(f"[Sweep] new best {val:.6f}")

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size ------------------------------------------------------
    n = 4096                 # length of the step‑function
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best = {'seq': None, 'val': float('inf')}

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep (more families / finer grids) -------
    # symmetric power‑law (U‑shape)
    for p in np.linspace(-8.0, -0.05, 60):
        _maybe_update(_candidate_power(p, n), n, best)

    # product‑power U‑shape
    for p in np.linspace(-8.0, -0.05, 40):
        _maybe_update(_candidate_product_power(p, n), n, best)

    # symmetric exponential
    for alpha in np.logspace(-5, 2.5, 45):
        _maybe_update(_candidate_exponential(alpha, n), n, best)

    # symmetric stretched‑exponential
    for alpha in np.logspace(-4, 1.5, 30):
        for beta in np.linspace(0.2, 4.5, 25):
            _maybe_update(_candidate_stretched(alpha, beta, n), n, best)

    # monotone left‑exponential
    for alpha in np.logspace(-5, 2.2, 40):
        _maybe_update(_candidate_left_exponential(alpha, n), n, best)

    # monotone left‑stretched
    for alpha in np.logspace(-5, 1.8, 35):
        for beta in np.linspace(0.3, 4.0, 20):
            _maybe_update(_candidate_left_stretched(alpha, beta, n), n, best)

    # left power‑law (including positive p)
    for p in np.linspace(-8.0, 3.0, 55):
        _maybe_update(_candidate_left_exponential(p, n) if p >= 0 else _candidate_left_power(p, n), n, best)

    # Gamma‑like
    for shape in np.linspace(0.2, 5.0, 14):
        for scale in np.logspace(-5, 0.7, 13):
            _maybe_update(_candidate_gamma(shape, scale, n), n, best)

    # Beta distributions (both parameters <1 → U‑shape)
    a_vals = np.logspace(-2, -0.1, 18)   # 0.01 … 0.79
    b_vals = np.logspace(-2, -0.1, 18)
    for a_par in a_vals:
        for b_par in b_vals:
            _maybe_update(_candidate_beta_general(a_par, b_par, n), n, best)

    # Beta (symmetric) with a=b in (0.05,0.9)
    for ab in np.linspace(0.05, 0.9, 17):
        _maybe_update(_candidate_beta_general(ab, ab, n), n, best)

    # Double‑Gaussian
    for sig in np.logspace(-3.5, -0.2, 20):
        _maybe_update(_candidate_double_gaussian(sig, n), n, best)

    # Cosine U‑shape
    _maybe_update(_candidate_cosine(n), n, best)

    # Log‑curve U‑shape
    _maybe_update(_candidate_log_curve(n), n, best)

    # Harmonic U‑shape
    _maybe_update(_candidate_harmonic(n), n, best)

    # Random spike patterns (grid & random)
    for k in [4, 8, 16, 32, 64, 128, 256, 512]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=False), n, best)
            _maybe_update(_candidate_spike(k, n, random=True), n, best)

    # Dirichlet random vectors (various concentrations)
    for conc in [0.005, 0.02, 0.05, 0.1, 0.3, 0.7, 1.5, 3.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a, n, best)

    # Mixture of two exponentials (random)
    for _ in range(300):
        alpha1 = 10 ** np.random.uniform(-5, 2)
        alpha2 = 10 ** np.random.uniform(-5, 2)
        w1 = np.random.rand()
        a_mix = w1 * _candidate_left_exponential(alpha1, n) + \
                (1 - w1) * _candidate_left_exponential(alpha2, n)
        a_mix /= a_mix.sum()
        _maybe_update(a_mix, n, best)

    # Warm‑start from a previously stored global variable if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, n, best)

    print(f"[Sweep] best after exploration: {best['val']:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max) -----------------------------------
    if best['seq'] is not None and time.time() < deadline - 25:
        print("[Phase‑Adam] smooth‑max Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 40.0)   # up to ~40 s
        a_adam, val_adam = _adam_optimize(
            best['seq'],
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=2_000_000,
        )
        if val_adam < best['val']:
            best['val'] = val_adam
            best['seq'] = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long sub‑gradient (exact‑max) ---------------------------------
    if best['seq'] is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] exponentiated‑gradient sub‑gradient")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best['seq'],
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=12_000_000,
            eta0=1.6,
        )
        if val_sub < best['val']:
            best['val'] = val_sub
            best['seq'] = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional annealing --------------------------------------------
    if best['seq'] is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] fine‑grained simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best['seq'],
            n,
            conv_fft_len,
            deadline,
            max_moves=400_000,
        )
        if val_ann < best['val']:
            best['val'] = val_ann
            best['seq'] = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best['seq'] is None:
        best['seq'] = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best['seq'], 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    # expose the best sequence for possible warm‑starts in later runs
    globals()['height_sequence_1'] = best_seq.tolist()

    return best_seq.tolist()
