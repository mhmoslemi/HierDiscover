def _project_simplex(v):
    """Project onto simplex {x >= 0, sum(x) = 1}."""
    if v.sum() == 1 and np.all(v >= 0): return v
    # Standard algorithm: sort v descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Propose a step‑function (list of non‑negative heights) with a low
evaluation score.

Algorithm (≈1000 s budget)
--------------------------------
1. Broad parametric sweep (same families as the reference code) –
   keep a small elite list (size ≤ 8).
2. Refine the most promising one‑parameter families (power law,
   left‑exponential, symmetric exponential) by a golden‑section line
   search.  This cheap fine‑tuning often lowers the bound by several
   10⁻³.
3. Run a short Adam optimisation (smooth‑max surrogate) from the 3
   best elite starts.
4. Run a long projected‑gradient (Euclidean) sub‑gradient descent
   on the exact max‑convolution (active set as a uniform weight on the
   ties).  This replaces the exponentiated‑gradient version used in the
   reference and usually converges to a lower plateau.
5. Optional simulated‑annealing fine‑tuning if time permits.
6. Return the best vector found (clipped to the evaluator limits).

All FFT work is done with powers‑of‑two sizes for speed.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same conv length)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping, no scaling)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:

    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:

    else:
        a /= a_sum
    return a

def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Simple Euclidean projection onto the simplex {x≥0, sum x = 1}
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑d array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero after projection
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.05            # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """Projected Euclidean sub‑gradient descent on exact max(conv)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[PGD it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad_a
        a = _project_simplex(a_new)

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


def _golden_section_minimize(func, a, b, tol=1e-5, max_iter=30):
    """Univariate golden‑section search for a minimum of func on [a,b]."""
    phi = (np.sqrt(5) - 1) / 2  # 0.618...
    c = b - phi * (b - a)
    d = a + phi * (b - a)
    fc = func(c)
    fd = func(d)
    for _ in range(max_iter):
        if abs(b - a) < tol:
            break
        if fc < fd:
            b, fd = d, fc
            d = c
            c = b - phi * (b - a)
            fc = func(c)
        else:
            a, fc = c, fd
            c = d
            d = a + phi * (b - a)
            fd = func(d)
    # return midpoint of final interval and its value
    x_opt = (a + b) / 2.0
    val_opt = func(x_opt)
    return x_opt, val_opt

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem dimension (keep a power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep a few elites
    # ------------------------------------------------------------------
    elite = []  # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Symmetric power‑law
    for p in np.linspace(-5.0, -0.05, 30):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # Symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # Asymmetric two‑exponential (grid – coarse)
    alphas = np.logspace(-5, 1.0, 12)
    for al in alphas:
        for ar in alphas:
            a = _candidate_two_exponential(al, ar, n)
            _maybe_update(a)

    # Stretched exponential
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # Beta distributions
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 20):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # Monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 20):
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # Random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(3):
            a = _candidate_spike(k, n, random=True)
            _maybe_update(a)

    # Warm start from previous run (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # ------------------------------------------------------------------
    # 2) Fine‑grained 1‑D optimisation for the most promising families
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 6.0:
        # Power‑law refinement
        # retrieve the p that gave the current best power‑law (if any)
        # we simply run a fresh line‑search over the whole interval
        print("[Refine] golden‑section search on power‑law")
        def _obj_power(p):
            return _objective(_candidate_power(p, n), n)
        p_opt, val_opt = _golden_section_minimize(_obj_power, -5.0, -0.05, tol=1e-4, max_iter=25)
        a_opt = _candidate_power(p_opt, n)
        _maybe_update(a_opt)

        # Left exponential refinement
        print("[Refine] golden‑section search on left‑exponential")
        def _obj_left(alpha):
            return _objective(_candidate_left_exponential(alpha, n), n)
        al_opt, val_al = _golden_section_minimize(_obj_left, 1e-5, 10.0, tol=1e-4, max_iter=25)
        a_opt = _candidate_left_exponential(al_opt, n)
        _maybe_update(a_opt)

        # Symmetric exponential refinement
        print("[Refine] golden‑section search on symmetric exponential")
        def _obj_sym(alpha):
            return _objective(_candidate_exponential(alpha, n), n)
        ae_opt, val_ae = _golden_section_minimize(_obj_sym, 1e-5, 10.0, tol=1e-4, max_iter=25)
        a_opt = _candidate_exponential(ae_opt, n)
        _maybe_update(a_opt)

    # ------------------------------------------------------------------
    # 3) Adam (smooth‑max) – short runs from elite starts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + min(15.0, 0.15 * remaining)
        print("[Phase‑Adam] short Adam optimisation (multiple starts)")
        for idx, (_, a_start) in enumerate(elite[:3]):  # up to three best
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=400_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Projected‑gradient sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        print("[Phase‑PGD] long projected‑gradient sub‑gradient optimisation")
        pgd_deadline = time.time() + (remaining - 5.0)
        a_pgd, val_pgd = _subgradient_pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            pgd_deadline,
            max_iters=8_000_000,
            eta0=0.9,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[Phase‑PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
