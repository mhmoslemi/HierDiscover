"""
search for a non‑negative step‑function that minimizes
evaluate_sequence.

The algorithm:
1. Wide parametric sweep with a fine grid for power‑law, exponential,
   stretched‑exponential, beta‑families and mixtures with a uniform
   baseline, plus many random Dirichlet draws.  The best few candidates
   are stored as elite.
2. For each elite start a short Adam optimisation of the smooth‑max
   surrogate (log‑sum‑exp) is performed.  The runs share the remaining
   time budget equally.
3. A long exact‑max exponentiated‑gradient sub‑gradient descent is
   applied to the current best sequence.
4. If time permits a simple simulated‑annealing fine‑tuner is executed.
   The best sequence found is returned.
"""

import time
import math
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _objective(a, n):
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# candidate generators

def _candidate_power(p, n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def _candidate_exponential(alpha, n):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a

def _candidate_stretched(alpha, beta, n):
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a

def _candidate_beta(a_par, b_par, n):
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a

def _candidate_double_gaussian(sig, n):
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a

def _candidate_left_exponential(alpha, n):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a

def _candidate_left_power(p, n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def _candidate_inverse_power(p, n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def _candidate_spike(k, n):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    a /= a.sum()
    return a

def _candidate_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)

def _candidate_mixture(shape_func, shape_args, lam, n):
    # lam: weight of shape, 1-lam weight of uniform
    shape = shape_func(*shape_args, n)
    uniform = _candidate_uniform(n)
    a = lam * shape + (1.0 - lam) * uniform
    a_sum = a.sum()
    if a_sum == 0.0:
        a = uniform
    else:
        a /= a_sum
    return a

# optimisation helpers

def _update_elite(elite, val, a, max_size=12):
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=300000):
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1 = 0.9
    beta2 = 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.05
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional print
            if it % 50000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 80000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val

def _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=5000000, eta0=0.9):
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= total

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set
        tol = 1e-12
        active = np.isclose(conv, max_c, atol=tol, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _anneal_optimize(a0, n, conv_fft_len, deadline, max_moves=200000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.25
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1

    return best_a, best_val

# main function

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluate_sequence score.  The method performs a large parametric
    sweep, short Adam runs on the best candidates, a long exact‑max
    exponentiated‑gradient descent and a final simulated‑annealing
    fine‑tuning, all respecting the given time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    elite = []

    def _maybe_update(a):
        val = _objective(a, n)
        # early abandon if not better than worst elite
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # 1. fine power‑law sweep
    for p in np.linspace(-5.0, -0.05, 200):
        if time.time() >= deadline:
            break
        a = _candidate_power(p, n)
        _maybe_update(a)

    # 2. finer exponential sweep
    for alpha in np.logspace(-5, 1.0, 200):
        if time.time() >= deadline:
            break
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # 3. stretched exponential grid
    alphas = np.logspace(-4, 0.5, 30)
    betas = np.linspace(0.3, 3.0, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # 4. beta family random draws (both symmetric and asymmetric)
    for _ in range(150):
        if time.time() >= deadline:
            break
        a_par = 10 ** np.random.uniform(-2, 1)   # between 0.01 and 10
        b_par = 10 ** np.random.uniform(-2, 1)
        a = _candidate_beta(a_par, b_par, n)

"""
Search for a non‑negative step‑function (list of heights) that minimises
the public `evaluate_sequence` score.

Algorithm
---------
1.  Wide parametric sweep – generate many families of symmetric / monotone
    shapes (power‑law, exponential, stretched‑exponential, beta,
    double‑Gaussian, random spikes, Dirichlet draws) and mixtures of those
    shapes with a uniform baseline.  The best few (elite) candidates are
    retained.

2.  Adam optimisation – each elite start is refined with Adam on the
    smooth‑max surrogate (log‑sum‑exp) of the convolution peak.  The
    available time is split evenly among the elite starts.

3.  Long exponentiated‑gradient (exact‑max) sub‑gradient descent – applied
    to the current best sequence for the bulk of the remaining budget.

4.  Simulated‑annealing fine‑tuning – a cheap move‑based annealer tries
    to improve the final answer if any time is left.

All phases respect the overall time budget (default 1000 s) and the
function always returns a valid non‑negative list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a via FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping, no scaling)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Helper for elite list handling (no lambda, as required)
# ----------------------------------------------------------------------
def _elite_key(item):
    """Key function for sorting elite (by objective value)."""
    return item[0]


def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=_elite_key)
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=_elite_key)


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Edges‑heavy shape (p < 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_spike(k: int, n: int) -> np.ndarray:
    """k equal‑mass spikes placed uniformly at random."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    a /= a.sum()
    return a


def _candidate_uniform(n: int) -> np.ndarray:
    """Uniform distribution."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _candidate_mixture(shape_func, shape_args, lam: float, n: int) -> np.ndarray:
    """Mixture λ·shape + (1‑λ)·uniform (λ ∈ [0,1])."""
    shape = shape_func(*shape_args, n)
    uniform = _candidate_uniform(n)
    a = lam * shape + (1.0 - lam) * uniform
    total = a.sum()
    if total == 0.0:
        a = uniform
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # start from softmax params
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    eps_adam = 1e-8
    lr = 0.05
    lam = 0.05            # initial temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for bookkeeping)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                  n: int,
                                  conv_fft_len: int,
                                  corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 5_000_000,
                                  eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    total = a.sum()
    if total < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a /= total

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrored onto simplex)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 250_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4

    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling schedule
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.25
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that attains a low
    `evaluate_sequence` score.  The routine respects the supplied
    time budget (seconds) and always returns a valid list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # ------------------------------------------------------------------
    # Problem dimension (power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep – keep a set of elite candidates
    # ------------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        # quick reject if we already have enough elites and val is worse
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # ---- power‑law sweep (fine grid) ---------------------------------
    for p in np.linspace(-5.0, -0.05, 200):
        if time.time() >= deadline:
            break
        a = _candidate_power(p, n)
        _maybe_update(a)

    # ---- exponential sweep (fine grid) -------------------------------
    for alpha in np.logspace(-5, 1.0, 200):
        if time.time() >= deadline:
            break
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # ---- stretched exponential (grid) --------------------------------
    alphas = np.logspace(-4, 0.5, 30)
    betas = np.linspace(0.3, 3.0, 30)
    for alpha in alphas:
        for beta in betas:
            if time.time() >= deadline:
                break
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # ---- random beta distributions (symmetric & asymmetric) ----------
    for _ in range(150):
        if time.time() >= deadline:
            break
        a_par = 10 ** np.random.uniform(-2, 1)   # 0.01 … 10
        b_par = 10 ** np.random.uniform(-2, 1)
        a = _candidate_beta(a_par, b_par, n)
        _maybe_update(a)

    # ---- double Gaussian (mirrored) ----------------------------------
    for sig in np.logspace(-3, -0.1, 30):
        if time.time() >= deadline:
            break
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # ---- left‑exponential family --------------------------------------
    for alpha in np.logspace(-5, 1.2, 50):
        if time.time() >= deadline:
            break
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # ---- left‑power family --------------------------------------------
    for p in np.linspace(-4.0, 2.0, 50):
        if time.time() >= deadline:
            break
        a = _candidate_left_power(p, n)
        _maybe_update(a)

    # ---- inverse‑power (edges heavy) ---------------------------------
    for p in np.linspace(-5.0, -0.05, 50):
        if time.time() >= deadline:
            break
        a = _candidate_inverse_power(p, n)
        _maybe_update(a)

    # ---- random spikes (different sparsities) ------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            if time.time() >= deadline:
                break
            a = _candidate_spike(k, n)
            _maybe_update(a)

    # ---- Dirichlet draws (various concentration parameters) ---------
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        for _ in range(12):
            if time.time() >= deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            _maybe_update(a)

    # ---- mixtures with a uniform baseline -----------------------------
    mix_lams = [0.2, 0.4, 0.6, 0.8]

    # power‑law mixtures
    for p in [-0.2, -0.5, -1.0, -2.0]:
        for lam in mix_lams:
            if time.time() >= deadline:
                break
            a = _candidate_mixture(_candidate_power, (p,), lam, n)
            _maybe_update(a)

    # exponential mixtures
    for alpha in [0.01, 0.1, 1.0]:
        for lam in mix_lams:
            if time.time() >= deadline:
                break
            a = _candidate_mixture(_candidate_exponential, (alpha,), lam, n)
            _maybe_update(a)

    # ---- warm start from a previously stored global sequence ----------
    warm = globals().get("height_sequence_1")
    if warm is not None and time.time() < deadline:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        total = w.sum()
        if total >= 0.01:
            w /= total
            _maybe_update(w)

    # ------------------------------------------------------------------
    # Choose the best elite candidate as the current best
    # ------------------------------------------------------------------
    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found – using uniform distribution.")

    # ------------------------------------------------------------------
    # 2) Adam optimisation on elite starts (≈20 % of remaining budget)
    # ------------------------------------------------------------------
    time_now = time.time()
    remaining = deadline - time_now
    if remaining > 30.0:
        adam_budget = min(remaining * 0.2, 30.0)
        adam_deadline = time_now + adam_budget
        print("[Phase‑Adam] short Adam optimisation on elite starts")
        # limit to first few elites to keep per‑run time reasonable
        num_starts = min(5, len(elite))
        per_start_budget = adam_budget / max(1, num_starts)
        for idx, (_, a_start) in enumerate(elite[:num_starts]):
            if time.time() >= adam_deadline:
                break
            per_deadline = time.time() + per_start_budget
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                per_deadline,
                max_iters=400_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient descent (majority of remaining time)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 5.0
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=10_000_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 0.5:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 1e-12:
        best_seq = np.full(n, 1.0 / n)

    # Final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
