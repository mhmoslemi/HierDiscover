"""
Step‑function optimizer (improved).

The algorithm works in two scales:

* **Coarse scale (n≈2k)** – a dense parametric sweep over many
  families (power law, exponential, stretched, beta, double‑Gaussian,
  monotone families, sinusoidal / cosine shapes and simple polynomial
  forms).  The best candidate is kept and linearly up‑sampled to the
  final resolution.

* **Fine scale (n≈8k)** – the up‑sampled candidate is refined with
  three complementary optimisers:
      – Adam on a smooth‑max (log‑sum‑exp) surrogate,
      – exponentiated‑gradient sub‑gradient descent on the exact max,
      – a short simulated‑annealing polishing phase.
  The whole search respects the 1000 s budget and always returns a
  non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of the max‑convolution.
    Returns a vector of length len(a) which corresponds to
    d/d a  Σ_k w_k (a*a)_k .
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_sine_power(p: float, n: int) -> np.ndarray:
    """Half‑sine shape raised to a power p (p>0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)  # positive half‑sine
    a = np.maximum(a, 0.0) ** p
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_cosine_power(p: float, n: int) -> np.ndarray:
    """Symmetric cosine bump (1+cos)⁽ᵖ⁾."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1.0 + np.cos(np.pi * (2 * i + 1 - n) / n))
    a = np.maximum(a, 0.0) ** p
    a /= a.sum()
    return a


def _candidate_polynomial(p: float, q: float, n: int) -> np.ndarray:
    """Two‑parameter polynomial shape: (i+1)ᵖ·(n‑i)ᵠ."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p * (n - i) ** q
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_periodic(period: int, phase: float, offset: float, n: int) -> np.ndarray:
    """Simple non‑negative periodic pattern."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(2.0 * np.pi * (i + phase) / period) + offset
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate of the
    max‑convolution.  Uses a soft‑max (logits) re‑parameterisation so the
    iterate always stays on the simplex.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.2          # temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for occasional logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (increase smoothing)
        lam = min(lam * lam_factor, 1e8)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.85

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 3_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient with exponentiated‑gradient (mirror descent)
    on the simplex.  Step‑size ηₜ = η₀ / √(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numeric loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that moves a random fraction
    of mass between two coordinates.  Temperature decays linearly.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.15
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-18:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (≤25 %) of source mass
        frac = np.random.rand() * 0.25
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative sequence with a low evaluation score.
    The routine respects the supplied time budget (default 1000 s) and
    always returns a list that can be passed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ----------------------- coarse scale (n≈2k) -----------------------
    n_small = 2048
    conv_fft_small = _next_pow2(2 * n_small - 1)
    corr_fft_small = _next_pow2(3 * n_small - 2)

    best_small = None
    best_val_small = np.inf

    def _maybe_update_small(arr: np.ndarray):
        nonlocal best_small, best_val_small
        val = _objective(arr, n_small)
        if val < best_val_small:
            best_val_small = val
            best_small = arr.copy()
            print(f"[Sweep small] new best {val:.6f}")

    # ---- families ----------------------------------------------------
    # symmetric power‑law
    for p in np.linspace(-4.5, -0.05, 100):
        _maybe_update_small(_candidate_power(p, n_small))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.2, 120):
        _maybe_update_small(_candidate_exponential(alpha, n_small))

    # stretched exponential
    alphas = np.logspace(-4, 0.8, 30)
    betas = np.linspace(0.3, 3.0, 25)
    for alpha in alphas:
        for beta in betas:
            _maybe_update_small(_candidate_stretched(alpha, beta, n_small))

    # beta distributions
    a_pars = [0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update_small(_candidate_beta(a_par, b_par, n_small))

    # double Gaussian
    for sig in np.logspace(-3.0, -0.2, 30):
        _maybe_update_small(_candidate_double_gaussian(sig, n_small))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 110):
        _maybe_update_small(_candidate_left_exponential(alpha, n_small))

    # monotone left‑power
    for p in np.linspace(-4.5, -0.05, 100):
        _maybe_update_small(_candidate_left_power(p, n_small))

    # monotone left‑logistic
    mu = n_small / 2.0
    for beta in np.logspace(-3, 1.4, 70):
        _maybe_update_small(_candidate_left_logistic(beta, mu, n_small))

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.5, 70):
        _maybe_update_small(_candidate_left_linear(slope, n_small))

    # sinusoidal power family
    for p in np.linspace(0.5, 4.0, 35):
        _maybe_update_small(_candidate_sine_power(p, n_small))

    # cosine bump power family
    for p in np.linspace(0.5, 4.0, 35):
        _maybe_update_small(_candidate_cosine_power(p, n_small))

    # two‑parameter polynomial shape
    for p in np.linspace(-2.0, 2.0, 15):
        for q in np.linspace(-2.0, 2.0, 15):
            _maybe_update_small(_candidate_polynomial(p, q, n_small))

    # simple periodic patterns
    periods = [4, 8, 16, 32, 64, 128, 256, 512, 1024]
    for period in periods:
        for phase in [0.0, 0.25, 0.5, 0.75]:
            for offset in [0.0, 0.2, 0.5]:
                _maybe_update_small(_candidate_periodic(period, phase, offset, n_small))

    # random spike patterns
    for k in [16, 32, 64, 128, 256]:
        for _ in range(10):
            _maybe_update_small(_candidate_spike(k, n_small, random=True))

    # a few pure Dirichlet draws
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n_small))
        _maybe_update_small(a_rand)

    # warm start from previous run (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n_small:
            w = w[:n_small]
        elif w.shape[0] < n_small:
            w = np.pad(w, (0, n_small - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update_small(w)

    print(f"[Sweep small] best after exploration: {best_val_small:.6f}")

    # ---- up‑sample to the fine scale ---------------------------------
    n_big = 8192
    x_small = np.linspace(0, 1, n_small)
    x_big = np.linspace(0, 1, n_big)
    best_big = np.interp(x_big, x_small, best_small)
    best_big = np.maximum(best_big, 0.0)
    best_big /= best_big.sum()

    # ---------------------- fine scale (n≈8k) -----------------------
    conv_fft_big = _next_pow2(2 * n_big - 1)
    corr_fft_big = _next_pow2(3 * n_big - 2)

    best_seq = best_big.copy()
    best_val = _objective(best_seq, n_big)
    print(f"[Init fine] objective = {best_val:.6f}")

    # ---- 1) Adam polishing -----------------------------------------
    if time.time() < deadline - 35:
        print("[Phase‑Adam] start")
        adam_deadline = min(deadline - 5, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n_big,
            conv_fft_big,
            corr_fft_big,
            adam_deadline,
            max_iters=600_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ---- 2) multiple stochastic restarts with sub‑gradient ---------
    remaining = deadline - time.time()
    max_restarts = 7
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        # Choose a concentration that covers both sparse and dense regions
        concentration = float(np.random.choice([0.05, 0.2, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n_big, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 18.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0,
            n_big,
            conv_fft_big,
            corr_fft_big,
            slice_deadline,
            max_iters=2_200_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] improved to {val_sub:.6f}")
        remaining = deadline - time.time()

    # ---- 3) final long sub‑gradient refinement ----------------------
    if time.time() < deadline - 8:
        print("[Phase‑Subgrad] final long run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n_big,
            conv_fft_big,
            corr_fft_big,
            deadline,
            max_iters=12_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] final improvement to {val_sub:.6f}")

    # ---- 4) simulated annealing polishing -------------------------
    if time.time() < deadline - 2:
        print("[Phase‑Anneal] final anneal")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n_big,
            conv_fft_big,
            deadline,
            max_moves=500_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # -----------------------------------------------------------------
    # final sanity checks
    if best_seq is None:
        best_seq = np.full(n_big, 1.0 / n_big, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluation failed (evaluate_sequence not available).")

    return best_seq.tolist()
