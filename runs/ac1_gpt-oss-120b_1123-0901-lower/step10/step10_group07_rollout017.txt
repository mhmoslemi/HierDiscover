"""
propose_candidate
-----------------

Search for a non‑negative step function (list of floats) that minimizes

    2·n·max(a*a) / (sum(a))**2

where ``a*a`` is the full linear convolution of the sequence with itself.
The algorithm combines

* a dense parametric sweep (exponential families, mixtures, beta,
  logistic, power‑law, random spikes),
* many Dirichlet restarts refined by an exponentiated‑gradient sub‑gradient
  method (mirror‑descent on the simplex),
* a short Adam polishing phase on a smooth‑max surrogate,
* a final simulated‑annealing phase that runs until the time budget is
  exhausted.

The implementation respects the supplied time budget (default 1000 s),
prints intermediate progress information and always returns the best
sequence found (as a plain Python list).  It makes no assumptions about
the length of the sequence; the default problem size is n = 4096, but
the code works for any n ≥ 2.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of max‑convolution.
    ``w`` is a weight vector (normally an indicator of the active set).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (assumes a ≥ 0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^p (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape: 1/(1+exp(beta·(i‑mu)))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_mixed_exponential(alpha: float, lam: float, n: int) -> np.ndarray:
    """
    Convex combination uniform + left‑exponential:
        a = lam·uniform + (1‑lam)·exp(-α·i)
    Both components are normalised; the whole vector is renormalised.
    """
    uniform = np.full(n, 1.0 / n)
    expo = _candidate_left_exponential(alpha, n)
    a = lam * uniform + (1.0 - lam) * expo
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate while staying
    on the simplex via the soft‑max parametrisation ``a = softmax(logits)``.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging (cheap, uses FFT)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – slowly tighten the smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional decay of learning rate
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient / mirror descent on the simplex).
    The step size is ηₜ = η₀ / √(t+1).  Stops when the deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numeric loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that moves a random fraction
    of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        # a linear cooling schedule
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray, tag: str = "Sweep"):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 1) Parameteric sweep – dense families + fine grids

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n), "Power")

    # symmetric exponential
    for alpha in np.logspace(-5, 0.8, 120):
        _maybe_update(_candidate_exponential(alpha, n), "Exp-sym")

    # stretched exponential (symmetric)
    alphas = np.logspace(-5, 0, 30)
    betas = np.linspace(0.2, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n), "Stretched")

    # beta distributions (skew allowed)
    a_pars = np.logspace(-1, 1, 12)
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n), "Beta")

    # double Gaussian (symmetric)
    for sig in np.logspace(-3.5, -0.2, 30):
        _maybe_update(_candidate_double_gaussian(sig, n), "DoubleGauss")

    # monotone left‑exponential (dense grid around promising region)
    alphas = np.logspace(-6, -0.5, 120)  # 1e‑6 … 0.3
    for alpha in alphas:
        _maybe_update(_candidate_left_exponential(alpha, n), "Exp-left")

    # monotone left‑power (decreasing)
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_left_power(p, n), "Power-left")

    # monotone left‑logistic (sigmoid)
    mu = n / 2.0
    for beta in np.logspace(-4, 1.2, 80):
        _maybe_update(_candidate_left_logistic(beta, mu, n), "Logistic-left")

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.0, 80):
        _maybe_update(_candidate_left_linear(slope, n), "Linear-left")

    # mixed uniform + exponential – fine 2‑D grid
    for lam in np.linspace(0.0, 1.0, 13):
        for alpha in np.logspace(-5, -0.5, 30):
            _maybe_update(_candidate_mixed_exponential(alpha, lam, n), "Mix-UE")

    # random spike patterns (many repetitions)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(12):
            _maybe_update(_candidate_spike(k, n, random=True), "Spike")

    # a few pure random Dirichlet samples
    for _ in range(40):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand, "Dirichlet")

    # --------------------------------------------------------------
    # 2) Small Adam polishing (if any time left)
    if time.time() < deadline - 30 and best_seq is not None:
        print("[Phase‑Adam] start polishing")
        adam_deadline = min(deadline - 5, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline, max_iters=400_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Dirichlet restarts + sub‑gradient refinement
    remaining = deadline - time.time()
    max_restarts = 12
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        # concentration favouring either sparse or dense starts
        concentration = float(np.random.choice([0.1, 0.3, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 20.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, slice_deadline, max_iters=1_500_000, eta0=1.0
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # --------------------------------------------------------------
    # 4) Long sub‑gradient refinement on the current best
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] final long run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=10_000_000, eta0=0.9
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Simulated annealing fine‑tuning (runs until time runs out)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, deadline, max_moves=500_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # final sanity checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
