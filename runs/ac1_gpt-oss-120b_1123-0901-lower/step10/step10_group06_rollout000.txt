def simplex_projection(v):
    # v is 1D array
    # returns proj onto simplex sum=1, v_i>=0
    if v.sum() == 1 and np.all(v >=0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # find rho
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def mutate(a, sigma):
    logit = np.log(a + 1e-12)
    noise = np.random.normal(scale=sigma, size=logit.shape)
    new_logit = logit + noise
    exp = np.exp(new_logit - np.max(new_logit))
    a_new = exp/exp.sum()
    return a_new

"""
Evolutionary search for a low‑value step‑function.
-------------------------------------------------
1. Generate a diverse initial population using simple parametric families
   (exponential, power‑law, inverse‑power, spikes) and random Dirichlet draws.
2. Run a steady‑state evolutionary algorithm:
   * selection – take a parent from the current elite (size 8);
   * mutation – add Gaussian noise on the log‑probability (soft‑max) space,
     with a temperature that linearly cools from σ≈0.8 to σ≈0.01.
   * crossover – occasionally (≈10 % of moves) average two elite vectors.
   * replacement – if the offspring is better than the worst individual it
     replaces it; elite members are updated continuously.
   * the best individual found so far is kept globally.
3. When the time budget is almost exhausted, perform a short exact‑max
   exponentiated‑gradient sub‑gradient refinement on the current best
   candidate.
4. Return the final vector (as a plain list) – it is guaranteed to be
   non‑negative, sum‑normalised and within the evaluator’s clipping limits.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities (linear convolution via FFT)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (no clipping, no scaling)
#    O(a) = 2·n·max(conv_a) / (sum a)²
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return the exact evaluation score for a given vector a."""
    s = a.sum()
    if s < 1e-12:
        return np.inf
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a non‑negative vector, not yet normalised)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Heavy at the edges (p < 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int) -> np.ndarray:
    """k equal‑mass spikes on a random grid."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Simple mutation on the simplex via log‑space Gaussian noise
# ----------------------------------------------------------------------
def _mutate_logspace(a: np.ndarray, sigma: float) -> np.ndarray:
    """Perturb a (which lives on the simplex) by Gaussian noise on log‑probability."""
    # work in log‑probability to stay on the simplex after exponentiation
    logit = np.log(a + 1e-12)
    noise = np.random.normal(scale=sigma, size=logit.shape)
    new_logit = logit + noise
    # stable soft‑max
    shift = new_logit.max()
    expv = np.exp(new_logit - shift)
    a_new = expv / expv.sum()
    return a_new

    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine respects the given time budget and always

        if a.sum() == 0:
            a = np.full_like(a, 1.0 / a.size)
        else:
            a = a / a.sum()
        return _objective(a, n, conv_fft_len)

    # ------------------------------------------------------------------

    POP_SIZE = 32

    # ------------------------------------------------------------------
