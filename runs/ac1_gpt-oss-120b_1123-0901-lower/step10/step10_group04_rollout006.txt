"""
Optimiser for the step‑function evaluation problem.

The algorithm works in two phases:

1. **LP‑mixing phase** – starting from a good known sequence we repeatedly
   solve a linear program
        maximise    Σ b_j
        subject to  (f * b)[k] ≤ M   ∀k,   b ≥ 0
   where M = max(f * f) for the current sequence f.
   The solution g = b / Σb is mixed with f using a tiny line‑search on the
   mixing weight t.  Because the convolution term f * g is bounded by M,
   the convex combination (1−t)f + t g strictly reduces the true objective.
   Only a handful of LP iterations are needed (the matrix has a simple
   Toeplitz structure and is built once per iteration).

2. **Exact‑max sub‑gradient phase** – after the LP phase the remaining time
   is spent in an exponentiated‑gradient sub‑gradient descent on the true
   max‑norm 2 n·max(conv)/(∑a)².  This fine‑tunes the sequence to the
   attainable optimum.

Both stages respect the non‑negativity constraint, keep the vector
normalised to sum = 1 (the objective is scale‑invariant) and honour the
overall time budget.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import csr_matrix

# ----------------------------------------------------------------------
# FFT‑based convolution utilities ---------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _full_convolution_two(a: np.ndarray, b: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * b (both length n) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    fb = np.fft.rfft(b, fft_len)
    conv = np.fft.irfft(fa * fb, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser ---------------------------

def _subgradient_optimize_expgrad(a0: np.ndarray,
                                  n: int,
                                  fft_len: int,
                                  deadline: float,
                                  max_iters: int = 10_000_000) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the true max‑norm."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c  # sum(a) == 1 by construction

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # diminishing step size
        eta = 1.0 / np.sqrt(it + 1.0)

        # mirror‑descent (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad)

        # renormalise and guard against numerical drift
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val

        bounds = [(0, None)] * n

        try:
            res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds,
                          method='highs', options={'presolve': True})
        except Exception as e:
            print(f"[LP] exception {e!r}")
            break

        if not res.success:
            print(f"[LP] solver failed (status {res.status}); breaking")
            break

        b = res.x
        if b.sum() < 1e-12:
            print("[LP] trivial solution (all zero); aborting")
            break

        g = b / b.sum()                     # normalise to sum == 1

        # --------------------------------------------------------------
        # Determine best mixing coefficient t (simple grid + local refinement)
        conv_gg = _full_convolution(g, fft_len)
        conv_fg = _full_convolution_two(f, g, fft_len)

        # coarse grid
        t_grid = np.linspace(0.0, 1.0, 41)   # step = 0.025
        best_local_val = best_val
        best_t = 0.0

        for t in t_grid:
            coeff0 = (1.0 - t) ** 2
            coeff1 = 2.0 * t * (1.0 - t)
            coeff2 = t ** 2
            conv_new = coeff0 * conv_ff + coeff1 * conv_fg + coeff2 * conv_gg
            val = 2.0 * n * conv_new.max()   # sum == 1
            if val < best_local_val:
                best_local_val = val
                best_t = t

        if best_t == 0.0:
            print("[LP] no improvement found on this iteration")
            break

        # fine‑grained local search around the best coarse point
        fine_low = max(0.0, best_t - 0.025)
        fine_high = min(1.0, best_t + 0.025)
        t_grid_fine = np.linspace(fine_low, fine_high, 21)   # step ≈0.0025
        for t in t_grid_fine:
            coeff0 = (1.0 - t) ** 2
            coeff1 = 2.0 * t * (1.0 - t)
            coeff2 = t ** 2
            conv_new = coeff0 * conv_ff + coeff1 * conv_fg + coeff2 * conv_gg
            val = 2.0 * n * conv_new.max()
            if val < best_local_val:
                best_local_val = val
                best_t = t

        # --------------------------------------------------------------
        # Apply the mixture
        f = (1.0 - best_t) * f + best_t * g
        f = np.clip(f, 0.0, None)          # safety
        f = f / f.sum()                    # re‑normalise
        best_seq = f.copy()
        best_val = best_local_val
        lp_iter += 1

        print(f"[LP iter {lp_iter}] t={best_t:.4f}  objective={best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Sub‑gradient fine‑tuning ---------------------------------------
    if time.time() < deadline - 1:
        sub_deadline = deadline - 0.5
        print("[Phase‑Subgrad] final exponentiated‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(best_seq,
                                                       n,
                                                       fft_len,
                                                       sub_deadline,
                                                       max_iters=20_000_000)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks & return
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
