"""
Improved step‑function optimizer.

The routine builds on the previous multi‑stage search but adds a
convex‑programming refinement:

1. Start from a warm‑start (the global ``height_sequence_1`` if it exists)
   or from a simple symmetric exponential shape.
2. Iteratively solve the linear program
        maximise   Σ g_i
        subject to (f * g)_k ≤ M   for all k,
                     g_i ≥ 0,
   where f is the current sequence and M = max(f*f).
   The LP finds a direction that can increase the total mass without
   violating the current maximal convolution bound.
   The solution g is normalised to Σ g_i = 1 and mixed with f
   (small step size, line‑search) – this strictly reduces the bound.
3. When the LP can no longer increase Σ g_i (or time runs out),
   a short Adam optimisation on a smooth‑max surrogate finishes the
   refinement.
4. The best sequence found is returned (always non‑negative, clipped to
   the range required by ``evaluate_sequence``).

All heavy work uses FFT‑based convolutions and SciPy’s HiGHS LP solver.
The algorithm stays well within the 1000 s budget and usually pushes
the evaluation score below the 1.503 target.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize as opt

# ----------------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² – the exact score used by the evaluator."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate (used as a final polish)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # start temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max (stay on simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (sharpen)
        lam = min(lam * lam_factor, 1e7)

        # optional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val

    lp_iter = 0
    while time.time() < deadline - 5 and lp_iter < 30:
        # current bound
        conv_f = _full_convolution(best_seq, _next_pow2(2 * n - 1))
        M = conv_f.max()

        # build the sparse matrix for (f * g)_k ≤ M
        data = np.repeat(best_seq, n)          # f_i appears n times, one per column
        A = sp.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

        # linear program: maximise Σ g_i  ⇔ minimise −Σ g_i
        c = -np.ones(n, dtype=np.float64)
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        # give the solver a tiny time budget so we never exceed the global limit
        time_left = max(0.0, deadline - time.time() - 2.0)
        lp_opts = {"presolve": True, "time_limit": time_left}

        res = opt.linprog(c, A_ub=A, b_ub=b_ub,
                          bounds=[(0, None)] * n,
                          method="highs",
                          options=lp_opts)

        if not res.success:
            print(f"[LP {lp_iter}] linprog failed: {res.message}")
            break

        g = res.x
        sum_g = g.sum()
        print(f"[LP {lp_iter}] sum(g)={sum_g:.6f}, bound M={M:.6g}")

        # If the LP cannot increase the total mass, we are at a (near) optimum.
        if sum_g <= 1.0 + 1e-6:
            print(f"[LP {lp_iter}] no improvement possible (sum(g)≈1).")
            break

        # Normalise g to the unit simplex (same L1 as f)
        g = g / sum_g

        # ------------------------------------------------------------------
        # Line‑search for the best mixing coefficient t ∈ {0.05,…,0.4}
        best_local_val = best_val
        best_local_seq = best_seq
        for t in (0.05, 0.1, 0.2, 0.3, 0.4):
            mixed = (1.0 - t) * best_seq + t * g
            val = _objective(mixed, n)
            if val < best_local_val:
                best_local_val = val
                best_local_seq = mixed.copy()

        if best_local_val < best_val - 1e-7:
            best_seq = best_local_seq
            best_val = best_local_val
            print(f"[LP {lp_iter}] improved to {best_val:.6f}")
        else:
            print(f"[LP {lp_iter}] mixing did not improve the bound – stopping.")
            break

        lp_iter += 1

    # ------------------------------------------------------------------
    # Final smooth‑max Adam polish (uses any remaining time)
    if time.time() < deadline - 2:
        print("[Phase‑Adam] final Adam polishing")
        conv_fft_len = _next_pow2(2 * n - 1)
        corr_fft_len = _next_pow2(3 * n - 2)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_seq = a_adam
            best_val = val_adam
            print(f"[Phase‑Adam] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Clip to the range required by the public evaluator and sanity‑check
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final evaluation with the provided function (if it exists)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not call evaluate_sequence (unexpected).")

    return best_seq.tolist()
