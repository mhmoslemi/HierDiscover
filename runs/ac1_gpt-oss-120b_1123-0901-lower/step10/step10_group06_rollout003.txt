"""
Propose a non‑negative step‑function (list of heights) with a low evaluation
score.

The algorithm consists of four stages:

1. **Broad parametric sweep** – generate many symmetric and *asymmetric*
   families (power‑law, exponential, stretched‑exponential, beta,
   double‑Gaussian, random spikes, Dirichlet draws).  All candidates are
   normalized to unit sum and the exact objective (2·n·max(conv)/sum²) is
   evaluated.  The best K (default 12) are kept as *elite*.

2. **Adam optimisation on a smooth‑max surrogate** – each elite start is
   refined with a soft‑max parametrisation (logits) and an Adam optimiser.
   The temperature λ of the smooth‑max (log‑sum‑exp) is slowly increased,
   turning the surrogate into the true max.

3. **Exact‑max sub‑gradient descent (exponentiated‑gradient)** – a long
   run from the best candidate of stage 2.  The active set of indices that
   attain the maximum convolution is used to compute a sub‑gradient, and
   updates are performed by exponentiated‑gradient (mirror descent on the
   simplex).

4. **Targeted local search** – the final sequence is fine‑tuned by moving
   a small amount of mass away from the pair of entries that contributes
   the most to the current maximal convolution value.  Simple simulated‑
   annealing is used to avoid getting stuck.  This stage typically yields
   the last few percent improvement needed to beat the target bound.

All stages respect a global 1000 s budget (a small safety margin is kept).
The routine always returns a valid non‑negative list; the final sequence
is also evaluated with the public ``evaluate_sequence`` function for logging.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same conv length)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping, no scaling)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators (all produce a unit‑sum non‑negative vector)

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(‑α i) + exp(‑α (n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Stretched exponential a_i ∝ exp(‑α t^β) + exp(‑α (1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians of width *sig*."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone left‑hand exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone left‑hand power‑law."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Heavy at the edges: a_i ∝ (i+1)^p + (n‑i)^p with p<0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_asym_power(p_left: float, p_right: float, n: int) -> np.ndarray:
    """Asymmetric power‑law: left half uses p_left, right half uses p_right."""
    i = np.arange(n, dtype=np.float64)
    half = n // 2
    a = np.empty_like(i)
    a[:half] = (i[:half] + 1) ** p_left
    a[half:] = (n - i[half:]) ** p_right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_asym_exponential(alpha_left: float, alpha_right: float, n: int) -> np.ndarray:
    """Asymmetric exponential: left half decays with α_left, right half with α_right."""
    i = np.arange(n, dtype=np.float64)
    half = n // 2
    a = np.empty_like(i)
    a[:half] = np.exp(-alpha_left * i[:half])
    a[half:] = np.exp(-alpha_right * (n - 1 - i[half:]))
    a /= a.sum()
    return a

# Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max (log‑sum‑exp) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters (stay on simplex)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a bit larger than the default
    lam = 0.03                        # start temperature
    lam_factor = 1.025                # grow slowly

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 5e5)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.95,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set – indices attaining the max (loose tolerance for stability)
        tol = 1e-9
        active = np.isclose(conv, max_c, atol=tol, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1.0)

        # exponentiated‑gradient step (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – guard against loss of mass
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_search_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Targeted local search: repeatedly move a small amount of mass away from
    the pair of indices that contributes most to the currently maximal
    convolution entry.  Accept moves that improve the objective; otherwise
    accept with a decreasing temperature (simulated‑annealing style).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    T_start = 0.13
    T_end = 1e-5
    it = 0

    while time.time() < deadline and it < max_moves:
        # pick one of the largest convolution indices
        K = 4
        top_inds = np.argpartition(conv, -K)[-K:]          # unsorted top‑K
        k = int(top_inds[np.random.randint(K)])            # shift with large mass

        # range of i that participates in this shift
        i_min = max(0, k - (n - 1))
        i_max = min(n - 1, k)
        i_range = np.arange(i_min, i_max + 1, dtype=int)

        # product a_i * a_{k-i}
        prods = a[i_range] * a[k - i_range]
        if prods.size == 0:
            it += 1
            continue
        max_idx = int(np.argmax(prods))
        i = int(i_range[max_idx])
        j = int(k - i)

        # amount to move – a modest fraction of the smaller of the two
        move_cap = min(a[i], a[j])
        if move_cap <= 1e-12:
            it += 1
            continue
        delta = move_cap * (0.12 + 0.28 * np.random.rand())   # 0.12‑0.40 of move_cap

        # destination indices (avoid i and j)
        r = np.random.randint(n)
        while r == i or r == j:
            r = np.random.randint(n)
        s = np.random.randint(n)
        while s == i or s == j or s == r:
            s = np.random.randint(n)

        a2 = a.copy()
        a2[i] -= delta
        a2[j] -= delta
        a2[r] += delta
        a2[s] += delta

        # sanity – keep non‑negative
        if a2[i] < -1e-15 or a2[j] < -1e-15:
            it += 1
            continue
        a2[i] = max(a2[i], 0.0)
        a2[j] = max(a2[j], 0.0)

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
        else:
            # simulated‑annealing acceptance
            T = max(T_end, T_start * (1.0 - it / float(max_moves)))
            if np.random.rand() < math.exp(-(val2 - best_val) / T):
                a = a2
                conv = conv2

        it += 1

    return best_a, best_val

# Helper to keep a small set of elite candidates
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # --------------------------------------------------------------
    # Problem dimension (keep a power of two for fast FFT)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Wide parametric sweep – keep a few elite candidates
    # --------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Symmetric families
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_par in a_pars:
        for b_par in a_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))
    for sig in np.logspace(-3, -0.1, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))
    for alpha in np.logspace(-5, 1.2, 20):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-4.0, 2.0, 28):
        _maybe_update(_candidate_left_power(p, n))
    for p in np.linspace(-4.5, -0.1, 25):
        _maybe_update(_candidate_inverse_power(p, n))

    # --------------------------------------------------------------
    # Asymmetric families (new)
    # --------------------------------------------------------------
    p_vals = np.linspace(-5.0, -0.1, 12)
    for p_l in p_vals:
        for p_r in p_vals:
            _maybe_update(_candidate_asym_power(p_l, p_r, n))
    al_vals = np.logspace(-5, 0.8, 12)
    for a_l in al_vals:
        for a_r in al_vals:
            _maybe_update(_candidate_asym_exponential(a_l, a_r, n))

    # Random spikes / Dirichlet draws
    for k in [32, 64, 128, 256, 512]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True))
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for c in concs:
        _maybe_update(np.random.dirichlet(np.full(n, c)))

    # Warm‑start from a previously stored global (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # --------------------------------------------------------------
    # 2) Adam (smooth‑max) – give each elite a short dedicated run
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(25.0, 0.22 * remaining)
        print("[Phase‑Adam] short Adam optimisation (multiple starts)")
        for idx, (_, a_start) in enumerate(elite[:4]):  # up to four elite starts
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=500_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient descent (exponentiated‑gradient)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=8_000_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Targeted local search (fine‑tuning)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Local] targeted local‑search fine‑tuning")
        a_loc, val_loc = _local_search_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_loc < best_val:
            best_val = val_loc
            best_seq = a_loc.copy()
            print(f"[Phase‑Local] improved to {val_loc:.6f}")

    # --------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
