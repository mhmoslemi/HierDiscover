"""
Propose a non‑negative step‑function (list of heights) with a low evaluation score.

Algorithm
---------
1. **Fine‑grained parametric sweep** – generate a large set of monotone
   decreasing exponentials (a_i ∝ exp(-α·i)) and inverse‑powers
   (a_i ∝ (i+1)^p) with a dense grid of the controlling parameters.
   Each candidate is normalised to unit L1–norm and its exact objective
   2·n·max(conv)/sum² is evaluated via an FFT‑based convolution.
   The best few candidates are kept.

2. **Local refinement** – the best exponential is re‑searched on a tight
   log‑scale window around its optimal α, again evaluating the exact
   objective.

3. **Exact‑max sub‑gradient descent (exponentiated‑gradient)** – starting
   from the best candidate, perform a long mirror‑descent on the simplex:
   a ← a·exp(-η·∇) with a step‑size ηₜ = η₀/√(t+1).  The gradient of the
   max‑convolution is obtained by a cross‑correlation with a mask that
   selects the active (max‑attaining) indices.  The loop runs until the
   allocated time budget is exhausted, constantly remembering the best
   vector encountered.

4. **Safety & return** – clip values to the evaluator’s limits,
   sanity‑check with ``evaluate_sequence`` and return the best sequence.

The routine respects the overall time budget (default 1000 s) and always
returns a valid non‑negative list of heights.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper FFT utilities (power‑of‑two lengths)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution_fft(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation_fft(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for sub‑gradient (same conv length)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (identical to evaluate_sequence, but without clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution_fft(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a non‑negative vector, normalised to sum=1)
# ----------------------------------------------------------------------
def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone power law a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """k equal‑mass spikes placed at random locations."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = rng.choice(n, size=k, replace=False)
        a[pos] = 1.0 / k
    else:
        a[:] = 1.0 / n
    return a


# ----------------------------------------------------------------------
# Elite handling
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimension – a power of two for fast FFT (trade‑off speed / flexibility)
    # --------------------------------------------------------------
    n = 4096                     # keep the original length for compatibility
    fft_len = _next_pow2(2 * n - 1)
    # --------------------------------------------------------------
    # 1) Fine‑grained parametric sweep
    # --------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(vec):
        val = _objective(vec, n, fft_len)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, vec)

    # ----- monotone decreasing exponentials (dense log‑grid) -----
    print("[Sweep] exploring left‑exponential family")
    alphas = np.logspace(-6, -2, 300)          # 1e‑6 … 1e‑2
    for alpha in alphas:
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # ----- monotone power‑law (dense linear grid) -----
    print("[Sweep] exploring left‑power family")
    powers = np.linspace(-6.0, 4.0, 300)       # from heavy tails to steep decay
    for p in powers:
        a = _candidate_left_power(p, n)
        _maybe_update(a)

    # ----- random sparse spikes (different sparsities) -----
    print("[Sweep] exploring spike patterns")
    for k in [32, 64, 128, 256, 512]:
        for _ in range(3):
            a = _candidate_spike(k, n, rng)
            _maybe_update(a)

    # ----- warm‑start from a previously stored solution (if any) -----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, fft_len)
        print("[Sweep] no candidates found, using uniform")

    # --------------------------------------------------------------
    # 2) Local refinement of the best exponential (if it is the winner)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 10.0 and elite:
        # Identify if the top elite came from the exponential family
        # (we cannot know directly, so we simply redo a fine sweep
        # around the α that gave the current best value).
        # To obtain the α, we re‑search the exponential family using a
        # finer log‑grid centred on the best found α.
        # The best α is approximated by a line‑search on a small interval.
        print("[Refine] local fine‑grid exponential search")
        # Rough guess for a good α: take the α that gave the best value
        # by re‑evaluating the sweep – we keep the α that generated the
        # current best sequence (it is the only way without bookkeeping).
        # Simple heuristic: try a log‑grid around 1e‑4 … 1e‑3.
        fine_alphas = np.logspace(-5, -2, 500)
        for alpha in fine_alphas:
            cand = _candidate_left_exponential(alpha, n)
            val = _objective(cand, n, fft_len)
            if val < best_val:
                best_val = val
                best_seq = cand.copy()
                print(f"[Refine] improved to {val:.6f} at α={alpha:.2e}")

    # --------------------------------------------------------------
    # 3) Exact‑max sub‑gradient descent (exponentiated‑gradient)
    # --------------------------------------------------------------
    # Use as much time as remains, but leave a small safety margin.
    sub_deadline = deadline - 1.0
    if time.time() < sub_deadline:
        print("[SubGrad] long exponentiated‑gradient descent")
        a = np.maximum(best_seq, 0.0)
        if a.sum() < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        a /= a.sum()

        best_a = a.copy()
        best_score = best_val

        # Pre‑compute FFT length for convolution / cross‑correlation
        corr_fft_len = _next_pow2(3 * n - 2)

        it = 0
        eta0 = 0.85
        while time.time() < sub_deadline:
            conv = _full_convolution_fft(a, fft_len)
            max_c = conv.max()
            score = 2.0 * n * max_c / (a.sum() ** 2)

            if score < best_score:
                best_score = score
                best_a = a.copy()
                print(f"[SubGrad] it={it:07d} new best {score:.6f}")

            # active set (indices attaining the max, within tolerance)
            active = conv == max_c
            if not np.any(active):
                # should not happen, but guard
                active = np.isclose(conv, max_c, atol=1e-15)

            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()

            # gradient of the exact max
            grad = 2.0 * _cross_correlation_fft(w, a, corr_fft_len)

            s = a.sum()
            if s < 1e-12 or np.isnan(s):
                a = np.random.dirichlet(np.ones(n))
                s = a.sum()
            a /= s

            it += 1

        # keep the best found during sub‑gradient phase
        if best_score < best_val:
            best_val = best_score
            best_seq = best_a.copy()
            print(f"[SubGrad] final best {best_score:.6f}")

    # --------------------------------------------------------------
    # 4) Safety – ensure a valid return (non‑negative, clipped)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if available)
    try:

        print(f"[Result] final evaluated score = {final_score:.6f}")
    except Exception:   # pragma: no cover
        print("[Result] could not compute final score (evaluator missing)")

    return best_seq.tolist()
