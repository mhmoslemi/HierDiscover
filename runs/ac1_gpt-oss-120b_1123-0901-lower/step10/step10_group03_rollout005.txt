    conv = np.fft.irfft(fa * fa, fft_len)

    # correlation w ★ a of length len(a)
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

Step‑function optimizer.

    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

            w /= w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) long exponentiated‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n)
        best_val = _objective(best_seq, n)

    print("[Phase‑Subgrad] start exponentiated‑gradient optimisation")
    a_opt, val_opt = _subgradient_optimize_expgrad(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, eta0=2.5
    )
    if val_opt < best_val:
        best_val = val_opt
        best_seq = a_opt
        print(f"[Phase‑Subgrad] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – clip to the limits used by `evaluate_sequence`
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Optional sanity check with the public evaluator
    # ------------------------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not run evaluate_sequence – returning raw sequence.")

    return best_seq.tolist()
