        return conv.max()

"""
Hybrid optimizer for the step‑function evaluation problem.

The routine builds a diverse pool of start vectors (including an
arcsine‑shaped candidate that concentrates mass near the boundaries),
chooses the best one, and then refines it through several
phases:

1. Adam on a smooth‑max surrogate of the max‑convolution.
2. Exponentiated‑gradient sub‑gradient descent on the exact max.
3. Spectral‑flattening: the FFT magnitude of the vector is forced to be
   (almost) constant while keeping the ℓ₁‑norm fixed; after each inverse
   FFT the vector is projected onto the probability simplex.
4. Projected gradient descent on the exact max‑convolution.
5. A short simulated‑annealing local search.

All phases respect the global time budget (≈ 1000 s) and the routine
always returns the best sequence found before the deadline.
"""

import time
import math
import numpy as np
from typing import List, Optional

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a smooth‑max surrogate of max(conv).
    Returns (best_vector, best_objective) where the objective is the
    *exact* max‑convolution value (i.e. the evaluation score for sum=√(2n)).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.5                # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector (sum = 1)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact max‑convolution (objective)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {max_c:.8f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # sub‑gradient of smooth‑max w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # slowly tighten the smooth‑max (increase λ)
        lam = min(lam * lam_factor, 1e8)

        # learning‑rate decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Operates directly on the probability simplex (sum = 1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {max_c:.8f}")

        # active set where conv attains the current max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    Keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.8f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Forces the FFT magnitude to be (almost) constant while keeping the
    ℓ₁‑norm (sum) fixed. After each inverse FFT the vector is projected onto
    the non‑negative simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        if max_c < best_val:
            best_val = max_c
            best_a = a.copy()
            print(f"[Spectral it={it:07d}] new best {max_c:.8f}")

        A = np.fft.fft(a)
        energy = np.dot(a, a)            # ||a||_2^2

        # Preserve the DC component (the sum = 1 after simplex projection)
        if n > 1:
            # total energy in FFT domain = n * energy
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # Projection onto the simplex (non‑negative, Σ=1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> List[float]:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length of the vector – 4096 is a good trade‑off between quality and speed.
    n = 4096
    S = np.sqrt(2.0 * n)                     # target ℓ₁‑norm (makes objective = max‑conv)
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start if a previous construction is available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum() * S
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Helper functions
    def _normalise(v: np.ndarray) -> np.ndarray:
        """Scale v to have sum S (default) while keeping non‑negativity."""
        v = np.maximum(v, 0.0)
        s = v.sum()
        if s < 1e-12:
            v = np.random.dirichlet(np.ones_like(v))
            s = v.sum()
        return v * (S / s)

    # ------------------------------------------------------------------
    # Generate a diverse pool of start vectors
    pool: List[np.ndarray] = []

    # 1) Warm start and its shifts
    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # 2) Arcsine‑shaped candidate (mass near the boundaries)
    i = np.arange(n, dtype=np.float64) + 0.5
    pos = i / n
    arcsine = 1.0 / np.sqrt(pos * (1.0 - pos))
    arcsine = np.where(np.isfinite(arcsine), arcsine, 0.0)
    pool.append(_normalise(arcsine))

    # 3) Dirichlet draws (various concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_normalise(np.random.dirichlet(alpha * np.ones(n))))

    # 4) Exponential – left/right decay families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        pool.append(_normalise(shape))

    # 5) Power‑law family (U‑shaped)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            pool.append(_normalise(shape))

    # 6) Double‑Gaussian spikes at the ends
    x = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((x) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((x - (n - 1)) / (sigma * n)) ** 2)
        pool.append(_normalise(shape))

    # 7) Half‑wave sinusoid
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    pool.append(_normalise(sinusoid))

    # 8) Uniform
    pool.append(_normalise(np.full(n, 1.0 / n)))

    # 9) Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        pool.append(_normalise(a))

    # 10) Random sparse spikes (hand‑crafted)
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(_normalise(a))
    for _ in range(6):
        k = np.random.choice((32, 64, 128))
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(_normalise(a))

    # ------------------------------------------------------------------
    # Quick scan of the pool to pick the most promising start
    best_seq: Optional[np.ndarray] = None
    best_val = float("inf")
    for idx, cand in enumerate(pool):
        if time.time() > deadline:
            break
        # Ensure proper normalisation
        cand = _normalise(cand)
        conv = _full_convolution(cand, conv_fft_len)
        val = conv.max()
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Init {idx:03d}] new best {val:.8f}")

    # If for any reason we have no candidate, fall back to uniform.
    if best_seq is None:
        best_seq = _normalise(np.full(n, 1.0 / n))

    # ------------------------------------------------------------------
    # Phase I – Adam on smooth‑max surrogate (large budget)
    if time.time() < deadline - 5:
        sub_dead = min(deadline, time.time() + 250.0)
        best_seq, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=2_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = best_seq.copy()
            print(f"[Phase I] Adam improved best to {val_adam:.8f}")

    # ------------------------------------------------------------------
    # Phase II – Exact‑max exponentiated‑gradient sub‑gradient descent
    if time.time() < deadline - 5:
        sub_dead = deadline - 5
        best_seq, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=1_200_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = best_seq.copy()
            print(f"[Phase II] Sub‑gradient improved best to {val_sub:.8f}")

    # ------------------------------------------------------------------
    # Phase III – Spectral flattening (aggressive)
    if time.time() < deadline - 5:
        print("[Phase III] Spectral flattening start")
        sub_dead = deadline - 5
        best_seq, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_iters=800_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = best_seq.copy()
            print(f"[Phase III] Spectral improved best to {val_spec:.8f}")

    # ------------------------------------------------------------------
    # Phase IV – Projected gradient descent on exact max‑conv
    if time.time() < deadline - 5:
        print("[Phase IV] Projected GD start")
        sub_dead = deadline - 5
        best_seq, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = best_seq.copy()
            print(f"[Phase IV] GD improved best to {val_gd:.8f}")

    # ------------------------------------------------------------------
    # Phase V – Final simulated‑annealing refinement
    if time.time() < deadline - 5:
        print("[Phase V] Simulated annealing start")
        sub_dead = deadline - 5
        best_seq, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = best_seq.copy()
            print(f"[Phase V] Anneal improved best to {val_ann:.8f}")

    # ------------------------------------------------------------------
    # Optional symmetrisation (helps for the convolution‑type objective)
    if time.time() < deadline - 3:
        sym = (best_seq + best_seq[::-1]) * 0.5
        sym = _normalise(sym)
        conv = _full_convolution(sym, conv_fft_len)
        val_sym = conv.max()
        if val_sym < best_val:
            best_val = val_sym
            best_seq = sym
            print(f"[Symmetrisation] improved best to {val_sym:.8f}")

    # ------------------------------------------------------------------
    # Ensure feasibility for the external evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.8f} (length {len(best_seq)})")
    return best_seq.tolist()
