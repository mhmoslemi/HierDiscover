def projected_gradient_descent(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=10**6, eta0=1.5):
    a = np.copy(a0)
    best_a = a.copy()
    best_val = _objective(a, n)
    for it in range(max_iters):
        if time.time() >= deadline: break
        # compute convolution and active set
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # compute gradient of max convolution: use uniform w on active set
        active = np.isclose(conv, M, atol=1e-12)
        w = np.zeros_like(conv)
        # If many active maybe big; uniform weight reduces gradient amplitude.
        if active.any():
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)  # gradient of max_b
        # compute gradient of full objective: factor 2*n
        grad_obj = 2.0 * n * grad

        # Step size schedule: η = eta0 / sqrt(it+1)
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad_obj

        # Project onto simplex
        a_new = _proj_simplex(a_new)  # ensure nonnegative and sum=1

        # Evaluate
        val = _objective(a_new, n)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            # optional print
            if it % 5000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")

        # maybe also keep a = a_new for next iteration
        a = a_new

    return best_a, best_val

def proj_simplex(v):
    # ensures non-negative and sum=1
    # v may be any real
    # algorithm from Duchi et al. (2008).
    n = len(v)
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all non-positive
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1] - 1  # index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

Be careful: above yields sum to 1, guarantee non-negative.

def _pgd_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=10**7, eta0=0.05):
    a = a0.copy()
    # ensure sum=1
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        # compute conv
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (allow tolerance)
        active = np.isclose(conv, max_c, atol=1e-12)
        # compute w uniform on active
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0/active.sum()
        # gradient of max_b
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # gradient of objective = 2*n * grad
        grad_obj = 2.0 * n * grad
        # schedule
        eta = eta0 / np.sqrt(it + 1.0)
        # additive update
        a_new = a - eta * grad_obj
        # project onto simplex
        a_new = _proj_simplex(a_new)
        # evaluate
        val = _objective(a_new, n)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            # optionally print every 5000
            if it % 5000 == 0:
                print(f"[PGD it={it}] new best {val:.6f}")
        a = a_new
        it += 1
    return best_a, best_val

"""
Step‑function optimizer.

The routine consists of four stages:

1. Broad parametric sweep – many analytic families (exponential,
   power‑law, stretched‑exponential, Gamma, Beta, double‑Gaussian,
   random spikes, Dirichlet draws) are generated, normalized to sum=1
   and evaluated with a fast FFT‑based convolution.  The best
   candidate is kept.

2. Smooth‑max Adam – a short Adam optimisation on the log‑sum‑exp
   surrogate of the max‑convolution (soft‑max weights).  This pushes
   the solution quickly into a good basin while staying on the
   simplex.

3. Smooth projected‑gradient descent – from the Adam result we run a
   long additive gradient descent on the *exact* max‑convolution
   objective.  The gradient is obtained via a uniform weight over the
   active (maximal) indices; a smooth‑max version is used early on to
   obtain a better direction.  After each step we project back onto
   the probability simplex by simple clipping and renormalisation.
   The step size decays as η_t = η₀/√t and a temperature‑like
   parameter λ is increased multiplicatively so the method gradually
   focuses on the true max.

4. Simulated‑annealing fine‑tuning – random mass transfers between
   coordinates are attempted; moves that improve the objective are
   always accepted, otherwise they are accepted with a small
   probability that decays linearly in time.

The whole procedure respects the given time budget (default 1000 s)
and always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a) via FFT."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(a*a)/(∑a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    a /= a.sum()
    return a

# Simple simplex projection (clip‑then‑renorm) -------------------------

def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Clip to non‑negative and renorm to sum‑one (fast, not exact Euclidean)."""
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0.0 or np.isnan(s):
        return np.full_like(v, 1.0 / v.shape[0])
    return v / s


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate ---------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Smooth‑max projected gradient descent on the exact max --------------

def _smooth_pgd_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.05,
    lam0: float = 0.1,
    lam_factor: float = 1.02,
) -> tuple[np.ndarray, float]:
    """
    Additive projected‑gradient descent.
    Starts with a smooth‑max surrogate (λ) and gradually sharpens it.
    Projection onto the simplex is performed by clipping & renormalising.
    """
    a = a0.copy()
    a = _project_to_simplex(a)

    best_a = a.copy()
    best_val = _objective(a, n)

    lam = lam0
    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # fall back to uniform over active set
            active = np.isclose(conv, max_c, atol=1e-12)
            w = np.zeros_like(conv)
            if active.any():
                w[active] = 1.0 / active.sum()
        else:
            w /= w_sum

        # gradient of max_b (or its smooth approximation)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)   # ∂ max_b / ∂a
        grad_obj = 2.0 * n * grad                               # ∂ f / ∂a

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad_obj
        a_new = _project_to_simplex(a_new)

        val = _objective(a_new, n)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            if it % 5000 == 0:
                print(f"[SmoothPGD it={it:06d}] new best {val:.6f}")

        a = a_new
        lam = min(lam * lam_factor, 1e7)
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine tuner -----------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point -------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ---------------------------------------

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # families (same as earlier code, but a bit more compact)
    for p in np.linspace(-6.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-4, 2.5, 35):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))
    for alpha in np.logspace(-4, 2.0, 30):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-6.0, 2.0, 30):
        _maybe_update(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 12):
        for alpha in np.logspace(-4, 1.5, 25):
            _maybe_update(_candidate_left_stretched(alpha, beta, n))
    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            _maybe_update(_candidate_gamma(shape, scale, n))
    a_pars = np.logspace(-2, 0.5, 12)
    b_pars = np.logspace(0, 1.0, 12)
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))
    for sig in np.logspace(-3, -0.2, 15):
        _maybe_update(_candidate_double_gaussian(sig, n))
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))
    # random mixtures of two exponentials
    for _ in range(200):
        alpha1 = 10 ** np.random.uniform(-4, 2)
        alpha2 = 10 ** np.random.uniform(-4, 2)
        w1 = np.random.rand()
        a_mix = w1 * _candidate_left_exponential(alpha1, n) + \
                (1 - w1) * _candidate_left_exponential(alpha2, n)
        a_mix /= a_mix.sum()
        _maybe_update(a_mix)
    # Dirichlet draws with various concentrations
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # warm‑start from a previously‑found global variable (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] < n:
                w = np.pad(w, (0, n - w.shape[0]))
            else:
                w = w[:n]
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam refinement (smooth‑max) ---------------------------

    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑Adam] short Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Smooth‑max projected‑gradient descent on the exact max --------

    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑PGD] smooth projected‑gradient descent")
        a_pgd, val_pgd = _smooth_pgd_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            eta0=0.05,        # base step size
            lam0=0.1,
            lam_factor=1.02,
        )
        if val_pgd < best_val:
            best_val = val_pgd
            best_seq = a_pgd.copy()
            print(f"[Phase‑PGD] improved to {val_pgd:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning ---------------------

    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity check and return ------------------------------------

    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
