"""
Propose a low‑value step‑function for the evaluation routine.

Main ideas
---------
1. Broad parametric sweep over a rich family of shapes (power‑law,
   stretched‑exponential, beta, double‑Gaussian, bimodal Gaussian,
   inverse‑quadratic, random Dirichlet spikes, etc.).  Every candidate is
   normalised to sum = 1 and evaluated with a fast FFT‑based convolution.
2. Keep the best few candidates and run a long Adam optimisation on a
   smooth‑max surrogate of the true objective (smooth‑max temperature
   is increased gradually).  Several restarts are performed.
3. Refine the best vector with an exponentiated‑gradient sub‑gradient
   method on the exact max‑norm.  The run length is limited only by the
   remaining time budget.
4. If time permits, finish with a simulated‑annealing fine‑tuner that
   transfers a random fraction of mass between two positions.
All stages respect the 1000 s budget and always return a valid
non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) ----------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(conv)/(sum a)² – the true score used for comparison."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution on the grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_bimodal_gaussian(sig: float, offset_frac: float, n: int) -> np.ndarray:
    """Two symmetric Gaussians located at a fraction offset from each end."""
    i = np.arange(n, dtype=np.float64)
    pos1 = offset_frac * (n - 1)
    pos2 = (n - 1) - pos1
    a = np.exp(-0.5 * ((i - pos1) / sig) ** 2) + np.exp(-0.5 * ((i - pos2) / sig) ** 2)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (Dirichlet masses) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_inv_quad(n: int) -> np.ndarray:
    """Inverse‑quadratic shape a_i ∝ 1/((i+1)*(n‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1.0) * (n - i))
    a_sum = a.sum()
    a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam optimisation (smooth‑max + L₂ regularisation) ------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    lam_start: float = 0.08,
    lam_factor: float = 1.008,
    l2_weight: float = 0.13,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth‑max(conv) + λ·‖conv‑uniform‖₂² .
    Returns the best vector encountered (simplex‑projected) and its exact
    objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for monitoring
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # L₂ part
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 15_000_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning ---------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing that transfers a random fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem size – 4096 provides a good trade‑off between quality and speed
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    best_seq = None
    best_val = float("inf")
    candidates = []      # store many (value, vector) pairs for later restarts

    # ------------------------------------------------------------------
    # Helper to evaluate a candidate and keep the global best
    def _store(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        candidates.append((val, a.copy()))
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Very wide parametric sweep
    # ------------------------------------------------------------------
    # power‑law (U‑shaped)
    for p in np.linspace(-7.0, -0.01, 60):
        _store(_candidate_power(p, n))

    # pure exponentials
    for alpha in np.logspace(-6, 1.5, 70):
        _store(_candidate_exponential(alpha, n))

    # stretched exponentials
    alphas = np.logspace(-4, 0.7, 30)
    betas = np.linspace(0.2, 4.0, 20)
    for a_alpha in alphas:
        for b_beta in betas:
            _store(_candidate_stretched(a_alpha, b_beta, n))

    # beta‑distribution draws
    for _ in range(200):
        a_par = 10 ** np.random.uniform(-2, 2)
        b_par = 10 ** np.random.uniform(-2, 2)
        _store(_candidate_beta(a_par, b_par, n))

    # double Gaussian anchored at the ends
    for sig in np.logspace(-3.0, -0.1, 30):
        _store(_candidate_double_gaussian(sig, n))

    # bimodal Gaussian with offset from the ends
    for sig in np.logspace(-3.0, -0.1, 20):
        for off in np.linspace(0.05, 0.45, 10):
            _store(_candidate_bimodal_gaussian(sig, off, n))

    # left‑hand exponential (monotone)
    for alpha in np.logspace(-6, 1.5, 40):
        _store(_candidate_left_exponential(alpha, n))

    # left‑hand power (monotone)
    for p in np.linspace(-5.0, 1.0, 30):
        _store(_candidate_left_power(p, n))

    # sine‑shaped bumps
    for pow_ in np.linspace(0.5, 5.0, 30):
        _store(_candidate_sine(pow_, n))

    # random spikes of various sparsities
    for k in [32, 64, 128, 256, 512, 1024]:
        for _ in range(5):
            _store(_candidate_spike(k, n, random=True))

    # inverse‑quadratic shape
    _store(_candidate_inv_quad(n))

    # warm start from a previously stored global variable, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() > 1e-12:
            w = w / w.sum()
            _store(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # Keep only the top few seeds for the costly refinement stage
    candidates.sort(key=lambda x: x[0])
    top_seeds = [vec for (_, vec) in candidates[:5]]

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max + L₂) – multiple restarts
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        remaining = deadline - time.time()
        # allocate roughly 35 % of the remaining budget to Adam, split evenly
        per_restart = remaining * 0.35 / max(1, len(top_seeds))
        for idx, start_vec in enumerate(top_seeds, 1):
            if time.time() >= deadline - 30:
                break
            print(f"[Adam] restart {idx}/{len(top_seeds)}")
            adam_deadline = time.time() + per_restart
            a_adam, val_adam = _adam_optimize(
                start_vec,
                n,
                fft_len,
                adam_deadline,
                max_iters=600_000,
                lam_start=0.05,
                lam_factor=1.01,
                l2_weight=0.15,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Adam] improved global best to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 10:
        print("[Subgrad] exhaustive optimisation")
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=15_000_000,
            eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved global best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing fine‑tune
    # ------------------------------------------------------------------
    if time.time() < deadline - 1:
        print("[Anneal] final fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=500_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved global best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Return a valid non‑negative list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final score (evaluator missing).")

    return best_seq.tolist()
