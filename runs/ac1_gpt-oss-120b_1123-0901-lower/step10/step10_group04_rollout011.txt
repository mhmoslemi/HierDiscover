"""
Propose a step‑function (non‑negative sequence) with a low
2·n·max(conv)/(sum a)² value.

The routine works as follows:

1. **Broad parametric sweep** – a rich collection of analytic windows
   (power‑law, (stretched) exponential, beta, Gaussian, Hann/Hamming/
   Blackman/Kaiser) together with random Dirichlet / spike patterns.
   All candidates are normalised to unit sum and evaluated with an
   FFT‑based convolution; the best few are retained.

2. **Adam‑type refinement** – each retained candidate is optimised with
   an Adam optimiser on a *smooth‑max* surrogate of the max‑norm plus a
   quadratic penalty that pushes the convolution towards a uniform
   distribution.  A strong L2‑weight (≈0.5) and a slowly increasing
   temperature (lam) are used to drive the solution toward a flat
   self‑convolution.

3. **Exponentiated‑gradient sub‑gradient** – the best vector from the
   Adam stage is further refined by exact sub‑gradient descent on the
   true max‑norm using the classic exponentiated‑gradient update
   (mirror‑descent on the probability simplex).

4. **Optional annealing** – a light simulated‑annealing pass transfers
   small mass fractions between random positions to escape any remaining
   shallow local minima.

All stages respect the supplied time budget (≈1000 s).  The final
sequence is clipped to the evaluator’s admissible range and returned
as a plain Python list.
"""

import time
import numpy as np
import heapq

# -------------------------------------------------------------
# FFT utilities
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear self‑convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for gradients (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# -------------------------------------------------------------
# Exact objective (no clipping)
# -------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(conv)/(sum a)² for a (sum a > 0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators (all return a normalised non‑negative vector)
# -------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_gaussian(sigma: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    a = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    a /= a.sum()
    return a


def _candidate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4.0 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


def _candidate_kaiser(beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = 2.0 * i / (n - 1) - 1.0  # map to [-1,1]
    w = np.i0(beta * np.sqrt(1.0 - x * x)) / np.i0(beta)
    w = np.clip(w, 0.0, None)
    w /= w.sum()
    return w


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# -------------------------------------------------------------
# Adam optimiser (smooth‑max + strong L2 regularisation)
# -------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
    lam_start: float = 0.01,
    lam_factor: float = 1.01,
    l2_weight: float = 0.5,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact conv for tracking
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L2 uniformity part -----
        resid = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(resid, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Exact‑max exponentiated‑gradient sub‑gradient
# -------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active (maximal) indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Gentle simulated‑annealing fine‑tuning (optional)
# -------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# -------------------------------------------------------------
# Main routine
# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5  # safety margin

    # Choose a dimension large enough to allow a flatter convolution.
    n = 8192
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    best_seq = None
    best_val = float("inf")
    top_k = 12
    top_candidates = []                     # min‑heap (worst on top)

    def _push_candidate(a: np.ndarray):
        nonlocal best_seq, best_val, top_candidates
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new global best {val:.6f}")
        # keep a bounded heap of the best vectors
        if len(top_candidates) < top_k:
            heapq.heappush(top_candidates, (-val, a.copy()))
        else:
            # heap stores negative value (so the largest value is on top)
            if -val > top_candidates[0][0]:
                heapq.heapreplace(top_candidates, (-val, a.copy()))

    # ---------------------------------------------------------
    # 1) Parametric sweep – diverse analytic families
    # ---------------------------------------------------------
    # Power‑law (both negative & positive exponents)
    for p in np.linspace(-6.0, 6.0, 48):
        if time.time() >= deadline - 5:
            break
        _push_candidate(_candidate_power(p, n))

    # Symmetric exponential
    for alpha in np.logspace(-8, 2, 30):
        if time.time() >= deadline - 5:
            break
        _push_candidate(_candidate_exponential(alpha, n))

    # Stretched exponential
    alphas = np.logspace(-4, 0.5, 15)
    betas = np.linspace(0.2, 3.0, 12)
    for a_alpha in alphas:
        for b_beta in betas:
            if time.time() >= deadline - 5:
                break
            _push_candidate(_candidate_stretched(a_alpha, b_beta, n))

    # Beta distributions
    for _ in range(80):
        if time.time() >= deadline - 5:
            break
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _push_candidate(_candidate_beta(a_par, b_par, n))

    # Gaussian (centered)
    for sigma_frac in np.linspace(0.05, 0.3, 12):
        if time.time() >= deadline - 5:
            break
        sigma = sigma_frac * n
        _push_candidate(_candidate_gaussian(sigma, n))

    # Classic windows
    _push_candidate(_candidate_hann(n))
    _push_candidate(_candidate_hamming(n))
    _push_candidate(_candidate_blackman(n))
    for beta in np.linspace(2.0, 10.0, 7):
        if time.time() >= deadline - 5:
            break
        _push_candidate(_candidate_kaiser(beta, n))

    # Random Dirichlet (different concentration)
    for alpha in [0.1, 0.5, 1.0, 2.0]:
        for _ in range(5):
            if time.time() >= deadline - 5:
                break
            a = np.random.dirichlet(np.full(n, alpha))
            _push_candidate(a)

    # Spike patterns (varying sparsities)
    for k in [64, 128, 256, 512, 1024]:
        for _ in range(4):
            if time.time() >= deadline - 5:
                break
            _push_candidate(_candidate_spike(k, n, random=True))

    # Warm‑start from a previously stored construction if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _push_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ---------------------------------------------------------
    # 2) Adam refinement – strong L2 drive toward a flat conv
    # ---------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        # allocate roughly 35 % of remaining budget across top candidates
        remaining = deadline - time.time()
        allocate = remaining * 0.35
        per_cand = allocate / max(1, len(top_candidates))
        print("[Phase‑Adam] refining top candidates")
        for neg_val, a0 in top_candidates:
            if time.time() >= deadline - 20:
                break
            cand_deadline = min(deadline - 5, time.time() + per_cand)
            a_adam, val_adam = _adam_optimize(
                a0,
                n,
                fft_len,
                cand_deadline,
                max_iters=5_000_000,
                lam_start=0.01,
                lam_factor=1.01,
                l2_weight=0.5,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Adam] improved to {val_adam:.6f}")

    # ---------------------------------------------------------
    # 3) Exact‑max sub‑gradient (exponentiated‑gradient)
    # ---------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑Subgrad] exact max sub‑gradient optimisation")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=15_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ---------------------------------------------------------
    # 4) Optional annealing fine‑tune
    # ---------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final simulated‑annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ---------------------------------------------------------
    # Final sanitisation and evaluation
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
