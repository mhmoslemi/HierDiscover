"""
Hybrid smoothed‑max gradient descent optimizer.

The algorithm works with a fixed length (default = 4096, same as the
reference solution) and keeps the mass normalised to 1 (the objective
is invariant under scaling).  It builds a modest pool of warm‑starts
(uniform, Dirichlet draws, monotone‑sorted Dirichlets, block‑constant
vectors and the previous best solution if available).  For each start
it runs a continuation‑method optimiser:

    * an “soft‑max’’ of the convolution is used as a smooth surrogate
      for the ‖a*a‖∞ term; the smoothing parameter β is increased
      from a tiny value (≈1) to a large one (≈500) so that the method
      first follows a gently‑shaped landscape and later focuses on the
      true peak.

    * the gradient of the smooth surrogate is obtained analytically
      via a single cross‑correlation (FFT‑based, O(N log N)).
      The update follows an exponentiated‑gradient step
      a←a·exp(‑η·∇) and the vector is re‑normalised to keep
      ∑a_i = 1 and a_i ≥ 0.

    * a simple Adam‑like schedule (η ∝ 1/√t) gives a rapidly
      decreasing step‑size while still allowing occasional large
      moves early on.

The loop respects a global deadline (default 1000 s).  The best
sequence seen over all restarts is returned as a plain Python list.
"""

import time
import numpy as np

# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the gradient."""
    # rev_a is a reversed – this realises correlation with w
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _smooth_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """
    Soft‑max weights for a given β.
    w_k ∝ exp(β·conv_k).  The largest entry gets most weight,
    but for small β the distribution is almost uniform.
    """
    # shift to avoid overflow
    m = conv.max()
    exp_term = np.exp(beta * (conv - m))
    w = exp_term / exp_term.sum()
    return w


# -------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    evaluate_sequence(seq).  The routine respects a total time
    budget (≈ 1000 s) and returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---------------------------------------------------------
    # problem size – the reference uses n = 4096
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)      # 8192
    corr_fft_len = _next_pow2(3 * n - 2)      # 12288

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ---------------------------------------------------------
    # 1️⃣  Build a modest pool of diversified starters
    pool: list[np.ndarray] = []

    # warm‑start from the previous best if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # uniform – a useful baseline
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a few Dirichlet draws (different concentrations)
    for alpha in (0.1, 0.3, 0.7, 1.5, 3.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # monotone decreasing draws – sort a Dirichlet in decreasing order
    for alpha in (0.5, 1.0, 2.0):
        v = np.random.dirichlet(alpha * np.ones(n))
        pool.append(np.sort(v)[::-1])

    # piece‑wise constant block vectors (8, 16, 32 blocks)
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        a = a / a.sum()
        pool.append(a)

    # sparse spikes – useful to explore extreme shapes
    for k in (64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        a = a / a.sum()
        pool.append(a)

    # ---------------------------------------------------------
    # 2️⃣  Quick evaluation of starters – keep the best so far
    def _objective_from_seq(seq: np.ndarray) -> float:
        """Exact objective 2·n·max(conv) for a normalised (∑=1) sequence."""
        conv = _full_convolution(seq, conv_fft_len)
        return float(2.0 * n * conv.max())

    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        val = _objective_from_seq(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{idx:02d}] new best {val:.6f}")

    # ---------------------------------------------------------
    # 3️⃣  Smoothed‑max gradient descent (continuation)
    # -----------------------------------------------------------------
    # hyper‑parameters (tuned empirically)
    max_outer_iters = 40            # how many restarts
    max_inner_iters = 8000          # iterations per restart (adjusted by time)
    beta_start = 0.8                # start of smoothing (close to uniform)
    beta_end = 500.0                # final β – approximates hard max
    eta0 = 0.7                      # base learning rate for EG step
    eps = 1e-12                     # to avoid divide‑by‑zero

    outer = 0
    while time.time() < deadline and outer < max_outer_iters:
        # pick a random starter (biased towards current best)
        if best_seq is not None and np.random.rand() < 0.4:
            a = best_seq.copy()
        else:
            a = pool[np.random.randint(len(pool))].copy()

        # modest random perturbation before each run – helps escape flat regions
        if np.random.rand() < 0.6:
            i, j = np.random.choice(n, size=2, replace=False)
            delta = 0.3 * a[i]   # move up to 30 % of mass from i to j
            a[i] -= delta
            a[j] += delta

        # ensure positivity & normalisation
        a = np.maximum(a, eps)
        a = a / a.sum()

        # inner optimisation loop
        for it in range(max_inner_iters):
            if time.time() >= deadline - 0.2:
                break

            # compute convolution (exact) for the true objective
            conv = _full_convolution(a, conv_fft_len)
            cur_val = float(2.0 * n * conv.max())
            if cur_val < best_val:
                best_val = cur_val
                best_seq = a.copy()
                print(f"[Refine outer={outer:02d} it={it:05d}] new best {cur_val:.6f}")

            # smooth‑max weights (β follows a linear schedule)
            beta = beta_start + (beta_end - beta_start) * (it / max_inner_iters)
            w = _smooth_weights(conv, beta)

            # gradient of the smooth surrogate (2·a★w)
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # EG step size – decreasing like 1/√t
            step = eta0 / np.sqrt(it + 1.0)

            # exponentiated‑gradient update (clipping avoids overflow)
            delta = -step * grad
            delta = np.clip(delta, -50.0, 50.0)
            a = a * np.exp(delta)

            # re‑normalise, protect against degenerate zero‑mass
            s = a.sum()
            if s < eps or np.isnan(s) or np.isinf(s):
                a = np.full(n, 1.0 / n, dtype=np.float64)
            else:
                a = a / s

        outer += 1

    # ---------------------------------------------------------
    # 4️⃣  Short stochastic polishing – random mass swaps
    if time.time() < deadline - 0.5 and best_seq is not None:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        cur = float(2.0 * n * conv.max())
        best_local = cur
        best_a_local = a.copy()

        # simple annealing schedule
        T0 = 0.03
        T_end = 1e-5
        moves = 120_000
        for mv in range(moves):
            if time.time() >= deadline - 0.2:
                break
            T = max(T_end, T0 * (1.0 - mv / moves))

            i, j = np.random.randint(n, size=2)
            if i == j or a[i] < eps:
                continue

            # move a fraction of mass from i to j
            frac = np.random.rand() * 0.4   # up to 40 % of the source mass
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = float(2.0 * n * conv2.max())

            if val2 < best_local:
                best_local = val2
                best_a_local = a2.copy()
                a = a2
                print(f"[Anneal move {mv}] improved to {val2:.6f}")
            else:
                # Metropolis acceptance
                if np.random.rand() < np.exp(-(val2 - cur) / T):
                    a = a2
                # otherwise keep old a
            cur = val2

        if best_local < best_val:
            best_val = best_local
            best_seq = best_a_local.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ---------------------------------------------------------
    # 5️⃣  Final sanity & return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s hard limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
