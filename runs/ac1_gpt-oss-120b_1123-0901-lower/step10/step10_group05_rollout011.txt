def propose_candidate(seed=0, budget_s=1000):
    import time, numpy as np
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.1

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    best_seq = None
    best_val = float('inf')

    # add warm start if exists
    if 'height_sequence_1' in globals():
        warm = globals()['height_sequence_1']
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            w = np.resize(w, n)  # pad/truncate maybe not needed. Use np.pad
        w = np.clip(w, 0, 1000)
        if w.sum() >= 0.01:
            w = w / w.sum()
            val = objective(w, n, conv_fft_len)
            best_seq = w.copy()
            best_val = val
            print(f"[Warm start] value {val:.6f}")

    # Stage 1: Exhaustive grid of param families
    # Beta grid:
    xs = np.linspace(0, 1, n, dtype=np.float64)
    for p in np.logspace(-2, 1.5, 20): # from 0.01 to ~31.6
        for q in np.logspace(-2, 1.5, 20):
            shape = (xs**p) * ((1-xs)**q)
            if shape.sum() == 0:
                continue
            shape = shape / shape.sum()
            val = objective(shape, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Beta grid(p={p:.3g},q={q:.3g})] new best {val:.6f}")

    # Power law grid:
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-8, -0.05, 50):
        # shape = prod ** alpha
        # use logs to avoid overflow.
        shape = np.exp(alpha * np.log(prod))
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum()
        val = objective(shape, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
            print(f"[Power α={alpha:.3f}] new best {val:.6f}")

    # Random Dirichlet draws
    for conc in [0.01,0.02,0.05,0.1,0.2,0.5,1.0]:
        for _ in range(30):
            shape = np.random.dirichlet(conc * np.ones(n))
            val = objective(shape, n, conv_fft_len)
            if val < best_val:
                best_val = val
                best_seq = shape.copy()
                print(f"[Dirichlet conc={conc}] new best {val:.6f}")

    # now we have best_seq from coarse search

    # Stage 2: parameter local search for best Beta shape
    # Identify if best_seq looks like Beta? We may just perform generic param random search.
    # We'll simply run random proposals for remaining time. 
    it = 0
    while time.time() < deadline - 5:
        # randomly pick shape generator
        choice = np.random.randint(4)
        if choice == 0: # beta
            p = 10**np.random.uniform(-2, 1.5)
            q = 10**np.random.uniform(-2, 1.5)
            shape = (xs**p) * ((1-xs)**q)
        elif choice == 1: # power
            alpha = np.random.uniform(-10, -0.01)
            shape = np.exp(alpha * np.log(prod))
        elif choice == 2: # mix two betas
            p1 = 10**np.random.uniform(-2, 1.5)
            q1 = 10**np.random.uniform(-2, 1.5)
            p2 = 10**np.random.uniform(-2, 1.5)
            q2 = 10**np.random.uniform(-2, 1.5)
            w = np.random.rand()
            shape = w * (xs**p1) * ((1-xs)**q1) + (1-w) * (xs**p2) * ((1-xs)**q2)
        else: # Dirichlet
            conc = 10**np.random.uniform(-2, 1)  # 0.01 to 10
            shape = np.random.dirichlet(conc * np.ones(n))
        if shape.sum() == 0:
            continue
        shape = shape / shape.sum()
        val = objective(shape, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = shape.copy()
            print(f"[Random {it}] new best {val:.6f}")
        it += 1

    # Stage 3: Exponentiated gradient refinement
    # Use best_seq as start; run steps up to remaining time
    # similar to earlier but maybe more iterations

    a = best_seq.copy()
    step_const = 0.9
    it = 0
    while time.time() < deadline - 2:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[EG it={it}] new best {val:.6f}")
        # active set
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, _next_pow2(3*n-2))
        step = step_const / np.sqrt(it+1)
        # ensure no underflow
        a_new = a * np.exp(-step * grad)  # mirror descent
        s = a_new.sum()
        if s > 0:
            a_new = a_new / s
            a = a_new
        it += 1

    # Stage 4: Simulated annealing polishing
    a = best_seq.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val_sa = val
    best_a_sa = a.copy()
    T_start = 0.05
    T_end = 1e-6
    moves = int(2e5)  # 200k moves
    for mv in range(moves):
        if time.time() > deadline - 0.5:
            break
        T = max(T_end, T_start * (1.0 - mv / moves))
        # choose two indices i != j
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j:
            continue
        # move a random fraction of mass from i to j
        mass_i = a[i]
        if mass_i <= 0:

        delta = mass_i * np.random.rand() * 0.5
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()
        if val2 < best_val_sa:
            best_val_sa = val2
            best_a_sa = a2.copy()
            a = a2
            print(f"[SA move {mv}] new best {val2:.6f}")
        else:
            # maybe accept
            if np.random.rand() < np.exp(-(val2 - val)/T):
                a = a2
        val = val2
    if best_val_sa < best_val:
        best_val = best_val_sa
        best_seq = best_a_sa.copy()
        print(f"[SA final] best {best_val:.6f}")

    # Final clip
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Returning best value {best_val:.6f} (len={len(best_seq)})")
    return best_seq.tolist()

Hybrid optimizer for the max‑convolution bound.

The search proceeds in four stages:

   analytically‑parameterised families (U‑shaped beta PDFs,

    fa = np.fft.rfft(a, fft_len)

    return conv[: 2 * a.shape[0] - 1]

    """Cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]

                log_shape = alpha * np.log(prod)
                shape = np.exp(log_shape)
            shape = np.nan_to_num(shape, nan=0.0, posinf=1e30, neginf=0.0)

        else:                    # mixture of two betas
            p1 = 10 ** np.random.uniform(-2.5, 0.5)
            q1 = 10 ** np.random.uniform(-2.5, 0.5)
            p2 = 10 ** np.random.uniform(-2.5, 0.5)
            q2 = 10 ** np.random.uniform(-2.5, 0.5)
            w = np.random.rand()
            with np.errstate(divide="ignore", over="ignore", invalid="ignore"):
                log1 = (p1 - 1.0) * np.log(xs_adj) + (q1 - 1.0) * np.log(1.0 - xs_adj)
                log2 = (p2 - 1.0) * np.log(xs_adj) + (q2 - 1.0) * np.log(1.0 - xs_adj)
                shape = w * np.exp(log1) + (1.0 - w) * np.exp(log2)
            shape = np.nan_to_num(shape, nan=0.0, posinf=1e30, neginf=0.0)

        if shape.sum() < 1e-12:
            continue
        a = shape / shape.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Random it={it}] new best {val:.6f}")
        it += 1

    # ------------------------------------------------------------------
    # 3️⃣ Mirror‑descent (exponentiated‑gradient) refinement
    # -------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    a = best_seq.copy()
    step_const = 0.9
    eg_it = 0
    while time.time() < deadline - 4:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[EG it={eg_it}] new best {val:.6f}")

        # active set of indices attaining the maximum
        active = np.isclose(conv, max_c, atol=1e-14)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule η_t = c / √t
        eta = step_const / np.sqrt(eg_it + 1.0)
        a_new = a * np.exp(-eta * grad)          # multiplicative update
        s = a_new.sum()
        if s > 0.0:
            a = a_new / s
        eg_it += 1

    # ------------------------------------------------------------------
    # 4️⃣ Simulated‑annealing polish (mass‑transfer moves)
    # ----------------------------------------------------
    a = best_seq.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val_sa = val
    best_a_sa = a.copy()

    T_start = 0.07
    T_end = 1e-6
    max_moves = 180_000                     # generous but bounded

    for mv in range(max_moves):
        if time.time() > deadline - 1.0:
            break
        # linear temperature schedule
        T = max(T_end, T_start * (1.0 - mv / max_moves))

        i = np.random.randint(n)
        j = np.random.randint(n)
        if i == j or a[i] <= 1e-14:
            continue

        # move a random fraction of mass from i to j (max 40 % of a[i])
        delta = a[i] * (np.random.rand() * 0.4)
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val_sa:
            best_val_sa = val2
            best_a_sa = a2.copy()
            a = a2
            print(f"[SA move {mv}] improved to {val2:.6f}")
        else:
            # Metropolis acceptance for uphill moves
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
        val = val2

    if best_val_sa < best_val:
        best_val, best_seq = best_val_sa, best_a_sa.copy()
        print(f"[SA final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final clean‑up and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respects evaluator limits
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
