"""
Hybrid optimisation with random restarts and a final discrete‐move polishing.

The routine:
1. Loads the global warm‑start `height_sequence_1` (if present) or falls back to a
   uniform distribution of length n=4096 (normalised to sum‑one).
2. Repeatedly performs *trials* until the time budget is exhausted.
   Each trial starts from either the warm‑start (first trial) or a fresh
   Dirichlet random vector with a randomly chosen concentration parameter.
3. Inside a trial a three‑stage optimiser is run:
   • Stage A – Adam on a smooth‑max surrogate of the max‑convolution.
   • Stage B – Exact‑max sub‑gradient descent using exponentiated gradients.
   • Stage C – A fast coordinate‑move hill‑climber (random “mass‑shifts”)
     that further reduces the peak of the convolution.
4. After a trial the best vector found so far is kept.  A short cyclic‑shift
   scan is performed on the overall best vector before returning.
5. The function always returns a plain Python list of non‑negative numbers
   and respects the supplied time budget (≈ 1000 s).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback – uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation 2·n·max(a*a) for a with sum(a)=1.
    (The denominator Σa² cancels because we keep Σa = 1 throughout.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max conversion
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          corr_fft_len: int,
                          deadline: float,
                          max_iters: int = 10_000_000,
                          eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _random_move_optimize(a0: np.ndarray,
                          n: int,
                          conv_fft_len: int,
                          deadline: float,
                          move_frac: float = 0.02,
                          max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climber: repeatedly transfer a small fraction of mass from a
    randomly chosen source index to a random destination.  The move is kept
    only if it reduces the exact objective.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_moves:
        # pick source with probability proportional to its mass
        src = np.random.choice(n, p=a)
        if a[src] <= 0.0:
            continue
        dst = np.random.randint(n)
        if src == dst:
            continue

        delta = move_frac * a[src]
        if delta <= 1e-15:
            continue

        a_new = a.copy()
        a_new[src] -= delta
        a_new[dst] += delta

        # maintain simplex – sum unchanged, non‑negativity already ensured
        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a_new.copy()
            a = a_new
            if it % 10_000 == 0:
                print(f"[Move it={it:06d}] improved to {val:.6f}")
        it += 1

    return best_a, best_val


def _cyclic_shift_refine(a: np.ndarray,
                         n: int,
                         conv_fft_len: int) -> np.ndarray:
    """Try a handful of deterministic cyclic shifts and keep the best."""
    best_a = a
    best_val = _objective(a, n, conv_fft_len)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2,
              5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")
    return best_a


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns the best sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    # safety margin to avoid overshoot
    overall_deadline = time.time() + budget_s - 0.3

    # --------------------------------------------------------------
    # Problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Global best holder
    global_best_a = None
    global_best_val = float("inf")

    trial_idx = 0
    # Keep trying new starts while we have enough time left for a meaningful trial
    min_trial_time = 30.0                     # seconds
    while time.time() + min_trial_time < overall_deadline:
        trial_idx += 1
        # ----- initialise a starting vector for this trial -----
        if trial_idx == 1:
            # first trial: warm start if available, else uniform
            warm = globals().get("height_sequence_1")
            if warm is not None:
                a0 = np.asarray(warm, dtype=np.float64)
                if a0.size > n:
                    a0 = a0[:n]
                elif a0.size < n:
                    a0 = np.pad(a0, (0, n - a0.size), mode="constant")
            else:
                a0 = np.full(n, 1.0 / n, dtype=np.float64)
        else:
            # later trials: random Dirichlet with a random concentration parameter
            alpha = np.random.uniform(0.3, 3.0)          # controls how peaked the start is
            a0 = np.random.dirichlet(np.full(n, alpha))
        # normalise to the simplex
        a0 = np.maximum(a0, 0.0)
        a0_sum = a0.sum()
        if a0_sum < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
            a0_sum = a0.sum()
        a0 /= a0_sum

        # ----- allocate the time budget for this trial -----
        remain = overall_deadline - time.time()
        # we split the remaining time for the trial in three parts:
        #   20 % Adam, 60 % sub‑gradient, 20 % random moves (if time permits)
        t_adam = time.time() + 0.20 * remain
        t_sub  = time.time() + 0.80 * remain      # sub‑gradient will take the bulk
        t_move = overall_deadline                  # whatever is left after sub‑grad

        # ----- Stage A : Adam optimisation -----
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, t_adam, max_iters=600_000
        )
        if val_adam < global_best_val:
            global_best_val, global_best_a = val_adam, a_adam.copy()
            print(f"[Trial {trial_idx}] Adam gave new global best {val_adam:.6f}")

        # ----- Stage B : Sub‑gradient (exponentiated‑gradient) -----
        a_sub, val_sub = _subgradient_optimize(
            a_adam, n, conv_fft_len, corr_fft_len, t_sub,
            max_iters=12_000_000, eta0=1.2
        )
        if val_sub < global_best_val:
            global_best_val, global_best_a = val_sub, a_sub.copy()
            print(f"[Trial {trial_idx}] Sub‑grad gave new global best {val_sub:.6f}")

        # ----- Stage C : Discrete random‑move polishing (if any time left) -----
        if time.time() < overall_deadline:
            a_move, val_move = _random_move_optimize(
                a_sub, n, conv_fft_len, overall_deadline,
                move_frac=0.02, max_moves=300_000
            )
            if val_move < global_best_val:
                global_best_val, global_best_a = val_move, a_move.copy()
                print(f"[Trial {trial_idx}] Random moves gave new global best {val_move:.6f}")

        # Small safety print
        current_time = time.time()
        print(f"[Trial {trial_idx}] elapsed {current_time - (overall_deadline - budget_s):.1f}s, "
              f"global best {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Final fine‑tuning on the best candidate found so far
    if global_best_a is not None:
        # short final Adam pass (≈5 % of any remaining budget)
        final_dead = min(overall_deadline, time.time() + 0.05 * (overall_deadline - time.time()))
        a_final, val_final = _adam_optimize(
            global_best_a, n, conv_fft_len, corr_fft_len, final_dead,
            max_iters=200_000
        )
        if val_final < global_best_val:
            global_best_val, global_best_a = val_final, a_final.copy()
            print(f"[Final Adam] improved to {val_final:.6f}")

        # cyclic‑shift sweep
        global_best_a = _cyclic_shift_refine(global_best_a, n, conv_fft_len)
        global_best_val = _objective(global_best_a, n, conv_fft_len)
        print(f"[Final] objective after shift sweep {global_best_val:.6f}")

    # --------------------------------------------------------------
    # Clip to allowed range and final sanity check
    if global_best_a is None:
        # Fallback – uniform distribution (should never happen)
        global_best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best = np.clip(global_best_a, 0.0, 1000.0)

    if best.sum() < 0.01:
        best = best + 1e-6
        best /= best.sum()

    # Report using the official evaluator (purely for logging)
    try:
        eval_val = evaluate_sequence(best.tolist())
        print(f"Final evaluation {eval_val:.6f} (length {len(best)})")
    except Exception as exc:
        print(f"Evaluation failed: {exc}")

    return best.tolist()
