def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n - 1)
    corr_fft_len = _next_pow2(3*n - 2)

    elite = []

    # define a helper to attempt add candidate
    def try_candidate(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # param sweep
    for p in np.linspace(-5, -0.05, 30):
        a = _candidate_power(p, n)
        try_candidate(a)
    for alpha in np.logspace(-5, 1.0, 30):
        a = _candidate_exponential(alpha, n)
        try_candidate(a)
    # Add left exponential monotone
    for alpha in np.logspace(-5, 1.2, 30):
        a = _candidate_left_exponential(alpha, n)
        try_candidate(a)
    # Add left power
    for p in np.linspace(-4, 2, 28):
        a = _candidate_left_power(p, n)
        try_candidate(a)
    # Add stretch exponential combos
    for alpha in np.logspace(-4, 0.5, 12):
        for beta in np.linspace(0.3, 3.0, 12):
            a = _candidate_stretched(alpha, beta, n)
            try_candidate(a)
    # add beta distribution grid
    for a_par in [0.05,0.1,0.2,0.5,1,2,5,10]:
        for b_par in [0.05,0.1,0.2,0.5,1,2,5,10]:
            a = _candidate_beta(a_par, b_par, n)
            try_candidate(a)
    # add double gaussian
    for sig in np.logspace(-3, -0.1, 20):
        a = _candidate_double_gaussian(sig, n)
        try_candidate(a)
    # random dirichlet with various concentration
    for c in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        a = np.random.dirichlet(np.full(n, c))
        try_candidate(a)
    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            a = _candidate_spike(k, n, random=True)

    # include a mixture candidate to broaden
    for _ in range(50):
        alpha = np.exp(np.random.uniform(np.log(1e-5), np.log(20)))

        t = np.random.rand()
        a_exp = np.exp(-alpha * np.arange(n, dtype=np.float64))

        best_val = _objective(best_seq, n)

    # Phase 2: refine parameters of top candidate(s) using local hill climbing
    # pick top 2 elites for refinement
    remaining = deadline - time.time()
    for idx in range(min(2, len(elite))):

        cur_a = cur_seq.copy()
        improved = True

            if new_val < cur_val:

        # after local hill climbing, possibly update elite

2. **Local log‑space refinement** – starting from each elite element we
   perform a short stochastic hill‑climb on the soft‑max (logits) of the
   vector.  This adds a cheap perturbation step that often yields a small
   improvement without expensive gradient computation.

3. **Adam on the smooth‑max surrogate** – for the top‑3 elite vectors we
   run a moderate‑length Adam optimiser on a log‑sum‑exp surrogate of the
   convolution maximum.  The temperature is increased gradually, allowing
   the optimiser to escape shallow basins.

4. **Multi‑restart exact‑max sub‑gradient descent** – the current best
   vector (and occasional random Dirichlet starts) are fed to an
   exponentiated‑gradient sub‑gradient method that minimises the true
   ``max(conv)``.  Restarts are performed until the time budget is almost
   exhausted; each run gets roughly 30 s.

5. **Simulated‑annealing fine‑tuning** – a final annealing pass makes tiny
   mass transfers between random indices, accepting uphill moves with a
   decreasing temperature schedule.

All phases respect the global deadline (≈1000 s).  The returned list is
clipped to the allowed range and a final call to ``evaluate_sequence``
verifies the score.
"""

import time
import math
import random
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for sub‑gradient.
    Returns a vector of length len(a) corresponding to the gradient of
    ``∑_k w_k (a*a)_k`` with respect to ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate (log‑sum‑exp) of ``max(conv)``.
    ``a0`` is already normalised to sum = 1.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # parameters for soft‑max (stays on simplex)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.05            # start temperature (smooth‑max sharpness)
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for book‑keeping
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (2·cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
    Works on the simplex via a mirror‑descent step.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – guard against numerical loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Elite handling
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    a valid list of heights.
    """
    np.random.seed(seed)
    random.seed(seed)

    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # --------------------------------------------------------------
    # Problem dimension (power of two for fast FFT)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Wide parametric sweep – keep a few elites
    # --------------------------------------------------------------
    elite = []      # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Symmetric power‑law (negative exponents concentrate at the ends)
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))

    # Symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    # Stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # Beta distributions
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # Monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 30):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # Monotone left‑power
    for p in np.linspace(-4.0, 2.0, 30):
        _maybe_update(_candidate_left_power(p, n))

    # Random spikes of various sparsities
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Dirichlet draws with a range of concentration parameters
    for c in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        _maybe_update(np.random.dirichlet(np.full(n, c)))

    # Simple mixtures of exponential and power‑law (adds extra flexibility)
    for _ in range(50):
        alpha = np.exp(np.random.uniform(np.log(1e-5), np.log(20.0)))
        p = np.random.uniform(-5.0, 2.0)
        t = np.random.rand()
        exp_part = np.exp(-alpha * np.arange(n, dtype=np.float64))
        pow_part = (np.arange(n, dtype=np.float64) + 1) ** p
        mix = t * exp_part + (1.0 - t) * pow_part
        mix = np.maximum(mix, 0.0)
        mix /= mix.sum()
        _maybe_update(mix)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found – using uniform")

    # --------------------------------------------------------------
    # 2) Stochastic refinement on the simplex (logits perturbation)
    # --------------------------------------------------------------
    for idx in range(min(2, len(elite))):
        cur_val, cur_seq = elite[idx]
        cur_a = cur_seq.copy()
        improved = True
        step_factor = 0.30
        while time.time() < deadline - 0.2 and improved:
            improved = False
            logits = np.log(cur_a + 1e-15)
            noise = np.random.randn(n) * step_factor
            new_logits = logits + noise
            new_a = np.exp(new_logits)
            new_a /= new_a.sum()
            new_val = _objective(new_a, n)
            if new_val < cur_val:
                cur_val = new_val
                cur_a = new_a
                improved = True
                step_factor *= 0.9
                print(f"[Refine elite {idx}] improved to {new_val:.6f}")
        _maybe_update(cur_a)
        if cur_val < best_val:
            best_val, best_seq = cur_val, cur_a.copy()
            print(f"[Refine elite {idx}] new overall best {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Adam optimisation on a smooth‑max surrogate (few short runs)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(20.0, 0.2 * remaining)
        print("[Phase‑Adam] short Adam optimisation (multiple starts)")
        for i in range(min(3, len(elite))):
            if time.time() >= adam_deadline:
                break
            a_start = elite[i][1]
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=500_000,
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Phase‑Adam] start {i} improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Long exact‑max sub‑gradient descent with many restarts
    # --------------------------------------------------------------
    while time.time() < deadline - 5.0:
        sub_start = time.time()
        # With 30 % probability start from a fresh random Dirichlet
        if np.random.rand() < 0.3:
            a0 = np.random.dirichlet(np.full(n, 0.1))
        else:
            a0 = best_seq.copy()
        sub_deadline = min(deadline - 2.0, sub_start + 30.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=2_000_000, eta0=0.9
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Subgrad restart] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=200_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and return
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluation failed (evaluate_sequence unavailable).")

    return best_seq.tolist()
