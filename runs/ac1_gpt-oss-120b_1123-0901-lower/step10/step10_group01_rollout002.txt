    max_c = conv.max()

"""
Improved optimiser for the step‑function evaluation problem.

The algorithm works in four phases:

1.  Build a diverse set of start vectors (warm‑start, uniform,
    Dirichlet, Gaussian, spikes …) on the probability simplex.
2.  For each start run a continuation of L‑BFGS‑B optimisations on a
    smooth approximation of the objective:
       f_λ(a) = 2·n·log‑sum‑exp(λ·(a*a)) / λ .
    The smoothing parameter λ is increased (10 → 50 → 200 → 800 → 3000)
    so the optimiser first sees a smooth landscape and then zooms onto
    the true max‑convolution.
    Gradients are obtained analytically via FFT‑based cross‑correlation.
3.  Refine the best solution with a cheap stochastic local‑search that
    moves a small fraction of mass between two coordinates.
4.  If time permits, up‑sample the best 4096‑vector to 8192 points,
    repeat phase 2 on the finer grid and keep the better of the two.

All heavy operations (convolution, cross‑correlation, soft‑max) are
implemented with NumPy/FFT and are fully vectorised.  The code respects
the 1000 s time budget and never returns an empty list.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# Helper utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _softmax(y: np.ndarray) -> np.ndarray:
    """Numerically stable soft‑max."""
    m = np.max(y)
    e = np.exp(y - m)
    return e / np.sum(e)


def _smooth_obj_and_grad(y: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         lam: float):
    """
    Smooth surrogate of the objective and its gradient w.r.t. the logits y.

    f_λ(a) = 2·n·log‑sum‑exp(λ·conv(a))/λ,
    where conv(a)=a*a.
    """
    a = _softmax(y)

    # convolution a*a
    conv = _full_convolution(a, conv_fft_len)

    # smooth max via log‑sum‑exp
    max_c = np.max(conv)
    exp_term = np.exp(lam * (conv - max_c))          # safe shifting
    sum_exp = np.sum(exp_term)
    log_sum_exp = max_c + np.log(sum_exp)           # = log ∑exp(lam·conv)

    obj = 2.0 * n * log_sum_exp / lam               # smooth objective value

    # gradient of smooth max: w = softmax(lam·conv)
    w = exp_term / sum_exp

    # ∇_a ( w·conv ) = 2·(w ★ a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # back‑prop through soft‑max
    avg = np.dot(a, grad_a)
    grad_y = a * (grad_a - avg)

    return obj, grad_y


def _local_search(a: np.ndarray,
                  n: int,
                  deadline: float,
                  max_moves: int = 200_000) -> np.ndarray:
    """
    Simple stochastic hill‑climbing on the simplex.
    Moves a random fraction (≤ 5 %) of mass from i to j.
    """
    best_a = a.copy()
    best_val = float('inf')
    # initial evaluation
    best_val = evaluate_sequence(best_a.tolist())

    moves = 0
    while time.time() < deadline and moves < max_moves:
        i = np.random.randint(n)
        if best_a[i] < 1e-14:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue

        # transfer a small fraction of the weight at i to j
        frac = np.random.rand() * 0.05          # ≤ 5 %
        delta = best_a[i] * frac

        a2 = best_a.copy()
        a2[i] -= delta
        a2[j] += delta

        val = evaluate_sequence(a2.tolist())
        if val < best_val:
            best_val = val
            best_a = a2
            # optional print for progress
            if moves % 20_000 == 0:
                print(f"[Local] move {moves:06d} improved to {val:.6f}")

        moves += 1

    return best_a


def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n))


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _upsample(a_small: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation from len(a_small) to new_n points, re‑normalised."""
    old_n = a_small.shape[0]
    x_old = np.linspace(0, 1, old_n, endpoint=False)
    x_new = np.linspace(0, 1, new_n, endpoint=False)
    a_big = np.interp(x_new, x_old, a_small)
    a_big = np.clip(a_big, 0.0, None)
    s = a_big.sum()
    if s < 1e-14:
        a_big = np.full(new_n, 1.0 / new_n, dtype=np.float64)
    else:
        a_big = a_big / s
    return a_big


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found before the time budget expires.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Base problem size (first optimisation on 4096 points)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Build a pool of start vectors (all are on the simplex)
    pool = []

    # warm start (if a previous construction is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            # linear interpolation to required size
            warm = _upsample(warm, n)
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            pool.append(warm.copy())
            # also try a couple of cyclic shifts of the warm start
            for shift in (n // 4, n // 2, 3 * n // 4):
                pool.append(np.roll(warm, shift))

    # a few deterministic families
    pool.append(_make_uniform(n))
    pool.append(_make_triangle(n))
    for alpha in (0.05, 0.2, 0.8, 1.5, 3.0):
        pool.append(_make_dirichlet(n, alpha))
    for sigma in (0.04, 0.1, 0.25, 0.5):
        pool.append(_make_gaussian(n, sigma))
    for gamma in (0.2, 0.6, 1.5):
        pool.append(_make_exponential(n, gamma))
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float('inf')

    def try_update(a_vec: np.ndarray):
        """Evaluate a_vec and keep it if it improves the global best."""
        nonlocal best_seq, best_val
        val = evaluate_sequence(a_vec.tolist())
        if val < best_val:
            best_val = val
            best_seq = a_vec.copy()
            print(f"[Update] New best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – L‑BFGS continuation on a smooth surrogate
    lam_schedule = (10.0, 50.0, 200.0, 800.0, 3000.0)
    max_iter_per_stage = 800   # ~max function‑evaluations per *lam*

    for start_id, a0 in enumerate(pool):
        if time.time() > deadline:
            break
        # initialise logits (softmax parameters)
        eps = 1e-12
        y = np.log(a0 + eps)

        for lam in lam_schedule:
            if time.time() > deadline:
                break
            # L‑BFGS‑B on the smooth objective
            try:
                res = minimize(_smooth_obj_and_grad,
                               y,
                               args=(n, conv_fft_len, corr_fft_len, lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter_per_stage,
                                        'ftol': 1e-12,
                                        'disp': False})
            except Exception as exc:
                print(f"[L‑BFGS] exception (lam={lam}): {exc}")
                break

            y = res.x  # warm‑start the next λ‑stage with the new logits
            a_opt = _softmax(y)
            try_update(a_opt)

        # After the λ‑schedule keep the final a_opt for possible later use
        # (no additional work needed here).

    # ------------------------------------------------------------------
    # Phase II – cheap stochastic local search (moves up to 5 % of a mass)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Local] Starting stochastic local refinement")
        best_seq = _local_search(best_seq, n, deadline, max_moves=200_000)

    # ------------------------------------------------------------------
    # Phase III – optional up‑sampling to a finer grid (8192 points)
    if best_seq is not None and time.time() + 30.0 < deadline:
        try:
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            print("[Upsample] Interpolating to 8192 points")
            a_big = _upsample(best_seq, n_big)

            # run the same continuation of L‑BFGS on the finer grid
            y_big = np.log(a_big + 1e-12)
            for lam in lam_schedule:
                if time.time() > deadline:
                    break
                res = minimize(_smooth_obj_and_grad,
                               y_big,
                               args=(n_big,
                                     conv_fft_len_big,
                                     corr_fft_len_big,
                                     lam),
                               method='L-BFGS-B',
                               jac=True,
                               options={'maxiter': max_iter_per_stage,
                                        'ftol': 1e-12,
                                        'disp': False})
                y_big = res.x

            a_big_opt = _softmax(y_big)
            # evaluate the finer‑grid candidate (the objective includes the larger n)
            val_big = evaluate_sequence(a_big_opt.tolist())
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big_opt.copy()
                print(f"[Upsample] New best after fine grid {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] failure: {e}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
