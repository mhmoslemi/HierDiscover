"""
Propose a low‑score step‑function.

The algorithm consists of four stages:

1. **Broad parametric sweep** – a large collection of simple families
   (exponential, power‑law, stretched‑exponential, Gamma‑like, Beta,
   double‑Gaussian, random spikes, mixtures of exponentials, etc.) is
   generated.  Every candidate is normalised to sum = 1 and its exact
   objective 2·n·max(a*a) is computed via an FFT‑based convolution.
   The best shape is retained.

2. **Random mixture search** – the best shape is refined by stochastically
   sampling a few low‑dimensional families that mix two basic patterns
   (exponential + uniform, two exponentials, exponential + power‑law).
   Millions of cheap candidates are evaluated while reserving time for
   the final optimisation phases.

3. **Exact‑max exponentiated‑gradient descent** – a long sub‑gradient run
   (mirror‑descent on the simplex) drives the solution toward a local
   optimum of the true max‑convolution objective.

4. **Optional simulated‑annealing fine‑tuning** – a cheap annealer swaps
   small fractions of mass between entries to escape tiny local traps.

All stages respect the global time budget (default 1000 s) and the final
return is a list of non‑negative coefficients ready for `evaluate_sequence`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β)+exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----- mixture families (new) -----------------------------------------
def _candidate_mix_exp_uniform(alpha: float, w: float, n: int) -> np.ndarray:
    """Mix exponential (symmetric) with uniform: w·exp + (1‑w)·1."""
    i = np.arange(n, dtype=np.float64)
    exp_part = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    uniform = np.ones_like(exp_part)
    a = w * exp_part + (1.0 - w) * uniform
    a /= a.sum()
    return a


def _candidate_mix_two_exp(alpha1: float, alpha2: float, w: float, n: int) -> np.ndarray:
    """Two symmetric exponentials mixed with weight w."""
    i = np.arange(n, dtype=np.float64)
    e1 = np.exp(-alpha1 * i) + np.exp(-alpha1 * (n - 1 - i))
    e2 = np.exp(-alpha2 * i) + np.exp(-alpha2 * (n - 1 - i))
    a = w * e1 + (1.0 - w) * e2
    a /= a.sum()
    return a


def _candidate_mix_exp_power(alpha: float, p: float, w: float, n: int) -> np.ndarray:
    """Mix symmetric exponential with power‑law tail."""
    i = np.arange(n, dtype=np.float64)
    exp_part = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    power_part = (i + 1) ** p + (n - i) ** p
    power_part = np.maximum(power_part, 0.0)
    a = w * exp_part + (1.0 - w) * power_part
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate (log‑sum‑exp)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient (mirror descent on the simplex)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active indices attaining the max (with tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # Exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # Renormalise safely
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing style local search."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # Transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    a valid sequence of coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Helper to possibly update the current best
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (similar to the reference implementation)
    # ----- symmetric families ------------------------------------------------
    for p in np.linspace(-6.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-4, 2.5, 35):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # ----- monotone families -------------------------------------------------
    for alpha in np.logspace(-4, 2.0, 30):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-6.0, 2.0, 30):
        _maybe_update(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 12):
        for alpha in np.logspace(-4, 1.5, 25):
            _maybe_update(_candidate_left_stretched(alpha, beta, n))

    # ----- Gamma‑like --------------------------------------------------------
    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            _maybe_update(_candidate_gamma(shape, scale, n))

    # ----- Beta‑distribution -------------------------------------------------
    a_pars = np.logspace(-2, 0.5, 12)   # 0.01 … 3.2
    b_pars = np.logspace(0, 1.0, 12)    # 1 … 10
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # ----- double Gaussian ---------------------------------------------------
    for sig in np.logspace(-3, -0.2, 15):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # ----- random spikes -----------------------------------------------------
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # ----- mixtures of two exponentials (random) ----------------------------
    for _ in range(200):
        alpha1 = 10**np.random.uniform(-4, 2)
        alpha2 = 10**np.random.uniform(-4, 2)
        w1 = np.random.rand()
        a_mix = w1 * _candidate_left_exponential(alpha1, n) + \
                (1 - w1) * _candidate_left_exponential(alpha2, n)
        a_mix = np.maximum(a_mix, 0.0)
        a_mix /= a_mix.sum()
        _maybe_update(a_mix)

    # ----- Dirichlet random vectors (various concentrations) ---------------
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # ----- warm start from a previously stored global -----------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Random mixture search (time‑boxed)
    # Reserve ~200 s for the final sub‑gradient stage + annealing.
    mix_deadline = deadline - 250.0
    if mix_deadline > time.time():
        print("[MixSearch] starting random mixture search")
        while time.time() < mix_deadline:
            typ = np.random.choice(["exp_uniform", "two_exp", "exp_power"])
            if typ == "exp_uniform":
                alpha = 10**np.random.uniform(-4, 3)          # 1e‑4 … 1e3
                w = np.random.rand()
                a = _candidate_mix_exp_uniform(alpha, w, n)
            elif typ == "two_exp":
                alpha1 = 10**np.random.uniform(-4, 3)
                alpha2 = 10**np.random.uniform(-4, 3)
                w = np.random.rand()
                a = _candidate_mix_two_exp(alpha1, alpha2, w, n)
            else:  # exp_power
                alpha = 10**np.random.uniform(-4, 3)
                p = np.random.uniform(-6.0, 2.0)
                w = np.random.rand()
                a = _candidate_mix_exp_power(alpha, p, w, n)

            _maybe_update(a)

    # ------------------------------------------------------------------
    # 3) Short Adam refinement (smooth‑max) --------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑Adam] short Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 30.0)  # up to ~30 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Long sub‑gradient descent (exact max) -------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning ---------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
