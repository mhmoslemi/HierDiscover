def _lowdim_optimize(n, conv_fft_len, corr_fft_len, deadline,
                     L=64, max_iters=300_000):
    # initialize basis phi: constant + cosine basis
    i = np.arange(n)
    phi = np.empty((n, L), dtype=np.float64)
    phi[:,0] = 1.0
    for l in range(1, L):
        phi[:,l] = np.cos(2 * np.pi * l * i / n)  # real basis
    # initialize c to zeros
    c = np.zeros(L)
    m = np.zeros(L)
    v = np.zeros(L)
    beta1=0.9; beta2=0.999; eps=1e-8
    lr = 0.05
    best_a = None; best_val = np.inf

    while time.time()<deadline and max_iters>0:
        # compute logits and softmax a
        logits = phi @ c
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # compute conv and max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LowDim it={max_iters}] new best {val:.6f}")

        # gradient of max (exact)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)  # gradient of max_c

        # backprop through softmax
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # gradient w.r.t c
        grad_c = phi.T @ grad_logits

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_c
        v = beta2 * v + (1 - beta2) * (grad_c **2)
        m_hat = m / (1 - beta1 ** (max_iters+1))
        v_hat = v / (1 - beta2 ** (max_iters+1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        c -= step

        max_iters -= 1

    return best_a, best_val

"""
Hybrid global–local optimizer for the step‑function problem.

The search consists of four distinct phases that together respect a
≈ 1000 s time budget:

1. **Diverse initial pool** – uniform, Dirichlet draws, exponential,
   sinusoidal, piece‑wise constant, sparse spikes and a family of
   Beta‑shaped (U‑ or bell‑shaped) candidates (including the arcsine
   law).  The previous best sequence (global variable ``height_sequence_1``)
   is also injected.

2. **Low‑dimensional Fourier‑basis Adam** – the distribution is
   parametrised via a soft‑max over a small set of cosine bases
   (≈ 64 parameters).  Adam minimises the exact max‑convolution
   (sub‑gradient via the active convolution index).  This quickly finds
   a promising “shape” that is later refined.

3. **Projected max‑convolution descent** – exact sub‑gradient steps on
   the full‑dimension vector, projected onto the probability simplex.
   A handful of the best candidates from the pool (including the result
   of phase 2) are polished by this method.

4. **Spectral flattening + final refinements** – (i) force the FFT
   magnitude to be (almost) constant while preserving the DC component;
   (ii) a short projected‑gradient pass; (iii) a simulated‑annealing
   local search.  After each phase the best objective value is kept.

The objective is
    2·n·max(conv(a,a)) / (sum(a)**2)
and the algorithm always returns a non‑empty sequence before the
deadline (otherwise the evaluator would give –∞).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback to uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv)/sum(a)^2."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    return 2.0 * n * conv.max() / (np.sum(a) ** 2)

# Candidate factories
def _beta_shape(n: int, alpha: float, beta: float) -> np.ndarray:
    """Discrete Beta(α,β) density on {0,…,n‑1}, normalised to sum = 1."""
    i = np.arange(n, dtype=np.float64) + 0.5
    x = i / n
    eps = 1e-12
    # Clip away from 0/1 for negative exponents
    x = np.clip(x, eps, 1 - eps)
    shape = (x ** alpha) * ((1 - x) ** beta)
    if shape.sum() == 0:
        shape = np.full(n, 1.0 / n)
    else:
        shape /= shape.sum()
    return shape


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of starter vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws (different concentrations)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    s = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    s = np.maximum(s, 0.0)
    s /= s.sum()
    cand.append(s)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Beta‑family (including arcsine with α=β=‑0.5)
    alphas = (-0.8, -0.5, -0.2, 0.2, 0.5)
    betas = (-0.8, -0.5, -0.2, 0.2, 0.5)
    for a in alphas:
        for b in betas:
            cand.append(_beta_shape(n, a, b))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates made of a few equally‑spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

# Low‑dimensional Fourier‑basis Adam optimiser
def _lowdim_optimize(
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    L: int = 64,
    max_iters: int = 300_000,
) -> tuple[np.ndarray | None, float]:
    """Optimize a soft‑max over a cosine basis (L parameters) with Adam."""
    i = np.arange(n, dtype=np.float64)

    # Build cosine basis (including a constant term)
    phi = np.empty((n, L), dtype=np.float64)
    phi[:, 0] = 1.0
    for l in range(1, L):
        phi[:, l] = np.cos(2.0 * np.pi * l * i / n)

    c = np.zeros(L, dtype=np.float64)          # parameters to optimise
    m = np.zeros_like(c)
    v = np.zeros_like(c)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    best_a = None
    best_val = np.inf

    it = 0
    while time.time() < deadline and it < max_iters:
        logits = phi @ c
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LowDim it={it:06d}] new best {val:.6f}")

        # exact‑max sub‑gradient
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_logits = a * (grad_a - avg)

        # gradient w.r.t. low‑dim parameters
        grad_c = phi.T @ grad_logits

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_c
        v = beta2 * v + (1.0 - beta2) * (grad_c ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        c -= step

        it += 1

    return best_a, best_val

# Exact max‑convolution projected gradient descent
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on max‑convolution (exact sub‑gradient)."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)      # a slightly larger base step
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

# Spectral flattening (constant‑magnitude FFT) with simplex projection
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Iteratively enforce (almost) constant FFT magnitude, projecting onto simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = np.inf

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- flatten magnitude ----
        A = np.fft.fft(a)                               # full complex FFT
        energy = np.dot(a, a)                           # L2‑norm²
        if n > 1:
            # preserve DC (= sum = 1).  Remaining energy is spread uniformly.
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                                 # keep DC component
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # projection onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

# Simulated‑annealing refinement on the simplex
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """Local stochastic search: move a random fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of a[i] to a[j]
        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising ``evaluate_sequence``.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found before the deadline.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Length – 4096 is a compromise between quality and per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best) if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a large pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # A few random Dirichlet permutations
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # Add low‑dim Fourier‑basis Adam result (global shape search)
    if time.time() < deadline - 30.0:
        print("[Phase 0] Low‑dimensional Fourier‑basis Adam optimisation")
        a_low, val_low = _lowdim_optimize(
            n, conv_fft_len, corr_fft_len, deadline, L=64, max_iters=250_000
        )
        if a_low is not None:
            pool.append(a_low)

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 12   # explore only the most promising starts

    # ------------------------------------------------------------------
    # Phase I – refine a handful of promising starts with exact max‑conv GD
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # allocate a slice of the remaining budget (≈ 25 s per candidate)
        sub_dead = min(deadline, time.time() + 25.0)

        a_gd, val_gd = _max_conv_descent(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Explore {idx}] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase II – spectral flattening
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase II] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – a short projected‑gradient refinement after flattening
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Final projected‑gradient descent")
        a_gd2, val_gd2 = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=150_000
        )
        if val_gd2 < best_val:
            best_val = val_gd2
            best_seq = a_gd2.copy()
            print(f"[Phase III] improved to {val_gd2:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated‑annealing fine‑tuning
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase IV] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase IV] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)

    # Clip to evaluator’s allowed range and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
