"""
Propose a low‑score non‑negative step function.

Algorithm outline
-----------------
1. Gather a diverse pool of initial distributions:
   - warm start (if supplied) and its cyclic shifts
   - Dirichlet draws with many concentration values
   - Exponential, Gaussian, power‑law, triangular, uniform families
   - Additional flexible families (Beta‑pdf, sine‑shape, cosine‑shape,
     double‑exponential centred at the middle).

2. Quick screening: evaluate each raw candidate (no optimisation) and keep the
   `top_k` (default 10) that already have the smallest convolution max.

3. Refine each of the selected candidates with a short Adam run
   (smooth‑max surrogate) followed by a modest exponentiated‑gradient
   sub‑gradient phase.  Keep the best vector found.

4. Spend the bulk of the remaining time on a long exponentiated‑gradient
   optimisation of the current best (the most effective refinement in the
   previous state‑of‑the‑art).

5. Finish with a short simulated‑annealing pass to escape possible
   local minima.

All optimisation is performed on the probability simplex (the evaluation
function is scale‑invariant).  FFT‑based convolution and cross‑correlation are
used for speed.  The routine respects the 1000‑second time budget and always
returns a valid sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories (more families added)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Discretised Beta(α,β) density on {0,…,n‑1}."""
    eps = 1e-12
    x = np.linspace(0.0, 1.0, n)
    x = np.clip(x, eps, 1 - eps)
    a = (x ** (alpha - 1)) * ((1 - x) ** (beta - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a.astype(np.float64)


def _make_sine(n, power=1):
    """a_i ∝ sin(π·i/(n‑1))^p, zero at the ends, peaked in the middle."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    a = a ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _make_cosine(n, power=2):
    """a_i ∝ max(cos(π·i/(n‑1)),0)^p – a bell‑shaped curve."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (n - 1))
    a = np.clip(a, 0.0, None) ** power
    a /= a.sum()
    return a

# Optimisers (same as in the reference code, only tiny hyper‑parameter tweaks)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

    top_k = 12
    top_candidates = [a for _, a in quick_scores[:top_k]]
    print(f"[Screen] selected {len(top_candidates)} promising seeds")

    # ------------------------------------------------------------------
    # Phase I – short Adam + ExpGrad refinements on the top seeds
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    for idx, a0 in enumerate(top_candidates):
        if time.time() >= deadline:
            break
        # Small time slice per candidate (≈20 s)
        sub_dead = min(deadline, time.time() + 20.0)

        # Adam (smooth‑max surrogate)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=150_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Seed {idx}] Adam improved to {val_adam:.6f}")

        # A brief exponentiated‑gradient sub‑gradient phase
        if time.time() < deadline - 5:
            sub_dead2 = min(deadline, time.time() + 15.0)
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead2,
                max_iters=300_000, eta0=0.8,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Seed {idx}] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – long exponentiated‑gradient run on the current best
    if best_seq is not None and time.time() < deadline - 5:
        print("[Final] long exponentiated‑gradient refinement")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=1_200_000, eta0=0.6,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Final] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Phase III – short simulated‑annealing polishing
    if best_seq is not None and time.time() < deadline - 5:
        print("[Final] simulated‑annealing polishing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Final] anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure we always return a valid list
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
