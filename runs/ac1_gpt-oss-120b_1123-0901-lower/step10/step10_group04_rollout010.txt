def solve_g(f, M):
    n = len(f)
    conv_len = 2*n - 1
    # Build A_ub: (conv_len) rows, each with <= n nonzeros.
    data = []
    rows = []
    cols = []
    for k in range(conv_len):
        # i from max(0,k-(n-1)) to min(k,n-1)
        i_start = max(0, k-(n-1))
        i_end = min(k, n-1)
        # g_j index is j = k - i
        for i in range(i_start, i_end+1):
            j = k - i
            data.append(f[i])
            rows.append(k)
            cols.append(j)
    A_ub = sp.csr_matrix((data, (rows, cols)), shape=(conv_len, n))
    b_ub = np.full(conv_len, M)
    # objective: maximize sum(g) => minimize -sum(g)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None
    g = res.x
    return g

def _build_conv_matrix(f):
    n = f.shape[0]
    conv_len = 2 * n - 1
    # diag offsets 0..n-1
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(conv_len, n), format='csr')
    return A

def _lp_max_sum_under_bound(f, M):
    n = len(f)
    A = _build_conv_matrix(f)
    b = np.full(2*n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize -sum
    bounds = [(0, None) for _ in range(n)]
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    # release matrix memory
    del A
    if not res.success:
        return None
    return res.x

def _lp_refine(f, n, deadline, max_iters=5):
    best = f.copy()
    best_val = _objective(f, n)
    it = 0
    while time.time() < deadline - 0.5 and it < max_iters:
        M = _full_convolution(f, _next_pow2(2*n-1)).max()
        g = _lp_max_sum_under_bound(f, M)
        if g is None:
            break
        # scale g to same sum
        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        scale = sum_f / sum_g
        g_scaled = g * scale
        # mixing factor
        t = 0.2
        f_new = (1 - t) * f + t * g_scaled
        val_new = _objective(f_new, n)
        if val_new < best_val:
            best_val = val_new
            best = f_new.copy()
            print(f"[LP it={it}] new best {val_new:.6f}")
        f = f_new
        it += 1
    return best, best_val

"""
Step‑function optimiser – enhanced with linear‑programming (LP) refinement.

The routine consists of four stages:

1. **Parametric sweep** – generate many candidates from a rich family of
   analytic shapes (power‑law, (stretched) exponential, beta‑distribution,
   double‑Gaussian, sine‑bump, random spikes).  Each candidate is normalised
   to a unit sum and the exact objective is evaluated via an FFT‑based
   convolution.

2. **LP‑refinement** – starting from the best candidate,
   repeatedly solve a linear programme that maximises the total mass
   of a new distribution *g* under the linear constraints
        (f ★ g) ≤ M    (M = max(f ★ f))
   and g ≥ 0.
   The solution is scaled to the current total mass of *f* and mixed in
   with a small step size.  This drives the mass into positions that are
   “inactive’’ for the current maximal convolution, often lowering the
   peak.

3. **Long Adam optimisation** – a smooth‑max surrogate (with an
   increasing temperature) is minimised using Adam on the simplex
   (soft‑max parametrisation).  The best exact objective seen is kept.

4. **Exact‑max sub‑gradient (exponentiated‑gradient)** – a pure
   sub‑gradient descent on the true max‑norm further refines the solution,
   followed by a short simulated‑annealing fine‑tune.

All stages respect the 1000 s time budget and the routine always returns
a valid non‑negative list of coefficients.
"""

import time
import numpy as np
import scipy.sparse as sp
import scipy.optimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation (smooth‑max + L₂ regularisation) ------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lam_start: float = 0.1,
    lam_factor: float = 1.01,
    l2_weight: float = 0.12,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (optional) ---------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear programming refinement -----------------------------------------

def _build_conv_matrix(f: np.ndarray) -> sp.csr_matrix:
    """
    Build the Toeplitz matrix A such that A @ g == f ★ g.
    Shape: (2·n‑1) × n   where n = len(f).
    """
    n = f.shape[0]
    conv_len = 2 * n - 1
    # each diagonal offset d (0 … n‑1) is constant f[d]
    data = [np.full(n, f[d], dtype=np.float64) for d in range(n)]
    offsets = np.arange(n)
    A = sp.diags(data, offsets, shape=(conv_len, n), format='csr')
    return A


def _lp_max_sum_under_bound(f: np.ndarray, M: float) -> np.ndarray | None:
    """
    Solve   max  Σ g_j   subject to   f ★ g ≤ M , g ≥ 0.
    Returns the optimal g (may be None if LP fails/unbounded).
    """
    n = f.shape[0]
    A = _build_conv_matrix(f)
    b = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise -sum
    bounds = [(0, None) for _ in range(n)]

    res = scipy.optimize.linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True}
    )
    # free the potentially large matrix early
    del A

    if not res.success:
        return None
    return res.x


def _lp_refine(f0: np.ndarray, n: int, deadline: float,
               max_iters: int = 6) -> tuple[np.ndarray, float]:
    """
    Repeatedly apply the LP‑step described above.
    Returns the best vector encountered and its exact objective.
    """
    f = f0.copy()
    best_a = f.copy()
    best_val = _objective(f, n)

    it = 0
    while time.time() < deadline - 0.5 and it < max_iters:
        # current maximal convolution value
        M = _full_convolution(f, _next_pow2(2 * n - 1)).max()
        g = _lp_max_sum_under_bound(f, M)
        if g is None:
            break

        sum_f = f.sum()
        sum_g = g.sum()
        if sum_g == 0:
            break
        scale = sum_f / sum_g
        g_scaled = g * scale

        t = 0.22                     # mixing weight
        f_new = (1.0 - t) * f + t * g_scaled
        val_new = _objective(f_new, n)

        if val_new < best_val:
            best_val = val_new
            best_a = f_new.copy()
            print(f"[LP it={it}] new best {val_new:.6f}")

        f = f_new
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem dimension – keep it modest for speed
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ---------------------------------------
    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # systematic families
    for p in np.linspace(-5.0, -0.05, 40):
        _try_candidate(_candidate_power(p, n))

    for alpha in np.logspace(-6, 1.2, 45):
        _try_candidate(_candidate_exponential(alpha, n))

    alphas = np.logspace(-4, 0.6, 20)
    betas = np.linspace(0.2, 3.5, 15)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n))

    for _ in range(120):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _try_candidate(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3.0, -0.1, 25):
        _try_candidate(_candidate_double_gaussian(sig, n))

    for alpha in np.logspace(-6, 1.5, 30):
        _try_candidate(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-4.0, 1.0, 30):
        _try_candidate(_candidate_left_power(p, n))

    for pow_ in np.linspace(0.5, 4.5, 20):
        _try_candidate(_candidate_sine(pow_, n))

    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            _try_candidate(_candidate_spike(k, n, random=True))

    # warm start from a previously stored sequence if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑programming refinement ---------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑LP] LP refinement")
        lp_deadline = min(deadline - 5,
                          time.time() + 0.25 * (deadline - time.time()))
        best_seq, best_val = _lp_refine(best_seq, n, lp_deadline,
                                        max_iters=6)
        print(f"[Phase‑LP] best after LP steps: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Long Adam refinement (smooth‑max + L₂) -----------------------
    if best_seq is not None and time.time() < deadline - 25:
        print("[Phase‑Adam] long Adam refinement")
        adam_deadline = min(deadline - 5,
                            time.time() + 0.35 * (deadline - time.time()))
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.05,
            lam_factor=1.005,
            l2_weight=0.10,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact‑max sub‑gradient optimisation (exponentiated‑gradient)-
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] exhaustive sub‑gradient optimisation")
        sub_deadline = deadline - 2.0   # tiny margin for final check
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=14_000_000,
            eta0=0.90,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning (if time permits) --
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid, non‑negative return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
