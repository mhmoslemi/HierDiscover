"""
Optimiser for the step‑function evaluation problem.

The optimiser works in three phases:

1️⃣  **Massive parametric sweep** – we enumerate a rich collection of
    families (U‑shaped power‑laws, one‑sided exponentials,
    beta‑distributions and mixtures with the uniform distribution).
    Every candidate is normalised to ℓ₁‑norm = 1 and its exact
    objective is obtained with a fast FFT‑based convolution.
    The best few vectors are kept.

2️⃣  **Exponentiated‑gradient sub‑gradient descent** – starting from the
    best vector found in the sweep we run a long mirror‑descent
    optimisation of the true max‑norm (no smooth surrogate).  The
    update keeps the iterate on the simplex, uses a diminishing
    stepsize ηₜ = η₀/√t and records the best vector seen.

3️⃣  **Very cheap random fine‑tuning** – a few thousand random mass‑transfer
    moves (move a fraction of the probability mass from one index to another)
    are tried; any improvement is accepted.  This stage cleans up the final
    few digits.

All stages respect the global time budget (default 1000 s) and the routine
always returns a non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1] (a_par,b_par>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_mix_uniform(base: np.ndarray, mix: float) -> np.ndarray:
    """Mixture λ·base + (1‑λ)·uniform, renormalised."""
    n = base.shape[0]
    uniform = np.full(n, 1.0 / n)
    a = mix * base + (1.0 - mix) * uniform
    a_sum = a.sum()
    if a_sum == 0:
        a = uniform
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient (mirror descent) ------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the true max‑norm.
    Returns best vector and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a, n, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of the max‑norm w.r.t. a
        grad = 2.0 * np.fft.irfft(
            np.fft.rfft(w, fft_len) * np.fft.rfft(np.concatenate([a, np.zeros_like(a)]), fft_len),
            fft_len)[:n]

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)
        a_sum = a.sum()
        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search entry point ----------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the allotted time budget (seconds) and always
    returns a valid list of coefficients.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5         # safety margin

    # problem size – 4096 is a good compromise between speed and quality
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # --------------------------------------------------------------
    # 1) Very dense parametric sweep
    # --------------------------------------------------------------
    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {tag}] new best {val:.6f}")

    # uniform (baseline)
    _try_candidate(np.full(n, 1.0 / n), "uniform")

    # ------- power‑law (U‑shaped) ---------------------------------
    for p in np.linspace(-12.0, -0.2, 71):
        a = _candidate_power(p, n)
        _try_candidate(a, f"pow{p:.2f}")

        # also try mixtures with the uniform distribution
        for lam in np.linspace(0.0, 1.0, 6):
            b = _candidate_mix_uniform(a, lam)
            _try_candidate(b, f"pow{p:.2f}_mix{lam:.2f}")

    # ------- symmetric exponential ---------------------------------
    for alpha in np.logspace(-5, 2, 60):
        a = _candidate_exponential(alpha, n)
        _try_candidate(a, f"exp{alpha:.3e}")

        for lam in np.linspace(0.0, 1.0, 6):
            b = _candidate_mix_uniform(a, lam)
            _try_candidate(b, f"exp{alpha:.3e}_mix{lam:.2f}")

    # ------- left‑hand exponential (monotone) ---------------------
    for alpha in np.logspace(-5, 2, 40):
        a = _candidate_left_exponential(alpha, n)
        _try_candidate(a, f"lexp{alpha:.3e}")

        for lam in np.linspace(0.0, 1.0, 6):
            b = _candidate_mix_uniform(a, lam)
            _try_candidate(b, f"lexp{alpha:.3e}_mix{lam:.2f}")

    # ------- beta‑distribution families ----------------------------
    # grid on (a,b) with a,b in [0.05, 20] on a log‑scale
    a_vals = np.logspace(-1.3, 1.3, 16)   # 0.05 … 20
    b_vals = np.logspace(-1.3, 1.3, 16)
    for a_par in a_vals:
        for b_par in b_vals:
            a = _candidate_beta(a_par, b_par, n)
            _try_candidate(a, f"beta{a_par:.2f}_{b_par:.2f}")

            for lam in np.linspace(0.0, 1.0, 4):
                b = _candidate_mix_uniform(a, lam)
                _try_candidate(b, f"beta{a_par:.2f}_{b_par:.2f}_mix{lam:.2f}")

    # ------- random Dirichlet spikes (sparse patterns) ----------
    for k in (64, 128, 256):
        for _ in range(12):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            a /= a.sum()
            _try_candidate(a, f"spike{k}")

    # warm start from a previously stored solution, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w, "warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Sub‑gradient mirror‑descent (exact max) -------------------
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] starting exact max sub‑gradient optimisation")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, fft_len, sub_deadline,
            max_iters=8_000_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 3) Very cheap random fine‑tuning (mass transfer) ------------
    # --------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 0.5:
        print("[Phase‑FineTune] random mass‑transfer hill‑climb")
        a = best_seq.copy()
        val = _objective(a, n, fft_len)
        best_local = a.copy()
        best_local_val = val

        # simple annealing schedule – temperature falls linearly
        T0 = 0.07
        it = 0
        while time.time() < deadline:
            T = max(0.0, T0 * (1.0 - it / 150_000.0))
            i = np.random.randint(n)
            if a[i] < 1e-18:
                it += 1
                continue
            j = np.random.randint(n)
            if i == j:
                it += 1
                continue
            frac = np.random.rand() * (0.25 if T > 0 else 0.0)
            delta = a[i] * frac
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            val2 = _objective(a2, n, fft_len)
            if val2 < best_local_val:
                best_local_val = val2
                best_local = a2.copy()
                a = a2
                val = val2
                # print progress only occasionally
                if it % 20000 == 0:
                    print(f"[Fine‑Tune it={it}] new best {val2:.6f}")
            else:
                # accept with small probability while temperature > 0
                if T > 0 and np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2
            it += 1

        if best_local_val < best_val:
            best_val = best_local_val
            best_seq = best_local.copy()
            print(f"[Phase‑FineTune] improved to {best_local_val:.6f}")

    # --------------------------------------------------------------
    # Ensure a valid return
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # final safety clipping (required by evaluate_sequence)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # show the final score with the public evaluator (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not run evaluate_sequence – returning raw sequence.")

    return best_seq.tolist()
