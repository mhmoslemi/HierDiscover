def project_to_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

def _expgrad_momentum_optimize(a0, n, fft_len, deadline, max_iters, eta0=1.0, beta=0.9):

    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    m = np.zeros_like(a)

    it = 0

    while time.time() < deadline and it < max_iters:

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()

        val = 2.0*n*max_c/(a.sum()**2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 500_000 == 0:
                print(f"[ExpGradMomentum it={it}] new best {val}")

        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0/active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # momentum
        m = beta * m + (1 - beta) * grad

        eta = eta0 / np.sqrt(it + 1.0)

        if a_sum < 1e-12 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        it += 1

    return best_a, best_val

"""
Propose a low‑value step‑function.

The optimiser works in four stages:

1. **Fast low‑resolution sweep** – generate many candidates on a
   modest grid (n_small = 2048) from a rich family of analytic shapes
   (power‑law, exponential, stretched‑exponential, beta, double‑Gaussian,
   monotone, sine, random spikes, random Dirichlet).  The best few are
   retained.

2. **Upsampling** – the retained low‑resolution vectors are linearly
   interpolated to the target size (n_big = 4096) and normalised.

3. **Adam on a smooth‑max surrogate** – starting from the best up‑sampled
   vector we minimise a smooth approximation of the true objective:
        smooth_max(conv) = (1/λ)·log Σ exp(λ·conv).
   The optimisation is performed on soft‑max (log‑it) parameters and
   uses a slowly increasing temperature λ (≈10⁵ at the end).  Only the
   smooth‑max term is used (no L² regularisation) – this focuses the
   search on the true maximum.

4. **Momentum exponentiated‑gradient on the exact max** – the final
   refinement runs a mirror‑descent update with exponential moving‑average
   of the exact sub‑gradient (active‑set of the convolution).  The step
   size decays as η₀/√t.  The best vector seen throughout the run is
   returned.

All stages respect the global time budget (default 1000 s) and the
function always returns a non‑negative list of coefficients.
"""

# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for gradients:
          (w ★ a)[i] = Σ_k w[k]·a[i−k] .
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_random_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Random Dirichlet with concentration alpha (>0)."""
    a = np.random.gamma(alpha, size=n)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max only) -------------------------------------

def _adam_optimize_smoothmax(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lam_start: float = 0.05,
    lam_factor: float = 1.005,
    lr: float = 0.07,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate true objective occasionally
        if it % 50000 == 0:
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Adam it={it}] new best {val:.6f}")

        # convolution and smooth‑max weights
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # mirror‑descent (soft‑max) transform
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam parameter update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e5)

        # occasional learning‑rate decay
        if it % 150_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient with momentum (exact max) ---------------------

def _expgrad_momentum_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.9,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the exact objective using an exponential‑moving
    average of the sub‑gradient (active set of the convolution).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    m = np.zeros_like(a)   # momentum accumulator
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 500_000 == 0:
                print(f"[ExpGrad it={it}] new best {val:.6f}")

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # momentum update
        m = beta * m + (1.0 - beta) * grad

        # step size schedule
        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update (mirrored on the simplex)
        a = a * np.exp(-eta * m)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (optional) ---------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Very simple annealing that moves a small fraction of mass between
    two randomly chosen positions.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine respects the supplied time budget.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # --------------------------------------------------------------
    # Parameters (resolution)
    n_big = 4096          # final dimension
    n_small = 2048        # quick sweep dimension
    fft_len_small = _next_pow2(2 * n_small - 1)
    fft_len_big = _next_pow2(2 * n_big - 1)

    # --------------------------------------------------------------
    # 1) Fast low‑resolution sweep ---------------------------------
    top_k = 8
    top_small = []   # list of (value, vector)

    def _push_small(a_small: np.ndarray):
        val = _objective(a_small, n_small)
        if len(top_small) < top_k or val < top_small[-1][0]:
            top_small.append((val, a_small.copy()))
            top_small.sort(key=lambda x: x[0])
            if len(top_small) > top_k:
                top_small.pop()

    # systematic families (reduced grid for speed)
    for p in np.linspace(-5.0, -0.05, 30):
        _push_small(_candidate_power(p, n_small))

    for alpha in np.logspace(-6, 1.0, 30):
        _push_small(_candidate_exponential(alpha, n_small))

    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.2, 3.5, 10)
    for a_alpha in alphas:
        for b_beta in betas:
            _push_small(_candidate_stretched(a_alpha, b_beta, n_small))

    for _ in range(60):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _push_small(_candidate_beta(a_par, b_par, n_small))

    for sig in np.logspace(-3.0, -0.2, 20):
        _push_small(_candidate_double_gaussian(sig, n_small))

    for alpha in np.logspace(-6, 1.2, 25):
        _push_small(_candidate_left_exponential(alpha, n_small))

    for p in np.linspace(-4.0, 1.0, 20):
        _push_small(_candidate_left_power(p, n_small))

    for pow_ in np.linspace(0.5, 4.5, 15):
        _push_small(_candidate_sine(pow_, n_small))

    # random spikes of varying sparsity
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _push_small(_candidate_spike(k, n_small, random=True))

    # random Dirichlet draws (different concentrations)
    for _ in range(200):
        conc = np.random.uniform(0.01, 5.0)
        _push_small(_candidate_random_dirichlet(conc, n_small))

    print(f"[Sweep] finished; best small‑scale value {top_small[0][0]:.6f}")

    # --------------------------------------------------------------
    # 2) Upsample best small‑scale candidates to the target size -----
    best_seq = None
    best_val = float("inf")
    for val_small, a_small in top_small:
        # simple linear interpolation to n_big points
        x_small = np.linspace(0, 1, n_small)
        x_big = np.linspace(0, 1, n_big)
        a_big = np.interp(x_big, x_small, a_small)
        a_big = np.clip(a_big, 0.0, None)
        if a_big.sum() < 1e-12:
            a_big = np.full(n_big, 1.0 / n_big)
        else:
            a_big /= a_big.sum()
        val_big = _objective(a_big, n_big)
        if val_big < best_val:
            best_val = val_big
            best_seq = a_big.copy()
    print(f"[Upsample] best after up‑sampling {best_val:.6f}")

    # --------------------------------------------------------------
    # 3) Adam refinement on smooth‑max --------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + remaining * 0.30
        print("[Phase‑Adam] smooth‑max Adam start")
        a_adam, val_adam = _adam_optimize_smoothmax(
            best_seq,
            n_big,
            fft_len_big,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.05,
            lam_factor=1.005,
            lr=0.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Exact‑max exponentiated‑gradient with momentum --------------
    remaining = deadline - time.time()
    if remaining > 10.0:
        sub_deadline = time.time() + remaining * 0.65
        print("[Phase‑Subgrad] exact‑max momentum EG start")
        a_sub, val_sub = _expgrad_momentum_optimize(
            best_seq,
            n_big,
            fft_len_big,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.9,
            beta=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Final optional simulated‑annealing fine‑tune ----------------
    remaining = deadline - time.time()
    if remaining > 5.0:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n_big,
            fft_len_big,
            deadline,
            max_moves=int(remaining * 200),   # roughly 200 moves per second
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Final sanity – clip and return ---------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print("Could not evaluate final sequence:", exc)

    return best_seq.tolist()
