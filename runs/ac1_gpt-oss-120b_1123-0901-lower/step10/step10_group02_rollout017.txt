"""
Hybrid simulated‑annealing + projected‑gradient optimizer.

1. Initialise a non‑negative step vector (warm start if available,
   otherwise a sparse Dirichlet sample) and normalise it to sum = 1.
2. Compute the full self‑convolution once via FFT.
3. Run a long simulated‑annealing loop that repeatedly transfers a small
   amount of mass from one index to another.
   *   The move size decays linearly from 50 % of the donor mass to
       0.05 % as the time budget runs out.
   *   The temperature also decays linearly, allowing occasional uphill
       moves early on.
   *   Half of the proposals are “targeted”: the donor index belongs to
       the pair that contributes most to the current maximal convolution
       entry, which directly attacks the bottleneck.
   *   The convolution is updated **incrementally** (O(n) work) using the
       exact formula
       ``conv ← conv + 2·δ·(shift_j – shift_i) + δ²·e_{2i}+δ²·e_{2j}–2δ²·e_{i+j}``,
       so millions of iterations fit into the 1000 s budget.
4. Every few thousand iterations the convolution is recomputed from
   scratch (FFT) to remove accumulated floating‑point drift.
5. After the annealing stage a short projected‑gradient descent (using
   the exact max‑convolution sub‑gradient) refines the best candidate.
6. Finally a handful of cyclic shifts are tested; the best sequence is
   returned as a plain Python list.

The whole routine respects the supplied time budget and always returns a
valid non‑negative sequence.
"""

import time
import math
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)
