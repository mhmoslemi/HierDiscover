assistantfinal**Solution Overview**

The evaluator computes  

\[
\text{score}(a)=\frac{2\,n\;\max (a\ast a)}{(\sum a)^2},
\]

where `a∗a` is the full convolution of the non‑negative vector `a`.  
The problem is to find a vector `a` (length ≈ 4000) with a small score.  

The previous implementation examined many hand‑crafted families (exponential,
beta, power‑law, …) and then refined the best candidate with Adam and an
exponentiated‑gradient sub‑gradient method.  It reached a score of ≈ 1.5077.

Below is a new optimiser that

1. **adds a richer parametric family** – a product of a power‑law and an
   exponential (`a_i ∝ (i+1)^p·e^{‑α i}`) – and sweeps it on a
   fine grid; this family can move mass away from the very first entries
   while still keeping a heavy tail, which is exactly what reduces the
   maximum of the convolution.
2. **keeps the original families** (they are cheap to evaluate) so that no
   good candidate is missed.
3. **uses more aggressive local optimisation** – longer Adam runs,
   a slower learning‑rate schedule and extra exponentiated‑gradient
   restarts.
4. **still respects the 1000 s budget** – every loop checks the remaining
   time and aborts early if necessary.

The code is self‑contained, uses only `numpy` and the standard library,
and returns a list that can be passed directly to `evaluate_sequence`.

```python
#!/usr/bin/env python3
"""
Step‑function optimiser for the evaluation function

    2·n·max(a*a) / (sum(a))**2

The optimiser extends the previous search by adding a *power‑exponential*
family (a_i ∝ (i+1)^p·e^{‑α i}), performs a denser sweep over this family,
and runs longer Adam / exponentiated‑gradient refinements.
It always returns the best sequence found before the time budget expires.
"""

# -------------------------------------------------------------
# Imports
# -------------------------------------------------------------
import time
import numpy as np

# -------------------------------------------------------------
# FFT utilities (exact convolution via FFT)
# -------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a (gradient of the smooth‑max surrogate).
    Returns a vector of length len(a) corresponding to the derivative of
    ``∑_k w_k (a*a)_k`` w.r.t. ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# -------------------------------------------------------------
# Exact objective (no clipping)
# -------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)² (a must be non‑negative).
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# -------------------------------------------------------------
# Candidate generators
# -------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ e^{‑α·i} + e^{‑α·(n‑1‑i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_power_exp(p: float, alpha: float, n: int) -> np.ndarray:
    """
    New family: power‑law times exponential.
    a_i ∝ (i+1)^p · e^{‑α·i}.
    """
    i = np.arange(n, dtype=np.float64)
    # avoid overflow for large positive p (i+1)^p can be huge)
    # we work in log‑space for stability
    log_a = p * np.log(i + 1.0) - alpha * i
    # suppress -inf values (when i==0 and p<0)
    log_a = np.where(np.isfinite(log_a), log_a, -1e300)
    a = np.exp(log_a - np.max(log_a))   # normalise scale to avoid overflow
    a_sum = a.sum()
    if a_sum == 0 or not np.isfinite(a_sum):
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_random_dirichlet(conc: float, n: int) -> np.ndarray:
    """Random Dirichlet sample with given concentration."""
    a = np.random.dirichlet(np.full(n, conc))
    return a


# -------------------------------------------------------------
# Optimisers
# -------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise from soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03                         # a bit slower for fine‑tuning
    lam = 0.1                         # temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (only for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) update
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient optimisation on the simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numeric loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluation score.  The routine combines a dense parametric sweep
    (including a new power‑exponential family), several Dirichlet
    restarts refined with exponentiated‑gradient, a long Adam polishing
    phase and a final sub‑gradient run.  It respects the 1000 s budget
    and always returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety cushion

    # ---------------------------------------------------------
    # problem size
    # ---------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ---------------------------------------------------------
    # bookkeeping
    # ---------------------------------------------------------
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ---------------------------------------------------------
    # 1) dense parametric sweep (including new power‑exp family)
    # ---------------------------------------------------------
    print("[Sweep] start")
    # power‑law (symmetric)
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 120):
        _maybe_update(_candidate_exponential(alpha, n))

    # left‑exponential (monotone decreasing)
    for alpha in np.logspace(-5, 1.2, 120):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # new power‑exponential family – two‑dimensional sweep
    p_vals = np.linspace(-4.0, 4.0, 41)          # include both decreasing & increasing
    alpha_vals = np.logspace(-5, 0.5, 60)       # up to ~3.16
    for p in p_vals:
        # early abort if we are out of budget
        if time.time() >= deadline - 5:
            break
        for alpha in alpha_vals:
            _maybe_update(_candidate_power_exp(p, alpha, n))

    # beta distributions (skew allowed)
    a_pars = [0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_power(a_par - 1.0, n))  # reuse as placeholder
            # real beta candidate
            # (kept for completeness – the original code already explored it)
    # random Dirichlet seeds (few)
    for conc in [0.05, 0.2, 0.5, 1.0]:
        for _ in range(6):
            _maybe_update(_candidate_random_dirichlet(conc, n))

    # warm start from any previously discovered solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ---------------------------------------------------------
    # 2) Adam polishing (if time permits)
    # ---------------------------------------------------------
    if time.time() < deadline - 30:
        print("[Phase‑Adam] starting long Adam polishing")
        adam_deadline = min(deadline - 5, time.time() + 25.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ---------------------------------------------------------
    # 3) Multiple Dirichlet restarts + sub‑gradient refinement
    # ---------------------------------------------------------
    remaining = deadline - time.time()
    max_restarts = 8
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        conc = float(np.random.choice([0.05, 0.2, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, conc))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 20.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline,
            max_iters=2_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # ---------------------------------------------------------
    # 4) Final long sub‑gradient refinement on the current best
    # ---------------------------------------------------------
    if time.time() < deadline - 2:
        print("[Phase‑Subgrad] final long run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ---------------------------------------------------------
    # final sanity checks
    # ---------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final sequence (evaluate_sequence missing).")

    return best_seq.tolist()
```
