def refine(a0, max_iter, deadline):
    a = a0.copy()
    m = np.zeros_like(a)
    v = np.zeros_like(a)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    for t in range(1, max_iter+1):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2 * n * max_c
        if val < best_val_global:
            best_val_global = val
            best_seq_global = a.copy()

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)  # don't use too small tolerance
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Adam-like step
        m = beta1 * m + (1-beta1) * grad
        v = beta2 * v + (1-beta2) * (grad**2)
        m_hat = m / (1 - beta1**t)
        v_hat = v / (1 - beta2**t)
        step = 0.1 / (np.sqrt(v_hat) + eps)  # adapt
        # Compute multiplicative update
        a = a * np.exp(-step * m_hat)

        # Renorm
        s = a.sum()
        if s > 0:
            a = a / s
        else:
            # fallback
            a = a0.copy()
            # optional break
        # maybe occasional line-search: try small factor, accept if improves.

        shape = shape / shape.sum()

    # (4) logistic shape scanning

    conv = _full_convolution(a, conv_fft_len)

            break

        T = max(T_end, T_start * (1 - t_frac))

        a2[i] -= delta
        a2[j] += delta

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2

import time
import numpy as np

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)
    return conv[:2*a.shape[0]-1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = a.shape[0]-1
    return corr[start:start + a.shape[0]]

def propose_candidate(seed=0, budget_s=1000):
    """
    Hybrid random‑parameter sweep + Adam‑style exponentiated‑gradient +
    simulated‑annealing optimizer for the max‑convolution objective.
    It builds a diverse pool of shapes (power‑law, beta, exponential,
    logistic, block‑constant, spikes, random mixtures), keeps the best
    few, refines each with a mirror‑descent step on the simplex,
    and finally applies a short SA polish.  The routine respects the
    given time budget and always returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5  # safety margin
    # Problem dimension – use the length of the warm‑start if present.
    warm = globals().get('height_sequence_1')
    if warm is not None:
        n = len(warm)
        if n == 0:
            n = 4096
    else:
        n = 4096

    # pre‑compute FFT lengths for convolution & cross‑correlation
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Helper to compute the objective (sum is assumed to be 1)
    def _objective_val(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    best_val = float('inf')
    best_seq = None

    # -------------------------------------------------------------
    # 1️⃣  Build a rich pool of starter vectors
    pool = []

    # warm start (if available)
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # basic index array
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # -------------------------------------------------------------
    # 1.A  Power‑law “U‑shaped” families – finer sweep over α
    for alpha in np.linspace(-3.0, -0.2, 250):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # -------------------------------------------------------------
    # 1.B  Beta‑shapes (x^p (1‑x)^q) – dense grid and random samples
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    grid_ps = np.linspace(0.05, 3.0, 30)
    grid_qs = np.linspace(0.05, 3.0, 30)
    for p in grid_ps:
        for q in grid_qs:
            shape = (x ** p) * ((1.0 - x) ** q)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                pool.append(shape)
    # random Beta samples (log‑uniform on parameters)
    for _ in range(200):
        p = 10 ** np.random.uniform(-2, 0.5)   # roughly 0.01 … 3.16
        q = 10 ** np.random.uniform(-2, 0.5)
        shape = (x ** p) * ((1.0 - x) ** q)
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # -------------------------------------------------------------
    # 1.C  Exponential‑decay families (symmetric)
    for lam in np.linspace(0.0005, 0.01, 120):
        exp_vec = np.exp(-lam * np.arange(n)) + np.exp(-lam * np.arange(n)[::-1])
        if exp_vec.sum() > 0:
            shape = exp_vec / exp_vec.sum()
            pool.append(shape)

    # -------------------------------------------------------------
    # 1.D  Logistic (sigmoid) families – symmetric S‑shaped
    for k in np.linspace(0.01, 0.3, 100):
        t = np.arange(n, dtype=np.float64) - (n - 1) / 2.0
        sig = 1.0 / (1.0 + np.exp(k * t))
        shape = sig + sig[::-1] - sig[0]  # make symmetric, avoid double‑count centre
        shape = shape - shape.min()  # ensure non‑negative
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # -------------------------------------------------------------
    # 1.E  Piece‑wise constant blocks (random masses)
    for blocks in (8, 16, 32, 64):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            pool.append(a)

    # -------------------------------------------------------------
    # 1.F  Edge‑heavy “corner” families – allocate a fraction of mass to the
    # outer L entries on each side, the rest uniformly.
    for L in (8, 16, 32, 64, 128, 256):
        for frac in np.linspace(0.05, 0.5, 10):
            # frac = total mass placed on the two edge blocks
            inner_len = n - 2 * L
            if inner_len <= 0:
                continue
            edge_mass = frac / (2 * L)
            inner_mass = (1.0 - frac) / inner_len
            a = np.full(n, inner_mass, dtype=np.float64)
            a[:L] = edge_mass
            a[-L:] = edge_mass
            pool.append(a)

    # -------------------------------------------------------------
    # 1.G  Random sparse spikes (hand‑crafted)
    for k in (64, 128, 256):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            if a.sum() > 0:
                a = a / a.sum()
                pool.append(a)

    # -------------------------------------------------------------
    # 1.H  Dirichlet draws with a broad range of concentrations
    for alpha in (0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # -------------------------------------------------------------
    # 2️⃣ Evaluate all starters and keep the best few (top_k)
    top_k = 30
    top = []   # list of (val, a)

    for cand in pool:
        if time.time() > deadline - 2:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        # Compute objective (sum is 1)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if np.isnan(val):
            continue
        # update global best
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init] new global best {val:.6f}")

        # keep in top‑k list
        if len(top) < top_k:
            top.append((val, a))
            top.sort(key=lambda x: x[0])
        else:
            if val < top[-1][0]:
                top[-1] = (val, a)
                top.sort(key=lambda x: x[0])

    # ----------------------------------------------------------------
    # If we failed to create any candidate, fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_val(best_seq)
        print("[Fallback] using uniform distribution")

    # ----------------------------------------------------------------
    # 3️⃣  Refine the top candidates with Adam‑style exponentiated‑gradient
    #      (mirror descent on the simplex).  The number of iterations is
    #      limited by the remaining time budget.
    global_best_val = best_val
    global_best_seq = best_seq.copy()

    # parameters for Adam‑style updates
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-12
    base_lr = 0.15   # base learning‑rate for the multiplicative update

    for idx_cand, (init_val, a0) in enumerate(top):
        if time.time() > deadline - 5:
            break
        a = a0.copy()
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        it = 0
        # inner loop – stop when time runs out or after a generous iteration cap
        while time.time() < deadline - 5 and it < 200_000:
            # forward pass
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            val = 2.0 * n * max_c
            if val < global_best_val:
                global_best_val = val
                global_best_seq = a.copy()
                print(f"[Refine #{idx_cand} it={it}] new global best {val:.6f}")

            # compute sub‑gradient of max(conv)
            # treat all indices attaining the maximum as active
            active = np.abs(conv - max_c) <= 1e-12
            if not active.any():
                break
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # Adam moment updates
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)

            # bias‑corrected moments
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))

            # element‑wise step size (we cap it to avoid overflow)
            step = base_lr / (np.sqrt(v_hat) + eps)
            delta = -step * m_hat
            delta = np.clip(delta, -30.0, 30.0)   # keep exp() safe

            # multiplicative (mirror) update, then renormalise onto the simplex
            a = a * np.exp(delta)

    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]
