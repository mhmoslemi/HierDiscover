"""
Hybrid optimiser with random restarts.

Key ideas
---------
1. Warm‑start from the global `height_sequence_1` if present,
   otherwise fall back to the uniform simplex distribution.
2. Perform a number of cheap Adam runs starting from random Dirichlet
   vectors.  Each run is limited to a few ten‑thousand iterations and
   keeps the best candidate seen so far.  Random restarts help to
   escape the basin of attraction of the previous best solution.
3. Run a long exponentiated‑gradient sub‑gradient phase (the same
   exact max‑convolution sub‑gradient used in the reference code) on
   the current best vector.  The time allocated to this stage is
   dominant because it is most effective at reducing the max‑convolution.
4. Refine the solution with a second, longer Adam pass and a short
   projected‑gradient descent on the exact objective.
5. A final cheap cyclic‑shift sweep is performed; the best shifted
   version is returned.

All stages respect the supplied time budget (`budget_s` ≈ 1000 s) and
always finish with a valid non‑negative simplex vector.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length = 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.05,
    lam0: float = 0.2,
    lam_factor: float = 1.04,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).

    Parameters
    ----------
    a0 : np.ndarray
        Starting point (already normalised to sum‑one).
    lr, lam0, lam_factor : float
        Learning‑rate and temperature schedule for the smooth‑max.
    max_iters : int
        Upper bound on inner iterations (the actual number is cut by the deadline).

    Returns
    -------
    best_a, best_val
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    lam = lam0
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a (always on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional logging
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – increase slowly towards a hard max
        lam = min(lam * lam_factor, 1e7)

        # very gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, up to numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # multiplicative EG update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        # compute sub‑gradient (uniform weight over all active indices)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # keep a tiny safety margin
    deadline = time.time() + max(0.0, budget_s - 0.3)

    # ------------------------------------------------------------------
    # problem size – the original code used 4096, we keep it as a good compromise
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s                     # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Allocate budget
    total_budget = deadline - time.time()
    if total_budget <= 0:
        return best_a.tolist()

    # Stage A – cheap random‑restart Adam runs (≈20 % of total time)
    restart_dead = min(deadline, time.time() + 0.20 * total_budget)
    restart_iter = 0
    while time.time() < restart_dead:
        # generate a fresh start – either Dirichlet or a small perturbation of current best
        if np.random.rand() < 0.7:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            noise = np.random.normal(scale=0.02, size=n)
            a0 = np.maximum(best_a + noise, 0.0)
            a0 = a0 / a0.sum()
        # limited Adam run – a few ten‑thousand iterations are enough to explore
        max_its = 30_000
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, restart_dead,
            max_iters=max_its,
            lr=0.07,
            lam0=0.15,
            lam_factor=1.05,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Restart {restart_iter}] improved to {val_adam:.6f}")
        restart_iter += 1

    # ------------------------------------------------------------------
    # Stage B – long exponentiated‑gradient sub‑gradient phase (≈60 %)
    remaining = deadline - time.time()
    if remaining > 0:
        sub_dead = min(deadline, time.time() + 0.60 * remaining)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=10_000_000,
            eta0=1.1,
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Stage B] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage C – second Adam pass (≈10 %)
    remaining = deadline - time.time()
    if remaining > 0:
        adam2_dead = min(deadline, time.time() + 0.10 * remaining)
        a_adam2, val_adam2 = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam2_dead,
            max_iters=300_000,
            lr=0.03,
            lam0=0.4,
            lam_factor=1.03,
        )
        if val_adam2 < best_val:
            best_val, best_a = val_adam2, a_adam2.copy()
            print(f"[Stage C] Second Adam improved to {val_adam2:.6f}")

    # ------------------------------------------------------------------
    # Stage D – short projected gradient descent (≈5 %)
    remaining = deadline - time.time()
    if remaining > 0:
        gd_dead = min(deadline, time.time() + 0.05 * remaining)
        a_gd, val_gd = _max_conv_descent(
            best_a, n, conv_fft_len, corr_fft_len, gd_dead,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val, best_a = val_gd, a_gd.copy()
            print(f"[Stage D] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Final cyclic‑shift sweep (deterministic, cheap)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # Report via the official evaluation routine (does not affect the return value)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
