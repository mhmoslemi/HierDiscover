"""
Optimiser for ``evaluate_sequence``.

The algorithm consists of the following stages (all respect the time budget,
default 1000 s):

1. **Very dense parametric sweep** – a large collection of analytic families
   (exponential, power‑law, stretched‑exponential, beta, logistic,
   linear, quadratic, cubic, reciprocal, piecewise constant) is sampled on
   a fine grid.  The best candidate is kept.

2. **Mirror‑descent on the simplex** – we minimise the smooth‑max (log‑sum‑exp)
   surrogate of ``max(conv)`` using an Adam optimiser that works on the
   soft‑max (exponentiated‑gradient) parametrisation ``a = softmax(logits)``.
   The optimiser runs for a few tens of seconds with a temperature schedule
   that quickly sharpens the surrogate.

3. **Multiple random restarts** – many Dirichlet draws are refined with a
   pure exponentiated‑gradient sub‑gradient method (step‑size ηₜ≈η₀/√t).  Each
   run is allowed a few seconds; the best result of all restarts replaces the
   current incumbent.

4. **Final simulated‑annealing polishing** – a classic annealing that moves a
   random fraction of mass between two coordinates.  The schedule starts at
   ``T≈0.2`` and cools geometrically.

All stages keep the iterates non‑negative and normalise to a unit‑sum.
The final list is returned; it can be fed directly to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of the smooth‑max
    surrogate of ``max(conv)``.
    Returns a vector of length len(a) corresponding to the derivative of
    ``∑_k w_k (a*a)[k]`` w.r.t. ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact value of the evaluation function."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate families ----------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p   (p can be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """a_i ∝ exp(-α·(i/(n‑1))**β)  (symmetric, mirrored)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on a regular grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape (clipped at zero)."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_quadratic(slope: float, n: int) -> np.ndarray:
    """Quadratic decreasing: a_i ∝ (1 - slope·i/(n‑1))²."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 - slope * i / (n - 1)
    a = np.maximum(0.0, base) ** 2
    a /= a.sum()
    return a


def _candidate_cubic(slope: float, n: int) -> np.ndarray:
    """Cubic decreasing: a_i ∝ (1 - slope·i/(n‑1))³."""
    i = np.arange(n, dtype=np.float64)
    base = 1.0 - slope * i / (n - 1)
    a = np.maximum(0.0, base) ** 3
    a /= a.sum()
    return a


def _candidate_inverse(alpha: float, n: int) -> np.ndarray:
    """Reciprocal shape: a_i ∝ 1/(i+α)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + alpha)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """Adam optimisation of the smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.5                 # initial temperature
    lam_factor = 1.025

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # smooth‑max weight
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – go to a very sharp max
        lam = min(lam * lam_factor, 1e7)

        # keep best
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional debug print (can be silenced)
            # print(f"[Adam it={it:06d}] new best {val:.6f}")

        # very occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _expgrad_subgradient(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         corr_fft_len: int,
                         deadline: float,
                         max_iters: int = 3_000_000,
                         eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient mirror‑descent on the exact max‑convolution."""
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices achieving the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against pathological loss of mass
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
        else:
            a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 400_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_a = a.copy()
    best_val = val

    T_start = 0.2
    T_end = 1e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤25 %) of mass from i to j
        frac = np.random.rand() * 0.25
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function (list of non‑negative heights) that minimises
    ``evaluate_sequence``.  The routine respects the given time budget (seconds)
    and always returns the best sequence discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # problem size – 2048 gives a good speed/quality trade‑off
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray, tag: str = "Sweep"):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1. Dense parametric sweep
    # ------------------------------------------------------------------

    # power‑law (both decreasing and symmetric)
    for p in np.linspace(-5.0, -0.05, 140):
        _maybe_update(_candidate_power(p, n), "PowDec")
    for p in np.linspace(0.05, 5.0, 100):
        # symmetric version (mirror)
        _maybe_update(_candidate_power(p, n) + _candidate_power(p, n)[::-1], "PowSym")

    # exponential decreasing
    for a in np.logspace(-5, 2.0, 150):
        _maybe_update(_candidate_exponential(a, n), "ExpDec")

    # stretched exponential (symmetric)
    alphas = np.logspace(-4, 0.5, 60)
    betas = np.linspace(0.2, 3.5, 40)
    for a in alphas:
        for b in betas:
            _maybe_update(_candidate_stretched(a, b, n), "StrExp")

    # beta distributions (allow skew)
    a_pars = [0.05, 0.1, 0.3, 0.6, 1.0, 2.0, 5.0, 12.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n), "Beta")

    # double Gaussian
    for sig in np.logspace(-3.5, -0.3, 40):
        _maybe_update(_candidate_double_gaussian(sig, n), "DGauss")

    # monotone families (linear, quadratic, cubic, logistic, reciprocal)
    for slope in np.linspace(0.2, 4.0, 100):
        _maybe_update(_candidate_left_linear(slope, n), "LinDec")
        _maybe_update(_candidate_quadratic(slope, n), "QuadDec")
        _maybe_update(_candidate_cubic(slope, n), "CubicDec")
    for beta in np.logspace(-3, 1.5, 80):
        _maybe_update(_candidate_left_logistic(beta, n / 2.0, n), "LogiDec")

    for alpha in np.logspace(-5, 1.0, 120):
        _maybe_update(_candidate_inverse(alpha, n), "InvDec")

    # random spike patterns (different cardinalities)
    for k in [64, 128, 256, 512, 1024]:
        for _ in range(6):
            _maybe_update(_candidate_spike(k, n, random=True), "SpikeRnd")

    # a few pure Dirichlet draws
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand, "Dirichlet")

    # possible warm start from a previous global variable
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "Warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2. Adam polishing (smooth‑max surrogate)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30:
        adam_deadline = min(deadline - 5, time.time() + 18.0)
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len,
                                          corr_fft_len,
                                          adam_deadline,
                                          max_iters=500_000)
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3. Multiple Dirichlet restarts with exact sub‑gradient
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    max_restarts = 12
    for r in range(max_restarts):
        if time.time() >= deadline - 8:
            break
        concentration = float(np.random.choice([0.05, 0.2, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        rest_deadline = time.time() + min(remaining / (max_restarts - r), 22.0)
        a_sub, val_sub = _expgrad_subgradient(a0, n,
                                               conv_fft_len,
                                               corr_fft_len,
                                               rest_deadline,
                                               max_iters=2_200_000,
                                               eta0=1.0)
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 4. Final annealing refinement
    # ------------------------------------------------------------------
    if time.time() < deadline - 2:
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          deadline,
                                          max_moves=350_000)
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # 5. Final sanity checks
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final evaluation using the provided scorer (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final sequence (evaluate_sequence missing).")

    return best_seq.tolist()
