def _project_simplex(v):
    # Project onto probability simplex (nonnegative sum = 1)
    # using algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.where(u + (1 - cssv) / (np.arange(1, len(v)+1)) > 0)[0][-1]
    theta = (cssv[rho] - 1) / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

def _gradient_descent_optimize(a0, n, fft_len, deadline, eta0=0.5):
    a = a0.copy()
    # ensure sum=1
    s = a.sum()
    if s <= 0:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s
    best_a = a.copy()
    best_val = _objective(a, n)
    iteration = 0
    while time.time() < deadline:
        # compute conv
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # active mask
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break
        w = active.astype(float) / active.sum()
        # gradient of max (subgradient)
        grad = 2.0 * _cross_correlation(w, a, fft_len)
        # step size
        eta = eta0 / np.sqrt(iteration + 1.0)
        # gradient descent update
        a_new = a - eta * grad
        # project onto simplex
        a_new = _project_simplex(a_new)
        a = a_new
        # evaluate
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if iteration % 1000 == 0:
                print(f"[GD it={iteration:06d}] new best {val:.6f}")
        iteration += 1
        # maybe break if iteration too many
        if iteration > 5_000_000:
            break
    return best_a, best_val

2. **Projected gradient descent on the true max‑norm** – for each retained
   vector we run a fast sub‑gradient method that works directly on the
   simplex (Euclidean projection after each step).  The gradient is a
   sub‑gradient of the `max` of the self‑convolution, obtained from the
   active set of maximal indices.  A decreasing stepsize η_t = η₀/√t
   guarantees convergence.  Several random Dirichlet restarts are used
   to explore different basins.

3. **Deterministic hill‑climbing (incremental mass transfer)** – starting
   from the best point found, repeatedly move a small fraction of mass
   from the index with the largest sub‑gradient (most responsible for the
   current peak) to the index with the smallest sub‑gradient.  The
   convolution is updated incrementally (‑2 δ a and +2 δ a slices) so that
   each move costs O(n) instead of a full FFT.  The objective is accepted
   only when it strictly improves; the loop runs until the time budget
   expires.

The routine respects the given time budget (default ≈ 1000 s),
always returns a non‑negative list, and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep only the first 2*len(a)-1 entries (linear convolution)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Projection onto the probability simplex ------------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the simplex {x >= 0, sum x = 1}.
    Implements the algorithm of Duchi et al. (2008).
    """
    n = v.shape[0]
    if n == 0:
        return v
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u)
    rho = np.nonzero(u + (1.0 - cssv) / (np.arange(1, n + 1)) > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = (cssv[rho] - 1.0) / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑norm --------------------

def _gd_optimize(a0: np.ndarray, n: int, fft_len: int, deadline: float,
                 eta0: float = 0.6, max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on 2·n·max(conv)/(sum a)².
    The gradient used is a sub‑gradient of the max‑norm.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if not active.any():
            break
        w = active.astype(float) / active.sum()
        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a_new = a - eta * grad
        a_new = _project_simplex(a_new)

        a = a_new
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 500 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Deterministic hill‑climbing with incremental mass transfer -----------

def _hill_climb(a0: np.ndarray, n: int, fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly transfer a small fraction of mass from the
    index with largest sub‑gradient to the one with smallest.  Convolution is
    updated incrementally (O(n) per move).  Only improving moves are kept.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    best_a = a.copy()
    best_val = _objective(a, n)

    moves = 0
    conv_len = 2 * n - 1

    while time.time() < deadline:
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        w = active.astype(float) / active.sum()
        grad = 2.0 * _cross_correlation(w, a, fft_len)

        donor = int(np.argmax(grad))
        receiver = int(np.argmin(grad))

        if a[donor] < 1e-15:
            # pick any non‑zero donor
            nz = np.where(a > 1e-15)[0]
            if nz.size == 0:
                break
            donor = int(np.random.choice(nz))
        if donor == receiver:
            receiver = (receiver + 1) % n

        # move a modest fraction of the donor mass
        delta = a[donor] * np.random.uniform(0.01, 0.04)

        # store old a for safety (used only for conv update)
        a_old = a.copy()

        # Incremental update of the convolution
        conv[donor:donor + n] -= 2.0 * delta * a_old
        conv[receiver:receiver + n] += 2.0 * delta * a_old
        conv[2 * donor] += delta * delta
        conv[2 * receiver] += delta * delta
        conv[donor + receiver] -= delta * delta

        # Update the mass vector
        a[donor] -= delta
        a[receiver] += delta
        a = np.clip(a, 0.0, None)

        # Renormalise occasionally to avoid drift
        if moves % 2000 == 0:
            s = a.sum()
            if abs(s - 1.0) > 1e-12:
                a /= s
                conv = _full_convolution(a, fft_len)

        cur_val = 2.0 * n * conv.max() / (a.sum() ** 2)

        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            print(f"[HC] new best {cur_val:.6f} after {moves} moves")
        else:
            # reject move – revert (very cheap because we have a_old)
            a = a_old
            # revert convolution exactly
            conv[donor:donor + n] += 2.0 * delta * a_old
            conv[receiver:receiver + n] -= 2.0 * delta * a_old
            conv[2 * donor] -= delta * delta
            conv[2 * receiver] -= delta * delta
            conv[donor + receiver] += delta * delta

        moves += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Respects the time budget and always returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # problem size – a modest length gives a good speed/quality trade‑off
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    best_seq = None
    best_val = float("inf")

    # -------------------- 1) parametric sweep --------------------
    # Systematic families (a reduced but still rich selection)
    # power‑law
    for p in np.linspace(-5.0, -0.05, 30):
        a = _candidate_power(p, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # exponential
    for alpha in np.logspace(-6, 1.2, 35):
        a = _candidate_exponential(alpha, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # stretched exponential
    for alpha in np.logspace(-4, 0.6, 12):
        for beta in np.linspace(0.2, 3.5, 10):
            a = _candidate_stretched(alpha, beta, n)
            v = _objective(a, n)
            if v < best_val:
                best_val, best_seq = v, a.copy()
                print(f"[Sweep] new best {v:.6f}")

    # beta distributions (random draws)
    for _ in range(80):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        a = _candidate_beta(a_par, b_par, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # double Gaussian
    for sig in np.logspace(-3.0, -0.1, 20):
        a = _candidate_double_gaussian(sig, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # monotone left‑exponential
    for alpha in np.logspace(-6, 1.5, 20):
        a = _candidate_left_exponential(alpha, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # monotone left‑power
    for p in np.linspace(-4.0, 1.0, 20):
        a = _candidate_left_power(p, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # sine bumps
    for power in np.linspace(0.5, 4.5, 12):
        a = _candidate_sine(power, n)
        v = _objective(a, n)
        if v < best_val:
            best_val, best_seq = v, a.copy()
            print(f"[Sweep] new best {v:.6f}")

    # random spikes with varying sparsity
    for k in [32, 64, 128, 256]:
        for _ in range(5):
            a = _candidate_spike(k, n, random=True)
            v = _objective(a, n)
            if v < best_val:
                best_val, best_seq = v, a.copy()
                print(f"[Sweep] new best {v:.6f}")

    # warm start from a previously stored construction (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            v = _objective(w, n)
            if v < best_val:
                best_val, best_seq = v, w.copy()
                print(f"[Sweep] new best (warm) {v:.6f}")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Projected gradient descent (multiple restarts)
    remaining = deadline - time.time()
    if remaining > 30.0 and best_seq is not None:
        print("[Phase‑GD] projected gradient descent with restarts")
        # start points: best from sweep + a few random Dirichlet vectors
        init_vectors = [best_seq]
        for _ in range(4):
            init_vectors.append(np.random.dirichlet(np.ones(n)))
        per_vec_time = remaining * 0.55 / len(init_vectors)

        for idx, a0 in enumerate(init_vectors):
            if time.time() >= deadline:
                break
            vec_deadline = min(deadline - 0.2, time.time() + per_vec_time)
            print(f"  [GD] restart {idx} (time budget {vec_deadline - time.time():.1f}s)")
            a_opt, v_opt = _gd_optimize(a0, n, fft_len, vec_deadline, eta0=0.7)
            if v_opt < best_val:
                best_val, best_seq = v_opt, a_opt.copy()
                print(f"  [GD] restart {idx} improved to {v_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Deterministic hill‑climbing fine‑tune (if time permits)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase‑HC] final hill‑climbing refinement")
        hc_deadline = deadline - 2.0
        a_opt, v_opt = _hill_climb(best_seq, n, fft_len, hc_deadline)
        if v_opt < best_val:
            best_val, best_seq = v_opt, a_opt.copy()
            print(f"[Phase‑HC] improved to {v_opt:.6f}")

    # --------------------------------------------------------------
    # Final sanity checks and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Evaluate with the public checker (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print(f"Could not run evaluate_sequence: {exc}")

    return best_seq.tolist()
