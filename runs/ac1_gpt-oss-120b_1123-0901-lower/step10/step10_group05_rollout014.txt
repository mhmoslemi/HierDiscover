"""
Hybrid soft‑max / Adam optimizer for the max‑convolution objective.

The algorithm works with a fixed length (default 4096) but can be
changed by editing the constant ``n``.  It proceeds as follows:

1. Build a small pool of diverse initial vectors (warm‑start,
   power‑law ends‑heavy, Beta‑shapes, Dirichlet draws).
2. Keep the best of those as a starting point.
3. Run a continuation method: for a series of increasing
   “soft‑max” temperatures `beta` we minimise the smoothed
   objective  Fβ(a)=∑_k w_k·(a*a)_k with
   w_k∝exp(beta·(a*a)_k).  The gradient of Fβ is
   g=2·(w ★ a) and is fed into an Adam optimiser that updates the
   vector with an exponentiated‑gradient step (keeps positivity).
   The step size decays like 1/√t.
4. The whole refinement is performed from several random restarts
   (occasionally perturbed copies of the current best) until the
   time budget is exhausted.
5. A short simulated‑annealing polishing phase is applied as a final
   touch‑up.

The returned sequence is guaranteed to be non‑negative, sums to 1,
and respects the evaluator’s clipping limits.  Intermediate best
values and timings are printed for monitoring.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns an array of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) that minimises
    the evaluation metric used in the challenge.  The routine respects the
    given time budget (≈ 1000 s) and always returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (same as in the reference solution)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helper – objective for a normalised vector (sum = 1)
    def _objective_val(a_vec: np.ndarray) -> float:
        conv = _full_convolution(a_vec, conv_fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 1️⃣  Build a modest pool of starter vectors
    pool = []

    # warm start, if supplied externally
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            pool.append(w)

    # power‑law ends‑heavy (α < 0)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in (-3.0, -4.0, -5.0):
        shape = prod ** alpha
        shape /= shape.sum()
        pool.append(shape)

    # Beta‑shapes (a_i ∝ x^p (1‑x)^q)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for p, q in ((0.5, 0.5), (1.0, 2.0), (2.0, 5.0)):
        shape = (x ** p) * ((1.0 - x) ** q)
        if shape.sum() > 0:
            shape /= shape.sum()
            pool.append(shape)

    # Random Dirichlet draws (a few)
    for _ in range(5):
        shape = np.random.dirichlet(np.full(n, 0.5))
        pool.append(shape)

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of starters – keep the best one
    best_seq = None
    best_val = float("inf")
    for i, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        if a.sum() < 0.01:
            continue
        a /= a.sum()
        val = _objective_val(a)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{i:03d}] value {val:.6f}")

    # fallback to uniform if nothing survived
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective_val(best_seq)
        print("[Fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Refinement with soft‑max smoothing + Adam (continuation)
    def _refine(start_vec: np.ndarray) -> None:
        """Run the Adam‑based continuation from a given start vector."""
        nonlocal best_seq, best_val

        a = start_vec.copy()

        # Adam hyper‑parameters
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        t = 0
        base_lr = 0.5

        # soft‑max temperature schedule (increase → focus on the max)
        beta_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0, 50.0,
                         100.0, 200.0, 500.0, 1000.0]

        for beta in beta_schedule:
            # a generous inner budget – will be cut by the global time limit
            max_iter = 20000
            for _ in range(max_iter):
                if time.time() > deadline - 1.0:
                    return
                t += 1

                # convolution and current value
                conv = _full_convolution(a, conv_fft_len)
                max_c = conv.max()
                val = 2.0 * n * max_c
                if val < best_val:
                    best_val = val
                    best_seq = a.copy()
                    print(f"[Refine β={beta:.1f} it={t}] new best {val:.6f}")

                # ----- soft‑max weights -------------------------------------------------
                # shift by the current maximum for numerical stability
                shift = max_c
                exp_arg = beta * (conv - shift)
                exp_arg = np.clip(exp_arg, -50, 50)      # avoid overflow/underflow
                w = np.exp(exp_arg)
                w_sum = w.sum()
                if w_sum == 0 or not np.isfinite(w_sum):
                    w = np.ones_like(conv) / conv.size
                else:
                    w /= w_sum

                # ----- gradient (g = 2·(w ★ a)) -----------------------------------------
                grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

                # ----- Adam update (exponentiated‑gradient form) -----------------------
                m = beta1 * m + (1.0 - beta1) * grad
                v = beta2 * v + (1.0 - beta2) * (grad * grad)

                m_hat = m / (1.0 - beta1 ** t)
                v_hat = v / (1.0 - beta2 ** t)

                lr = base_lr / np.sqrt(t)               # simple decay
                step = lr / (np.sqrt(v_hat) + eps)

                # exponentiated‑gradient step (keeps positivity)
                a = a * np.exp(-step * m_hat)

                # renormalise to the simplex
                a = np.clip(a, 0.0, 1000.0)
                s = a.sum()
                if s < 1e-12:
                    a = np.full_like(a, 1.0 / n)
                else:
                    a /= s

    # ------------------------------------------------------------------
    # 4️⃣  Multiple restarts until time runs out
    while time.time() < deadline - 2.0:
        # 30 % of the time we perturb the current best, otherwise start anew
        if best_seq is not None and np.random.rand() < 0.3:
            start = best_seq * (1.0 + 0.02 * (np.random.rand(n) - 0.5))
        else:
            # fresh Dirichlet draw (moderately diffuse)
            start = np.random.dirichlet(np.full(n, 0.5))
        start = np.clip(start, 0.0, 1000.0)
        start /= start.sum()
        _refine(start)

    # ------------------------------------------------------------------
    # 5️⃣  Very short simulated‑annealing polish (helps escape tiny pits)
    if time.time() < deadline - 1.0 and best_seq is not None:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        best_val_sa = cur_val
        best_a_sa = a.copy()

        T_start, T_end = 0.05, 1e-5
        max_moves = 50000
        for mv in range(max_moves):
            if time.time() > deadline - 0.5:
                break
            T = max(T_end, T_start * (1.0 - mv / max_moves))
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue
            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of mass at i
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta
            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()
            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[Anneal move {mv}] improved to {val2:.6f}")
            else:
                if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                    a = a2
        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣  Final clean‑up and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)   # matches evaluator’s limits
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
