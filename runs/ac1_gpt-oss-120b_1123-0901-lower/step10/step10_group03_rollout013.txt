"""
Step‑function optimizer (search for a low evaluation score).

The optimizer works in four stages:

1. **Broad parametric sweep** – many families (exponential, power‑law,
   stretched‑exponential, Gamma‑like, Beta, cosine‑power, double‑Gaussian,
   random spikes etc.).  Each candidate is normalised to sum = 1 and its exact
   objective 2·n·max(a*a)/(∑a)² is computed (FFT based).  The best candidate is
   kept.

2. **Smooth‑max Adam** – a short Adam optimisation on the log‑sum‑exp (smooth‑max)
   surrogate of the max‑convolution.  Soft‑max parametrisation guarantees that
   the iterates stay on the simplex.

3. **Adam on the true max** – from the best candidate a second Adam run is
   performed, this time using the exact sub‑gradient of the max‑convolution.
   Adam’s momentum accelerates convergence compared to the plain
   exponentiated‑gradient used previously.

4. **Greedy local refinement** – a cheap hill‑climbing phase that repeatedly
   transfers a tiny fraction of mass from coordinates with a positive gradient
   (increase the max) to coordinates with a negative gradient (decrease the max).
   This fine‑tunes the solution and typically yields a few × 10⁻³ improvement.

All phases respect the overall time budget (default 1000 s) and the routine
always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for sub‑gradient (same length as a).

    Implements the operation described in the original code:
    reverse a, FFT‑multiply with w and extract the centred slice.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_cosine_pow(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (1.0 - np.cos(2.0 * np.pi * i / (n - 1))) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sine_pow(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _adam_smoothmax_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (on the simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.5          # start temperature (larger than the previous 0.1)
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (track)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam‑smooth it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # gradient w.r.t logits (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # gradual sharpening of the smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional LR decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _adam_max_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation directly on the exact max‑convolution.
    Operates on the probability simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03               # a bit smaller than the smooth‑max phase
    it = 0
    best_a = a0.copy()
    best_val = _objective(a0, n)

    while time.time() < deadline and it < max_iters:
        # soft‑max to enforce non‑negativity and sum=1
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Adam‑max it={it:06d}] new best {val:.6f}")

        # indices attaining the maximum (active set)
        active = np.isclose(conv, max_c, atol=1e-12)
        if not active.any():
            # should not happen, but protect
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()   # uniform over active indices

        # exact sub‑gradient of max(conv) w.r.t a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # map to gradient over logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        it += 1

    return best_a, best_val


def _local_refine_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Greedy hill‑climbing refinement.
    Repeatedly transfers a tiny fraction of mass from a coordinate with a
    positive gradient (increase the max) to one with a negative gradient
    (decrease the max).  Accepts occasional uphill moves via a simple
    simulated‑annealing schedule.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    # temperature schedule for occasional uphill moves
    T_start = 0.05
    T_end = 1e-5

    for it in range(max_moves):
        if time.time() >= deadline:
            break

        # recompute only when needed (every few iterations) – cheap enough here
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # pick source (largest positive gradient) and destination (most negative)
        pos_idx = np.argmax(grad_a)
        neg_idx = np.argmin(grad_a)

        if a[pos_idx] < 1e-14:
            continue

        # transfer a small fraction (≤2 %) of the source mass
        frac = np.random.rand() * 0.02
        delta = a[pos_idx] * frac

        a2 = a.copy()
        a2[pos_idx] -= delta
        a2[neg_idx] += delta
        a2 /= a2.sum()  # renormalise (protect against round‑off)

        val2 = 2.0 * n * _full_convolution(a2, conv_fft_len).max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            # simulated‑annealing acceptance
            T = max(T_end, T_start * (1.0 - it / float(max_moves)))
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

    return best_a, best_val

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep ------------------------------------
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ----- symmetric families ------------------------------------------------
    for p in np.linspace(-6.0, -0.1, 30):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-4, 2.5, 35):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # ----- monotone families -------------------------------------------------
    for alpha in np.logspace(-4, 2.0, 30):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-6.0, 2.0, 30):
        _maybe_update(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 12):
        for alpha in np.logspace(-4, 1.5, 25):
            _maybe_update(_candidate_left_stretched(alpha, beta, n))

    # ----- Gamma‑like --------------------------------------------------------
    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            _maybe_update(_candidate_gamma(shape, scale, n))

    # ----- Beta‑distribution -------------------------------------------------
    a_pars = np.logspace(-2, 0.5, 12)
    b_pars = np.logspace(0, 1.0, 12)
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # ----- double Gaussian ---------------------------------------------------
    for sig in np.logspace(-3, -0.2, 15):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # ----- cosine / sine powered families ------------------------------------
    for p in np.linspace(0.2, 4.0, 15):
        _maybe_update(_candidate_cosine_pow(p, n))
        _maybe_update(_candidate_sine_pow(p, n))

    # ----- random spike patterns ---------------------------------------------
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # ----- mixtures of two exponentials (random) ----------------------------
    for _ in range(200):
        alpha1 = 10 ** np.random.uniform(-4, 2)
        alpha2 = 10 ** np.random.uniform(-4, 2)
        w1 = np.random.rand()
        a_mix = w1 * _candidate_left_exponential(alpha1, n) + \
                (1 - w1) * _candidate_left_exponential(alpha2, n)
        a_mix /= a_mix.sum()
        _maybe_update(a_mix)

    # ----- Dirichlet random vectors (various concentrations) ----------------
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # ----- warm‑start from a previously‑found global variable -------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam refinement (smooth‑max) --------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑1‑Adam‑smooth] short smooth‑max Adam")
        adam_deadline = min(deadline - 5, time.time() + 30.0)  # up to ~30 s
        best_seq, val_adam = _adam_smoothmax_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = best_seq.copy()
            print(f"[Phase‑1‑Adam‑smooth] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam optimisation on the exact max ---------------------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑2‑Adam‑max] optimiser on the true max")
        max_deadline = deadline - 5
        best_seq, val_max = _adam_max_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            max_deadline,
            max_iters=10_000_000,
        )
        if val_max < best_val:
            best_val = val_max
            best_seq = best_seq.copy()
            print(f"[Phase‑2‑Adam‑max] improved to {val_max:.6f}")

    # ------------------------------------------------------------------
    # 4) Greedy local refinement --------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑3‑Refine] greedy local optimisation")
        best_seq, val_refine = _local_refine_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_refine < best_val:
            best_val = val_refine
            best_seq = best_seq.copy()
            print(f"[Phase‑3‑Refine] improved to {val_refine:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
