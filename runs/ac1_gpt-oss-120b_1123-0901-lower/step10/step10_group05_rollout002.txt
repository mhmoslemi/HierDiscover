"""
Hybrid parametric + smooth‑gradient optimizer for the max‑convolution
objective.

The algorithm proceeds in three phases, all respecting the global time
budget (≈ 1000 s).

1️⃣  **Broad parametric sweep** – generate many candidate shapes
    (power‑law ends‑heavy, beta‑like, uniform, and mixtures thereof),
    normalise each to sum = 1 and keep the best (lowest exact objective
    2·n·max(a★a)).

2️⃣  **Smooth‑gradient refinement** – starting from the best candidate,
    minimise a smooth approximation
        fβ(a)= (1/β)·log Σk exp(β·(a★a)k)
    using exponentiated‑gradient (multiplicative) updates with an
    Adam‑style schedule.  The smoothing parameter β is slowly increased,
    so the iterations first follow a gentle landscape and end up
    optimising the true max‑convolution.

3️⃣  **Short simulated‑annealing polish** – a lightweight stochastic
    local search (mass transfers between two coordinates) that can
    escape tiny local pits.

All helper functions are defined at module level; no closures or lambdas
are used.  The final sequence is clipped to the evaluator limits
(non‑negative, ≤ 1000) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    Returns the first (2·len(a)‑1) entries.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) of length len(a).
    Implemented as convolution of w with the reversed a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a★a) for a vector a with sum = 1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _smooth_gradient(a: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     beta: float) -> np.ndarray:
    """
    Gradient of the smooth approximation
        fβ(a) = (1/β) log Σk exp(β·(a★a)k)
    with respect to a.  Returns ∇fβ(a).
    """
    conv = _full_convolution(a, conv_fft_len)          # length 2n‑1
    # softmax weights over the convolution entries
    w = np.exp(beta * conv)
    w_sum = w.sum()
    if w_sum == 0.0:
        # numerical safeguard – fall back to hard max weighting
        w = np.isclose(conv, conv.max()).astype(float)
        w /= w.sum()
    else:
        w /= w_sum
    # gradient = 2·(w ★ a)
    grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
    return grad

    the evaluation function.  The routine respects the given time budget
    and always returns the best sequence found.

    Parameters
    ----------
    seed : int, optional
        Seed for NumPy's RNG to make the search deterministic.
    budget_s : float, optional
        Time budget in seconds (default 1000).

    Returns
    -------
    list[float]
        The best sequence (length n) discovered, as a plain Python list.
    """

    deadline = time.time() + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem size – keep the same dimension used by the reference solution
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Helper for normalisation & objective
    def eval_and_update(cand: np.ndarray, tag: str = "") -> None:
        nonlocal best_seq, best_val
        if cand.ndim != 1 or cand.shape[0] != n:
            return
        # clip to allowable range before normalisation (just in case)
        cand = np.clip(cand, 0.0, 1000.0)
        s = cand.sum()
        if s < 1e-12:
            return
        cand = cand / s                     # enforce sum = 1
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Sweep {tag}] new best {val:.6f}")

    # ---- warm start from previous best if available -----------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        eval_and_update(w, "warm")

    # ---- uniform baseline ------------------------------------------
    uniform = np.full(n, 1.0 / n, dtype=np.float64)
    eval_and_update(uniform, "uniform")

    # ---- power‑law ends‑heavy (i·(n‑i+1))^α -------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # fine grid for α around the known good region
    for alpha in np.linspace(-2.5, -0.05, 250):
        shape = prod ** alpha
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        eval_and_update(shape, f"alpha={alpha:.4f}")

    # ---- mixture of the best power‑law shape with uniform ---------
    # (we will have a candidate for the best α after the loop)
    if best_seq is not None:
        best_alpha_shape = best_seq
        for lam in np.linspace(0.0, 1.0, 21):
            mixed = lam * best_alpha_shape + (1.0 - lam) * uniform
            # both components already sum to 1 → mixed also sums to 1
            eval_and_update(mixed, f"mixλ={lam:.2f}")

    # ---- beta‑like shapes  x^p (1-x)^q  ---------------------------
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    # random sampling of (p,q) on a log‑scale
    for _ in range(300):
        # draw p,q ∈ [0.01, 10] logarithmically
        logp = np.random.uniform(-2, 1)   # 10^{-2}=0.01 … 10^{1}=10
        logq = np.random.uniform(-2, 1)
        p = 10 ** logp
        q = 10 ** logq
        shape = (x ** p) * ((1.0 - x) ** q)
        if shape.sum() == 0:
            continue
        shape /= shape.sum()
        eval_and_update(shape, f"beta p={p:.3f} q={q:.3f}")

    # ---- random mixtures of two power‑law shapes ------------------
    alphas = np.linspace(-2.5, -0.05, 30)
    for _ in range(200):
        a1, a2 = np.random.choice(alphas, size=2, replace=False)
        shape1 = prod ** a1
        shape2 = prod ** a2
        shape1 /= shape1.sum()
        shape2 /= shape2.sum()
        lam = np.random.rand()
        mixed = lam * shape1 + (1.0 - lam) * shape2
        eval_and_update(mixed, f"mixα {a1:.2f}/{a2:.2f}")

    # --------------------------------------------------------------
    # 2️⃣  Smooth‑gradient (exponentiated‑gradient) refinement
    # --------------------------------------------------------------
    if best_seq is None:
        # fall back to uniform if everything failed (should never happen)
        best_seq = uniform.copy()
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # Ensure strictly positive entries (required for multiplicative updates)
    eps = 1e-15
    a = best_seq.copy()
    a = np.maximum(a, eps)
    a /= a.sum()

    # Parameters for the smooth optimisation
    step_const = 0.7          # initial step‑size factor
    beta = 5.0                # start with mild smoothing
    beta_max = 500.0
    it = 0
    max_iters = 250_000

    while time.time() < deadline - 2.0 and it < max_iters:
        # exact objective for logging / best‑track
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Refine it={it:06d}] new best exact {val:.6f}")

        # smooth gradient (log‑sum‑exp surrogate)
        grad = _smooth_gradient(a, n, conv_fft_len, corr_fft_len, beta)

        # step size – 1/√t schedule
        step = step_const / np.sqrt(it + 1.0)

        # multiplicative (exponentiated) update
        log_factor = -step * grad
        # avoid overflow/underflow in the exponential
        log_factor = np.clip(log_factor, -50.0, 50.0)
        a_new = a * np.exp(log_factor)

        # re‑normalise to the simplex
        s = a_new.sum()
        if s == 0.0 or np.isnan(s):
            # catastrophic numerical issue – abort refinement
            break
        a = a_new / s

        # slowly increase the smoothing parameter so we end up optimising the true max
        beta = min(beta * 1.001, beta_max)

        it += 1

    # --------------------------------------------------------------
    # 3️⃣  Very short simulated‑annealing polish (optional)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        best_val_sa = cur_val
        best_a_sa = a.copy()

        T_start = 0.07
        T_end = 1e-5
        max_moves = 70_000
        for mv in range(max_moves):
            if time.time() > deadline - 1.0:
                break
            # linear temperature schedule
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            # move a random fraction from i to j
            delta = a[i] * (np.random.rand() * 0.4)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[SA move {mv}] improved to {val2:.6f}")
            else:
                # Metropolis criterion
                if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                    a = a2
                cur_val = val2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[SA final] best {best_val:.6f}")

    # --------------------------------------------------------------
    # Finalisation
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = uniform.copy()
        best_val = _objective(best_seq, n, conv_fft_len)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
