"""
Propose a step‑function (non‑negative sequence) that minimizes

    2·n·max(conv(a,a)) / (sum(a))²

where conv is the linear convolution.  The search consists of:

1.  A very broad parametric sweep (beta, power‑law, exponential,
    stretched‑exponential, sine, and random Dirichlet spikes).  All
    candidates are normalised to sum = 1 and the exact objective is
    evaluated with an FFT‑based convolution.  The best few candidates
    are kept.

2.  A long Adam optimisation on a smooth surrogate
        smooth_max(conv) + λ·‖conv‑uniform‖₂² .
    The surrogate is differentiable; the update is performed in the
    log‑softmax (mirror‑descent) parametrisation which guarantees a
    simplex‑constrained non‑negative vector.

3.  An exponentiated‑gradient sub‑gradient descent on the true
    max‑norm.  The active set of indices attaining the current maximum
    provides a sub‑gradient via cross‑correlation.

All stages are timed; the routine always returns a valid sequence
within the given budget (default ≈ 1000 s).  The final sequence is
clipped to the range required by ``evaluate_sequence`` before being
returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------

def _candidate_beta(alpha: float, beta: float, n: int) -> np.ndarray:
    """
    Beta‑distribution shape on [0,1]; alpha,beta>0.
    Points are the cell centres (i+0.5)/n.
    """
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (alpha - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (beta - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^{power}."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k equal‑mass spikes (Dirichlet if random)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max + L₂ regularisation) ---------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
    lam_start: float = 0.2,
    lam_factor: float = 1.01,
    l2_weight: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth‑max(conv) + λ·‖conv – uniform‖₂² .
    Returns the best vector encountered (on the simplex) and its exact
    objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # log‑softmax parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    # uniform target for L₂ regulariser
    uniform_target = np.full(2 * n - 1,
                             1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution (for bookkeeping)
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max (soft‑max) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L₂ part -----
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        # combined gradient w.r.t. a
        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – smooth‑max becomes sharper
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.90

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 15_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:08d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirrored on the simplex)
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget (seconds) and always returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem dimension – a modest size keeps the FFT cheap
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep -----------------------------------
    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray, label: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {label}] new best {val:.6f}")

    # ---- Beta grid (symmetric and asymmetric) ----
    beta_vals = np.geomspace(0.05, 5.0, num=30)
    for alpha in beta_vals:
        for beta in beta_vals:
            _try_candidate(_candidate_beta(alpha, beta, n), f"beta({alpha:.3g},{beta:.3g})")

    # ---- Power‑law (edge‑heavy) ----
    for p in np.linspace(-6.0, -0.02, 70):
        _try_candidate(_candidate_power(p, n), f"power({p:.3f})")

    # ---- Simple exponential decay (symmetric) ----
    for alpha in np.logspace(-7, 1.5, 80):
        _try_candidate(_candidate_exponential(alpha, n), f"exp({alpha:.3e})")

    # ---- Stretched exponential (two‑parameter) ----
    alphas = np.logspace(-4, 0.6, 15)
    betas = np.linspace(0.2, 3.5, 12)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n),
                           f"strexp({a_alpha:.3e},{b_beta:.3f})")

    # ---- Sine‑shaped bumps (varying power) ----
    for pwr in np.linspace(0.4, 5.0, 22):
        _try_candidate(_candidate_sine(pwr, n), f"sine({pwr:.3f})")

    # ---- Random Dirichlet spikes (different sparsities) ----
    for k in [16, 32, 64, 128, 256]:
        for _ in range(6):
            _try_candidate(_candidate_spike(k, n, random=True),
                           f"spike{k}")

    # ---- Left‑only monotone families (useful for asymmetric shapes) ----
    for alpha in np.logspace(-6, 1.5, 40):
        _try_candidate(_candidate_exponential(alpha, n), f"leftexp({alpha:.3e})")
    for p in np.linspace(-5.0, 2.0, 35):
        _try_candidate(_candidate_power(p, n), f"leftpow({p:.3f})")

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max + L₂) -----------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] Starting Adam refinement")
        # devote up to 30 % of remaining time to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(remaining * 0.30, 250.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=2_000_000,
            lam_start=0.2,
            lam_factor=1.01,
            l2_weight=0.20,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient optimisation (exponentiated‑gradient)-
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] Starting sub‑gradient optimisation")
        sub_deadline = deadline - 1.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=20_000_000,
            eta0=1.5,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final sanity & return ----------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # ensure feasibility for the public evaluator
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    s = best_seq.sum()
    if s < 1e-8:
        best_seq = np.full_like(best_seq, 1.0 / n)
    else:
        best_seq = best_seq / s          # normalise (scale‑invariant)

    # optional verification with the given evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluator missing).")

    return best_seq.tolist()
