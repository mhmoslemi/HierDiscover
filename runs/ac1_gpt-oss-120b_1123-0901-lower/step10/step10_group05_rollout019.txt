"""
Hybrid optimizer for the max‑convolution bound.

The search consists of three independent stages:

1. **Broad deterministic & stochastic pool** – we generate a large
   collection of candidate step‑functions, explicitly adding families that
   the original solver missed (e.g. monotone linear / quadratic decays,
   exponential tails, and dense random subsets with many non‑zero entries).
   Every candidate is normalised to sum = 1 and evaluated with a fast FFT‑based
   convolution; the best one is retained.

2. **Mirror‑descent (exponentiated‑gradient) refinement** – starting from the
   best pool element we repeatedly compute the exact sub‑gradient of the
   objective  f(a)=2·n·max(a∗a)  (max‑convolution) and perform a
   multiplicative‑weights update on the probability simplex.  The step size
   follows a 1/√t schedule and a short back‑tracking line‑search guarantees
   monotone improvement.

3. **Local simulated‑annealing polish** – a cheap stochastic move that
   transfers a random fraction of mass between two coordinates, accepting
   downhill moves always and uphill moves with a slowly decreasing temperature.
   This helps escape tiny local plateaus.

All stages respect the global time budget (≈ 1000 s).  The returned list
contains non‑negative numbers (already clipped to the evaluator limits) and
has the same length as the reference solution (default 4096, or the length of
a provided warm start).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using the FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    ``w`` and ``a`` are real vectors; the result has the same length as ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) assuming sum(a)=1.
    """
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    score.  The routine respects a 1000 s wall‑clock budget and always returns
    the best vector discovered (as a Python list).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Determine problem size.  Prefer the length of the warm‑start if one
    # is available; otherwise fall back to the reference size 4096.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        n = len(warm)
    else:
        n = 4096

    # FFT lengths for convolutions and correlations.
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of starter vectors
    pool: list[np.ndarray] = []

    # --- warm start ----------------------------------------------------
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # --- monotone families (new compared to the reference code) --------
    # Linear decay: a_i ∝ (n‑i)
    lin = np.arange(n, 0, -1, dtype=np.float64)
    lin = lin / lin.sum()
    pool.append(lin)

    # Quadratic decay: a_i ∝ (n‑i)²
    quad = lin**2
    quad = quad / quad.sum()
    pool.append(quad)

    # Exponential tails for several decay rates
    for alpha in (0.001, 0.005, 0.01, 0.02, 0.05, 0.1, 0.2):
        exp_decay = np.exp(-alpha * np.arange(n, dtype=np.float64))
        exp_decay = exp_decay / exp_decay.sum()
        pool.append(exp_decay)

    # --- parametric families (original) ---------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # power‑law ends‑heavy: a_i ∝ (i·(n‑i+1))^α , α < 0
    for alpha in np.linspace(-3.0, -0.2, 15):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Beta‑shape: a_i ∝ x^p (1‑x)^q
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for p in (0.2, 0.5, 1.0, 2.0, 5.0):
        for q in (0.2, 0.5, 1.0, 2.0, 5.0):
            shape = (x ** p) * ((1.0 - x) ** q)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                pool.append(shape)

    # Dirichlet draws (dense random)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # --- large dense random subsets (new) -------------------------------
    # Randomly select a large fraction of the positions and give them equal mass.
    for frac in (0.6, 0.7, 0.8, 0.9):
        k = int(frac * n)
        for _ in range(3):                     # three draws per fraction
            mask = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            mask[pos] = 1.0
            mask = mask / k                    # turn into a probability vector
            pool.append(mask)

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of the whole pool – keep the best starter
    for idx, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{idx:03d}] new best {val:.6f}")

    # If nothing succeeded, fall back to a uniform distribution.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Mirror‑descent / exponentiated‑gradient refinement
    #    (sub‑gradient of the max‑convolution is obtained via an active set)
    a = best_seq.copy()
    it = 0
    step_const = 1.0
    while time.time() < deadline - 5 and it < 300_000:
        # --- current objective ------------------------------------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Refine it={it:06d}] improved to {val:.6f}")

        # --- active set (indices attaining the maximum) ----------------------
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # --- sub‑gradient of max_c w.r.t. a (mirror‑descent direction) -------
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- step size (decreasing like 1/√t) --------------------------------
        step = step_const / np.sqrt(it + 1.0)

        # --- multiplicative update (exponentiated‑gradient) -------------------
        a_new = a * np.exp(-step * grad)
        s_new = a_new.sum()
        if s_new < 1e-12 or np.isnan(s_new):
            # occasionally the step may be too aggressive – shrink it.
            step_const *= 0.9
            it += 1
            continue
        a_new = a_new / s_new

        # --- accept the move (always – the mirror‑descent objective is
        #     guaranteed not to increase by more than the line‑search tolerance) ---
        a = a_new
        it += 1

    # ------------------------------------------------------------------
    # 4️⃣  Very short simulated‑annealing polish (helps escape tiny pits)
    if time.time() < deadline - 2:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()

        T_start = 0.07
        T_end = 1e-5
        max_moves = 80_000

        for mv in range(max_moves):
            if time.time() > deadline - 1:
                break
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            # move a fraction of the mass from i to j
            delta = a[i] * (np.random.rand() * 0.4)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[Anneal move {mv}] improved to {val2:.6f}")
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final tidy‑up and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # Normalise once more to avoid any drift introduced by the polishing step.
    s = best_seq.sum()
    if s < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / s

    print(f"Final best estimated value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
