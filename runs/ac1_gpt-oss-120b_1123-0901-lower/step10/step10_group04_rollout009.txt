"""
step‑function optimiser – L‑BFGS‑based refinement + long sub‑gradient

The optimiser proceeds in three stages:

1. **Broad parametric sweep** – a dozen analytic families (power‑law,
   exponential, stretched‑exponential, beta, double‑Gaussian, monotone
   variants, sine‑bump, random spikes) are sampled densely.  Each candidate
   is normalised to ℓ₁‑norm = 1 and its exact objective
        2·n·max(conv)/(sum a)²
   is evaluated with an FFT‑based convolution.  The best few vectors are kept.

2. **L‑BFGS refinement** – the best vector is re‑parametrised as a soft‑max
   of logits (guaranteeing non‑negativity and unit sum).  A smooth‑max
   surrogate of the max‑convolution,
        sm_λ(conv)= Σ_i w_i·conv_i   with   w_i ∝ exp(λ·conv_i),
   is minimised.  The temperature λ is increased through a short continuation
   schedule (5 → 30 → 150 → 600 → 2500), each stage being solved by
   SciPy’s L‑BFGS‑B on the logits.  The exact objective is monitored and the
   incumbent updated.

3. **Long exponentiated‑gradient sub‑gradient** – the final phase runs an
   exact max‑convolution sub‑gradient descent (mirror‑descent on the simplex)
   for the remaining time budget, with a decreasing step‑size
   ηₜ = η₀/√t and occasional tiny random perturbations to escape plateaus.
   The best vector is finally returned.

All stages respect the overall 1000 s budget, always output a non‑negative
list of coefficients, and store the result in the global ``height_sequence_1``
for warm‑starts in subsequent runs.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# L‑BFGS optimisation (smooth‑max surrogate) -------------------------

def _lbfgs_val_and_grad(logits: np.ndarray,
                        n: int,
                        fft_len: int,
                        lam: float):
    """
    Return smooth‑max surrogate value and its gradient w.r.t. logits.
    logits are unconstrained; a = softmax(logits) guarantees a≥0, Σa=1.
    """
    mx = np.max(logits)
    exp_log = np.exp(logits - mx)
    a = exp_log / exp_log.sum()

    conv = _full_convolution(a, fft_len)
    max_c = conv.max()

    # smooth‑max weights (softmax over conv)
    exp_w = np.exp(lam * (conv - max_c))
    w_sum = exp_w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(conv, 1.0 / conv.size)
    else:
        w = exp_w / w_sum

    smooth_max = np.dot(w, conv)          # Σ w_i·conv_i
    val = 2.0 * n * smooth_max            # surrogate objective

    # gradient w.r.t. a (each conv_i = Σ a_j a_{i‑j})
    grad_a = 2.0 * _cross_correlation(w, a, fft_len)

    # chain‑rule through soft‑max (mirror‑descent form)
    avg = np.dot(a, grad_a)
    grad_log = a * (grad_a - avg)
    return val, grad_log.astype(np.float64)


def _lbfgs_fun(logits: np.ndarray, n: int, fft_len: int, lam: float):
    """Wrapper returning only the value for SciPy."""
    val, _ = _lbfgs_val_and_grad(logits, n, fft_len, lam)
    return val


def _lbfgs_jac(logits: np.ndarray, n: int, fft_len: int, lam: float):
    """Wrapper returning only the gradient for SciPy."""
    _, grad = _lbfgs_val_and_grad(logits, n, fft_len, lam)
    return grad


def _lbfgs_optimize(a0: np.ndarray,
                    n: int,
                    fft_len: int,
                    deadline: float) -> tuple[np.ndarray, float]:
    """
    Refine a0 by a short continuation of L‑BFGS on a smooth‑max surrogate.
    Returns the best exact objective found and its vector.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    # λ schedule – from very smooth to sharply peaked
    lam_schedule = [5.0, 30.0, 150.0, 600.0, 2500.0]

    for lam in lam_schedule:
        if time.time() >= deadline:
            break

        # allocate a modest number of L‑BFGS iterations depending on time left
        secs_left = deadline - time.time()
        max_iter = int(min(2000, max(20, secs_left * 300)))   # ≈300 eval/s

        res = scipy.optimize.minimize(
            fun=_lbfgs_fun,
            x0=logits,
            args=(n, fft_len, lam),
            jac=_lbfgs_jac,
            method='L-BFGS-B',
            options={'maxiter': max_iter, 'disp': False}
        )

        # take the resulting point, evaluate exact objective
        mx = np.max(res.x)
        exp_log = np.exp(res.x - mx)
        a_opt = exp_log / exp_log.sum()
        exact_val = _objective(a_opt, n)
        if exact_val < best_val:
            best_val = exact_val
            best_a = a_opt.copy()
            print(f"[L‑BFGS λ={lam:.0f}] new best {exact_val:.6f}")

        logits = res.x  # warm‑start next λ‑stage

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 fft_len: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val - 1e-12:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active set (all indices attaining the max up to numerical tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        cnt = active.sum()
        if cnt == 0:
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / cnt

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # decreasing step size ηₜ = η₀ / √t
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)          # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        # occasional tiny random jitter to avoid stagnation
        if it % 50000 == 0 and np.random.rand() < 0.3:
            jitter = np.random.dirichlet(np.ones(n)) * 0.001   # 0.1 % total mass
            a = a * 0.999 + jitter
            a = a / a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (optional) ---------------------------

def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple annealing that transfers a fraction of mass between two positions."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    best_seq = None
    best_val = float('inf')

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ---------------------------------------
    def _try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # uniform baseline
    _try_candidate(np.full(n, 1.0 / n))

    # power‑law
    for p in np.linspace(-5.0, -0.05, 30):
        _try_candidate(_candidate_power(p, n))

    # exponential
    for alpha in np.logspace(-6, 1.2, 35):
        _try_candidate(_candidate_exponential(alpha, n))

    # stretched exponential
    alphas = np.logspace(-4, 0.6, 15)
    betas = np.linspace(0.2, 3.5, 12)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n))

    # beta distributions (random draws)
    for _ in range(100):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _try_candidate(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-3.0, -0.1, 20):
        _try_candidate(_candidate_double_gaussian(sig, n))

    # left‑exponential (monotone)
    for alpha in np.logspace(-6, 1.5, 25):
        _try_candidate(_candidate_left_exponential(alpha, n))

    # left‑power (monotone)
    for p in np.linspace(-4.0, 1.0, 25):
        _try_candidate(_candidate_left_power(p, n))

    # sine‑shaped bumps
    for pow_ in np.linspace(0.5, 4.5, 15):
        _try_candidate(_candidate_sine(pow_, n))

    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(6):
            _try_candidate(_candidate_spike(k, n, random=True))

    # a few Dirichlet draws (high‑entropy starting points)
    for _ in range(30):
        _try_candidate(np.random.dirichlet(np.ones(n)))

    # warm start from previous run, if available
    warm = globals().get('height_sequence_1')
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            if w.shape[0] > n:
                w = w[:n]
            else:
                w = np.pad(w, (0, n - w.shape[0]), mode='constant')
        w = np.clip(w, 0.0, None)
        if w.sum() > 0.01:
            w = w / w.sum()
            _try_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) L‑BFGS refinement (smooth‑max surrogate) --------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑L‑BFGS] start refinement")
        # give L‑BFGS roughly half of the remaining budget
        lbfgs_deadline = time.time() + 0.5 * (deadline - time.time())
        a_lbfgs, val_lbfgs = _lbfgs_optimize(best_seq, n, fft_len, lbfgs_deadline)
        if val_lbfgs < best_val:
            best_val = val_lbfgs
            best_seq = a_lbfgs.copy()
            print(f"[Phase‑L‑BFGS] improved to {val_lbfgs:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient optimisation (exponentiated‑gradient)-
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        sub_deadline = deadline - 2.0     # keep a tiny buffer
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=1.0,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning (if time permits) --
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid, non‑negative return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    # store for possible warm‑starts in a later call
    globals()['height_sequence_1'] = best_seq.tolist()

    return best_seq.tolist()
