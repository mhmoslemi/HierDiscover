"""
Mirror‑descent optimizer with soft‑max smoothing of the max‑convolution.
Starts from a diverse pool of parametric candidates (including the warm
start), then iteratively minimizes a smooth approximation
    φ_λ(a)= (1/λ) log Σ_k exp(λ (a∗a)_k)
using exponentiated‑gradient updates.  λ is increased during the run,
turning the smooth surrogate into the true L_∞ objective.  The best
vector seen (by the exact evaluation) is returned.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of Σ w_k (a*a)_k.
    The result has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a with sum=1."""
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()

The routine

* builds a diverse pool of candidate vectors (including the provided warm‑start),
* picks the best one as a start,
* then minimises   φ_λ(a) = (1/λ)·log∑_k exp(λ·(a∗a)_k)   by exponentiated‑gradient
  updates (a_i ← a_i·exp(−η·∇φ_λ)_i) while increasing λ so that φ_λ → max(a∗a).
* a tiny random jitter is injected every 50 k iterations to escape flat plateaus.
The best (exact) objective 2·n·max(a∗a) seen so far is kept and finally
returned as a Python list.  All numbers are non‑negative and the vector is
normalised (sum = 1); the evaluator’s clipping therefore never activates.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a∗a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient of
    Σ_k w_k·(a∗a)_k.  Result has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

    """Exact objective 2·n·max(a∗a) for a with sum = 1."""
