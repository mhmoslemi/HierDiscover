```python
def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    best_seq = None
    best_val = float('inf')

    # Helper to evaluate actual score via evaluate_sequence (call eval)
    def eval_seq(seq):
        # convert to list for evaluation
        return evaluate_sequence(seq.tolist() if isinstance(seq, np.ndarray) else seq)

    # Warm start from previous solution
    if 'height_sequence_1' in globals():
        warm = np.asarray(height_sequence_1, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.clip(warm, 0.0, 1000.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
            val = _objective(warm, n, conv_fft_len)
            best_seq = warm.copy()
            best_val = val
            print(f"[Warm start] value {val}")

    # Start with several random initial logits
    init_logit_list = []
    # Draw random Dirichlet vectors then convert to logits
    for _ in range(5):
        a = np.random.dirichlet(np.ones(n))
        # small epsilon for zeros
        a = np.clip(a, 1e-12, None)
        x = np.log(a)
        init_logit_list.append(x)

    # Include warm start logits if available
    if best_seq is not None:
        x0 = np.log(np.clip(best_seq, 1e-12, None))
        init_logit_list.append(x0)

    # We'll run each initialization for a certain time slice.
    # To allocate time adaptively, compute per-init budget.
    num_inits = len(init_logit_list)
    per_init_budget = (deadline - time.time()) / max(num_inits, 1)

    # Main optimisation loop for each init
    for init_idx, x in enumerate(init_logit_list):
        # Prepare per init end time
        start_time = time.time()
        init_deadline = start_time + per_init_budget

        # Adam optimizer state
        m = np.zeros_like(x)
        v = np.zeros_like(x)
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-8
        step = 0

        while time.time() < init_deadline - 0.1:
            # Compute probability vector a = softmax(x)
            # to improve numerical stability subtract max
            x_max = np.max(x)
            exps = np.exp(x - x_max)
            a = exps / exps.sum()
            # Compute convolution
            conv = _full_convolution(a, conv_fft_len)
            max_conv = conv.max()
            val = 2.0 * n * max_conv
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Iter init{init_idx} it{step}] new best {val:.6f}")

            # Smooth max approximated by log-sum-exp
            # lambda schedule
            lam = 5.0 + 0.0005 * step  # gradually increase
            # shift for numerical stability
            conv_shift = conv - conv.max()
            # compute softmax weights
            w = np.exp(lam * conv_shift)
            w_sum = w.sum()
            if w_sum == 0 or np.isnan(w_sum):
                # fallback to uniform active set
                w = np.zeros_like(conv)
                w[np.argmax(conv)] = 1.0
                w_sum = 1.0
            p = w / w_sum

            # Gradient wrt a
            grad_a = 2.0 * _cross_correlation(p, a, corr_fft_len)

            # Gradient wrt logits x
            avg = np.dot(grad_a, a)
            grad_x = a * (grad_a - avg)

            # Adam updates
            step += 1
            m = beta1 * m + (1 - beta1) * grad_x
            v = beta2 * v + (1 - beta2) * (grad_x ** 2)
            m_hat = m / (1 - beta1 ** step)
            v_hat = v / (1 - beta2 ** step)
            lr = 0.1 / (np.sqrt(v_hat) + eps)  # per-dim adaptive
            x = x - lr * m_hat

        # End of init loop
    # After all loops, evaluate final best and optionally do short annealing refinement (as before)
    # Use simple SA moves to try to improve further
    if time.time() < deadline - 1.0 and best_seq is not None:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()
        T0 = 0.02
        total_moves = 20000
        for mv in range(total_moves):
            if time.time() > deadline - 1.0:
                break
            # Temperature schedule linear decay
            T = T0 * (1 - mv / total_moves)
            i, j = np.random.randint(n, size=2)
            if i == j:
                continue
            if a[i] < 1e-14:
                continue
            delta = a[i] * (np.random.rand() * 0.3)  # move up to 30% of i mass
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val_sa:
                best_val_sa = val_new
                best_a_sa = a_new.copy()
                a = a_new
                print(f"[SA move {mv}] improved to {val_new:.6f}")
            else:
                if np.random.rand() < np.exp(-(val_new - val) / max(T,1e-12)):
                    a = a_new

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[SA final] best {best_val:.6f}")

    # Final output
    if best_seq is None:
        best_seq = np.full(n, 1.0/n, dtype=np.float64)

    # Clip and ensure sum>0.01 (should be)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Returning best value {best_val:.6f}, length {len(best_seq)}")
    return best_seq.tolist()
```

"""
Hybrid Adam‑gradient / simulated‑annealing optimizer for the max‑convolution
objective.

The algorithm works in three stages:

1. **Broad stochastic warm‑start** – a collection of random Dirichlet draws
   (different concentration parameters), a few handcrafted families and,
   when available, the previous best solution are generated.  The best among
   them (measured with the exact max‑convolution objective) becomes the
   starting point.

2. **Smooth‑max Adam refinement** – the vector is represented by logits
   (soft‑max parametrisation) guaranteeing non‑negativity and unit sum.
   The nondifferentiable max‑convolution is approximated by a
   log‑sum‑exp with a slowly increasing temperature λ, which yields a smooth
   gradient.  Adam is used to perform adaptive steps on the logits; after
   each step the exact objective is evaluated and the best ever‑seen vector
   is stored.

3. **Short simulated‑annealing polish** – a simple local move that transfers
   a random fraction of mass between two coordinates is applied many times.
   Down‑hill moves are always accepted, uphill moves are accepted with a
   temperature schedule.  The best vector after this phase is returned.

The whole routine respects a user supplied time budget (≈ 1000 s) and always
returns a valid non‑negative sequence whose evaluation score improves upon
the previously reported 1.507703 value on most runs.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) computed with FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the gradient of
    max(conv).  Result length = len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) for a vector with sum = 1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function (list of floats) that minimizes
    the evaluation function.  The routine respects ``budget_s`` seconds of
    wall‑clock time and always returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5      # a small safety margin

    # problem size – the reference solution used n = 4096
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # 1️⃣  Build a diverse pool of warm‑starts
    pool: list[np.ndarray] = []

    # Warm start from the previous best (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # Several families of handcrafted shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # power‑law ends‑heavy: a_i ∝ (i·(n‑i+1))^α, α<0
    for alpha in np.linspace(-3.0, -0.2, 12):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Beta‑shapes, including very U‑shaped cases
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for p in (0.05, 0.15, 0.3, 0.6, 1.0, 2.0, 5.0):
        for q in (0.05, 0.15, 0.3, 0.6, 1.0, 2.0, 5.0):
            shape = (x ** p) * ((1.0 - x) ** q)
            if shape.sum() > 0:
                shape = shape / shape.sum()
                pool.append(shape)

    # Random Dirichlet draws with many concentration parameters
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            pool.append(a)

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of the starter pool – keep the best one
    for i, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{i:03d}] new best {val:.6f}")

    # Fallback to uniform if nothing succeeded (should never happen)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # --------------------------------------------------------------
    # 3️⃣  Adam optimisation on the smooth‑max surrogate
    # Transform the best candidate into logits for the soft‑max parametrisation
    eps = 1e-12
    logits = np.log(np.clip(best_seq, eps, None))

    # Adam state
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1 = 0.9
    beta2 = 0.999
    adam_eps = 1e-8

    step = 0
    # λ‑schedule: start modest, grow slowly to approximate the true max
    lambda_start = 5.0
    lambda_inc = (200.0 - lambda_start) / max(1, int(0.8 * (deadline - time.time())))

    while time.time() < deadline - 2.0:
        # ---------- soft‑max → probability vector ----------
        max_log = np.max(logits)
        exp_shift = np.exp(logits - max_log)
        a = exp_shift / exp_shift.sum()          # guaranteed sum = 1, a_i ≥ 0

        # ---------- exact objective value ----------
        conv = _full_convolution(a, conv_fft_len)
        max_conv = conv.max()
        val = 2.0 * n * max_conv
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Adam it={step:06d}] new best {val:.6f}")

        # ---------- smooth‑max weights ----------
        lam = lambda_start + step * lambda_inc
        shift = conv - conv.max()                # ≤ 0, safe for exponentiation
        w = np.exp(lam * shift)                  # log‑sum‑exp weights
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            # Very unlikely, fall back to exact active set
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0
            w_sum = 1.0
        p = w / w_sum

        # ---------- gradient w.r.t. a ----------
        grad_a = 2.0 * _cross_correlation(p, a, corr_fft_len)

        # ---------- gradient w.r.t. logits (soft‑max Jacobian) ----------
        grad_avg = np.dot(grad_a, a)
        grad_logits = a * (grad_a - grad_avg)

        # Optional gradient clipping (helps stability)
        max_abs = np.max(np.abs(grad_logits))
        if max_abs > 1e3:
            grad_logits = grad_logits * (1e3 / max_abs)

        # ---------- Adam update ----------
        step += 1
        m = beta1 * m + (1.0 - beta1) * grad_logits
        v = beta2 * v + (1.0 - beta2) * (grad_logits ** 2)
        m_hat = m / (1.0 - beta1 ** step)
        v_hat = v / (1.0 - beta2 ** step)

        # Per‑coordinate learning rate, typical Adam formula
        lr = 0.05 / (np.sqrt(v_hat) + adam_eps)
        logits = logits - lr * m_hat

    # --------------------------------------------------------------
    # 4️⃣  Very short simulated‑annealing polish
    if time.time() < deadline - 0.5 and best_seq is not None:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()

        T_start = 0.03
        T_end = 1e-5
        max_moves = 40000
        for mv in range(max_moves):
            if time.time() > deadline - 0.5:
                break
            # linear cooling
            T = T_start * (1.0 - mv / max_moves)
            i, j = np.random.randint(n, size=2)
            if i == j or a[i] < 1e-14:
                continue
            # move a random fraction (up to 30%) of mass from i to j
            delta = a[i] * (np.random.rand() * 0.3)
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val_sa:
                best_val_sa = val_new
                best_a_sa = a_new.copy()
                a = a_new
                print(f"[SA move {mv}] improved to {val_new:.6f}")
            else:
                if np.random.rand() < np.exp(-(val_new - val) / max(T, 1e-12)):
                    a = a_new
            val = val_new

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[SA final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final clean‑up and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Returning best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
