    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    # initial population
    population = []
    # Get warm if exists
    warm = globals().get('height_sequence_1')
    initial = _initial_candidates(n, None) + _spike_candidates(n)
    # add warm if present
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] != n:
            w = np.resize(w, n) # or pad/truncate
        w = np.maximum(w, 0)
        if w.sum() > 0:
            w = w / w.sum()
        population.append((w, compute_obj(w)))
    # add some from initial
    for a in initial[:100]: # limit
        a = np.maximum(a, 0)
        if a.sum() < 0.01:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = compute_obj(a)
        population.append((a, val))
    # sort and keep top pop_size
    pop_size = 30
    population = sorted(population, key=lambda x: x[1])[:pop_size]
    best_seq, best_val = population[0]

    generation = 0
    while time.time() < deadline:
        generation += 1
        # create child
        # select parents via tournament
        if np.random.rand() < 0.7:
            # crossover
            p1 = population[np.random.randint(pop_size)][0]
            p2 = population[np.random.randint(pop_size)][0]
            alpha = np.random.rand()
            child = alpha * p1 + (1 - alpha) * p2
        else:
            # mutation
            parent = population[np.random.randint(pop_size)][0]
            # random choice among mutation types
            r = np.random.rand()
            if r < 0.33:
                # Gaussian noise
                child = parent + np.random.normal(scale=0.02, size=n)
                child = np.maximum(child, 0)
            elif r < 0.66:
                # random phase mutation
                child = _random_phase_mutation(parent)
            else:
                # spike-like random redistribution
                child = np.copy(parent)
                # pick random two indices and shift mass

                    child[i] -= delta
                    child[j] += delta
            # ensure positivity; clip negative approx

        child_sum = child.sum()
        if child_sum <= 0:

            child = child / child_sum

def _random_phase_mutation(a):
    # a is a numpy array length n
    # compute rfft of zero-padded length conv_fft_len
    # return new a after random phase and projection
    # The length for rfft is conv_fft_len
    A = np.fft.rfft(a, n=conv_fft_len)

"""
Hybrid evolutionary optimizer for the max‑convolution objective.

The algorithm builds a modest population of non‑negative vectors (step
functions) and repeatedly mutates them:
  * Gaussian noise added to a parent,
  * convex combination (crossover) of two parents,
  * a random‑phase perturbation performed in the Fourier domain.
After each mutation the candidate is projected onto the probability simplex.
The best individuals are kept and periodically refined by a short
projected‑gradient descent on the exact max‑convolution.  When the time budget
is almost exhausted a longer Adam optimisation and a spectral‑flattening
pass are applied to the current best vector.  The routine always returns the
best sequence found before it is terminated (the time budget is ≈ 1000 s).
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, n=fft_len)
    conv = np.fft.irfft(fa * fa, n=fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, n=fft_len)
    fr = np.fft.rfft(rev_a, n=fft_len)
    corr = np.fft.irfft(fw * fr, n=fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective (2·n·max(conv) / (sum(a)²))."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return float("inf")
    return 2.0 * n * max_c / (s * s)


def _random_phase_mutation(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Keep |FFT|, randomise all non‑DC phases, then project onto simplex."""
    A = np.fft.rfft(a, n=fft_len)
    mag = np.abs(A)
    phi = np.random.rand(len(A)) * 2.0 * np.pi
    phi[0] = 0.0                     # DC stays real
    A_new = mag * np.exp(1j * phi)
    a_new = np.fft.irfft(A_new, n=fft_len)[: len(a)]
    a_new = np.maximum(a_new, 0.0)
    tot = a_new.sum()
    if tot < 1e-12:
        a_new = np.full_like(a_new, 1.0 / len(a_new))
    else:
        a_new /= tot
    return a_new


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """A diverse set of starter vectors."""
    cand = []
    if warm is not None:
        cand.append(warm.copy())
    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))
    # Symmetric exponential
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)
    # Power‑law shapes
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)
    # Double‑Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * (i / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)
    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)
    # Uniform
    cand.append(np.full(n, 1.0 / n))
    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse vectors composed of a few equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)
    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1
    return best_a, best_val


def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # initial temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Iteratively force the FFT magnitude (except DC) to be (almost) constant."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        A = np.fft.fft(a)
        energy = np.dot(a, a)

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s
        it += 1
    return best_a, best_val

    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
