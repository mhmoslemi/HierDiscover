"""
Hybrid optimizer for the max‑convolution objective.

The search proceeds in two stages:

1. **Broad parametric sweep** – a large and diverse pool of analytic
   profiles (beta‑distributions, window functions, ramps, power‑laws,
   Dirichlet draws, sparse spikes, etc.) is generated, each profile is
   normalised to the probability simplex and its exact objective
   (2·n·max(a∗a)) is evaluated via an FFT‑based convolution.  The best
   candidate from this cheap sweep seeds the refinement stage.

2. **Refinement** – the seeded vector is improved by a succession of
   optimisation passes that respect the time budget (≈ 1000 s):
   * Adam optimiser on a smooth‑max surrogate of the max‑convolution.
   * Exponentiated‑gradient sub‑gradient descent on the exact max‑conv.
   * “Spectral‑flattening’’ (alternating projection onto the set of
     vectors with (almost) constant FFT magnitude and onto the simplex).
   * Projected gradient descent on the exact max‑convolution.
   * Simulated‑annealing style local search with tiny mass transfers.

All iterates stay non‑negative and are re‑projected onto the simplex
after every update, guaranteeing a valid input for ``evaluate_sequence``.
The function always returns the best sequence encountered before the
deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, conv_fft_len: int) -> float:
    """Exact objective (2·n·max(conv)) assuming Σa = 1."""
    n = a.shape[0]
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()

# Optimisers
def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of max(conv)."""
    logits = np.log(a0 + 1e-15)        # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ----- obtain current a from logits -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- surrogate evaluation -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # surrogate smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of surrogate -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- keep track of best exact value -----
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- schedule -----
        lam = min(lam * lam_factor, 1e9)
        if it % 80_000 == 0 and it > 0:
            lr *= 0.92
        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 500_000,
                                 eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (allow a tiny tolerance so the set is not empty)
        tol = max(1e-12, 1e-8 * max_c)
        active = np.isclose(conv, max_c, atol=tol, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(a0: np.ndarray,
                               n: int,
                               conv_fft_len: int,
                               deadline: float,
                               max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Alternating projection between the simplex and the set of vectors whose
    FFT magnitudes (except the DC component) are (almost) constant.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # exact objective on current iterate
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50_000 == 0:
                print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ----- spectral flattening step -----
        A = np.fft.fft(a)                     # full complex FFT
        energy = np.dot(a, a)                 # ||a||_2^2

        # Preserve the DC term (sum = 1).  Redistribute remaining energy uniformly.
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        # Damping factor: gradually enforce the constant magnitude.
        damp = min(1.0, 0.5 + 0.5 * it / max_iters)   # from 0.5 → 1.0
        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            target = const_mag * np.exp(1j * np.angle(A[1:]))
            A_new[1:] = damp * target + (1 - damp) * A[1:]

        a_new = np.real(np.fft.ifft(A_new))

        # enforce non‑negativity and simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 250_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        # active set (use a small tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 40_000 == 0:
                print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 120_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a small fraction of mass from i to j
        frac = np.random.rand() * 0.2
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1
    return best_a, best_val

def _beta_distribution(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑distribution shaped probability vector (discrete)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    shape = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    shape = np.where(np.isfinite(shape), shape, 0.0)
    if shape.sum() == 0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    shape /= shape.sum()
    return shape


def _window_candidate(n: int, kind: str, invert: bool = False) -> np.ndarray:
    """Generate a standard window (hanning, hamming, blackman)."""
    if kind == "hanning":
        w = np.hanning(n)
    elif kind == "hamming":
        w = np.hamming(n)
    elif kind == "blackman":
        w = np.blackman(n)
    else:
        raise ValueError(f"Unsupported window: {kind}")
    if invert:
        w = np.max(w) - w  # simple inversion (keeps non‑negative)
    w = np.clip(w, 0.0, None)
    if w.sum() == 0:
        return np.full(n, 1.0 / n)
    w /= w.sum()
    return w


def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a wide pool of starter vectors."""
    cand = []

    # warm start (if any) and few rolled copies
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families (varying exponent)
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.5, -0.3, 7):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian (tuned widths)
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.08, 0.15, 0.25):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Inverted Hann / Hamming / Blackman windows
    for wnd in ("hanning", "hamming", "blackman"):
        cand.append(_window_candidate(n, wnd, invert=True))

    # Standard windows (non‑inverted) – they serve as diverse baselines
    for wnd in ("hanning", "hamming", "blackman"):
        cand.append(_window_candidate(n, wnd, invert=False))

    # Triangular (Bartlett) windows – both regular and inverted
    tri = np.bartlett(n)
    tri = np.clip(tri, 0.0, None)
    tri /= tri.sum()
    cand.append(tri)
    cand.append(_project_simplex(np.max(tri) - tri))   # inverted triangle

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Beta‑distribution grid
    for a in (0.2, 0.5, 1.0, 2.0, 5.0):
        for b in (0.2, 0.5, 1.0, 2.0, 5.0):
            cand.append(_beta_distribution(n, a, b))

    # Sparse spike patterns
    for k in (32, 64, 128, 256):
        a_spike = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a_spike[pos] = 1.0 / k
        cand.append(a_spike)

    # Random piece‑wise constant blocks
    for blocks in (8, 12, 16):
        block_len = n // blocks
        a_blk = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a_blk[b * block_len:(b + 1) * block_len] = mass
        if a_blk.sum() > 0:
            a_blk = a_blk / a_blk.sum()
        cand.append(a_blk)

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Very sparse candidates – handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget (≈ 1000 s) and always returns the
    best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # safety margin to avoid overshooting the judge's timeout
    deadline = time.time() + budget_s - 0.5

    # --------------------------------------------------------------
    # Problem size – 4096 is a good trade‑off between quality and speed.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (if a previous best is stored globally)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Broad parametric sweep (cheap evaluation)
    pool = _initial_candidates(n, warm) + _spike_candidates(n)
    # Shuffle to avoid systematic bias
    np.random.shuffle(pool)

    best_seq = None
    best_val = float("inf")
    eval_cnt = 0
    for cand in pool:
        if time.time() >= deadline:
            break
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s                       # simplex normalisation
        val = _objective(a, conv_fft_len)   # cheap exact objective
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")
        eval_cnt += 1

    print(f"Parametric sweep evaluated {eval_cnt} candidates, best {best_val:.6f}")

    # --------------------------------------------------------------
    # Deep optimisation – allocate the remaining budget
    # Phase I: Adam on smooth‑max surrogate
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase I] Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase I] Adam improved to {val_adam:.6f}")

    # Phase II: Exponentiated‑gradient sub‑gradient descent
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] Sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=600_000, eta0=0.6,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase II] Sub‑grad improved to {val_sub:.6f}")

    # Phase III: Spectral flattening (alternating projections)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] Spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_iters=350_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Phase III] Spectral improved to {val_spec:.6f}")

    # Phase IV: Projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase IV] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=300_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Phase IV] GD improved to {val_gd:.6f}")

    # Phase V: Simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase V] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=150_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase V] Anneal improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Final safety clipping (evaluate_sequence caps values to [0,1000])
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
