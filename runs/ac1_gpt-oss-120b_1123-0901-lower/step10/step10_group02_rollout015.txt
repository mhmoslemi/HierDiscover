"""
Hybrid random‑restart + gradient optimizer for the max‑convolution bound.

The procedure:
1. Gather a few diverse initial vectors (warm start, uniform, a family of
   symmetric beta‑shaped densities and a handful of Dirichlet samples).
   All vectors are projected onto the probability simplex (sum = 1).
2. Evaluate each candidate with the exact max‑convolution objective
   (`_objective`).  Keep the best one as a starting point.
3. Run a long exponentiated‑gradient sub‑gradient phase (the same as in the
   reference code) on the current best vector.  The phase runs until the
   global time budget is almost exhausted.
4. Fine‑tune the solution with a short Adam optimisation on the smooth‑max
   surrogate of the max‑convolution.  The temperature parameter `lam`
   is increased up to a very large value to approach the true max.
5. Perform a tiny stochastic local‑search (move a tiny amount of mass from
   one coordinate to another) for the last few seconds – this can often
   hop out of a shallow local minimum.
6. Try a few cyclic shifts of the final vector (the linear convolution changes
   under a shift) and keep the best shift.
7. Clip the result to the allowed range, renormalise, evaluate with the
   official `evaluate_sequence` (only for reporting), and return the list.

The whole routine respects the supplied time budget (≈ 1000 s) and always
returns a valid non‑negative sequence.
"""

import time
import numpy as np

# -------------------------------------------------------------
# Basic utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact evaluation 2·n·max(a*a) for a with sum(a)=1.
    (The final `evaluate_sequence` uses the same scaling.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisers (same building blocks as the reference implementation)
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02
    lam_max = 1e12           # push very close to the true max

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp trick)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, lam_max)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # a small safety margin so we never overshoot the deadline
    deadline = time.time() + budget_s - 0.3

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 0) collect a few diverse initial candidates
    candidates = []

    # warm start (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-6:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        candidates.append(a)

    # uniform distribution
    candidates.append(np.full(n, 1.0 / n, dtype=np.float64))

    # a family of symmetric beta‑shaped densities (U‑shaped for α<1)
    alphas = [0.55, 0.60, 0.65, 0.70, 0.75, 0.80, 0.85, 0.90, 0.95]
    x = (np.arange(n) + 0.5) / n
    for alpha in alphas:
        a = np.power(x, alpha - 1.0) * np.power(1.0 - x, alpha - 1.0)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s > 0:
            a = a / s
            candidates.append(a)

    # a handful of random Dirichlet draws with varied concentration
    for _ in range(20):
        conc = np.random.uniform(0.5, 3.0)
        a = np.random.dirichlet(np.full(n, conc))
        candidates.append(a)

    # --------------------------------------------------------------
    # 1) pick the best among the cheap candidates (exact eval)
    best_a = None
    best_val = np.inf
    for a in candidates:
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Init candidate] objective {val:.6f}")

    # --------------------------------------------------------------
    # 2) long sub‑gradient optimisation (the heavy‑weight part)
    # Reserve a few seconds for final tweaks and reporting.
    sub_deadline = deadline - 8.0
    if sub_deadline > time.time():
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_a, n, conv_fft_len, corr_fft_len, sub_deadline,
            max_iters=5_000_000, eta0=1.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Subgrad full] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 3) short Adam fine‑tuning (smooth‑max with very high temperature)
    adam_deadline = deadline - 4.0
    if adam_deadline > time.time():
        a_adam, val_adam = _adam_optimize(
            best_a, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Adam fine] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) tiny stochastic local search for the last few seconds
    local_deadline = deadline - 0.5
    while time.time() < local_deadline:
        # pick two distinct indices and move a small random chunk
        i, j = np.random.choice(n, 2, replace=False)
        delta = min(best_a[i], 1e-3) * np.random.rand()
        if delta <= 0:
            continue
        cand = best_a.copy()
        cand[i] -= delta
        cand[j] += delta
        # re‑project to the simplex (preserves non‑negativity)
        cand = _project_simplex(cand)
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Local move] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 5) try a few cyclic shifts (linear convolution changes with a shift)
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # 6) final sanity checks and return
    best_a = np.clip(best_a, 0.0, 1000.0)
    # ensure exact simplex normalisation (stability)
    best_a = _project_simplex(best_a)

    # report the official evaluation (mainly for logging)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
