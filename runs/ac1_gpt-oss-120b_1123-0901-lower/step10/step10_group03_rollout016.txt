"""
Propose a low‑score step‑function.

The routine follows three ideas that improve on the reference implementation:

1. **Very broad parametric sweep** – many families (exponential, power, stretched,
   Gamma, Beta, double‑Gaussian, mixtures of exponentials, sparse Dirichlet,
   random spikes, …) are sampled on a fine grid.  The best shape so far is kept.
2. **Multi‑restart exact‑max sub‑gradient descent** – starting from the best
   sweep candidate **and** from a number of random initialisations (Dirichlet,
   sparse vectors) a short exponentiated‑gradient run is performed.  The best
   result of all restarts is retained.
3. **Long final sub‑gradient pass & cheap annealing** – a long exact‑max
   (mirror‑descent) optimisation is launched from the best vector, followed by
   a simulated‑annealing style fine‑tuner.

All steps respect the global time‑budget (default 1000 s) and the routine
always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_mixture_exponential(num: int, n: int) -> np.ndarray:
    """Mixture of `num` left‑exponential components with random rates."""
    alphas = 10 ** np.random.uniform(-4, 2, size=num)
    weights = np.random.rand(num)
    weights /= weights.sum()
    mix = np.zeros(n, dtype=np.float64)
    for w, a in zip(weights, alphas):
        mix += w * _candidate_left_exponential(a, n)
    mix /= mix.sum()
    return mix


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimisation using exponentiated updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set for the max entry (or set of ties)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise safely
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing style local search.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

    for shape in np.linspace(0.2, 4.0, 12):
        for scale in np.logspace(-4, 0.5, 12):
            _maybe_update(_candidate_gamma(shape, scale, n))

    # Beta distribution (left‑skewed)
    for a_par in np.logspace(-2, 0.5, 12):
        for b_par in np.logspace(0, 1.0, 12):
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # Double Gaussian
    for sig in np.logspace(-3, -0.2, 15):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # Random spikes
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Mixtures of exponentials – more extensive than the reference
    for _ in range(1000):
        mix = _candidate_mixture_exponential(num=2, n=n)
        _maybe_update(mix)

    # Mixtures of three exponentials (still cheap)
    for _ in range(300):
        mix = _candidate_mixture_exponential(num=3, n=n)
        _maybe_update(mix)

    # Random Dirichlet vectors with several concentrations
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # Attempt to include any previously saved good solution
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑restart exact‑max sub‑gradient descent --------------------
    # Allocate roughly half of the remaining budget to many short restarts.
    remaining = deadline - time.time()
    restart_budget = min(remaining * 0.45, 300.0)   # at most 5 min for restarts
    restart_deadline = time.time() + restart_budget
    restart_counter = 0

    while time.time() < restart_deadline:
        # choose a diverse random start
        if np.random.rand() < 0.6:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            # sparse random support (size between 32 and 128)
            k = np.random.randint(32, 129)
            a0 = np.zeros(n, dtype=np.float64)
            idx = np.random.choice(n, size=k, replace=False)
            a0[idx] = np.random.rand(k)
            a0 /= a0.sum()
        # short sub‑gradient run (≈15 s)
        short_deadline = min(deadline - 5.0, time.time() + 15.0)
        a_tmp, val_tmp = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, short_deadline,
            max_iters=2_000_000, eta0=0.9
        )
        restart_counter += 1
        if val_tmp < best_val:
            best_val = val_tmp
            best_seq = a_tmp.copy()
            print(f"[Restart #{restart_counter}] improved to {val_tmp:.6f}")

    print(f"[Restarts] performed {restart_counter} short runs, best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient optimisation -----------------------
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑Subgrad] long optimisation")
        a_final, val_final = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.2,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Phase‑Subgrad] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional annealing fine‑tuning ---------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Clean‑up / final safety checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
