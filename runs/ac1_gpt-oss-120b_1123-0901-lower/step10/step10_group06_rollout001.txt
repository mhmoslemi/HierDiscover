"""
Optimisation of a non‑negative step‑function for the evaluation metric

    score = 2·n·max(conv(a,a)) / (sum a)²

The algorithm proceeds in three phases:

1. **Diverse parametric sweep** – a large collection of symmetric,
   monotone and mixed shapes (power‑law, exponential, stretched‑exponential,
   beta, double‑Gaussian, sinusoidal, random spikes, Dirichlet draws, and
   random convex combinations of the previous families) is generated.
   Each candidate is normalised to unit sum and its exact objective is
   evaluated.  The best `ELITE_SIZE` vectors are retained.

2. **Momentum exponentiated‑gradient sub‑gradient descent** – starting from
   the best elite candidate we perform a long mirror‑descent on the simplex
   using the exact sub‑gradient of the max‑convolution (active set weights
   uniform on the maximising shifts).  A momentum term (β≈0.9) accelerates
   convergence and the step‑size follows ηₜ = η₀/√(t+1).

3. **Optional simulated‑annealing fine‑tuning** – if time permits a short
   annealing pass moves small fractions of mass between random positions,
   accepting downhill moves always and uphill moves with a temperature schedule.

All phases respect the overall `budget_s` (default 1000 s).  The routine
always returns a valid non‑negative list; the final score is printed via
`evaluate_sequence`.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same conv length)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, sum is assumed ≈ 1)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p with p < 0 (heavier at the edges)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_sine(n: int) -> np.ndarray:
    """Half‑sine shape – zero at the boundaries, smooth interior bump."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 1) / (n + 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_cosine_half(n: int) -> np.ndarray:
    """1‑minus‑cosine half wave – small at the left, larger at the right."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.cos(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

# Elite handling
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite list (kept sorted by val, smallest first)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Momentum exponentiated‑gradient sub‑gradient optimiser
# ----------------------------------------------------------------------
def _subgradient_momentum_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.6,
    beta: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Mirror‑descent on the simplex with momentum, using the exact max‑conv gradient."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    m = np.zeros_like(a)  # momentum buffer

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[SubGrad-Mom it={it:07d}] new best {val:.6f}")

        # --- active‑set weights -------------------------------------------------
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # --- sub‑gradient (cross‑correlation) ----------------------------------
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- momentum update ----------------------------------------------------
        m = beta * m + (1.0 - beta) * grad

        # --- step‑size (decaying) -----------------------------------------------
        eta = eta0 / np.sqrt(it + 1.0)

        # --- exponentiated‑gradient (mirror descent) -----------------------------
        a = a * np.exp(-eta * m)

        # --- renormalise ---------------------------------------------------------
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated annealing (lightweight) – optional fine‑tuning
# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing: move a random fraction of mass between two sites."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # Problem dimension – a power of two gives fast FFTs.
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – populate a diverse elite set
    # ------------------------------------------------------------------
    elite = []          # List of (objective, vector)
    ELITE_SIZE = 12

    def _maybe_update(a_vec):
        val = _objective(a_vec, n)
        if len(elite) and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a_vec, max_size=ELITE_SIZE)

    # ---- systematic families -------------------------------------------------
    # symmetric power‑law (negative exponents concentrate mass at the ends)
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta distributions (wide range)
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-3, -0.1, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 20):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # left‑power (allow negative/positive exponents)
    for p in np.linspace(-4.0, 2.0, 28):
        _maybe_update(_candidate_left_power(p, n))

    # inverse‑power (edges heavy)
    for p in np.linspace(-4.5, -0.1, 25):
        _maybe_update(_candidate_inverse_power(p, n))

    # sinusoidal and cosine‑half shapes (new)
    _maybe_update(_candidate_sine(n))
    _maybe_update(_candidate_cosine_half(n))

    # random spike patterns (different sparsities)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Dirichlet draws with a range of concentration parameters
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]
    for c in concs:
        _maybe_update(np.random.dirichlet(np.full(n, c)))

    # ------------------------------------------------------------------
    # 1b) Random convex combinations of already generated bases
    # ------------------------------------------------------------------
    base_vectors = [vec for _, vec in elite]  # start from the elite pool
    rng = np.random.default_rng(seed + 1)
    for _ in range(80):
        idx = rng.choice(len(base_vectors), size=2, replace=False)
        a1, a2 = base_vectors[idx[0]], base_vectors[idx[1]]
        alpha = rng.random()
        mix = alpha * a1 + (1.0 - alpha) * a2
        mix = np.maximum(mix, 0.0)
        s = mix.sum()
        if s < 1e-12:
            continue
        mix /= s
        _maybe_update(mix)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # ------------------------------------------------------------------
    # 2) Long momentum exponentiated‑gradient sub‑gradient optimisation
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 15.0:
        sub_deadline = time.time() + remaining * 0.9  # keep a tiny buffer for final steps
        print("[Phase‑SubGrad] momentum exponentiated‑gradient optimisation")
        a_sub, val_sub = _subgradient_momentum_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            eta0=0.6,
            beta=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑SubGrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 3) Optional simulated‑annealing fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety checks and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    if best_seq.sum() < 0.01:          # safeguard – re‑normalise if needed
        best_seq = np.full_like(best_seq, 1.0 / best_seq.size)

    # final public evaluation (if the function is available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
