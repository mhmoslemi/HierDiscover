Algorithm Overview
------------------

import time

import numpy as np
from scipy import special  # for the Kaiser window (i0)

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale invariant, sum can be arbitrary)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p with p < 0 (edges heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_hann(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def _candidate_hamming(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_blackman(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (
        0.42
        - 0.5 * np.cos(2 * np.pi * i / (n - 1))
        + 0.08 * np.cos(4 * np.pi * i / (n - 1))
    )
    a /= a.sum()
    return a


def _candidate_kaiser(beta: float, n: int) -> np.ndarray:
    """Kaiser window with shape parameter beta (beta>=0)."""
    i = np.arange(n, dtype=np.float64)
    alpha = (n - 1) / 2.0
    t = (i - alpha) / alpha
    a = special.i0(beta * np.sqrt(1 - t * t))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sqrt(n: int) -> np.ndarray:
    """a_i ∝ sqrt(i+1)  (monotone increasing)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sqrt(i + 1.0)
    a /= a.sum()
    return a


def _candidate_inv_sqrt(n: int) -> np.ndarray:
    """a_i ∝ 1/sqrt(i+1)  (monotone decreasing)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 1.0)
    a /= a.sum()
    return a

# Optimisers
# ----------------------------------------------------------------------
def _adam_max_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser **on the exact max‑convolution**.
    The mirror‑descent is performed on logits (soft‑max), so iterates stay
    on the simplex.  The gradient is the sub‑gradient of max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise in log‑softmax space
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.09                     # a bit larger than in the original code

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # compute convolution and exact max
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam‑max it={it:06d}] new best {val:.6f}")

        # sub‑gradient of max(conv): uniform on active indices
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        cnt = active.sum()
        if cnt > 0:
            w[active] = 1.0 / cnt

        # gradient wrt a (exact)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent gradient (log‑space)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        logits -= step

        # occasional learning‑rate decay
        if it % 120_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max exponentiated‑gradient descent (mirror descent on the simplex)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        cnt = active.sum()
        if cnt > 0:
            w[active] = 1.0 / cnt

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing refinement – random mass transfers."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.30
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Elite handling

def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite list (kept sorted by val, lowest first)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best sequence
    (as a plain Python list) that it discovered.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # a small safety margin

    # ------------------------------------------------------------------
    # Problem dimension – we stay at a power of two for fast FFTs
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)   # >= convolution length + n

    # ------------------------------------------------------------------
    # 1) Wide parametric sweep – collect an elite set
    # ------------------------------------------------------------------
    elite = []   # list of tuples (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # --- Existing families (already used in previous run) -----------------
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))

    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_par in a_pars:
        for b_par in a_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3, -0.1, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    for alpha in np.logspace(-5, 1.2, 20):
        _maybe_update(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-4.0, 2.0, 28):
        _maybe_update(_candidate_left_power(p, n))

    for p in np.linspace(-4.5, -0.1, 25):
        _maybe_update(_candidate_inverse_power(p, n))

    # --- New families (cosine‑based windows, Kaiser, sine, sqrt) ---------
    _maybe_update(_candidate_hann(n))
    _maybe_update(_candidate_hamming(n))
    _maybe_update(_candidate_blackman(n))

    for beta in [0.0, 2.0, 5.0, 8.0, 12.0, 16.0]:
        _maybe_update(_candidate_kaiser(beta, n))

    _maybe_update(_candidate_sine(n))
    _maybe_update(_candidate_sqrt(n))
    _maybe_update(_candidate_inv_sqrt(n))

    # --- Random sparse spikes ------------------------------------------------
    for k in [32, 64, 128, 256, 512]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True))

    # --- Dirichlet draws with varied concentration ----------------------------
    for c in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        a = np.random.dirichlet(np.full(n, c))
        _maybe_update(a)

    # --- Convex combinations of elite members (introduce diversity) ---------
    if elite:
        elites_arrays = [vec for _, vec in elite]
        for _ in range(12):
            a1, a2 = np.random.choice(elites_arrays, size=2, replace=False)
            lam = np.random.rand()
            a = lam * a1 + (1.0 - lam) * a2
            a /= a.sum()
            _maybe_update(a)

    # --- Warm start from a previously stored global (if present) -------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found – using uniform distribution")

    # ------------------------------------------------------------------
    # 2) Adam on exact max‑convolution – short runs from the top elites
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(25.0, 0.18 * remaining)
        print("[Phase‑Adam] exact‑max Adam optimisation (multiple starts)")
        for idx, (_, a_start) in enumerate(elite[:4]):   # up to 4 starts
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_max_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=350_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long exponentiated‑gradient sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 6.0:
        print("[Phase‑Subgrad] long exponentiated‑gradient optimisation")
        sub_deadline = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Simulated‑annealing fine‑tuning (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine‑tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, respects limits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respect evaluator caps

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
