import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two >= x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa*fa, fft_len)
    return conv[: 2*a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start+a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2*n*max(convolution) (assumes sum(a)=1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Adam on smooth‑max of the self‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.05                 # start temperature
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress print
            if it % 10000 == 0:
                print(f"[Adam iter={it}] best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e6)

        if it % 80000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val

def _max_conv_descent(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent directly on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14)
        w = np.zeros_like(conv)
        if active.any():

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[GD iter={it}] new best {val:.6f}")

        it += 1

    return best_a, best_val

def _spectral_random_search(a0: np.ndarray,
                           n: int,

    """Exploit the constant‑magnitude Fourier manifold via random phases."""

        it += 1

    pool.append(np.full(n, 1.0 / n))                         # uniform

    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    for k in (32, 64, 128, 256):
        a = np.zeros(n)
        pos = np.random.choice(n, k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        pool.append(a)

    best_a: np.ndarray | None = None
    best_val = float("inf")
    max_initial = 8                     # a few candidates for the Adam stage

    # ------------------------------------------------------------------
    # Phase I – Adam optimisation from each start
    for idx, a0 in enumerate(pool[:max_initial]):
        if time.time() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        sub_dead = min(deadline, time.time() + 28.0)    # ~28 s per candidate
        a_opt, val_opt = _adam_optimize(a0, n, conv_fft_len,
                                        corr_fft_len, sub_dead)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print(f"[Init {idx}] Adam improved best to {val_opt:.6f}")

    if best_a is None:
        best_a = np.full(n, 1.0 / n)
        best_val = _objective(best_a, n, conv_fft_len)

    # ------------------------------------------------------------------
    # Phase II – random‑phase spectral exploration
    if time.time() < deadline - 5.0:
        print("[Phase] Spectral random‑phase exploration")
        a_spec, val_spec = _spectral_random_search(best_a, n,
                                                   conv_fft_len,
                                                   deadline - 5.0)
        if val_spec < best_val:
            best_val = val_spec
            best_a = a_spec.copy()
            print(f"[Phase] Spectral random improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase III – final projected gradient descent fine‑tuning
    if time.time() < deadline - 2.0:
        print("[Phase] Final projected gradient descent")
        a_gd, val_gd = _max_conv_descent(best_a, n, conv_fft_len,
                                        corr_fft_len, deadline - 2.0)
        if val_gd < best_val:
            best_val = val_gd
            best_a = a_gd.copy()
            print(f"[Phase] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # safety fallback
    if best_a is None:
        best_a = np.full(n, 1.0 / n)

    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best objective {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
