"""
Propose a low‑value step‑function for the “max‑self‑convolution” challenge.

The search works in three stages:

1. **Massive parametric sweep** – a dense grid (and some random draws)
   over a few flexible families (power‑law, exponential, power·exp,
   beta‑distribution, sparse spikes).  All candidates are normalised to
   sum = 1 and evaluated with an FFT‑based convolution.  The 15 best
   candidates are retained.

2. **Soft‑max Adam refinement** – each retained candidate is refined
   with Adam on a smooth‑max surrogate of the true objective.  The
   temperature of the smooth‑max (`lam`) is increased slowly so that
   the optimisation first explores a smooth landscape and then focusses
   on the true maximum.  No L₂ regulariser is used (the pure smooth‑max
   works better for the final polishing).

3. **Exact‑max exponentiated‑gradient sub‑gradient** – the best result
   from the Adam phase is taken as start point for a long mirror‑descent
   (exponentiated‑gradient) run on the *real* max‑norm objective.
   The step‑size decays as 1/√t.  When time runs out the best sequence is
   returned (always a non‑negative list).

All stages respect the 1000 s budget and the routine never returns an
empty or invalid list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a using FFT (length 2·len(a)-1).
    Returns the central part of length 2·len(a)-1.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _candidate_power_exp(p: float, alpha: float, n: int) -> np.ndarray:
    """Product of power‑law and exponential, mirrored."""
    i = np.arange(n, dtype=np.float64)
    a = ((i + 1) ** p) * np.exp(-alpha * i) + ((n - i) ** p) * np.exp(-alpha * (n - 1 - i))
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on the discrete grid."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes placed on a length‑n grid (Dirichlet masses)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max -----------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lam_start: float = 0.05,
    lam_factor: float = 1.008,
    l2_weight: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth_max(conv)  +  λ·‖conv – uniform‖₂² .
    The L₂ term is optional (set λ=0 for pure smooth‑max).
    Returns the best vector encountered (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    # uniform target for the optional L₂ term (unused if λ=0)
    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution (for tracking)
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- optional L₂ regulariser -----
        if l2_weight > 0.0:
            resid = conv - uniform_target
            grad_l2 = 2.0 * _cross_correlation(resid, a, fft_len)
            grad = grad_smooth + l2_weight * grad_l2
        else:
            grad = grad_smooth

        # mirror‑descent (Adam) on logits
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – sharpen smooth‑max
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient sub‑gradient -------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative (exponentiated‑gradient) update
        a = a * np.exp(-eta * grad)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated annealing (optional fine‑tuning) --------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very light annealing that transfers a fraction of mass between two cells."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.15
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.30   # up‑to‑30 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # small safety margin

    # problem size – a modest power of two works well for FFT speed
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep the 15 best candidates
    top_n = 15
    top_candidates = []  # list of (value, array)

    def _maybe_add(cand: np.ndarray, val: float):
        """Insert candidate into the sorted top‑list if it is good enough."""
        nonlocal top_candidates
        if len(top_candidates) < top_n:
            top_candidates.append((val, cand.copy()))
            top_candidates.sort(key=lambda x: x[0])
        elif val < top_candidates[-1][0]:
            top_candidates[-1] = (val, cand.copy())
            top_candidates.sort(key=lambda x: x[0])

    # Helper that safely evaluates and updates the incumbent list
    def _try_candidate(a: np.ndarray):
        val = _objective(a, n)
        if val < best_val[0]:
            best_val[0] = val
            best_seq[0] = a.copy()
            print(f"[Sweep] new best {val:.6f}")
        _maybe_add(a, val)

    # keep global best for quick logging
    best_val = [float("inf")]
    best_seq = [None]

    # ----- power‑law sweep (very fine) -----
    for p in np.linspace(-10.0, -0.05, 300):
        if time.time() > deadline:
            break
        _try_candidate(_candidate_power(p, n))

    # ----- exponential sweep (fine) -----
    for alpha in np.logspace(-6, 2, 300):
        if time.time() > deadline:
            break
        _try_candidate(_candidate_exponential(alpha, n))

    # ----- power·exp random draws (moderate) -----
    for _ in range(200):
        if time.time() > deadline:
            break
        p = np.random.uniform(-6.0, 2.0)
        alpha = 10 ** np.random.uniform(-6, 1)
        _try_candidate(_candidate_power_exp(p, alpha, n))

    # ----- beta‑distribution draws (focus on U‑shapes) -----
    for _ in range(300):
        if time.time() > deadline:
            break
        a_par = 10 ** np.random.uniform(-2.0, 1.0)   # 0.01 … 10
        b_par = 10 ** np.random.uniform(-2.0, 1.0)
        _try_candidate(_candidate_beta(a_par, b_par, n))

    # ----- random Dirichlet draws (various concentrations) -----
    for conc in [0.1, 0.2, 0.5, 1.0, 2.0]:
        for _ in range(40):
            if time.time() > deadline:
                break
            a = np.random.dirichlet(np.full(n, conc))
            a = np.clip(a, 0.0, None)
            a /= a.sum()
            _try_candidate(a)

    # ----- sparse spike patterns -----
    for k in [32, 64, 128, 256, 512]:
        for _ in range(20):
            if time.time() > deadline:
                break
            _try_candidate(_candidate_spike(k, n, random=True))

    # ----- warm‑start from previously discovered sequence (if any) -----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w)

    print(f"[Sweep] best after exploration: {best_val[0]:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement on the (up to) top candidates
    # allocate roughly 30 % of the remaining budget to Adam
    time_left = deadline - time.time()
    if time_left > 30 and top_candidates:
        adam_budget = min(time_left * 0.30, 120.0)   # at most 2 min for Adam phase
        per_candidate_deadline = start_time + adam_budget
        print("[Phase‑Adam] start refinement")
        for idx, (val0, cand) in enumerate(top_candidates):
            if time.time() >= per_candidate_deadline:
                break
            # each candidate gets a slice of the Adam budget (≈ 1 min total)
            sub_deadline = min(per_candidate_deadline, time.time() + 25.0)
            a_adam, val_adam = _adam_optimize(
                cand,
                n,
                fft_len,
                sub_deadline,
                max_iters=400_000,
                lam_start=0.05,
                lam_factor=1.008,
                l2_weight=0.0,          # pure smooth‑max
            )
            if val_adam < best_val[0]:
                best_val[0] = val_adam
                best_seq[0] = a_adam.copy()
                print(f"[Phase‑Adam] candidate {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max exponentiated‑gradient sub‑gradient optimisation
    time_left = deadline - time.time()
    if time_left > 20 and best_seq[0] is not None:
        print("[Phase‑Subgrad] start exact max optimisation")
        sub_deadline = deadline - 2.0   # leave a tiny margin for final sanity check
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq[0],
            n,
            fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=1.2,
        )
        if val_sub < best_val[0]:
            best_val[0] = val_sub
            best_seq[0] = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Light annealing fine‑tune (if time permits)
    if time.time() < deadline - 1 and best_seq[0] is not None:
        print("[Phase‑Anneal] final fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq[0],
            n,
            fft_len,
            deadline,
            max_moves=150_000,
        )
        if val_ann < best_val[0]:
            best_val[0] = val_ann
            best_seq[0] = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid, non‑negative return
    if best_seq[0] is None:
        best_seq[0] = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq[0] = np.clip(best_seq[0], 0.0, 1000.0)

    # final sanity check using the public evaluator (if it exists)
    try:
        final_score = evaluate_sequence(best_seq[0].tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence not available).")

    return best_seq[0].tolist()
