def refine_step(a, M, t=0.2):
    # compute b using greedy allocation respecting a * b <= M
    n = len(a)
    cur = np.zeros(2*n-1)  # current convex sum of constraints
    b = np.zeros_like(a)
    # order indices by ascending a
    idx = np.argsort(a)  # small a first
    for i in idx:
        if a[i] == 0:
            b[i] = 0
            continue
        # compute max possible for b_i
        window = slice(i, i+n)  # indices i...i+n-1 in cur
        available = M - cur[window]
        max_possible = np.min(available) / a[i]
        if max_possible <= 0:
            continue

"""
Hybrid optimizer with a new “greedy LP‑refinement’’ stage.

The routine:
1. Loads a warm start (global `height_sequence_1`) or falls back to a uniform
   distribution of length n = 4096 and normalises it to a probability simplex.
2. Runs a short Adam optimisation on a smooth‑max surrogate (≈ 10 % of the
   time budget) to get a decent starting point.
3. Executes several epochs of a greedy “linear‑program’’ refinement:
   –  For the current sequence a compute M = max(a*a).
   –  Construct a vector b that maximises ∑b_i under the linear constraints
      a * b ≤ M (every component of the convolution a★b must stay below M).
      The constraints have a band‑structure; a simple O(n²) greedy algorithm
      (adding mass to the right‑most position of each sliding window) solves
      this problem optimally.
   –  Mix a towards the normalised b with a small stepsize t and re‑project
      onto the simplex.
   This step reduces the peak of the convolution without changing the
   scaling of a, and is repeated many times (the bulk of the budget).
4. Finishes with a short projected gradient descent on the exact max‑convolution
   (≈ 5 % of the budget) and a few cyclic shifts.
5. Returns the best sequence found.  The whole procedure respects the ~1000 s
   budget and always returns a valid non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lr0: float = 0.05,
    lam0: float = 0.1,
    lam_factor: float = 1.02,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Parameters `lr0`, `lam0`, `lam_factor` allow some flexibility in the schedule.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)            # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr0
    lam = lam0

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a in the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _greedy_lp_refine(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    t: float = 0.1,
) -> tuple[np.ndarray, float]:
    """
    Greedy linear‑program refinement.
    Constructs a vector b that maximises Σb under the constraints a★b ≤ M,
    where M = max(a*a).  Returns a new a ← (1‑t)a + t·b (re‑projected onto the simplex)
    together with its objective value.
    """
    conv = _full_convolution(a, conv_fft_len)
    M = conv.max()

    # current convolution contributions of the (still‑zero) b
    cur = np.zeros(2 * n - 1, dtype=np.float64)
    b = np.zeros_like(a)

    # Process indices in order of increasing a_i (small a gets more room)
    idx = np.argsort(a)

    for i in idx:
        ai = a[i]
        if ai <= 0.0:
            continue
        start = i
        end = i + n
        # maximum extra amount that can be added to b_i without violating any bound
        avail = M - cur[start:end]
        min_avail = avail.min()
        if min_avail <= 0.0:
            continue
        max_possible = min_avail / ai
        if max_possible <= 0.0:
            continue
        b[i] = max_possible
        cur[start:end] += ai * max_possible

    # Normalise b to a probability simplex for a smooth mixture.
    sb = b.sum()
    if sb > 0.0:
        b = b / sb
    else:
        b = a.copy()

    # Convex combination and projection
    a_new = (1.0 - t) * a + t * b
    a_new = np.maximum(a_new, 0.0)
    a_new = a_new / a_new.sum()

    val = _objective(a_new, n, conv_fft_len)
    return a_new, val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Uses a short Adam run, followed by many greedy‑LP refinement epochs,
    a brief gradient‑descent mop‑up and a few cyclic shifts.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # safety margin

    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.clip(a, 0.0, 1000.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()                     # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – a brief Adam pass (≈10 % of the remaining budget)
    t_rem = deadline - time.time()
    if t_rem > 5.0:
        stage_dead = time.time() + min(0.10 * t_rem, 30.0)
        a_adam, val_adam = _adam_optimize(
            best_a,
            n,
            conv_fft_len,
            corr_fft_len,
            stage_dead,
            lr0=0.07,          # a slightly larger learning rate than the default
            lam0=0.05,
            lam_factor=1.02,
            max_iters=200_000,
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Stage 1] Adam improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # Greedy LP refinement loop – the core of the improvement.
    # We keep running refinements with a decreasing step size `t` until time runs out.
    t_values = [0.2, 0.15, 0.1, 0.07, 0.05, 0.03, 0.02, 0.01]
    t_idx = 0
    iter_count = 0
    while time.time() < deadline:
        if t_idx >= len(t_values):
            t_idx = len(t_values) - 1
        t_step = t_values[t_idx]

        a_new, val_new = _greedy_lp_refine(best_a, n, conv_fft_len, t=t_step)

        if val_new < best_val:
            best_val, best_a = val_new, a_new.copy()
            print(f"[Greedy t={t_step:.3f} it={iter_count}] new best {val_new:.6f}")

        # slowly decrease the step size after a few successful improvements
        if iter_count % 15 == 0 and t_idx < len(t_values) - 1:
            t_idx += 1

        iter_count += 1
        # safety break if we have performed a huge number of cheap iterations
        if iter_count > 5000:
            break

    # --------------------------------------------------------------
    # Short projected gradient descent mop‑up (≈5 % of remaining budget)
    t_rem = deadline - time.time()
    if t_rem > 10.0:
        stage_dead = time.time() + min(0.05 * t_rem, 30.0)
        a_gd, val_gd = _max_conv_descent(
            best_a, n, conv_fft_len, corr_fft_len, stage_dead, max_iters=150_000
        )
        if val_gd < best_val:
            best_val, best_a = val_gd, a_gd.copy()
            print(f"[Stage GD] improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Final cheap cyclic‑shift local search
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Clip to allowed range and final sanity check
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # should never happen
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # --------------------------------------------------------------
    # Final official evaluation (only for reporting)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
