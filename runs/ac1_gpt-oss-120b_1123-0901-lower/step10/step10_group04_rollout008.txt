"""
Propose a low‑score step‑function.

The algorithm works in three stages:

1️⃣  A *dense parametric sweep* over simple families that are known to
    give good U‑shaped distributions (symmetric power‑laws and Beta
    densities).  The sweep uses a very fine grid for the power‑law
    exponent and a large set of random Beta parameters.  The best
    candidate from this stage is used as a warm‑start.

2️⃣  A *long Adam optimiser* on a smooth‑max surrogate of the true
    objective.  The surrogate is `log‑sum‑exp(λ·conv)` with a
    temperature λ that is increased exponentially until it is huge
    (≈10⁸), thereby turning the smooth‑max into the true max.  The
    optimisation works in the log‑softmax domain, guaranteeing the
    simplex constraints.  Several short restarts from noisy perturbations
    of the best candidate are performed.

3️⃣  A *mirror‑descent sub‑gradient* (exponentiated‑gradient) on the
    exact max‑norm, followed by a tiny simulated‑annealing fine‑tune.
    The best sequence found at any time is returned.

The whole routine respects the supplied time budget (default 1000 s)
and always yields a non‑negative list of coefficients.  The final
score is printed using the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact eval 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------

# pre‑computed logarithms for power‑law generation (used many times)
def _precompute_power_logs(n: int):
    i = np.arange(n, dtype=np.float64)
    log_i_plus = np.log(i + 1.0)           # log(i+1)
    log_n_i = np.log(n - i)               # log(n-i)
    return log_i_plus, log_n_i


def _candidate_power(p: float, n: int, log_i_plus, log_n_i) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    a = np.exp(p * log_i_plus) + np.exp(p * log_n_i)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_mixed_power(p: float, q: float, lam: float,
                          n: int, log_i_plus, log_n_i) -> np.ndarray:
    """Linear mix of two symmetric power‑law shapes."""
    a1 = np.exp(p * log_i_plus) + np.exp(p * log_n_i)
    a2 = np.exp(q * log_i_plus) + np.exp(q * log_n_i)
    a = lam * a1 + (1.0 - lam) * a2
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    s = a.sum()
    a /= s
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   fft_len: int,
                   deadline: float,
                   max_iters: int = 800_000,
                   lam_start: float = 0.08,
                   lam_factor: float = 1.015,
                   l2_weight: float = 0.10) -> tuple[np.ndarray, float]:
    """
    Adam on a smooth‑max surrogate + small L2 regularisation.
    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max params
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    # uniform target for the optional L2 term
    uniform_target = np.full(2 * n - 1,
                             1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution – needed only for tracking & L2 term
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:06d}]   new best {val:.6f}")

        # ----- smooth‑max weights (log‑sum‑exp) -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- optional L₂ regularisation towards a flat convolution -----
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through the soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – smooth‑max becomes sharper
        lam = min(lam * lam_factor, 1e7)

        # occasional LR decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 fft_len: int,
                                 deadline: float,
                                 max_iters: int = 6_000_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector found and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}]  new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # renormalise (guard against numerical drift)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                    n: int,
                    fft_len: int,
                    deadline: float,
                    max_moves: int = 250_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tune."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.30   # up to 30 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ----------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns a valid list of coefficients.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem dimension – 4096 is a good trade‑off between quality & speed
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1) Fine‑grained parametric sweep ---------------------------------
    best_seq = None
    best_val = float('inf')

    # pre‑compute logs for the power‑law family (used many times)
    log_i_plus, log_n_i = _precompute_power_logs(n)

    def _try_candidate(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {tag}] new best {val:.6f}")

    # ----- symmetric power‑law -------------------------------------------------
    # coarse grid (500 points) first
    coarse_ps = np.linspace(-5.0, -0.05, 500)
    for p in coarse_ps:
        _try_candidate(_candidate_power(p, n, log_i_plus, log_n_i), f"pow({p:.3f})")
        if time.time() >= deadline:
            break

    # refine around the best exponent
    if best_seq is not None:
        # extract the exponent that gave the best value (approx.)
        # we keep the last p examined as a proxy; for a proper rebound we
        # re‑evaluate the coarse sweep results
        # (the coarse loop already kept the best; we just reuse that p)
        # Perform a finer scan in a narrow window ±0.1
        fine_ps = np.linspace(best_p - 0.1, best_p + 0.1, 400) \
            if 'best_p' in locals() else np.linspace(-1.0, -0.2, 400)
        for p in fine_ps:
            _try_candidate(_candidate_power(p, n, log_i_plus, log_n_i), f"pow_fine({p:.4f})")
            if time.time() >= deadline:
                break

    # ----- random mixtures of two power‑laws ---------------------------------
    for _ in range(300):
        p = np.random.uniform(-5.0, -0.05)
        q = np.random.uniform(-5.0, -0.05)
        lam = np.random.rand()
        _try_candidate(_candidate_mixed_power(p, q, lam, n,
                                            log_i_plus, log_n_i),
                       f"mix(p={p:.2f},q={q:.2f},l={lam:.2f})")
        if time.time() >= deadline:
            break

    # ----- Beta distributions (random parameters) ----------------------------
    for _ in range(800):
        a_par = 10 ** np.random.uniform(-2.0, 0.8)   # roughly 0.01 … 6.3
        b_par = 10 ** np.random.uniform(-2.0, 0.8)
        _try_candidate(_candidate_beta(a_par, b_par, n), f"beta({a_par:.3f},{b_par:.3f})")
        if time.time() >= deadline:
            break

    # ----- left‑exponential (monotone) ---------------------------------------
    for alpha in np.logspace(-6, 1.0, 30):
        _try_candidate(_candidate_left_exponential(alpha, n), f"lexp({alpha:.2e})")
        if time.time() >= deadline:
            break

    # incorporate a warm start if the notebook already provided one
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w, "warm")

    if best_seq is None:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidate improved – using uniform")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max + tiny L2) -------------------------
    if time.time() < deadline - 30:
        print("[Phase‑Adam] start")
        # give Adam at most 35 % of remaining time
        adam_deadline = min(deadline - 5,
                            time.time() + 0.35 * (deadline - time.time()))
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.08,
            lam_factor=1.015,
            l2_weight=0.10,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

        # a few short restarts from noisy neighbours
        for r in range(3):
            if time.time() >= deadline - 10:
                break
            noise = np.random.randn(n) * 0.02
            pert = np.exp(noise) * best_seq
            pert /= pert.sum()
            sub_deadline = min(deadline - 3,
                               time.time() + 0.10 * (deadline - time.time()))
            a_tmp, val_tmp = _adam_optimize(
                pert,
                n,
                fft_len,
                sub_deadline,
                max_iters=200_000,
                lam_start=0.06,
                lam_factor=1.020,
                l2_weight=0.12,
            )
            if val_tmp < best_val:
                best_val = val_tmp
                best_seq = a_tmp.copy()
                print(f"[Adam restart {r}] improved to {val_tmp:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient (exponentiated‑gradient) -----------------
    if time.time() < deadline - 5:
        print("[Phase‑Subgrad] start")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tune -------------------------
    if time.time() < deadline - 1:
        print("[Phase‑Anneal] final fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity & return ------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # evaluate with the public function (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:   # pragma: no cover
        final_score = None
        print("Could not evaluate final score (evaluate_sequence missing).")

    return best_seq.tolist()
