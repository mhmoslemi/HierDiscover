"""
Hybrid search for the minimal value of
    2 * n * max( conv(a,a) ) / ( sum(a) ** 2 )
where `a` is a non‑negative step function (list of floats).
The search is performed on the probability simplex (sum = 1) because the
objective is scale‑invariant – the final sequence is simply re‑scaled to
satisfy the evaluator’s clipping rules.

Main ideas
---------
1. Build a diverse pool of start vectors (uniform, Dirichlet draws,
   exponential, Gaussian, Beta‑shaped, and a warm‑start if available).
2. Quickly evaluate each start (FFT based convolution) and keep the
   `k_best` candidates.
3. For each of the `k_best` candidates run a long **exponentiated‑gradient
   sub‑gradient** phase (the cheapest but very effective optimiser).
4. Refine the current global best with a few rounds of **projected gradient
   descent** and a brief **simulated‑annealing** local search.
5. If time permits, up‑sample to a finer grid (8192 points) and run a short
   Adam optimisation to squeeze out the last few digits.

All heavy work is done with FFT‑based convolutions; the algorithm adapts
its phases to the remaining time budget (≈ 1000 s) and never returns an
empty sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for (sub‑)gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta distribution on a regular grid, clipped to non‑negative."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.power(x, alpha - 1.0) * np.power(1.0 - x, beta - 1.0)
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = _make_uniform(n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Objective helpers

def _objective_val(a: np.ndarray, conv_fft_len: int) -> float:
    """Return 2·n·max(conv(a,a)) assuming Σa = 1."""
    n = a.shape[0]
    max_c = _full_convolution(a, conv_fft_len).max()
    return 2.0 * n * max_c


# ----------------------------------------------------------------------
# Optimisers

def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 0.8,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best iterate (still on the simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # active constraints (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)          # EG update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution objective.
    Keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Very light simulated‑annealing on the simplex.
    Randomly shifts a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of floats) with a low
    evaluation score under the provided `evaluate_sequence` function.
    The routine uses a multi‑phase hybrid optimiser and respects the
    1000 s time budget.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem size – we keep the original 4096 points (good trade‑off)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (previous best, if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a large, diverse pool of starting vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Uniform
    pool.append(_make_uniform(n))

    # A bunch of Dirichlet draws with a wide range of concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))
    for _ in range(10):
        a = 10 ** np.random.uniform(-3, 1)                # 1e‑3 … 10
        pool.append(_make_dirichlet(n, a))

    # Exponential tails
    for gamma in (0.05, 0.1, 0.3, 0.6, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))

    # Double‑Gaussian shapes
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.6):
        pool.append(_make_gaussian(n, sigma))

    # Beta‑shaped families (symmetric and skewed)
    for a, b in [(0.5, 0.5), (1.0, 1.0), (2.0, 2.0), (0.3, 3.0),
                 (3.0, 0.3), (0.8, 2.5), (2.5, 0.8)]:
        pool.append(_make_beta(n, a, b))

    # Simple piece‑wise linear triangle
    i = np.arange(1, n + 1, dtype=np.float64)
    tri = i * (n - i + 1)
    tri /= tri.sum()
    pool.append(tri)

    # ------------------------------------------------------------------
    # Quick evaluation of the pool – keep the best `k_keep`
    k_keep = 7
    best_vectors = []          # list of (value, vector)
    for vec in pool:
        if time.time() >= deadline:
            break
        vec = np.maximum(vec, 0.0)
        if vec.sum() < 0.01:
            continue
        vec = vec / vec.sum()
        val = _objective_val(vec, conv_fft_len)
        best_vectors.append((val, vec))
    # sort and keep the top candidates
    best_vectors.sort(key=lambda x: x[0])
    best_vectors = best_vectors[:k_keep]

    # ------------------------------------------------------------------
    # Global best trackers
    best_seq = None
    best_val = float("inf")
    for val, vec in best_vectors:
        if val < best_val:
            best_val, best_seq = val, vec.copy()
    print(f"[Init] best coarse value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase I – long exponentiated‑gradient runs on the retained candidates
    for idx, (_, a0) in enumerate(best_vectors):
        if time.time() >= deadline:
            break
        sub_dead = min(deadline, time.time() + 45.0)   # allocate ~45 s per start
        a_opt, val_opt = _expgrad_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            eta0=0.9, max_iters=3_000_000
        )
        if val_opt < best_val:
            best_val, best_seq = val_opt, a_opt.copy()
            print(f"[ExpGrad #{idx}] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – fine‑tune the current champion with projected gradient descent
    if best_seq is not None and time.time() < deadline - 8:
        sub_dead = min(deadline, time.time() + 30.0)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000,
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase III – a short simulated annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        sub_dead = min(deadline, time.time() + 12.0)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead, max_moves=100_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – final long exponentiated‑gradient run (takes the remaining budget)
    if best_seq is not None and time.time() < deadline - 2:
        print("[Final ExpGrad] running to the end")
        a_final, val_final = _expgrad_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            eta0=0.6, max_iters=10_000_000,
        )
        if val_final < best_val:
            best_val, best_seq = val_final, a_final.copy()
            print(f"[Final ExpGrad] new best {val_final:.6f}")

    # ------------------------------------------------------------------
    # Optional up‑sampling (if any decent time remains)
    if best_seq is not None and time.time() + 20 < deadline:
        try:
            print("[Upsample] 8192‑point refinement")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)
            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()
            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big_opt, val_big = _expgrad_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, eta0=0.7, max_iters=3_000_000,
            )
            if val_big < best_val:
                best_val, best_seq = val_big, a_big_opt.copy()
                print(f"[Upsample] improved to {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] skipped – {e}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
