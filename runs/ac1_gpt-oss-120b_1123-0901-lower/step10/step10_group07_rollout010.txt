```python
def _lp_refine(a, n, conv_fft_len, deadline, max_iter=30, step=0.2):
    a = a.copy()
    while time.time() < deadline and max_iter>0:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # find active indices with conv >= M - eps
        eps = 1e-12
        active = np.where(conv >= M - eps)[0]
        # Build constraint matrix rows: for each s in active, row_j = a[s - j] if valid else 0.
        m = active.shape[0]
        # Build A matrix with shape (m, n) dense? Use np.zeros
        A = np.zeros((m, n))
        for row_idx, s in enumerate(active):
            # indices j where i = s - j is in [0, n-1]; i ranges 0 to n-1 => j = s - i.
            # So for j from max(0, s-(n-1)) to min(n-1, s)
            j_start = max(0, s - (n - 1))
            j_end = min(n - 1, s)
            # i = s - j
            # corresponding a_i = a[s - j]
            # Fill row
            A[row_idx, j_start:j_end+1] = a[s - np.arange(j_start, j_end+1)]
        # Solve LP: maximize sum g => minimize -sum g.
        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # fall back: break
            break
        g = res.x
        # normalize g to sum 1 if sum not zero
        s = g.sum()
        if s < 1e-12:
            # fallback random restart
            g = np.random.dirichlet(np.ones(n))
            s = g.sum()
        g = g / s
        # convex combination
        a = (1 - step) * a + step * g
        max_iter -= 1
    return a
```

```python
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    eps_coeff = 1e-12
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # active indices where conv is within tolerance
        # tolerance relative to M
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]
        # Build constraint matrix A (len(active) x n)
        m = active.shape[0]
        if m == 0:
            break
        # Use modified a to avoid zero coefficients
        a_mod = np.maximum(a, eps_coeff)
        A = np.zeros((m, n))
        for row_idx, s in enumerate(active):
            # compute j range
            j_start = max(0, s - (n - 1))
            j_end = min(n - 1, s)
            # fill row
            # indices for j: [j_start, ..., j_end]
            # corresponding a_i: a_mod[s - j]
            idxs = np.arange(j_start, j_end + 1)
            A[row_idx, j_start:j_end + 1] = a_mod[s - idxs]
        # Solve LP: maximize sum(g) -> minimize -sum(g)
        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n
        # Solve using HiGHS method
        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # In case of unbounded or failure, break
            break
        g = res.x
        s = g.sum()
        if s < 1e-12:
            # fallback: random distribution
            g = np.random.dirichlet(np.ones(n))
            s = g.sum()
        g = g / s  # normalize
        # convex combination update
        a = (1 - step) * a + step * g
        # evaluate objective
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP] iter={it} improved to {val:.6f}")
        # optional: gradually reduce step size
        step = max(step * 0.99, 0.01)
        it += 1
    return best_a, best_val
```

"""
Step‑function optimizer for the evaluation function
    2·n·max(a*a) / (sum(a))**2

The search proceeds in three stages:

1.  A very dense parametric sweep over many one‑parameter families
    (exponential, power‑law, stretched‑exponential, beta,
    double‑Gaussian, cosine, harmonic, etc.).  The best candidate
    from this cheap exploration is kept.

2.  A small “convex‑LP” refinement that implements the classic
    iteration of the original paper:
        – compute the current maximal convolution value M,
        – build the active set of shifts where the maximum is attained,
        – solve a linear programme that maximises the total mass of a new
          step‑function g under the constraints (a * g)[s] ≤ M for all
          active shifts,
        – move the current solution a a little towards g.
    The LP has at most a few constraints (the active set is usually
    size 1–10) and is solved with SciPy’s HiGHS interior‑point method.
    Several iterations (until the time budget runs out) quickly lower M.

3.  A final long exponentiated‑gradient sub‑gradient run (mirror‑descent
    on the simplex) that fine‑tunes the solution after the LP phase.

All steps respect the supplied time budget (default 1000 s) and always
return a feasible non‑negative list that can be fed to
``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a (needed for the gradient of the max‑convolution).
    The result has length ``len(a)`` and corresponds to the derivative of
    ``max(w·(a*a))`` w.r.t. ``a`` when ``w`` is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)² (no clipping).
    ``a`` is assumed to be non‑negative.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^p (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape: 1/(1+exp(beta·(i‑mu)))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_cosine(n: int) -> np.ndarray:
    """Cosine‑shaped decreasing window (non‑negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (np.cos(np.pi * i / (n - 1)) + 1.0) / 2.0
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_harmonic(p: float, n: int) -> np.ndarray:
    """Harmonic‑law: a_i ∝ 1/(i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / ((i + 1) ** p)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient (exponentiated‑gradient)."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _lp_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    step: float = 0.3,
    max_iters: int = 200,
) -> tuple[np.ndarray, float]:
    """
    Convex‑LP refinement based on the classic iteration described in the
    problem statement.  Each iteration solves a tiny linear programme
    (active constraints only) and moves the current solution a little
    toward the optimiser g.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n)
    eps_coeff = 1e-12
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        # active set – indices where the maximum is (almost) attained
        active = np.where(conv >= M - 1e-12)[0]
        m = active.shape[0]
        if m == 0:
            break

        # avoid zero coefficients – they would make the LP unbounded
        a_mod = np.maximum(a, eps_coeff)

        # build constraint matrix A (m × n)
        A = np.zeros((m, n))
        for row_idx, s in enumerate(active):
            j_start = max(0, s - (n - 1))
            j_end = min(n - 1, s)
            idxs = np.arange(j_start, j_end + 1)
            A[row_idx, j_start:j_end + 1] = a_mod[s - idxs]

        # maximise sum(g)  ≡  minimise -∑ g
        c = -np.ones(n)
        b_ub = np.full(m, M)
        bounds = [(0, None)] * n

        res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
        if not res.success:
            # If the LP is unbounded or fails, fall back to a random move
            g = np.random.dirichlet(np.ones(n))
        else:
            g = res.x

        s = g.sum()
        if s < 1e-12:
            g = np.random.dirichlet(np.ones(n))
            s = g.sum()
        g = g / s                       # re‑normalise to unit sum

        # convex combination update
        a = (1 - step) * a + step * g

        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP refine it={it:03d}] improved to {val:.6f}")

        # slowly decay the step size
        step = max(step * 0.97, 0.01)
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and returns the best sequence
    found.  It combines a dense parametric sweep, a convex‑LP refinement,
    a final sub‑gradient polish and an optional simulated‑annealing pass.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5       # a small safety margin

    # --------------------------------------------------------------
    # problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # --------------------------------------------------------------
    # 1) Very dense parametric sweep

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.01, 120):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 140):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential (symmetric)
    alphas = np.logspace(-4, 0.6, 40)
    betas = np.linspace(0.2, 3.0, 30)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta distributions (skew allowed)
    a_pars = [0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian (symmetric)
    for sig in np.logspace(-3.0, -0.2, 30):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 120):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # monotone left‑power (decreasing)
    for p in np.linspace(-4.0, -0.01, 120):
        _maybe_update(_candidate_left_power(p, n))

    # monotone left‑logistic (sigmoid)
    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 80):
        _maybe_update(_candidate_left_logistic(beta, mu, n))

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.0, 80):
        _maybe_update(_candidate_left_linear(slope, n))

    # cosine‑shaped decreasing window
    _maybe_update(_candidate_cosine(n))

    # harmonic‑law family
    for p in np.linspace(0.1, 4.0, 80):
        _maybe_update(_candidate_harmonic(p, n))

    # random spike patterns (many repetitions)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            _maybe_update(_candidate_spike(k, n, random=True))

    # a few pure random Dirichlet samples
    for _ in range(20):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand)

    # warm start from a possible previous global variable
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Convex‑LP refinement (core of the classic algorithm)
    if time.time() < deadline - 30:
        print("[Phase‑LP] starting convex‑LP refinement")
        lp_deadline = min(deadline - 5, time.time() + 25.0)
        a_lp, val_lp = _lp_refine(best_seq, n, conv_fft_len, lp_deadline,
                                  step=0.3, max_iters=200)
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[Phase‑LP] improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # 3) Adam polishing (short, just in case)
    if time.time() < deadline - 30:
        print("[Phase‑Adam] short Adam polishing")
        adam_deadline = min(deadline - 5, time.time() + 12.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 4) Stochastic Dirichlet restarts + sub‑gradient refinement
    remaining = deadline - time.time()
    max_restarts = 5
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        concentration = float(np.random.choice([0.1, 0.3, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 20.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline,
            max_iters=1_500_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # --------------------------------------------------------------
    # 5) Final long sub‑gradient run on the current best
    if time.time() < deadline - 3:
        print("[Phase‑Subgrad] final long sub‑gradient run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 6) Very short simulated‑annealing polishing (if any time left)
    if time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # final sanity checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
