def _solve_lp_for_g(f: np.ndarray, M: float, rows, cols) -> np.ndarray:
    n = f.shape[0]
    data = f[cols]  # using broadcasting, f[cols] shape same as rows length

    if not res.success:
        # fallback to uniform
        g = np.full(n, 1.0/n)
    else:
        g = res.x
    return g

"""
Step‑function optimizer for the evaluation function

    score = 2·n·max(a*a) / (sum(a))² .

The algorithm consists of three phases:

1. **Dense parametric sweep** – a variety of analytic families
   (exponential, power‑law, stretched‑exponential, beta, Gaussian,
   sinusoid, raised‑cosine, linear, random spikes) is sampled on a fine
   grid.  The best candidate is kept.

2. **Linear‑programming refinement** – given the current best shape ``f``,
   solve the LP

        max   1ᵀ g
        s.t.  f ★ g ≤ M          ( ★ is the linear convolution )
               g ≥ 0 ,

   where ``M = max(f★f)``.  The solution ``g`` is mixed with ``f``
   (convex combination) and the process is repeated a few times.
   This procedure is known to strictly decrease the worst‑case
   autocorrelation and helps escape the basin of the simple gradient
   methods.

3. **Long exponentiated‑gradient sub‑gradient run** – a final
   mirror‑descent optimisation on the simplex (the same update used in
   the reference implementation) is performed for the remaining time
   budget.

All steps respect the elapsed wall‑clock time (default 1000 s) and
always return a non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the sub‑gradient.
    The result has length ``len(a)`` and corresponds to d/d a of
    ``max(w·(a*a))`` when ``w`` is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (a ≥ 0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_raised_cosine(width: float, n: int) -> np.ndarray:
    """A raised cosine bell, width ∈ (0,1] controls support."""
    i = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.where(np.abs(i - 0.5) <= width / 2,
                 0.5 * (1 + np.cos(2 * np.pi * (i - 0.5) / width)),
                 0.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate with a soft‑max parametrisation."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient (mirror‑descent) on the simplex."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Linear‑programming refinement -----------------------------------------

def _build_conv_matrix_indices(n: int):
    """Pre‑compute row/col indices for the convolution matrix (size (2n‑1)×n)."""
    rows = np.concatenate([np.arange(i, i + n, dtype=np.int32) for i in range(n)])
    cols = np.concatenate([np.full(n, i, dtype=np.int32) for i in range(n)])
    return rows, cols


def _solve_lp_for_g(f: np.ndarray, M: float, rows, cols):
    """
    Solve  max 1ᵀg  s.t.  f★g ≤ M, g ≥ 0.
    Returns the optimal g (numpy array of shape (n,)).
    """
    from scipy import sparse
    from scipy.optimize import linprog

    n = f.shape[0]
    data = f[cols]                   # each entry equals the coefficient f_i
    A = sparse.csr_matrix((data, (rows, cols)), shape=(2 * n - 1, n))

    c = -np.ones(n)                  # minimise –1ᵀg  <=> maximise 1ᵀg
    b = np.full(2 * n - 1, M, dtype=np.float64)

    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to a simple feasible point (uniform)
        g = np.full(n, 1.0 / n)
    else:
        g = res.x
    return g


def _lp_refine(a: np.ndarray, n: int, rows, cols,
               deadline: float, max_iters: int = 6) -> np.ndarray:
    """
    Perform a few LP‑based improvement steps.
    Each step solves the LP for g and mixes it into the current a.
    """
    conv_fft_len = _next_pow2(2 * n - 1)
    eps = 1e-12
    best_a = a.copy()
    best_val = _objective(best_a, n)

    mixing = 0.5
    for it in range(max_iters):
        if time.time() >= deadline - 0.5:
            break
        # current bound M
        val = _objective(best_a, n)
        M = val / (2.0 * n)               # because val = 2·n·M / 1²  (sum=1)

        g = _solve_lp_for_g(best_a, M, rows, cols)
        # enforce strict positivity to keep the LP bounded in later steps
        g = np.maximum(g, eps)
        g /= g.sum()

        # convex combination
        new_a = (1.0 - mixing) * best_a + mixing * g
        new_a = np.maximum(new_a, eps)
        new_a /= new_a.sum()

        new_val = _objective(new_a, n)
        if new_val < best_val:
            best_val = new_val
            best_a = new_a
            print(f"[LP it={it}] improved to {new_val:.6f}")

        # slowly reduce mixing to stabilise
        mixing *= 0.8
        if mixing < 0.05:
            mixing = 0.05
    return best_a


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the given
    evaluation metric.  The routine respects the supplied time budget
    (seconds) and always returns a feasible sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # problem size --------------------------------------------------------
    n = 2048                                    # moderately large, still cheap for LP
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # pre‑compute convolution matrix indices for the LP phase (if possible)
    try:
        rows, cols = _build_conv_matrix_indices(n)
        SCIPY_AVAILABLE = True
    except Exception:
        SCIPY_AVAILABLE = False
        rows = cols = None

    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # -------------------- 1) dense parametric sweep --------------------
    # symmetric families
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n))

    for alpha in np.logspace(-5, 1.0, 120):
        _maybe_update(_candidate_exponential(alpha, n))

    alphas = np.logspace(-4, 0.6, 30)
    betas = np.linspace(0.2, 3.0, 25)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    a_pars = [0.1, 0.4, 0.8, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3.0, -0.2, 25):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone families (left‑hand side)
    for alpha in np.logspace(-5, 1.2, 100):
        _maybe_update(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_left_power(p, n))

    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 70):
        _maybe_update(_candidate_left_logistic(beta, mu, n))

    for slope in np.linspace(0.1, 3.0, 70):
        _maybe_update(_candidate_left_linear(slope, n))

    # new sinusoidal / raised‑cosine families
    for pow_ in [1, 2, 4]:
        _maybe_update(_candidate_sine(pow_, n))

    for width in np.linspace(0.2, 1.0, 9):
        _maybe_update(_candidate_raised_cosine(width, n))

    # random spikes – try a few different sparsities
    for k in [16, 32, 64, 128, 256]:
        for _ in range(5):
            _maybe_update(_candidate_spike(k, n, random=True))

    # a few pure Dirichlet draws
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand)

    # incorporate any previously‑found global (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # -------------------- 2) LP refinement (if possible) --------------------
    if SCIPY_AVAILABLE and best_seq is not None and time.time() < deadline - 30:
        print("[LP] starting linear‑programming refinement")
        lp_deadline = deadline - 20.0
        best_seq = _lp_refine(best_seq, n, rows, cols, lp_deadline, max_iters=8)
        best_val = _objective(best_seq, n)
        print(f"[LP] best after refinement: {best_val:.6f}")

    # -------------------- 3) Adam polishing (optional, short) ------------
    if time.time() < deadline - 20:
        print("[Phase‑Adam] polishing with Adam")
        adam_deadline = min(deadline - 5, time.time() + 12.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=350_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # -------------------- 4) Long sub‑gradient run -----------------------
    if time.time() < deadline - 5:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=12_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ----------------------------------------------------------------------
    # final sanity checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
