"""
Propose a non‑negative step‑function (list of heights) with a low evaluation score.

The algorithm combines a wide parametric sweep, several Adam runs on a
smooth‑max surrogate, long exponentiated‑gradient (sub‑gradient) descent
with a multi‑active‑set gradient, and a final short annealing phase.
It works for any power‑of‑two length (default 8192) and always returns a
valid sequence before the time budget expires.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for the gradient of
    sum_i a_i a_{k-i}.  Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (same as evaluate_sequence, but without the safety checks)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families (all return a non‑negative vector with sum == 1)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = (x ** (a_par - 1)) * ((1 - x) ** (b_par - 1))
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Heavy at the edges, p<0 (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_logarithmic(n: int) -> np.ndarray:
    """a_i ∝ 1/ln(i+2) + 1/ln(n-i+1)  (edges heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.log(i + 2.0) + 1.0 / np.log(n - i + 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_arcsin(n: int) -> np.ndarray:
    """a_i ∝ 1/√(i+0.5) + 1/√(n-i-0.5)  (edges heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt(i + 0.5) + 1.0 / np.sqrt(n - i - 0.5)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_piecewise_linear(mid: float, slope: float, n: int) -> np.ndarray:
    """Linear decay from 1 at i=0 to 0 at i=mid, then constant 0."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / mid)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """k random spikes (Dirichlet masses)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = rng.choice(n, size=k, replace=False)
        masses = rng.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_dirichlet(conc: float, n: int, rng: np.random.Generator) -> np.ndarray:
    """Full Dirichlet draw with concentration parameter conc."""
    a = rng.dirichlet(np.full(n, conc))
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Helper to keep a small set of elite candidates
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted ascending by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

# Optimisation primitives
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.04            # start temperature (smaller → smoother)
    lam_factor = 1.04

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max -> keep a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – become sharper over time
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay occasionally
        if it % 120_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_multi(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
    rel_tol: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent using a multi‑active set.
    All shifts whose value is within `rel_tol` of the current maximum
    receive equal weight.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set: all shifts within rel_tol of the max
        threshold = max_c * (1.0 - rel_tol)
        active = conv >= threshold
        if active.any():
            w = active.astype(float)
            w /= w.sum()
        else:
            w = np.zeros_like(conv)
            w[np.argmax(conv)] = 1.0

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numerical loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimension – power of two for fast FFT
    # --------------------------------------------------------------
    n = 8192                     # larger than before (more freedom)
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Broad parametric sweep – keep a few elites
    # --------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(a_vec):
        val = _objective(a_vec, n)
        _update_elite(elite, val, a_vec, max_size=8)

    # Symmetric power‑law (negative exponent concentrates mass at the ends)
    for p in np.linspace(-6.0, -0.05, 35):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # Symmetric exponential
    for alpha in np.logspace(-6, 1.2, 35):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # Stretched exponential (α,β)
    alphas = np.logspace(-5, 0.6, 14)
    betas = np.linspace(0.25, 3.5, 14)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # Beta distributions
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 22):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # Monotone left‑exponential
    for alpha in np.logspace(-6, 1.4, 22):
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # Monotone left‑power (allow negative exponents)
    for p in np.linspace(-5.0, 2.5, 30):
        a = _candidate_left_power(p, n)
        _maybe_update(a)

    # Inverse‑power (edges heavy)
    for p in np.linspace(-5.0, -0.1, 30):
        a = _candidate_inverse_power(p, n)
        _maybe_update(a)

    # Logarithmic edges‑heavy
    a = _candidate_logarithmic(n)
    _maybe_update(a)

    # Arcsine‑type edges‑heavy
    a = _candidate_arcsin(n)
    _maybe_update(a)

    # Random spikes of various sparsities
    for k in [32, 64, 128, 256, 512, 1024]:
        for _ in range(3):
            a = _candidate_spike(k, n, rng)
            _maybe_update(a)

    # Dirichlet draws with several concentration parameters
    for conc in [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]:
        a = _candidate_dirichlet(conc, n, rng)
        _maybe_update(a)

    # Warm‑start from a previously stored global (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates – using uniform")

    # --------------------------------------------------------------
    # 2) Adam optimisation (smooth‑max) on the top elite candidates
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 40.0:
        adam_deadline = time.time() + min(30.0, 0.18 * remaining)
        print("[Phase‑Adam] soft‑max Adam optimisation")
        for idx, (_, a_start) in enumerate(elite[:4]):  # up to four starts
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=600_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Long exponentiated‑gradient sub‑gradient descent (multi‑active)
    # --------------------------------------------------------------
    if time.time() < deadline - 8.0:
        print("[Phase‑Subgrad] long multi‑active sub‑gradient optimisation")
        sub_deadline = deadline - 6.0
        # We'll perform a few restarts; each restart runs until sub_deadline is hit.
        while time.time() < sub_deadline - 1.0:
            # 70 % chance to continue from current best, 30 % chance a fresh Dirichlet start
            if np.random.rand() < 0.7:
                a_start = best_seq
            else:
                a_start = _candidate_dirichlet(0.5, n, rng)

            a_sub, val_sub = _subgradient_optimize_multi(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=10_000_000,
                eta0=1.0,
                rel_tol=0.015,   # a tighter active set than before
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Phase‑Subgrad] restart improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    if time.time() < deadline - 2.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=220_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check (if the public evaluator is present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score – evaluator not available.")

    return best_seq.tolist()
