"""Improved search for a low‑evaluation step function.

The algorithm consists of four stages:
1. Wide parametric sweep (exponential, power‑law, beta, Gaussian, spikes)
   on a reduced size (n≈1024) to obtain a diverse elite set.
2. Smooth‑max Adam optimisation from the best elites (fast surrogate).
3. Exact‑max exponentiated‑gradient sub‑gradient descent (long run).
4. A few LP‑boost iterations (as described in the paper) which
   solve  “maximise total mass  subject to  (f * g) ≤ max(f * f)”.
   This drives the convolution towards a flat shape.
Finally the best small‑size vector is up‑sampled to the target length
(4096) and returned.

All heavy work respects the time budget (default 1000 s).  The code
uses NumPy, SciPy (linprog) and FFT for fast convolutions.
"""

import time
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import coo_matrix

# FFT utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)² (sum a is assumed >0)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce sum‑to‑1 vectors)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Heavy at the edges (p < 0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes with Dirichlet masses."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam on smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.05            # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 50_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient descent (exponentiated‑gradient updates)
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, within tolerance)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – guard against numerical loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# LP‑boost step (maximise total mass under convolution bound)
# ----------------------------------------------------------------------
def _lp_boost_step(f: np.ndarray, M: float, row_idx, col_idx) -> np.ndarray:
    """
    Solve   max Σ b_j   s.t.   conv(f,b) ≤ M ,  b ≥ 0.
    Return the optimal vector b (may be all zeros if LP fails).
    """
    n = f.shape[0]
    data = np.repeat(f, n)
    A = coo_matrix((data, (row_idx, col_idx)), shape=(2 * n - 1, n)).tocsr()
    b_ub = np.full(2 * n - 1, M, dtype=np.float64)
    c = -np.ones(n, dtype=np.float64)          # maximise sum → minimise –sum
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=(0, None), method="highs")
    if res.success:
        return res.x
    else:
        # LP failure – fall back to a zero vector (no improvement)
        return np.zeros_like(f)


# ----------------------------------------------------------------------
# Helper to keep a small set of elite candidates
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite (kept sorted by val, smallest first)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    a valid non‑negative list.  The strategy is:
      * coarse sweep on a reduced size (n≈1024),
      * Adam optimisation on the smooth‑max surrogate,
      * long exact‑max sub‑gradient descent,
      * a few LP‑boost iterations,
      * up‑sample to the target length (4096) and final sanity check.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Working dimension – smaller than the final 4096 for speed
    # ------------------------------------------------------------------
    n_small = 1024                     # power of two → fast FFT
    conv_fft_len = _next_pow2(2 * n_small - 1)
    corr_fft_len = _next_pow2(3 * n_small - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep a small elite pool
    # ------------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n_small)
        # early abort if clearly worse than current worst elite entry
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Symmetric power‑law (negative exponents concentrate mass at ends)
    for p in np.linspace(-5.0, -0.05, 30):
        a = _candidate_power(p, n_small)
        _maybe_update(a)

    # Symmetric exponential (search a finer grid)
    for alpha in np.logspace(-4, 1.0, 40):
        a = _candidate_exponential(alpha, n_small)
        _maybe_update(a)

    # Stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n_small)
            _maybe_update(a)

    # Beta distributions – wide range of shape parameters
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n_small)
            _maybe_update(a)

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 20):
        a = _candidate_double_gaussian(sig, n_small)
        _maybe_update(a)

    # Monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 20):
        a = _candidate_left_exponential(alpha, n_small)
        _maybe_update(a)

    # Monotone left‑power (allow negative exponents)
    for p in np.linspace(-4.0, 2.0, 28):
        a = _candidate_left_power(p, n_small)
        _maybe_update(a)

    # Inverse‑power (edges heavy)
    for p in np.linspace(-4.5, -0.1, 25):
        a = _candidate_inverse_power(p, n_small)
        _maybe_update(a)

    # Random spike patterns (different sparsities)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            a = _candidate_spike(k, n_small, random=True)
            _maybe_update(a)

    # Dirichlet draws with additional extreme concentrations
    for c in [0.01, 0.02, 0.05, 0.1, 0.2]:
        a = np.random.dirichlet(np.full(n_small, c))
        _maybe_update(a)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration (n={n_small}): {best_val:.6f}")
    else:
        best_seq = np.full(n_small, 1.0 / n_small, dtype=np.float64)
        best_val = _objective(best_seq, n_small)
        print("[Sweep] no candidates found – using uniform")

    # ------------------------------------------------------------------
    # 2) Adam (smooth‑max) – short runs from the top elites
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 20.0:
        adam_deadline = time.time() + min(30.0, 0.2 * remaining)
        print("[Phase‑Adam] short Adam optimisation (multiple starts)")
        for idx, (_, a_start) in enumerate(elite[:4]):  # up to 4 starts
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_optimize(
                a_start,
                n_small,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=300_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long exponentiated‑gradient sub‑gradient descent (exact max)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n_small,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=15_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) LP‑boost iterations (few but powerful)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑LP] LP‑boost refinement")
        # Pre‑compute the sparse index pattern for the constraint matrix
        n = n_small
        row_idx = np.empty(n * n, dtype=np.int32)
        col_idx = np.empty(n * n, dtype=np.int32)
        ptr = 0
        for d in range(n):
            rows_d = np.arange(d, d + n, dtype=np.int32)
            row_idx[ptr : ptr + n] = rows_d
            col_idx[ptr : ptr + n] = np.arange(n, dtype=np.int32)
            ptr += n
        # Simple schedule for the mixing parameter t
        t = 0.25
        for it_lp in range(8):
            if time.time() >= deadline:
                break
            M = _full_convolution(best_seq, conv_fft_len).max()
            b = _lp_boost_step(best_seq, M, row_idx, col_idx)
            b_sum = b.sum()
            if b_sum < 1e-12:
                print("[Phase‑LP] LP produced negligible mass – stop")
                break
            g = b / b_sum                     # renormalise to unit sum
            # gentle move towards g
            best_seq = (1.0 - t) * best_seq + t * g
            # recompute objective for reporting
            val = _objective(best_seq, n_small)
            if val < best_val:
                best_val = val
                print(f"[Phase‑LP] iteration {it_lp} new best {val:.6f}")
            # slowly decay mixing weight
            t *= 0.85

    # ------------------------------------------------------------------
    # 5) Upsample to the target length (4096) and final sanity check
    # ------------------------------------------------------------------
    target_n = 4096
    x_small = np.linspace(0.0, 1.0, n_small)
    x_target = np.linspace(0.0, 1.0, target_n)
    best_seq_big = np.interp(x_target, x_small, best_seq)
    best_seq_big = np.maximum(best_seq_big, 0.0)
    best_seq_big /= best_seq_big.sum()

    # Clip to the limits imposed by evaluate_sequence (just a safety)
    best_seq_big = np.clip(best_seq_big, 0.0, 1000.0)

    # Final evaluation (may raise if evaluate_sequence is missing)
    try:
        final_score = evaluate_sequence(best_seq_big.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as e:
        print("Final evaluation could not be performed:", e)

    return best_seq_big.tolist()
