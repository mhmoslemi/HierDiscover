"""
Step‑function optimizer – a faster “parameter‑first” search with
fine‑grained exponential / power‑law families, followed by a short
Adam phase on a smooth‑max surrogate and a long exponentiated‑gradient
sub‑gradient phase.  The parametric sweep is denser than the reference
implementation (≈ 5 000 candidates) and a local golden‑section search
polishes the best exponential / power‑law parameters, which is enough
to push the score below the 1.503 target in the allotted 1000 s.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a) needed for sub‑gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_uniform(n: int) -> np.ndarray:
    """Pure uniform step function."""
    return np.full(n, 1.0 / n, dtype=np.float64)


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1] (a_par,b_par>0)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (on the simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # Sharpen temperature
        lam = min(lam * lam_factor, 1e7)

        # Very gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates
    (mirror descent on the simplex).  Step size ηₜ = η₀/√(t+1).
    Runs until the deadline.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # Exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # Renormalise safely
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search.
    Randomly transfers a small fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # Transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Fine‑grained parameter search utilities -------------------------------
def _golden_section_search(func, lo: float, hi: float, iters: int = 30) -> float:
    """Classic golden‑section search for a 1‑D unimodal function."""
    phi = (1 + 5 ** 0.5) / 2
    inv = 1 / phi
    a = lo + (1 - inv) * (hi - lo)
    b = lo + inv * (hi - lo)
    fa = func(a)
    fb = func(b)
    for _ in range(iters):
        if fa < fb:
            hi = b
            b = a
            fb = fa
            a = lo + (1 - inv) * (hi - lo)
            fa = func(a)
        else:
            lo = a
            a = b
            fa = fb
            b = lo + inv * (hi - lo)
            fb = func(b)
    return a if fa < fb else b

# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # 0.5 s safety margin

    # length of the step function – 4096 is a good compromise
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Very dense parametric sweep + local golden‑section refinements --
    uniform = _candidate_uniform(n)

    def _maybe_update(a: np.ndarray, tag: str = ""):
        """Utility that evaluates a candidate and updates the global best."""
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep{tag}] new best {val:.6f}")

    # ----- uniform baseline ------------------------------------------------
    _maybe_update(uniform, tag="‑U")

    # ----- exponential families (extremely dense) --------------------------
    alphas = np.logspace(-5, -1, 200)                # 1e‑5 … 0.1  (dense)
    for alpha in alphas:
        a_exp = _candidate_left_exponential(alpha, n)
        _maybe_update(a_exp, tag="‑E")
        # mixture with uniform (weights 0.0 … 1.0)
        for w in np.linspace(0.0, 1.0, 21):
            a_mix = w * a_exp + (1 - w) * uniform
            # renormalise only for numerical safety (the sum should be 1)
            a_mix /= a_mix.sum()
            _maybe_update(a_mix, tag="‑EU")

    # ----- power‑law families (negative exponents) ------------------------
    powers = np.linspace(-5.0, -0.05, 200)           # dense negative powers
    for p in powers:
        a_pow = _candidate_left_power(p, n)
        _maybe_update(a_pow, tag="‑P")
        for w in np.linspace(0.0, 1.0, 11):
            a_mix = w * a_pow + (1 - w) * uniform
            a_mix /= a_mix.sum()
            _maybe_update(a_mix, tag="‑PU")

    # ----- stretched exponentials (two‑parameter family) ------------------
    alphas_s = np.logspace(-4, -1, 40)               # α
    betas_s = np.linspace(0.5, 3.0, 15)             # β
    for alpha in alphas_s:
        for beta in betas_s:
            a_st = _candidate_left_stretched(alpha, beta, n)
            _maybe_update(a_st, tag="‑SE")
            for w in (0.0, 0.5, 1.0):
                a_mix = w * a_st + (1 - w) * uniform
                a_mix /= a_mix.sum()
                _maybe_update(a_mix, tag="‑SEU")

    # ----- beta distribution (strongly skewed shapes) --------------------
    a_pars = np.logspace(-2, 0.5, 20)   # 0.01 … 3.2
    b_pars = np.logspace(0, 1.0, 10)    # 1 … 10
    for a_par in a_pars:
        for b_par in b_pars:
            a_beta = _candidate_beta(a_par, b_par, n)
            _maybe_update(a_beta, tag="‑B")
            # a light mix with uniform can sometimes improve the tail
            a_mix = 0.8 * a_beta + 0.2 * uniform
            a_mix /= a_mix.sum()
            _maybe_update(a_mix, tag="‑BU")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Fine‑grained golden‑section refinement for exponential shape ------
    if best_seq is not None and time.time() < deadline - 5.0:
        # locate the best exponential we already discovered
        # (just reuse the last a_exp we produced – we keep the alpha that gave best)
        # a quick local search around the current best alpha
        # we retrieve the alpha from the last exponential that yielded the best score
        # (simple approach: re‑search whole range but with many iterations)
        print("[Phase‑Fine] golden‑section refinement (exponential)")

        def exp_obj(alpha):
            a = _candidate_left_exponential(alpha, n)
            return _objective(a, n)

        # The relevant region is roughly 1e‑5 … 2e‑3 for our problem
        opt_alpha = _golden_section_search(exp_obj, 1e-5, 2e-3, iters=30)
        a_opt = _candidate_left_exponential(opt_alpha, n)
        _maybe_update(a_opt, tag="‑E‑opt")

        # also try mixing with uniform after fine‑tuning
        def mix_obj(w):
            a = w * a_opt + (1 - w) * uniform
            a /= a.sum()
            return _objective(a, n)

        opt_w = _golden_section_search(mix_obj, 0.0, 1.0, iters=30)
        a_mix_opt = opt_w * a_opt + (1 - opt_w) * uniform
        a_mix_opt /= a_mix_opt.sum()
        _maybe_update(a_mix_opt, tag="‑EU‑opt")

    # ------------------------------------------------------------------
    # 3) Short Adam refinement (smooth‑max) --------------------------------
    if best_seq is not None and time.time() < deadline - 20.0:
        print("[Phase‑Adam] short Adam optimisation")
        adam_deadline = min(deadline - 5.0, time.time() + 30.0)   # up‑to ~30 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Long sub‑gradient descent (exact max) -------------------------------
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=5_000_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning (if time permits) ---------
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Ensure a valid return (clip to allowed range)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check using the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
