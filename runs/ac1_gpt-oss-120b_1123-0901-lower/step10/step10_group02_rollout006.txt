"""
Hybrid multi‑start optimizer for the step‑function bound.

1.  Generate a large set of cheap closed‑form candidates (Gaussian, cosine,
    sine‑power windows, and mixtures with the uniform distribution).  The best
    of these is kept as a warm start.

2.  Refine the warm start with many short exponentiated‑gradient sub‑gradient
    runs.  Each run starts from a slightly perturbed copy of the current best
    and uses a larger learning‑rate schedule (eta0≈2.5).  The best vector seen
    over all runs is retained.

3.  A short Adam pass on the smooth‑max surrogate (the same code as the
    reference) is applied to exploit any remaining smooth descent direction.

4.  A final projected‑gradient descent on the exact max‑convolution cleans up
    the result.

5.  A cheap cyclic‑shift sweep is performed and the best shifted vector is
    returned.

All stages respect the global time budget (≈ 1000 s) and always return a
non‑negative sequence whose sum is 1 (the evaluation function is scale‑invariant).
"""

import time
import numpy as np
from scipy.optimize import linprog  # Optional, not required for the main flow.


# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (returns length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Closed‑form families of candidate shapes
def _gaussian_shape(n: int, sigma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    centre = (n - 1) / 2.0
    g = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
    return g

def _cosine_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    c = 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
    return c

def _sin_power_shape(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    s = np.sin(np.pi * (i + 0.5) / n) ** gamma
    return s

# Optimisation primitives -------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 2.5,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val

    Search for a non‑negative step‑function with a low evaluation score.

    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)

    # safety margin to avoid overshoot
    deadline = time.time() + budget_s - 0.3

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s                     # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 0 : fast exploration of closed‑form families
    # ------------------------------------------------------------------
    stage0_dead = min(deadline, time.time() + 0.12 * (deadline - time.time()))
    while time.time() < stage0_dead:
        # pick a family at random
        family = rng.integers(0, 3)
        if family == 0:  # Gaussian with random sigma
            sigma = rng.uniform(n / 200.0, n / 2.0)           # ~[20, 2000]
            shape = _gaussian_shape(n, sigma)
        elif family == 1:  # Cosine window (no extra parameter)
            shape = _cosine_window(n)
        else:  # Sine‑power shape with random exponent
            gamma = rng.uniform(0.5, 5.0)
            shape = _sin_power_shape(n, gamma)

        # mix with uniform to keep enough mass in the tails
        mix = rng.uniform(0.0, 1.0)
        cand = mix * np.ones(n) + (1.0 - mix) * shape
        cand = np.maximum(cand, 0.0)
        cand_sum = cand.sum()
        if cand_sum < 1e-12:
            continue
        cand /= cand_sum

        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, cand.copy()
            print(f"[Stage 0] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 : many short exponentiated‑gradient runs from perturbed starts
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage1_dead = min(deadline, time.time() + 0.68 * remaining)   # about 60 % of rest
    while time.time() < stage1_dead:
        # run a short sub‑gradient optimisation (≈ a few seconds)
        run_dead = min(stage1_dead, time.time() + 5.0)  # 5 s per mini‑run
        # perturb current best slightly
        noise = rng.normal(scale=0.01, size=n)
        a0 = best_a + noise
        a0 = np.maximum(a0, 0.0)
        s0 = a0.sum()
        if s0 < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        else:
            a0 /= s0

        a_opt, val_opt = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, run_dead,
            max_iters=5_000_000, eta0=2.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Stage 1] improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 : short Adam pass on smooth‑max (helps fine‑tune)
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage2_dead = min(deadline, time.time() + 0.12 * remaining)   # ~10 % of rest
    a_adam, val_adam = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, stage2_dead, max_iters=300_000
    )
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Stage 2] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 : brief projected‑gradient descent on the exact max‑conv
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    stage3_dead = min(deadline, time.time() + 0.07 * remaining)   # ~5 % of rest
    a_gd, val_gd = _max_conv_descent(
        best_a, n, conv_fft_len, corr_fft_len, stage3_dead, max_iters=200_000
    )
    if val_gd < best_val:
        best_val, best_a = val_gd, a_gd.copy()
        print(f"[Stage 3] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Final cheap cyclic‑shift search
    # ------------------------------------------------------------------
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # should never happen, but be safe
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # compute official evaluation for reporting (no effect on return value)
    try:
        from __main__ import evaluate_sequence  # the host environment provides it
    except Exception:  # pragma: no cover
        evaluate_sequence = lambda seq: float('inf')
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
