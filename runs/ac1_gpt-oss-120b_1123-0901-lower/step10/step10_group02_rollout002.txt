import time, numpy as np

def _next_pow2(x): ...
def _full_convolution(a, fft_len): ...
def _cross_correlation(w, a, fft_len): ...
def _project_simplex(v): ...
def _objective(a, n, conv_fft_len): ...

def _expgrad_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, eta0=1.0):
    a = np.maximum(a0, 0.0)
    if a.sum() == 0:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val, best_a = val, a.copy()
            # occasional print if improvement
            if it % 100000 == 0:
                print(f"[ExpGrad it={it}] new best {val:.6f}")
        # active set (max indices)
        active = np.isclose(conv, max_c, atol=1e-14)
        # weight uniform over active
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        # gradient w.r.t a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
        # learning rate
        eta = eta0 / np.sqrt(it + 1.0)
        # exponentiated gradient update
        # guard against overflow
        # compute new a using log to avoid overflow
        a = a * np.exp(-eta * grad)
        # renorm to simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1
    return best_a, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3
    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)
    # get warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.pad(a0, (0, n-a0.size), mode='constant')
    else:
        a0 = np.full(n, 1.0/n)
    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()
    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # allocate time allotments
    total = deadline - time.time()
    # Phase A: first deep expgrad from warm start (maybe 15% time)
    phase_a_dead = min(deadline, time.time() + 0.15 * total)
    a_opt, val_opt = _expgrad_optimize(best_a, n, conv_fft_len, corr_fft_len, phase_a_dead, eta0=1.2)
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[Phase A] improved to {val_opt:.6f}")

    # Phase B: random restarts
    remaining = deadline - time.time()
    # We'll perform up to 4 restarts, each gets equal chunk
    n_restarts = 4
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # allocate chunk
        chunk_dead = min(deadline, time.time() + remaining / (n_restarts - r))
        # random start
        a_start = np.random.dirichlet(np.ones(n))
        a_opt, val_opt = _expgrad_optimize(a_start, n, conv_fft_len, corr_fft_len, chunk_dead, eta0=0.9)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Restart {r}] improved to {val_opt:.6f}")
        remaining = deadline - time.time()
    # Phase C: fine-tune with very small eta0

        a_opt, val_opt = _expgrad_optimize(best_a, n, conv_fft_len, corr_fft_len, fine_dead, eta0=0.5)
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Fine-tune] improved to {val_opt:.6f}")

    # Phase D: cyclic shifts
    shifts = [n//8, n//4, 3*n//8, n//2, 5*n//8, 3*n//4, 7*n//8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # final clipping and sanity check
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")

def _local_random_search(a, n, conv_fft_len, deadline, attempts=5000, epsilon_factor=0.02):
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

"""
Hybrid random‑restart exponentiated‑gradient optimizer.

The routine works on a probability vector `a` (non‑negative, Σa=1) of length `n`.
It repeatedly minimizes the quantity

        2·n·max( a * a )        (where * is the linear convolution)

which is exactly the value returned by `evaluate_sequence` (the denominator
Σa² cancels because the objective is scale‑invariant).

Algorithm outline
-----------------
1. Initialise `a` from the supplied warm start `height_sequence_1` or from a
   uniform distribution.
2. Run a relatively long exponentiated‑gradient (EG) phase starting from the
   warm start.
3. Perform several *random restarts*: each restart starts from a fresh
   Dirichlet sample and runs EG for a share of the remaining time budget.
   The best vector among all runs is kept.
4. A final fine‑tuning EG pass with a smaller learning‑rate is executed.
5. A cheap cyclic‑shift sweep is applied (the convolution is not shift‑invariant
   for a linear convolution).
6. A short random local‑search (pairwise mass transfers) is performed if any
   time is left.
7. The best vector is clipped to the allowed range and returned as a plain
   Python list.

All steps respect the global time budget (≈ 1000 s) and the returned sequence
is guaranteed to be non‑negative, finite and have a sum > 0.01, so
`evaluate_sequence` will never output `inf`.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient minimisation of the max‑convolution.
    Returns the best vector seen (still on the simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[ExpGrad it={it}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        # gradient of the (exact) max‑convolution w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size schedule (classic 1/√t)
        eta = eta0 / np.sqrt(it + 1.0)

        # EG update – clip exponent to avoid overflow
        delta = -eta * grad
        delta = np.clip(delta, -50, 50)
        a = a * np.exp(delta)

        # projection back onto the simplex
        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a = a / a_sum

        it += 1

    return best_a, best_val


def _local_random_search(
    a: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_trials: int = 5000,
    eps_factor: float = 0.02,
) -> tuple[np.ndarray, float]:
    """
    Very cheap hill‑climbing: move a small amount of mass from a random
    index i to a random index j and keep the move if it improves the objective.
    """
    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    trials = 0
    while trials < max_trials and time.time() < deadline:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        eps = eps_factor * best_a[i]
        if eps <= 0:
            continue
        new_a = best_a.copy()
        new_a[i] -= eps
        new_a[j] += eps
        # keep feasibility
        if new_a[i] < 0:
            continue
        val = _objective(new_a, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, new_a.copy()
        trials += 1
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    a valid sequence (as a plain Python list).
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.3   # tiny safety margin

    # ----- configuration -------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ----- warm start ----------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=np.float64)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.pad(a0, (0, n - a0.size), mode="constant")
    else:
        a0 = np.full(n, 1.0 / n, dtype=np.float64)

    a0 = np.maximum(a0, 0.0)
    if a0.sum() < 0.01:
        a0 = np.random.dirichlet(np.ones(n))
    a0 = a0 / a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ----- phase A : deep EG from warm start (≈15 % of time) -------------
    remaining = deadline - time.time()
    if remaining <= 0:
        return best_a.tolist()
    phase_a_dead = time.time() + 0.15 * remaining
    a_opt, val_opt = _expgrad_optimize(
        best_a, n, conv_fft_len, corr_fft_len, phase_a_dead, eta0=1.2
    )
    if val_opt < best_val:
        best_val, best_a = val_opt, a_opt.copy()
        print(f"[Phase A] improved to {val_opt:.6f}")

    # ----- phase B : several random restarts (≈60 % of time) -------------
    remaining = deadline - time.time()
    n_restarts = 5
    for r in range(n_restarts):
        if time.time() >= deadline:
            break
        # allocate a fair share of the still available time
        share = (deadline - time.time()) / (n_restarts - r)
        restart_dead = time.time() + share
        a_start = np.random.dirichlet(np.ones(n))
        a_opt, val_opt = _expgrad_optimize(
            a_start, n, conv_fft_len, corr_fft_len, restart_dead, eta0=0.9
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Restart {r}] improved to {val_opt:.6f}")

    # ----- phase C : fine‑tuning EG with a smaller learning rate ---------
    if time.time() < deadline:
        fine_dead = deadline
        a_opt, val_opt = _expgrad_optimize(
            best_a, n, conv_fft_len, corr_fft_len, fine_dead, eta0=0.5
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Fine‑tune] improved to {val_opt:.6f}")

    # ----- phase D : deterministic cyclic‑shift sweep --------------------
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ----- phase E : tiny random local search (if time permits) ---------
    if time.time() < deadline:
        a_opt, val_opt = _local_random_search(
            best_a, n, conv_fft_len, deadline, max_trials=8000, eps_factor=0.01
        )
        if val_opt < best_val:
            best_val, best_a = val_opt, a_opt.copy()
            print(f"[Local search] improved to {val_opt:.6f}")

    # ----- final sanitisation -------------------------------------------
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # safety net
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # ----- report final evaluation ---------------------------------------
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
