Hybrid optimizer for the “max‑convolution” objective.

The optimizer works in three stages:

1️⃣  **Extensive parametric sweep** – a large pool of candidate vectors is

2️⃣  **Exponentiated‑gradient refinement** – starting from the best pool

    probability simplex.  The exact sub‑gradient of the objective is obtained

    line‑search with a 1/√t schedule is used.  The loop runs until the global
    time budget (≈ 1000 s) is exhausted, always remembering the best ever seen

3️⃣  **Simulated‑annealing polish** – a short stochastic local search that
    repeatedly moves a random fraction of mass between two coordinates.
    Down‑hill moves are always accepted; uphill moves are accepted with a
    temperature schedule.  This phase helps to escape tiny local minima

The final sequence is clipped to the limits imposed by ``evaluate_sequence``

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a simplex‑normalised vector."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a (possibly signed) vector onto the probability simplex."""
    # Simple “clip‑and‑renorm”.  Guarantees non‑negativity and sum = 1.
    v = np.maximum(v, 0.0)
    s = v.sum()
    if s <= 0:
        # fall back to the uniform distribution
        return np.full_like(v, 1.0 / v.size, dtype=np.float64)
    return v / s

            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        pool.append(_project_to_simplex(a))

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of the whole pool – keep the best one
    for i, cand in enumerate(pool):
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Init #{i:04d}] new best {val:.6f}")

    # If the pool was empty (should never happen) fall back to uniform.
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient refinement (mirror descent on the simplex)
    it = 0
    step_const = 0.9                     # initial step‑size factor
    max_iters = 2_500_000                # generous limit; will be cut by time

    while time.time() < deadline - 5 and it < max_iters:
        a = best_seq.copy()

        # compute convolution once per outer iteration – heavy part
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        # store the best found so far (might be the previous best)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[EG it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the maximum (sub‑gradient support)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # step size – classic 1/√t schedule
        step = step_const / math.sqrt(it + 1.0)

        # back‑tracking line‑search (few trial factors)
        improved = False
        for factor in (1.0, 0.6, 0.3, 0.1, 0.05):
            a_new = a * np.exp(-step * factor * grad)
            a_new = _project_to_simplex(a_new)
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new <= val or factor == 0.05:
                # accept this trial
                a = a_new
                if val_new < best_val:
                    best_val = val_new
                    best_seq = a_new.copy()
                    print(f"[EG LS it={it:07d}] improved to {val_new:.6f}")
                improved = True
                break

        if not improved:
            # If none of the factors gave improvement, shrink the global step constant
            step_const *= 0.92

        it += 1

    # ------------------------------------------------------------------
    # 4️⃣  Short simulated‑annealing polish
    if time.time() < deadline - 2:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()

        # temperature schedule – fast cooling (the problem is well‑behaved)
        T_start = 0.07
        T_end = 3e-6
        max_moves = 80_000

        for mv in range(max_moves):
            if time.time() > deadline - 1:
                break
            # linear cooling
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            # pick two distinct indices, move a fraction of mass from i → j
            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            # fraction of mass to relocate (up to 40 % of a[i])
            delta = a[i] * (np.random.rand() * 0.4)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                val = val2
                print(f"[Anneal move {mv}] improved to {val2:.6f}")
            else:
                # Metropolis acceptance
                if np.random.rand() < math.exp(-(val2 - val) / T):
                    a = a2
                    val = val2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final clean‑up and return
    if best_seq is None:                      # safety net
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)   # respect the evaluator limits
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")

    # Optional sanity check – real evaluator (commented out for speed)
    # try:
    #     real_val = evaluate_sequence(best_seq.tolist())
    #     print("Evaluator reports:", real_val)
    # except Exception:
    #     pass

    return best_seq.tolist()
