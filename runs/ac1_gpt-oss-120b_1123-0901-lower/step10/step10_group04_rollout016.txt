"""
Propose a low‑value step‑function for evaluate_sequence.

The optimiser works in four stages:

1. **Rich parametric sweep** – a large family of analytically‑defined shapes
   (power‑law, product‑power, (mirrored) exponential, stretched‑exponential,
   symmetric “U‑exponential”, beta, double‑Gaussian, sine, random spikes,
   Dirichlet draws) is sampled on a grid of parameters.  Every candidate is
   normalised to ℓ₁‑norm = 1 and evaluated exactly with an FFT‑based convolution.
   The best few candidates are retained.

2. **Adam on a smooth‑max surrogate** – starting from the current incumbent,
   Adam is run on
       smooth_max(conv) + λ·‖conv – uniform‖₂²
   where smooth_max uses a temperature‑controlled soft‑max.  The update is
   performed in the log‑softmax domain (mirror descent on the simplex) which
   guarantees non‑negativity and unit sum.

3. **Exact‑max exponentiated‑gradient sub‑gradient descent** – the true
   objective 2·n·max(conv)/‖a‖₁² is minimised by an exponentiated‑gradient
   update that uses the active set where the convolution attains its maximum.
   The best vector over the whole run is kept.

4. **Optional annealing fine‑tune** – a cheap simulated‑annealing that moves a
   small fraction of mass between two random positions.

All stages respect the supplied time budget (default ≈ 1000 s).  The routine
always returns a valid list of non‑negative numbers.

import time

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""

    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a

def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_u_exponential(alpha: float, n: int) -> np.ndarray:
    """
    Symmetric U‑exponential a_i ∝ exp(-α·min(i, n‑1‑i)),
    i.e. large at the two ends, small in the centre.
    """
    i = np.arange(n, dtype=np.float64)
    d = np.minimum(i, n - 1 - i)
    a = np.exp(-alpha * d)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    # safe powers
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation (smooth‑max + L2 regularisation) --------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 700_000,
    lam_start: float = 0.07,
    lam_factor: float = 1.008,
    l2_weight: float = 0.13,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the surrogate smooth‑max + L² term."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # simplex via soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for monitoring
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L2 regulariser -----
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # mirror‑descent step (log‑softmax)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 150_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 8_000_000,
    eta0: float = 1.1,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient descent on the true max‑norm."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (optional) ---------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing that shifts a fraction of mass between two bins."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises `evaluate_sequence`.
    The routine respects the supplied time budget and always returns a valid
    list of coefficients.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5          # tiny safety margin

    # Problem size – 4096 gives a good trade‑off between quality and speed.
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1️⃣ Extensive parametric sweep
    # ------------------------------------------------------------------

    # Power‑law (negative exponents = U‑shape)
    for p in np.linspace(-10.0, -0.02, 45):
        _try_candidate(_candidate_power(p, n))

    # Power‑product (more aggressive U‑shape)
    for p in np.linspace(-5.0, -0.05, 30):
        _try_candidate(_candidate_power_product(p, n))

    # Symmetric exponential (mirrored)
    for alpha in np.logspace(-6, 2.5, 60):
        _try_candidate(_candidate_exponential(alpha, n))

    # U‑exponential (high at edges, low in centre)
    for alpha in np.logspace(-6, 2.0, 55):
        _try_candidate(_candidate_u_exponential(alpha, n))

    # Stretched exponential
    alphas = np.logspace(-4, 0.8, 25)
    betas = np.linspace(0.2, 3.5, 18)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n))

    # Beta distribution – systematic symmetric sweep
    for a_par in np.logspace(-2.5, 1.2, 70):   # covers 0.003 to ~16
        _try_candidate(_candidate_beta(a_par, a_par, n))

    # Double Gaussian (two ends)
    for sig in np.logspace(-3.5, -0.2, 30):
        _try_candidate(_candidate_double_gaussian(sig, n))

    # Left‑exponential (monotone decreasing)
    for alpha in np.logspace(-6, 1.8, 40):
        _try_candidate(_candidate_left_exponential(alpha, n))

    # Left‑power (monotone)
    for p in np.linspace(-5.0, 2.0, 30):
        _try_candidate(_candidate_left_power(p, n))

    # Sine‑shaped bump (central bump – still useful as a baseline)
    for pow_ in np.linspace(0.5, 5.0, 25):
        _try_candidate(_candidate_sine(pow_, n))

    # Random Dirichlet spikes (different sparsities)
    for k in [16, 32, 64, 128, 256]:
        for _ in range(6):
            _try_candidate(_candidate_spike(k, n, random=True))

    # Fully random Dirichlet (dense)
    for _ in range(10):
        a = np.random.dirichlet(np.ones(n))
        _try_candidate(a)

    # Warm start from any previously stored sequence
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Adam refinement on smooth‑max surrogate
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] starting smooth‑max Adam")
        adam_deadline = min(deadline - 5,
                            time.time() + 0.35 * (deadline - time.time()))
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.06,
            lam_factor=1.009,
            l2_weight=0.14,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Exact‑max sub‑gradient (exponentiated‑gradient)
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] launching exponentiated‑gradient")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Optional annealing fine‑tune
    # ------------------------------------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=220_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
