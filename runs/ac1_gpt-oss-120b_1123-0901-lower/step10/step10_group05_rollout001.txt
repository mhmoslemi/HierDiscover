"""
Hybrid random‑restart + exponentiated‑gradient + simulated‑annealing optimizer.

* Phase 0 – Warm‑start: use the previously‑found construction if available.
* Phase 1 – Broad parametric sweep:
    – power‑law ends‑heavy shapes a_i ∝ (i·(n‑i+1))^α  (α < 0)
    – symmetric beta shapes a_i ∝ x^p·(1‑x)^q
    – Dirichlet draws with random concentration
    – piece‑wise constant blocks
    – sparse spikes
  All candidates are normalised to sum = 1 and the best one is kept.
* Phase 2 – Intensive exponentiated‑gradient (mirror‑descent) refinement.
  The true sub‑gradient of max(a*a) is obtained via a cross‑correlation with an
  active‑set weight vector.  A decreasing step size 𝛾_t≈c/√t together with a
  tiny back‑tracking line‑search is applied many thousands of times.
* Phase 3 – Short simulated‑annealing polish to escape possible local minima.
  Random mass transfers between two coordinates are accepted according to a
  linear cooling schedule.

The routine respects a total time budget (≈ 1000 s) and always returns the
best sequence found.  All numbers are clipped to the evaluator’s limits
(≥ 0, ≤ 1000) before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) assuming sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _candidate_from_array(arr: np.ndarray) -> np.ndarray:
    """Clip, make non‑negative, normalise to sum = 1."""
    arr = np.clip(arr, 0.0, 1000.0)
    s = arr.sum()
    if s < 1e-12:
        return None
    return arr / s


def _generate_powerlaw(n: int, alpha: float) -> np.ndarray:
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    shape = prod ** alpha
    return _candidate_from_array(shape)


def _generate_beta(n: int, p: float, q: float) -> np.ndarray:
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    shape = (x ** p) * ((1.0 - x) ** q)
    return _candidate_from_array(shape)


def _generate_dirichlet(n: int, conc: float) -> np.ndarray:
    return np.random.dirichlet(conc * np.ones(n))


def _generate_blocks(n: int, blocks: int) -> np.ndarray:
    block_len = n // blocks
    a = np.zeros(n, dtype=np.float64)
    masses = np.random.dirichlet(np.ones(blocks))
    for b in range(blocks):
        a[b * block_len:(b + 1) * block_len] = masses[b]
    return _candidate_from_array(a)


def _generate_spikes(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    pos = np.random.choice(n, size=k, replace=False)
    masses = np.random.dirichlet(np.ones(k))
    a[pos] = masses
    return _candidate_from_array(a)


def _refine_exponentiated_gradient(a0: np.ndarray,
                                   n: int,
                                   conv_fft_len: int,
                                   corr_fft_len: int,
                                   deadline: float,
                                   max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """Mirror‑descent (exponentiated gradient) on the max‑convolution objective."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    step_const = 0.9

    while time.time() < deadline - 2 and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[EG it={it}] new best {val:.6f}")

        # active set for the max entry
        active = np.isclose(conv, max_c, atol=1e-14)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / np.sum(active)

        # sub‑gradient of max(conv) w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = step_const / np.sqrt(it + 1.0)

        # simple back‑tracking line‑search (few factors)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            a_new = a * np.exp(-step * factor * grad)
            s = a_new.sum()
            if s < 1e-12:
                continue
            a_new = a_new / s
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new <= val or factor == 0.05:
                a = a_new
                improved = True
                break
        if not improved:
            step_const *= 0.9  # shrink step if nothing helped

        it += 1

    return best_a, best_val


def _simulated_annealing(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         moves: int = 70_000) -> tuple[np.ndarray, float]:
    """Very short SA polish moving a random mass fraction between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_a = a.copy()
    best_val = val

    T_start = 0.07
    T_end = 1e-5

    for mv in range(moves):
        if time.time() > deadline - 1:
            break
        T = max(T_end, T_start * (1.0 - mv / moves))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        delta = a[i] * (np.random.rand() * 0.4)  # up to 40 % of source mass
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            print(f"[SA move {mv}] improved to {val2:.6f}")
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
        val = val2

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem size – we keep the reference size (4096) which works well
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # helper to test a candidate and possibly update the current best
    def consider(cand: np.ndarray):
        nonlocal best_seq, best_val
        if cand is None:
            return
        s = cand.sum()
        if s < 0.01:
            return
        cand = cand / s
        val = _objective(cand, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = cand.copy()
            print(f"[Init] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 0️⃣ Warm‑start (if a previous construction is present)
    warm = globals().get("height_sequence_1")
    if isinstance(warm, (list, tuple, np.ndarray)):
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        consider(_candidate_from_array(w))

    # ------------------------------------------------------------------
    # 1️⃣ Broad parametric sweep (very cheap)
    # Power‑law ends‑heavy
    alphas = np.linspace(-4.0, -0.1, 200)          # dense sweep
    for alpha in alphas:
        consider(_generate_powerlaw(n, alpha))

    # Symmetric beta shapes (p = q)
    ts = np.linspace(0.05, 5.0, 150)
    for t in ts:
        consider(_generate_beta(n, t, t))

    # Dirichlet draws with random concentration
    while time.time() < deadline - 30:
        conc = 10 ** np.random.uniform(-2, 1)      # from 0.01 to 10
        consider(_generate_dirichlet(n, conc))

    # Piece‑wise constant blocks
    while time.time() < deadline - 30:
        blocks = np.random.choice([8, 16, 32, 64])
        consider(_generate_blocks(n, blocks))

    # Sparse spikes
    while time.time() < deadline - 30:
        k = np.random.randint(30, 500)
        consider(_generate_spikes(n, k))

    # Fallback to uniform if nothing succeeded
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 2️⃣ Intensive exponentiated‑gradient refinement on the current best
    best_seq, best_val = _refine_exponentiated_gradient(
        best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=2_000_000
    )
    print(f"[After EG] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3️⃣ Short simulated‑annealing polish
    if time.time() < deadline - 2:
        best_seq, best_val = _simulated_annealing(
            best_seq, n, conv_fft_len, deadline, moves=70_000
        )
        print(f"[After SA] best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Final clean‑up
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
