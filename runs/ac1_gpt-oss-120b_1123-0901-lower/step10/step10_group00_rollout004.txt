def random_phase_vector(n):
    # number of rfft output components = n//2+1
    N_half = n//2+1
    phases = np.random.rand(N_half) * 2*np.pi
    # Force DC phase 0 and magnitude = 0? We'll set DC as 0 (since sum after projection)
    # Set amplitude for frequencies >0 to 1 * exp(i phi)
    # For continuity real sequence, we just use rfft with complex numbers.
    A = np.zeros(N_half, dtype=np.complex128)
    A[0] = 0.0 # we will adjust after projection
    # amplitude constant for k>0
    A[1:] = np.exp(1j * phases[1:]) # magnitude 1
    a = np.fft.irfft(A, n=n)
    # Now shift to positivity: project onto simplex.
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12:

    return a

def _spectral_flatten_optimize(a0, n, conv_fft_len, deadline, max_iters=200_000):
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0*n*max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # Compute FFT (full length n)
        A = np.fft.rfft(a, n=n)  # length n//2+1
        # Compute energy (l2 norm squared)
        energy = np.dot(a, a)
        # Compute constant magnitude for non-DC frequencies:
        # Using Parseval: (|A0|^2 + 2*sum_{k=1}^{n/2-1} |A_k|^2 + extra) = n * energy
        # We'll preserve DC value and distribute remaining energy uniformly.
        # Number of non-DC terms in rfft (including Nyquist if even) = len(A) - 1
        nd = len(A) - 1
        if nd > 0:
            rem_energy = max(0.0, n * energy - (np.abs(A[0])**2))
            const_mag = np.sqrt(rem_energy / nd)
        else:
            const_mag = 0.0

        # Randomize phases
        phases = np.random.rand(nd) * 2*np.pi
        A_new = np.empty_like(A)
        A_new[0] = A[0]  # keep DC (complex)
        A_new[1:] = const_mag * np.exp(1j * phases)

        a = np.fft.irfft(A_new, n=n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        it += 1
    return best_a, best_val

"""
Hybrid optimizer for the non‑negative step‑function problem.

The search proceeds in several timed phases:

1. **Random‑phase spectral exploration** – generate many candidates whose
   discrete Fourier transform has (almost) unit magnitude and random phase,
   then project onto the probability simplex.  These vectors tend to produce
   a very flat self‑convolution, and a large number of cheap trials quickly
   discovers a good starting point.

2. **Adam optimisation** on a smooth‑max surrogate of the max‑convolution,
   starting from the best candidate found so far.  The optimiser works in the
   log‑softmax parametrisation and follows the schedule used in the reference
   implementation but with a larger iteration budget.

3. **Projected gradient descent** on the exact max‑convolution (sub‑gradient
   method).  This further reduces the max‑convolution value while keeping the
   iterate on the probability simplex.

4. **Spectral‑flattening with random phases** – repeatedly enforce a constant
   magnitude of the (real) Fourier spectrum while re‑randomising the phases.
   After each inverse FFT the vector is projected back onto the simplex.

5. **Simulated‑annealing refinement** – a classical local‑search that moves a
   random fraction of mass between two coordinates, accepting improving moves
   and occasionally worsening moves according to a temperature schedule.

All phases respect a global time budget (≈ 1000 s).  The routine always returns
the best sequence seen before the deadline.  The returned list contains only
non‑negative numbers ≤ 1000, as required by ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic linear‑algebra helpers
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Random‑phase spectral candidate generator
# ----------------------------------------------------------------------
def _random_phase_sequence(n: int) -> np.ndarray:
    """
    Generate a random‑phase vector whose real FFT has unit magnitude on all
    non‑DC frequencies, then project onto the probability simplex.
    """
    # Length of rfft output for a real vector of length n
    h = n // 2 + 1
    phases = np.random.rand(h - 1) * 2.0 * np.pi
    A = np.empty(h, dtype=np.complex128)
    A[0] = 0.0                     # DC component will be fixed by the simplex projection
    A[1:] = np.exp(1j * phases)    # unit magnitude, random phase
    a = np.fft.irfft(A, n=n)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 1e-12 or np.isnan(s):
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Projected gradient descent on the exact max‑convolution
# ----------------------------------------------------------------------
def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Spectral‑flattening with random phases
# ----------------------------------------------------------------------
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Iterate:
        * keep the DC term,
        * enforce a constant magnitude on all non‑DC Fourier coefficients,
        * draw fresh random phases each iteration,
        * project back onto the probability simplex.
    The best iterate (measured by the exact max‑convolution) is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ----- random‑phase constant‑magnitude projection -----
        # full rfft length
        A = np.fft.rfft(a, n=n)
        energy = np.dot(a, a)                     # ||a||_2^2
        nd = len(A) - 1                           # number of non‑DC bins
        if nd > 0:
            # preserve DC amplitude, spread the rest of the energy uniformly
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / nd)
        else:
            const_mag = 0.0

        # new random phases for all non‑DC bins
        phases = np.random.rand(nd) * 2.0 * np.pi
        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        A_new[1:] = const_mag * np.exp(1j * phases)

        a = np.fft.irfft(A_new, n=n)
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing local search
# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """Classic simulated‑annealing on the simplex (mass moves between two slots)."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4          # move up to 40 % of mass at i
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimising
    ``evaluate_sequence``.  The routine respects the global time budget
    (≈ 1000 s) and always returns the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # Length of the vector – 4096 offers a good trade‑off (quality vs speed)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best, if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]), mode="constant")
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase A – massive random‑phase spectral exploration
    phase_a_deadline = min(deadline, time.time() + budget_s * 0.25)
    it = 0
    while time.time() < phase_a_deadline:
        a = _random_phase_sequence(n)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c           # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Phase A it={it}] new best {val:.6f}")
        it += 1

    # Also evaluate the warm start (if any)
    if warm is not None:
        conv = _full_convolution(warm, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_seq = warm.copy()
            print(f"[Warm] initial best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase B – Adam optimisation (smooth‑max surrogate)
    if best_seq is not None and time.time() < deadline - 5:
        sub_dead = min(deadline, time.time() + budget_s * 0.25)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=600_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase C – projected sub‑gradient descent on the exact max‑convolution
    if best_seq is not None and time.time() < deadline - 5:
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline, max_iters=300_000
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase D – spectral flattening with random phases (more iterations)
    if best_seq is not None and time.time() < deadline - 5:
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase E – simulated‑annealing refinement (longer run)
    if best_seq is not None and time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=150_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] final best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * (_full_convolution(best_seq, conv_fft_len).max())
        print(f"[Fallback] val {best_val:.6f}")

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
