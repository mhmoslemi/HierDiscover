"""
Propose a non‑negative step‑function (list of heights) with a low evaluation score.

Algorithm:
1.  Broad parametric sweep – generate a diverse set of symmetric / monotone
    sequences (power‑law, exponential, stretched‑exponential, beta,
    double‑Gaussian, sparse spikes).  Each candidate is normalised to sum = 1.
    The K=8 best candidates are kept (elite list).

2.  Hill‑climbing with incremental convolution updates.
    Starting from each elite candidate we perform a large number of
    “mass‑transfer” moves: a small fraction of the mass at a source index
    `i` is moved to a target index `j`.  The effect on the full convolution
    `a*a` can be updated in O(n) time using the analytic formula
    Δconv = 2·δ·(shift_j(a) − shift_i(a)) + δ²·(e_{2i}+e_{2j}−2e_{i+j}),
    where δ is the transferred mass.  A move is accepted only when the
    peak of the convolution decreases.  The best vector found across all
    starts is kept.

3.  Optional long sub‑gradient (exponentiated‑gradient) descent
    on the exact max‐convolution, using the code from the previous
    baseline.  This fine‑tunes the result if time permits.

All phases respect the overall time budget (default 1000 s) and the
function always returns a valid non‑negative list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a

def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a

def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a

def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a

def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p with p < 0 (edges heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of exact max (cross‑correlation)
        # use FFT for speed
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = n - 1
        grad_a = 2.0 * corr[start : start + n]

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = None,
    max_frac: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing (mass‑transfer) optimiser.
    Moves mass δ = frac * a[i] from a source index i to a target j.
    The convolution is updated incrementally in O(n) time.
    Only moves that strictly decrease max(conv) are kept.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    conv = np.convolve(a, a)  # full convolution (length 2n‑1)
    cur_max = conv.max()
    best_a = a.copy()
    best_val = _objective(a, n)

    moves = 0
    start_time = time.time()
    while time.time() < deadline:
        if max_moves is not None and moves >= max_moves:
            break
        # pick source i with probability proportional to a (larger values more likely)
        # to keep it simple we use a weighted random draw
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue
        # pick a different target uniformly
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1
        # fraction of source mass to move
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # --- incremental update of convolution (copy + slice ops) ---
        new_conv = conv.copy()
        # shift contributions
        new_conv[i:i + n] -= 2.0 * delta * a
        new_conv[j:j + n] += 2.0 * delta * a
        # delta^2 contributions
        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

        new_max = new_conv.max()
        if new_max < cur_max - 1e-15:   # strict improvement
            # accept
            conv = new_conv
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[HillClimb move={moves}] new best {val:.6f}")
        moves += 1
        # occasional progress report
        if moves % 100_000 == 0:
            elapsed = time.time() - start_time
            print(f"[HillClimb] moves={moves}, elapsed={elapsed:.1f}s, best={best_val:.6f}")

    return best_a, best_val

def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """Edges‑heavy shape: a_i ∝ (i+1)^p + (n-i)^p with p < 0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Optimisers
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # gradient of exact max via cross‑correlation (FFT)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = n - 1
        grad_a = 2.0 * corr[start : start + n]

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _hill_climb_optimize(
    a0: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = None,
    max_frac: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Hill‑climbing optimiser that repeatedly transfers a random fraction of
    mass from a source index `i` to a target index `j`.  The convolution
    a*a is updated analytically in O(n) time; only moves that strictly lower
    max(conv) are kept.
    """
    a = a0.copy()
    a = np.maximum(a, 0.0)
    a /= a.sum()
    conv = np.convolve(a, a)            # full convolution (length 2n‑1)
    cur_max = conv.max()
    best_a = a.copy()
    best_val = _objective(a, n)

    moves = 0
    start_time = time.time()
    while time.time() < deadline:
        if max_moves is not None and moves >= max_moves:
            break

        # source index – weighted by current mass (larger entries are tried more)
        i = np.random.choice(n, p=a / a.sum())
        if a[i] < 1e-14:
            continue

        # target index – any other position
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # fraction of source mass to move
        frac = np.random.rand() * max_frac
        delta = a[i] * frac
        if delta <= 0.0:
            continue

        # --- incremental update of the convolution (copy + slice ops) ---
        new_conv = conv.copy()
        new_conv[i : i + n] -= 2.0 * delta * a
        new_conv[j : j + n] += 2.0 * delta * a
        new_conv[2 * i] += delta * delta
        new_conv[2 * j] += delta * delta
        new_conv[i + j] -= 2.0 * delta * delta

        new_max = new_conv.max()
        if new_max < cur_max - 1e-15:        # strict improvement
            conv = new_conv
            a[i] -= delta
            a[j] += delta
            cur_max = new_max
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[HillClimb move={moves}] new best {val:.6f}")

        moves += 1
        if moves % 100_000 == 0:
            elapsed = time.time() - start_time
            print(f"[HillClimb] moves={moves}, elapsed={elapsed:.1f}s, best={best_val:.6f}")

    return best_a, best_val


# ----------------------------------------------------------------------
# Elite handling helper
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite, keeping the list sorted by increasing val."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(budget_s: float = 1000, seed: int = 0) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    a valid list of heights.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 0.5   # tiny safety margin

    # --------------------------------------------------------------
    # Problem dimension (use power‑of‑two for FFT speed)
    # --------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        n = len(warm)
    else:
        n = 4096
    if (n & (n - 1)) != 0:               # ensure power of two
        n = _next_pow2(n)

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Broad parametric sweep – keep a few elite candidates
    # --------------------------------------------------------------
    elite = []

    def _maybe_update(a):
        val = _objective(a, n)
        _update_elite(elite, val, a)

    # symmetric power‑law (edges heavy)
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential
    for alpha in np.logspace(-4, 0.5, 8):
        for beta in np.linspace(0.3, 3.0, 8):
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta family
    for a_par in [0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
        for b_par in [0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-3, -0.1, 12):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.1, 12):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # inverse‑power (edges heavy)
    for p in np.linspace(-4.5, -0.2, 20):
        _maybe_update(_candidate_inverse_power(p, n))

    # random spikes
    for k in [32, 64, 128, 256]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True))

    # warm start from previously stored solution
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates generated; falling back to uniform")

    # --------------------------------------------------------------
    # 2) Hill‑climbing mass‑transfer optimisation
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        hill_deadline = time.time() + min(remaining * 0.6, 250.0)
        print("[Phase‑HillClimb] starting hill‑climbing")
        # try the three best elite starts independently
        for idx, (_, a_start) in enumerate(elite[:3]):
            if time.time() >= hill_deadline:
                break
            a_hc, val_hc = _hill_climb_optimize(
                a_start, n, hill_deadline, max_moves=None, max_frac=0.5
            )
            if val_hc < best_val:
                best_val = val_hc
                best_seq = a_hc.copy()
                print(f"[Phase‑HillClimb] start {idx} improved to {val_hc:.6f}")

    # --------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient optimisation (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=8_000_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Final safety checks and public evaluation
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
