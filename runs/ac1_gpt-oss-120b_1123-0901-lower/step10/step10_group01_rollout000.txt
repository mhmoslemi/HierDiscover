def _targeted_transfer_optimize(a, n, conv_fft_len, corr_fft_len, deadline):
    best_a = a.copy()
    conv = _full_convolution(best_a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    while time.time() < deadline:
        conv = _full_convolution(best_a, conv_fft_len)
        max_c = conv.max()
        # active indices
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, best_a, corr_fft_len)  # gradient of max_c wrt a

        # find indices
        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        # if gradient direction minimal
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        # step size epsilon
        eps = min(best_a[i_high] * 0.5, 0.001)  # maybe adjust dynamic
        # ensure positive low index
        if eps <= 0:
            break

        # try move
        a_new = best_a.copy()
        a_new[i_high] -= eps
        a_new[i_low] += eps
        # ensure non-negative and sum unchanged
        if a_new[i_high] < -1e-12:  # should not happen
            continue

        # compute new objective
        conv_new = _full_convolution(a_new, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()
        if val_new < best_val - 1e-12:  # improve
            best_val = val_new
            best_a = a_new
            # optionally allow some randomization: continue
        else:
            # reduce eps and try again up to few attempts
            # maybe try smaller eps
            for factor in (0.5, 0.2, 0.1):
                eps2 = eps * factor
                if eps2 < 1e-12:
                    break
                a2 = best_a.copy()
                a2[i_high] -= eps2
                a2[i_low] += eps2
                if a2[i_high] < 0:
                    continue
                conv2 = _full_convolution(a2, conv_fft_len)
                val2 = 2.0 * n * conv2.max()
                if val2 < best_val:
                    best_val = val2
                    best_a = a2
                    break
        it += 1
        # optional break after many iterations
        if it >= 200000:
            break
    return best_a, best_val

def _make_beta(n, alpha, beta):
    i = (np.arange(n) + 0.5) / n
    a = i**(alpha-1) * (1-i)**(beta-1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    a_sum = a.sum()
    if a_sum <= 0:
        a = np.full(n, 1.0/n)
    else:
        a = a / a_sum
    return a

"""
Hybrid optimiser with extra fine‑grained mass‑transfer refinement.

The algorithm
----------------
1. Build a diverse pool of candidate step‑functions.
   • Warm‑start (if provided), several Dirichlet draws,
   • exponential, Gaussian, power‑law, triangular, uniform,
   • symmetric Beta distributions (including the arcsine shape),
   • simple cosine‑shaped wave.

2. For each candidate (up to a small number) run a quick Adam optimisation
   on a smooth‑max surrogate of the peak of the self‑convolution.
   If time permits, follow it by a longer exponentiated‑gradient sub‑gradient
   phase.

3. Keep the best vector found so far (normalised to sum = 1).

4. Refine that vector with projected gradient descent on the exact
   max‑convolution (convex‑but‑nonsmooth) for a few seconds.

5. **New:** Perform a targeted mass‑transfer local search.
   At each step we compute the exact gradient of the current maximum
   convolution (the average cross‑correlation over the active indices)
   and move a small amount of mass from the coordinate with the largest
   gradient to the one with the smallest gradient.  The step size is
   adaptively reduced until an improvement is found.  This simple
   sub‑gradient step often yields the tiny reductions needed to beat the
   previous bound.

6. If any time remains we run a short simulated‑annealing pass.

7. The final vector is clipped to the evaluator’s allowed range and
   returned as a plain Python list.

The whole routine respects the 1000 s budget and prints intermediate
objective values for monitoring.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and abs(v.sum() - 1.0) < 1e-12:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_beta(n, alpha, beta):
    """Symmetric (or asymmetric) Beta pdf on a uniform grid of mid‑points."""
    # avoid exact 0/1 to keep finite values
    i = (np.arange(n) + 0.5) / n
    a = np.power(i, alpha - 1) * np.power(1 - i, beta - 1)
    a[np.isnan(a) | np.isinf(a)] = 0.0
    s = a.sum()
    if s <= 0:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_cosine(n):
    """Simple cosine‐bell shape."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) / 2
    a /= a.sum()
    return a

# Optimisers (adapted from previous state‑of‑the‑art)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp surrogate of max(conv).
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c                     # sum(a)=1 ⇒ denominator =1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 70_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Plain projected gradient descent on the exact max‑convolution.
    Uses a diminishing step size and keeps the iterate on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 100_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.07
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.45
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# New fine‑grained mass‑transfer optimiser

def _mass_transfer_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the coordinate with the
    largest gradient (w.r.t. the current max‑convolution) to the one with
    the smallest gradient, accepting moves that strictly reduce the exact
    objective.  This often yields the tiny improvements needed for the
    final bound.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if not np.any(active):
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_high = int(np.argmax(grad))
        i_low = int(np.argmin(grad))
        # If gradients are almost equal we are near a stationary point
        if grad[i_high] <= grad[i_low] + 1e-15:
            break

        # try a few step sizes (fractions of the source mass)
        improved = False
        src_mass = a[i_high]
        for frac in (0.5, 0.3, 0.15, 0.07, 0.03, 0.01):
            eps = src_mass * frac
            if eps <= 0:
                continue
            a_new = a.copy()
            a_new[i_high] -= eps
            a_new[i_low] += eps
            # safety
            if a_new[i_high] < -1e-14:
                continue
            # compute exact objective
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                a = a_new
                best_a = a_new.copy()
                best_val = val_new
                improved = True
                print(f"[MassTransfer it={it:06d}] new best {val_new:.6f}")
                break
        if not improved:
            # no improvement with any fraction – give up on this pair
            # try a random perturbation to escape possible plateau
            i = np.random.randint(n)
            j = np.random.randint(n)
            if i != j and a[i] > 1e-12:
                eps = a[i] * 0.05
                a[i] -= eps
                a[j] += eps
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ 1000 s.  Returns the best sequence (list of floats).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a rich pool of diverse initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0, 4.0):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families
    for sigma in (0.04, 0.1, 0.2, 0.4, 0.8):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law families (negative exponents give decreasing tails)
    for expo in (-3.0, -2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))
    pool.append(_make_cosine(n))

    # Sparse spike patterns
    for k in (32, 64, 128, 256):
        pool.append(_make_spikes(n, k))

    # Beta distributions (including arcsine shape)
    for alpha, beta in [(0.5, 0.5), (0.8, 0.8), (1.0, 1.0),
                        (2.0, 2.0), (3.0, 3.0), (5.0, 5.0), (10.0, 10.0)]:
        pool.append(_make_beta(n, alpha, beta))

    # A few completely random Dirichlet draws
    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)  # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 12   # number of starts we will fully optimise

    # ------------------------------------------------------------------
    # Phase I – deep optimisation of a few promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate a slice of the remaining budget (≈30 s for Adam)
        sub_dead = min(deadline, time.time() + 27.0)

        # 1) Adam on smooth‑max surrogate
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient phase (if time permits)
        if time.time() < deadline - 8:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.7,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 12:
        print("[Fine] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=250_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase III – mass‑transfer fine‑grained local search
    if best_seq is not None and time.time() < deadline - 6:
        print("[MassTransfer] Fine‑grained refinement")
        a_mt, val_mt = _mass_transfer_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_mt < best_val:
            best_val = val_mt
            best_seq = a_mt.copy()
            print(f"[MassTransfer] improved best to {val_mt:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – simulated annealing (short run)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase V – optional up‑sampling to a finer grid (if time permits)
    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Upsample] Trying finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=150_000,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Upsample] refined best {val_big:.6f}")
        except Exception as e:
            print(f"[Upsample] exception: {e}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final exact evaluation (using the provided evaluate_sequence)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final exact score {final_score:.6f} (length {len(best_seq)})")
    except Exception as e:
        print(f"Final evaluation failed: {e}")

    return best_seq.tolist()
