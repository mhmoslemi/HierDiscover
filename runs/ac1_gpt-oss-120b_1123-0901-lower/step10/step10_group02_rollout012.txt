def propose_candidate(seed=0, budget_s=1000):
    """
    Hybrid optimizer: fast Adam warmup, long exponentiated‑gradient subgradient,
    followed by a pairwise‑mass‑balancing local search and cyclic‑shift polishing.
    """
    import time, numpy as np

    deadline = time.time() + budget_s - 0.3
    np.random.seed(seed)

    n = 4096
    conv_fft_len = _next_pow2(2*n-1)
    corr_fft_len = _next_pow2(3*n-2)

    # warm start
    warm = globals().get('height_sequence_1')
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n-a.size), mode='constant')
    else:
        a = np.full(n, 1.0/n, dtype=np.float64)

    # make simplex
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # share conv lengths computed see above functions.

    # Phase 1: short Adam (5% time)
    phase_dead = min(deadline, time.time() + 0.05*(deadline - time.time()))
    a_adam, val_adam = _adam_optimize(best_a, n, conv_fft_len, corr_fft_len,
                                      phase_dead, max_iters=200_000)
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Phase 1] Adam improved to {val_adam:.6f}")

    # Phase 2: long subgradient (80% time)
    phase_dead = min(deadline, time.time() + 0.80*(deadline - time.time()))
    a_sub, val_sub = _subgradient_optimize_expgrad(best_a, n, conv_fft_len,
                                                  corr_fft_len, phase_dead,
                                                  max_iters=20_000_000,
                                                  eta0=1.5)
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Phase 2] Subgrad improved to {val_sub:.6f}")

    # Phase 3: pairwise balancing local search (remaining time)
    remaining = deadline - time.time()
    if remaining > 5.0:
        a_bal, val_bal = _pairwise_balance_opt(best_a, n, conv_fft_len,
                                               corr_fft_len, deadline)
        if val_bal < best_val:
            best_val, best_a = val_bal, a_bal.copy()
            print(f"[Phase 3] Pairwise balance improved to {val_bal:.6f}")

    # final cyclic shift search
    shifts = [n//8, n//4, 3*n//8, n//2, 5*n//8, 3*n//4, 7*n//8]

def _pairwise_balance_opt(a0: np.ndarray, n: int, conv_fft_len: int,
                          corr_fft_len: int, deadline: float) -> tuple[np.ndarray, float]:
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Bal it={it}] new best {val:.6f}")

        # active indices where conv is max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if not active.any():
            # fallback to single max
            active = np.array([conv.argmax()], dtype=int)
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # choose indices

        if grad[i_max] <= grad[i_min] + 1e-16:

    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val

"""
Hybrid optimizer for the step‑function bound.

The routine combines three complementary phases:

1.  A short Adam run on a smooth‑max surrogate (≈ 5 % of the time budget).
    This quickly moves the candidate into a basin of attraction.

2.  A long exponentiated‑gradient sub‑gradient phase (≈ 80 % of the budget).
    The exact max‑convolution is used, with an adaptive step‑size η_t.

3.  A cheap pairwise‑mass‑balancing local search (≈ the remaining time).
    Using the sub‑gradient of the current max‑convolution we repeatedly move a
    small amount of probability mass from the most harmful index to the least
    harmful one.  This often squeezes a few extra points out of the plateau.

Finally a short set of cyclic‑shifts is tried and the best vector is returned
as a plain Python list.  The whole routine obeys the supplied time budget
(~1000 s) and always produces a non‑negative sequence normalised to Σa_i = 1.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.5,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _pairwise_balance_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simple local search that repeatedly moves a small amount of mass from the
    index with the largest sub‑gradient to the index with the smallest one.
    It works in the exact max‑convolution landscape and often yields a tiny
    extra improvement after the long sub‑gradient phase.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Bal it={it:06d}] new best {val:.6f}")

        # identify active constraints (indices where max is attained)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        else:
            w[conv.argmax()] = 1.0  # fallback, should rarely happen

        # sub‑gradient of the max‑convolution w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        i_max = int(np.argmax(grad))
        i_min = int(np.argmin(grad))

        # stop if gradient is flat (unlikely but safe)
        if grad[i_max] <= grad[i_min] + 1e-16:
            break

        # amount of probability mass to move (20 % of the donor, capped)
        eps = min(a[i_max] * 0.20, 5e-5)
        if eps <= 0:
            break

        a[i_max] -= eps
        a[i_min] += eps

        # occasional re‑projection to the simplex to counteract drift
        if it % 1000 == 0:
            a = _project_simplex(a)

        it += 1

    # final projection and evaluation
    a = _project_simplex(a)
    final_val = _objective(a, n, conv_fft_len)
    if final_val < best_val:
        best_val = final_val
        best_a = a.copy()
    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a plain Python list) found.
    """
    np.random.seed(seed)
    # safety margin to avoid overshoot
    deadline = time.time() + budget_s - 0.3

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a = a / s                     # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # --------------------------------------------------------------
    # ---- Phase 1 : very short Adam (≈5 % of remaining budget) ----
    phase_dead = min(deadline, time.time() + 0.05 * (deadline - time.time()))
    a_adam, val_adam = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, phase_dead, max_iters=200_000
    )
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Phase 1] Adam improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # ---- Phase 2 : long exponentiated‑gradient sub‑gradient (≈80 %) ----
    phase_dead = min(deadline, time.time() + 0.80 * (deadline - time.time()))
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_a, n, conv_fft_len, corr_fft_len, phase_dead,
        max_iters=20_000_000, eta0=1.5,
    )
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Phase 2] Subgrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # ---- Phase 3 : cheap pairwise‑balance local search (rest of time) ----
    if time.time() < deadline:
        a_bal, val_bal = _pairwise_balance_opt(best_a, n, conv_fft_len,
                                               corr_fft_len, deadline)
        if val_bal < best_val:
            best_val, best_a = val_bal, a_bal.copy()
            print(f"[Phase 3] Pairwise balance improved to {val_bal:.6f}")

    # --------------------------------------------------------------
    # Final cheap cyclic‑shift local search
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2,
              5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Clip to allowed range and final sanity check
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # safety net
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # Evaluate with the official function (just for reporting)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
