"""
Propose a low‑score step‑function.

The algorithm consists of four phases:

1. **Broad parametric sweep** – many families (exponential, stretched,
   power‑law, Gamma, Beta, double‑Gaussian, left‑hand monotone, random spikes)
   are sampled on a coarse grid.  The best global candidate is kept, and for
   each family we also remember its own best parameters.

2. **Local refinement of the most promising families** – the few families
   with the lowest values are optimised in continuous (log) parameter space
   using a cheap Nelder‑Mead search (SciPy).  This discovers a finer‑tuned
   shape that often beats the coarse grid.

3. **Smooth‑max Adam optimisation** – starting from the best sequence found
   so far, a soft‑max (log‑sum‑exp) surrogate of the max‑convolution is
   minimised with Adam for the remaining time budget.  The temperature (λ)
   is increased slowly so the surrogate approaches the true max.

4. **Exact‑max exponentiated‑gradient sub‑gradient descent** – the final
   stage runs a long exponentiated‑gradient (mirror‑descent) on the true
   max‑convolution objective until the deadline.

All phases respect the total 1000 s budget, always return a non‑negative
list whose sum is positive, and finally report the public `evaluate_sequence`
score.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (on the simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.02
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # Temperature schedule
        lam = min(lam * lam_factor, 1e9)

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Step size ηₜ = η₀/√(t+1).  Runs until the deadline.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # Active set (indices attaining max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # Renormalise safely
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.15
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper for local parameter optimisation --------------------------------
class _TimeOutException(Exception):
    pass


def _make_candidate_from_log(name: str, x: np.ndarray, n: int) -> np.ndarray:
    """Decode log‑parameter vector `x` into a concrete candidate."""
    if name == "left_stretched":
        alpha = np.exp(x[0])
        beta = np.exp(x[1])
        return _candidate_left_stretched(alpha, beta, n)
    if name == "stretched":
        alpha = np.exp(x[0])
        beta = np.exp(x[1])
        return _candidate_stretched(alpha, beta, n)
    if name == "left_exponential":
        alpha = np.exp(x[0])
        return _candidate_left_exponential(alpha, n)
    if name == "exponential":
        alpha = np.exp(x[0])
        return _candidate_exponential(alpha, n)
    if name == "left_power":
        p = x[0]
        return _candidate_left_power(p, n)
    if name == "power":
        p = x[0]
        return _candidate_power(p, n)
    if name == "gamma":
        shape = np.exp(x[0])
        scale = np.exp(x[1])
        return _candidate_gamma(shape, scale, n)
    if name == "beta":
        a_par = np.exp(x[0])
        b_par = np.exp(x[1])
        return _candidate_beta(a_par, b_par, n)
    if name == "double_gaussian":
        sig = np.exp(x[0])
        return _candidate_double_gaussian(sig, n)
    raise ValueError(f"Unknown family {name!r}")


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")
    # per‑family best parameters (value, param‑tuple)
    best_by_family = {}

    # ------------------------------------------------------------------
    # Helper to update globals -------------------------------------------------
    def _update(name: str, a: np.ndarray, val: float, params: tuple | None = None):
        nonlocal best_seq, best_val, best_by_family
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Global] new best {val:.6f} from {name}")
        if params is not None:
            rec = best_by_family.get(name)
            if rec is None or val < rec[0]:
                best_by_family[name] = (val, params)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (coarse) ------------------------------------
    # ---- symmetric families ------------------------------------------------
    for p in np.linspace(-6.0, -0.1, 30):
        a = _candidate_power(p, n)
        val = _objective(a, n)
        _update("power", a, val, (p,))

    for alpha in np.logspace(-4, 2.5, 35):
        a = _candidate_exponential(alpha, n)
        val = _objective(a, n)
        _update("exponential", a, val, (alpha,))

    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            val = _objective(a, n)
            _update("stretched", a, val, (alpha, beta))

    # ---- monotone families -------------------------------------------------
    for alpha in np.logspace(-4, 2.0, 30):
        a = _candidate_left_exponential(alpha, n)
        val = _objective(a, n)
        _update("left_exponential", a, val, (alpha,))

    for p in np.linspace(-6.0, 2.0, 30):
        a = _candidate_left_power(p, n)
        val = _objective(a, n)
        _update("left_power", a, val, (p,))

    betas = np.linspace(0.5, 4.0, 12)
    for beta in betas:
        for alpha in np.logspace(-4, 1.5, 25):
            a = _candidate_left_stretched(alpha, beta, n)
            val = _objective(a, n)
            _update("left_stretched", a, val, (alpha, beta))

    # ---- Gamma‑like --------------------------------------------------------
    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            a = _candidate_gamma(shape, scale, n)
            val = _objective(a, n)
            _update("gamma", a, val, (shape, scale))

    # ---- Beta‑distribution -------------------------------------------------
    a_pars = np.logspace(-2, 0.5, 12)
    b_pars = np.logspace(0, 1.0, 12)
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            val = _objective(a, n)
            _update("beta", a, val, (a_par, b_par))

    # ---- double Gaussian ---------------------------------------------------
    for sig in np.logspace(-3, -0.2, 15):
        a = _candidate_double_gaussian(sig, n)
        val = _objective(a, n)
        _update("double_gaussian", a, val, (sig,))

    # ---- random spike patterns ---------------------------------------------
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(3):
            a = _candidate_spike(k, n, random=True)
            val = _objective(a, n)
            _update(f"spike_{k}", a, val, (k,))

    # ---- warm‑start from previously discovered sequence --------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            val = _objective(w, n)
            _update("warm_start", w, val, None)

    print(f"[Sweep] best after coarse exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Local refinement of the most promising families --------------------
    #   (Nelder‑Mead on log‑parameters)
    remaining = deadline - time.time()
    if remaining > 30:
        # pick up to 5 families with lowest values
        sorted_fams = sorted(best_by_family.items(), key=lambda kv: kv[1][0])
        top_fams = [name for name, _ in sorted_fams[:5]]
        print("[Refine] local optimisation of families:", top_fams)

        for fam in top_fams:
            val0, params0 = best_by_family[fam]
            # encode params into log‑space where appropriate
            init = []
            for p in params0:
                # parameters that are strictly positive will be optimised in log‑space
                if p > 0 and fam not in ("left_power", "power"):
                    init.append(np.log(p))
                else:
                    init.append(p)
            init = np.array(init, dtype=np.float64)

            # allocate a sub‑deadline for this family (max 20 % of remaining)
            fam_deadline = time.time() + min(25.0, remaining * 0.2)

            def _obj(x):
                if time.time() > fam_deadline:
                    raise _TimeOutException()
                cand = _make_candidate_from_log(fam, x, n)
                return _objective(cand, n)

            try:
                res = minimize(
                    _obj,
                    init,
                    method="Nelder-Mead",
                    options={"maxfev": 800, "fatol": 1e-9, "xatol": 1e-9, "disp": False},
                )
                if not res.success:
                    continue
                cand = _make_candidate_from_log(fam, res.x, n)
                val = _objective(cand, n)
                _update(f"{fam}_refined", cand, val, None)
                print(f"[Refine {fam}] improved to {val:.6f}")
            except _TimeOutException:
                print(f"[Refine {fam}] timed out")
                continue
            remaining = deadline - time.time()
            if remaining < 5:
                break

    # ------------------------------------------------------------------
    # 3) Smooth‑max Adam refinement (more iterations) -----------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 20:
        print("[Phase‑Adam] extended Adam optimisation")
        adam_deadline = time.time() + remaining * 0.35  # ~35 % of time left
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=5_000_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Long exact‑max sub‑gradient descent --------------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 10:
        print("[Phase‑Subgrad] long sub‑gradient optimisation")
        subgrad_deadline = time.time() + remaining * 0.5   # use ~50 % of what is left
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            subgrad_deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tuning ---------------------------
    remaining = deadline - time.time()
    if best_seq is not None and remaining > 2:
        print("[Phase‑Anneal] final simulated‑annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks --------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # ------------------------------------------------------------------
    # Evaluate with the public scorer (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
