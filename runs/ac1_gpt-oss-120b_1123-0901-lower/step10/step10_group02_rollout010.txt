"""
Hybrid optimizer for the step‑function bound with extra random‑restart
and local‑search phases.

The routine:
1. Starts from the previous best (global ``height_sequence_1``) if it exists,
   otherwise from a uniform simplex vector.
2. A short Adam phase on a smooth‑max surrogate gives a quick improvement.
3. The bulk of the budget is spent on an exponentiated‑gradient sub‑gradient
   optimisation of the exact max‑convolution.
4. Several cheap random‑restart sub‑gradient runs are launched to escape
   possible local minima.
5. A simple two‑index perturbation hill‑climb (local search) uses the
   remaining time.
6. Finally a few cyclic shifts are tested and the best sequence is returned.
All steps respect the supplied time budget (≈ 1000 s) and always return a
non‑negative simplex‑normalised sequence.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real; w has length 2·len(a)‑1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) for a on the simplex."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
        a0: np.ndarray,
        n: int,
        conv_fft_len: int,
        corr_fft_len: int,
        deadline: float,
        max_iters: int = 10_000_000,
        eta0: float = 1.2) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _local_search(a0: np.ndarray,
                  n: int,
                  conv_fft_len: int,
                  deadline: float,
                  max_trials: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly pick two indices and move a random
    fraction of mass from one to the other.  Accept only improvements.
    """
    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    it = 0
    while time.time() < deadline and it < max_trials:
        i, j = np.random.randint(0, n, size=2)
        if i == j:
            continue
        # transfer a random fraction (up to 30 %) of the mass at i to j
        eps = np.random.rand() * 0.3 * best_a[i]
        if eps <= 0:
            continue
        a_new = best_a.copy()
        a_new[i] -= eps
        a_new[j] += eps
        # evaluate
        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_new
        it += 1
    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget (≈ 1000 s) and always
    returns the best sequence found (as a plain Python list).
    """
    np.random.seed(seed)
    # safety margin to avoid overshoot
    deadline = time.time() + budget_s - 0.3

    # ------------------------------------------------------------------
    # Problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    s = a.sum()
    if s < 0.01:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s                                   # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage 1 – quick Adam (≈10 % of remaining budget)
    now = time.time()
    remaining = deadline - now
    stage1_dead = min(deadline, now + 0.10 * remaining)
    a_adam, val_adam = _adam_optimize(best_a, n, conv_fft_len,
                                      corr_fft_len, stage1_dead,
                                      max_iters=300_000)
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Stage 1] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Stage 2 – long sub‑gradient optimisation (≈70 % of remaining)
    now = time.time()
    remaining = deadline - now
    stage2_dead = min(deadline, now + 0.70 * remaining)
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_a, n, conv_fft_len, corr_fft_len,
        stage2_dead, max_iters=10_000_000, eta0=1.0)
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Stage 2] Subgrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – several cheap random‑restart sub‑gradient runs
    # (use whatever time is still left after stage 2)
    now = time.time()
    while time.time() < deadline - 5.0:            # keep ~5 s for final phases
        sub_dead = time.time() + min(8.0, deadline - time.time() - 5.0)
        a_start = np.random.dirichlet(np.ones(n))
        a_r, v_r = _subgradient_optimize_expgrad(
            a_start, n, conv_fft_len, corr_fft_len,
            sub_dead, max_iters=2_000_000, eta0=1.2)
        if v_r < best_val:
            best_val, best_a = v_r, a_r.copy()
            print(f"[Random restart] improved to {v_r:.6f}")

    # ------------------------------------------------------------------
    # Stage 4 – simple local search (use whatever time remains)
    now = time.time()
    if now < deadline:
        a_local, v_local = _local_search(best_a, n, conv_fft_len, deadline)
        if v_local < best_val:
            best_val, best_a = v_local, a_local.copy()
            print(f"[Local search] improved to {v_local:.6f}")

    # ------------------------------------------------------------------
    # Stage 5 – cyclic‑shift mop‑up
    shifts = [n // 8, n // 4, n // 2, 3 * n // 4]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # safety
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # Report using the official evaluator (for logging only)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
