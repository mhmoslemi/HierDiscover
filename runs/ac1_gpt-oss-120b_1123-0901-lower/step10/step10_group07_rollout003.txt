def _de_optimize(obj_func, bounds, deadline, popsize=20, max_gen=200, F=0.8, CR=0.7, seed=0):
   rng = np.random.default_rng(seed)
   dim = len(bounds)
   # Initialise population
   pop = np.empty((popsize, dim))
   for i, (lo, hi) in enumerate(bounds):
       pop[:, i] = rng.uniform(lo, hi, size=popsize)
   # Evaluate
   pop_vals = np.empty(popsize)
   for i in range(popsize):
        pop_vals[i] = obj_func(pop[i])
   best_idx = np.argmin(pop_vals)
   best = pop[best_idx].copy()
   best_val = pop_vals[best_idx]
   gen = 0
   while time.time() < deadline and gen < max_gen:
       for i in range(popsize):
           # Mutation
           idxs = list(range(popsize))
           idxs.remove(i)
           r1, r2, r3 = rng.choice(idxs, 3, replace=False)
           mutant = pop[r1] + F * (pop[r2] - pop[r3])
           # Ensure bounds
           mutant = np.clip(mutant, [b[0] for b in bounds], [b[1] for b in bounds])
           # Crossover
           cross = rng.random(dim) < CR
           # Ensure at least one parameter from mutant
           if not np.any(cross):
               cross[rng.integers(dim)] = True
           trial = np.where(cross, mutant, pop[i])
           # Clip again
           trial = np.clip(trial, [b[0] for b in bounds], [b[1] for b in bounds])
           val = obj_func(trial)
           if val < pop_vals[i]:
               pop[i] = trial
               pop_vals[i] = val
               if val < best_val:
                   best_val = val
                   best = trial.copy()
                   # print progress
                   print(f"[DE gen={gen} ind={i}] new best {best_val:.6f}")
       gen += 1
   return best, best_val
```

"""
Propose a non‑negative step‑function (list of heights) that minimizes

    2·n·max(conv(a,a)) / (sum(a))**2

The algorithm follows several complementary stages while respecting a
global time budget (default 1000 s).

1. **Quick parametric sweep** – a coarse grid over several analytic
   families (exponential, power‑law, beta, stretched‑exponential and a
   double‑Gaussian) is evaluated.  The best shape is kept.

2. **Local refinement** – the best parameters of each family are
   refined with SciPy's bounded L‑BFGS‑B optimiser acting on the exact
   objective (no surrogate).

3. **Mixture optimisation** – a differential‑evolution (DE) search over
   a convex mixture of the four families (exponential, power‑law,
   beta, double‑Gaussian) explores a richer shape space.

4. **Adam polishing** – starting from the best candidate found so far,
   a mirror‑descent Adam optimiser minimises a smooth‑max surrogate of
   the max‑convolution while staying on the simplex via a soft‑max
   parametrisation.

5. **Exponentiated‑gradient sub‑gradient** – a long run of the exact‑max
   sub‑gradient method (mirror descent on the simplex) is performed.

6. **Simulated‑annealing fine‑tuning** – a short annealing phase makes
   tiny local adjustments.

Every stage checks the remaining time and aborts early if necessary,
always returning the best feasible sequence encountered.
"""

import time
import numpy as np
from math import log10, exp

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using real FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of the smoothed max.
    Returns a vector of length len(a) corresponding to the derivative of
    ``max(w·(a*a))`` w.r.t. ``a`` when ``w`` is the (normalised) active‑set
    indicator.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum a)² (no clipping).
    ``a`` is assumed to be non‑negative and already normalised
    (sum(a) may be any positive number – the ratio is scale‑invariant).
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Analytic families (all return a normalised non‑negative vector) ----

def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Differential Evolution (DE) over a mixture of families --------------

def _mix_shapes(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a shape as a convex mixture of the four analytic families.
    ``params`` layout (9‑dimensional):
        0‑3 : raw non‑negative mixture weights (will be normalised)
        4   : log10(alpha) for the exponential
        5   : power‑law exponent p
        6   : log10(a_par) for beta
        7   : log10(b_par) for beta
        8   : log10(sig) for the double‑Gaussian
    """
    # mixture weights (force non‑negative, then normalise)
    w_raw = np.maximum(params[0:4], 0.0)
    w_sum = w_raw.sum()
    if w_sum == 0.0:
        w = np.full(4, 0.25)
    else:
        w = w_raw / w_sum

    # internal parameters
    alpha = 10.0 ** params[4]
    p = params[5]
    a_par = 10.0 ** params[6]
    b_par = 10.0 ** params[7]
    sig = 10.0 ** params[8]

    # individual shapes (already normalised)
    s0 = _candidate_exponential(alpha, n)
    s1 = _candidate_power(p, n)
    s2 = _candidate_beta(a_par, b_par, n)
    s3 = _candidate_double_gaussian(sig, n)

    seq = w[0] * s0 + w[1] * s1 + w[2] * s2 + w[3] * s3
    # re‑normalise (tiny numerical drift)
    seq_sum = seq.sum()
    if seq_sum <= 0.0:
        seq = np.full(n, 1.0 / n)
    else:
        seq = seq / seq_sum
    return seq


def _de_optimize(obj_func, bounds, deadline,
                 popsize: int = 30, max_gen: int = 200,
                 F: float = 0.8, CR: float = 0.7, seed: int = 0):
    """
    Simple (DE/rand/1/bin) differential‑evolution optimiser.
    Stops when the deadline is reached or ``max_gen`` generations have
    been processed.
    Returns the best parameter vector and its objective value.
    """
    rng = np.random.default_rng(seed)
    dim = len(bounds)
    lo = np.array([b[0] for b in bounds])
    hi = np.array([b[1] for b in bounds])

    # initialise population uniformly inside the bounds
    pop = rng.uniform(lo, hi, size=(popsize, dim))
    pop_vals = np.empty(popsize, dtype=np.float64)
    for i in range(popsize):
        pop_vals[i] = obj_func(pop[i])

    best_idx = int(np.argmin(pop_vals))
    best = pop[best_idx].copy()
    best_val = pop_vals[best_idx]

    gen = 0
    while time.time() < deadline and gen < max_gen:
        for i in range(popsize):
            # --- mutation -------------------------------------------------
            idxs = list(range(popsize))
            idxs.remove(i)
            r1, r2, r3 = rng.choice(idxs, 3, replace=False)
            mutant = pop[r1] + F * (pop[r2] - pop[r3])
            mutant = np.clip(mutant, lo, hi)

            # --- crossover ------------------------------------------------
            cross = rng.random(dim) < CR
            if not np.any(cross):
                cross[rng.integers(dim)] = True
            trial = np.where(cross, mutant, pop[i])
            trial = np.clip(trial, lo, hi)

            # --- selection ------------------------------------------------
            val = obj_func(trial)
            if val < pop_vals[i]:
                pop[i] = trial
                pop_vals[i] = val
                if val < best_val:
                    best_val = val
                    best = trial.copy()
                    print(f"[DE gen={gen} ind={i}] new best {best_val:.6f}")
        gen += 1
    return best, best_val


# ----------------------------------------------------------------------
# Adam optimiser on the smooth‑max surrogate ---------------------------

def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 5_000_000) -> tuple[np.ndarray, float]:
    """
    Adam (mirror‑descent) on the log‑sum‑exp surrogate of ``max(conv)``,
    keeping the iterate on the probability simplex via a soft‑max
    parametrisation ``a = softmax(logits)``.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.08                               # a bit larger than before
    lam = 0.5                                # initial temperature for smooth‑max
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ---- soft‑max ----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- exact objective (for logging) ----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ---- smooth‑max weights (log‑sum‑exp) ----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w = w / w_sum

        # ---- gradient of surrogate ----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ---- chain rule through soft‑max (mirror‑descent) ----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update on logits ----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule (smooth‑max gets sharper) ----
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.94

        it += 1
    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient / mirror descent) -----

def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 4_000_000,
                                 eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient on the exact max‑convolution.
    Step size ηₜ = η₀ / √(t+1).  The procedure respects the deadline.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set where convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val

def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_moves: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Simple annealing that repeatedly moves a random fraction (≤30 %) of the
    mass from one index to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1
    return best_a, best_val

# Main entry point ------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # problem dimensions -------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # bookkeeping --------------------------------------------------------
    best_seq = None
    best_val = np.inf

    def _maybe_update(seq: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(seq, n)
        if val < best_val:
            best_val = val
            best_seq = seq.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 0) warm start from a previously saved global variable (if any) ----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    # ------------------------------------------------------------------
    # 1) quick parametric sweep (coarse grid) ------------------------
    # exponential
    for alpha in np.logspace(-5, 1.0, 80):
        _maybe_update(_candidate_exponential(alpha, n))
    # power‑law (negative exponents)
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n))
    # stretched exponential (two‑parameter)
    alphas = np.logspace(-4, 0.5, 24)
    betas = np.linspace(0.4, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))
    # beta distributions
    a_vals = [0.2, 0.5, 1.0, 2.0, 5.0, 8.0]
    b_vals = a_vals
    for a_par in a_vals:
        for b_par in b_vals:
            _maybe_update(_candidate_beta(a_par, b_par, n))
    # double Gaussian
    for sig in np.logspace(-3.0, -0.2, 25):
        _maybe_update(_candidate_double_gaussian(sig, n))

    print(f"[Sweep] best after coarse exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) local refinement of each family with SciPy (if available) -----
    try:
        from scipy.optimize import minimize

        # exponential refinement
        def _obj_exp(x):
            return _objective(_candidate_exponential(np.abs(x[0]), n), n)

        res = minimize(_obj_exp,
                       x0=np.array([1.0]),
                       bounds=[(1e-6, 10.0)],
                       method='L-BFGS-B')
        if res.success:
            seq = _candidate_exponential(abs(res.x[0]), n)
            _maybe_update(seq)

        # power‑law refinement
        def _obj_pow(x):
            return _objective(_candidate_power(x[0], n), n)

        res = minimize(_obj_pow,
                       x0=np.array([-1.0]),
                       bounds=[(-5.0, -0.001)],
                       method='L-BFGS-B')
        if res.success:
            seq = _candidate_power(res.x[0], n)
            _maybe_update(seq)

        # beta refinement
        def _obj_beta(x):
            return _objective(_candidate_beta(10.0 ** x[0], 10.0 ** x[1], n), n)

        res = minimize(_obj_beta,
                       x0=np.array([0.0, 0.0]),
                       bounds=[(-1.0, 1.0), (-1.0, 1.0)],
                       method='L-BFGS-B')
        if res.success:
            seq = _candidate_beta(10.0 ** res.x[0], 10.0 ** res.x[1], n)
            _maybe_update(seq)

        # stretched‑exponential refinement (2‑D)
        def _obj_stretched(x):
            return _objective(_candidate_stretched(10.0 ** x[0], x[1], n), n)

        res = minimize(_obj_stretched,
                       x0=np.array([0.0, 1.0]),
                       bounds=[(-4.0, 0.5), (0.2, 3.0)],
                       method='L-BFGS-B')
        if res.success:
            alpha = 10.0 ** res.x[0]
            beta = res.x[1]
            seq = _candidate_stretched(alpha, beta, n)
            _maybe_update(seq)

        # double‑Gaussian refinement
        def _obj_gauss(x):
            return _objective(_candidate_double_gaussian(10.0 ** x[0], n), n)

        res = minimize(_obj_gauss,
                       x0=np.array([-1.0]),
                       bounds=[(-3.0, -0.2)],
                       method='L-BFGS-B')
        if res.success:
            seq = _candidate_double_gaussian(10.0 ** res.x[0], n)
            _maybe_update(seq)

    except Exception as e:
        # SciPy not available – simply continue
        print(f"[Refine] SciPy not available or failed: {e}")

    print(f"[Refine] best after local optimiser: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Differential‑Evolution over a mixture of families -------------
    time_left = deadline - time.time()
    if time_left > 30.0:
        # bounds for the 9‑dimensional DE parameter vector
        de_bounds = [
            (0.0, 1.0), (0.0, 1.0), (0.0, 1.0), (0.0, 1.0),     # mixture weights
            (-6.0, 1.0),                                        # log10(alpha)  exp
            (-5.0, -0.001),                                     # p  power‑law
            (-1.0, 1.0),                                        # log10(a_par)  beta
            (-1.0, 1.0),                                        # log10(b_par)  beta
            (-3.0, -0.2)                                        # log10(sig)  double Gaussian
        ]

        def _de_obj(par):
            seq = _mix_shapes(par, n)
            return _objective(seq, n)

        de_deadline = deadline - 5.0
        best_params, best_de_val = _de_optimize(_de_obj, de_bounds,
                                                de_deadline,
                                                popsize=30,
                                                max_gen=250,
                                                seed=seed)
        seq_de = _mix_shapes(best_params, n)
        _maybe_update(seq_de)
        print(f"[DE] best after mixture optimisation: {best_de_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Adam polishing (smooth‑max surrogate) -------------------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        print("[Phase‑Adam] long Adam polishing")
        adam_deadline = time.time() + min(180.0, time_left - 5.0)
        a_adam, val_adam = _adam_optimize(best_seq, n,
                                          conv_fft_len, corr_fft_len,
                                          adam_deadline,
                                          max_iters=5_000_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 5) Exponentiated‑gradient sub‑gradient refinement -----------------
    time_left = deadline - time.time()
    if time_left > 30.0 and best_seq is not None:
        print("[Phase‑Subgrad] final long sub‑gradient run")
        sub_deadline = time.time() + min(300.0, time_left - 5.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(best_seq, n,
                                                       conv_fft_len,
                                                       corr_fft_len,
                                                       sub_deadline,
                                                       max_iters=10_000_000,
                                                       eta0=0.9)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 6) Simulated‑annealing fine‑tuning (short) ------------------------
    time_left = deadline - time.time()
    if time_left > 5.0 and best_seq is not None:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(best_seq, n,
                                          conv_fft_len,
                                          deadline,
                                          max_moves=250_000)
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # final sanity checks ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final evaluation via the provided wrapper (may apply clipping etc.)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (evaluator missing).")

    return best_seq.tolist()
