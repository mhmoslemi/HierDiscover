def anneal_seq(a, max_moves, deadline):
    best_a = a.copy()
    best_val_a = evaluate(a)
    current_a = a.copy()
    for mv in range(max_moves):
        if time.time() > deadline - 0.1:
            break
        # pick i with non-zero mass
        i = np.random.choice(n, p=current_a/current_a.sum())  # prefer mass
        # pick j uniformly maybe
        j = np.random.randint(n)
        if i == j:
            continue
        # transfer fraction up to 30% of a[i]
        delta = current_a[i] * np.random.rand() * 0.3
        a2 = current_a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = evaluate(a2)
        if val2 < best_val_a:
            best_val_a = val2
            best_a = a2.copy()
            current_a = a2
        else:
            # Metropolis acceptance
            T = max(1e-5, 0.05 * (1.0 - mv/max_moves))
            if np.random.rand() < np.exp(-(val2 - evaluate(current_a)) / T):
                current_a = a2
    return best_a, best_val_a

    corr_fft_len = _next_pow2(3*n-2)

    def normalize(a):
        a = np.clip(a, 0.0, 1000.0)
        s = a.sum()
        if s < 0.01:
            return None
        return a / s

    def obj_val(a):
        conv = _full_convolution(a, conv_fft_len)
        return 2.0 * n * conv.max()

    # collect candidates
    candidates = []

    # warm start

        w = np.asarray(globals()['height_sequence_1'], dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        else:
            w = np.pad(w, (0, n-w.shape[0]))
        w_norm = normalize(w)
        if w_norm is not None:
            candidates.append(w_norm)

    # param families
    idx = np.arange(1, n+1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # power-law: fine grid
    for alpha in np.linspace(-6.0, -0.05, 200):
        shape = prod ** alpha
        shape_norm = normalize(shape)
        if shape_norm is not None:
            candidates.append(shape_norm)

"""
Hybrid parametric‑sweep + exponentiated‑gradient + annealing optimizer.

1.  Generate a large, diverse pool of candidate step‑functions.
    Families include power‑law tails, Beta‑shapes, exponential tails,
    mixtures of power‑law and uniform, Dirichlet draws and random block‑spikes.
    All candidates are normalised to sum = 1 (the objective is scale‑invariant).

2.  Evaluate every candidate with a fast FFT based convolution and keep the
    best few (default 5) for local refinement.

3.  Refine each of the top candidates with an exponentiated‑gradient (EG)
    descent on the simplex.  The sub‑gradient of the max‑convolution is
    obtained by a cross‑correlation with a uniform weight on the active
    (max‑attaining) convolution indices.  An adaptive step‑size
    (∝ 1/√t) and a simple back‑tracking line‑search guarantee monotone
    improvement.

4.  A short simulated‑annealing polish moves a random fraction of mass
    between two coordinates, always accepting downhill moves and
    occasionally uphill moves according to a temperature schedule.

The routine respects the supplied time budget (≈ 1000 s) and always returns
the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (~1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # Problem size – we keep the size used in the reference solution.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Helper utilities -------------------------------------------------
    def _norm(arr: np.ndarray) -> np.ndarray | None:
        """Clip, enforce non‑negativity, normalise to sum = 1."""
        arr = np.clip(arr, 0.0, 1000.0)
        s = arr.sum()
        if s < 0.01 or np.isnan(s):
            return None
        return arr / s

    def _objective(arr: np.ndarray) -> float:
        """Exact objective 2·n·max(a*a) (sum is 1 after normalisation)."""
        conv = _full_convolution(arr, conv_fft_len)
        return 2.0 * n * conv.max()

    # ------------------------------------------------------------------
    # 1️⃣ Build a diverse pool of starter vectors ----------------------
    pool: list[np.ndarray] = []

    if "height_sequence_1" in globals():

        else:

        w = _norm(w)

    idx = np.arange(1, n + 1, dtype=np.float64)

        a = start_vec.copy()

        it = 0

            active = np.isclose(conv, max_c, atol=1e-14)

            w[active] = 1.0 / active.sum()

            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
