        if time.time() > stage1_end:
            break
        # Random init Dirichlet
        a0 = np.random.dirichlet(np.ones(n))

            print(f"[Restart {restart}] new best {val_sub:.6f}")

        # Optionally small Adam
        adam_dead = min(deadline, time.time() + (stage1_end - time.time())/ (5 - restart))
        a_adam, val_adam = _adam_optimize(
            a_sub, n, conv_len, corr_len, adam_dead, max_iters=200_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Restart {restart} Adam] new best {val_adam:.6f}")

    # Stage2: long subgradient starting from best_a
    remaining = deadline - time.time()
    if remaining > 0:
        sub_dead = deadline
        a_refined, val_refined = _subgradient_optimize_expgrad(
            best_a, n, conv_len, corr_len, sub_dead, max_iters=30_000_000, eta0=1.2

            best_val, best_a = val_refined, a_refined.copy()
            print(f"[Refine] improved to {val_refined:.6f}")

    # Stage3: local pairwise swaps
    remaining = deadline - time.time()
    if remaining > 0:
        # Hill climbing moves
        a = best_a.copy()
        conv_len_full = conv_len
        # evaluate conv once per move may be heavy; we compute objective each step

                a = a_new

"""
Hybrid optimizer with an additional “flatten‑conv” stage.

The routine:
1. Loads a warm start (global `height_sequence_1`) or starts from a uniform
   distribution of length n = 4096 and normalises it to the probability simplex.
2. **Flatten‑conv phase** – Adam optimisation of the *variance* of the
   self‑convolution (i.e. minimises  Σ (a*a)[k]²).  Flattening the convolution
   makes the maximal entry smaller and provides a better starting point for the
   original max‑convolution problem.
3. **Long sub‑gradient phase** – exact max‑convolution sub‑gradient descent
   (exponentiated‑gradient) on the original objective.
4. **Refinement** – a short Adam pass on the original max‑objective.
5. **Local hill‑climbing** – random pairwise mass transfers for any remaining
   time.
6. A few cyclic shifts are tested and the best sequence is returned.

All steps respect the overall time budget (≈ 1000 s) and always return a
non‑negative sequence.  The implementation is self‑contained and uses only NumPy.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the *smooth‑max* surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _adam_optimize_L(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser that minimises the L2‑norm of the self‑convolution
    (i.e. Σ (a*a)[k]²).  Flattening the convolution helps to lower the max‑value."""
    a = a0.copy()
    a = a / a.sum()
    m = np.zeros_like(a)
    v = np.zeros_like(a)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)          # c = a*a
        # Gradient of Σ c_k² is 4 * (c ★ a)
        grad = 4.0 * _cross_correlation(conv, a, corr_fft_len)

        # Adam update on the raw vector (no softmax)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        a -= lr * m_hat / (np.sqrt(v_hat) + eps_adam)

        # projection onto the simplex
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 50000 == 0:
                print(f"[Adam‑L it={it:06d}] new best {val:.6f}")

        if it % 50000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # safety margin to avoid overshoot
    deadline = time.time() + budget_s - 0.3

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size), mode="constant")
    else:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    a = np.maximum(a, 0.0)
    if a.sum() < 0.01:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()                     # normalise to the simplex

    best_a = a.copy()
    best_val = _objective(best_a, n, conv_fft_len)
    print(f"[Init] objective {best_val:.6f}")

    # --------------------------------------------------------------
    # Stage 1 – flatten‑conv (Adam on Σ(conv²))
    stage1_dead = min(deadline, time.time() + 0.10 * (deadline - time.time()))
    a_flat, val_flat = _adam_optimize_L(
        best_a, n, conv_fft_len, corr_fft_len, stage1_dead, max_iters=300_000
    )
    if val_flat < best_val:
        best_val, best_a = val_flat, a_flat.copy()
        print(f"[Stage 1] Flatten‑Adam improved to {val_flat:.6f}")

    # --------------------------------------------------------------
    # Stage 2 – long exact‑max sub‑gradient (≈65 % of budget)
    stage2_dead = min(deadline, time.time() + 0.65 * (deadline - time.time()))
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_a, n, conv_fft_len, corr_fft_len, stage2_dead,
        max_iters=8_000_000, eta0=1.2,
    )
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Stage 2] Subgrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Stage 3 – short Adam on the *max* objective (≈15 % of budget)
    stage3_dead = min(deadline, time.time() + 0.15 * (deadline - time.time()))
    a_adam2, val_adam2 = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, stage3_dead, max_iters=300_000
    )
    if val_adam2 < best_val:
        best_val, best_a = val_adam2, a_adam2.copy()
        print(f"[Stage 3] Adam on max improved to {val_adam2:.6f}")

    # --------------------------------------------------------------
    # Stage 4 – cheap local hill‑climbing with random pairwise transfers
    remaining = deadline - time.time()
    if remaining > 0:
        a = best_a.copy()
        it = 0
        while time.time() < deadline:
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue
            # move a small random fraction of mass from i to j
            eps = np.random.rand() * 0.05          # up to 5 %
            delta = eps * a[i]
            if delta <= 0:
                continue
            a_new = a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # safety check
            if a_new[i] < 0:
                continue
            val_new = _objective(a_new, n, conv_fft_len)
            if val_new < best_val:
                best_val = val_new
                best_a = a_new.copy()
                a = a_new
                if it % 1000 == 0:
                    print(f"[Hill it={it}] new best {val_new:.6f}")
            it += 1

    # --------------------------------------------------------------
    # Final cheap cyclic‑shift local search
    shifts = [n // 8, n // 4, 3 * n // 8, n // 2, 5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # --------------------------------------------------------------
    # Clip to allowed range and final sanity check
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                     # should never happen
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    # Final evaluation (for logging only)
    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")

    return best_a.tolist()
