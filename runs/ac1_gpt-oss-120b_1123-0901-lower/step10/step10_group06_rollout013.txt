"""
Propose a non‑negative step‑function (list of heights) minimizing
evaluate_sequence().

Algorithm
---------
1.  Wide parametric sweep – generate many symmetric/monotone families
    (power‑law, exponential, stretched‑exponential, beta, double‑Gaussian,
    random spikes, Dirichlet draws).  All candidates are normalised to sum=1.
    The best K=8 candidates are kept (elite list).

2.  Short Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    Each elite start runs a few‑hundred‑thousand Adam steps; the best
    result becomes the new current solution.

3.  Fast hill‑climbing – exact convolution is maintained and updated
    incrementally when a small amount of mass ε is moved from a source
    index i to a target index j:
        conv ← conv + 2·ε·(a shifted to j) – 2·ε·(a shifted to i)
        a[i] -= ε ; a[j] += ε
    The source index is usually taken from the pair that contributes the
    most to the current maximal convolution value; the target is chosen
    uniformly.  Moves that lower the maximal convolution are always kept,
    occasional uphill moves are accepted with a tiny probability.  The
    loop runs until the time budget expires (or the target bound 1.503 is
    reached).

4.  The best sequence found is clipped to the evaluator limits and
    returned.  All steps respect the global 1000 s budget.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimisation on the smooth‑max surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.04                      # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam] new best {val:.6f} it={it}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Fast hill‑climbing on the exact maximum
# ----------------------------------------------------------------------
def _hillclimb(a_start: np.ndarray, conv_start: np.ndarray,
               n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Repeatedly move a small amount of mass from the index that most
    contributes to the current maximal convolution value to a random
    location, accepting only moves that lower the peak (with a tiny
    acceptance probability for uphill steps).  The convolution array is
    updated incrementally using the exact linear rule:
        conv ← conv + 2·ε·(a shifted to j) – 2·ε·(a shifted to i)
    """
    a = a_start.copy()
    conv = conv_start.copy()
    best_a = a.copy()
    best_val = 2.0 * n * conv.max()

    # small probability for occasional uphill moves
    uphill_prob = 0.001

    while time.time() < deadline:
        # -------- pick source index ---------------------------------
        # 70 % of the time pick the element that contributes most to the
        # current peak; otherwise pick a random index weighted by mass.
        if np.random.rand() < 0.7:
            k_max = int(conv.argmax())
            i_lo = max(0, k_max - (n - 1))
            i_hi = min(n - 1, k_max)
            idxs = np.arange(i_lo, i_hi + 1, dtype=np.int64)
            partners = k_max - idxs
            products = a[idxs] * a[partners]
            i = int(idxs[products.argmax()])
        else:
            if a.sum() == 0.0:
                break
            i = int(np.random.choice(n, p=a / a.sum()))
        if a[i] <= 1e-16:
            continue

        # -------- pick target index ---------------------------------
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1   # ensure i != j

        # -------- choose transfer amount -----------------------------
        # transfer up to 10 % of the source mass (but never more than half)
        eps = a[i] * (0.05 + 0.05 * np.random.rand())
        eps = min(eps, a[i] * 0.5)
        if eps < 1e-12:
            continue

        # -------- compute candidate convolution ----------------------
        conv_cand = conv.copy()
        conv_cand[j:j + n] += 2.0 * eps * a
        conv_cand[i:i + n] -= 2.0 * eps * a
        new_max = conv_cand.max()
        cur_max = conv.max()

        # -------- acceptance -----------------------------------------
        accept = False
        if new_max < cur_max - 1e-14:
            accept = True
        elif np.random.rand() < uphill_prob:
            accept = True

        if accept:
            # commit move
            conv = conv_cand
            a[i] -= eps
            a[j] += eps

            val = 2.0 * n * new_max
            if val < best_val:
                best_val = val
                best_a = a.copy()
                # early exit if we reached the requested bound
                if best_val < 1.503:
                    break

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper to keep a small set of elite candidates
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # tiny safety margin

    # ------------------------------------------------------------------
    # Length of the vector – power of two for fast FFTs.
    # Larger n gives a finer search space; 8192 is a good compromise.
    # ------------------------------------------------------------------
    n = 8192
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Parametric sweep – keep a few elites
    # ------------------------------------------------------------------
    elite = []  # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Symmetric power‑law (negative exponents concentrate at ends)
    for p in np.linspace(-5.0, -0.05, 30):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # Symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # Stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # Beta densities – broad shape exploration
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_par in a_pars:
        for b_par in a_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 20):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # Monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 20):
        a = _candidate_left_exponential(alpha, n)
        _maybe_update(a)

    # Monotone left‑power (allow negatives)
    for p in np.linspace(-4.0, 2.0, 28):
        a = _candidate_left_power(p, n)
        _maybe_update(a)

    # Inverse‑power (edges heavy)
    for p in np.linspace(-4.5, -0.1, 25):
        a = _candidate_inverse_power(p, n)
        _maybe_update(a)

    # Random spikes with several sparsities
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            a = _candidate_spike(k, n, random=True)
            _maybe_update(a)

    # Dirichlet draws (various concentration parameters)
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for c in concs:
        a = np.random.dirichlet(np.full(n, c))
        _maybe_update(a)

    # Warm start from a previously stored global (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # ------------------------------------------------------------------
    # 2) Short Adam optimisation on the smooth‑max surrogate
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(15.0, 0.12 * remaining)
        print("[Phase‑Adam] short Adam optimisation")
        for idx, (_, a_start) in enumerate(elite[:3]):
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=300_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Fast hill‑climbing on the exact max (dominant phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase‑Hillclimb] exact‑max hill‑climbing")
        # compute exact convolution once
        conv0 = _full_convolution(best_seq, conv_fft_len)
        a_hc, val_hc = _hillclimb(best_seq, conv0, n, deadline - 4.0)
        if val_hc < best_val:
            best_val = val_hc
            best_seq = a_hc.copy()
            print(f"[Hillclimb] improved to {val_hc:.6f}")

    # ------------------------------------------------------------------
    # 4) Safety – ensure a valid return (non‑negative, finite, within limits)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
