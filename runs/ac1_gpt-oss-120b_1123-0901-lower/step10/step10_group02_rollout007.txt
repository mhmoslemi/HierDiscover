"""
Hybrid multi‑restart optimizer for the max‑convolution bound.

The routine mixes several ideas that complement the reference implementation:

* Quick random / parametric restarts (uniform, Dirichlet, exponential, linear,
  quadratic) and a short exp‑gradient phase to discover diverse basins.
* A longer “main’’ phase consisting of
    - one Adam pass on a smooth‑max surrogate,
    - a very long exponentiated‑gradient sub‑gradient pass,
    - a second Adam pass,
    - a projected gradient‑descent polishing step.
* An aggressive shift / reversal sweep (deterministic + random) that exploits
  the fact that a circular shift changes the linear convolution.
* All stages respect the supplied wall‑clock budget (≈1000 s) and always
  return a valid non‑negative sequence.

The code is written with only top‑level functions (no closures) and relies on
NumPy and the already‑imported ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility FFT helpers ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value 2·n·max(a*a) for a with Σa = 1."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max re‑parameterisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # very slow learning‑rate decay – helps fine‑tuning
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)
        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val

def _exp_decay_seq(n: int, r: float) -> np.ndarray:
    """Exponential decreasing sequence a_i ∝ r**i (sum normalised to 1)."""
    if r <= 0.0:
        raise ValueError("r must be > 0")
    if r == 1.0:
        return np.full(n, 1.0 / n, dtype=np.float64)
    p = np.arange(n, dtype=np.float64)
    r_pow = r ** p
    denom = (1.0 - r ** n) / (1.0 - r)          # geometric series sum
    return r_pow / denom


def _linear_decay_seq(n: int) -> np.ndarray:
    """Linearly decreasing sequence a_i ∝ (n-i)."""
    idx = np.arange(n, dtype=np.float64)
    a = n - idx
    return a / a.sum()


def _quadratic_decay_seq(n: int) -> np.ndarray:
    """Quadratically decreasing sequence a_i ∝ (n-i)^2."""
    idx = np.arange(n, dtype=np.float64)
    a = (n - idx) ** 2
    return a / a.sum()

    restart_cnt = 0
    while time.time() < explore_deadline:
        restart_cnt += 1
        # pick an initializer
        mode = np.random.randint(5)
        if mode == 0:
            a0 = np.random.dirichlet(np.ones(n))
        elif mode == 1:
            a0 = np.full(n, 1.0 / n)
        elif mode == 2:
            r = 0.98 + 0.02 * np.random.rand()          # 0.98‑1.00
            a0 = _exp_decay_seq(n, r)
        elif mode == 3:
            a0 = _linear_decay_seq(n)
        else:
            a0 = _quadratic_decay_seq(n)

        # short sub‑gradient run (few seconds, limited iterations)
        sub_dead = time.time() + min(5.0, explore_deadline - time.time())
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=200_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val, best_a = val_sub, a_sub.copy()
            print(f"[Restart {restart_cnt}] new best {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Main optimisation pipeline (budget split dynamically)
    remaining = deadline - time.time()

    # Stage 1: Adam (≈20 % of remaining)
    stage1_dead = time.time() + 0.20 * remaining
    a_adam, val_adam = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, stage1_dead, max_iters=400_000
    )
    if val_adam < best_val:
        best_val, best_a = val_adam, a_adam.copy()
        print(f"[Stage 1] Adam improved to {val_adam:.6f}")

    # Stage 2: long sub‑gradient (≈55 % of remaining)
    remaining = deadline - time.time()
    stage2_dead = time.time() + 0.55 * remaining
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_a, n, conv_fft_len, corr_fft_len, stage2_dead,
        max_iters=10_000_000, eta0=1.0,
    )
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Stage 2] Subgrad improved to {val_sub:.6f}")

    # Stage 3: second Adam (≈10 % of remaining)
    remaining = deadline - time.time()
    stage3_dead = time.time() + 0.10 * remaining
    a_adam2, val_adam2 = _adam_optimize(
        best_a, n, conv_fft_len, corr_fft_len, stage3_dead, max_iters=300_000
    )
    if val_adam2 < best_val:
        best_val, best_a = val_adam2, a_adam2.copy()
        print(f"[Stage 3] Second Adam improved to {val_adam2:.6f}")

    # Stage 4: projected GD (≈10 % of remaining)
    remaining = deadline - time.time()
    stage4_dead = time.time() + 0.10 * remaining
    a_gd, val_gd = _max_conv_descent(
        best_a, n, conv_fft_len, corr_fft_len, stage4_dead, max_iters=300_000
    )
    if val_gd < best_val:
        best_val, best_a = val_gd, a_gd.copy()
        print(f"[Stage 4] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Shift & reversal sweep
    shifts = [0, n // 8, n // 4, 3 * n // 8, n // 2,
              5 * n // 8, 3 * n // 4, 7 * n // 8]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # reversal
    a_rev = best_a[::-1]
    val = _objective(a_rev, n, conv_fft_len)
    if val < best_val:
        best_val, best_a = val, a_rev.copy()
        print(f"[Reverse] improved to {val:.6f}")

    # a few random shifts while we still have a handful of seconds left
    while time.time() < deadline - 0.1:
        sh = np.random.randint(0, n)
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[RandShift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
```
