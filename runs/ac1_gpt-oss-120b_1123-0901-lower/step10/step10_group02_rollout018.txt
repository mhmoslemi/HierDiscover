"""
Hybrid power‑law search for a low‑evaluation step function.

The routine:
1.  Load a warm‑start vector (if the global `height_sequence_1` exists) and keep it
    as a baseline candidate.
2.  Search a one‑parameter family a_i ∝ (i+1)^(-γ) (γ∈[0.4,0.7]) on the simplex.
    For each γ we build the vector, normalise it to sum = 1 and evaluate the exact
    objective 2·n·max(a*a) (the same quantity used by `evaluate_sequence`).
3.  Keep the best γ‑vector.
4.  Refine the best vector with a tiny projected‑gradient / random‑exchange
    local search for the remaining time budget.
5.  Return the best sequence as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(a*a) for a with sum(a)=1."""
    conv = _full_convolution(a, fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    Uses a power‑law family (i+1)^(-γ) as a strong baseline and a tiny
    random‑exchange local refinement.  The whole routine respects the supplied
    time budget (≈ 1000 s) and always returns a valid list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.3          # tiny safety margin

    # problem size (the benchmark uses n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Warm start (if available) – just to have a baseline comparison
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a_warm = np.asarray(warm, dtype=np.float64)
        if a_warm.size != n:
            a_warm = np.resize(a_warm, n)          # truncate / pad with zeros
        a_warm = np.maximum(a_warm, 0.0)
        if a_warm.sum() < 0.01:
            a_warm = np.random.dirichlet(np.ones(n))
        a_warm /= a_warm.sum()
        best_a = a_warm.copy()
        best_val = _objective(best_a, n, conv_fft_len)
        print(f"[Warm] objective {best_val:.6f}")
    else:
        # uniform vector as a trivial start
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_a, n, conv_fft_len)
        print(f"[Init] uniform objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Grid search over the power‑law exponent γ
    #    γ ∈ [0.40, 0.70] is a safe interval that contains the optimum for n≈4k.
    gammas = np.linspace(0.40, 0.70, 151)          # step ≈0.002
    idx = np.arange(1, n + 1, dtype=np.float64)   # i+1 for 1‑based indexing

    for gamma in gammas:
        # early abort if time is almost exhausted
        if time.time() > deadline:
            break

        weights = idx ** (-gamma)                 # raw (i+1)^(-γ)
        a = weights / weights.sum()               # normalise to simplex
        val = _objective(a, n, conv_fft_len)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Gamma] γ={gamma:.4f} improved to {val:.6f}")

    # ------------------------------------------------------------------
    # 3) Very cheap stochastic local refinement (random exchanges)
    #    Only runs if we still have a noticeable amount of time left.
    #    The step size is gradually reduced.
    remaining = deadline - time.time()
    if remaining > 1.0:
        # parameters for the random‑exchange loop
        step = 0.2                # relative size of transferred mass
        decay = 0.9995            # multiplicative decay per accepted move
        it = 0

        while time.time() < deadline:
            # pick two distinct indices
            i, j = np.random.randint(0, n, size=2)
            if i == j:
                continue

            # propose moving a fraction of a[i] to a[j]
            max_delta = min(step * best_a[i], best_a[i])
            if max_delta <= 0:
                continue
            delta = np.random.rand() * max_delta

            a_new = best_a.copy()
            a_new[i] -= delta
            a_new[j] += delta
            # re‑normalise to avoid accumulation of floating‑point errors
            a_new /= a_new.sum()

            val_new = _objective(a_new, n, conv_fft_len)

            if val_new < best_val:
                best_val, best_a = val_new, a_new.copy()
                step *= decay                      # shrink step after a success
                # occasional printout
                if it % 10000 == 0:
                    print(f"[Refine it={it}] new best {val_new:.6f}")
            it += 1
            # safeguard against endless loop if step becomes extremely tiny
            if step < 1e-8:
                break

    # ------------------------------------------------------------------
    # 4) Final sanity checks and reporting
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:                         # should never happen
        best_a = best_a + 1e-6
        best_a /= best_a.sum()

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
