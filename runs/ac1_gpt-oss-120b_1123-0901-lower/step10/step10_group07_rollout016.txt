"""
Step‑function optimizer for the provided ``evaluate_sequence``.
----------------------------------------------------------------
The routine combines:

1. A dense parametric sweep over a wide variety of analytic families
   (exponential, power‑law, stretched‑exponential, beta, Gaussian,
   logistic, linear, random support spikes).  Every candidate is
   normalised to a probability vector (sum = 1) and its exact objective
   ``2·n·max(conv)/(sum²)`` is evaluated.

2. A large number of stochastic restarts.  For each restart a random
   support (a subset of the indices) is sampled and a Dirichlet
   distribution assigns masses on that support.  The start vector is
   refined by two complementary optimisation stages:
      • Adam on a smooth‑max surrogate (soft‑max temperature increase)
        while representing the vector as a soft‑max of unconstrained
        logits (mirror‑descent on the simplex).
      • An exponentiated‑gradient (mirror descent) sub‑gradient optimiser
        on the exact max‑convolution objective.

   The number of restarts and the number of inner iterations are
   chosen adaptively to fill the 1000 s budget.

3. A final short simulated‑annealing polishing phase that moves a random
   fraction of mass between two coordinates.

All helper functions are top‑level (no closures).  The code respects the
time limit by checking ``time.time()`` against a deadline and always
returns the best admissible sequence found.
"""

import time
import numpy as np

# --------------------------------------------------------------
# FFT utilities -------------------------------------------------

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)−1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation of selector w with a (gradient of smooth‑max).
    ``w`` has the same length as the full convolution of ``a`` with itself.
    The result has length ``len(a)``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# --------------------------------------------------------------
# Objective ----------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# --------------------------------------------------------------
# Simplex projection (Duchi et al.) -----------------------------

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if v.ndim != 1:
        raise ValueError("Simplex projection expects a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1] - 1
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# --------------------------------------------------------------
# Candidate generators -------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ e^{−αi}+e^{−α(n−1−i)}."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_gaussian(mean: float, sigma: float, n: int) -> np.ndarray:
    """Plain Gaussian (not mirrored)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - mean) / sigma) ** 2)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_random_support(k: int, n: int) -> np.ndarray:
    """k spikes on a random support, Dirichlet masses on that support."""
    a = np.zeros(n, dtype=np.float64)
    pos = np.random.choice(n, size=k, replace=False)
    masses = np.random.dirichlet(np.ones(k))
    a[pos] = masses
    a /= a.sum()
    return a


# --------------------------------------------------------------
# Optimisers ----------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    ``a = softmax(logits)`` enforces the simplex.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise logits
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report (quiet to save time)
            # print(f"[Adam it={it}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent update for logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (gradually approach true max)
        lam = min(lam * lam_factor, 1e7)

        # decay learning rate occasionally
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _expgrad_optimize(a0: np.ndarray,
                      n: int,
                      conv_fft_len: int,
                      corr_fft_len: int,
                      deadline: float,
                      max_iters: int = 5_000_000,
                      eta0: float = 0.9) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on simplex) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # print(f"[ExpGrad it={it}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        # uniform weight over the active indices (sub‑gradient)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (stay on simplex)
        a = a * np.exp(-eta * grad_a)

        # re‑normalise (projection onto simplex)
        a_sum = a.sum()
        if a_sum <= 0 or np.isnan(a_sum):
            a = np.random.dirichlet(np.ones(n))
            a_sum = a.sum()
        a /= a_sum

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that moves a random fraction
    of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling schedule
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a small random fraction (≤20 %)
        frac = np.random.rand() * 0.2
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative sequence with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # -----------------------------------------------------------
    # problem dimensions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -----------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # -----------------------------------------------------------
    # 1) dense parametric sweep
    # (the sweep is deliberately lightweight – we keep the most promising families)

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.05, 80):
        _maybe_update(_candidate_power(p, n), "Pow")

    # symmetric exponential
    for alpha in np.logspace(-5, 0.8, 100):
        _maybe_update(_candidate_exponential(alpha, n), "Exp")

    # stretched exponential
    for alpha in np.logspace(-4, 0.5, 30):
        for beta in np.linspace(0.3, 2.5, 20):
            _maybe_update(_candidate_stretched(alpha, beta, n), "StE")

    # beta distributions
    a_pars = [0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n), "Beta")

    # double Gaussian
    for sig in np.logspace(-3.0, -0.3, 20):
        _maybe_update(_candidate_double_gaussian(sig, n), "DGau")

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.0, 80):
        _maybe_update(_candidate_left_exponential(alpha, n), "LExp")

    # monotone left‑power
    for p in np.linspace(-4.0, -0.05, 80):
        _maybe_update(_candidate_left_power(p, n), "LPow")

    # left logistic
    mu = n / 2.0
    for beta in np.logspace(-3, 1.2, 60):
        _maybe_update(_candidate_left_logistic(beta, mu, n), "Logi")

    # left linear
    for slope in np.linspace(0.1, 3.0, 50):
        _maybe_update(_candidate_left_linear(slope, n), "Lin")

    # Gaussian centre (non‑mirrored)
    for sigma in np.logspace(-2.5, 0.0, 25):
        _maybe_update(_candidate_gaussian(mean=n/2, sigma=sigma, n=n), "Gauss")

    # random sparse spikes (different densities)
    for k in [64, 128, 256, 512, 1024]:
        for _ in range(6):
            _maybe_update(_candidate_random_support(k, n), f"Spike{k}")

    # a couple of pure Dirichlet samples (dense)
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand, "Dir")

    # -----------------------------------------------------------
    # 2) Adam polishing on the current best (if time permits)
    if best_seq is not None and time.time() < deadline - 35:
        print("[Phase‑Adam] polishing best candidate with Adam")
        adam_deadline = min(deadline - 5, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # -----------------------------------------------------------
    # 3) Stochastic restarts with extended sub‑gradient optimisation
    remaining = deadline - time.time()
    # allocate roughly equal time slices to each restart
    n_restarts = 8
    for r in range(n_restarts):
        if time.time() >= deadline - 5:
            break
        slice_deadline = time.time() + min(remaining / (n_restarts - r), 30.0)

        # draw a random support size (log‑uniform between 128 and 2048)
        k = int(2 ** np.random.uniform(np.log2(128), np.log2(2048)))
        a0 = _candidate_random_support(k, n)

        # sub‑gradient optimisation (long run)
        a_sub, val_sub = _expgrad_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline,
            max_iters=3_000_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")

        remaining = deadline - time.time()

    # -----------------------------------------------------------
    # 4) Final long sub‑gradient run on the overall best sequence
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] final long sub‑gradient refinement")
        a_sub, val_sub = _expgrad_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=12_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # -----------------------------------------------------------
    # 5) Very short simulated‑annealing polishing (if any time left)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] brief annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=180_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # -----------------------------------------------------------
    # final checks & return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # clip to the limits enforced by ``evaluate_sequence`` (just for safety)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Unable to compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
