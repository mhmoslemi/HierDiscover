"""
propose_candidate – a multi‑stage optimizer for the max‑convolution bound.

Stage 1 – Broad random search:
    * Generates thousands of candidates from a wide variety of parametric families:
        – Beta(p,q) densities (p,q drawn log‑uniformly in (0.01,5)).
        – Power‑law “ends‑heavy’’ shapes a_i ∝ (i·(n‑i+1))^α with α drawn log‑uniformly
          in (‑10,‑0.05).
        – Symmetric exponential decays a_i ∝ exp(‑λ·i)+exp(‑λ·(n‑1‑i)), λ random.
        – Powered sine profiles a_i ∝ sin(π·(i+0.5)/n)^β.
        – Random Dirichlet draws with concentration α∈(0.01,5).
    * Every candidate is normalised to unit sum, evaluated by the exact
      convolution (FFT) and the objective 2·n·max(conv).
    * The best candidate (smallest objective) is retained.

Stage 2 – Projected gradient descent (PGD):
    * Starts from the best candidate and iteratively reduces the maximal
      convolution entry using the sub‑gradient of the active (largest) entry.
    * Uses a diminishing step η_t=η₀/√t and an exact Euclidean projection onto
      the probability simplex (ensuring non‑negativity and unit sum).
    * Simple back‑tracking line‑search (five trial multipliers) accelerates
      convergence.  The loop runs until the global time budget is nearly exhausted.

Stage 3 – Light simulated‑annealing polish (optional):
    * Randomly moves a fraction of mass between two coordinates, always accepting
      downhill moves and occasionally uphill ones with a geometric temperature schedule.

All stages respect the global 1000 s budget.  The returned sequence is clipped
to the evaluator’s limits (0 ≤ a_i ≤ 1000) and returned as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) assuming sum(a)=1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    {x | x_i ≥ 0, Σ x_i = 1}.
    """
    # Implements the algorithm of Wang & Carreira‑Perpiñán (2013).
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    # Find the rho where u_i > (cssv_i / i)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # All entries negative – project onto the vertex (1,0,...,0)
        w = np.zeros_like(v)
        w[np.argmax(v)] = 1.0
        return w
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            shape = np.random.dirichlet(alpha * np.ones(n))

        try_candidate(shape)

    # ------------------------------------------------------------------
    # 3️⃣ Projected Gradient Descent refinement (sub‑gradient of max‑conv)
    # ------------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)

    a = best_seq.copy()
    step_const = 1.0                         # initial step size multiplier
    it = 0
    max_iters = 500_000
    while time.time() < deadline - 30.0 and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[PGD it={it:06d}] new best {val:.6f}")

        # index of the (first) maximal convolution entry
        k_star = int(np.argmax(conv))

        # gradient of conv[k_star] w.r.t. a: g_i = 2 * a_{k_star-i} if in range
        grad = np.zeros_like(a)
        i_min = max(0, k_star - (n - 1))
        i_max = min(n - 1, k_star)
        idx_i = np.arange(i_min, i_max + 1, dtype=np.int64)
        grad[idx_i] = 2.0 * a[k_star - idx_i]

        # step size schedule (1/√t)
        step = step_const / np.sqrt(it + 1.0)

        # back‑tracking line‑search (few trial factors)
        improved = False
        for factor in (1.0, 0.5, 0.2, 0.1, 0.05):
            a_new = a - step * factor * grad
            a_new = _project_onto_simplex(a_new)          # maintain sum=1, non‑negative
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new <= val:
                a = a_new
                improved = True
                if val_new < best_val:
                    best_val = val_new
                    best_seq = a_new.copy()
                    print(f"[PGD LS it={it:06d}] val {val_new:.6f}")
                break
        if not improved:
            # if no trial succeeded, shrink global step constant a little
            step_const *= 0.9

        it += 1

    # ------------------------------------------------------------------
    # 4️⃣ Very short simulated‑annealing polish (optional)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()

        T_start = 0.07
        T_end = 1e-5
        moves = 70_000
        for mv in range(moves):
            if time.time() > deadline - 1.0:
                break
            T = max(T_end, T_start * (1.0 - mv / moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of a[i]
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[SA move {mv}] improved to {val2:.6f}")
            else:
                if np.random.rand() < np.exp(-(val2 - val) / T):
                    a = a2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[SA final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final clean‑up
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
```

This implementation expands the search space beyond the previous code (log‑uniform sampling of Beta and power‑law exponents, symmetric exponential shapes, powered‑sine families and many Dirichlet draws) and uses a simple projected‑gradient descent on the max‑convolution entry followed by a short annealing polish. The broader initial sweep together with the more aggressive Euclidean‑projection updates gives a higher chance of beating the previous bound of 1.507703 and reaching the target ≈ 1.503.
