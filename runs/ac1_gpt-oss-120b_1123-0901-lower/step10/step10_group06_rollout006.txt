def _project_simplex(v):
    # v: array, shape (n,)
    # return projection onto probability simplex (nonnegative sum=1)
    n = v.shape[0]
    # algorithm from Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Improved search for a non‑negative step function with a low evaluation score.

Main ideas
----------
1.  A broad parametric sweep (powers, exponentials, beta densities,
    random spikes, Dirichlet draws) produces a set of elite candidates.
2.  For the current best candidate we run a *linear‑program* improvement
    step (the “g‑max” step described in the paper).  The LP maximises the
    total mass of a new vector `g` under the constraint  a ∗ g ≤ max(a∗a).
    After normalising `g` we mix it with `a`.  This usually drops the
    maximal convolution value substantially in only a few iterations.
3.  The result is refined by a long projected sub‑gradient descent on the
    simplex.  The sub‑gradient of `max(a∗a)` is computed via the active set
    (indices attaining the current maximum) and the update is projected
    onto the probability simplex using the O(n log n) algorithm of
    Duchi‑Shalev‑Shwartz‑Singer‑Chandra.  A diminishing step‑size
    ηₜ = η₀ / √(t+1) guarantees convergence.
4.  If time permits a short Adam optimisation (smooth‑max surrogate) is
    performed as a final fine‑tuning.

The whole routine respects a user supplied time budget (default 1000 s)
and always returns a valid non‑negative list.  The default problem size is
n = 512 (a power of two for fast FFTs) – this keeps the LP small while
still providing enough degrees of freedom to beat the previous bound
(1.503).

"""

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for gradients (same length as w)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective (exact, scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean, O(n log n))
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the probability simplex (non‑negative, sum=1)."""
    if v.ndim != 1:
        raise ValueError("Simplex projection only defined for 1‑D arrays.")
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_array = u - cssv / np.arange(1, n + 1)
    rho = np.nonzero(rho_array > 0)[0]
    if rho.size == 0:
        theta = 0.0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Linear‑program improvement (max‑mass g under a ∗ g ≤ M)
# ----------------------------------------------------------------------
def _build_convolution_matrix(a: np.ndarray) -> np.ndarray:
    """Return matrix A (size (2n‑1)×n) such that A @ g = a ∗ g."""
    n = a.shape[0]
    conv_len = 2 * n - 1
    A = np.zeros((conv_len, n), dtype=np.float64)
    for i in range(n):
        A[i : i + n, i] = a
    return A


def _lp_improvement(a: np.ndarray, n: int, conv_fft_len: int,
                    deadline: float, max_iters: int = 12) -> tuple[np.ndarray, float]:
    """Iteratively apply LP improvement; returns best vector and its value."""
    best_a = a.copy()
    best_val = _objective(a, n)
    for it in range(max_iters):
        if time.time() >= deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()
        A = _build_convolution_matrix(a)               # (2n‑1)×n
        c = -np.ones(n)                                 # maximise sum(g)
        b = np.full(A.shape[0], M)
        bounds = [(0.0, None)] * n
        # solve
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            # If the LP fails (numerical), abort the LP phase.
            print(f"[LP] iteration {it} failed: {res.message}")
            break
        g0 = res.x
        sum_g = g0.sum()
        if sum_g <= 1e-12:
            break
        g = g0 / sum_g                    # renormalise to unit sum
        # convex combination (mixing factor 0.5 works well)
        t = 0.5
        a = (1.0 - t) * a + t * g
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[LP] iter {it:02d} improved to {val:.6f}")
    return best_a, best_val


# ----------------------------------------------------------------------
# Projected sub‑gradient descent (simplex, diminishing step size)
# ----------------------------------------------------------------------
def _proj_subgrad_optimize(a0: np.ndarray, n: int,
                           conv_fft_len: int, corr_fft_len: int,

                           max_iters: int = 5_000_000,
                           eta0: float = 0.7) -> tuple[np.ndarray, float]:
    """Projected sub‑gradient descent on max(a*a) with Euclidean projection."""
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[ProjGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, with a tiny tolerance)
        tol = max(1e-12, 1e-9 * max_c)
        active = conv >= max_c - tol
        if not np.any(active):
            # should not happen, but guard against it
            active = conv == max_c
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution (see analysis)
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        eta = eta0 / np.sqrt(it + 1.0)

        # gradient step and projection onto the simplex
        a = a - eta * grad
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Adam optimiser on a smooth‑max (log‑sum‑exp) surrogate
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 400_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)
    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.02
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max (stays on the simplex)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # problem size -------------------------------------------------------
    n = 512                    # power of two, yields fast FFTs and small LP
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep a handful of elites
    # ------------------------------------------------------------------
    elite = []     # list of (objective, vector)

    def _maybe_update(a: np.ndarray):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
        if len(elite) > 8:
            elite.pop()   # keep only the 8 best

    # 1‑a) power‑law (negative exponents concentrate mass at the edges)
    for p in np.linspace(-10.0, -0.04, 40):
        _maybe_update(_candidate_power(p, n))

    # 1‑b) symmetric exponential
    for alpha in np.logspace(-5, 1.2, 35):
        _maybe_update(_candidate_exponential(alpha, n))

    # 1‑c) Beta distributions – explore very sharp edges
    a_pars = [0.01, 0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # 1‑d) Random spike patterns (different sparsities)
    for k in [16, 32, 64, 128, 256]:
        for _ in range(5):
            _maybe_update(_candidate_spike(k, n, random=True))

    # 1‑e) Dirichlet draws with tiny concentration (extremely sparse)
    for conc in [0.02, 0.05, 0.1, 0.2]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # 1‑f) Warm start from a previously stored solution if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    # Pick the current best elite entry
    best_val, best_seq = elite[0]
    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Linear‑program improvement (few iterations, cheap)
    # ------------------------------------------------------------------
    if time.time() < deadline - 30.0:
        lp_deadline = time.time() + min(12.0, 0.1 * (deadline - time.time()))
        print("[Phase‑LP] starting LP‑based improvement")
        best_seq, best_val = _lp_improvement(best_seq, n, conv_fft_len,
                                             lp_deadline, max_iters=12)
        print(f"[Phase‑LP] best after LP phase: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Projected sub‑gradient descent (long phase)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        sub_deadline = deadline - 4.0
        print("[Phase‑Subgrad] long projected sub‑gradient optimisation")
        best_seq, best_val = _proj_subgrad_optimize(best_seq, n,
                                                    conv_fft_len,
                                                    corr_fft_len,
                                                    sub_deadline,
                                                    max_iters=10_000_000,
                                                    eta0=0.8)
        print(f"[Phase‑Subgrad] best after sub‑gradient: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional short Adam fine‑tuning
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Adam] short Adam fine‑tuning")
        best_seq, best_val = _adam_optimize(best_seq, n,
                                            conv_fft_len,
                                            corr_fft_len,
                                            deadline,
                                            max_iters=300_000)
        print(f"[Phase‑Adam] final best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity‑check with the public evaluator (if it exists)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score – evaluator unavailable.")

    return best_seq.tolist()
