```python
import time
import numpy as np
import scipy.optimize

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    # compute linear convolution using real FFT
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]

def _softmax(logits):
    mx = logits.max()
    exp = np.exp(logits - mx)
    sum_exp = exp.sum()
    return exp / sum_exp

def _smooth_max_val_and_grad(logits, lam, n, conv_fft_len, corr_fft_len):
    a = _softmax(logits)
    conv = _full_convolution(a, conv_fft_len)
    # stable max
    max_c = conv.max()
    shift = conv - max_c
    # raw weights for softmax on conv
    w_raw = np.exp(lam * shift)
    w_sum_raw = w_raw.sum()
    if w_sum_raw == 0 or not np.isfinite(w_sum_raw):
        w = np.full_like(conv, 1.0 / conv.size)
        w_sum_raw = 1.0
    else:
        w = w_raw / w_sum_raw
    smooth_max = max_c + np.log(w_sum_raw) / lam
    # gradient w.r.t a
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
    # chain rule through softmax (logits)
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)
    val = 2.0 * n * smooth_max
    return val, grad_logits
```

"""
Propose a non‑negative step‑function (probability vector) with a low
`evaluate_sequence` score.

Algorithm
---------
* Build a diverse pool of initial probability vectors (Dirichlet draws,
  exponential, Gaussian, spikes, warm‑start …).
* For each start run a *continuation* optimisation:
    - Represent the vector as a soft‑max of unrestricted logits.
    - Minimise a smooth approximation of `max(conv(a,a))`
      using the log‑sum‑exp with increasing temperature λ.
    - The smooth objective is differentiable; gradients are obtained
      analytically via FFT‑based convolution and cross‑correlation.
    - Optimise the smooth objective with L‑BFGS‑B (scipy.optimize), which
      exploits quasi‑Newton curvature information and converges faster than
      first‑order methods.
* After the λ‑continuation perform a short exponentiated‑gradient
  sub‑gradient phase (exact max) to polish the solution.
* Keep the best vector (lowest exact objective) seen across all starts.
* Return the best vector as a plain Python list (the evaluator will
  clip to the allowed range).

The whole routine respects a global time budget (default 1000 s) and
guarantees a non‑empty return.
"""

import time
import numpy as np
import scipy.optimize

# ----------------------------------------------------------------------
# Basic FFT utilities
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _softmax(logits: np.ndarray) -> np.ndarray:
    """Stable soft‑max."""
    mx = logits.max()
    exp = np.exp(logits - mx)
    return exp / exp.sum()


# ----------------------------------------------------------------------
# Smooth max (log‑sum‑exp) objective and gradient (for L‑BFGS)

def _smooth_max_val_and_grad(
    logits: np.ndarray,
    lam: float,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
) -> tuple[float, np.ndarray]:
    """
    Returns (value, gradient) of the smoothed objective
        2 * n * (log‑sum‑exp(lam * conv) / lam)
    where conv = a * a with a = softmax(logits).
    """
    a = _softmax(logits)
    conv = _full_convolution(a, conv_fft_len)

    # max of conv for numerical stability
    max_c = conv.max()
    shift = conv - max_c                          # ≤ 0
    w_raw = np.exp(lam * shift)                  # exp(lam*(conv-max))
    w_sum_raw = w_raw.sum()
    if w_sum_raw == 0 or not np.isfinite(w_sum_raw):
        # fallback to uniform weights (unlikely but safe)
        w = np.full_like(conv, 1.0 / conv.size)
        w_sum_raw = 1.0
    else:
        w = w_raw / w_sum_raw

    # smooth max = max_c + log(w_sum_raw) / lam
    smooth_max = max_c + np.log(w_sum_raw) / lam

    # gradient wrt a: 2 * (w ★ a)
    grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

    # chain rule through the soft‑max parametrisation
    avg = np.dot(a, grad_a)
    grad_logits = a * (grad_a - avg)

    val = 2.0 * n * smooth_max
    return val, grad_logits

# Exponentiated‑gradient sub‑gradient optimiser (exact max)

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient on the exact max‑convolution.
    Returns the best vector seen and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)          # multiplicative update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val

# Candidate factories (simple families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a

        pool.append(_make_gaussian(n, sigma))

    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))

    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    for _ in range(6):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # Continuation schedule for the smooth‑max temperature λ
    lambda_schedule = [0.5, 1.0, 2.0, 5.0, 10.0, 20.0,
                       50.0, 100.0, 200.0, 500.0, 1000.0]

    # ------------------------------------------------------------------
    # Main loop over candidates
    for idx, a0 in enumerate(pool):
        if time.time() > deadline:
            break

        # Normalise to the probability simplex
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 1e-12:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Initialise logits for the soft‑max parametrisation
        eps = 1e-15
        logits = np.log(a0 + eps)

        # Continuation with increasing λ
        for lam in lambda_schedule:
            if time.time() > deadline:
                break

            # L‑BFGS optimisation of the smoothed objective
            try:
                # maxiter per λ – a modest budget; L‑BFGS already uses curvature
                max_iter = 250
                res = scipy.optimize.minimize(
                    fun=_smooth_max_val_and_grad,
                    x0=logits,
                    args=(lam, n, conv_fft_len, corr_fft_len),
                    method="L-BFGS-B",
                    jac=True,
                    options={"maxiter": max_iter, "disp": False},
                )
            except Exception as e:
                print(f"[Candidate {idx}] L‑BFGS error (λ={lam}): {e}")
                continue

            logits = res.x
            a = _softmax(logits)

            # Exact objective (sum=1 ⇒ denominator = 1)
            conv = _full_convolution(a, conv_fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                print(f"[Candidate {idx}] λ={lam:.1f} improved best = {val:.6f}")

        # Short exponentiated‑gradient polishing (exact max)
        if best_seq is not None and time.time() < deadline - 5:
            a_ref, val_ref = _subgradient_optimize_expgrad(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                deadline,
                max_iters=200_000,
                eta0=0.8,
            )
            if val_ref < best_val:
                best_val = val_ref
                best_seq = a_ref.copy()
                print(f"[Candidate {idx}] ExpGrad polish improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Final long polishing phase if time permits
    if best_seq is not None and time.time() < deadline - 5:
        print("[Final] Long exponentiated‑gradient refinement")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=800_000,
            eta0=0.6,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Final] refinement improved best to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Guard against empty result
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0  # uniform value

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Done – best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
