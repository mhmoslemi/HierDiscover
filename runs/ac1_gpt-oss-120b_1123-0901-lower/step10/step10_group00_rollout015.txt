def _lp_improve(a, n, conv_fft_len, deadline, max_iter=3):
    best = a.copy()
    best_val = evaluate_sequence(best.tolist()) # use evaluate function
    for it in range(max_iter):
        if time.time() > deadline - 5:
            break
        M = _full_convolution(a, conv_fft_len).max()  # compute max conv
        # Build sparse matrix A_ub
        rows = []
        cols = []
        datas = []
        for i in range(n):
            a_i = a[i]
            if a_i == 0:
                continue
            start = i
            end = i + n  # exclusive
            # columns j = start...end-1 map to j-start = index in [0,n-1]
            # Row indices k from start to end-1, column indices = k - i
            # Add n entries for each i
            rows.extend(range(start, end))
            cols.extend(range(0, n))
            datas.extend([a_i] * n)
        A_ub = sp.csr_matrix((datas, (rows, cols)), shape=(2*n-1, n))
        # bound vector:
        b_ub = np.full(2*n-1, M)
        # objective:
        c = -np.ones(n) # minimize -sum => maximize sum
        # bounds for b: (0, None)
        res = spopt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * n, method='highs')
        if not res.success:
            print("LP failed")
            break
        b = res.x
        sum_b = b.sum()
        if sum_b <= 1e-12:
            break
        # Scale down to sum a
        scale = a.sum() / sum_b
        g = b * scale
        # Project onto simplex (nonneg) but scaling already ensures sum = sum a.
        # But might have small negative? b >=0 ensures nonneg.
        # Evaluate g
        val_g = evaluate_sequence(g.tolist())
        if val_g < best_val:
            best_val = val_g
            best = g.copy()
            print(f"[LP it={it}] improved to {val_g:.6f}")
        # Mix a and g
        a = 0.7 * a + 0.3 * g
        # Normalize sum
        a = a / a.sum() * (a.sum()) # Actually sum same? Already sum = sum a.
        # We may also re-normalize to 1 (scale invariance).
        a = a / a.sum()
    return best, best_val

"""
Hybrid step‑function optimizer with a dedicated linear‑programming refinement.

The routine follows three main stages (all respecting the 1000 s budget):

1.  A diverse pool of starting vectors (uniform, Dirichlet, sinusoid,
    exponential, spike‑patterns, etc.) is generated.  For each starter the
    algorithm performs a long Adam optimisation on a smooth‑max surrogate of
    the max‑convolution, followed by an exponentiated‑gradient sub‑gradient
    phase (exact max).  The best candidate is retained.

2.  Spectral flattening: the FFT magnitude of the current vector is forced to
    be (almost) constant while the DC component (the sum) is kept unchanged;
    after each inverse FFT the vector is projected onto the probability
    simplex.  This operation tends to flatten the autocorrelation and usually
    improves the objective further.

3.  Linear‑programming improvement:  given the current vector `a` with
    `M = max(a*a)`, we solve
        maximise   Σ b_j
        subject to a * b ≤ M   (component‑wise convolution)
                  b_j ≥ 0.
    The LP is built with a sparse Toeplitz convolution matrix (size
    (2n‑1) × n).  The solution `b` is scaled back to the original sum,
    mixed with `a` and the process is repeated a few times.  This step is
    inspired by the classic “LP update” described in the problem statement
    and typically yields a noticeable drop in the evaluation value.

If time permits, a short projected‑gradient descent and simulated‑annealing
pass are performed for final polishing.  The function always returns the
best sequence found before the deadline (as a plain Python list).  All
vectors are kept non‑negative and normalised to sum = 1, because the
evaluation function is homogeneous of degree 0.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of the max‑convolution."""
    eps = 1e-15
    logits = np.log(a0 + eps)   # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # initial temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float, max_iters: int = 300_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the current max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray, n: int,
                     conv_fft_len: int, corr_fft_len: int,
                     deadline: float, max_iters: int = 150_000) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, deadline: float,
                    max_moves: int = 60_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential edge families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def _spectral_flatten_optimize(a0: np.ndarray, n: int,
                              conv_fft_len: int, deadline: float,
                              max_iters: int = 200_000) -> tuple[np.ndarray, float]:
    """Force the FFT magnitude to be (almost) constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)               # full complex FFT
        energy = np.dot(a, a)           # ||a||_2^2

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

# Linear‑programming refinement (requires SciPy)
try:
    import scipy.sparse as _sp
    import scipy.optimize as _spopt
    _SCIPY_OK = True
except Exception:  # pragma: no cover
    _SCIPY_OK = False


def _build_conv_matrix(a: np.ndarray) -> _sp.csr_matrix:
    """Build the (2n‑1)×n convolution matrix for a fixed left vector a.
    Row i, column j contains a[i‑j] if i‑j ≥ 0 else 0.
    The matrix is lower‑triangular with n non‑zero diagonals."""
    n = a.shape[0]
    offsets = np.arange(n)                       # 0 … n‑1
    data = [np.full(2 * n - 1 - d, a[d], dtype=np.float64) for d in range(n)]
    return _sp.diags(data, offsets, shape=(2 * n - 1, n), format='csr')


def _lp_improve(a0: np.ndarray, n: int, conv_fft_len: int,
                deadline: float, max_iter: int = 5) -> tuple[np.ndarray, float]:
    """Iteratively solve the LP  max Σb  s.t. a*b ≤ M, b ≥ 0.
    Mix the resulting b back into a and keep the best iterate."""
    if not _SCIPY_OK:
        # SciPy not available – skip this phase
        return a0, float("inf")

    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        M = conv.max()

        # Build the convolution matrix for the current a
        A = _build_conv_matrix(a)

        # Linear program: maximise Σb   <=>  minimise -Σb
        c = -np.ones(n, dtype=np.float64)
        bounds = [(0, None)] * n
        b_ub = np.full(2 * n - 1, M, dtype=np.float64)

        res = _spopt.linprog(c, A_ub=A, b_ub=b_ub,
                             bounds=bounds, method='highs')
        if not res.success:
            print(f"[LP it={it}] linprog failed: {res.message}")
            break

        b = res.x
        sum_b = b.sum()
        if sum_b <= 0.0:
            break

        # Scale b back to the current total mass (which is 1)
        scale = a.sum() / sum_b
        g = b * scale                         # still non‑negative, sum(g)=sum(a)

        # Evaluate candidate g
        val_g = evaluate_sequence(g.tolist())
        if val_g < best_val:
            best_val = val_g
            best_a = g.copy()
            print(f"[LP it={it}] new best {val_g:.6f}")

        # Mix a and g (convex combination) and re‑normalise
        a = 0.6 * a + 0.4 * g
        total = a.sum()
        if total <= 0 or np.isnan(total):
            a = np.random.dirichlet(np.ones(n))
        else:
            a = a / total

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # --------------------------------------------------------------
    # Vector length – 2048 gives a good trade‑off between quality and speed
    n = 2048
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start from a previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # --------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Additional shuffled Dirichlet variants
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 5                     # invest more time per candidate

    # --------------------------------------------------------------
    # Phase I – deep optimisation of a handful of promising starts
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Allocate a slice of the remaining budget (≈ 30 s per candidate)
        sub_dead = min(deadline, time.time() + 30.0)

        # 1) Adam on smooth‑max surrogate
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=400_000
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # 2) Exponentiated‑gradient sub‑gradient phase (if time permits)
        if time.time() < deadline - 5:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=500_000, eta0=0.8,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase II – spectral flattening (new component)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Spectral] Starting spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, n, conv_fft_len, deadline, max_iters=250_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # --------------------------------------------------------------
    # Phase III – linear‑programming refinement
    if best_seq is not None and time.time() < deadline - 20:
        print("[LP] Starting LP refinement")
        a_lp, val_lp = _lp_improve(
            best_seq, n, conv_fft_len, deadline, max_iter=4
        )
        if val_lp < best_val:
            best_val = val_lp
            best_seq = a_lp.copy()
            print(f"[LP] improved to {val_lp:.6f}")

    # --------------------------------------------------------------
    # Phase IV – projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 10:
        print("[GD] Projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase V – simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] Final simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final check (in case rounding introduced tiny mass loss)
    if best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    final_val = evaluate_sequence(best_seq.tolist())
    print(f"Final best value {final_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
