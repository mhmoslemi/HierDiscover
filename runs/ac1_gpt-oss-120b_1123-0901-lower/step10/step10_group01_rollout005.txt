"""
Hybrid multi‑stage optimiser for the step‑function evaluation problem.

Stage 1 – massive random sampling of simple parametric families (Dirichlet,
exponential, Gaussian, power‑law, random spikes, mixtures).  Each candidate is
evaluated via a fast FFT‑based convolution and the top‑K (default 5) are kept.

Stage 2 – each of the elite candidates is refined by a short Adam run on the
smooth‑max surrogate, an exponentiated‑gradient sub‑gradient phase and finally a
projected gradient descent on the exact max‑convolution.  The best refined
solution is remembered.

Stage 3 – if time permits a stochastic local‑search phase perturbs the current
best solution and applies a brief projected‑gradient descent;  improvements are
kept.

All optimisation works on the probability simplex (sum = 1);  the evaluation
function is scale‑invariant, so this does not affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT helpers -------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using real FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w★a needed for (sub‑)gradient computation."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

# Families of random initial vectors ------------------------------------
def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    """k distinct spike locations, equal mass."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_random_spikes(n):
    """Random number of spikes with random masses."""
    k = np.random.randint(20, max(21, n // 8))
    pos = np.random.choice(n, size=k, replace=False)
    weights = np.random.rand(k)
    a = np.zeros(n, dtype=np.float64)
    a[pos] = weights
    s = a.sum()
    if s < 1e-12:
        a = _make_uniform(n)
    else:
        a /= s
    return a


def _make_mixed_exponential(n):
    """Mixture of two mirrored exponentials."""
    gamma1 = np.random.uniform(0.05, 3.0)
    gamma2 = np.random.uniform(0.05, 3.0)
    a1 = _make_exponential(n, gamma1)
    a2 = _make_exponential(n, gamma2)[::-1]  # mirrored
    w = np.random.rand()
    a = w * a1 + (1 - w) * a2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _random_shape(n):
    """Draw a random shape from several families."""
    r = np.random.rand()
    if r < 0.12:
        # Dirichlet with wide concentration range
        alpha = 10 ** np.random.uniform(-3, 1)
        return _make_dirichlet(n, alpha)
    elif r < 0.24:
        # Exponential symmetric
        gamma = np.random.uniform(0.05, 2.5)
        return _make_exponential(n, gamma)
    elif r < 0.36:
        # Gaussian centred at the middle
        sigma = np.random.uniform(0.02, 0.35)
        return _make_gaussian(n, sigma)
    elif r < 0.48:
        # Power‑law
        expo = np.random.uniform(-2.0, -0.3)
        return _make_powerlaw(n, expo)
    elif r < 0.60:
        # Random spikes
        return _make_random_spikes(n)
    elif r < 0.72:
        # Mixed exponentials (asymmetric)
        return _make_mixed_exponential(n)
    elif r < 0.84:
        # Two‑Gaussian mixture with random means
        sigma = np.random.uniform(0.02, 0.3)
        mu1 = np.random.uniform(0, n - 1)
        mu2 = np.random.uniform(0, n - 1)
        i = np.arange(n, dtype=np.float64)
        g1 = np.exp(-0.5 * ((i - mu1) / (sigma * n)) ** 2)
        g2 = np.exp(-0.5 * ((i - mu2) / (sigma * n)) ** 2)
        w = np.random.rand()
        a = w * g1 + (1 - w) * g2
        a = np.maximum(a, 0.0)
        a /= a.sum()
        return a
    else:
        # Uniform fallback
        return _make_uniform(n)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters: int = 150_000):
    """Adam on a log‑sum‑exp smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max parametrisation
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # smooth‑max of convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # weights for the smooth‑max
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95
        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len,
                                 deadline, max_iters: int = 250_000,
                                 eta0: float = 0.8):
    """Exact‑max sub‑gradient with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1
    return best_a, best_val


def _max_conv_descent(a0, n, conv_fft_len, corr_fft_len,
                     deadline, max_iters: int = 120_000):
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1
    return best_a, best_val

# Main search routine ---------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine builds a large pool of random shapes, keeps the elite few,
    refines them with a chain of Adam → exponentiated‑gradient → projected‑GD,
    and finally performs a stochastic local‑search refinement.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5          # safety margin

    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start (if a previous construction is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Stage 1 – massive random sampling
    top_k = 5
    elite = []                     # list of (value, vector)
    best_val = float("inf")
    best_seq = None

    phase1_dead = deadline - 300.0   # reserve ~5 min for later refinements
    it_rand = 0
    while time.time() < phase1_dead:
        if warm is not None and it_rand == 0:
            a = warm.copy()
        else:
            a = _random_shape(n)

        # quick exact evaluation (no clipping needed – values are safe)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Random {it_rand:05d}] new global best {val:.6f}")

        # maintain a small elite set
        if len(elite) < top_k or val < elite[-1][0]:
            elite.append((val, a.copy()))
            elite.sort(key=lambda x: x[0])
            if len(elite) > top_k:
                elite.pop()
        it_rand += 1

    # ------------------------------------------------------------------
    # Stage 2 – refinement of the elite candidates
    for idx, (init_val, a0) in enumerate(elite):
        if time.time() >= deadline - 60:          # keep some slack for final phase
            break

        print(f"[Refine #{idx}] starting from init val {init_val:.6f}")

        # allocate a slice of the remaining budget (≈ 60 s per candidate)
        sub_dead = min(deadline - 5.0, time.time() + 55.0)

        # 1) Adam on smooth‑max (shorter run)
        a_adam, val_adam = _adam_optimize(a0, n, conv_fft_len,
                                          corr_fft_len, sub_dead,
                                          max_iters=100_000)
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Refine #{idx}] Adam improved to {val_adam:.6f}")

        # 2) Exact‑max sub‑gradient (exponentiated‑gradient)
        if time.time() < deadline - 30:
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=200_000, eta0=0.7)
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Refine #{idx}] ExpGrad improved to {val_sub:.6f}")

        # 3) Projected gradient descent (final polish)
        if time.time() < deadline - 10:
            a_gd, val_gd = _max_conv_descent(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=150_000)
            if val_gd < best_val:
                best_val = val_gd
                best_seq = a_gd.copy()
                print(f"[Refine #{idx}] GD improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Stage 3 – stochastic local‑search perturbations
    while time.time() < deadline - 5:
        # small random perturbation of the current best
        sigma = np.random.uniform(1e-5, 1e-3) * (1.0 / np.sqrt(n))
        a0 = best_seq + sigma * np.random.randn(n)
        a0 = np.maximum(a0, 0.0)
        s = a0.sum()
        if s < 1e-12:
            continue
        a0 /= s

        sub_dead = min(deadline, time.time() + 6.0)
        a_ref, val_ref = _max_conv_descent(
            a0, n, conv_fft_len, corr_fft_len,
            sub_dead, max_iters=30_000)

        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Perturb] improved best to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to evaluator's allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
