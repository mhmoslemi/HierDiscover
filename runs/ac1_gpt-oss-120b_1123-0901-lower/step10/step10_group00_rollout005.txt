def _lp_phase(a0, deadline):
    f = a0.copy()
    n = len(f)
    best_f = f.copy()
    best_val = float('inf')
    iter_cnt = 0
    while time.time() < deadline:
        # compute conv and current objective
        conv = np.convolve(f, f)
        M = conv.max()
        val = 2.0 * n * M
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP it={iter_cnt}] new best {val:.6f}")
        # Build convolution matrix
        A = np.empty((2*n-1, n), dtype=np.float64)
        # vectorizable fill using loops:
        for i in range(n):
            A[i:i+n, i] = f
        # Solve LP
        c = -np.ones(n, dtype=np.float64)  # maximize sum(g)
        b = np.full(2*n-1, M, dtype=np.float64)
        bounds = [(0, None)] * n
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break
        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            print("[LP] zero solution")
            break
        g = g / g_sum
        # update with mixing
        t = 0.2  # could adapt
        f = (1 - t) * f + t * g
        # re-normalize
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s <= 0:
            f = np.random.dirichlet(np.ones(n))
            s = f.sum()
        f = f / s
        iter_cnt += 1
        # optionally reduce t
        if iter_cnt % 5 == 0:
            t = max(0.02, t * 0.9)
    return best_f, best_val

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    final_n = 4096
    base_n = final_n // 4  # 1024
    
    # Build warm start if exists
    warm = globals().get('height_sequence_1')
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        # truncate or pad to final_n
        if warm.shape[0] != final_n:
            if warm.shape[0] > final_n:
                warm = warm[:final_n]
            else:
                warm = np.pad(warm, (0, final_n - warm.shape[0]))
        warm = np.clip(warm, 0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        # downsample to base_n by averaging per block
        factor = final_n // base_n
        warm_down = warm.reshape(base_n, factor).mean(axis=1)
        warm_down = warm_down / warm_down.sum()
    else:
        warm_down = None

    # initial a0
    if warm_down is not None:
        a0 = warm_down
    else:
        a0 = np.random.dirichlet(np.ones(base_n))

    # stage 1: LP smoothing
    stage1_dead = min(deadline, time.time() + 300.0)  # 5 min for LP
    f_lp, val_lp = _lp_phase(a0, stage1_dead)
    best_seq = f_lp.copy()
    best_val = val_lp

    # stage 2: upsample to final_n
    up_factor = final_n // base_n
    f_up = np.repeat(best_seq, up_factor)
    f_up = f_up / f_up.sum()  # sum 1
    # compute objective after upsampling
    n = final_n
    conv = np.convolve(f_up, f_up)
    val_up = 2 * n * conv.max()
    if val_up < best_val:
        best_val = val_up
        best_seq = f_up.copy()
        print(f"[Upsample] new best {best_val:.6f}")

    # recompute FFT lengths
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # Stage 3: subgradient optimization (exponential gradient)
    if time.time() < deadline - 30:
        sub_dead = deadline - 5
        a_sub, val_sub = _subgradient_optimize_expgrad(best_seq, n, conv_fft_len, corr_fft_len, sub_dead, max_iters=800_000, eta0=0.8)
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()

    # Stage 4: spectral flattening
    if time.time() < deadline - 20:
        spec_dead = deadline - 5
        a_spec, val_spec = _spectral_flatten_optimize(best_seq, n, conv_fft_len, spec_dead, max_iters=250_000)

    print(f"Final best value {best_val:.6f}")

"""
Hybrid optimizer that adds a Frank‑Wolfe (LP‑based) stage to the previous
gradient / spectral pipeline.

1.  Low‑dimensional Frank‑Wolfe loop (n≈1024)
    – For the current vector f compute M = max(f⋆f).
    – Solve the linear programme
          maximise   Σ g_i
          s.t.      (f⋆g)[k] ≤ M   ∀k
                    g_i ≥ 0
      The solution g points towards a direction that reduces the
      L∞‑norm of the autocorrelation.
    – Update f ← (1−t)f + t·g (t shrinks gradually) and repeat.
    – The best iterate is kept.

2.  Upsample the low‑dimensional result to the target length
    (4096) by block‑wise replication (piece‑wise constant).  This spreads
    the mass and often lowers the maximal convolution value.

3.  Refine the full‑size vector with the already‑tested stages:
       • Exponentiated‑gradient sub‑gradient descent on the exact max‑conv.
       • Spectral‑flattening (force |FFT| to be (almost) constant).
       • Simple simulated‑annealing local search.

The whole routine respects the global time budget (≈1000 s) and always
returns the best sequence found.
"""

import time
import numpy as np
from scipy import optimize as opt

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _build_conv_matrix(f: np.ndarray) -> np.ndarray:
    """
    Build the (2n‑1)×n matrix A such that A @ g = f⋆g (full linear convolution).
    Row k (0‑based) contains f_i at column j = k-i whenever 0≤i,j<n.
    """
    n = f.shape[0]
    rows = 2 * n - 1
    A = np.zeros((rows, n), dtype=np.float64)
    for i in range(n):
        A[i:i + n, i] = f
    return A


# ----------------------------------------------------------------------
def _lp_phase(f0: np.ndarray, deadline: float):
    """
    Frank‑Wolfe (LP) phase on a low‑dimensional vector.
    Returns the best vector encountered (sum=1) and its objective value.
    """
    f = f0.copy()
    n = f.shape[0]
    best_f = f.copy()
    best_val = float("inf")
    it = 0
    t = 0.2                     # initial mixing weight

    while time.time() < deadline:
        conv = np.convolve(f, f)
        M = conv.max()
        val = 2.0 * n * M       # sum(f)=1 ⇒ objective = 2n·max(conv)
        if val < best_val:
            best_val = val
            best_f = f.copy()
            print(f"[LP it={it:03d}] new best {val:.6f}")

        # Build the linear constraints (f⋆g ≤ M)
        A = _build_conv_matrix(f)          # (2n‑1)×n
        c = -np.ones(n, dtype=np.float64)  # maximise Σ g_i  → minimise −Σ g_i
        b = np.full(A.shape[0], M, dtype=np.float64)
        bounds = [(0, None)] * n

        # Solve the LP
        res = opt.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
        if not res.success:
            print("[LP] linprog failed:", res.message)
            break

        g = res.x
        g_sum = g.sum()
        if g_sum <= 0:
            print("[LP] zero solution")
            break
        g = g / g_sum                     # project onto the simplex

        # Convex combination step
        f = (1.0 - t) * f + t * g
        f = np.maximum(f, 0.0)
        s = f.sum()
        if s <= 0:
            f = np.random.dirichlet(np.ones(n))
            s = f.sum()
        f = f / s

        it += 1
        # slowly shrink the step size
        if it % 5 == 0:
            t = max(0.02, t * 0.9)

    return best_f, best_val


# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max exponentiated‑gradient sub‑gradient method.
    Keeps the iterate on the probability simplex.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the current max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude of the vector to be (almost) constant
    while keeping the DC component (the sum) unchanged.
    After each inverse FFT the vector is projected back onto the probability simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        A = np.fft.fft(a)               # full complex FFT
        energy = np.dot(a, a)           # ||a||_2^2

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Target length and low‑dimensional size for the LP stage
    final_n = 4096
    low_n = final_n // 4          # 1024 → good trade‑off between quality & speed
    up_factor = final_n // low_n

    # ------------------------------------------------------------------
    # Warm start (if a previous construction is available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > final_n:
            warm = warm[:final_n]
        elif warm.shape[0] < final_n:
            warm = np.pad(warm, (0, final_n - warm.shape[0]))
        warm = np.clip(warm, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        # Down‑sample to low_n by averaging each block
        warm_low = warm.reshape(low_n, up_factor).mean(axis=1)
        warm_low = warm_low / warm_low.sum()
    else:
        warm_low = None

    # ------------------------------------------------------------------
    # Initial low‑dimensional vector
    if warm_low is not None:
        a0_low = warm_low
    else:
        a0_low = np.random.dirichlet(np.ones(low_n))

    # ------------------------------------------------------------------
    # Stage I – Frank‑Wolfe LP smoothing on the low‑dimensional vector
    stage1_dead = min(deadline, time.time() + 300.0)   # allocate ~5 min
    best_low, best_val_low = _lp_phase(a0_low, stage1_dead)
    best_seq = best_low.copy()
    best_val = best_val_low
    print(f"[Stage I] best low‑dim value {best_val:.6f}")

    # ------------------------------------------------------------------
    # Stage II – upsample to the final length
    best_up = np.repeat(best_seq, up_factor)
    best_up = best_up / best_up.sum()               # keep Σa = 1
    conv_up = np.convolve(best_up, best_up)
    val_up = 2.0 * final_n * conv_up.max()
    if val_up < best_val:
        best_val = val_up
        best_seq = best_up.copy()
    print(f"[Upsample] value {val_up:.6f}")

    # ------------------------------------------------------------------
    # Pre‑compute FFT lengths for the full‑size problem
    conv_fft_len = _next_pow2(2 * final_n - 1)
    corr_fft_len = _next_pow2(3 * final_n - 2)

    # ------------------------------------------------------------------
    # Stage III – exponentiated‑gradient sub‑gradient refinement
    if time.time() < deadline - 30:
        sub_dead = deadline - 5.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, final_n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=600_000, eta0=0.8,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Stage IV – spectral flattening
    if time.time() < deadline - 20:
        spec_dead = deadline - 5.0
        a_spec, val_spec = _spectral_flatten_optimize(
            best_seq, final_n, conv_fft_len, spec_dead, max_iters=250_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Spectral] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Stage V – simulated‑annealing final polish
    if time.time() < deadline - 5:
        a_ann, val_ann = _anneal_optimize(
            best_seq, final_n, conv_fft_len, deadline, max_moves=80_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safety fallback
    if best_seq is None or best_seq.size == 0:
        best_seq = np.full(final_n, 1.0 / final_n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
