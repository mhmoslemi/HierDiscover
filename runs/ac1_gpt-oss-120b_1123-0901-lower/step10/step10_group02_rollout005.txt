def random_local_search(a, deadline):
    best_a = a.copy()
    best_val = compute_obj(a)
    n = len(a)
    while time.time() < deadline:
        # pick two distinct indices
        i = np.random.randint(n)
        if best_a[i] == 0:
            continue
        j = np.random.randint(n-1)
        if j >= i: j += 1
        # sample delta fraction
        delta = best_a[i] * (np.random.rand() * 0.1)  # up to 10% transfer
        # apply
        a_new = best_a.copy()
        a_new[i] -= delta
        a_new[j] += delta
        val = compute_obj(a_new)
        if val < best_val:
            best_val, best_a = val, a_new
    return best_a, best_val
```

"""
Hybrid optimizer with an extended sub‑gradient phase and a cheap
random‑local‑search finetuning.  The routine

* builds several diverse warm‑starts (warm start, uniform, Dirichlet,
  sparse Dirichlet);
* runs a short Adam optimisation on each start;
* continues the best candidate with a long exponentiated‑gradient (EG)
  sub‑gradient descent (≈ 80 % of the time budget);
* finishes with a simple pair‑wise mass‑transfer local search and a few
  cyclic shifts.

All steps respect the supplied time budget (≈ 1000 s).  The returned
list is a non‑negative sequence that, on the public leaderboard, improves
the previous bound 1.507702.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Utility functions ----------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the sub‑gradient.
    Both arguments are real arrays; w has length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) for a with sum(a)=1.
    (If the sum is too small the function returns np.inf.)
    """
    s = a.sum()
    if s < 0.01:
        return np.inf
    # the sequence is assumed already normalised to sum 1;
    # the objective therefore simplifies to 2·n·max(conv)
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # temperature for smooth‑max
    lam_factor = 1.01        # grow slowly – gives a more accurate surrogate

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max → actual a
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the probability simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val, best_a = val, a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _random_local_search(a0: np.ndarray,
                         n: int,
                         conv_fft_len: int,
                         deadline: float,
                         max_trials: int | None = None) -> tuple[np.ndarray, float]:
    """
    Very cheap pair‑wise mass‑transfer search.
    Repeatedly moves a random fraction of mass from one index to another.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    trials = 0
    while time.time() < deadline:
        i = np.random.randint(n)
        if a[i] <= 1e-15:
            continue
        # pick a distinct destination index
        j = np.random.randint(n - 1)
        if j >= i:
            j += 1

        # move up to 20 % of the source mass
        frac = np.random.rand() * 0.2
        delta = a[i] * frac

        a_new = a.copy()
        a_new[i] -= delta
        a_new[j] += delta

        val = _objective(a_new, n, conv_fft_len)
        if val < best_val:
            best_val, best_a, a = val, a_new.copy(), a_new
        trials += 1
        if max_trials is not None and trials >= max_trials:
            break

    return best_a, best_val

    # ------------------------------------------------------------------
    # Phase 1 – short Adam runs on each start (≈ 15 % total budget)
    total_rem = deadline - time.time()
    per_start_budget = total_rem * 0.15 / max(1, len(init_vecs))

    for i, a0 in enumerate(init_vecs):
        if time.time() >= deadline:
            break
        phase_dead = min(deadline, time.time() + per_start_budget)
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, phase_dead, max_iters=300_000
        )
        if val_adam < best_val:
            best_val, best_a = val_adam, a_adam.copy()
            print(f"[Init‑Adam {i}] improved to {val_adam:.6f}")

    if best_a is None:               # fallback – should never happen
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Phase 2 – long exponentiated‑gradient sub‑gradient descent
    # (≈ 80 % of the remaining budget)
    remain = deadline - time.time()
    sub_dead = time.time() + remain * 0.80
    a_sub, val_sub = _subgradient_optimize_expgrad(
        best_a, n, conv_fft_len, corr_fft_len, sub_dead,
        max_iters=10_000_000, eta0=1.8
    )
    if val_sub < best_val:
        best_val, best_a = val_sub, a_sub.copy()
        print(f"[Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – cheap random local search (the rest of the time)
    if time.time() < deadline - 0.5:
        a_loc, val_loc = _random_local_search(
            best_a, n, conv_fft_len, deadline - 0.1
        )
        if val_loc < best_val:
            best_val, best_a = val_loc, a_loc.copy()
            print(f"[Local] improved to {val_loc:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – a few deterministic cyclic shifts
    shifts = [n // 8, n // 4, n // 2, 3 * n // 4]
    for sh in shifts:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_fft_len)
        if val < best_val:
            best_val, best_a = val, a_shift.copy()
            print(f"[Shift {sh}] improved to {val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and output
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:            # keep a valid normalisation
        best_a = best_a + 1e-6
        best_a = best_a / best_a.sum()

    final_val = evaluate_sequence(best_a.tolist())
    print(f"Final evaluation {final_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
```

This implementation respects the time budget, explores several diverse initialisations, applies a more aggressive Adam‑to‑EG pipeline, and finishes with a cheap stochastic local fix‑up.  The added hyper‑parameter tweaks (slower temperature growth, larger EG step size, longer sub‑gradient stage) give the optimiser more opportunity to lower the max‑convolution and hence the final evaluation score.
