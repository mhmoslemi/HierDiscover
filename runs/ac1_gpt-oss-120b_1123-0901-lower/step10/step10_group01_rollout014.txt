"""
Improved optimizer for the step‑function evaluation.
- Starts from the best known warm‑start (if any) or a random Dirichlet.
- Performs a long Adam run on a smooth‑max surrogate of the max‑convolution.
- Switches to an exact exponential‑gradient sub‑gradient phase.
- Both phases run until the time budget is exhausted, keeping the best vector
  seen.  No external files or network access is used.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT, returns length 2·len(a)‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradient of max‑conv."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the evaluation
    function.  The routine runs for up to ``budget_s`` seconds and returns the
    best sequence it found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start (previous best) if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        print("[Init] warm start used")
    else:
        a = np.random.dirichlet(np.ones(n))
        print("[Init] random start")

    # keep best ever seen
    best_a = a.copy()
    conv_a = _full_convolution(best_a, fft_len)
    best_val = 2.0 * n * conv_a.max()
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam on smooth‑max surrogate
    eps = 1e-15
    logits = np.log(best_a + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    it = 0
    # allocate roughly 60 % of the time to this phase
    phase1_deadline = time.time() + 0.6 * (deadline - time.time())
    while time.time() < phase1_deadline:
        # ---- obtain probability vector from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ---- convolution and objective
        conv_a = _full_convolution(a, fft_len)
        max_c = conv_a.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ---- smooth‑max weights for gradient
        w = np.exp(lam * (conv_a - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ---- gradient of smooth‑max wrt a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # ---- back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ---- Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ---- temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # optional learning‑rate decay
        if (it + 1) % 200_000 == 0:
            lr *= 0.9

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – exact max sub‑gradient with exponentiated‑gradient updates
    a = best_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    it = 0
    while time.time() < deadline:
        conv_a = _full_convolution(a, fft_len)
        max_c = conv_a.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv_a, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv_a)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv_a)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        eta = 0.6 / np.sqrt(it + 1.0)   # step size schedule

        a = a * np.exp(-eta * grad_a)  # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence (clip to allowed range just in case)
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()

"""
Improved optimiser for the step‑function evaluation.

Strategy
--------
1. Warm start – use the previously found best sequence (global variable
   ``height_sequence_1``) if it is present; otherwise start from a random
   Dirichlet vector on the probability simplex.
2. Phase‑1 – Adam optimisation on a smooth‑max surrogate of the
   max‑convolution.  The temperature ``lam`` is increased gradually so that
   the smooth‑max approaches the true max.  The whole run is limited to
   roughly 60 % of the total time budget.
3. Phase‑2 – Exact sub‑gradient descent with exponentiated‑gradient updates
   on the true max‑convolution.  This continues until the deadline.
4. The best vector seen in any phase is clipped to the evaluator’s allowed
   range and returned as a plain ``list[float]``.

Both phases keep the current iterate on the probability simplex
({x≥0, Σx=1}) because the evaluation function is scaling‑invariant.
All heavy work (convolutions, cross‑correlations) is performed with FFTs
for speed.  No external files, network access or hidden state is used.
"""

import time
import numpy as np


def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT; returns length 2·len(a)‑1."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient of max‑conv."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto {x≥0, Σx = 1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the provided
    ``evaluate_sequence`` objective.  The function runs for at most
    ``budget_s`` seconds and returns the best sequence it discovered.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a small safety margin

    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Initialise from warm start (if available) or a random Dirichlet.
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a = np.asarray(warm, dtype=np.float64)
        if a.size > n:
            a = a[:n]
        elif a.size < n:
            a = np.pad(a, (0, n - a.size))
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 0.01:
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s
        print("[Init] warm start used")
    else:
        a = np.random.dirichlet(np.ones(n))
        print("[Init] random start")

    # best solution tracked throughout the run
    best_a = a.copy()
    conv_a = _full_convolution(best_a, fft_len)
    best_val = 2.0 * n * conv_a.max()
    print(f"[Init] objective {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 1 – Adam optimisation on a smooth‑max surrogate.
    eps = 1e-15
    logits = np.log(best_a + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    it = 0
    phase1_deadline = time.time() + 0.6 * (deadline - time.time())
    while time.time() < phase1_deadline:
        # ----- probability vector from logits
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- compute convolution and objective
        conv_a = _full_convolution(a, fft_len)
        max_c = conv_a.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights (soft‑max over the convolution entries)
        w = np.exp(lam * (conv_a - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of smooth‑max w.r.t. a
        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- back‑propagation through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule (approach the true max)
        lam = min(lam * lam_factor, 1e7)

        # optional learning‑rate decay
        if (it + 1) % 200_000 == 0:
            lr *= 0.9

        it += 1

    # ------------------------------------------------------------------
    # Phase 2 – Exact sub‑gradient descent using exponentiated‑gradient updates.
    a = best_a.copy()
    a = np.maximum(a, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    it = 0
    while time.time() < deadline:
        conv_a = _full_convolution(a, fft_len)
        max_c = conv_a.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # ----- active set: indices where the convolution attains its maximum
        active = np.isclose(conv_a, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv_a)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv_a)

        grad_a = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- exponentiated‑gradient step (step size decays with √t)
        eta = 0.6 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    # ------------------------------------------------------------------
    # Return the best sequence encountered (clip to the allowed range).
    best_a = np.clip(best_a, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
