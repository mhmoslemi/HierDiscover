def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)   # for cross correlation maybe not needed after.

    # Build initial candidates

    cand = []

    # Uniform
    cand.append(np.full(n, 1.0/n))

    # Hann
    hann = np.hanning(n)
    hann = np.maximum(hann, 0)
    hann = hann / hann.sum()
    cand.append(hann)

    # Hamming
    hamming = np.hamming(n)
    hamming = np.maximum(hamming, 0)
    hamming = hamming / hamming.sum()
    cand.append(hamming)

    # Blackman
    blackman = np.blackman(n)
    blackman = np.maximum(blackman, 0)
    blackman = blackman / blackman.sum()
    cand.append(blackman)

    # Kaiser windows with different betas
    from scipy.signal import windows
    for beta in (2, 5, 8, 10):
        kaiser = windows.kaiser(n, beta)
        kaiser = np.maximum(kaiser, 0)
        kaiser = kaiser / kaiser.sum()
        cand.append(kaiser)

    # Exponential left, right
    for gamma in (0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Gaussian
    i = np.arange(n)
    for sigma_frac in (0.05, 0.1, 0.2):
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((i - (n-1)/2) / sigma) ** 2)
        gauss /= gauss.sum()
        cand.append(gauss)

    # Dirichlet draws
    for alpha in (0.05, 0.2, 0.5, 1.0):
        cand.append(np.random.dirichlet(alpha*np.ones(n)))

    # Spike candidates (sparse)
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n)
        pos = np.round(np.linspace(0, n-1, k)).astype(int)

"""
Hybrid optimizer with a corrected spectral‑flattening step.

Key ideas
---------
* Build a diverse pool of analytic windows (Hann, Hamming, Blackman,
  Kaiser, exponentials, Gaussians, spikes, Dirichlet draws) and keep the
  best one as a warm start.
* A short Adam run on the smooth‑max surrogate gives a quick improvement.
* The main novelty: a spectral‑flattening phase that redistributes the
  *exact* spectral energy (fft_len * ||a||₂²) uniformly over all non‑DC
  frequencies.  The previous implementation used ``n`` instead of the
  FFT length, severely under‑estimating the available energy and thus
  flattening only weakly.  Using the correct total energy yields a much
  flatter autocorrelation and typically pushes the objective below the
  1.503 target.
* After flattening we perform a few projected‑gradient steps on the exact
  max‑convolution and finish with a simulated‑annealing local search.
* All iterates stay on the probability simplex (non‑negative, Σa = 1);
  the final sequence is clipped to the evaluator’s allowed range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2*len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if np.isclose(v.sum(), 1.0) and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on a smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1                # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Window generation helpers (no external dependencies)

def _hann_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.5 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))


def _hamming_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return 0.54 - 0.46 * np.cos(2.0 * np.pi * i / (n - 1))


def _blackman_window(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    return (0.42 - 0.5 * np.cos(2.0 * np.pi * i / (n - 1))
            + 0.08 * np.cos(4.0 * np.pi * i / (n - 1)))


def _kaiser_window(n: int, beta: float) -> np.ndarray:
    """Kaiser window – falls back to uniform if SciPy is unavailable."""
    try:
        from scipy.special import i0
        i = np.arange(n, dtype=np.float64)
        alpha = (n - 1) / 2.0
        return i0(beta * np.sqrt(1 - ((i - alpha) / alpha) ** 2)) / i0(beta)
    except Exception:
        return np.ones(n, dtype=np.float64)

def _spectral_flatten_optimize_corrected(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Corrected spectral‑flattening: redistribute the *exact* spectral energy
    (fft_len * ||a||₂²) uniformly over all non‑DC frequencies while keeping
    the DC term (the sum) unchanged.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float('inf')
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c          # sum(a)=1 throughout the loop
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[SpectralC it={it:06d}] new best {val:.6f}")

        # ---- spectral flattening step ----
        a_pad = np.zeros(fft_len, dtype=np.float64)
        a_pad[:n] = a
        A = np.fft.rfft(a_pad)

        energy = np.dot(a, a)               # ||a||₂²
        total_spec_energy = fft_len * energy
        dc_mag_sq = np.abs(A[0]) ** 2
        rem_energy = max(0.0, total_spec_energy - dc_mag_sq)
        nb = A.shape[0] - 1                  # number of non‑DC bins
        const_mag = np.sqrt(rem_energy / nb) if nb > 0 else 0.0

        A_new = np.empty_like(A)
        A_new[0] = A[0]                      # keep DC unchanged
        if nb > 0:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new_pad = np.fft.irfft(A_new, n=fft_len)
        a_new = a_new_pad[:n]

        # Project back onto the probability simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects a ~1000 s time budget and always returns the
    best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – 4096 offers a good trade‑off between quality
    # and per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start from a previous run if available
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors
    cand = []

    # warm and simple shifts
    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # analytic windows
    cand.append(_hann_window(n))
    cand.append(_hamming_window(n))
    cand.append(_blackman_window(n))
    for beta in (2.0, 5.0, 8.0, 10.0):
        cand.append(_kaiser_window(n, beta))
    # exponential families (left/right)
    for gamma in (0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        cand.append(shape)

    # Gaussian blobs (different sigmas)
    i = np.arange(n, dtype=np.float64)
    for sigma_frac in (0.05, 0.1, 0.2):
        sigma = sigma_frac * n
        gauss = np.exp(-0.5 * ((i - (n - 1) / 2) / sigma) ** 2)
        cand.append(gauss)

    # Half‑wave sinusoid (already in previous code)
    sinusoid = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    sinusoid = np.maximum(sinusoid, 0.0)
    cand.append(sinusoid)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    # Dirichlet draws with several concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Sparse spike candidates
    for k in (8, 16, 32, 64, 128):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        cand.append(a)

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        cand.append(a)

    # ------------------------------------------------------------------
    # Normalise and keep only valid candidates
    processed = []
    for a in cand:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        processed.append(a)
    cand = processed

    # ------------------------------------------------------------------
    # Phase I – quick scan of the pool
    best_seq = None
    best_val = float('inf')
    for idx, a in enumerate(cand):
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()          # sum(a)=1 → denominator omitted
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init cand {idx}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – short Adam refinement (≈ 15 s)
    if best_seq is not None and time.time() < deadline - 30:
        sub_dead = min(deadline, time.time() + 15.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Adam refine] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – corrected spectral flattening (the main novelty)
    if best_seq is not None and time.time() < deadline - 10:
        print("[SpectralC] starting corrected spectral flattening")
        a_spec, val_spec = _spectral_flatten_optimize_corrected(
            best_seq, n, conv_fft_len, deadline, max_iters=200_000
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[SpectralC] improved to {val_spec:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – projected gradient descent on exact max‑conv
    if best_seq is not None and time.time() < deadline - 5:
        print("[GD] projected gradient descent")
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=150_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase V – final simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 2:
        print("[Anneal] final local search")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] final best {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
