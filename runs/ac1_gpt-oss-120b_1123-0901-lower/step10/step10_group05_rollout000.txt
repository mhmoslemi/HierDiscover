def project_simplex(v):
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Gradient‑based smoothed max‑convolution optimizer.

The routine builds a diverse pool of starter probability vectors
(including the previous warm‑start if present), selects the best one,
and then runs a projected Adam optimizer on a smooth approximation of the
max‑convolution objective.
The smoothness is controlled by a soft‑max temperature that is increased
during the run, so the algorithm first sees a well‑behaved landscape and
gradually focuses on the true max entry.
A brief simulated‑annealing polishing phase follows.
The returned list is non‑negative, summed to a positive value and clipped
to the evaluator’s limits.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex
    {x | x_i ≥ 0, sum_i x_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = ind[cond].max()
    theta = cssv[cond].max() / rho
    w = np.maximum(v - theta, 0.0)
    return w

    while time.time() < deadline - 2.0 and it < max_iters:
        # linear schedule for the soft‑max temperature
        beta = beta_start + (beta_end - beta_start) * (it / max_iters)

        # convolution a * a (exact)
        conv = _full_convolution(best_seq, conv_fft_len)

        # soft‑max weights (smooth approximation of the active set)
        max_c = conv.max()
        # to avoid overflow we shift by the maximum value
        w_exp = np.exp(beta * (conv - max_c))
        w = w_exp / w_exp.sum()

        # gradient of the smooth objective: 2 * (w ★ a)
        grad = 2.0 * _cross_correlation(w, best_seq, corr_fft_len)

        # Adam update (note: we *minimize* the objective → descent step)
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad ** 2)

        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps)

        a_new = best_seq - step
        a_proj = _project_simplex(a_new)

        # enforce symmetry – the optimal solution is known to be symmetric
        a_sym = 0.5 * (a_proj + a_proj[::-1])
        a_sym = _project_simplex(a_sym)   # keep on the simplex

        # evaluate
        conv_new = _full_convolution(a_sym, conv_fft_len)
        val_new = 2.0 * n * conv_new.max()

        if val_new < global_best_val:
            global_best_val = val_new
            global_best_seq = a_sym.copy()
            print(f"[Iter {it:06d}] new best {val_new:.6f}")

        # accept the move if it does not worsen the current point
        if val_new <= best_val:
            best_seq = a_sym
            best_val = val_new
        else:
            # occasional probabilistic acceptance to escape tiny pits
            if rng.random() < 0.0005:
                best_seq = a_sym
                best_val = val_new

        it += 1

    # ------------------------------------------------------------------
    # 4️⃣  Very short simulated‑annealing polish (helps escape remaining pits)
    if time.time() < deadline - 1.0:
        a = global_best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_local_seq = a.copy()
        best_local_val = val

        T_start = 0.07
        T_end   = 1e-5
        max_moves = 80_000
        for mv in range(max_moves):
            if time.time() > deadline - 0.5:
                break
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = rng.integers(n)
            if a[i] < 1e-14:
                continue
            j = rng.integers(n)
            if i == j:
                continue

            # move a random fraction (up to 40 %) of mass from i to j
            delta = a[i] * (rng.random() * 0.4)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_local_val:
                best_local_val = val2
                best_local_seq = a2.copy()
                a = a2
                val = val2
                print(f"[Anneal {mv}] improved to {val2:.6f}")
            else:
                if rng.random() < np.exp(-(val2 - val) / T):
                    a = a2
                    val = val2

        if best_local_val < global_best_val:
            global_best_val = best_local_val
            global_best_seq = best_local_seq.copy()
            print(f"[Anneal final] best {global_best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final clean‑up
    final_seq = np.clip(global_best_seq, 0.0, 1000.0)
    print(f"Final best value {global_best_val:.6f} (length {len(final_seq)})")
    return final_seq.tolist()
