    # Stage: Local random refinement (SA)
    if best_a is not None and time.time() < deadline - 0.5:
        best_a, best_val = _local_refine(best_a, n, conv_fft_len, deadline)
        print(f"[Refine] finished with best {best_val:.6f}")

    # Ensure valid return
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
    best_a = np.clip(best_a, 0.0, 1000.0)
    # optionally scale to avoid extremely small sum? sum >=0.01 re ensure
    s = best_a.sum()
    if s < 0.01:
        best_a = best_a + (0.01 - s) / n
    return best_a.tolist()

"""
Hybrid optimizer for the max‑convolution objective.

The routine works in four phases, always respecting a global time budget
(≈ 1000 s).  A pool of random Dirichlet start vectors (including a warm‑start
if available) is first screened to obtain a decent incumbent.  Then an
exponentiated‑gradient sub‑gradient method minimizes the exact
max‑convolution (the objective 2·n·max(a*a) / (∑a)²).  A short “spectral‑flatten’’
step forces the magnitude of the DFT of the vector to be (almost) constant,
which tends to equalise the autocorrelation.  Finally a simulated‑annealing
local search repeatedly moves a tiny fraction of mass between two coordinates
and accepts improving moves (or uphill moves with a decaying temperature).
All iterates are kept on the probability simplex, guaranteeing non‑negative
entries and a total sum ≥ 0.01 (the evaluation function is homogeneous, so we
normalise to sum = 1).  The best vector found before the deadline is returned
as a plain Python list.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a using FFT (length 2·len(a)‑1).
    Returns the first 2·len(a)‑1 entries of the circular convolution.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the sub‑gradient.
    Both arguments are real; the result has length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact value of 2·n·max(a*a) (sum a is assumed to be 1)."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
def _subgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the exact max‑convolution.
    Operates on the simplex (∑a = 1, a ≥ 0).  Returns the best iterate seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices attaining the current maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if not active.any():
            # numerical safety – treat everything as active
            active = np.ones_like(conv, dtype=bool)

        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()

        # sub‑gradient of the max‑convolution
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # exponentiated‑gradient update on the simplex
        eta = 0.5 / np.sqrt(it + 1.0)   # diminishing step size
        factor = np.exp(-eta * grad)
        # avoid overflow/under‑flow
        factor = np.where(np.isfinite(factor), factor, 0.0)
        a = a * factor
        s = a.sum()
        if s < 1e-12 or not np.isfinite(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the magnitude of the DFT of the vector to be constant
    (except for the DC component).  After each inverse FFT the vector is
    projected back onto the probability simplex.  The best iterate is kept.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # monitor objective
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # ---- spectral flattening step ----
        A = np.fft.fft(a)                     # full complex FFT (length n)
        energy = np.dot(a, a)                 # ||a||_2^2

        # Preserve the DC term (sum = 1).  The remaining energy is spread uniformly.
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                       # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # back to the simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or not np.isfinite(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _local_refine(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing style local search.
    At each step a random fraction of mass is moved from one coordinate to
    another while keeping the total sum unchanged.  Moves are accepted if they
    improve the objective, otherwise with probability exp(‑Δ / T) where T
    cools linearly to a small floor.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)
    cur_val = best_val
    cur_a = a.copy()

    # temperature schedule – start a small fraction of the current gap
    init_T = max(1e-6, (best_val - 1.0) * 0.05)
    start = time.time()
    while time.time() < deadline:
        elapsed = time.time() - start
        total = deadline - start
        # linear cooling to a tiny final temperature
        T = max(1e-8, init_T * (1.0 - elapsed / total))

        i = np.random.randint(n)
        if cur_a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a small fraction of the mass at i to j
        eps = cur_a[i] * np.random.rand() * 0.05   # ≤ 5 % of a[i]
        if eps <= 0.0:
            continue

        new_a = cur_a.copy()
        new_a[i] -= eps
        new_a[j] += eps

        # evaluate
        new_val = _objective(new_a, n, conv_fft_len)
        delta = new_val - cur_val
        if delta < 0.0 or np.random.rand() < np.exp(-delta / T):
            cur_a = new_a
            cur_val = new_val
            if new_val < best_val:
                best_val = new_val
                best_a = new_a.copy()

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # safety margin – leave a short buffer before the hard deadline
    deadline = time.time() + budget_s - 0.5

    # ------------------------------------------------------------------
    # Problem size – 4096 gives a good quality/price trade‑off.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_a: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1️⃣ Warm start (if a previous run produced `height_sequence_1`).
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            val = _objective(w, n, conv_fft_len)
            best_a = w
            best_val = val
            print(f"[Init] Warm start value {val:.6f}")

    # ------------------------------------------------------------------
    # 2️⃣ Random Dirichlet pool – cheap screening.
    pool_size = 120
    alphas = (0.05, 0.2, 0.5, 1.0, 2.0, 5.0)
    for _ in range(pool_size):
        if time.time() > deadline:
            break
        alpha = np.random.choice(alphas)
        a = np.random.dirichlet(alpha * np.ones(n))
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a
            print(f"[Init] Random seed improved to {val:.6f}")

    if best_a is None:
        # Fallback – uniform distribution.
        best_a = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_a, n, conv_fft_len)

    # ------------------------------------------------------------------
    # 3️⃣ Sub‑gradient exponentiated‑gradient phase (main optimisation).
    sub_dead = min(deadline, time.time() + 0.60 * (deadline - time.time()))
    best_a, best_val = _subgrad_optimize(
        best_a, n, conv_fft_len, corr_fft_len, sub_dead
    )
    print(f"[Subgrad] Completed – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣ Spectral flattening (optional – consume a fraction of the remaining time).
    if time.time() < deadline - 10.0:
        spec_dead = min(deadline, time.time() + 0.20 * (deadline - time.time()))
        best_a, best_val = _spectral_flatten_optimize(
            best_a, n, conv_fft_len, spec_dead
        )
        print(f"[Spectral] Completed – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final local refinement (simulated annealing style).
    if time.time() < deadline - 2.0:
        ref_dead = deadline
        best_a, best_val = _local_refine(best_a, n, conv_fft_len, ref_dead)
        print(f"[Refine] Completed – best value {best_val:.6f}")

    # ------------------------------------------------------------------
    # 6️⃣ Return a valid list (clip to the evaluator's allowed range).
    if best_a is None:
        best_a = np.full(n, 1.0 / n, dtype=np.float64)

    best_a = np.clip(best_a, 0.0, 1000.0)

    # sanity: make sure the sum is not too small (evaluate_sequence would reject)
    if best_a.sum() < 0.01:
        # add a tiny uniform mass
        best_a += (0.01 - best_a.sum()) / n

    print(f"Final best value {best_val:.6f} (length {len(best_a)})")
    return best_a.tolist()
