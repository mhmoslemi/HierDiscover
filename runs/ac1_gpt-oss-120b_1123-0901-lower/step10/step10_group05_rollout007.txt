def proj_simplex(v):
    # ensures nonnegative sum=1
    # Based on algorithm by Duchi et al.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond].max()
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid deterministic‑annealing optimizer for the max‑self‑convolution problem.

1. Build a rich pool of starter vectors (uniform, Dirichlet draws,
   Beta‑shapes – including negative exponents, power‑law ends‑heavy,
   piece‑wise constant blocks, Gaussian bumps).  All are normalised to
   unit sum and the best one is kept.

2. Refine the best starter with a *soft‑max* (log‑sum‑exp) smoothing of the
   max‑convolution.  For a temperature parameter β we minimise

        L_β(a) = (1/β)·log Σ_k exp(β·(a*a)_k)

   whose gradient is
        ∇L_β = 2·(w ★ a)   with   w_k ∝ exp(β·(a*a)_k).

   An exponentiated‑gradient step (multiplicative update) is performed,
   followed by normalisation and enforced symmetry a_i = a_{n‑1‑i}.
   β is increased gradually (deterministic‑annealing schedule) so the
   optimisation gradually focuses on the true L∞ objective.

3. A short stochastic‑annealing polish moves a random fraction of mass
   between two coordinates, accepting downhill moves always and uphill
   moves with a temperature schedule.

All steps respect the global time budget (≈ 1000 s).  The returned
sequence is non‑negative, summed to 1 (scale‑invariant for the target
evaluator) and clipped to the evaluator’s limits.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Result length = len(a).  Uses FFT of length “fft_len”.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step function that minimises the evaluator."""
    np.random.seed(seed)
    now = time.time()
    deadline = now + budget_s - 0.5          # safety margin for the evaluator

    # ------------------------------------------------------------------
    # Problem size – 4096 works well and keeps the runtime reasonable.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse pool of starter vectors
    pool: list[np.ndarray] = []

    # warm start (if a previous run deposited a sequence)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64).ravel()
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # uniform distribution – baseline
    pool.append(np.full(n, 1.0 / n))

    # Dirichlet draws (different concentration parameters)
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Random Beta‑shapes (allowing mildly negative exponents)
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    eps = 1e-12
    for _ in range(30):
        p = np.random.uniform(-0.8, 5.0)
        q = np.random.uniform(-0.8, 5.0)
        shape = (np.power(x + eps, p) * np.power(1.0 - x + eps, q))
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Power‑law “ends‑heavy” shapes   a_i ∝ (i·(n‑i+1))^α , α<0
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-3.0, -0.2, 12):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Piece‑wise constant random blocks
    for blocks in (8, 16, 32, 64):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
            pool.append(a)

    # Gaussian bumps centred at the middle
    for sigma in (n / 8, n / 6, n / 4):
        i = np.arange(n, dtype=np.float64)
        centre = (n - 1) / 2.0
        shape = np.exp(-0.5 * ((i - centre) / sigma) ** 2)
        shape = shape / shape.sum()
        pool.append(shape)

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of all starters – keep the best one
    best_seq: np.ndarray | None = None

        if s < 0.01:

        a = a / s

        print("[fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Deterministic‑annealing smooth‑max refinement
    a = best_seq.copy()

    beta_schedule = np.concatenate([

        np.linspace(30.0, 80.0, 10)     # almost the true max
    ])

    for beta in beta_schedule:
        if time.time() >= deadline - 3:
            break
        # a few inner iterations per β (adjusted dynamically)
        inner_limit = 2000
        for _ in range(inner_limit):
            if time.time() >= deadline - 3:
                break

            conv = _full_convolution(a, conv_fft_len)
            maxc = conv.max()

            grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

            # ----- exponentiated‑gradient step -----
            step = 0.7 / np.sqrt(total_iters + 1.0)
            a_new = a * np.exp(-step * grad)

            s = a_new.sum()
            if s <= 0.0 or np.isnan(s):
                # numerical trouble – shrink the step and continue
                step *= 0.5
                continue
            a_new = a_new / s

            # ----- enforce exact symmetry (keeps the search within the
            #       subspace that we expect to contain the optimum) -----
            a_new = (a_new + a_new[::-1]) / 2.0
            a_new = a_new / a_new.sum()

            # ----- evaluate -----
            conv_new = _full_convolution(a_new, conv_fft_len)
            val_new = 2.0 * n * conv_new.max()
            if val_new < best_val - 1e-12:
                best_val = val_new
                best_seq = a_new.copy()
                print(f"[beta {beta:.1f} it {total_iters:06d}] new best {val_new:.6f}")

            a = a_new
            total_iters += 1

    # ------------------------------------------------------------------
    # 4️⃣  Very short stochastic‑annealing polish
    if time.time() < deadline - 2:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        best_val_sa = val
        best_a_sa = a.copy()

        T_start = 0.07
        T_end = 1e-6
        max_moves = 70000
        for mv in range(max_moves):
            if time.time() > deadline - 1:

            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            # move a random fraction (up to 30 %) of mass from i to j
            delta = a[i] * (np.random.rand() * 0.3)
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa - 1e-12:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2

            best_seq = best_a_sa.copy()

    # 5️⃣  Final clean‑up and return
    if best_seq is None:

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
