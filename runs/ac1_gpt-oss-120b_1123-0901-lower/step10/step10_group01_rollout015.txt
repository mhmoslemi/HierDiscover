def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    n = 4096
    # warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] != n:
            if warm.shape[0] > n:
                warm = warm[:n]
            else:
                warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    if warm is None:
        # default to uniform
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a = warm.copy()
    # ensure sum constraints
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    # compute convolution
    conv = np.convolve(a, a)
    best_val = 2.0 * n * conv.max()
    best_seq = a.copy()
    print(f"Initial value {best_val:.6f}")

    # Local search
    max_iters = 2_000_000   # limit
    it = 0
    # dynamic step size parameters
    max_step = 0.2
    min_step = 0.01
    while time.time() < deadline and it < max_iters:
        # Choose source index with largest a
        i = int(np.argmax(a))
        # Choose target index with smallest a; ensure not same
        j = int(np.argmin(a))
        if i == j:
            # choose random different
            j = (j + 1) % n
        # Choose delta
        step_factor = max_step * (1 - it / max_iters) + min_step
        delta = a[i] * step_factor * np.random.rand()
        if delta <= 1e-12:
            it += 1
            continue
        # Apply move to conv copy
        conv_new = conv.copy()
        # linear part
        conv_new[i:i+n] -= 2.0 * delta * a
        conv_new[j:j+n] += 2.0 * delta * a
        # quadratic corrections
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        conv_new[idx2i] += delta * delta
        conv_new[idx2j] += delta * delta
        conv_new[idxij] -= 2.0 * delta * delta
        # Clip to non-negative just in case
        # (optional)
        # conv_new = np.maximum(conv_new, 0.0)

        new_val = 2.0 * n * conv_new.max()
        if new_val < best_val - 1e-12:   # improvement significant
            # accept
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_val = new_val
            best_seq = a.copy()
            if it % 1000 == 0:
                print(f"[Local it={it}] improved to {best_val:.6f}")
        # optionally, if not improved, we could accept with temperature but skip
        it += 1

    # Ensure final sequence compliance
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    # After modifications sum should stay 1; re-normalize for safety
    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full(n, 1.0 / n)
    else:
        best_seq = best_seq / total
    print(f"Final best value {best_val:.6f}")
    return best_seq.tolist()

"""
Local‑search optimiser for the max‑convolution objective.

The evaluation function
    E(a) = 2·n·max_k (a*a)[k]      with Σ a = 1 , a_i ≥ 0
is scale‑invariant.  Starting from the best known sequence (the global
variable ``height_sequence_1`` – if it exists) we perform a very fast
coordinate‑wise hill climbing:

* pick the index i with the largest coefficient and the index j with the
  smallest coefficient,
* move a small amount δ from i to j,
* update the full convolution a*a **incrementally** using only O(n)
  vector operations (no FFT inside the loop),
* keep the move only if the objective is improved.

The step size δ is a fraction of a_i and decays linearly with the
iteration count, allowing a coarse global search at the beginning and a
fine‑grained refinement later.  The whole routine respects a user‑supplied
time budget (default 1000 s) and returns the best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1)."""
    return np.convolve(a, a)


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    `evaluate_sequence`.  Uses a simple hill‑climbing move that transfers a
    small amount of mass from a large coefficient to a small one, updating
    the convolution incrementally.  The algorithm starts from the previously
    best sequence (global ``height_sequence_1``) if available, otherwise from
    a uniform distribution.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem size – 4096 works well for the benchmark
    n = 4096

    # ------------------------------------------------------------------
    # Warm start: use the previous best if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            a = warm / warm.sum()
        else:
            a = None
    else:
        a = None

    # If no valid warm start, fall back to a uniform vector
    if a is None:
        a = np.full(n, 1.0 / n, dtype=np.float64)

    # ------------------------------------------------------------------
    # Initialise convolution and best‑so‑far bookkeeping
    conv = _full_convolution(a)                     # length 2n‑1
    best_val = 2.0 * n * conv.max()                # current objective
    best_seq = a.copy()
    print(f"[Init] objective = {best_val:.6f}")

    # ------------------------------------------------------------------
    # Hill‑climbing parameters
    max_iters = 2_500_000                         # hard cap; the time limit stops us earlier
    it = 0
    max_step = 0.18                               # initial maximum fraction of a[i] to move
    min_step = 0.008                              # final minimum fraction
    report_every = 10_000

    # ------------------------------------------------------------------
    # Main local‑search loop
    while time.time() < deadline and it < max_iters:
        # 1) choose source (largest coefficient) and target (smallest)
        i = int(np.argmax(a))
        j = int(np.argmin(a))
        if i == j:            # should never happen, guard anyway
            j = (j + 1) % n

        # 2) step size – linearly decaying fraction of a[i]
        frac = max_step * (1.0 - it / max_iters) + min_step
        delta = a[i] * frac * np.random.rand()   # 0 ≤ δ ≤ frac·a[i]

        if delta < 1e-14:     # nothing to move – skip
            it += 1
            continue

        # 3) incremental update of the convolution
        conv_new = conv.copy()
        # linear part: 2·δ·(shift_j(a) – shift_i(a))
        conv_new[i:i + n] -= 2.0 * delta * a
        conv_new[j:j + n] += 2.0 * delta * a
        # quadratic correction (δ² on the three affected positions)
        idx2i = 2 * i
        idx2j = 2 * j
        idxij = i + j
        conv_new[idx2i] += delta * delta
        conv_new[idx2j] += delta * delta
        conv_new[idxij] -= 2.0 * delta * delta

        # 4) evaluate the candidate objective
        cand_val = 2.0 * n * conv_new.max()

        # 5) accept only improving moves (greedy descent)
        if cand_val < best_val - 1e-12:
            # update the current state
            a[i] -= delta
            a[j] += delta
            conv = conv_new
            best_val = cand_val
            best_seq = a.copy()

            if it % report_every == 0:
                print(f"[Iter {it:07d}] improved to {best_val:.6f}")

        it += 1

    # ------------------------------------------------------------------
    # Final clean‑up: ensure non‑negativity, normalise, and store globally
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    total = best_seq.sum()
    if total < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        best_seq = best_seq / total

    # make the sequence available for the next run
    globals()["height_sequence_1"] = best_seq.tolist()

    print(f"Final objective {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
