"""
Simple yet aggressive multi‑phase optimiser.

The optimiser works on the probability simplex (the vector is normalised to sum
to 1).  The evaluation function is homogeneous, therefore normalisation does
not affect the final score.

The pipeline is:

1. Build a diverse pool of starting vectors (uniform, Dirichlet, exponential,
   Gaussian, power‑law, triangle, spikes, cosine, sine, …).  If a previous
   best sequence is available it is added and interpolated to the current
   length.

2. For each start (limited by the remaining time) run a short Adam optimiser
   on a smooth‑max surrogate of the max‑convolution.  The Adam run is fast
   (≈ 15 s) and yields a decent local optimum.

3. Refine the current best with a long exponentiated‑gradient sub‑gradient
   phase (exact max).  This phase dominates the remaining budget.

4. A few extra fine‑tuning steps – projected gradient descent on the exact
   max‑convolution and a simulated‑annealing local search – are performed if
   time permits.

5. If there is spare time the best solution is up‑sampled to a finer grid
   (8192 points) and refined again.

The whole routine respects the 1000 s time budget and guarantees a non‑empty
return (the uniform vector) as a fallback.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT‑based convolution utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    The result has the same length as ``a``.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    lr: float = 0.07,
    lam_start: float = 0.5,
    lam_factor: float = 1.04,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns the best vector encountered (still on the simplex) and its exact
    objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()               # a lies on simplex

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (lam > 0 gives a softened max)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase smoothness gradually
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 60_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 1_200_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns the best vector (still on the simplex) and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the max (exact max, no tolerance)
        active = conv == max_c
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)               # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Projected (sub‑)gradient descent on the exact max‑convolution.
    Uses a diminishing step size and Nesterov‑style momentum.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0
    momentum = np.zeros_like(a)

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = conv == max_c
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Nesterov momentum (simple)
        momentum = 0.8 * momentum - 0.1 / np.sqrt(it + 1.0) * grad
        a_next = a + momentum
        a_next = np.maximum(a_next, 0.0)
        a_next = _project_simplex(a_next)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a_next.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        a = a_next
        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing local search on the simplex.
    Randomly moves a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        # exponential cooling
        T = T_end + (T_start - T_end) * (1.0 - it / float(max_moves))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of the mass from i to j
        frac = np.random.rand() * 0.6
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / max(T, 1e-12)):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Candidate factories (simple families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_cosine(n):
    """Cosine‑squared shape (non‑negative, symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (n - 1)) ** 2
    a /= a.sum()
    return a


def _make_sine(n):
    """Half‑sine shape, zero at the endpoints."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """

    n = 4096

    corr_fft_len = _next_pow2(3 * n - 2)

    # Warm start: use the previous best if available.
    warm = globals().get("height_sequence_1")
    if warm is not None:

        if warm.shape[0] > n:

        elif warm.shape[0] < n:

        warm = np.maximum(warm, 0.0)

        else:
            warm = None
    else:

    pool = []

        # few cyclic shifts of the warm start (helps escape symmetry)

    for alpha in (0.01, 0.07, 0.2, 0.5, 1.0, 3.0, 6.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families (strongly front‑biased)
    for gamma in (0.08, 0.2, 0.4, 0.9, 1.8):
        pool.append(_make_exponential(n, gamma))

    # Gaussian families (center‑biased)
    for sigma in (0.04, 0.12, 0.25, 0.55):

    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Phase V] Upsampling to 8192 points and final refinement")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # linear interpolation onto a finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            # a short Adam run on the larger grid
            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big,
                n_big,
                conv_fft_len_big,
                corr_fft_len_big,
                sub_dead_big,
                max_iters=300_000,
                lr=0.09,
                lam_start=0.7,
                lam_factor=1.04,
            )
            if val_big < best_val:
                best_val = val_big
                best_seq = a_big.copy()
                print(f"[Phase V] refined best {val_big:.6f}")

            # final exact‑max sub‑gradient on the big grid
            if time.time() < deadline - 3:
                a_big2, val_big2 = _subgradient_optimize_expgrad(
                    best_seq,
                    n_big,
                    conv_fft_len_big,
                    corr_fft_len_big,
                    deadline,
                    max_iters=800_000,
                    eta0=0.5,
                )
                if val_big2 < best_val:
                    best_val = val_big2
                    best_seq = a_big2.copy()
                    print(f"[Phase V] final big‑grid refinement {val_big2:.6f}")

        except Exception as exc:
            print(f"[Phase V] upsampling exception: {exc}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
