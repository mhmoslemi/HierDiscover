"""
Hybrid optimiser with expanded families and adaptive scheduling.

The goal is to minimise
    2·n·max(conv(a,a)) / (sum(a)²)

which, after normalising a to a probability vector (sum = 1),
is simply 2·n·max(conv(a,a)).  The optimiser works on the
probability simplex, uses several initial families (including
asymmetric exponentials and logistic decays), and runs a
pipeline of:

* Adam on a smooth‑max surrogate,
* Exponentiated‑gradient sub‑gradient descent,
* Projected gradient descent on the exact max,
* Simulated‑annealing fine‑tuning,
* A final long refinement stage.

All stages respect the global time budget (≈ 1000 s) and keep the
best sequence found.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
#  Low‑level utilities (no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
#  Candidate factories (simple families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    """Symmetric exponential (both ends)."""
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_exponential_one_sided(n, gamma):
    """Monotonically decreasing exponential, normalised."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-gamma * i)
    a /= a.sum()
    return a


def _make_logistic(n, beta, shift):
    """Logistic decay: 1/(1+exp(beta*(i-shift)))"""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - shift)))
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    """Power‑law a_i = (i+1)^expo, normalised."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
#  Optimisers (soft‑max Adam, exponentiated‑gradient, GD, anneal)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=400_000, lr0=0.07, lam0=0.01,
                   lam_factor=1.015, lr_decay=0.97):
    """
    Adam optimiser on the smooth‑max surrogate of max(conv).
    Returns (best_a, best_val) where best_val = 2·n·max(conv).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = lr0
    lam = lam0

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= lr_decay

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0, n, conv_fft_len, corr_fft_len,
                                 deadline, max_iters=500_000,
                                 eta0=1.0):
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns (best_a, best_val).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 30_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0, n, conv_fft_len, corr_fft_len,
                     deadline, max_iters=200_000):
    """
    Projected gradient descent on the exact max‑convolution.
    Returns (best_a, best_val).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize(a0, n, conv_fft_len,
                    deadline, max_moves=150_000):
    """
    Simulated‑annealing local search on the simplex.
    Returns (best_a, best_val).
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

#  Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Returns the best sequence found within the allotted time.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # tiny safety margin

    # problem size – 4096 works well for the existing helper functions
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (if a previous solution is present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a rich pool of start vectors

    pool = []

    if warm is not None:
        pool.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Symmetric families (as used before)
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))

    # -------- new asymmetric families ---------------------------------
    for gamma in (0.05, 0.1, 0.2, 0.4, 0.8, 1.5):
        pool.append(_make_exponential_one_sided(n, gamma))

    for beta in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0):
        for shift in (n // 3, n // 2, 2 * n // 3):
            pool.append(_make_logistic(n, beta, shift))

    for expo in (-0.3, -0.5, -0.8, -1.2, -1.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))

    # Random sparse spike patterns
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    # Completely random Dirichlet draws
    for _ in range(12):
        alpha = 10 ** np.random.uniform(-3, 1)   # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq = None
    best_val = float("inf")
    max_candidates = 30          # we have many starts, but we will trim later

    # ------------------------------------------------------------------
    # Phase I – quick Adam sweeps on a large number of starts
    print("[Phase I] quick Adam passes")
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # give each start at most 8 s of Adam time (or less if we run out)
        sub_dead = min(deadline, time.time() + 8.0)

        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=300_000, lr0=0.07, lam0=0.01,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase II – deeper optimisation of the current best
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] deep Adam refinement")
        sub_dead = min(deadline, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=600_000, lr0=0.06, lam0=0.005,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Deep Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase III – exponentiated‑gradient sub‑gradient phase
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] long ExpGrad")
        sub_dead = min(deadline, time.time() + 120.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000, eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[ExpGrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase IV – projected gradient descent on the exact max
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase IV] projected GD")
        sub_dead = min(deadline, time.time() + 120.0)
        a_gd, val_gd = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=500_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[GD] improved to {val_gd:.6f}")

    # ------------------------------------------------------------------
    # Phase V – simulated‑annealing refinement
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase V] annealing")
        sub_dead = min(deadline, time.time() + 120.0)
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, sub_dead,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final long sub‑gradient pass (very cheap, runs until time runs out)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Final] long ExpGrad")
        a_ref, val_ref = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=1_200_000, eta0=0.5,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Final ExpGrad] improved to {val_ref:.6f}")

    # ------------------------------------------------------------------
    # Guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
