assistantfinal**Solution Overview**

The goal is to return a non‑negative list `a` that minimises  

\[
\text{score}(a)=\frac{2\,n\; \max\limits_{k}(a*a)[k]}{\bigl(\sum_i a_i\bigr)^2},
\]

where `a*a` is the linear convolution of `a` with itself.
The value does **not** depend on an overall scaling of `a`; only the *shape* of the
distribution matters.

The previous implementation already explores many parametric families and then
optimises the best one with Adam‑smooth‑max and a sub‑gradient scheme.
To push the bound further we add three main ideas

1. **Larger support** – using `n = 8192` (instead of 4096) gives a finer
   discretisation and a theoretical lower bound of `≈ 1.0001`.  The extra
   points are cheap to handle because the convolution is performed with an FFT.

2. **Broader and finer sweep** – we generate many more candidates, keep the
   *k* best of them (`k = 8`) and start a dedicated optimisation from each.
   This reduces the chance of being trapped in a poor local minimum.

3. **Restartable sub‑gradient descent** – after a first long sub‑gradient phase
   we perturb the current best solution, re‑normalise and continue.  The
   optimiser therefore performs several “restarts’’ inside the remaining time
   budget.

The programme respects the 1000 s budget, prints progress, stores the best
vector in the global variable `height_sequence_1` (so that later runs can warm‑start
from it) and finally returns the list of coefficients.

```python
"""
Step‑function optimiser (improved version).

* Uses a longer support (n = 8192) – the theoretical lower bound is ≈ 1.0001.
* Performs a very wide parametric sweep and keeps the 8 best candidates.
* Runs a short Adam smoothing on each of those candidates.
* Executes a restartable exponentiated‑gradient sub‑gradient descent that
  repeatedly perturbs the current best solution.
* Finishes with a cheap simulated‑annealing fine‑tuner.
All work is bounded by the supplied time budget (default 1000 s) and the
function always returns a valid non‑negative list.
"""

import time
import numpy as np
import heapq

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² – exact, no clipping."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Families of candidate step functions -----------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_harmonic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (i + 1) + 1.0 / (n - i)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** power
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max -----------------------------------------

def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 600_000) -> tuple[np.ndarray, float]:
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.2
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent on logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional gentle learning‑rate decay
        if it % 150_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimiser (restartable) ---------

def _subgradient_restartable(a0: np.ndarray, n: int,
                            conv_fft_len: int, corr_fft_len: int,
                            deadline: float,
                            eta0: float = 1.3,
                            restart_interval: int = 30_000) -> tuple[np.ndarray, float]:
    """
    Performs exponentiated‑gradient sub‑gradient descent.
    Every `restart_interval` iterations the current best solution is
    perturbed multiplicatively (log‑normal noise) and the descent continues.
    The whole loop stops when the global deadline is reached.
    """
    best = a0.copy()
    best_val = _objective(best, n)

    a = best.copy()
    it = 0
    while time.time() < deadline:
        # full convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best = a.copy()
            print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:  # should not happen, but keep a safe gradient
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad_a)

        # renormalise, protect against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # restart with a cheap perturbation
        if (it + 1) % restart_interval == 0:
            # multiplicative log‑normal noise around the current best
            noise = np.random.lognormal(mean=0.0, sigma=0.2, size=n)
            a = best * noise
            a = np.maximum(a, 0.0)
            a /= a.sum()

        it += 1

    return best, best_val


# ----------------------------------------------------------------------
# Tiny simulated‑annealing fine‑tuner ----------------------------------

def _anneal_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                     deadline: float, max_moves: int = 150_000) -> tuple[np.ndarray, float]:
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 8e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a step‑function with a low evaluation score.

    The routine works in four phases:
    1. Very broad parametric sweep (keeps the 8 best candidates).
    2. Short Adam‑smooth‑max refinement on each of those candidates.
    3. Long exponentiated‑gradient sub‑gradient descent with periodic restarts.
    4. Optional simulated‑annealing fine tuning.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4   # small safety margin

    # ----- problem size -------------------------------------------------
    n = 8192                       # larger support → better theoretical bound
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # keep the k best candidates from the sweep
    top_k = 8
    top_heap = []                  # min‑heap with entries (-value, vector)

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val, top_heap
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new global best {val:.6f}")
        if len(top_heap) < top_k:
            heapq.heappush(top_heap, (-val, a.copy()))
        else:
            # heap[0] holds the worst (largest negative) value
            if val < -top_heap[0][0]:
                heapq.heapreplace(top_heap, (-val, a.copy()))

    # ------------------------------------------------------------------
    # 1) Extensive parametric sweep ------------------------------------
    # symmetric families
    for p in np.linspace(-6.0, -0.1, 40):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-4, 3.0, 45):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 14)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # monotone families
    for alpha in np.logspace(-4, 2.5, 35):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-6.0, 2.0, 35):
        _maybe_update(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 14):
        for alpha in np.logspace(-4, 1.8, 30):
            _maybe_update(_candidate_left_stretched(alpha, beta, n))

    # Gamma‑like
    for shape in np.linspace(0.2, 4.0, 15):
        for scale in np.logspace(-4, 0.5, 15):
            _maybe_update(_candidate_gamma(shape, scale, n))

    # Beta‑distribution (strong left skew → useful)
    for a_par in np.logspace(-2, 0.5, 15):
        for b_par in np.logspace(0, 1.0, 15):
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # Double Gaussian bumps
    for sig in np.logspace(-3, -0.2, 18):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # Harmonic family (new)
    _maybe_update(_candidate_harmonic(n))

    # Sine‑power family (new)
    for power in np.linspace(0.5, 4.0, 16):
        _maybe_update(_candidate_sine(power, n))

    # Random spikes
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Mixtures of two exponentials
    for _ in range(250):
        alpha1 = 10 ** np.random.uniform(-4, 2)
        alpha2 = 10 ** np.random.uniform(-4, 2)
        w1 = np.random.rand()
        mix = w1 * _candidate_left_exponential(alpha1, n) + \
              (1 - w1) * _candidate_left_exponential(alpha2, n)
        mix /= mix.sum()
        _maybe_update(mix)

    # Dirichlet draws with various concentration parameters
    for conc in [0.01, 0.05, 0.1, 0.3, 0.6, 1.0, 2.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # Warm start from previous run (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement on every top candidate -------------------------
    # give each candidate up to ~6 s (or whatever time is left)
    remaining = deadline - time.time()
    if remaining > 10 and top_heap:
        per_candidate = min(6.0, remaining / max(1, len(top_heap)))
        new_top = []
        while top_heap and time.time() < deadline - 2:
            neg_val, cand = heapq.heappop(top_heap)
            cand_deadline = min(deadline, time.time() + per_candidate)
            a_adam, val_adam = _adam_optimize(
                cand, n, conv_fft_len, corr_fft_len, cand_deadline,
                max_iters=800_000,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Adam] improved global best to {val_adam:.6f}")
            new_top.append((-val_adam, a_adam.copy()))
        # rebuild heap from the refined candidates (keep at most top_k)
        top_heap = heapq.nsmallest(top_k, new_top)

    # ------------------------------------------------------------------
    # 3) Long restartable sub‑gradient phase -----------------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] long restartable sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_restartable(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            eta0=1.3,
            restart_interval=25_000,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved global best to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional annealing fine‑tune ------------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine‑tuner")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved global best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation -------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Global variable for possible warm‑starts in future runs
    global height_sequence_1
    height_sequence_1 = best_seq.tolist()

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
```
