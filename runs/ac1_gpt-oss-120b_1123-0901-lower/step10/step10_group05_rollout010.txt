def evaluate_sequence(sequence):
    # checks: list, non-empty, each element int/float, no nan/inf, convert to float, clip negatives to 0, clip max to 1000.
    n = len(sequence)

"""
Hybrid multi‑start smoothed exponentiated‑gradient optimizer.

* Build a large diversified pool of starter vectors (warm‑start, Beta
  densities, power‑law ends‑heavy shapes, Dirichlet draws, random spikes,
  block‑constant vectors).  All vectors are projected to the simplex
  (non‑negative, sum = 1) and evaluated with the fast FFT‑based objective

        Obj(a) = 2·n·max(a∗a)          (n = len(a),   sum(a)=1)

  The best few candidates (default 3) are kept for refinement.

* For each kept candidate run a smoothed mirror‑descent (exponentiated‑gradient)
  loop.  At iteration *t* we form a temperature β(t) and replace the hard
  active‑set weights of the max‑convolution by a softmax
  `w_i ∝ exp(conv_i / β)`.  The gradient of the smoothed objective is

        grad = 2·( w ★ a )                (★ = linear cross‑correlation)

  The update is `a ← a * exp(‑η·grad)`, followed by a projection onto the
  probability simplex.  The step size η decays like 1/√t, while β is
  geometrically cooled (β←β·γ).  All convolutions are performed with
  FFTs, re‑using pre‑computed lengths.

* After the gradient phase a short simulated‑annealing polish moves a random
  fraction of mass between two coordinates, accepting uphill moves with a
  linear temperature schedule.

The whole routine respects the supplied time budget (≈ 1000 s) and always
returns the best sequence found.  The returned list is non‑negative,
sums to ≈ 1, and is clipped to the evaluator’s [0, 1000] range.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for the gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) assuming sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _softmax_weights(conv: np.ndarray, beta: float) -> np.ndarray:
    """Return w_i ∝ exp(conv_i / beta).  If beta≤0 use active‑set uniform."""
    if beta <= 0.0:
        mx = conv.max()
        mask = np.isclose(conv, mx, atol=1e-14)
        w = np.zeros_like(conv)
        cnt = mask.sum()
        if cnt > 0:
            w[mask] = 1.0 / cnt
        return w
    # stable softmax around the maximum
    mx = conv.max()
    exp_arg = (conv - mx) / beta
    w = np.exp(exp_arg)
    s = w.sum()
    if s == 0.0:
        w = np.ones_like(conv) / conv.size
    else:
        w = w / s
    return w


def _refine_candidate(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Smoothed exponentiated‑gradient refinement.

    Returns the best vector seen and its value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    step_const = 0.7          # initial step‑size multiplier
    beta = 0.02               # initial temperature for the softmax
    beta_min = 1e-5
    beta_decay = 0.985        # geometric cooling per 5000 iterations

    it = 0
    while time.time() < deadline - 2.0 and it < max_iters:
        # current convolution
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # soft‑max weights
        w = _softmax_weights(conv, beta)

        # gradient of the smoothed objective
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # diminishing step size
        step = step_const / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        factor = np.exp(-step * grad)
        a = a * factor
        s = a.sum()
        if s <= 1e-14 or np.isnan(s):
            a = best_a.copy()
            s = a.sum()
        a = a / s

        # temperature schedule
        if it % 5000 == 0 and beta > beta_min:
            beta = max(beta * beta_decay, beta_min)

        it += 1

    return best_a, best_val


def _anneal_polish(
    a0: np.ndarray,
    n: int,

    deadline: float,
    max_moves: int = 80_000,
) -> tuple[np.ndarray, float]:
    """
    Very short simulated‑annealing polish (mass transfers).
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    T_start = 0.06

        if time.time() > deadline - 1.0:

        # linear temperature schedule
        T = max(T_end, T_start * (1.0 - mv / max_moves))

        i = np.random.randint(n)
        if a[i] < 1e-14:

        j = np.random.randint(n)

            # up‑sample / down‑sample linearly to the target length
            x_old = np.linspace(0, 1, w.shape[0])
            x_new = np.linspace(0, 1, n)
            w = np.interp(x_new, x_old, w)
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # ---- parametric families -----------------------------------------
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)

    # Beta‑shape (including a fine random search)
    for p in np.linspace(0.05, 2.5, 12):
        for q in np.linspace(0.05, 2.5, 12):
            shape = (x ** (p - 1)) * ((1.0 - x) ** (q - 1))
            if shape.sum() > 0:
                shape = shape / shape.sum()
                pool.append(shape)

    # random Beta draws
    for _ in range(30):
        p = 10 ** np.random.uniform(-2.0, 0.6)   # roughly 0.01 … 4
        q = 10 ** np.random.uniform(-2.0, 0.6)
        shape = (x ** (p - 1)) * ((1.0 - x) ** (q - 1))
        shape = shape / shape.sum()
        pool.append(shape)

    # power‑law ends‑heavy (i·(n‑i+1))^α
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-5.0, -0.1, 30):
        shape = prod ** alpha
        if shape.sum() > 0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Dirichlet draws (dense)
    for alpha in (0.02, 0.05, 0.1, 0.2, 0.5, 1.0):
        pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Random sparse spikes – many of them
    for k in (256, 512, 1024):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
        pool.append(a)

    # Piece‑wise constant blocks
    for blocks in (32, 64, 128):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        a = a / a.sum()
        pool.append(a)

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of all starters – keep the best few
    best_seq: np.ndarray | None = None
    best_val = float("inf")
    top_candidates: list[tuple[np.ndarray, float]] = []   # (vector, value)

    for i, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{i:04d}] new global best {val:.6f}")
        # keep a shortlist of the most promising vectors
        top_candidates.append((a, val))
    # keep only the k best distinct vectors
    top_candidates.sort(key=lambda x: x[1])
    top_candidates = top_candidates[:3]  # few for refinement
    print(f"Initial best value {best_val:.6f} (length {n}) – refining {len(top_candidates)} candidates.")

    # ------------------------------------------------------------------
    # 3️⃣  Smoothed exponentiated‑gradient refinement on the shortlist
    for idx, (cand, val0) in enumerate(top_candidates):
        if time.time() > deadline - 10.0:
            break
        print(f"[Refine start] candidate {idx} with start value {val0:.6f}")
        refined, refined_val = _refine_candidate(
            cand, n, conv_fft_len, corr_fft_len, deadline
        )
        if refined_val < best_val:
            best_val = refined_val
            best_seq = refined.copy()
            print(f"[Refine done] candidate {idx} improved to {refined_val:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Very short simulated‑annealing polish
    if best_seq is not None and time.time() < deadline - 2.0:
        print("[Anneal start] polishing the current best.")
        a_polished, val_polished = _anneal_polish(best_seq, n, conv_fft_len, deadline)
        if val_polished < best_val:
            best_val = val_polished
            best_seq = a_polished.copy()
            print(f"[Anneal done] improved to {val_polished:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final clean‑up and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
