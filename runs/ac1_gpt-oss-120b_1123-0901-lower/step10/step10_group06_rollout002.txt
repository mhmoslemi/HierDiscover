def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same conv length)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]

# Exact objective (scale‑invariant)

def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate
    `log‑sum‑exp(λ·conv)`.  The smoothing parameter λ grows exponentially.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters (stays on the simplex)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.05            # start temperature
    lam_factor = 1.03

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----------- soft‑max projection ------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----------- exact objective for bookkeeping ------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----------- smooth‑max weights ------------
        # w ∝ exp(λ·(conv - max_c))  – numerically stable
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----------- gradient of surrogate ------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----------- mirror‑descent (soft‑max) gradient ------------
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----------- Adam update ------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----------- temperature schedule ------------
        lam = min(lam * lam_factor, 1e7)

        # ----------- learning‑rate decay (occasional) ------------
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                 n: int,
                                 conv_fft_len: int,
                                 corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 eta0: float = 0.85) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient descent with exponentiated‑gradient (mirror‑descent)
    updates.  An exponential moving average of past gradients is kept to
    obtain a quasi‑adaptive step‑size (Adam‑like momentum).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    # moments for Adam‑style scaling (only first moment is used)
    grad_ema = np.zeros_like(a)

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # ----------- active set (indices attaining the max) ------------
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # ----------- sub‑gradient ------------
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # EMA of gradient (momentum)
        gamma = 0.9
        grad_ema = gamma * grad_ema + (1.0 - gamma) * grad_a

        # ----------- step‑size (decaying) ------------
        eta = eta0 / np.sqrt(it + 1.0)

        # ----------- exponentiated‑gradient update on the simplex ------------
        a = a * np.exp(-eta * grad_ema)

        # ----------- renormalisation ------------
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 400_000) -> tuple[np.ndarray, float]:
    """
    Simulated‑annealing fine‑tuning.  The temperature decays
    exponentially from `T_start` to `T_end`.  Each move redistributes a
    random fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        # exponential cooling
        T = T_end + (T_start - T_end) * math.exp(-5.0 * it / max_moves)

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction of i's mass towards j
        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            # Metropolis acceptance
            if np.random.rand() < math.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Elite set management
# ----------------------------------------------------------------------
def _update_elite(elite: list, val: float, a: np.ndarray, max_size: int = 12):
    """Insert (val, a) into elite list (kept sorted by val, low → high)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

# Main entry point
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # Problem dimension – keep a power of two for fast FFT
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Broad parametric sweep + random permutations
    # --------------------------------------------------------------
    ELITE_SIZE = 12
    elite = []  # [(value, vector)]

    def _maybe_update(a):
        val = _objective(a, n)
        # early cut‑off: discard if clearly worse than current worst elite
        if elite and len(elite) == ELITE_SIZE and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a, max_size=ELITE_SIZE)

    # symmetric power‑law (negative exponents concentrate mass at the ends)
    for p in np.linspace(-6.0, -0.05, 40):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.2, 40):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 16)
    betas = np.linspace(0.3, 3.5, 16)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # Beta distributions – wide exploration of shape parameters
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0, 20.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # Double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 28):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # Monotone left‑exponential
    for alpha in np.logspace(-5, 1.3, 28):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # Monotone left‑power (allows negative exponents)
    for p in np.linspace(-4.5, 2.5, 36):
        _maybe_update(_candidate_left_power(p, n))

    # Inverse‑power (edges heavy)
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_inverse_power(p, n))

    # Random spike patterns (different sparsities)
    for k in [16, 32, 64, 128, 256, 512]:
        for _ in range(6):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Dirichlet draws with various concentration parameters
    concs = [0.02, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for c in concs:
        _maybe_update(_candidate_dirichlet(c, n))

    # ------------------------------------------------------------------
    # Random permutations of the current elite vectors (break symmetry)
    # ------------------------------------------------------------------
    perm_reps = 5
    for _, vec in elite[:ELITE_SIZE // 2]:   # only the very best few
        for _ in range(perm_reps):
            perm = np.random.permutation(n)
            a_perm = vec[perm]
            # re‑normalise (tiny numerical drift may appear)
            a_perm = np.maximum(a_perm, 0.0)
            if a_perm.sum() < 1e-12:
                continue
            a_perm /= a_perm.sum()
            _maybe_update(a_perm)

    # ------------------------------------------------------------------
    # Warm‑start from a previously stored global (if present)
    # ------------------------------------------------------------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates – using uniform")

    # --------------------------------------------------------------
    # 2) Adam optimisation on smooth‑max (multiple starts)
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_budget = min(30.0, 0.25 * remaining)
        adam_deadline = time.time() + adam_budget
        print("[Phase‑Adam] short Adam optimisation (multiple starts)")
        # we start from the top `ADAM_STARTS` elites
        ADAM_STARTS = min(4, len(elite))
        for idx in range(ADAM_STARTS):
            if time.time() >= adam_deadline:
                break
            a_start = elite[idx][1]
            a_adam, val_adam = _adam_optimize(
                a_start, n, conv_fft_len, corr_fft_len, adam_deadline,
                max_iters=400_000
            )
            if val_adam < best_val:
                best_val, best_seq = val_adam, a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Long exact‑max exponentiated‑gradient sub‑gradient descent
    # --------------------------------------------------------------
    if time.time() < deadline - 6.0:
        sub_deadline = deadline - 6.0
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,
            sub_deadline, max_iters=12_000_000, eta0=0.80
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 4) Exhaustive simulated‑annealing fine‑tuning (as much time as left)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        ann_deadline = deadline
        moves_allowed = int(6e5)  # generous number if time permits
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, ann_deadline, max_moves=moves_allowed
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # --------------------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if it is available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final sequence (evaluator missing).")

    return best_seq.tolist()
