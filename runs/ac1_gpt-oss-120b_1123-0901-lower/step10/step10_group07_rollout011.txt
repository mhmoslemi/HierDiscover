assistantfinal**Solution Overview**

The goal is to minimise  

\[
\phi(a)=\frac{2\,n\;\max (a*a)}{(\sum_i a_i)^2},
\]

where `a*a` is the linear convolution of the non‑negative vector `a`.  
The scaling of `a` does not change the score, therefore we can keep
`∑a = 1` and minimise `2·n·max(a*a)`.

The programme follows three stages  

1. **Extensive parametric sweep** – many analytical families of
   step‑functions are sampled (power‑law, exponential, stretched‑exponential,
   beta‑distribution, sinusoid, cosine‑squared, centred‑exponential,
   centred‑Gaussian,…). The best vector found is kept.

2. **Local optimisation of the family parameters** – the best family
   from the sweep is refined with a small‑dimensional bound‑constrained
   optimisation (`scipy.optimize.minimize` with L‑BFGS‑B).  
   This usually reduces the score by a few ‰.

3. **Stochastic refinement** – a handful of random Dirichlet
   initialisations are improved with two complementary gradient‑based
   methods  

   * **Adam optimisation** on a smooth‑max surrogate (log‑sum‑exp) while
     staying on the simplex via a soft‑max parametrisation, and  

   * **Exponentiated‑gradient sub‑gradient descent** on the exact
     max‑convolution.  

   The best result of all runs is finally polished with a short
   simulated‑annealing pass.

All steps respect the 1000 s budget and the code never returns an
invalid list – the sequence is always non‑negative and renormalised
before the final call to `evaluate_sequence`.

```python
"""
Step‑function optimiser for the evaluation function:
    2·n·max(a*a) / (sum(a))**2

The algorithm proceeds in three phases:

1. A dense parametric sweep over many analytic families (power‑law,
   exponential, stretched‑exponential, beta, sinusoid, cosine‑squared,
   centred‑exponential, centred‑Gaussian, etc.).  The best candidate
   is stored.

2. Local refinement of the parameters of the best family with a
   bound‑constrained optimiser (L‑BFGS‑B).  The objective is the exact
   2·n·max‑convolution score, computed by FFT.

3. Stochastic improvement: several random Dirichlet starts are
   refined first by a short Adam run on a smooth‑max surrogate and
   afterwards by an exponentiated‑gradient sub‑gradient method.
   The final best sequence is polished by a brief simulated‑annealing
   local search.

All work respects the supplied time budget (default 1000 s) and the
returned list is always a valid, non‑negative step‑function that can be
fed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for the gradient of max‑convolution.
    Result length = len(a) and corresponds to the derivative of
    ``max(w·(a*a))`` w.r.t. ``a`` when ``w`` is an indicator of the
    active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)².
    ``a`` is assumed to be non‑negative and already normalised
    (∑a = 1); the normalisation is not enforced here for speed.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families ----------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Power‑law a_i ∝ (i+1)^p + (n‑i)^p (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution on [0,1] (skew allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_sinusoid(n: int) -> np.ndarray:
    """Positive sinusoid (zero at the edges, peak in the middle)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n)          # sin runs from ≈0 to 1 back to ≈0
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_cosine_squared(n: int) -> np.ndarray:
    """Cosine‑squared bump: a_i = sin²(π·i/(n‑1))."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * i / (n - 1)) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_center_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential centred in the middle."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-alpha * np.abs(i - mid))
    a /= a.sum()
    return a


def _candidate_center_gaussian(sig_frac: float, n: int) -> np.ndarray:
    """Symmetric Gaussian centred in the middle.
       sig_frac is the standard deviation as a fraction of n."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    sigma = sig_frac * n
    a = np.exp(-0.5 * ((i - mid) / sigma) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate while
    staying on the simplex via the soft‑max parametrisation
    ``a = softmax(logits)``.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # full convolution (exact objective for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional decay of learning rate
        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient / mirror descent on the simplex).
    Step size ηₜ = η₀ / √(t+1).  Stops when the deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numeric loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that moves a random fraction
    of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Parameter refinement --------------------------------------------------

def _refine_parameters(
    func,
    init_params,
    bounds,
    n,
    deadline,
) -> tuple[np.ndarray, float]:
    """
    Small‑dimensional bound‑constrained optimisation of the parameters
    of a family ``func`` (signature ``func(n, *params)``) using
    L‑BFGS‑B.  If the time budget is exhausted, the original parameters
    are returned unchanged.
    """
    remaining = deadline - time.time()
    if remaining < 5.0:
        return None, np.inf

    maxfun = 200  # a few hundred objective calls are enough

    def obj(p):
        if time.time() > deadline:
            return np.inf
        seq = func(n, *p)
        return _objective(seq, n)

    try:
        res = minimize(
            obj,
            np.array(init_params, dtype=np.float64),
            method="L-BFGS-B",
            bounds=bounds,
            options={"maxfun": maxfun, "disp": False},
        )
        if not res.success:
            # still try the result – it may be better than the start
            pass
        best_seq = func(n, *res.x)
        best_val = _objective(best_seq, n)
        return best_seq, best_val
    except Exception as e:
        print("[Refine] optimisation failed:", e)
        return None, np.inf


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.  It combines a dense parametric
    sweep, local parameter optimisation, stochastic Dirichlet restarts
    refined by Adam and exponentiated‑gradient sub‑gradients, and a final
    simulated‑annealing polishing phase.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safe margin

    # -------------------------- problem size ---------------------------
    # Use n = 4096 (fast enough for many evaluations) – the algorithm
    # works for any n, replace with a larger value if the budget permits.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # -------------------------- bookkeeping ----------------------------
    best_seq = None
    best_val = np.inf
    best_info = None          # (func, params) for the current best candidate

    def _maybe_update(a: np.ndarray, info):
        nonlocal best_seq, best_val, best_info
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            best_info = info
            print(f"[Sweep] new best {val:.6f}")

    # -------------------------- 1) Parametric sweep --------------------
    print("[Phase 1] Parametric sweep started")

    # power‑law
    for p in np.linspace(-4.5, -0.01, 80):
        _maybe_update(_candidate_power(p, n), ("power", (p,)))

    # exponential
    for alpha in np.logspace(-5, 1.0, 120):
        _maybe_update(_candidate_exponential(alpha, n), ("exp", (alpha,)))

    # stretched exponential
    alphas = np.logspace(-4, 0.6, 30)
    betas = np.linspace(0.2, 3.0, 25)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n), ("stretched", (alpha, beta)))

    # beta distributions – search a grid with finer resolution
    a_vals = np.geomspace(0.2, 15.0, 30)
    b_vals = a_vals
    for a_par in a_vals:
        for b_par in b_vals:
            _maybe_update(_candidate_beta(a_par, b_par, n), ("beta", (a_par, b_par)))

    # sinusoid (no parameters)
    _maybe_update(_candidate_sinusoid(n), ("sinusoid", ()))

    # cosine‑squared bump
    _maybe_update(_candidate_cosine_squared(n), ("cos_sq", ()))

    # centred exponential
    for alpha in np.logspace(-5, 1.0, 80):
        _maybe_update(_candidate_center_exponential(alpha, n), ("center_exp", (alpha,)))

    # centred Gaussian (sigma expressed as a fraction of n)
    for sig_frac in np.logspace(-3.5, -0.3, 50):
        _maybe_update(_candidate_center_gaussian(sig_frac, n), ("center_gauss", (sig_frac,)))

    # random spikes – a few different sparsities
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True), ("spike", (k,)))

    # pure Dirichlet draws
    for _ in range(20):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand, ("rand_dirichlet", ()))

    # warm start from a possible previous global variable
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, ("warm", ()))

    print(f"[Phase 1] best after sweep: {best_val:.6f}")

    # ----------------------- 2) Refine parameters ----------------------
    if best_info is not None and best_info[0] in {
        "power", "exp", "stretched", "beta",
        "center_exp", "center_gauss"
    }:
        func_name, params = best_info
        # map to the actual function and bounds
        if func_name == "power":
            func = _candidate_power
            bounds = [(-6.0, -0.001)]
        elif func_name == "exp":
            func = _candidate_exponential
            bounds = [(1e-6, 10.0)]
        elif func_name == "stretched":
            func = _candidate_stretched
            bounds = [(1e-6, 10.0), (0.1, 5.0)]
        elif func_name == "beta":
            func = _candidate_beta
            bounds = [(0.05, 30.0), (0.05, 30.0)]
        elif func_name == "center_exp":
            func = _candidate_center_exponential
            bounds = [(1e-6, 10.0)]
        elif func_name == "center_gauss":
            func = _candidate_center_gaussian
            bounds = [(1e-4, 0.8)]
        else:
            func = None
            bounds = None

        if func is not None:
            print("[Phase 2] Local optimisation of family parameters")
            refined_seq, refined_val = _refine_parameters(
                func,
                list(params),
                bounds,
                n,
                deadline,
            )
            if refined_seq is not None and refined_val < best_val:
                best_val = refined_val
                best_seq = refined_seq.copy()
                best_info = (func_name, refined_seq)   # keep for later use
                print(f"[Phase 2] improved to {refined_val:.6f}")

    # ------------------------ 3) Adam polishing (if time) ------------
    if time.time() < deadline - 30:
        print("[Phase 3] Adam polishing of current best")
        sub_deadline = time.time() + min(20.0, deadline - time.time() - 10.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase 3] Adam improved to {val_adam:.6f}")

    # ------------------- 4) Stochastic Dirichlet restarts ------------
    print("[Phase 4] Stochastic Dirichlet restarts")
    remaining = deadline - time.time()
    max_restarts = 6
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        concentration = float(np.random.choice([0.1, 0.3, 0.7, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 22.0)

        # Adam phase (short)
        a_adam, _ = _adam_optimize(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline - 5.0,
            max_iters=200_000,
        )

        # Sub‑gradient phase (long)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a_adam,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline,
            max_iters=1_500_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")

        remaining = deadline - time.time()

    # ------------------- 5) Final sub‑gradient refinement ------------
    if time.time() < deadline - 3:
        print("[Phase 5] Final long sub‑gradient run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase 5] improved to {val_sub:.6f}")

    # ------------------- 6) Simulated‑annealing polishing ------------
    if time.time() < deadline - 1:
        print("[Phase 6] Final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase 6] improved to {val_ann:.6f}")

    # ------------------- final sanity checks -------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final evaluation through the provided security‑checked function
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
```
