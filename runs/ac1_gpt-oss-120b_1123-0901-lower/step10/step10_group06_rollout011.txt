def _hill_climb(a, n, conv_fft_len, deadline, max_steps=100000):
    best = a.copy()
    best_val = _objective(best, n)
    while time.time() < deadline and max_steps>0:
        i = np.random.randint(n)
        j = np.random.randint(n)
        if i==j or best[i] < 1e-12: continue
        # random transfer fraction
        frac = np.random.rand() * 0.1
        delta = best[i] * frac
        a2 = best.copy()
        a2[i] -= delta
        a2[j] += delta
        # ensure sum unchanged
        val = _objective(a2, n)
        if val < best_val:
            best = a2
            best_val = val
            # optionally print improvement
        max_steps -= 1
    return best, best_val

def _candidate_sidon_spike(num_spikes, n):
    sums = set()
    positions = []
    # random order of indices
    perm = np.random.permutation(n)
    for idx in perm:
        # compute new sums with existing positions
        ok = True
        for p in positions:
            if idx + p in sums or 2*idx in sums:
                ok = False
                break
        if ok:
            # add idx
            for p in positions:
                sums.add(idx + p)
            sums.add(2*idx)
            positions.append(idx)
            if len(positions) >= num_spikes:
                break
    # construct a vector
    a = np.zeros(n)
    if positions:
        w = 1.0/len(positions)
        a[positions] = w
        # ensure sum=1 after normalization
        a /= a.sum()
    return a

"""
propose_candidate
-----------------

Search for a non‑negative step‑function (list of heights) that gives a low
value of the public ``evaluate_sequence`` score.

The algorithm proceeds in four stages:

1. **Broad parametric sweep** – generate a wide variety of deterministic
   shapes (power‑law, exponential, stretched‑exponential, beta, cosine,
   triangular, double‑Gaussian) and a few random Dirichlet / spike
   patterns.  Each candidate is normalised to unit sum and the exact
   objective 2·n·max(conv)/(sum²) is evaluated with an FFT‑based
   convolution.  The K best candidates (K=8) are kept.

2. **Sidon‑like spike constructions** – a greedy algorithm builds a set
   of indices whose pairwise sums collide rarely (exact Sidon or
   “low‑collision” sets).  The corresponding sparse vectors are also
   added to the elite pool.

3. **Multiple exponentiated‑gradient sub‑gradient runs** – for every
   elite start we run the exact‑max sub‑gradient descent (multiplicative
   update on the simplex) for a time slice of the remaining budget.
   The step size η decays as η₀/√t (η₀≈0.6).  The best vector found over
   all runs is kept.

4. **Final local fine‑tuning** – a short hill‑climbing loop moves a
   random fraction of mass between two positions whenever the move
   reduces the objective.

All phases respect the global time budget (default 1000 s).  The routine
always returns a valid list of non‑negative floats that can be fed to
``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant, no clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    if a.sum() < 1e-12:
        return np.inf
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (unit‑sum, non‑negative)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_cosine(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.cos(np.pi * i / (n - 1)) ** 2
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_triangular(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = 1.0 - np.abs(2.0 * i / (n - 1) - 1.0)
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p with p < 0 (edges heavy)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_sidon_spikes(target_size: int, n: int) -> np.ndarray:
    """
    Greedy construction of an (almost) Sidon set of indices.
    Returns a sparse vector with equal mass on the selected positions.
    """
    sums = set()
    positions = []
    order = np.random.permutation(n)
    for idx in order:
        ok = True
        # ensure every new sum is new
        for p in positions:
            if idx + p in sums:
                ok = False
                break
        if ok and (2 * idx) not in sums:
            # accept index
            for p in positions:
                sums.add(idx + p)
            sums.add(2 * idx)
            positions.append(idx)
            if len(positions) >= target_size:
                break
    a = np.zeros(n, dtype=np.float64)
    if positions:
        w = 1.0 / len(positions)
        a[positions] = w
        a /= a.sum()
    return a


def _candidate_low_collision_spikes(target_size: int, n: int, max_coll: int = 2) -> np.ndarray:
    """
    Greedy construction where each possible sum may appear at most
    ``max_coll`` times.  Useful when a true Sidon set of the requested
    size cannot exist.
    """
    sum_cnt = {}
    positions = []
    order = np.random.permutation(n)
    for idx in order:
        ok = True
        # check pairwise sums
        for p in positions:
            s = idx + p
            if sum_cnt.get(s, 0) >= max_coll:
                ok = False
                break
        if not ok:
            continue
        # self‑sum
        s = 2 * idx
        if sum_cnt.get(s, 0) >= max_coll:
            continue
        # accept
        for p in positions:
            s = idx + p
            sum_cnt[s] = sum_cnt.get(s, 0) + 1
        sum_cnt[2 * idx] = sum_cnt.get(2 * idx, 0) + 1
        positions.append(idx)
        if len(positions) >= target_size:
            break
    a = np.zeros(n, dtype=np.float64)
    if positions:
        w = 1.0 / len(positions)
        a[positions] = w
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[SubGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max, within a tight tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise – guard against numerical loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _hill_climb(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very simple hill‑climbing: move a random fraction of mass between two indices."""
    a = a0.copy()
    best_val = _objective(a, n)
    it = 0
    while time.time() < deadline and it < max_moves:
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.12  # modest transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        val2 = _objective(a2, n)
        if val2 < best_val:
            a = a2
            best_val = val2
        it += 1
    return a, best_val


# ----------------------------------------------------------------------
# Utility for elite handling
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns a valid
    list of heights (float).  The implementation combines a broad parametric
    sweep, Sidon‑set spike constructions, multiple exponentiated‑gradient
    sub‑gradient runs and a final hill‑climbing refinement.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size – a power of two is fast for FFTs
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # 1) Wide parametric sweep – keep a handful of elite candidates
    # --------------------------------------------------------------
    elite = []   # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Deterministic families
    for p in np.linspace(-5.0, -0.05, 25):
        _maybe_update(_candidate_power(p, n))
    for alpha in np.logspace(-5, 1.0, 25):
        _maybe_update(_candidate_exponential(alpha, n))
    alphas = np.logspace(-4, 0.5, 9)
    betas = np.linspace(0.3, 3.0, 9)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_par in a_pars:
        for b_par in a_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))
    for sig in np.logspace(-3, -0.1, 16):
        _maybe_update(_candidate_double_gaussian(sig, n))
    for alpha in np.logspace(-5, 1.2, 16):
        _maybe_update(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-4.0, 2.0, 22):
        _maybe_update(_candidate_left_power(p, n))
    for p in np.linspace(-4.5, -0.1, 22):
        _maybe_update(_candidate_inverse_power(p, n))
    _maybe_update(_candidate_cosine(n))
    _maybe_update(_candidate_triangular(n))

    # Random Dirichlet draws (different concentration parameters)
    for conc in [0.02, 0.05, 0.1, 0.2, 0.5]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)

    # Random spike patterns
    for k in [32, 64, 96, 128, 192, 256]:
        for _ in range(3):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Sidon‑like spike sets (exact Sidon up to the feasible size)
    for target in [48, 56, 64]:
        _maybe_update(_candidate_sidon_spikes(target, n))

    # Low‑collision spike sets (allow up to 2 collisions)
    for target in [80, 96, 112]:
        _maybe_update(_candidate_low_collision_spikes(target, n, max_coll=2))

    # Warm start from a previously stored global (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        else:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    # Pick the best elite after sweep
    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found – using uniform distribution")

    # --------------------------------------------------------------
    # 2) Sub‑gradient optimisation from elite starts
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        # allocate roughly 80 % of the remaining budget to sub‑gradient runs
        sub_budget = remaining * 0.8
        n_restarts = min(len(elite), 5)       # up to five independent runs
        per_run = sub_budget / n_restarts
        for idx in range(n_restarts):
            if time.time() >= deadline:
                break
            start_vec = elite[idx][1]
            sub_deadline = time.time() + per_run
            a_opt, val_opt = _subgradient_optimize_expgrad(
                start_vec,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                max_iters=10_000_000,
                eta0=0.6,
            )
            if val_opt < best_val:
                best_val, best_seq = val_opt, a_opt.copy()
                print(f"[SubGrad‑run {idx}] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Final hill‑climbing fine‑tuning (if time left)
    # --------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[HillClimb] final local refinement")
        a_hc, val_hc = _hill_climb(best_seq, n, conv_fft_len, deadline, max_moves=250_000)
        if val_hc < best_val:
            best_val, best_seq = val_hc, a_hc.copy()
            print(f"[HillClimb] improved to {val_hc:.6f}")

    # --------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final sanity check with the public evaluator (if it exists)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score – evaluator unavailable.")

    return best_seq.tolist()
