def project_simplex(v):
    # ensure nonnegative and sum to 1.
    # Clip negative to zero first? Actually algorithm works with any real.
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    # find rho
    rho = np.nonzero(u - cssv / np.arange(1, n+1) > 0)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Hybrid smooth‑max gradient descent with simplex projection.

* Build a diverse pool of starter vectors (warm start, uniform, power‑law,
  beta‑shapes, geometric, Dirichlet draws, random block patterns).
* Evaluate each candidate with a fast FFT‑based convolution and keep the best.
* Run several restarts of a smooth‑max gradient descent:
    – The true objective  f(a)=2·n·max(a∗a) is non‑smooth.
    – Replace the max by a log‑sum‑exp (smooth_max) with a temperature τ.
    – The gradient of smooth_max is 2·corr(w,a) where w are the softmax
      weights of the convolution entries.
    – An Adam‑style update is performed on the unrestricted variables,
      then the vector is projected onto the probability simplex
      (non‑negative, sum = 1).
    – τ is shrunk gradually so the algorithm approaches the original
      non‑smooth problem.
* A very short simulated‑annealing polish moves a random fraction of mass
  between two positions, accepting uphill moves with a cooling schedule.
* The best sequence (clipped to the evaluator’s limits) is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for the gradient.
    Returns a vector of length len(a) corresponding to the middle part
    of the convolution of w with the reversed a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _smooth_max_and_weights(conv: np.ndarray, tau: float):
    """
    Returns smooth_max = τ·log Σexp(conv_i/τ) and the softmax weights
    w_i = exp(conv_i/τ) / Σexp(conv_i/τ).
    """
    max_c = conv.max()
    # stability: subtract max_c before exponentiation
    shifted = (conv - max_c) / tau
    exp_shifted = np.exp(shifted)
    sum_exp = exp_shifted.sum()
    smooth = max_c + tau * np.log(sum_exp)
    weights = exp_shifted / sum_exp
    return smooth, weights


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of v onto the probability simplex
    {x | x_i ≥ 0, Σx_i = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → return uniform vector
        return np.full(n, 1.0 / n, dtype=v.dtype)
    rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) under the assumption Σa = 1.
    (The sum constraint is enforced externally.)
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # ------------------------------------------------------------------
    # 3️⃣  Smooth‑max gradient descent (multiple restarts)
    # ----------------------------------------------------
    # We work with the normalized simplex (Σa = 1).  The objective to
    # minimise is  f(a)=2·n·max(a*a) .
    # Instead we minimise the smoothed version  f_τ(a)=2·n·smooth_max(conv)
    # and gradually shrink τ.
    max_restarts = 6
    inner_limit = 150_000                     # hard cap for inner iterations
    it_global = 0
    while time.time() < deadline - 10 and max_restarts > 0:
        # ----- initialise a new restart ---------------------------------
        if max_restarts == max_restarts:      # first restart uses current best
            a = best_seq.copy()
        else:
            # random perturbation of the current best
            a = best_seq + 0.02 * np.random.randn(n)
            a = _project_simplex(a)

        # Adam state
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        beta1, beta2 = 0.9, 0.999
        eps = 1e-12
        t = 0
        tau_factor = 0.02                     # initial τ = 0.02 * max_c

        while time.time() < deadline - 2 and t < inner_limit:
            # ----- compute convolution and current max --------------------
            conv = _full_convolution(a, conv_fft_len)
            max_c = conv.max()
            obj = 2.0 * n * max_c               # exact objective (Σa = 1)
            if obj < best_val:
                best_val = obj
                best_seq = a.copy()
                print(f"[GD it={it_global:06d}] new best {obj:.6f}")

            # ----- smooth‑max and its weights ----------------------------
            tau = max(tau_factor * max_c, 1e-8)   # keep τ positive
            _, w = _smooth_max_and_weights(conv, tau)

            # ----- gradient of the smoothed objective --------------------
            # grad_f = 4·n·corr(w,a)
            grad = 4.0 * n * _cross_correlation(w, a, corr_fft_len)

            # ----- Adam update -------------------------------------------
            t += 1
            it_global += 1
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** t)
            v_hat = v / (1.0 - beta2 ** t)

            # learning‑rate schedule (larger at the beginning, decays)
            lr = 0.2 / (np.sqrt(t) + 1.0)

            a = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # ----- simplex projection (keeps Σa = 1, a_i ≥ 0) -----------
            a = _project_simplex(a)

            # gradually tighten τ (anneal)
            if t % 2000 == 0:
                tau_factor *= 0.95
                if tau_factor < 1e-4:
                    tau_factor = 1e-4

        max_restarts -= 1

    # ------------------------------------------------------------------
    # 4️⃣  Very short simulated‑annealing polish
    # ----------------------------------------------------
    if time.time() < deadline - 1:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        best_val_sa = 2.0 * n * conv.max()
        best_a_sa = a.copy()

        T_start = 0.07
        T_end = 1e-5
        max_moves = 80_000
        for mv in range(max_moves):
            if time.time() > deadline - 1:
                break
            # linear cooling
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of mass
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            conv2 = _full_convolution(a2, conv_fft_len)
            val2 = 2.0 * n * conv2.max()

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[Anneal mv={mv}] improved to {val2:.6f}")
            else:
                if np.random.rand() < np.exp(-(val2 - best_val_sa) / T):
                    a = a2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣  Final post‑processing and return
    # ----------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator's limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
