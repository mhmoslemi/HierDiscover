def _spectral_flatten_optimize_long(a, n, fft_len, deadline, eval_interval=100, max_iters=1e9):
    a = np.copy(a)
    best_a = a.copy()
    best_val = 2 * n * np.max(_full_convolution(a, fft_len))
    it = 0
    while time.time() < deadline and it < max_iters:
        # Compute Fourier.
        A = np.fft.fft(a)
        energy = np.dot(a, a)
        rem_energy = max(0.0, n * energy - (abs(A[0])**2))
        if n > 1:
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        # Build new Fourier with constant magnitude.
        A_new = np.empty_like(A)
        A_new[0] = A[0]
        A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # Project onto simplex
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12:
            # Reinitialize random
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        if it % eval_interval == 0:
            conv = _full_convolution(a, fft_len)
            max_c = conv.max()
            val = 2 * n * max_c
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Spectral it={it}] new best {val:.6f}")

        it += 1
    return best_a, best_val
```

def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)

    conv_fft_len = _next_pow2(2*n - 1)

    scored = []

    # allocate time for each candidate spectral flatten
    for i in range(topK):
        if time.time() > deadline - 5:
            break
        val, a_start = scored[i]
        sub_deadline = deadline - 5.0 - (topK - i - 1) * 5.0 # ensure some slack; maybe 5s per later step
        # we also ensure leaving at least 5 seconds for final annealing
        a_opt, val_opt = _spectral_flatten_optimize(a_start, n, conv_fft_len, sub_deadline)

            best_seq = a_opt.copy()

    # final annealing
    if best_seq is not None and time.time() < deadline - 1:
        a_ann, val_ann = _anneal_optimize(best_seq, n, conv_fft_len, deadline, max_moves=100_000)

    # clip and return list
    best_seq = np.clip(best_seq, 0.0, 1000.0)

The algorithm works in four phases under a global time budget
(~1000 s).  It builds a small but diverse pool of seeds (uniform,
Dirichlet draws, block‑wise pieces, sparse spikes and a warm start if
available).  Each seed is evaluated cheaply using the exact convolution
and the best few are refined by a long alternating‑projection phase
that forces the FFT magnitude (except the DC component) to be (almost)
constant – this “spectral flattening’’ minimizes the peak of the
self‑convolution.  The best iterate found is finally polished by a very
light simulated‑annealing move‑based local search.

All vectors are kept on the probability simplex (non‑negative,
∑aᵢ=1) because the evaluation function is homogeneous of degree 0.
The routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        # a few circular shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with several concentration parameters
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Exponential decays from both sides
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        shape = left + left[::-1]
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law shaped vectors
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian bumps
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform distribution
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates made of a handful of equally‑spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out


# ----------------------------------------------------------------------
def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    eval_interval: int = 500,
) -> tuple[np.ndarray, float]:
    """
    Alternating‑projection “spectral flatten’’ iteration.
    Starts from a0 (already on the simplex) and repeatedly:
        1. Enforces a constant magnitude for all non‑DC Fourier coefficients.
        2. Projects back onto the non‑negative simplex.
    The best iterate (lowest 2·n·max(conv)) is returned.
    """
    a = a0.copy()
    best_a = a.copy()
    # initial objective (sum = 1 ⇒ denominator = 1)
    conv = _full_convolution(a, fft_len)
    best_val = 2.0 * n * conv.max()
    it = 0

    while time.time() < deadline:
        # ----- spectral flatten step -----
        A = np.fft.fft(a)                     # full complex FFT
        energy = np.dot(a, a)                 # ℓ₂² of a
        # Parseval: sum|A|² = n * energy
        rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
        const_mag = np.sqrt(rem_energy / (n - 1)) if n > 1 else np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                       # keep DC unchanged
        A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # ----- projection onto the probability simplex -----
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        # ----- keep track of the best value -----
        if it % eval_interval == 0:
            conv = _full_convolution(a, fft_len)
            val = 2.0 * n * conv.max()
            if val < best_val:
                best_val = val
                best_a = a.copy()
                print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # occasional tiny random perturbation to escape shallow traps
        if it % 10000 == 0 and it > 0:
            pert = a + np.random.normal(scale=1e-4, size=n)
            pert = np.maximum(pert, 0.0)
            pert /= pert.sum()
            a = pert

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
) -> tuple[np.ndarray, float]:
    """
    Very lightweight simulated‑annealing move based local search.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - 2.0 * n * conv.max()) / T):
                a = a2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always
    returns the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    # small safety margin to avoid overrunning the external time limit
    deadline = time.time() + budget_s - 0.5

    # Length – 4096 is a good compromise between quality and per‑iteration cost.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # Warm start handling (if a previous run stored a solution)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # ------------------------------------------------------------------
    # Build a modest pool of diverse start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # ------------------------------------------------------------------
    # Quick evaluation of all seeds; keep the best few
    scored = []
    for a in pool:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:
            continue
        a = a / a.sum()
        conv = _full_convolution(a, conv_fft_len)
        val = 2.0 * n * conv.max()
        scored.append((val, a))

    scored.sort(key=lambda x: x[0])
    top_k = min(5, len(scored))                # we will refine only the top K

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – heavy spectral‑flattening on the most promising seeds
    for idx in range(top_k):
        if time.time() > deadline - 10:       # keep a few seconds for final polish
            break
        init_val, a_start = scored[idx]

        # allocate a slice of the remaining budget for this candidate
        sub_deadline = min(deadline - 5.0, time.time() + 30.0)   # ≈30 s per candidate

        a_opt, val_opt = _spectral_flatten_optimize(
            a_start, n, conv_fft_len, sub_deadline, eval_interval=1000
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Refine {idx}] spectral improved to {val_opt:.6f}")

    # ------------------------------------------------------------------
    # Phase II – a final short simulated‑annealing polish (if time permits)
    if best_seq is not None and time.time() < deadline - 1:
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=80_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Anneal] final improvement to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = 2.0 * n * (1.0 / n)  # value of the uniform distribution = 2

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
