Pseudo:

def proj_simplex(v):
    """Project onto the probability simplex of sum 1 and nonnegative."""
    # ensure non-negative after projection.
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

"""
Step‑function optimizer (improved)

The approach builds on the previous implementation and adds a few
enhancements that tend to push the bound a little lower:

1. **Richer parametric sweep** – we add beta‑distributions with very
   small shape parameters, a flexible “gamma‑like’’ family
   a_i ∝ (i+1)^p·exp(‑αi) and a sinusoidal family.
2. **More Adam polishing rounds** – after the sweep we run several
   Adam optimisations (soft‑max parametrisation) from the best
   candidates found, each for a few seconds.
3. **Increased sub‑gradient restarts** – we allocate more time to
   stochastic Dirichlet restarts, using a broader range of concentration
   parameters.
4. **Final annealing with a larger move budget** – we keep the simulated
   annealing but allow it to run until the time budget is exhausted.

All helpers are defined at top‑level (no closures).  The routine respects
the supplied time budget (default 1000 s) and always returns a
non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of the max‑convolution.
    Result length = len(a).  ``w`` is a weight vector of length 2·len(a)-1.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a) / (sum a)² (no clipping).
    ``a`` must be non‑negative and sum(a)>0.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes, either at random positions or evenly spaced."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_gamma(p: float, alpha: float, n: int) -> np.ndarray:
    """Gamma‑like shape: (i+1)^p · exp(‑α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p * np.exp(-alpha * i)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_sine(p: float, n: int) -> np.ndarray:
    """Sinusoidal power: sin(π·(i+½)/n)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** p
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a

# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimisation on the smooth‑max (log‑sum‑exp) surrogate.
    The parametrisation ``a = softmax(logits)`` guarantees non‑negativity
    and unit sum.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters

    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.2          # temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_500_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient / mirror‑descent sub‑gradient method.
    The step size decays as ηₜ = η₀ / √(t+1).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set
        active = np.isclose(conv, max_c, atol=1e-13, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update on simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, safeguard against collapse
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 5_000_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated annealing that transfers a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.2
    T_end = 1e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        # linear cooling
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-18:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # move a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # problem dimension – 4096 gives a good trade‑off between quality
    # and speed; larger n can be tried later if time permits.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = np.inf

    # ------------------------------------------------------------------
    # 1) dense parametric sweep (extended families)

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.01, 120):
        cand = _candidate_power(p, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 140):
        cand = _candidate_exponential(alpha, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # stretched exponential
    alphas = np.logspace(-4, 0.6, 40)
    betas = np.linspace(0.2, 3.0, 30)
    for alpha in alphas:
        for beta in betas:
            cand = _candidate_stretched(alpha, beta, n)
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Sweep] new best {val:.6f}")

    # beta distributions – extended to very small shape parameters
    a_pars = [0.02, 0.05, 0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            cand = _candidate_beta(a_par, b_par, n)
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Sweep] new best {val:.6f}")

    # double Gaussian
    for sig in np.logspace(-3.0, -0.2, 30):
        cand = _candidate_double_gaussian(sig, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 120):
        cand = _candidate_left_exponential(alpha, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # monotone left‑power
    for p in np.linspace(-4.0, -0.01, 120):
        cand = _candidate_left_power(p, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # monotone left‑logistic
    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 80):
        cand = _candidate_left_logistic(beta, mu, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.0, 80):
        cand = _candidate_left_linear(slope, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # gamma‑like family (power·exp)
    for p in np.linspace(-3.0, 3.0, 30):
        for alpha in np.logspace(-5, 0.5, 25):
            cand = _candidate_gamma(p, alpha, n)
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Sweep] new best {val:.6f}")

    # sinusoidal power family
    for p in np.linspace(-3.0, -0.2, 20):
        cand = _candidate_sine(p, n)
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # random spike patterns
    for k in [32, 64, 128, 256, 512]:
        for _ in range(8):
            cand = _candidate_spike(k, n, random=True)
            val = _objective(cand, n)
            if val < best_val:
                best_val, best_seq = val, cand.copy()
                print(f"[Sweep] new best {val:.6f}")

    # a few pure Dirichlet draws
    for _ in range(30):
        cand = np.random.dirichlet(np.ones(n))
        val = _objective(cand, n)
        if val < best_val:
            best_val, best_seq = val, cand.copy()
            print(f"[Sweep] new best {val:.6f}")

    # warm‑start from a previous global variable if it exists
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            val = _objective(w, n)
            if val < best_val:
                best_val, best_seq = val, w.copy()
                print(f"[Sweep] new best from warm start {val:.6f}")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Multiple Adam polishing runs
    remaining = deadline - time.time()
    adam_runs = 0
    while remaining > 30 and adam_runs < 3:
        print(f"[Phase‑Adam] run {adam_runs+1}")
        adam_deadline = time.time() + min(15.0, remaining - 5.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=400_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")
        adam_runs += 1
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 3) Stochastic Dirichlet restarts + sub‑gradient refinement
    remaining = deadline - time.time()
    max_restarts = 8
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        conc = float(np.random.choice([0.02, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, conc))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 20.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0,
            n,
            conv_fft_len,
            corr_fft_len,
            slice_deadline,
            max_iters=1_500_000,
            eta0=1.0,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 4) Final long sub‑gradient run on current best
    if time.time() < deadline - 3:
        print("[Phase‑Subgrad] final long sub‑gradient run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.0,
        )
        if val_sub < best_val:
            best_val, best_seq = val_sub, a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Final simulated‑annealing polishing (run until time runs out)
    if time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=5_000_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # sanity checks and final evaluation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluation failed (evaluate_sequence not available).")

    return best_seq.tolist()
