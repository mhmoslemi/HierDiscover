"""
Step‑function optimizer (≈1000 s budget)

Goal
----
Find a non‑negative sequence `a` (length n) that minimises

    2·n·max(a∗a) / (sum(a))²

where a∗a is the linear convolution of a with itself.  The problem is
scale‑invariant, so we keep the sum fixed to 1 during the optimisation.

Method
------
1. **Very dense parametric sweep** – the sweep now contains a fine
   grid for the monotone exponential family (the family that is known to
   be close to optimal) and a random search over mixtures of two
   exponentials.  The best candidate from the sweep is saved.

2. **Long Adam polishing** – a smooth‑max surrogate (log‑sum‑exp) of the
   peak convolution is optimised with Adam while the temperature is
   increased very slowly (λ←λ·1.005).  The variables are parametrised by
   a soft‑max (mirror descent on the simplex) which guarantees the
   non‑negativity and normalisation constraints.

3. **Extended exponentiated‑gradient (EG) sub‑gradient** – an exact
   sub‑gradient of the max‑convolution is used.  A larger number of
   iterations and a slower decay of the step size (η₀ / √t) are used
   than in the previous implementation.

4. **Optional short simulated‑annealing polish** – a few thousand
   random mass‑transfer moves are tried to escape any remaining tiny
   local bumps.

All steps respect the global time budget; the routine always returns
the best sequence found so far (or a uniform fallback).  The code uses
only NumPy and SciPy and can be run on a single CPU core.
"""

import time
import numpy as np
import scipy.sparse as sp
from scipy.optimize import linprog  # not used in the final version but kept for extensibility

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a (needed for the gradient of the max–convolution).
    Result length = len(a) and corresponds to the derivative of
    ``max(w·(a*a))`` w.r.t. ``a`` when ``w`` is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)² (no clipping).
    ``a`` is assumed to be non‑negative.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_mixed_exponential(alpha1: float, alpha2: float, w: float, n: int) -> np.ndarray:
    """Mixture of two left‑exponential tails."""
    i = np.arange(n, dtype=np.float64)
    a = w * np.exp(-alpha1 * i) + (1.0 - w) * np.exp(-alpha2 * i)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_random_dirichlet(conc: float, n: int) -> np.ndarray:
    """Random Dirichlet start (concentration ``conc``)."""
    a = np.random.dirichlet(np.full(n, conc))
    # ensure we are not too close to zero after clipping
    if a.sum() < 1e-12:
        a = np.full(n, 1.0 / n)
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lr: float = 0.05,
    lam_start: float = 0.05,
    lam_factor: float = 1.005,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp) of the max‑convolution.
    Parameters are deliberately chosen for a very gentle temperature schedule.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # evaluate exact objective (for logging)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (slowly sharpen)
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient / mirror descent on the simplex).
    The step size is ηₜ = η₀ / √(t+1).  Stops when the deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    a_sum = a.sum()
    if a_sum < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        a_sum = a.sum()
    a = a / a_sum

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update on the simplex
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numeric loss
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search that moves a random fraction of mass
    between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.2
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    the best sequence found.  It combines a dense exponential sweep,
    random mixtures of exponentials, a long Adam polishing stage,
    extended exponentiated‑gradient refinement and a short annealing polish.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # ------------------------------------------------------------------
    # problem dimensions
    n = 4096                                   # length of the sequence
    conv_fft_len = _next_pow2(2 * n - 1)       # FFT length for convolutions
    corr_fft_len = _next_pow2(3 * n - 2)       # FFT length for cross‑correlation

    # ------------------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Fine exponential sweep (known to be close to optimum)

    # centre of the theoretically interesting region: delta ≈ 2/n
    alpha_center = 2.0 / n
    alphas_fine = np.linspace(alpha_center * 0.4, alpha_center * 2.5, 2500)
    print("[Sweep] fine left‑exponential search")
    for a in alphas_fine:
        _maybe_update(_candidate_left_exponential(a, n))

    # ------------------------------------------------------------------
    # 2) Random mixtures of two exponentials (adds extra degrees of freedom)
    print("[Sweep] random mixed‑exponential search")
    for _ in range(300):
        alpha1 = 10 ** np.random.uniform(-5, 0)          # between 1e‑5 and 1
        alpha2 = 10 ** np.random.uniform(-5, 0)
        w = np.random.rand()
        _maybe_update(_candidate_mixed_exponential(alpha1, alpha2, w, n))

    # ------------------------------------------------------------------
    # 3) A few random Dirichlet starts (used later for refinement)
    print("[Sweep] random Dirichlet seeds")
    for _ in range(30):
        conc = float(np.random.choice([0.1, 0.3, 0.5, 1.0, 2.0, 5.0]))
        _maybe_update(_candidate_random_dirichlet(conc, n))

    # ------------------------------------------------------------------
    # 4) Warm‑start from a previously stored solution, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        w_sum = w.sum()
        if w_sum >= 0.01:
            w = w / w_sum
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5) Adam polishing (allowing a generous amount of time)
    if time.time() < deadline - 40:
        print("[Phase‑Adam] start long Adam polishing")
        adam_deadline = min(deadline - 5, time.time() + 30.0)  # keep a small safety buffer
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=1_200_000,
            lr=0.05,
            lam_start=0.05,
            lam_factor=1.005,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 6) Extended exponentiated‑gradient sub‑gradient refinement
    if time.time() < deadline - 15:
        print("[Phase‑Subgrad] long EG refinement")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=8_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 7) Final simulated‑annealing polish (very short, just in case)
    if time.time() < deadline - 2:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # final sanity checks and evaluation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
