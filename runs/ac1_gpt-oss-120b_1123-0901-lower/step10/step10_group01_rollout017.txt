"""
Propose a low‑value step‑function for the given evaluate_sequence.

The algorithm follows a multi‑stage strategy:

1. **Diverse initial pool** – many analytic families (uniform,
   exponential, Gaussian, power‑law, triangular, spikes) with dense
   grids of parameters, plus a handful of random Dirichlet draws.
   Every candidate is normalised to the probability simplex
   (sum = 1) and its exact objective
   `val = 2·n·max(conv(a,a))` is computed using an FFT‑based
   convolution.

2. **Select promising seeds** – the best `k` seeds (default 8) are kept
   for deep optimisation.

3. **Adam on a smooth‑max surrogate** – each seed is optimised with
   Adam on the log‑soft‑max parametrisation of the simplex.
   The temperature of the smooth‑max (`lam`) is increased slowly,
   allowing the optimizer to first view a softened objective and then
   focus on the true max.

4. **Exponentiated‑gradient sub‑gradient phase** – after Adam a long
   exact‑max sub‑gradient run (exponential‑gradient updates) reduces
   the objective further.

5. **Projected gradient descent** – a few hundred thousand steps of
   plain PGD fine‑tune the iterate.

6. **Simulated annealing** – a stochastic local‑search that moves a
   random fraction of mass between two coordinates, with an annealing
   schedule, often escapes shallow local minima.

7. **Random perturbations** – if time permits the current best is
   perturbed with small Gaussian noise and the whole pipeline is
   re‑run on the perturbed start.

8. **Fallback** – if nothing is found (should never happen) a uniform
   vector is returned.

All heavy work is done with FFT‑based convolutions (O(n log n)),
projected onto the simplex to keep the feasibility constraints.
The total wall‑time is limited by the `budget_s` argument (≈1000 s).

The returned list is a non‑negative sequence; the evaluation function
treats it as a (scaled‑invariant) step‑function, so any scaling of the
vector does not affect the score.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Implements convolution of w with reversed a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection on the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # fallback: uniform distribution
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_optimize(a0, n, conv_fft_len, corr_fft_len, deadline,
                   max_iters=300_000):
    """
    Adam optimiser on the smooth‑max surrogate.
    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-14
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2 = 0.9, 0.999
    lr = 0.08                                 # slightly higher initial LR
    lam = 0.05                                 # initial temperature
    lam_factor = 1.01                         # slower temperature rise

    best_a = a0.copy()
    best_val = float("inf")

    for it in range(max_iters):
        if time.time() > deadline:
            break

        # ----- soft‑max to obtain a on the simplex ----------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- evaluate exact objective -----
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weight vector -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ----- gradient of the smooth‑max -----
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # ----- chain rule through soft‑max -----
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # ----- Adam update -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # ----- temperature schedule -----
        lam = min(lam * lam_factor, 1e7)

        # ----- occasional learning‑rate decay -----
        if (it + 1) % 75_000 == 0:
            lr *= 0.9

    return best_a, best_val


def _expgrad_optimize(a0, n, conv_fft_len, corr_fft_len,
                      deadline, max_iters=600_000, eta0=0.8):
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    Returns best vector (still on the simplex) and its objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")

    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if (it & 0xFFFF) == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the maximum)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / math.sqrt(it + 1.0)

        # EG update (stay on the simplex automatically)
        a = a * np.exp(-eta * grad)
        s = a.sum()
        if s == 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

    return best_a, best_val


def _pgd_optimize(a0, n, conv_fft_len, corr_fft_len,
                  deadline, max_iters=200_000):
    """
    Projected gradient descent on the exact max‑convolution.
    Simple diminishing step size.
    """
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    for it in range(max_iters):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.12 / math.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[PGD  it={it:06d}] new best {val:.6f}")

    return best_a, best_val


def _anneal_optimize(a0, n, conv_fft_len,
                    deadline, max_moves=150_000):
    """
    Simulated‑annealing refinement on the simplex.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.09
    T_end = 5e-5

    for it in range(max_moves):
        if time.time() > deadline:
            break
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        frac = np.random.rand() * 0.45   # move up to 45 % of source mass
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val, best_a = val2, a2.copy()
            a, val = a2, val2
        else:
            if np.random.rand() < math.exp(-(val2 - val) / T):
                a, val = a2, val2

    return best_a, best_val

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    return np.random.dirichlet(alpha * np.ones(n)).astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a

        pool.append(_make_exponential(n, gamma))

    # Gaussian families (linear sweep of sigma)
    for sigma in np.linspace(0.02, 0.6, num=15):
        pool.append(_make_gaussian(n, sigma))

    # power‑law families (negative exponents, fine grid)
    for expo in np.linspace(-2.5, -0.3, num=16):
        pool.append(_make_powerlaw(n, expo))

    # triangle (single deterministic shape)
    pool.append(_make_triangle(n))

    # spike patterns (varying sparsity)
    for k in (32, 64, 128, 256, 512):
        pool.append(_make_spikes(n, k))

    # random Dirichlet draws – a richer set than before
    for _ in range(30):
        alpha = 10 ** np.random.uniform(-3, 1)   # 1e‑3 … 10
        pool.append(_make_dirichlet(n, alpha))

    # --------------------------------------------------------------
    # quick evaluation of the raw pool (no gradients)
    def _objective(a_vec):
        conv = _full_convolution(a_vec, conv_fft_len)
        return 2.0 * n * conv.max()

    vals = []
    for a in pool:
        a = np.maximum(a, 0.0)
        if a.sum() < 0.01:          # degenerate – replace
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        vals.append(_objective(a))

    # keep the best K seeds for deep optimisation
    K = 8
    top_idx = np.argsort(vals)[:K]
    start_vectors = [pool[i] for i in top_idx]

    # --------------------------------------------------------------
    # Global best tracking
    best_seq = None
    best_val = float("inf")

    # --------------------------------------------------------------
    # Phase I – deep optimisation of each promising start
    for idx, a0 in enumerate(start_vectors):
        if time.time() > deadline:
            break
        a0 = np.maximum(a0, 0.0)
        a0 = a0 / a0.sum()

        # allow roughly 30 s for Adam on each seed, but respect the global budget
        sub_dead = min(deadline, time.time() + 30.0)

        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=350_000
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Seed {idx}] Adam improved to {val_adam:.6f}")

        # sub‑gradient phase (longer than before)
        if time.time() < deadline - 5:
            sub_dead2 = deadline
            a_sub, val_sub = _expgrad_optimize(
                best_seq, n, conv_fft_len, corr_fft_len,
                sub_dead2, max_iters=800_000, eta0=0.7
            )
            if val_sub < best_val:
                best_val, best_seq = val_sub, a_sub.copy()
                print(f"[Seed {idx}] ExpGrad improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # Phase II – fine‑tune the current best with projected gradient descent
    if best_seq is not None and time.time() < deadline - 10:
        print("[Fine] Projected gradient descent")
        a_gd, val_gd = _pgd_optimize(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=300_000
        )
        if val_gd < best_val:
            best_val, best_seq = val_gd, a_gd.copy()
            print(f"[Fine] GD improved to {val_gd:.6f}")

    # --------------------------------------------------------------
    # Phase III – stochastic local search (simulated annealing)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Anneal] Simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline,
            max_moves=180_000
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Anneal] improved best to {val_ann:.6f}")

    # --------------------------------------------------------------
    # Phase IV – random perturbations + re‑optimisation (if time permits)
    while time.time() < deadline - 20:
        # small Gaussian jitter on the current best, re‑normalise
        jitter = np.random.normal(scale=0.002, size=n)
        a_pert = best_seq + jitter
        a_pert = np.maximum(a_pert, 0.0)
        if a_pert.sum() < 0.01:
            a_pert = np.random.dirichlet(np.ones(n))
        a_pert = a_pert / a_pert.sum()

        # short Adam run to recover feasibility
        sub_dead = min(deadline, time.time() + 12.0)
        a_tmp, val_tmp = _adam_optimize(
            a_pert, n, conv_fft_len, corr_fft_len,
            sub_dead, max_iters=150_000
        )
        if val_tmp < best_val:
            best_val, best_seq = val_tmp, a_tmp.copy()
            print(f"[Perturb] new best {val_tmp:.6f}")

        # give the sub‑gradient phase a quick shot
        if time.time() < deadline - 5:
            a_tmp2, val_tmp2 = _expgrad_optimize(
                best_seq, n, conv_fft_len, corr_fft_len,
                deadline, max_iters=250_000, eta0=0.6
            )
            if val_tmp2 < best_val:
                best_val, best_seq = val_tmp2, a_tmp2.copy()
                print(f"[Perturb] sub‑grad improved to {val_tmp2:.6f}")

    # --------------------------------------------------------------
    # Optional up‑sampling to a finer grid (if we still have time)
    if best_seq is not None and time.time() + 30 < deadline:
        try:
            print("[Upsample] Explore finer grid (8192 points)")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            x_old = np.linspace(0.0, 1.0, n, endpoint=False)
            x_big = np.linspace(0.0, 1.0, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big = a_big / a_big.sum()

            sub_dead_big = min(deadline, time.time() + 12.0)
            a_big, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big, max_iters=200_000
            )
            if val_big < best_val:
                best_val, best_seq = val_big, a_big.copy()
                print(f"[Upsample] improved best to {val_big:.6f}")
        except Exception as exc:
            print(f"[Upsample] failed: {exc}")

    # --------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = _make_uniform(n)

    # final sanitisation (the evaluator clamps anyway)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
