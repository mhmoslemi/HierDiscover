a = a0.copy()
if a.sum() == 0: a = np.random.dirichlet(...)

best_a = a.copy(); best_val = _objective(a,n)
it=0
lam=lam0
while time.time() < deadline and it < max_iter:
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    val = 2*n*max_c/(a.sum()**2)  # or _objective(a)
    if val < best_val:
        best_val = val
        best_a = a.copy()
        print(f"[EG it={it}] new best {val:.6f}")
    # compute smooth weights
    w = np.exp(lam * (conv - max_c))
    w_sum = w.sum()
    if w_sum == 0 or np.isnan(w_sum):
        w = np.full_like(w, 1.0 / w.size)
    else:
        w /= w_sum
    grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
    eta = eta0 / np.sqrt(it+1)
    a = a * np.exp(-eta * grad)

        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s
    # update lambda
    lam = min(lam * lam_factor, 1e7)

def _next_pow2(x):
    return 1 << (x - 1).bit_length()

def _full_convolution(a, fft_len):
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _objective(a, n):
    conv = _full_convolution(a, _next_pow2(2*n-1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p, n):
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a

def _candidate_exponential(alpha, n):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a

def _candidate_stretched(alpha, beta, n):
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a

    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)

def _smooth_exponential_gradient_descent(a0, n, conv_fft_len, corr_fft_len,
                                        deadline, eta0=0.2, lam0=0.5,
                                        lam_factor=1.02, max_iter=10_000_000):
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()
    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    lam = lam0
    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[SmoothSGD it={it}] new best {val:.6f}")

        # smooth weights (soft‑max style)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        # project back onto simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        lam = min(lam * lam_factor, 1e7)

        it += 1
    return best_a, best_val

def _anneal_optimize(a0, n, conv_fft_len, deadline,
                    max_moves=250_000):
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()
    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.3
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)
        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1
    return best_a, best_val

"""
Step‑function optimizer for the evaluation metric

    2·n·max(conv(a,a)) / (sum a)²

The algorithm proceeds in four stages inside the given time budget:

1. **Dense parametric sweep** – a large collection of analytic families
   (exponential, power‑law, stretched‑exponential, beta, Gaussian,
   classic window functions, left‑sine, mixed exponential‑power, random
   spikes, random Dirichlet samples) is generated and the best candidate
   is kept.

2. **Smooth‑max exponentiated‑gradient descent** – starting from the best
   candidate, we minimise the ‑∞‑norm of the convolution by optimising the
   smooth‑max surrogate
        Sₗ(a)=log ∑ₖ exp(l·(convₖ−max(conv))).
   The update is a mirror‑descent step on the simplex:
        a←a·exp(‑η·∇Sₗ(a))   (followed by normalisation).
   The temperature *l* grows slowly, while the stepsize η decays as
   1/√t.  This phase occupies the bulk of the remaining time.

3. **Stochastic Dirichlet restarts** – a handful of random Dirichlet
   initialisations are refined with the same smooth‑max optimiser in a
   shortened time slice, providing extra chances to escape local minima.

4. **Simulated‑annealing polishing** – a short local‑search swaps a small
   fraction of mass between random indices to fine‑tune the final solution.

The routine always returns a non‑negative list that can be fed directly to
``evaluate_sequence``.  All computations respect the supplied budget
(≤ 1000 s) and the code uses only NumPy and the standard library.
"""

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (gradient of the smooth‑max)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective 2·n·max(a*a)/(sum a)² (scale‑invariant)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^p, p<0."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape: 1/(1+exp(beta·(i‑mu)))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_hann(n: int) -> np.ndarray:
    """Hann window (symmetric, zero at the ends)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a /= a.sum()
    return a


def _candidate_hamming(n: int) -> np.ndarray:
    """Hamming window (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.54 - 0.46 * np.cos(2 * np.pi * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_blackman(n: int) -> np.ndarray:
    """Blackman window (symmetric)."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42
         - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_sine(n: int) -> np.ndarray:
    """Monotone half‑sine shape (large at the left, decays to the right)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (n - i) / (2 * n))
    a /= a.sum()
    return a


def _candidate_exponential_power(alpha: float, p: float, n: int) -> np.ndarray:
    """Mixed shape a_i ∝ exp(-α·i)·(i+1)^p (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) * ((i + 1) ** p)
    a = np.maximum(a, 0.0)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _smooth_exponential_gradient_descent(a0: np.ndarray, n: int,
                                        conv_fft_len: int, corr_fft_len: int,
                                        deadline: float,
                                        eta0: float = 0.2,
                                        lam0: float = 0.5,
                                        lam_factor: float = 1.02,
                                        max_iter: int = 10_000_000):
    """
    Exponentiated‑gradient descent on the smooth‑max surrogate of the
    convolution peak.  The update is performed on the simplex via
    multiplicative weights; the temperature λ grows geometrically.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    lam = lam0

    while time.time() < deadline and it < max_iter:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10000 == 0:
                print(f"[SmoothSGD it={it}] new best {val:.6f}")

        # smooth‑max weights (soft‑max on the convolution values)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        # projection back onto the simplex
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        lam = min(lam * lam_factor, 1e7)

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int, deadline: float,
                    max_moves: int = 250_000):
    """
    Simple simulated‑annealing that repeatedly moves a random fraction of the
    mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0

    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue
        frac = np.random.rand() * 0.3          # ≤30 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1

    return best_a, best_val

# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises
    ``2·n·max(a*a)/(sum a)²``.  The routine respects the time budget and
    always returns a feasible list.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # problem size ------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Very dense parametric sweep ------------------------------------
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n))

    for alpha in np.logspace(-5, 1.0, 100):
        _maybe_update(_candidate_exponential(alpha, n))

    for alpha in np.logspace(-4, 0.6, 30):
        for beta in np.linspace(0.2, 3.0, 20):
            _maybe_update(_candidate_stretched(alpha, beta, n))

    pars = [0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    for a_par in pars:
        for b_par in pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3.0, -0.2, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    for alpha in np.logspace(-5, 1.2, 80):
        _maybe_update(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_left_power(p, n))

    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 60):
        _maybe_update(_candidate_left_logistic(beta, mu, n))

    for slope in np.linspace(0.1, 3.0, 40):
        _maybe_update(_candidate_left_linear(slope, n))

    # classic windows (symmetric)
    _maybe_update(_candidate_hann(n))
    _maybe_update(_candidate_hamming(n))
    _maybe_update(_candidate_blackman(n))

    # monotone half‑sine
    _maybe_update(_candidate_left_sine(n))

    # mixed exponential‑power shapes
    for alpha in np.logspace(-5, 0.5, 30):
        for p in np.linspace(-4.0, -0.1, 20):
            _maybe_update(_candidate_exponential_power(alpha, p, n))

    # random spikes
    for k in [32, 64, 128, 256]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # random Dirichlet samples
    for _ in range(15):
        _maybe_update(np.random.dirichlet(np.ones(n)))

    # optional warm start from a global variable (previous run)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max exponentiated‑gradient descent on the best candidate
    if best_seq is not None and time.time() < deadline - 5:
        phase_deadline = min(deadline - 5,
                             time.time() + 0.7 * (deadline - time.time()))
        print("[Phase‑SGD] smooth‑max gradient descent")
        a_sgd, val_sgd = _smooth_exponential_gradient_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            phase_deadline,
            eta0=0.2,
            lam0=0.5,
            lam_factor=1.02,
            max_iter=5_000_000
        )
        if val_sgd < best_val:
            best_val = val_sgd
            best_seq = a_sgd.copy()
            print(f"[Phase‑SGD] improved to {val_sgd:.6f}")

    # ------------------------------------------------------------------
    # 3) Random Dirichlet restarts, each refined by the same optimiser
    remaining = deadline - time.time()
    max_restarts = 4
    for r in range(max_restarts):
        if time.time() >= deadline - 5:
            break
        concentration = float(np.random.choice([0.05, 0.1, 0.3, 0.6, 1.0, 2.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 12.0)
        a_rst, val_rst = _smooth_exponential_gradient_descent(
            a0, n, conv_fft_len, corr_fft_len,
            slice_deadline,
            eta0=0.3,
            lam0=0.3,
            lam_factor=1.025,
            max_iter=2_000_000
        )
        if val_rst < best_val:
            best_val = val_rst
            best_seq = a_rst.copy()
            print(f"[Restart {r}] new best {val_rst:.6f}")
        remaining = deadline - time.time()

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing polishing (if time permits)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=300_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # final sanity checks & reporting
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print(f"Could not compute final score: {exc}")

    return best_seq.tolist()
