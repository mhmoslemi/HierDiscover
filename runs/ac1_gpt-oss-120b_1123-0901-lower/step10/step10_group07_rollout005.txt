"""
Hybrid optimizer for the step‑function bound.

The routine performs:
1.  A dense parametric sweep over many analytic families (power‑law,
    exponential, stretched‑exponential, beta‑distribution, double‑Gaussian,
    monotone left‑exponential/power/logistic/linear) and a new mixed
    exponential‑power family.
2.  A large number of short stochastic Dirichlet restarts.  Each restart is
    refined either by exponentiated‑gradient mirror‑descent (the original
    sub‑gradient) *or* by Euclidean projected gradient descent – both
    operate on the exact ``max``‑convolution sub‑gradient.
    The step‑size η is annealed as η₀/√t with a random η₀ per restart,
    increasing the chance of escaping shallow minima.
3.  A final polishing stage that first runs a short Adam optimisation on a
    smooth‑max surrogate and then a long projected‑gradient run until the
    time budget expires.

All helpers are top‑level functions, the code respects the 1000 s budget and
always returns a non‑negative list suitable for ``evaluate_sequence``.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution ``a * a`` (length 2·len(a)-1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation of a weight mask ``w`` with ``a``.
    The result has length ``len(a)`` and corresponds to the gradient of
    ``max(w·(a*a))`` w.r.t. ``a`` when ``w`` is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)² (no clipping).
    ``a`` is assumed to be non‑negative and summing to 1 (the scaling is
    irrelevant for the ratio, but keeping sum=1 simplifies the code).
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^p (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape: 1/(1+exp(beta·(i‑mu)))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_mixed(alpha: float, p: float, n: int) -> np.ndarray:
    """Mixed exponential‑power shape: a_i ∝ exp(-α·i)·(i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) * (i + 1) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a[:] = 1.0 / n
    else:
        a /= a_sum
    return a

def _proj_onto_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        rho = 0
    else:
        rho = np.where(cond)[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate while staying
    on the simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # initialise soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution for logging
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (sharpening)
        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent (mirror descent on the simplex).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Exp‑Grad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update, stays on the simplex implicitly
        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _proj_subgradient_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.7,
) -> tuple[np.ndarray, float]:
    """
    Euclidean projected sub‑gradient descent on the simplex.
    The update is a = Proj_{simplex}(a - η·grad).
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Proj‑Grad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a_tmp = a - eta * grad_a
        a = _proj_onto_simplex(a_tmp)

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the provided time budget and always returns a
    feasible sequence (list of floats) that can be fed to ``evaluate_sequence``.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # small safety margin

    # ------------------------------------------------------------------
    # problem dimension (the original runs used 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray, info: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {info}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Dense parametric sweep (including a new mixed family)

    # power‑law (symmetric)
    for p in np.linspace(-4.0, -0.01, 120):
        _maybe_update(_candidate_power(p, n), "power")

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 140):
        _maybe_update(_candidate_exponential(alpha, n), "exp")

    # stretched exponential (symmetric)
    alphas = np.logspace(-4, 0.6, 40)
    betas = np.linspace(0.2, 3.0, 30)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n), "stretch")

    # beta distributions
    a_pars = [0.1, 0.2, 0.4, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n), "beta")

    # double Gaussian
    for sig in np.logspace(-3.0, -0.2, 30):
        _maybe_update(_candidate_double_gaussian(sig, n), "gauss")

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 120):
        _maybe_update(_candidate_left_exponential(alpha, n), "lexp")

    # monotone left‑power
    for p in np.linspace(-4.0, -0.01, 120):
        _maybe_update(_candidate_left_power(p, n), "lpow")

    # monotone left‑logistic
    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 80):
        _maybe_update(_candidate_left_logistic(beta, mu, n), "logi")

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.0, 80):
        _maybe_update(_candidate_left_linear(slope, n), "lin")

    # random spikes (few dozen attempts)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(6):
            _maybe_update(_candidate_spike(k, n, random=True), f"sp{k}")

    # mixed exponential‑power family (coarse grid)
    for p in np.linspace(-4.0, -0.01, 40):
        for alpha in np.logspace(-5, 0.5, 30):
            _maybe_update(_candidate_mixed(alpha, p, n), "mix")

    # a few random Dirichlet draws
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand, "rand")

    # incorporate a possible warm start from a previous run
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam polishing (if time permits)
    if time.time() < deadline - 30:
        print("[Phase‑Adam] polishing")
        adam_deadline = min(deadline - 5, time.time() + 20.0)
        a_adam, val_adam = _adam_optimize(
            best_seq, n, conv_fft_len, corr_fft_len, adam_deadline,
            max_iters=600_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Many short stochastic restarts – both optimisers
    while time.time() < deadline - 5:
        remaining = deadline - time.time()
        # allocate a small slice (max 6 s) – enough for a few thousand updates
        slice_len = min(remaining * 0.5, 6.0)

        # random Dirichlet concentration (favoring sparse or dense starts)
        concentration = float(
            np.random.choice([0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0])
        )
        a0 = np.random.dirichlet(np.full(n, concentration))

        # pick optimiser at random
        if np.random.rand() < 0.55:          # favour the exponentiated‑gradient
            eta0 = np.random.uniform(0.6, 1.4)
            a_opt, val_opt = _subgradient_optimize_expgrad(
                a0, n, conv_fft_len, corr_fft_len,
                time.time() + slice_len,
                max_iters=2_000_000,
                eta0=eta0,
            )
            opt_name = "ExpGrad"
        else:
            eta0 = np.random.uniform(0.4, 1.0)
            a_opt, val_opt = _proj_subgradient_optimize(
                a0, n, conv_fft_len, corr_fft_len,
                time.time() + slice_len,
                max_iters=2_000_000,
                eta0=eta0,
            )
            opt_name = "ProjGrad"

        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Restart {opt_name}] new best {val_opt:.6f}")

    # ------------------------------------------------------------------
    # 4) Final long projected‑gradient run (full remaining budget)
    if time.time() < deadline - 2:
        print("[Phase‑FinalProj] long projected‑gradient descent")
        a_final, val_final = _proj_subgradient_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=12_000_000,
            eta0=0.7,
        )
        if val_final < best_val:
            best_val = val_final
            best_seq = a_final.copy()
            print(f"[Phase‑FinalProj] improved to {val_final:.6f}")

    # ------------------------------------------------------------------
    #   final sanity checks and evaluation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score = {final_score:.6f}")
    except Exception:
        print("Could not compute final score (evaluator missing).")

    return best_seq.tolist()
