"""
Hybrid random‑Dirichlet + smooth‑max Adam + exponentiated‑gradient + annealing.

The optimiser proceeds in four phases, each respecting the remaining time
budget:

1. **Exploratory random search** – draws many candidates from a rich
   family (Dirichlet distributions with random concentration,
   sparse “spike’’ patterns, piecewise‑constant blocks and monotone
   exponentials).  The best few vectors are kept.

2. **Smooth‑max Adam refinement** – starting from the incumbent the
   surrogate
        S(a) = log‑sum‑exp(λ·(a*a)) + μ·‖(a*a) – uniform‖₂²
   is minimised with Adam in the soft‑max (log‑probability) parametrisation,
   guaranteeing the simplex constraints.  λ is increased slowly, μ
   pushes the convolution towards a flat target.

3. **Exact‑max exponentiated‑gradient sub‑gradient** – the true max‑norm
   objective (2·n·max(a*a)/(∑a)²) is minimised by mirror‑descent on the
   simplex using the sub‑gradient obtained from the active set of maximal
   convolution indices.

4. **Mass‑transfer simulated annealing** – a final fine‑tuning stage that
   repeatedly moves a random fraction of mass between two positions,
   accepting worsening moves with a temperature schedule.

The routine always returns a non‑negative list; the final score is printed
via the provided ``evaluate_sequence`` function.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation ``w ⋆ a`` needed for gradients.
    Both arguments are real; ``w`` has length 2·len(a)‑1.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """Exact evaluation 2·n·max(conv)/(∑a)² (no clipping)."""
    conv = _full_convolution(a, fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_dirichlet(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Draw a Dirichlet vector with symmetric concentration ``alpha``."""
    # Gamma sampling works for any positive alpha
    g = rng.gamma(alpha, 1.0, size=n)
    s = g.sum()
    if s == 0.0:
        g[:] = 1.0
        s = n
    return g / s


def _candidate_spike(n: int, rng: np.random.Generator, k: int) -> np.ndarray:
    """k spikes on a grid of length n with Dirichlet masses."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = rng.choice(n, size=k, replace=False)
        masses = rng.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    return a


def _candidate_piecewise_constant(n: int, rng: np.random.Generator, p: int) -> np.ndarray:
    """Piecewise constant function with ``p`` blocks."""
    # Random split points
    cuts = sorted(rng.integers(1, n, size=p - 1))
    seg_lengths = np.diff([0] + cuts + [n])
    masses = rng.dirichlet(np.ones(p))
    a = np.empty(n, dtype=np.float64)
    start = 0
    for length, mass in zip(seg_lengths, masses):
        a[start:start + length] = mass / length
        start += length
    return a


def _candidate_left_exponential(n: int, rng: np.random.Generator, alpha: float) -> np.ndarray:
    """Monotonically decreasing exponential a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    s = a.sum()
    if s == 0.0:
        a[:] = 1.0 / n
        s = 1.0
    return a / s


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max + L2 penalty) ------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lam_start: float = 0.05,
    lam_factor: float = 1.005,
    l2_weight: float = 0.20,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate
        smooth‑max(conv) + l2_weight·‖conv – uniform‖₂² .
    Returns the best vector found (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                 # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n, fft_len)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max -> simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (soft‑max over conv) -----
        w = np.exp(lam * conv)
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L2 uniformity part -----
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        # combined gradient
        grad = grad_smooth + l2_weight * grad_l2

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # schedule λ → harder max
        lam = min(lam * lam_factor, 1e7)

        # occasional decay of learning rate
        if it % 150_000 == 0 and it > 0:
            lr *= 0.92

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 12_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient step
        a = a * np.exp(-eta * grad)

        # renormalise, guard against numerical drift
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning ---------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple annealing that transfers a small fraction of mass between two
    positions.  The temperature is linearly cooled.
    """
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.15
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.30       # up to 30 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns a valid list of coefficients.
    """
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5        # safety margin

    # problem size ------------------------------------------------------
    n = 4096                     # length of the step function
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Random exploratory sweep ---------------------------------------
    print("[Stage 1] random exploratory sweep")
    stage1_deadline = start_time + 0.18 * (deadline - start_time)

    while time.time() < stage1_deadline:
        # pick a generation mode uniformly at random
        mode = rng.integers(0, 4)

        if mode == 0:                      # Dirichlet with random concentration
            alpha = 10 ** rng.uniform(-3.0, 2.0)   # 1e‑3 … 1e2
            a = _candidate_dirichlet(n, rng, alpha)

        elif mode == 1:                    # sparse spikes
            k = int(rng.choice([32, 64, 128, 256, 512, 1024]))
            a = _candidate_spike(n, rng, k)

        elif mode == 2:                    # piecewise‑constant blocks
            p = int(rng.integers(2, 9))    # 2 … 8 blocks
            a = _candidate_piecewise_constant(n, rng, p)

        else:                              # monotone exponential (non‑symmetric)
            alpha = 10 ** rng.uniform(-4.0, 2.0)   # 1e‑4 … 1e2
            a = _candidate_left_exponential(n, rng, alpha)

        # evaluate and possibly update incumbent
        val = _objective(a, n, fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Stage 1] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Adam refinement (smooth‑max + L2 uniformity) -------------------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Stage 2] Adam refinement")
        # allocate up to 40 % of remaining budget
        remaining = deadline - time.time()
        adam_deadline = min(deadline - 5, time.time() + 0.40 * remaining)

        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=900_000,
            lam_start=0.05,
            lam_factor=1.005,
            l2_weight=0.22,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Stage 2] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max exponentiated‑gradient optimisation ------------------
    if best_seq is not None and time.time() < deadline - 8:
        print("[Stage 3] exponentiated‑gradient sub‑gradient")
        sub_deadline = deadline - 3.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=14_000_000,
            eta0=0.85,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Stage 3] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing fine‑tuning -------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Stage 4] final annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=400_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Stage 4] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Sanity & final evaluation -----------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # optional sanity check with the public evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        final_score = None
        print("Could not compute final score:", exc)

    return best_seq.tolist()
