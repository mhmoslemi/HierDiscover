"""
Improved optimiser for the step‑function “max‑convolution” problem.

Strategy
--------
1. Build a diverse pool of *parameterised* initial vectors (exponential,
   double‑exponential, beta‑type, cosine‑bump, uniform, Dirichlet draws,
   spikes, and the optional warm start).
2. Evaluate every candidate cheaply with an FFT‑based convolution
   (objective = 2·n·max(conv) on the probability simplex).
   Keep the best vector.
3. Refine the best vector with a *smooth‑Lp* exponentiated‑gradient scheme.
   We minimise the surrogate  Σ conv_i^p  for a sequence of increasing p
   (2 → 4 → 8 → 16 → 32).  The update is
        a ← a·exp(‑η·grad)   (followed by simplex projection)
   where   grad = 2·p·cross_corr(conv^{p‑1}, a).
   The step‑size η decays as 1/√t.
4. If time permits, perform a simulated‑annealing local search that moves a
   random fraction of mass between two coordinates.
5. Return the best vector (clipped to the evaluator’s allowed range).

All heavy work stays inside the 1000 s time budget; the algorithm never
calls ``evaluate_sequence`` directly – it uses the same fast FFT‑based
objective as the optimiser, guaranteeing a fast search even for n=4096.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities ------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a * a (length 2·len(a)‑1) using FFT.
    a is assumed to be a real 1‑D array.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation (w ★ a) needed for gradients.
    w – length 2n‑1 , a – length n .
    Returns a length‑n array.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex Σx=1, x≥0."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective (fast analogue of evaluate_sequence) -----------------------

def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """
    Compute 2·n·max(conv) for a probability vector a (sum = 1).
    This equals the value returned by evaluate_sequence for the same a,
    because the denominator (sum a)^2 = 1.
    """
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
# Candidate factories ---------------------------------------------------

def _make_uniform(n: int) -> np.ndarray:
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n: int, alpha: float) -> np.ndarray:
    a = np.random.dirichlet(np.full(n, alpha))
    return a.astype(np.float64)


def _make_exponential(n: int, gamma: float) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    a = left + left[::-1]
    a /= a.sum()
    return a


def _make_double_exponential(n: int, gamma1: float, gamma2: float, w: float) -> np.ndarray:
    """Symmetric mixture  w·exp(-γ₁·i) + (1‑w)·exp(-γ₂·i) ."""
    i = np.arange(n, dtype=np.float64)
    left = w * np.exp(-gamma1 * i) + (1.0 - w) * np.exp(-gamma2 * i)
    a = left + left[::-1]
    a /= a.sum()
    return a


def _make_beta(n: int, alpha: float, beta: float) -> np.ndarray:
    """Beta‑like density on the discrete grid."""
    i = np.arange(n, dtype=np.float64) + 0.5
    t = i / n
    # protect against zero‑power blows (α,β can be <1)
    eps = 1e-12
    shape = np.maximum(t, eps) ** (alpha - 1.0) * np.maximum(1.0 - t, eps) ** (beta - 1.0)
    shape = np.maximum(shape, 0.0)
    if shape.sum() == 0:
        shape = np.ones_like(shape)
    a = shape / shape.sum()
    return a


def _make_cosine_positive(n: int) -> np.ndarray:
    """Cosine bump that is non‑negative (mass near the two sides)."""
    i = np.arange(n, dtype=np.float64)
    # centre at (n‑1)/2
    cos_vals = np.cos(np.pi * (i - (n - 1) / 2.0) / (n - 1))
    cos_vals = np.maximum(cos_vals, 0.0)
    if cos_vals.sum() == 0:
        cos_vals = np.ones_like(cos_vals)
    a = cos_vals / cos_vals.sum()
    return a.astype(np.float64)


def _make_spikes(n: int, k: int) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _lp_expgrad_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    p: int,
    deadline: float,
    max_iters: int = 200_000,
    eta0: float = 0.6,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient descent on the smooth Lp surrogate:
        L_p(a) = Σ_i (conv_i)^p ,   conv = a * a .
    The gradient is   grad = 2·p·cross_corr(conv^{p‑1}, a).
    The iterate stays on the probability simplex via multiplicative
    updates and normalisation.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    for it in range(max_iters):
        if time.time() >= deadline:
            break

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_a = a.copy()
            # occasional progress report
            if it % 50000 == 0:
                print(f"[LpGrad p={p} it={it:06d}] new best {cur_val:.6f}")

        # weights w_i = conv_i^{p‑1}
        w = np.power(conv, p - 1, dtype=np.float64)
        grad = 2.0 * p * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # multiplicative update (exponentiated gradient)
        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a = a / s

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing on the simplex.
    Moves a random fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
        else:
            w = None
    else:
        w = None

    # ------------------------------------------------------------------
    # Build a rich pool of parametrised candidates
    pool = []

    # Warm start and simple cyclic shifts
    if w is not None:
        pool.append(w.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(w, shift))

    # Uniform baseline
    pool.append(_make_uniform(n))

    # Simple symmetric cosine bump
    pool.append(_make_cosine_positive(n))

    # Exponential families (fine grid on γ)
    for gamma in np.linspace(0.02, 0.80, 38):
        pool.append(_make_exponential(n, gamma))

    # Double‑exponential mixtures (coarse grid)
    for g1 in (0.05, 0.10, 0.20):
        for g2 in (0.30, 0.60, 1.00):
            for w_mix in (0.3, 0.5, 0.7):
                pool.append(_make_double_exponential(n, g1, g2, w_mix))

    # Beta‑type shapes (α,β ∈ {0.3,0.5,1.0,2.0})
    for a in (0.3, 0.5, 1.0, 2.0):
        for b in (0.3, 0.5, 1.0, 2.0):
            pool.append(_make_beta(n, a, b))

    # Random sparse spike patterns
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    # Random Dirichlet draws (diverse concentration)
    for _ in range(12):
        alpha = 10 ** np.random.uniform(-3, 1)   # 1e‑3 … 10
        pool.append(_make_dirichlet(n, alpha))

    # ------------------------------------------------------------------
    best_seq: np.ndarray | None = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # Phase I – cheap exhaustive evaluation of the candidate pool
    print("[Phase I] evaluating candidate pool")
    for idx, a in enumerate(pool):
        if time.time() >= deadline:
            break
        a = np.maximum(a, 0.0)
        if a.sum() < 1e-12:
            a = np.random.dirichlet(np.ones(n))
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Pool {idx:03d}] new best {val:.6f}")

    # ------------------------------------------------------------------
    # Phase II – smooth‑Lp exponentiated‑gradient refinement
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase II] Lp exponentiated‑gradient refinement")
        # schedule of p‑values (larger p = sharper approximation of max)
        p_schedule = [2, 4, 8, 16, 32]
        # Rough allocation: each p gets roughly 1/len(p_schedule) of the remaining budget
        for p in p_schedule:
            if time.time() >= deadline - 5:
                break
            remain = deadline - time.time()
            # Cap iterations so we never exceed the remaining time
            # Empirically ~0.001 s per iteration (FFT + updates)
            max_iters = min(200_000, int(remain / 0.001) // len(p_schedule))
            best_seq, best_val = _lp_expgrad_optimize(
                best_seq,
                n,
                conv_fft_len,
                corr_fft_len,
                p,
                deadline,
                max_iters=max_iters,
                eta0=0.8,
            )
            print(f"[Phase II p={p}] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # Phase III – simulated‑annealing local search (if time left)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase III] simulated‑annealing refinement")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase III] improved best to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
