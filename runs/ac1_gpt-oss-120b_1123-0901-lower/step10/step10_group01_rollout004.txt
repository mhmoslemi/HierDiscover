"""
Hybrid optimiser – aggressive multi‑phase search.

The routine builds a diverse pool of initial step‑functions (Dirichlet draws,
exponential / Gaussian families, simple spikes, warm‑start if available) and
prunes it quickly with a cheap smooth‑max surrogate.  The most promising
candidates are then refined with a long Adam run on the soft‑max approximation
of the max‑convolution, followed by an exponentiated‑gradient sub‑gradient phase
and a plain projected‑gradient descent on the exact max‑convolution.  Finally
the best result is optionally up‑sampled to a finer grid (8192 points) and
re‑optimised.

All optimisation is performed on the probability simplex (Σa=1).  The
evaluation function `evaluate_sequence` is scale‑invariant, therefore the
simplex parametrisation does not affect the final score.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Low‑level convolution utilities

def _next_pow2(x: int) -> int:
    """Return the smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Candidate factories (simple families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers

def _adam_optimize(a0: np.ndarray,
                   n: int,
                   conv_fft_len: int,
                   corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 300_000,
                   lr: float = 0.05,
                   lam_start: float = 0.1,
                   lam_factor: float = 1.02) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate of max(conv)."""
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    lam = lam_start

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c               # sum(a)=1 ⇒ denominator = 1
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray,
                                  n: int,
                                  conv_fft_len: int,
                                  corr_fft_len: int,
                                  deadline: float,
                                  max_iters: int = 800_000,
                                  eta0: float = 0.8) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient phase using exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 25_000 == 0:
                print(f"[ExpGrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)       # exponentiated‑gradient update
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray,
                     n: int,
                     conv_fft_len: int,
                     corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """Plain projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.15 / np.sqrt(it + 1.0)   # a slightly larger initial step
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray,
                    n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.08
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

# Main entry point

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function minimizing `evaluate_sequence`.
    The algorithm proceeds through:
        1) generation of a diverse pool,
        2) cheap pre‑selection using a smooth‑max surrogate,
        3) deep Adam optimisation of the top few,
        4) long exponentiated‑gradient sub‑gradient,
        5) projected gradient descent,
        6) optional up‑sampling to 8192 points.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # keep a safety margin

    # --------------------------------------------------------------
    # Problem size – we work on 4096 points (fast FFT) and optionally
    # on 8192 points in the final up‑sampling phase.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # Warm start handling (previous best, if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None
    else:
        warm = None

    # --------------------------------------------------------------
    # Build a large, diverse pool of start vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with a broad range of concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families (front‑heavy)
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))

    # Double‑Gaussian families (smooth bell‑shapes)
    for sigma in (0.05, 0.15, 0.3, 0.6):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law families (heavy tail)
    for expo in (-2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple deterministic shapes
    pool.append(_make_triangle(n))
    pool.append(_make_uniform(n))

    # Random sparse spikes (different granularities)
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    # A few completely random Dirichlet draws
    for _ in range(12):
        alpha = 10 ** np.random.uniform(-3, 1)  # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # Random mixtures of a couple of spikes – often good seeds
    for _ in range(6):
        a = np.zeros(n)
        pos = np.random.choice(n, size=2, replace=False)
        w = np.random.rand(2)
        w /= w.sum()
        a[pos] = w
        pool.append(a)

    # Ensure every entry is a proper probability vector
    init_vectors = []
    for a in pool:
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12:
            continue
        a = a / s
        init_vectors.append(a)

    # --------------------------------------------------------------
    # Phase I – cheap surrogate screening (smooth‑max with small λ)
    best_seq = None
    best_est = float("inf")
    lam_screen = 5.0
    for idx, a in enumerate(init_vectors):
        if time.time() > deadline:
            break
        conv = _full_convolution(a, conv_fft_len)
        # smooth‑max surrogate (λ small → cheap)
        w = np.exp(lam_screen * (conv - conv.max()))
        w /= w.sum()
        est_val = 2.0 * n * np.dot(w, conv)   # smooth approximation of max
        if est_val < best_est:
            best_est = est_val
            best_seq = a.copy()
        if idx % 20 == 0:
            print(f"[Screen] processed {idx+1}/{len(init_vectors)} candidates, "
                  f"current best estimate {best_est:.6f}")

    # If nothing survived, fall back to uniform
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # --------------------------------------------------------------
    # Phase II – deep Adam optimisation on a few elite seeds
    # pick the current best plus a couple of random seeds for diversity
    elite = [best_seq]
    while len(elite) < 5:
        a = _make_dirichlet(n, np.random.uniform(0.01, 5.0))
        elite.append(a)

    for idx, a0 in enumerate(elite):
        if time.time() > deadline:
            break
        sub_dead = min(deadline, time.time() + 115.0)   # up to ~2 min per candidate
        a_opt, val_opt = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=800_000, lr=0.12, lam_start=0.12, lam_factor=1.03)
        if val_opt < best_est:
            best_est = val_opt
            best_seq = a_opt.copy()
            print(f"[Adam elite {idx}] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase III – long exponentiated‑gradient sub‑gradient refinement
    if best_seq is not None and time.time() < deadline - 10:
        print("[ExpGrad] long sub‑gradient phase")
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=1_400_000, eta0=0.5)
        if val_opt < best_est:
            best_est = val_opt
            best_seq = a_opt.copy()
            print(f"[ExpGrad] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase IV – projected gradient descent on exact max‑convolution
    if best_seq is not None and time.time() < deadline - 10:
        print("[GD] final projected gradient descent")
        a_opt, val_opt = _max_conv_descent(
            best_seq, n, conv_fft_len, corr_fft_len,
            deadline, max_iters=600_000)
        if val_opt < best_est:
            best_est = val_opt
            best_seq = a_opt.copy()
            print(f"[GD] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase V – simulated‑annealing polishing (optional, time‑aware)
    if best_seq is not None and time.time() < deadline - 15:
        print("[Anneal] final annealing")
        a_opt, val_opt = _anneal_optimize(
            best_seq, n, conv_fft_len,
            deadline, max_moves=250_000)
        if val_opt < best_est:
            best_est = val_opt
            best_seq = a_opt.copy()
            print(f"[Anneal] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # Phase VI – optional up‑sampling to a finer grid (8192 points)
    if best_seq is not None and time.time() + 40 < deadline:
        try:
            print("[Upsample] attempting 8192‑point refinement")
            n_big = 8192
            conv_fft_len_big = _next_pow2(2 * n_big - 1)
            corr_fft_len_big = _next_pow2(3 * n_big - 2)

            # interpolate linearly onto the finer grid
            x_old = np.linspace(0, 1, n, endpoint=False)
            x_big = np.linspace(0, 1, n_big, endpoint=False)
            a_big = np.interp(x_big, x_old, best_seq)

            a_big = np.clip(a_big, 0.0, 1000.0)
            a_big /= a_big.sum()

            sub_dead_big = min(deadline, time.time() + 35.0)
            a_big_opt, val_big = _adam_optimize(
                a_big, n_big, conv_fft_len_big, corr_fft_len_big,
                sub_dead_big,
                max_iters=400_000,
                lr=0.08, lam_start=0.1, lam_factor=1.02)
            if val_big < best_est:
                best_est = val_big
                best_seq = a_big_opt.copy()
                print(f"[Upsample] improved to {val_big:.6f}")
        except Exception as exc:
            print(f"[Upsample] exception: {exc}")

    # --------------------------------------------------------------
    # Ensure a valid return (non‑negative, ≤ 1000)
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # (optional) compute the exact score with the provided evaluator for logging
    try:
        exact_score = evaluate_sequence(best_seq.tolist())
        print(f"Exact score reported by evaluator: {exact_score:.6f}")
    except Exception:
        pass

    print(f"Final best estimate {best_est:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
