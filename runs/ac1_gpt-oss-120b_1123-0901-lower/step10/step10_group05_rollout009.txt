"""
Hybrid random‑shape + gradient + annealing optimizer.

The search proceeds in three stages:

1️⃣  Build a large, diverse pool of initial step‑functions
    (uniform, power‑law ends‑heavy, Beta‑shapes, exponential,
    Gaussian bumps, random Dirichlet draws and sparse random spikes).
    All candidates are normalised to sum = 1 and the best few (by the
    exact objective = 2·n·max(a*a)) are kept.

2️⃣  Refine each of the elite candidates.
    * Exponentiated‑gradient (EG) steps – multiplicative updates that stay
      on the probability simplex.
    * Additive projected‑gradient (PG) steps – subtract a sub‑gradient
      and project back onto the simplex using the O(n log n) projection
      algorithm.
    Both refinements respect a decreasing step‑size schedule and store
    the best vector ever seen.

3️⃣  A short simulated‑annealing polish starting from the globally best
    vector.  Random mass transfers between two coordinates are accepted
    according to a temperature schedule; downhill moves are always kept.

The routine is time‑capped (≈ 1000 s).  If the budget expires early the
currently best sequence is returned.  All numbers are clipped to the
evaluator limits (≥ 0, ≤ 1000) before returning.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)‑1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) required for the convolution
    sub‑gradient.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective_from_conv(conv: np.ndarray, n: int) -> float:
    """Exact objective value 2·n·max(conv) (sum of a is assumed 1)."""
    return 2.0 * n * conv.max()


def _simplex_projection(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, sum x = 1}.
    Uses the algorithm of Wang & Carreira‑Perpiñán (2013).
    """
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero or negative – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w

            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:
                continue

            delta = a[i] * (np.random.rand() * 0.4)   # move up to 40 % of mass at i
            a2 = a.copy()
            a2[i] -= delta
            a2[j] += delta

            val2 = _obj(a2)

            if val2 < best_val_sa:
                best_val_sa = val2
                best_a_sa = a2.copy()
                a = a2
                print(f"[Anneal move {mv}] improved to {val2:.6f}")
            else:
                # Metropolis acceptance
                if np.random.rand() < np.exp(-(val2 - cur_val) / T):
                    a = a2
                # keep current val for next iteration
            cur_val = val2

        if best_val_sa < best_val:
            best_val = best_val_sa
            best_seq = best_a_sa.copy()
            print(f"[Anneal final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 5️⃣ Final clean‑up and return ------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (≥0, ≤1000) and return as a Python list.
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
