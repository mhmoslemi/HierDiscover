"""
Propose a low‑evaluation step‑function.

Algorithm
---------
1. Broad parametric sweep – generate a variety of symmetric / monotone
   shapes (power‑law, exponential, stretched‑exponential, beta,
   double‑Gaussian, random spikes).  Keep the best few candidates.

2. Fine‑tuning with a fast simulated‑annealing / hill‑climbing
   routine that updates the convolution incrementally (O(n) per move)
   instead of recomputing it with FFT each step.  The move transfers a
   random fraction of the mass from one index to another; the
   convolution is updated using the exact rank‑1 formulas derived from
   (a+Δ)*(a+Δ).  The temperature is decreased linearly; occasional full
   recomputation removes accumulated numerical drift.

The whole search respects the 1000 s budget and always returns a valid
non‑negative list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (used for the initial sweep)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (no clipping, no scaling) – used for comparison only
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n, dtype=np.float64)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_inverse_power(p: float, n: int) -> np.ndarray:
    """a_i ∝ (i+1)^p + (n-i)^p with p < 0 (heavier at the edges)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a

# Elite handling utilities
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=8):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])

# Fast simulated‑annealing with exact convolution updates
# ----------------------------------------------------------------------
def _simulated_anneal(
    a_start: np.ndarray,
    n: int,
    deadline: float,
    max_moves: int = 8_000_000,
    max_frac: float = 0.3,
) -> tuple[np.ndarray, float]:
    """Hill‑climbing / simulated annealing with O(n) updates per move."""
    a = a_start.copy()
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    sum_a = a.sum()
    factor = 2.0 * n / (sum_a * sum_a)

    best_max = conv.max()
    best_val = factor * best_max
    best_a = a.copy()

    current_max = best_max
    moves = 0
    recompute_interval = 250_000  # full recompute to keep numerical stability

    # linear temperature schedule (from ~5% of current max down to 1e‑7)
    T_start = best_max * 0.05 + 1e-12
    T_end = best_max * 1e-7 + 1e-12

    while time.time() < deadline and moves < max_moves:
        # linear cooling
        T = max(T_end, T_start * (1.0 - moves / float(max_moves)))

        i = np.random.randint(n)
        if a[i] <= 0.0:
            moves += 1
            continue
        j = np.random.randint(n)
        if i == j:
            moves += 1
            continue

        old_i = a[i]
        old_j = a[j]
        delta = old_i * (np.random.rand() * max_frac)  # move a fraction of the mass
        if delta <= 0.0:
            moves += 1
            continue

        # ----- apply proposed move (using the exact rank‑1 update) -----
        # conv slice updates use the *old* a vector (still stored in `a` now)
        conv[i : i + n] -= 2.0 * delta * a
        conv[j : j + n] += 2.0 * delta * a
        conv[2 * i] += delta * delta
        conv[2 * j] += delta * delta
        conv[i + j] -= 2.0 * delta * delta

        # update the vector itself
        a[i] = old_i - delta
        a[j] = old_j + delta

        new_max = conv.max()

        # ----- acceptance test -----
        accept = False
        if new_max <= current_max:
            accept = True
        else:
            # occasional uphill moves
            if T > 0.0 and np.random.rand() < math.exp(-(new_max - current_max) / T):
                accept = True

        if accept:
            current_max = new_max
            if new_max < best_max:
                best_max = new_max
                best_val = factor * best_max
                best_a = a.copy()
                # progress report
                if moves % 200_000 == 0:
                    print(f"[Anneal move={moves}] new best {best_val:.6f}")
        else:
            # ----- revert the move (exact inverse operation) -----
            # restore a to the old state
            a[i] = old_i
            a[j] = old_j
            # revert conv using the same delta and the restored a
            conv[i : i + n] += 2.0 * delta * a
            conv[j : j + n] -= 2.0 * delta * a
            conv[2 * i] -= delta * delta
            conv[2 * j] -= delta * delta
            conv[i + j] += 2.0 * delta * delta
            # current_max stays unchanged

        moves += 1

        # periodic full recompute to combat numerical drift
        if moves % recompute_interval == 0:
            conv = _full_convolution(a, _next_pow2(2 * n - 1))
            current_max = conv.max()
            # keep best consistent
            if current_max < best_max:
                best_max = current_max
                best_val = factor * best_max
                best_a = a.copy()
                print(f"[Anneal recompute] improved to {best_val:.6f}")

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function (list of heights) with a low
    evaluation score.  The routine respects the given time budget and
    always returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5  # safety margin

    # --------------------------------------------------------------
    # Problem dimension (power‑of‑two length for fast FFT)
    # --------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)

    # --------------------------------------------------------------
    # 1) Wide parametric sweep – keep a few elites
    # --------------------------------------------------------------
    elite = []  # list of (objective, vector)

    def _maybe_update(a_vec):
        val = _objective(a_vec, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a_vec)

    # symmetric power‑law (negative exponents concentrate mass at the ends)
    for p in np.linspace(-5.0, -0.05, 30):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 30):
        _maybe_update(_candidate_exponential(alpha, n))

    # stretched exponential (α,β)
    alphas = np.logspace(-4, 0.5, 12)
    betas = np.linspace(0.3, 3.0, 12)
    for alpha in alphas:
        for beta in betas:
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta distributions (wide shape range)
    a_pars = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0, 10.0]
    for a_par in a_pars:
        for b_par in a_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian (mirrored)
    for sig in np.logspace(-3, -0.1, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 20):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # monotone left‑power (allow negative exponents)
    for p in np.linspace(-4.0, 2.0, 28):
        _maybe_update(_candidate_left_power(p, n))

    # inverse‑power (edges heavy)
    for p in np.linspace(-4.5, -0.1, 25):
        _maybe_update(_candidate_inverse_power(p, n))

    # random spike patterns (different sparsities)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _maybe_update(_candidate_spike(k, n, random=True))

    # Dirichlet draws with various concentration parameters
    concs = [0.05, 0.1, 0.2, 0.5, 1.0, 2.0]
    for c in concs:
        _maybe_update(np.random.dirichlet(np.full(n, c)))

    # warm start from previously stored global (if present)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    # pick the current best from the sweep
    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # --------------------------------------------------------------
    # 2) Intensive simulated‑annealing fine‑tuning
    # --------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        print("[Phase‑Anneal] intensive simulated‑annealing")
        a_fine, val_fine = _simulated_anneal(
            best_seq,
            n,
            deadline,
            max_moves=int(5e6),
            max_frac=0.25,
        )
        if val_fine < best_val:
            best_val, best_seq = val_fine, a_fine.copy()
            print(f"[Phase‑Anneal] improved to {best_val:.6f}")

    # --------------------------------------------------------------
    # Safety – ensure a valid return (non‑negative, finite, within limits)
    # --------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
