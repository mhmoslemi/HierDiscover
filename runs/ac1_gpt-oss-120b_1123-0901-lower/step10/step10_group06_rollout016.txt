    if a_sum < 1e-12: a = np.random.dirichlet(np.ones(n))

    it = 0

        # compute convolution
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
        # identify active indices
        active = np.isclose(conv, max_c, atol=0.0, rtol=1e-12)
        # compute w vector: uniform over active positions
        w = np.zeros_like(conv)
        if active.any():

        # gradient

            a_sum = a.sum()
        a /= a_sum
        it +=1

"""
Propose a step‑function (list of heights) with a low evaluation score.

Algorithm
---------
1. **Broad parametric sweep** – generate thousands of candidates from a
   rich family of shapes (asymmetric exponentials, power‑laws,
   stretched‑exponentials, beta densities, mixtures, random Dirichlet,
   spike patterns …).  Each candidate is normalised to unit sum and its
   exact objective (2·n·max(conv)/sum²) is evaluated with an FFT‑based
   convolution.  The best K=12 candidates are kept.

2. **Adam on a smooth‑max surrogate** – for the top few elite candidates
   a short Adam optimisation on the log‑sum‑exp surrogate of the
   convolution maximum is performed.  This quickly moves mass away from
   the most active constraints while staying on the simplex (soft‑max
   parametrisation).

3. **Exact‑max exponentiated‑gradient descent** – the current best
   sequence is then refined with a long sub‑gradient loop that uses the
   exact max‑convolution.  The update is an exponentiated‑gradient
   (mirror‑descent on the simplex) with a 1/√t step‑size schedule.

4. **Simulated‑annealing fine‑tuning** – a final simple annealing stage
   moves a random fraction of mass between two positions and accepts
   worsening moves with a temperature schedule.

The routine respects the supplied time budget (default 1000 s) and always
returns a non‑negative list that can be fed directly to ``evaluate_sequence``.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Low‑level FFT utilities
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (scale‑invariant)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, conv_fft_len)
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all return a unit‑sum non‑negative vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_right_exponential(alpha: float, n: int) -> np.ndarray:
    # simply reverse a left‑exponential
    return _candidate_left_exponential(alpha, n)[::-1]


def _candidate_combined_exponential(alpha: float, beta: float,
                                    w: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-alpha * i)
    right = np.exp(-beta * (n - 1 - i))
    a = w * left + (1.0 - w) * right
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑density discretised on {0,…,n‑1}."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_dirichlet(conc: float, n: int, rng: np.random.Generator) -> np.ndarray:
    a = rng.dirichlet(np.full(n, conc))
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, rng: np.random.Generator) -> np.ndarray:
    """k spikes with Dirichlet masses."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = rng.choice(n, size=k, replace=False)
        masses = rng.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_mix(v1: np.ndarray, v2: np.ndarray, w: float) -> np.ndarray:
    """Convex mixture of two vectors."""
    a = w * v1 + (1.0 - w) * v2
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full_like(a, 1.0 / a.shape[0])
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Helpers for elite management
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=12):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
# Optimisers
# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 1_000_000) -> tuple[np.ndarray, float]:
    """Adam on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07                         # a tad larger than the reference version
    lam = 0.05                        # start temperature for smooth‑max
    lam_factor = 1.04                 # increase a bit faster

    best_a = a0.copy()
    best_val = _objective(a0, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent (soft‑max) gradient
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (sharpen)
        lam = min(lam * lam_factor, 1e7)

        # learning‑rate decay every 80 k steps
        if it % 80_000 == 0 and it > 0:
            lr *= 0.94

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 5_000_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set – indices attaining the current max
        active = np.isclose(conv, max_c, atol=0.0, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        # sub‑gradient of max (2 * cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                     conv_fft_len: int,
                     deadline: float,
                     max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing fine‑tuning."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """Search for a non‑negative step‑function with a low evaluation score."""

    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Problem dimension (power of two for fast FFT)
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Wide parametric sweep – keep a few elites
    # ------------------------------------------------------------------
    elite = []                                   # list of (objective, vector)

    def _maybe_update(a: np.ndarray):
        val = _objective(a, n, conv_fft_len)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    sweep_deadline = start_time + min(0.2 * budget_s, 250.0)

    # ---- symmetric power‑law (covers many exponents) ----
    for p in np.linspace(-5.0, -0.05, 45):
        if time.time() > sweep_deadline:
            break
        _maybe_update(_candidate_power(p, n))

    # ---- left‑exponential (monotone) ----
    for alpha in np.logspace(-5, 2.0, 40):
        if time.time() > sweep_deadline:
            break
        _maybe_update(_candidate_left_exponential(alpha, n))

    # ---- right‑exponential (mirror of left) ----
    for alpha in np.logspace(-5, 2.0, 30):
        if time.time() > sweep_deadline:
            break
        _maybe_update(_candidate_right_exponential(alpha, n))

    # ---- combined asymmetric exponentials ----
    for _ in range(80):
        if time.time() > sweep_deadline:
            break
        alpha = 10 ** rng.uniform(-5, 2)
        beta = 10 ** rng.uniform(-5, 2)
        w = rng.random()
        _maybe_update(_candidate_combined_exponential(alpha, beta, w, n))

    # ---- stretched exponential (α,β) ----
    for _ in range(70):
        if time.time() > sweep_deadline:
            break
        alpha = 10 ** rng.uniform(-4, 0.5)
        beta = rng.uniform(0.3, 3.0)
        _maybe_update(_candidate_stretched(alpha, beta, n))

    # ---- beta distributions (shape parameters) ----
    for _ in range(80):
        if time.time() > sweep_deadline:
            break
        a_par = 10 ** rng.uniform(-2, 1.5)  # range roughly 0.01 … 30
        b_par = 10 ** rng.uniform(-2, 1.5)
        _maybe_update(_candidate_beta(a_par, b_par, n))

    # ---- double Gaussian (mirrored) ----
    for _ in range(70):
        if time.time() > sweep_deadline:

        sig = 10 ** rng.uniform(-4, -0.1)

    # ---- random Dirichlet draws (various concentrations) ----
    for _ in range(120):
        if time.time() > sweep_deadline:
            break
        conc = 10 ** rng.uniform(-2, 1)      # from 0.01 to 10
        _maybe_update(_candidate_dirichlet(conc, n, rng))

    # ---- random spike patterns (different sparsities) ----
    for k in [16, 32, 64, 128, 256, 512]:
        for _ in range(4):
            if time.time() > sweep_deadline:

            _maybe_update(_candidate_spike(k, n, rng))

    # ---- mixtures of already discovered elites (adds flexibility) ----

        for _ in range(100):

            w = rng.random()
            _maybe_update(_candidate_mix(v1, v2, w))

    # ---- warm start from a previously saved sequence (if present) ----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)

            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]

    else:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

                adam_deadline,

            best_seq,

            conv_fft_len,
            corr_fft_len,

        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        ann_deadline = deadline

            best_seq,
            n,
            conv_fft_len,
            ann_deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final safety checks and return
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final score (evaluate_sequence not available).")

    # expose the result for a potential warm‑start in a later run
    globals()["height_sequence_2"] = best_seq.tolist()

    return best_seq.tolist()
