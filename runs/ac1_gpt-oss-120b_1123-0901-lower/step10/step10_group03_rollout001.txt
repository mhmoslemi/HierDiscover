"""
Step‑function optimizer.

The algorithm proceeds in four stages:

1. **Broad parametric sweep** – a large collection of parametric families
   (exponential, power‑law, stretched‑exponential, Gamma‑like, Beta,
   double‑Gaussian, semi‑circle, sine‑power, random spikes, and mixtures)
   is generated.  Each candidate is normalised to sum = 1 and its exact
   objective 2·n·max(a*a) is computed via an FFT‑based convolution.
   The best shape so far is kept.

2. **Smooth‑max Adam** – from the best candidate we run a short Adam
   optimisation on the log‑sum‑exp (smooth‑max) surrogate of the max‑
   convolution.  The optimisation works on the simplex via a soft‑max
   parametrisation.  This pushes the solution into a better basin.

3. **Repeated exact‑max exponentiated‑gradient** – we run a long
   sub‑gradient (exponentiated‑gradient) phase from the current best
   vector.  When time permits we restart the sub‑gradient from random
   Dirichlet initialisations and keep the best result found.

4. **Intensive annealing** – a cheap simulated‑annealing fine‑tuner that
   repeatedly transfers a random fraction of mass between two coordinates.
   This stage is allowed to consume the remaining time budget.

All stages respect the overall time budget (default 1000 s) and the routine
always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a needed for sub‑gradient (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a) (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β) + exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_semicircle(p: float, n: int) -> np.ndarray:
    """Symmetric semi‑circular shape: a_i ∝ (1‑x²)^p,  x∈[-1,1]."""
    i = np.arange(n, dtype=np.float64)
    x = 2 * i / (n - 1) - 1.0
    a = np.maximum(0.0, (1 - x ** 2)) ** p
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine_power(p: float, n: int) -> np.ndarray:
    """Symmetric sine‑power shape: a_i ∝ sin^p(π·(i+0.5)/n)."""
    i = np.arange(n, dtype=np.float64)
    a = np.sin(np.pi * (i + 0.5) / n) ** p
    a = np.maximum(a, 0.0)
    a_sum = a.sum()
    if a_sum == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best vector (on the simplex) seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1          # temperature for smooth‑max
    lam_factor = 1.01

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # Exact objective (for tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # Smooth‑max weights (log‑sum‑exp)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # Gradient of the surrogate
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # Mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # Sharpening schedule
        lam = min(lam * lam_factor, 1e7)

        # Gentle learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates
    (mirror descent on the simplex).  Step size ηₜ = η₀/√(t+1).
    Runs until the deadline.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # Identify active indices attaining the max
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # Exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # Renormalise, with a safety net
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing style local search.
    Randomly transfers a fraction of mass between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        # Choose source from relatively large entries
        i = np.random.choice(np.where(a > 1e-12)[0])
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # Transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
        else:
            if np.random.rand() < np.exp(-(val2 - best_val) / T):
                a = a2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # Length of the step function – 4096 works well for the benchmark.
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep ------------------------------------
    # Symmetric families
    for p in np.linspace(-6.0, -0.1, 30):
        a = _candidate_power(p, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    for alpha in np.logspace(-4, 2.5, 35):
        a = _candidate_exponential(alpha, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    alphas = np.logspace(-3, 1.5, 20)
    betas = np.linspace(0.3, 4.0, 12)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            val = _objective(a, n)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] new best {val:.6f}")

    # Semi‑circle and sine‑power families (new)
    for p in np.linspace(0.5, 4.0, 15):
        a = _candidate_semicircle(p, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    for p in np.linspace(0.5, 4.0, 15):
        a = _candidate_sine_power(p, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # Monotone families
    for alpha in np.logspace(-4, 2.0, 30):
        a = _candidate_left_exponential(alpha, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    for p in np.linspace(-6.0, 2.0, 30):
        a = _candidate_left_power(p, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    for beta in np.linspace(0.5, 4.0, 12):
        for alpha in np.logspace(-4, 1.5, 25):
            a = _candidate_left_stretched(alpha, beta, n)
            val = _objective(a, n)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] new best {val:.6f}")

    # Gamma‑like
    shapes = np.linspace(0.2, 4.0, 12)
    scales = np.logspace(-4, 0.5, 12)
    for shape in shapes:
        for scale in scales:
            a = _candidate_gamma(shape, scale, n)
            val = _objective(a, n)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] new best {val:.6f}")

    # Beta‑distribution (left‑skewed)
    a_pars = np.logspace(-2, 0.5, 12)
    b_pars = np.logspace(0, 1.0, 12)
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            val = _objective(a, n)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] new best {val:.6f}")

    # Double Gaussian
    for sig in np.logspace(-3, -0.2, 15):
        a = _candidate_double_gaussian(sig, n)
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # Random spike patterns
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(4):
            a = _candidate_spike(k, n, random=True)
            val = _objective(a, n)
            if val < best_val:
                best_val, best_seq = val, a.copy()
                print(f"[Sweep] new best {val:.6f}")

    # Mixture of two exponentials
    for _ in range(400):
        alpha1 = 10 ** np.random.uniform(-4, 2)
        alpha2 = 10 ** np.random.uniform(-4, 2)
        w1 = np.random.rand()
        a_mix = w1 * _candidate_left_exponential(alpha1, n) + \
                (1 - w1) * _candidate_left_exponential(alpha2, n)
        a_mix /= a_mix.sum()
        val = _objective(a_mix, n)
        if val < best_val:
            best_val, best_seq = val, a_mix.copy()
            print(f"[Sweep] new best {val:.6f}")

    # Mixture of exponential + power‑law (new)
    for _ in range(200):
        alpha = 10 ** np.random.uniform(-4, 2)
        p = np.random.uniform(-6, 2)
        w = np.random.rand()
        a_mix = w * _candidate_left_exponential(alpha, n) + \
                (1 - w) * _candidate_left_power(p, n)
        a_mix /= a_mix.sum()
        val = _objective(a_mix, n)
        if val < best_val:
            best_val, best_seq = val, a_mix.copy()
            print(f"[Sweep] new best {val:.6f}")

    # Dirichlet random vectors (various concentrations)
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        val = _objective(a, n)
        if val < best_val:
            best_val, best_seq = val, a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # Warm start from a previously‑found global variable (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            val = _objective(w, n)
            if val < best_val:
                best_val, best_seq = val, w.copy()
                print(f"[Warm‑start] new best {val:.6f}")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Short Adam refinement (smooth‑max) -----------------------------
    if best_seq is not None and time.time() < deadline - 20:
        print("[Phase‑Adam] short Adam optimisation")
        adam_deadline = min(deadline - 5, time.time() + 30.0)   # up to ~30 s
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val, best_seq = val_adam, a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Repeated sub‑gradient optimisation with random restarts -------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] long sub‑gradient optimisation (with restarts)")

        # Main run starting from the current best
        a_cur, val_cur = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            eta0=1.5,
        )
        if val_cur < best_val:
            best_val, best_seq = val_cur, a_cur.copy()
            print(f"[Phase‑Subgrad] improved to {val_cur:.6f}")

        # Additional random‑restart runs while time permits
        while time.time() < deadline - 5:
            remaining = deadline - time.time()
            # give each restart at most 25 s
            sub_deadline = time.time() + min(25.0, remaining)
            a0 = np.random.dirichlet(np.ones(n))
            a_res, val_res = _subgradient_optimize_expgrad(
                a0,
                n,
                conv_fft_len,
                corr_fft_len,
                sub_deadline,
                eta0=1.5,
            )
            if val_res < best_val:
                best_val, best_seq = val_res, a_res.copy()
                print(f"[Phase‑Subgrad‑Restart] improved to {val_res:.6f}")

    # ------------------------------------------------------------------
    # 4) Intense annealing fine‑tuning ---------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=400_000,
        )
        if val_ann < best_val:
            best_val, best_seq = val_ann, a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks ------------------------------------------------
    if best_seq is None or best_seq.sum() < 0.01:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Final public evaluation (if the evaluator is available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print("Final evaluation failed:", exc)

    return best_seq.tolist()
