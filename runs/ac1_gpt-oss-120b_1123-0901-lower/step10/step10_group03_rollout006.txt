def de_optimize(gen_func, bounds, max_evals):
    dim = len(bounds)
    popsize = 10
    # initial population random within bounds
    pop = np.array([np.array([np.random.uniform(low, high) for (low,high) in bounds]) for _ in range(popsize)])
    # evaluate initial
    obj = np.array([_objective(gen_func(*ind, n=n), n) for ind in pop])
    best_idx = obj.argmin()
    best = pop[best_idx].copy()
    best_val = obj[best_idx]
    evals = popsize

    while evals < max_evals:
        for i in range(popsize):
            # mutation: select 3 distinct others
            idxs = list(range(popsize))
            idxs.remove(i)
            a,b,c = pop[np.random.choice(idxs, 3, replace=False)]
            # F scaling factor maybe 0.8
            F = 0.8
            mutant = np.clip(a + F * (b - c), [low for low,high in bounds], [high for low,high in bounds])
            # crossover: CR=0.9
            CR = 0.9
            cross = np.random.rand(dim) < CR
            if not cross.any():
                cross[np.random.randint(dim)] = True
            trial = np.where(cross, mutant, pop[i])
            # evaluate
            trial_val = _objective(gen_func(*trial, n=n), n)
            evals += 1
            # selection
            if trial_val < obj[i]:
                pop[i] = trial
                obj[i] = trial_val
                if trial_val < best_val:
                    best_val = trial_val
                    best = trial.copy()
            if evals >= max_evals or time.time() >= deadline:
                break
        if time.time() >= deadline:
            break
    return best, best_val
```

"""
Step‑function optimizer – refined version.

Main ideas
----------
1. A broad random search over several low‑dimensional parametric families
   (exponential, stretched‑exponential, double‑Gaussian, Gamma, Beta, …).
   For each family we first sample many points, keep the best and then
   perform a cheap Nelder‑Mead refinement in the (log‑scaled) parameter
   space.

2. The best sequence from (1) is fed to an exponentiated‑gradient
   sub‑gradient descent on the exact max‑convolution objective.
   This phase runs for the remainder of the time budget and dramatically
   reduces the objective.

3. A very light simulated‑annealing fine‑tuner is applied at the very end
   if any time is left.

All steps respect the 1000 s budget and always return a valid
non‑negative list.  The code is self‑contained and uses only numpy and
scipy (both are available in the execution environment).
"""

import time
import numpy as np
from scipy.optimize import minimize

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Cross‑correlation w ★ a (same length as a)."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β)+exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_mixture_exponential(alpha1: float, alpha2: float,
                                   w: float, n: int) -> np.ndarray:
    """Weighted mixture of two left‑exponential shapes (already normalised)."""
    a1 = _candidate_left_exponential(alpha1, n)
    a2 = _candidate_left_exponential(alpha2, n)
    # each sum to 1, mixture also sums to 1
    return w * a1 + (1.0 - w) * a2

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates
    (mirror descent on the simplex).  Step size ηₜ = η₀/√(t+1).
    Runs until the deadline.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 200_000 == 0:
                print(f"[Subgrad] it={it:,}  best={best_val:.6f}")

        # Active set for the max (may contain several indices)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing fine‑tuner."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2
        it += 1

    return best_a, best_val

def _sample_param(low, high, log):
    """Draw a single parameter – log‑scale if requested."""
    if log:
        return 10 ** np.random.uniform(np.log10(low), np.log10(high))
    else:
        return np.random.uniform(low, high)


def _random_search(gen, param_specs, n_samples, n):
    """
    Random sampling in the (possibly log‑scaled) parameter space.
    Returns the best (vector, value, params).
    """
    best_val = float("inf")
    best_a = None
    best_params = None
    for _ in range(n_samples):
        params = [_sample_param(low, high, log) for (low, high, log) in param_specs]
        a = gen(*params, n=n)
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a
            best_params = params
    return best_a, best_val, best_params


def _refine_nelder_mead(gen, param_specs, init_params, n, maxfev=300):
    """
    Local refinement with Nelder‑Mead.  The optimisation works in the
    log‑domain for parameters flagged as log‑scaled.
    """
    # transform initial guess
    x0 = []
    for p, (low, high, log) in zip(init_params, param_specs):
        x0.append(np.log10(p) if log else p)

    # bounds for clipping inside the objective
    bounds = [(np.log10(low) if log else low,
               np.log10(high) if log else high) for (low, high, log) in param_specs]

    def fun(x):
        # clip to bounds and undo log‑scale
        actual = []
        for xi, (low, high, log) in zip(x, param_specs):
            if log:
                xi = np.clip(xi, np.log10(low), np.log10(high))
                actual.append(10 ** xi)
            else:
                xi = np.clip(xi, low, high)
                actual.append(xi)
        a = gen(*actual, n=n)
        return _objective(a, n)

    res = minimize(fun, x0, method="Nelder-Mead",
                   options={"maxfev": maxfev, "disp": False})

    # build final params
    final_params = []
    for xi, (low, high, log) in zip(res.x, param_specs):
        if log:
            xi = np.clip(xi, np.log10(low), np.log10(high))
            final_params.append(10 ** xi)
        else:
            final_params.append(np.clip(xi, low, high))

    best_a = gen(*final_params, n=n)
    best_val = _objective(best_a, n)
    return best_a, best_val, final_params

    families = [
        # name, generator, list of (low, high, log) for each param
        ("exponential",
         _candidate_exponential,
         [(1e-5, 500.0, True)]),

        ("stretch",
         _candidate_stretched,
         [(1e-5, 200.0, True), (0.3, 6.0, False)]),

        ("double_gaussian",
         _candidate_double_gaussian,
         [(5e-4, 0.8, False)]),

        ("gamma",
         _candidate_gamma,
         [(0.1, 8.0, False), (1e-4, 5.0, True)]),

        ("beta",
         _candidate_beta,
         [(0.02, 8.0, True), (0.2, 8.0, True)]),

        ("left_exponential",
         _candidate_left_exponential,
         [(1e-5, 300.0, True)]),

        ("left_power",
         _candidate_left_power,
         [(-8.0, 4.0, False)]),

        ("mixture_exponential",
         _candidate_mixture_exponential,
         [(1e-5, 200.0, True), (1e-5, 200.0, True), (0.0, 1.0, False)])
    ]

    # coarse random sampling budgets (adjustable)
    samples_per_family = 200
    for name, gen, specs in families:
        if time.time() >= deadline:
            break
        print(f"[Search] sampling family {name}")

        a_rand, val_rand, params_rand = _random_search(
            gen, specs, samples_per_family, n)

        # local refinement (a small number of function evaluations)
        a_ref, val_ref, params_ref = _refine_nelder_mead(
            gen, specs, params_rand, n, maxfev=300)

        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Search] new overall best {best_val:.6f} from {name}")

    # ------------------------------------------------------------------
    # 2) Optional warm‑start from previous global variable ----------------
    # (helps if the earlier run already found something good)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            val_warm = _objective(w, n)
            if val_warm < best_val:
                best_val = val_warm
                best_seq = w.copy()
                print(f"[Warm‑start] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Long sub‑gradient optimisation (exact max) -----------------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] starting exponentiated‑gradient descent")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Very cheap simulated‑annealing fine‑tuning (if time permits) -----
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanity – ensure a non‑negative, bounded list -----------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Report the score obtained via the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence missing).")

    return best_seq.tolist()
