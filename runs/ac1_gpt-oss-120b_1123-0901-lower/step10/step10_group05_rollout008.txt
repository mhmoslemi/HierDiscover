def proj_simplex(v):
    # v: np.ndarray of floats, shape (n,)
    # project onto {x >= 0, sum x = 1}
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n+1)
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

    conv = np.fft.irfft(fa*fa, fft_len)
    return conv[:2*a.shape[0]-1]
```

Define cross-correlation:

```
def cross_correlation(w, a, fft_len):
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw*fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]
```

def proj_simplex(v):
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u - cssv / np.arange(1, n+1) > 0)[0]
    if rho.size == 0:
        # all negative, return uniform distribution
        return np.full(n, 1.0/n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid search for a low‑max‑convolution step function.

The algorithm consists of three phases:

1. **Broad parametric sweep** – generate a large, diverse set of
   candidates (warm‑start, power‑law / Beta families, Dirichlet draws,
   block‑constant and random‑spike patterns).  All candidates are
   normalised to ℓ₁‑norm = 1 and the best few are retained.

2. **Fine‑grained parametric refinement** – if the best candidate belongs
   to a parametric family (power‑law or Beta) we perform a local grid
   search on its continuous parameters to improve the objective.

3. **Projected‑gradient optimisation** – start from the best candidate
   and apply many steps of projected gradient descent on the convex
   objective 2·n·max(a∗a) (the sum is fixed, so only the maximum of the
   self‑convolution matters).  The gradient is the cross‑correlation of
   the active (max‑attaining) convolution entries with the current
   vector; after each step the iterate is projected back onto the simplex
   (non‑negative, sum = 1).  The step size follows a 1/√t schedule and a
   simple back‑tracking line‑search is used.

A short simulated‑annealing post‑process is applied before returning the
best sequence found within the allotted time budget (≈ 1000 s).  All
numbers are clipped to the evaluator’s limits (≥0, ≤1000).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation w ★ a needed for the sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact objective 2·n·max(a*a) under the assumption sum(a)=1."""
    conv = _full_convolution(a, conv_fft_len)
    return 2.0 * n * conv.max()


def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of vector v onto the probability simplex:
    { x ≥ 0 , sum x = 1 }.
    """
    v = np.asarray(v, dtype=np.float64)
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all entries are non‑positive → uniform distribution
        return np.full(n, 1.0 / n, dtype=np.float64)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function that minimises the
    evaluate_sequence() score.  The routine respects the given time budget
    and returns a Python list with the best sequence found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # problem size (the reference solution used n = 4096)
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # store the (value, vector, meta) triples of the current elite pool
    elite: list[tuple[float, np.ndarray, dict]] = []
    elite_max = 20                      # keep at most this many elites

    def _maybe_push(val: float, vec: np.ndarray, meta: dict):
        """Insert a candidate into the elite list if it is among the best."""
        nonlocal elite
        if val >= float("inf"):
            return
        if len(elite) < elite_max:
            elite.append((val, vec.copy(), meta.copy()))
            elite.sort(key=lambda x: x[0])
        else:
            if val < elite[-1][0]:
                elite[-1] = (val, vec.copy(), meta.copy())
                elite.sort(key=lambda x: x[0])

    # ------------------------------------------------------------------
    # 1️⃣  Warm‑start (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() > 0.01:
            w = w / w.sum()
            val = _objective(w, n, conv_fft_len)
            _maybe_push(val, w, {"type": "warm"})

    # ------------------------------------------------------------------
    # 2️⃣  Parametric families

    for alpha in np.linspace(-4.0, -0.1, 200):
        shape = prod ** alpha
        if shape.sum() == 0.0:

    # explore a dense grid on (0,2] for both parameters
    p_vals = np.geomspace(0.01, 2.0, 30)

            if shape.sum() == 0.0:

            shape = shape / shape.sum()
            val = _objective(shape, n, conv_fft_len)
            _maybe_push(val, shape,
                        {"type": "beta", "p": float(p), "q": float(q)})

    # ---- Dirichlet draws ------------------------------------------------
    for alpha in (0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0):
        vec = np.random.dirichlet(alpha * np.ones(n))
        val = _objective(vec, n, conv_fft_len)
        _maybe_push(val, vec, {"type": "dirichlet", "alpha": float(alpha)})

    # ---- Piece‑wise constant blocks ------------------------------------
    for blocks in (8, 16, 32, 64):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            a[b * block_len:(b + 1) * block_len] = np.random.rand()
        if a.sum() == 0.0:
            continue
        a = a / a.sum()
        val = _objective(a, n, conv_fft_len)
        _maybe_push(val, a, {"type": "blocks", "blocks": blocks})

    # ---- Random sparse spikes ------------------------------------------
    for k in (64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        pos = np.random.choice(n, size=k, replace=False)
        a[pos] = np.random.dirichlet(np.ones(k))
        val = _objective(a, n, conv_fft_len)
        _maybe_push(val, a, {"type": "spikes", "k": k})

    # ------------------------------------------------------------------
    # 3️⃣  Keep the best few elite candidates
    if not elite:
        # fallback: uniform distribution
        uniform = np.full(n, 1.0 / n, dtype=np.float64)
        elite.append((_objective(uniform, n, conv_fft_len), uniform,
                      {"type": "uniform"}))

    # ------------------------------------------------------------------
    # 4️⃣  Fine‑grained refinement of parametric elites
    #    (only for power‑law and Beta families)
    refined_elite = []
    for val, vec, meta in elite:
        if meta["type"] == "power":
            # local grid around the best alpha
            alpha0 = meta["alpha"]
            for delta in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01, 0.005):
                alphas = np.linspace(alpha0 - delta, alpha0 + delta, 41)
                for a in alphas:
                    shape = prod ** a
                    if shape.sum() == 0.0:
                        continue
                    shape = shape / shape.sum()
                    vnew = _objective(shape, n, conv_fft_len)
                    if vnew < val:
                        val = vnew
                        vec = shape
                        alpha0 = a
            refined_elite.append((val, vec, {"type": "power", "alpha": alpha0}))
        elif meta["type"] == "beta":
            p0, q0 = meta["p"], meta["q"]
            best_val, best_vec, best_p, best_q = val, vec, p0, q0
            # multi‑scale local search
            for delta in (0.5, 0.2, 0.1, 0.05, 0.02, 0.01):
                p_grid = np.linspace(max(best_p - delta, 1e-6), best_p + delta, 11)
                q_grid = np.linspace(max(best_q - delta, 1e-6), best_q + delta, 11)
                for p in p_grid:
                    for q in q_grid:
                        shape = (x ** p) * ((1.0 - x) ** q)
                        if shape.sum() == 0.0:
                            continue
                        shape = shape / shape.sum()
                        vnew = _objective(shape, n, conv_fft_len)
                        if vnew < best_val:
                            best_val, best_vec = vnew, shape

        else:
            refined_elite.append((val, vec, meta))

    # Re‑select the global best after refinement

    best_val, best_seq, best_meta = refined_elite[0]

    #    The step size follows η / sqrt(t) with back‑tracking.
    it = 0
    step_const = 0.8                     # initial constant for step size
    while time.time() < deadline - 2.0:

        conv = _full_convolution(a, conv_fft_len)

            step_const *= 0.9

        # keep the best ever seen

            print(f"[PGD it={it}] new best {cur_val:.6f}")

        # occasional random perturbation to escape plateaus

        # stop the PGD loop if the gradient disappears (practically converged)
        if np.linalg.norm(grad) < 1e-14:

    # ------------------------------------------------------------------
    # 6️⃣  Very short simulated‑annealing polish
    if time.time() < deadline - 1.0:
        a = best_seq.copy()
        conv = _full_convolution(a, conv_fft_len)
        cur_val = 2.0 * n * conv.max()
        best_val_sa = cur_val

        max_moves = 60000

                break
            T = max(T_end, T_start * (1.0 - mv / max_moves))

            i = np.random.randint(n)
            if a[i] < 1e-14:
                continue
            j = np.random.randint(n)
            if i == j:

            a2[i] -= delta

            print(f"[SA final] best {best_val:.6f}")

    # ------------------------------------------------------------------
    # 7️⃣  Final production: clip to evaluator limits and return
    best_seq = np.clip(best_seq, 0.0, 1000.0)
    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
