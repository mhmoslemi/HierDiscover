"""
Improved step‑function optimiser.

The optimiser consists of four stages:

1️⃣  A very broad parametric sweep.  In addition to the families used in the
    reference code we add logistic, Laplace and arcsine shapes and also many
    random Dirichlet / power‑law draws.  The best few candidates are kept.

2️⃣  Smooth‑max mirror‑descent (Adam on the log‑probabilities).  The objective
    2·n·max(conv) is approximated by the smooth‑max
        sm(β) = (1/β)·log Σ exp(β·convₖ)
    and we perform Adam updates on the log‑soft‑max parameters.  β is
    increased exponentially so that the surrogate converges to the true
    max.  This stage receives the bulk of the remaining time.

3️⃣  Exact max sub‑gradient (exponentiated‑gradient) – a few thousand
    iterations are used to polish the solution obtained in stage 2.

4️⃣  A short simulated‑annealing fine‑tuner that swaps mass between two
    positions.  It runs only if time permits.

All stages respect the overall 1000 s budget and the routine always returns
a valid non‑negative list of heights.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for gradients:
         (w ★ a)[i] = Σ_j w[j]·a[i‑j] .
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    return a / a.sum()


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    return a / a.sum()


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = np.nan_to_num(a * b, nan=0.0, posinf=0.0, neginf=0.0)
    return a / a.sum()


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    return a / a.sum()


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    return a / a.sum()


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (Dirichlet masses) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    return a / a.sum()


def _candidate_logistic(scale: float, n: int) -> np.ndarray:
    """Monotone logistic decay a_i = 1/(1+exp((i‑mid)/scale))."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = 1.0 / (1.0 + np.exp((i - mid) / scale))
    return a / a.sum()


def _candidate_laplace(scale: float, n: int) -> np.ndarray:
    """Laplace‑shape a_i = exp(-|i‑mid|/scale)."""
    i = np.arange(n, dtype=np.float64)
    mid = (n - 1) / 2.0
    a = np.exp(-np.abs(i - mid) / scale)
    return a / a.sum()


def _candidate_arcsine(n: int) -> np.ndarray:
    """Arcsine‑law a_i ∝ 1/√((i+½)(n‑i‑½))."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / np.sqrt((i + 0.5) * (n - i - 0.5))
    a = np.clip(a, 0.0, None)
    return a / a.sum()


def _candidate_mix(family_funcs, n, rng):
    """Random convex combination of a few basic families."""
    m = rng.integers(2, 5)               # 2‑4 components
    weights = rng.random(m)
    weights /= weights.sum()
    a = np.zeros(n, dtype=np.float64)
    for w, fn in zip(weights, rng.choice(family_funcs, size=m, replace=False)):
        a += w * fn(rng, n)              # each fn receives rng, n
    return a / a.sum()

# Optimisers ------------------------------------------------------------

def _smoothmax_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    beta_start: float = 0.5,
    beta_factor: float = 1.015,
    eta0: float = 0.07,
) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent (Adam) on the smooth‑max surrogate.
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

    best_a = a0.copy()
    best_val = _objective(a0, n)

    beta = beta_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # obtain probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for tracking
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max weights
        w = np.exp(beta * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w[:] = 1.0 / w.size
        else:
            w /= w_sum

        # gradient of smooth‑max w.r.t. a
        grad = 2.0 * _cross_correlation(w, a, fft_len)

        # chain‑rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = eta0 * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – smooth‑max becomes sharper
        beta = min(beta * beta_factor, 1e9)

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 6_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 250_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a fraction of mass between two positions."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val

# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    # Problem dimension – a compromise between expressivity and speed
    n = 4096
    fft_len = _next_pow2(2 * n - 1)

    # ------------------------------------------------------------------
    # 1) Very broad parametric sweep ----------------------------------

    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep {tag}] new best {val:.6f}")

    # deterministic families (slightly denser grids than before)
    for p in np.linspace(-10.0, 2.0, 60):
        _try_candidate(_candidate_power(p, n), f"pow{p:.2f}")

    for alpha in np.logspace(-8, 2, 55):
        _try_candidate(_candidate_exponential(alpha, n), f"exp{alpha:.2e}")

    alphas = np.logspace(-5, 1, 25)
    betas = np.linspace(0.1, 4.0, 20)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n), f"stre{a_alpha:.2e}_{b_beta:.2f}")

    for _ in range(250):
        _try_candidate(_rand_beta(rng, n), "randbeta")

    for sig in np.logspace(-4.0, -0.2, 30):
        _try_candidate(_candidate_double_gaussian(sig, n), f"dg{sig:.2e}")

    for alpha in np.logspace(-8, 2, 40):
        _try_candidate(_candidate_left_exponential(alpha, n), f"lexp{alpha:.2e}")

    for p in np.linspace(-6.0, 2.0, 40):
        _try_candidate(_candidate_left_power(p, n), f"lpow{p:.2f}")

    for pw in np.linspace(0.2, 6.0, 30):
        _try_candidate(_candidate_sine(pw, n), f"sine{pw:.2f}")

    for k in [16, 32, 64, 128, 256, 512]:
        for _ in range(12):
            _try_candidate(_candidate_spike(k, n, random=True), f"spk{k}")

    # extra families
    for scale in np.logspace(-2.5, 1.5, 25):
        _try_candidate(_candidate_logistic(scale, n), f"log{scale:.2e}")

    for scale in np.logspace(-2.5, 1.5, 25):
        _try_candidate(_candidate_laplace(scale, n), f"lap{scale:.2e}")

    _try_candidate(_candidate_arcsine(n), "arcs")

    # random Dirichlet draws (moderate concentration)
    for conc in np.logspace(-2, 1, 12):
        a = np.random.dirichlet(np.full(n, conc))
        _try_candidate(a, f"dir{conc:.2e}")

    # Warm start from previously stored sequence, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _try_candidate(w, "warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max Adam refinement (majority of remaining budget) ------
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] smooth‑max Adam refinement")
        # Allocate roughly 60 % of the remaining time to this stage
        adam_deadline = min(deadline - 5, time.time() + 0.6 * (deadline - time.time()))
        a_adam, val_adam = _smoothmax_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=2_000_000,
            beta_start=0.5,
            beta_factor=1.018,
            eta0=0.07,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient (exponentiated‑gradient)--------------
    if best_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Subgrad] exact‑max exponentiated‑gradient")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=8_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Final simulated‑annealing fine‑tune ---------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=300_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Post‑process and return ------------------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:  # pragma: no cover
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
