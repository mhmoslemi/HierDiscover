"""
propose_candidate
-----------------
A fast three‑stage optimiser for the step‑function lower bound.

Stage 1 – extremely broad parametric sweep.
          Includes all families from the reference code and several
          new asymmetric families (left‑stretched exponential,
          left logistic, random convex mixtures).  The best few
          candidates are retained.

Stage 2 – long “Adam” optimisation on a smooth‑max surrogate.
          The temperature (λ) of the smooth‑max is increased
          extremely slowly (factor 1.004) so the optimisation
          sees a very smooth landscape for most of the budget.
          No L2‑regularisation is used – we focus only on the
          max‑convolution term.

Stage 3 – exponentiated‑gradient sub‑gradient descent on the
          *exact* max‑norm.  The step‑size starts large (η₀≈1.2)
          and decays as 1/√t.  A tiny amount of simulated annealing
          (random perturbation of the log‑parameters) is mixed in
          to escape shallow plateaux.

The routine always returns a non‑negative list, respects the
time limit (default 1000 s) and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for gradients.
    Returns vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact objective: 2·n·max(conv) / (sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = _i[:n]
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = _i[:n]
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    t = _i[:n] / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:

    t = np.arange(n, dtype=np.float64) / (n - 1)

    a_sum = a.sum()

def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (Dirichlet masses) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a

    if f2 is _candidate_left_exponential:
        p2 = f2(alpha=10 ** np.random.uniform(-4, 1), n=n)
    elif f2 is _candidate_left_power:
        p2 = f2(p=np.random.uniform(-5, 0), n=n)
    elif f2 is _candidate_left_stretched:
        p2 = f2(alpha=10 ** np.random.uniform(-4, 0), beta=np.random.uniform(0.2, 3.5), n=n)
    elif f2 is _candidate_left_logistic:
        mu = np.random.uniform(0, n - 1)

# Adam optimiser (smooth‑max surrogate) --------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lam_start: float = 0.02,
    lam_factor: float = 1.004,
    l2_weight: float = 0.0,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth‑max(conv)  +  λ·‖conv – uniform‖₂² .
    Returns the best vector seen (exact objective).
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for book‑keeping
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Adam it={it:07d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L₂ regularisation (optional) -----
        if l2_weight > 0.0:
            residual = conv - uniform_target
            grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)
            grad = grad_smooth + l2_weight * grad_l2
        else:
            grad = grad_smooth

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) -----------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 6_000_000,
    eta0: float = 1.2,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns best vector and exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror‑descent on simplex)
        a = a * np.exp(-eta * grad)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        # occasional random jitter to escape plateaux
        if it % 500_000 == 0 and it > 0:
            jitter = np.random.normal(scale=0.02, size=n)
            a = np.clip(a + jitter, 0.0, None)
            a /= a.sum()

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple annealing fine‑tuning -----------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Transfer a small fraction of mass between two random indices."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    # temperature schedule (fast exponential decay)
    T_start = 0.15
    T_end = 3e-5

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))
        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.40   # up to 40 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main optimisation routine ---------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns a valid sequence.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # Problem dimension – 4096 gives a good trade‑off between quality
    # and speed (FFT length = 8192 → power‑of‑two).
    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ---------------------------------------
    best_seq = None
    best_val = float("inf")

    def _try_candidate(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ----- systematic families (same as reference) -----
    for p in np.linspace(-5.0, -0.05, 42):
        _try_candidate(_candidate_power(p, n))

    for alpha in np.logspace(-6, 1.2, 48):
        _try_candidate(_candidate_exponential(alpha, n))

    alphas = np.logspace(-4, 0.6, 22)
    betas = np.linspace(0.2, 3.5, 16)
    for a_alpha in alphas:
        for b_beta in betas:
            _try_candidate(_candidate_stretched(a_alpha, b_beta, n))

    for _ in range(150):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _try_candidate(_candidate_beta(a_par, b_par, n))

    for sig in np.logspace(-3.0, -0.1, 27):
        _try_candidate(_candidate_double_gaussian(sig, n))

    for alpha in np.logspace(-6, 1.5, 35):
        _try_candidate(_candidate_left_exponential(alpha, n))

    for p in np.linspace(-5.0, 2.0, 35):
        _try_candidate(_candidate_left_power(p, n))

    for pow_ in np.linspace(0.5, 5.0, 22):
        _try_candidate(_candidate_sine(pow_, n))

    # ----- new asymmetric families -----
    for alpha in np.logspace(-5, 0.5, 25):
        for beta in np.linspace(0.3, 3.0, 14):
            _try_candidate(_candidate_left_stretched(alpha, beta, n))

    for _ in range(120):
        mu = np.random.uniform(0, n - 1)
        sigma = np.random.uniform(0.01 * n, 0.5 * n)
        _try_candidate(_candidate_left_logistic(mu, sigma, n))

    # ----- random spike patterns (different sparsities) -----
    for k in [48, 96, 192, 384, 768]:
        for _ in range(10):
            _try_candidate(_candidate_spike(k, n, random=True))

    # ----- random mixtures of two families -----
    for _ in range(200):
        _try_candidate(_candidate_random_mixture(n))

    # ----- warm start from a globally stored sequence, if any -----
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _try_candidate(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Long Adam refinement (smooth‑max surrogate) -----------------
    if best_seq is not None and time.time() < deadline - 30:
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(0.45 * remaining, remaining - 20.0)
        print("[Phase‑Adam] running smooth‑max Adam optimisation")
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.02,
            lam_factor=1.004,
            l2_weight=0.0,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max exponentiated‑gradient sub‑gradient ----------------
    if best_seq is not None and time.time() < deadline - 10:
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(0.40 * remaining, remaining - 5.0)
        print("[Phase‑Subgrad] exact‑max exponentiated‑gradient")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=8_000_000,
            eta0=1.2,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional annealing fine‑tuning --------------------------------
    if best_seq is not None and time.time() < deadline - 2:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=500_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks & return --------------------------------------
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
