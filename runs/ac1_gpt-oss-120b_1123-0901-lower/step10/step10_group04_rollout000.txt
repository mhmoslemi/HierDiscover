"""
step_function_search.py

Search for a non‑negative step‑function (list of heights) that minimises

    2·n·max(conv(a,a)) / (sum(a)^2)

where conv is the linear convolution of the sequence with itself.
The algorithm combines:

* A broad parametric sweep over many analytic families (power‑law,
  exponential, stretched‑exponential, beta, double‑Gaussian, sine,
  window functions, spikes, …).  Each candidate is normalised to have
  sum = 1 and its exact objective is evaluated with a fast FFT‑based
  convolution.

* An Adam optimiser on a smooth‑max surrogate of the max‑norm (plus a
  tiny L₂ regulariser) that works in the soft‑max (log‑probability)
  parametrisation, guaranteeing the simplex constraints.

* An exponentiated‑gradient sub‑gradient descent on the true max‑norm.

* A cheap simulated‑annealing fine‑tuner.

The search is performed for several problem sizes (n = 4096, 8192,
12288) – larger n gives a finer discretisation of the underlying
continuous shape and typically yields a lower bound.  The routine
honours a user‑supplied time budget (default 1000 s) and always returns a
valid non‑negative list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv) / (sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    """Sine‑shaped bump a_i ∝ sin(π·(i+½)/n)^power."""
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (Dirichlet masses) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_hann(n: int) -> np.ndarray:
    """Hann window (zero at ends, smooth bump)."""
    i = np.arange(n, dtype=np.float64)
    a = 0.5 * (1 - np.cos(2 * np.pi * i / (n - 1)))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_blackman(n: int) -> np.ndarray:
    """Blackman window."""
    i = np.arange(n, dtype=np.float64)
    a = (0.42 - 0.5 * np.cos(2 * np.pi * i / (n - 1))
         + 0.08 * np.cos(4 * np.pi * i / (n - 1)))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_tent(width_frac: float, n: int) -> np.ndarray:
    """
    Symmetric triangular bump.
    width_frac ∈ (0,1] determines the support width relative to n.
    """
    if width_frac <= 0:
        width_frac = 0.001
    m = max(2, int(round(width_frac * n)))
    half = m // 2
    # linear ramp up
    left = np.linspace(0, 1, half, endpoint=False)
    # linear ramp down
    right = np.linspace(1, 0, m - half, endpoint=False)
    shape = np.concatenate([left, right])
    a = np.zeros(n, dtype=np.float64)
    start = (n - m) // 2
    a[start:start + m] = shape
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_cosine_bump(power: float, n: int) -> np.ndarray:
    """Cosine-shaped bump: a_i ∝ (1 - cos(2π·i/(n-1)))^power."""
    i = np.arange(n, dtype=np.float64)
    a = (1 - np.cos(2 * np.pi * i / (n - 1))) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max surrogate) -------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    lam_start: float = 0.08,
    lam_factor: float = 1.008,
    l2_weight: float = 0.13,
) -> tuple[np.ndarray, float]:
    """Adam on a smooth‑max + tiny L₂ regulariser surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to keep a on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # soft‑max weights (smooth max)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # L₂ regulariser gradient (optional, keeps optimisation stable)
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # mirror‑descent (soft‑max) chain rule
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exponentiated‑gradient sub‑gradient optimisation (true max) -------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 12_000_000,
    eta0: float = 0.95,
) -> tuple[np.ndarray, float]:
    """Exponentiated‑gradient sub‑gradient descent on the exact max‑norm."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning ---------------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing moving a small fraction of mass between two bins."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Helper: Upsample an existing sequence to a new size -----------------

def _upsample_sequence(old_seq: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of old_seq (length m) to length new_n."""
    m = old_seq.shape[0]
    if m == new_n:
        return old_seq.copy()
    x_old = np.linspace(0.0, 1.0, m)
    x_new = np.linspace(0.0, 1.0, new_n)
    new = np.interp(x_new, x_old, old_seq)
    new = np.clip(new, 0.0, None)
    s = new.sum()
    if s == 0:
        new = np.full(new_n, 1.0 / new_n)
    else:
        new = new / s
    return new

# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # safety margin

    # ------------------------------------------------------------------
    # Problem sizes to try – larger n gives a finer discretisation.
    # We start with modest sizes and increase if time permits.
    n_candidates = [4096, 8192, 12288]   # 12288 ≈ 3·4096 keeps FFT size manageable

    # Global best across all sizes
    best_seq = None
    best_val = float("inf")
    best_n = None

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep for each n
    for n in n_candidates:
        if time.time() >= deadline - 5:
            break
        print(f"\n=== Sweeping shape families for n = {n} ===")
        conv_len = 2 * n - 1
        fft_len = _next_pow2(conv_len)

        # helper to evaluate & possibly update incumbents
        def _try_candidate(a: np.ndarray, tag: str = "sweep"):
            nonlocal best_seq, best_val, best_n
            val = _objective(a, n)
            if val < best_val:
                best_val = val
                best_seq = a.copy()
                best_n = n
                print(f"[{tag}] new overall best {val:.6f} (n={n})")

        # ---- systematic families -------------------------------------------------
        # power‑law
        for p in np.linspace(-5.0, -0.05, 40):
            _try_candidate(_candidate_power(p, n), "power")
        # exponential
        for alpha in np.logspace(-6, 1.2, 45):
            _try_candidate(_candidate_exponential(alpha, n), "exp")
        # stretched exponential
        alphas = np.logspace(-4, 0.6, 20)
        betas = np.linspace(0.2, 3.5, 15)
        for a_alpha in alphas:
            for b_beta in betas:
                _try_candidate(_candidate_stretched(a_alpha, b_beta, n), "stretch")
        # beta distributions – random draws
        for _ in range(120):
            a_par = 10 ** np.random.uniform(-2.0, 1.5)
            b_par = 10 ** np.random.uniform(-2.0, 1.5)
            _try_candidate(_candidate_beta(a_par, b_par, n), "beta")
        # double Gaussian
        for sig in np.logspace(-3.0, -0.1, 25):
            _try_candidate(_candidate_double_gaussian(sig, n), "dblgauss")
        # left‑exponential (monotone)
        for alpha in np.logspace(-6, 1.5, 30):
            _try_candidate(_candidate_left_exponential(alpha, n), "leftexp")
        # left‑power (monotone)
        for p in np.linspace(-4.0, 1.0, 30):
            _try_candidate(_candidate_left_power(p, n), "leftpow")
        # sine‑shaped bumps
        for pow_ in np.linspace(0.5, 4.5, 20):
            _try_candidate(_candidate_sine(pow_, n), "sine")
        # window functions
        _try_candidate(_candidate_hann(n), "hann")
        _try_candidate(_candidate_blackman(n), "black")
        # triangular/tent shapes
        for w in np.linspace(0.1, 0.9, 17):
            _try_candidate(_candidate_tent(w, n), "tent")
        # cosine bump
        for pw in np.linspace(0.5, 5.0, 15):
            _try_candidate(_candidate_cosine_bump(pw, n), "cosbump")
        # random spike patterns
        for k in [32, 64, 128, 256, 512]:
            for _ in range(6):
                _try_candidate(_candidate_spike(k, n, random=True), "spike")
        # Warm start from previously stored global variable (if any)
        try:
            warm = globals().get("height_sequence_1")
            if warm is not None:
                w_arr = np.asarray(warm, dtype=np.float64)
                up = _upsample_sequence(w_arr, n)
                _try_candidate(up, "warm")
        except Exception:
            pass

        print(f"[Sweep] best after n={n}: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Long Adam refinement (smooth‑max + L₂)
    if best_seq is not None and time.time() < deadline - 25:
        print("[Phase‑Adam] starting Adam refinement")
        n = best_n
        fft_len = _next_pow2(2 * n - 1)
        # allocate at most 35 % of remaining time to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(0.35 * remaining, remaining - 5.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.07,
            lam_factor=1.007,
            l2_weight=0.10,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Exact‑max sub‑gradient optimisation (exponentiated‑gradient)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑Subgrad] starting exact‑max sub‑gradient optimisation")
        n = best_n
        fft_len = _next_pow2(2 * n - 1)
        remaining = deadline - time.time()
        sub_deadline = time.time() + max(0.5 * remaining, remaining - 2.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=15_000_000,
            eta0=0.95,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning (if time permits)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final simulated‑annealing fine‑tune")
        n = best_n
        fft_len = _next_pow2(2 * n - 1)
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity adjustments
    if best_seq is None:
        # fallback: uniform distribution of length 4096
        n = 4096
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # sanity check with the public evaluator (if it exists)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
