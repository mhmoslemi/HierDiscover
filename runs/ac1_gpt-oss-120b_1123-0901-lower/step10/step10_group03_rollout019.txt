def project_simplex(v):
    # Projects onto {x >=0, sum(x)=1}
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w

def _projected_subgradient_optimize(a0, n, conv_fft_len, corr_fft_len, deadline, max_iters=5_000_000, eta0=1.0):

   a = a0.copy()
   a = np.maximum(a,0)
   a /= a.sum()

   best_a, best_val = a.copy(), _objective(a,n)

   it=0
   while time.time()<deadline and it<max_iters:
       conv=_full_convolution(a, conv_fft_len)
       max_c=conv.max()
       # compute subgradient
       # active set: where conv == max_c (within tolerance)
       active = np.isclose(conv, max_c, atol=1e-12)
       # assign uniform weight over active indices
       w=np.zeros_like(conv)
       if active.any():
           w[active] = 1.0/active.sum()
       grad = 2.0 * _cross_correlation(w, a, corr_fft_len)
       # step size schedule
       eta = eta0 / np.sqrt(it+1)
       # gradient descent step
       a = a - eta * grad
       # project onto simplex
       # projection onto simplex ensures sum=1 and nonnegative
       # use algorithm:
       u = np.sort(a)[::-1]
       cssv = np.cumsum(u) - 1
       ind = np.arange(len(u))+1
       rho = ind[u - cssv/ind > 0]
       if rho.size == 0:
           theta = 0
       else:
           rho = rho[-1]
           theta = cssv[rho-1]/rho
       a = np.maximum(a - theta, 0)
       # track best
       val = _objective(a,n)
       if val < best_val:
           best_val = val
           best_a = a.copy()
           # optionally print progress

       it+=1
   return best_a, best_val

def _simplex_projection(v):
    """Project onto probability simplex."""
    # from https://arxiv.org/pdf/1309.1541.pdf
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    # Find rho
    rho = np.where(u - cssv / ind > 0)[0]
    if len(rho) == 0:
        theta = 0
    else:
        rho = rho[-1]
        theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

"""
Optimise a non‑negative step‑function (list of heights) for the
`evaluate_sequence` score.

The algorithm works in four phases within the given time budget:

1. **Broad parametric sweep** – generate many candidate families
   (exponential, power‑law, stretched‑exponential, Gamma, Beta,
   mixtures, random Dirichlet, logistic, etc.).  Each candidate is
   normalised to sum = 1 and evaluated with the exact objective
   `2·n·max(conv)/(∑a)²`.  The best candidate is kept.

2. **Smooth‑max Adam** – starting from the best candidate, run a
   Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate of the
   max‑convolution.  The optimiser works in the soft‑max (simplex)
   parametrisation to keep the iterates feasible.

3. **Projected sub‑gradient descent** – switch to the true max‑convolution
   objective.  At each iteration the sub‑gradient is
   ``grad = 2·cross_correlation(w, a)`` where ``w`` is the uniform weight
   on the active (max‑attaining) convolution indices.  An additive
   update ``a ← a – η·grad`` is performed and the vector is projected back
   onto the probability simplex (clip to ≥0 and renormalise).  The step
   size decays as ``η₀/√t``.

4. **Optional annealing** – a cheap simulated‑annealing finetuner moves a
   random fraction of mass between two coordinates, accepting
   downhill moves and occasionally uphill moves.

All phases respect the global deadline (default 1000 s).  The routine
always returns a valid list of non‑negative numbers; the final score is
printed for diagnostics.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation needed for the (sub‑)gradient:
    (w ★ a)[i] = Σ_k w[k]·a[i+k‑(n‑1)],  length = len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)²   (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power law a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential a_i ∝ exp(-α·i)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β)+exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_logistic(mu: float, tau: float, n: int) -> np.ndarray:
    """Logistic (sigmoid) decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp((i - mu) / tau))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_mixture_exponential(alpha1: float, alpha2: float,
                                   w1: float, n: int) -> np.ndarray:
    """Mixture w1·exp(-α1·i)+(1‑w1)·exp(-α2·i)."""
    i = np.arange(n, dtype=np.float64)
    a = w1 * np.exp(-alpha1 * i) + (1.0 - w1) * np.exp(-alpha2 * i)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -------------------------------------------------------------

def _adam_optimize(a0: np.ndarray, n: int,
                   conv_fft_len: int, corr_fft_len: int,
                   deadline: float,
                   max_iters: int = 2_000_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate.
    Returns the best feasible vector and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max params
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.03
    lam = 0.1           # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- simplex via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective (tracking)
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # mirror‑descent form (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (sharpen)
        lam = min(lam * lam_factor, 1e7)

        # gentle learning‑rate decay
        if (it + 1) % 200_000 == 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _projected_subgrad_optimize(a0: np.ndarray, n: int,
                               conv_fft_len: int, corr_fft_len: int,
                               deadline: float,
                               max_iters: int = 5_000_000,
                               eta0: float = 1.5) -> tuple[np.ndarray, float]:
    """
    Sub‑gradient descent on the true max‑convolution objective.
    Uses additive updates followed by a cheap projection onto the simplex
    (clip negatives and renormalise).  Returns the best vector seen.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 100_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a - eta * grad
        # simple simplex projection: clip & renormalise
        a = np.maximum(a, 0.0)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(a0: np.ndarray, n: int,
                    conv_fft_len: int,
                    deadline: float,
                    max_moves: int = 200_000) -> tuple[np.ndarray, float]:
    """
    Very cheap simulated‑annealing finetuner.  Randomly moves a fraction
    of mass between two coordinates; accepts downhill moves and accepts
    uphill moves with a decreasing temperature.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.2
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-15:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low `evaluate_sequence`
    score.  The routine respects the time budget and always returns a
    valid list (the best sequence found so far).
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5      # safety margin

    # problem size – 4096 works well and was used by the reference code
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep ----------------------------------------
    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # ---- families ----------------------------------------------------
    # left exponential – dense log‑grid
    for alpha in np.logspace(-5, 0.5, 2500):
        _maybe_update(_candidate_exponential(alpha, n))
        if time.time() >= deadline:
            break

    # left power law – dense linear grid
    for p in np.linspace(-6.0, 2.0, 2500):
        _maybe_update(_candidate_power(p, n))
        if time.time() >= deadline:
            break

    # left stretched exponential – random sampling
    for _ in range(2500):
        alpha = 10 ** np.random.uniform(-5, 1)
        beta = np.random.uniform(0.2, 4.0)
        _maybe_update(_candidate_left_stretched(alpha, beta, n))
        if time.time() >= deadline:
            break

    # Gamma‑like family – grid
    for shape in np.linspace(0.2, 4.0, 30):
        for scale in np.logspace(-4, 1, 30):
            _maybe_update(_candidate_gamma(shape, scale, n))
            if time.time() >= deadline:
                break

    # Beta distribution – grid
    for a_par in np.logspace(-2, 0.5, 20):
        for b_par in np.logspace(0, 1.0, 20):
            _maybe_update(_candidate_beta(a_par, b_par, n))
            if time.time() >= deadline:
                break

    # Double Gaussian – log‑grid on sigma
    for sig in np.logspace(-3, -0.2, 30):
        _maybe_update(_candidate_double_gaussian(sig, n))
        if time.time() >= deadline:
            break

    # Logistic decreasing – random parameters
    for _ in range(1500):
        mu = np.random.uniform(0, n * 0.9)
        tau = 10 ** np.random.uniform(-3, 0.5)
        _maybe_update(_candidate_logistic(mu, tau, n))
        if time.time() >= deadline:
            break

    # Random mixtures of two exponentials
    for _ in range(3000):
        alpha1 = 10 ** np.random.uniform(-5, 1)
        alpha2 = 10 ** np.random.uniform(-5, 1)
        w1 = np.random.rand()
        _maybe_update(_candidate_mixture_exponential(alpha1, alpha2, w1, n))
        if time.time() >= deadline:
            break

    # Random Dirichlet draws – varied concentration
    for conc in [0.01, 0.05, 0.1, 0.3, 0.5, 1.0, 2.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_update(a)
        if time.time() >= deadline:
            break

    # Spike patterns – various numbers of spikes
    for k in [8, 16, 32, 64, 128, 256, 512]:
        for _ in range(2):
            _maybe_update(_candidate_spike(k, n, random=True))
            if time.time() >= deadline:

    # Warm‑start from a previous global variable, if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:

        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Smooth‑max Adam refinement ------------------------------------
    if best_seq is not None and time.time() < deadline - 20:
        remaining = deadline - time.time()
        # give Adam up to 30 % of the remaining budget (but at most 45 s)
        adam_deadline = time.time() + min(45.0, 0.30 * remaining)

        a_adam, val_adam = _adam_optimize(

            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

        else:

        a_sub, val_sub = _projected_subgrad_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_deadline
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

        # break if we have less than a few seconds left
        if deadline - time.time() < 6.0:
            break

    # ------------------------------------------------------------------
    # 4) Optional annealing fine‑tuner ----------------------------------
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq, n, conv_fft_len, deadline, max_moves=250_000
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks & return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
