"""
Hybrid optimizer with composite smooth‑max and variance regularisation.

The optimizer runs the following pipeline under the 1000 s budget:

1. Initialise a probability vector a (warm start if available, otherwise a
   Dirichlet sample).  The vector lives on the simplex (sum = 1,
   a_i ≥ 0).

2. Run a custom Adam optimiser on a smooth approximation of the objective
   `2·n·max(a*a)`.  The smooth‑max is combined with a variance‑penalty that
   pushes the whole convolution towards a constant value.  The resulting
   iterate usually reaches a lower maximum than the plain Adam used in the
   reference code.

3. Apply a long “spectral‑flattening’’ phase: the FFT magnitude of the
   vector is forced to be (almost) constant while projecting back onto the
   simplex.  This reduces the peak of the autocorrelation further.

4. Finish with a projected gradient‑descent pass on the exact maximum
   convolution and a brief simulated‑annealing local search.

The routine always returns the best sequence found before the deadline.
"""

import time
import numpy as np

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()

def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]

def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]

def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_composite_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a composite surrogate:
      smooth‑max(conv)  +  beta * variance(conv).

    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)  # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06
    lam = 0.2                # start temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to retrieve the current probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # full convolution
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Composite Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # variance part (conv - mean)
        avg = conv.mean()
        var_err = conv - avg
        grad_var = 2.0 * _cross_correlation(var_err, a, corr_fft_len)

        # annealing weight for variance term
        beta = beta_start * max(0.0, 1.0 - it / max_iters)

        grad_a = grad_max + beta * grad_var

        # chain rule through the soft‑max
        avg_g = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg_g)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # increase temperature
        lam = min(lam * lam_factor, 1e7)

        # optional learning‑rate decay
        if it % 80_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val

def _spectral_flatten_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 250_000,
) -> tuple[np.ndarray, float]:
    """
    Iteratively force the FFT magnitude to be (almost) constant while
    staying on the simplex.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Spectral it={it:06d}] new best {val:.6f}")

        # FFT of current vector
        A = np.fft.fft(a)
        energy = np.dot(a, a)          # ||a||_2^2
        # redistribute energy uniformly over non‑DC frequencies
        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                # keep DC unchanged (sum = 1)
        if n > 1:
            # keep phase, replace magnitude
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        # project onto simplex (non‑negative, sum = 1)
        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val

def _max_conv_descent(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,

) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution.
    """

        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        a = np.maximum(a, 0.0)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val

"""
Hybrid optimiser with composite smooth‑max and variance regularisation.

The algorithm works under a 1000 s budget and proceeds in four phases:

1. Composite Adam optimisation – minimises a smooth‑max of the convolution
   plus a variance penalty that flattens the whole autocorrelation.
2. Spectral flattening – forces the FFT magnitude to be (almost) constant
   while staying on the probability simplex.
3. Projected gradient descent on the exact max‑convolution.
4. Simulated‑annealing refinement.

Each phase keeps the best vector encountered; the routine always returns a
valid non‑negative sequence before the deadline.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)

def _adam_composite_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 300_000,
    beta_start: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on a composite surrogate:
      smooth‑max(conv) + beta * variance(conv).

    Returns the best vector (on the simplex) and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)                     # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06                                     # initial learning rate
    lam = 0.2                                     # temperature for smooth‑max
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # ------------------- retrieve probability vector -------------------
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ------------------- evaluate exact objective ---------------------
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Composite Adam it={it:06d}] new best {val:.6f}")

        # ------------------- smooth‑max weights -------------------------
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # ------------------- gradients -----------------------------------
        grad_max = 2.0 * _cross_correlation(w, a, corr_fft_len)

        avg = conv.mean()
        var_err = conv - avg
        grad_var = 2.0 * _cross_correlation(var_err, a, corr_fft_len)

        beta = beta_start * max(0.0, 1.0 - it / max_iters)
        grad_a = grad_max + beta * grad_var

        # ------------------- chain rule through soft‑max -----------------
        avg_g = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg_g)

        # ------------------- Adam update ---------------------------------
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ------------------- temperature / learning‑rate schedule -------
        lam = min(lam * lam_factor, 1e7)
        if it % 80_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val
