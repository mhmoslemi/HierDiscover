    sequence = [max(0, x) for x in sequence]
    sequence = [min(1000.0, x) for x in sequence]

def simplex_projection(v):
    # v shape (n,)
    # Project onto simplex {x >=0, sum x = 1}
    # Implementation from literature:
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

"""
Hybrid optimizer for the max‑convolution objective.

The routine builds a diverse pool of warm‑starts (including a supplied
height_sequence_1), picks the best one and then refines it with two
phases:

1.  Simple exponentiated‑gradient descent (sub‑gradient of the max‑conv).
    A decreasing step‑size and a lightweight projection onto the simplex
    keep the iterates feasible.  This quickly drives the objective down.

2.  A long simulated‑annealing polish.  At each move a random fraction of
    mass is shifted from one coordinate to another; downhill moves are
    always accepted, uphill moves with a temperature schedule.  The move
    count is limited only by the remaining time budget.

Both phases use FFT‑based convolution / cross‑correlation, so each iteration
is O(n log n) with a very small constant.  The implementation respects the
global 1000 s budget and returns the best sequence found.
"""

import time
import numpy as np

# ---------------------------------------------------------------
# Helper functions (all top‑level, no closures)

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for sub‑gradient."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _objective(a: np.ndarray, n: int, fft_len: int) -> float:
    """
    Exact objective 2·n·max(a*a) assuming sum(a)=1.
    a must be non‑negative and sum to 1 (otherwise the value is scaled
    artificially – the evaluator is scale‑invariant).
    """
    conv = _full_convolution(a, fft_len)
    return 2.0 * n * conv.max()


def _project_onto_simplex(v: np.ndarray) -> np.ndarray:
    """
    Projects the vector v onto the probability simplex {x ≥ 0, Σ x = 1}.
    Implementation of the algorithm from:
    Wang & Carreira‑Perpiñán, “Projection onto the probability simplex:
    An efficient algorithm with a simple proof, and an application”.
    """
    # work on a copy to avoid side effects
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    # condition: u_i - cssv_i / i > 0
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are zero – return uniform
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0.0)
    return w


# ---------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (≈ 1000 s) and always returns
    the best sequence (as a Python list) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – 4096 works well for the time budget
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    best_seq = None          # type: np.ndarray | None
    best_val = float("inf")

    # -----------------------------------------------------------
    # 1️⃣  Build a diverse pool of starter vectors

    pool = []

    # warm start (if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64).copy()
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, 1000.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            pool.append(w)

    # uniform baseline
    pool.append(np.full(n, 1.0 / n, dtype=np.float64))

    # parametric families -------------------------------------------------
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)

    # power‑law ends‑heavy (α < 0).  A fine grid around α≈‑0.5 (arcsine shape)
    alphas = np.concatenate((
        np.linspace(-3.0, -0.2, 15),
        np.linspace(-0.8, -0.1, 15)
    ))
    for alpha in alphas:
        shape = prod ** alpha
        if shape.sum() > 0.0:
            shape = shape / shape.sum()
            pool.append(shape)

    # Beta‑shapes a_i ∝ x^p (1‑x)^q
    x = np.linspace(0.0, 1.0, n, dtype=np.float64)
    for p in (0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
        for q in (0.2, 0.5, 1.0, 2.0, 5.0, 10.0):
            shape = (x ** p) * ((1.0 - x) ** q)
            if shape.sum() > 0.0:
                shape = shape / shape.sum()
                pool.append(shape)

    # Dirichlet draws with several concentration parameters
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        for _ in range(3):
            pool.append(np.random.dirichlet(alpha * np.ones(n)))

    # Piece‑wise constant blocks (random block masses)
    for blocks in (8, 16, 32, 64):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        a = a / a.sum()
        pool.append(a)

    # Symmetric spikes – equally spaced tiny masses
    for k in (64, 128, 256):
        a = np.zeros(n, dtype=np.float64)
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
        pool.append(a)

    # Random sparse spikes
    for k in (64, 128, 256):
        for _ in range(2):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            a = a / a.sum()
            pool.append(a)

    # -----------------------------------------------------------
    # 2️⃣  Quick evaluation of all starters – keep the best one

    for i, cand in enumerate(pool):
        a = np.maximum(cand, 0.0)
        s = a.sum()
        if s < 0.01:
            continue
        a = a / s
        val = _objective(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Init #{i:03d}] new best {val:.6f}")

    # Fallback – should never happen
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n, conv_fft_len)
        print("[Fallback] using uniform distribution")

    # -----------------------------------------------------------
    # 3️⃣  Exponentiated‑gradient descent (sub‑gradient of max‑conv)

    a = best_seq.copy()
    it = 0
    step_const = 1.0            # initial step‑size constant
    while time.time() < deadline - 8:
        # --- compute convolution and current objective
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        cur_val = 2.0 * n * max_c
        if cur_val < best_val:
            best_val = cur_val
            best_seq = a.copy()
            print(f"[EG it={it:06d}] new best {cur_val:.6f}")

        # --- sub‑gradient (average over active indices)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if not active.any():
            break
        w = np.zeros_like(conv)
        w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # --- multiplicative (exponentiated) update + simplex projection
        step = step_const / np.sqrt(it + 1.0)
        a = a * np.exp(-step * grad)          # guarantee non‑negativity
        a = _project_onto_simplex(a)          # keep sum = 1
        it += 1
        # very occasional back‑off if the update made the max larger too much
        if it % 10000 == 0 and cur_val > best_val * 1.0005:
            step_const *= 0.8

    # -----------------------------------------------------------
    # 4️⃣  Simulated‑annealing polish (many local moves)

    a = best_seq.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val_sa = val
    best_a_sa = a.copy()

    # temperature schedule (exponential decay)
    T_start = 0.1
    T_end = 5e-6

    max_moves = 300_000
    for mv in range(max_moves):
        if time.time() > deadline - 1.0:
            break
        # linear cooling
        T = max(T_end, T_start * (1.0 - mv / max_moves))

        # pick source and destination indices
        i = np.random.randint(n)
        if a[i] < 1e-14:
            continue
        j = np.random.randint(n)
        if i == j:
            continue

        # move a random fraction of the source mass to the destination
        delta = a[i] * (np.random.rand() * 0.4)   # at most 40 % of the source
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        # evaluate the candidate
        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        # accept downhill always; uphill with Metropolis probability
        if val2 < best_val_sa:
            best_val_sa = val2
            best_a_sa = a2.copy()
            a = a2
            print(f"[SA move {mv:06d}] improved to {val2:.6f}")
        elif np.random.rand() < np.exp(-(val2 - val) / max(T, 1e-12)):
            a = a2

        # keep current value for the next Metropolis test
        val = val2 if val2 < val else val

    if best_val_sa < best_val:
        best_val = best_val_sa
        best_seq = best_a_sa.copy()
        print(f"[SA final] best {best_val:.6f}")

    # -----------------------------------------------------------
    # 5️⃣  Final clean‑up and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to evaluator limits (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
