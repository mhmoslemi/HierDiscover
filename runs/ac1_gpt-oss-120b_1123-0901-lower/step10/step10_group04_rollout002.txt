"""
Propose a low‑score non‑negative step function.

Strategy
--------
1. **Coarse parametric sweep on a modest grid (n≈2048).**
   A rich family of analytic windows (power‑law, exponential,
   stretched‑exponential, Beta, double‑Gaussian, monotone,
   sine‑bump, tent, parabolic, Hann/Hamming/Blackman and random
   spike patterns) is sampled.  Each candidate is normalised to
   L₁‑norm = 1 and evaluated with an exact FFT convolution.
   The best 5 “small” candidates are kept.

2. **Upsample to a large dimension (n≈16384).**
   The continuous shape is preserved by linear interpolation and
   re‑normalised.  The objective is re‑evaluated on the large grid
   and the best up‑sampled vector is selected as a seed.

3. **Long Adam refinement on the smooth‑max surrogate.**
   The surrogate is  `smooth_max(conv) + λ·‖conv‑uniform‖₂²`,
   smoothed via a temperature parameter that is slowly increased.
   Adam is run (mirrored‑gradient via a soft‑max parametrisation)
   until the allocated time slice ends.

4. **Exact‑max exponentiated‑gradient sub‑gradient descent.**
   Using the true `max(conv)` the active set is formed and an
   exponentiated‑gradient update (mirror‑descent on the simplex) is
   performed with a 1/√t step‑size schedule.

5. **(Optional) Light annealing** if a few seconds remain.

All phases respect the global time budget (`budget_s` seconds) and
always return a valid non‑negative list.  The routine works with a
single CPU core but can be trivially parallelised if desired.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))

# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a

def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a

    return a


def _candidate_parabolic(n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    x = (i - (n - 1) / 2) / ((n - 1) / 2)
    a = 1 - x ** 2
    a = np.clip(a, 0.0, None)
    a /= a.sum()
    return a


def _candidate_dirichlet(alpha: float, n: int) -> np.ndarray:
    """Random Dirichlet with concentration alpha."""
    a = np.random.dirichlet(np.full(n, alpha))
    return a


# ----------------------------------------------------------------------
# Upsampling ------------------------------------------------------------

def _upsample(a_small: np.ndarray, target_n: int) -> np.ndarray:
    """Linear interpolation from size |a_small| to target_n, re‑normalised."""
    small_n = a_small.shape[0]
    xs = np.linspace(0.0, 1.0, small_n)
    xt = np.linspace(0.0, 1.0, target_n)
    a_big = np.interp(xt, xs, a_small)
    a_big = np.clip(a_big, 0.0, None)
    s = a_big.sum()
    if s == 0:
        a_big = np.full(target_n, 1.0 / target_n)
    else:
        a_big /= s
    return a_big


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max + L2 regularisation) -----------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lam_start: float = 0.1,
    lam_factor: float = 1.01,
    l2_weight: float = 0.12,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth‑max(conv) + λ·‖conv‑uniform‖₂² .
    Returns the best vector encountered (on the simplex) and its exact
    objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    # uniform target for the L2 term
    uniform_target = np.full(2 * n - 1,
                             1.0 / (2 * n - 1),
                             dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution (used for tracking only)
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights -----
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max part
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L₂ part -----
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        # combined gradient w.r.t. a
        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule – smooth‑max becomes sharper
        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated‑gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # renormalise (guard against numerical drift)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuning (optional) ------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Very light annealing that moves a small fraction of mass between two indices."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    # tiny safety margin so the caller can still read the result
    deadline = start_time + budget_s - 0.5

    # ------------------------------------------------------------------
    # Phase 0 – dimensions
    #   - Small grid for cheap exploration
    #   - Large grid for final optimisation
    # ------------------------------------------------------------------
    n_small = 2048                   # fast sweep size
    n_large = 16384                  # final resolution
    fft_len_large = _next_pow2(2 * n_large - 1)

    # ------------------------------------------------------------------
    # Phase 1 – coarse parametric sweep on the small grid
    # ------------------------------------------------------------------
    best_small_val = float("inf")
    best_small_seq = None
    top_k = []          # (value, vector) – keep a few good seeds
    KMAX = 5

    def _keep_candidate(a_small: np.ndarray):
        nonlocal best_small_val, best_small_seq, top_k
        val = _objective(a_small, n_small)
        if val < best_small_val:
            best_small_val = val
            best_small_seq = a_small.copy()
            print(f"[Sweep] new best (small) {val:.6f}")
        # keep a shortlist
        if len(top_k) < KMAX:
            top_k.append((val, a_small.copy()))
        else:
            worst_idx = max(range(len(top_k)), key=lambda i: top_k[i][0])
            if val < top_k[worst_idx][0]:
                top_k[worst_idx] = (val, a_small.copy())

    # --- systematic families ------------------------------------------------
    # power‑law
    for p in np.linspace(-5.0, 2.0, 30):
        _keep_candidate(_candidate_power(p, n_small))

    # exponential
    for alpha in np.logspace(-6, 1.1, 30):
        _keep_candidate(_candidate_exponential(alpha, n_small))

    # stretched exponential
    alphas = np.logspace(-4, 0.5, 10)
    betas = np.linspace(0.2, 3.5, 10)
    for a_alpha in alphas:
        for b_beta in betas:
            _keep_candidate(_candidate_stretched(a_alpha, b_beta, n_small))

    # beta distribution (random draws)
    for _ in range(80):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _keep_candidate(_candidate_beta(a_par, b_par, n_small))

    # double Gaussian
    for sig in np.logspace(-3.0, -0.1, 20):
        _keep_candidate(_candidate_double_gaussian(sig, n_small))

    # left‑exponential (monotone)
    for alpha in np.logspace(-6, 1.4, 25):
        _keep_candidate(_candidate_left_exponential(alpha, n_small))

    # left‑power (monotone)
    for p in np.linspace(-4.0, 1.0, 25):
        _keep_candidate(_candidate_left_power(p, n_small))

    # sine‑shaped bump
    for pow_ in np.linspace(0.5, 5.0, 15):
        _keep_candidate(_candidate_sine(pow_, n_small))

    # classic windows
    _keep_candidate(_candidate_hann(n_small))
    _keep_candidate(_candidate_hamming(n_small))
    _keep_candidate(_candidate_blackman(n_small))
    _keep_candidate(_candidate_tent(n_small))
    _keep_candidate(_candidate_parabolic(n_small))

    # random spikes (different sparsities)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _keep_candidate(_candidate_spike(k, n_small, random=True))

    # random Dirichlet vectors (different concentrations)
    for alpha in [0.1, 0.3, 0.6, 1.0, 3.0]:
        for _ in range(3):
            _keep_candidate(_candidate_dirichlet(alpha, n_small))

    # warm start from a previously stored sequence, if any
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n_small:
            w = w[:n_small]
        elif w.shape[0] < n_small:
            w = np.pad(w, (0, n_small - w.shape[0]))
        w = np.clip(w, 0.0, None)
        if w.sum() >= 0.01:
            w /= w.sum()
            _keep_candidate(w)

    print(f"[Sweep] best after exploration (small) = {best_small_val:.6f}")

    # ------------------------------------------------------------------
    # Phase 2 – upsample shortlisted candidates to the large grid
    # ------------------------------------------------------------------
    best_large_seq = None
    best_large_val = float("inf")
    for val_small, seq_small in top_k:
        seq_large = _upsample(seq_small, n_large)
        val_large = _objective(seq_large, n_large)
        if val_large < best_large_val:
            best_large_val = val_large
            best_large_seq = seq_large.copy()
            print(f"[Upsample] new best {val_large:.6f}")

    # ------------------------------------------------------------------
    # Phase 3 – Adam refinement on the large grid
    # ------------------------------------------------------------------
    if best_large_seq is not None and time.time() < deadline - 30:
        # allocate roughly half the remaining budget to Adam
        remaining = deadline - time.time()
        adam_deadline = time.time() + min(remaining * 0.55, 400.0)  # cap at ~400 s
        print("[Phase‑Adam] refining with Adam on large grid")
        a_adam, val_adam = _adam_optimize(
            best_large_seq,
            n_large,
            fft_len_large,
            adam_deadline,
            max_iters=700_000,
            lam_start=0.09,
            lam_factor=1.009,
            l2_weight=0.13,
        )
        if val_adam < best_large_val:
            best_large_val = val_adam
            best_large_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # Phase 4 – Exact‑max sub‑gradient optimisation
    # ------------------------------------------------------------------
    if best_large_seq is not None and time.time() < deadline - 15:
        remaining = deadline - time.time()
        sub_deadline = time.time() + min(remaining * 0.70, 350.0)  # cap at ~350 s
        print("[Phase‑Subgrad] exact‑max exponentiated‑gradient")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_large_seq,
            n_large,
            fft_len_large,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.95,
        )
        if val_sub < best_large_val:
            best_large_val = val_sub
            best_large_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase 5 – optional final annealing (if time permits)
    # ------------------------------------------------------------------
    if best_large_seq is not None and time.time() < deadline - 5:
        print("[Phase‑Anneal] final annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_large_seq,
            n_large,
            fft_len_large,
            deadline,
            max_moves=250_000,
        )
        if val_ann < best_large_val:
            best_large_val = val_ann
            best_large_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity checks and return
    # ------------------------------------------------------------------
    if best_large_seq is None:
        best_large_seq = np.full(n_large, 1.0 / n_large, dtype=np.float64)

    # Guarantee non‑negativity and reasonable magnitude (evaluate_sequence will clip)
    best_large_seq = np.clip(best_large_seq, 0.0, 1000.0)

    # Verify with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_large_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_large_seq.tolist()
