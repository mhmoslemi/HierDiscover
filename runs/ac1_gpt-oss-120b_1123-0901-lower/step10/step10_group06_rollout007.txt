def _adam_continuation_optimize(a0, n, conv_len, corr_len, deadline, max_iters=500_000, lambda_start=0.01, lambda_factor=1.5, lambda_max=1e6):

    logits = np.log(a0 + eps)
    m,v = zeros
    it=0
    lam=lambda_start
    best_a = a0
    best_val = _objective(a0, n)
    while time < deadline and it < max_iters:
        # compute a from logits (softmax)
        a = np.exp(logits - np.max(logits))
        a /= a.sum()
        # enforce symmetry projection
        a = (a + a[::-1]) / 2.0
        # compute conv and max
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum()**2)
        if val < best_val:
            best_val = val; best_a = a.copy()
        # smooth max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0:
            w /= w_sum
        else:
            w[:] = 1.0 / w.size

        grad = 2.0 * _cross_correlation(w, a, corr_len)

        # mirror descent gradient (softmax)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        beta1,beta2 = 0.9,0.999
        m = beta1*m + (1-beta1)*grad_log
        v = beta2*v + (1-beta2)*(grad_log**2)
        m_hat = m/(1-beta1**(it+1))
        v_hat = v/(1-beta2**(it+1))
        lr = 0.05 # may schedule based on it
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        logits -= step

        # ramp up lambda
        lam = min(lam * lambda_factor, lambda_max)

        # occasionally decay lr
        if it % 100000 == 0 and it>0:
            lr *= 0.8

        it += 1

    return best_a, best_val

"""
Search for a low‑evaluation step‑function.

The algorithm consists of four phases:

1.  **Broad parametric sweep** – generate many symmetric / monotone shapes
    (power‑law, exponential, stretched‑exponential, beta, double‑Gaussian,
    random spikes, very flat/very spiky Dirichlet draws, …).  Each shape is
    normalised to unit ℓ₁‑norm and its exact objective
    ``2·n·max(conv)/sum²`` is evaluated with an FFT‑based convolution.
    The best few candidates are kept (elite list).

2.  **Adam with smooth‑max continuation** – starting from each elite candidate
    we run a soft‑max (log‑sum‑exp) surrogate of the max‑convolution.
    The temperature λ is increased geometrically (continuation) so the
    optimisation first smoothes the objective, then sharpens it.
    The optimisation is performed in the log‑softmax (mirror‑descent) space
    and a symmetry projection ``a ← (a + a[::-1]) / 2`` is applied each
    iteration because the optimal solution is expected to be symmetric.

3.  **Long exact‑max sub‑gradient descent (exponentiated‑gradient)** – the best
    candidate from phase 2 is refined with the true max‑convolution
    sub‑gradient.  The update ``a ← a·exp(−η·g)`` stays on the simplex;
    the step‑size follows the classic ηₜ = η₀/√(t+1) schedule.
    A symmetry projection is also applied.

4.  **Optional simulated‑annealing fine‑tuning** – a cheap mass‑transfer
    neighbourhood move is performed while the temperature is slowly lowered.
    Any improvement is kept.

All phases respect the overall 1000 s budget and always return a valid
non‑negative list.  The final sequence is clipped to ``[0, 1000]`` before
calling ``evaluate_sequence`` for a sanity check.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities (shared by the whole search)
# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) via FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for sub‑gradient (same length as a).
    Implements   grad_i = 2 · Σ_k w_k · a_{k‑i}.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective (the same as ``evaluate_sequence`` but without clipping)
# ----------------------------------------------------------------------
def _objective(a: np.ndarray, n: int) -> float:
    """Return 2·n·max(a*a) / (sum a)²."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators (all produce a non‑negative, unit‑sum vector)
# ----------------------------------------------------------------------
def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta density on {0,…,n‑1} (continuous analogue, then discretised)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = (
            np.random.choice(n, size=k, replace=False)
            if random
            else np.round(np.linspace(0, n - 1, k)).astype(int)
        )
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_dirichlet(conc: float, n: int) -> np.ndarray:
    a = np.random.dirichlet(np.full(n, conc))
    a /= a.sum()
    return a

# Optimisers
# ----------------------------------------------------------------------
def _adam_continuation_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 500_000,
    lam_start: float = 0.05,
    lam_factor: float = 1.05,
    lam_max: float = 1e6,
) -> tuple[np.ndarray, float]:
    """
    Adam on the smooth‑max (log‑sum‑exp) surrogate.
    Temperature λ is increased geometrically (continuation) until ``lam_max``.
    Symmetry projection is applied each iteration.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.06

    lam = lam_start

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to stay on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # enforce symmetry (a_i = a_{n‑1‑i})
        a = (a + a[::-1]) * 0.5

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()

        # smooth‑max (log‑sum‑exp) weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum > 0.0 and np.isfinite(w_sum):
            w /= w_sum
        else:
            w[:] = 1.0 / w.size

        # gradient of the surrogate (mirrored into log‑space)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update on logits
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))
        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule (become sharper)
        lam = min(lam * lam_factor, lam_max)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.85

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.85,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient descent with exponentiated‑gradient updates."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()

        # active set (indices attaining the max, within a tiny tolerance)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # enforce symmetry (helps convergence)
        a = (a + a[::-1]) * 0.5

        # renormalise
        s = a.sum()
        if s < 1e-12 or not np.isfinite(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Simple simulated‑annealing: redistribute a random fraction of mass."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.28
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Elite handling
# ----------------------------------------------------------------------
def _update_elite(elite, val, a, max_size=10):
    """Insert (val, a) into elite list (kept sorted by val)."""
    if len(elite) < max_size:
        elite.append((val, a.copy()))
        elite.sort(key=lambda x: x[0])
    else:
        if val < elite[-1][0]:
            elite[-1] = (val, a.copy())
            elite.sort(key=lambda x: x[0])


# ----------------------------------------------------------------------
def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always
    returns the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # tiny safety margin

    # ------------------------------------------------------------------
    # Problem dimension – keep a power of two for fast FFT
    # ------------------------------------------------------------------
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # 1) Wide parametric sweep – keep a handful of elites
    # ------------------------------------------------------------------
    elite = []  # list of (objective, vector)

    def _maybe_update(a):
        val = _objective(a, n)
        if elite and val >= elite[-1][0]:
            return
        _update_elite(elite, val, a)

    # Power‑law (both ends heavy) – extend the exponent range
    for p in np.linspace(-10.0, -0.01, 40):
        a = _candidate_power(p, n)
        _maybe_update(a)

    # Exponential decay (symmetric)
    for alpha in np.logspace(-6, 2.0, 40):
        a = _candidate_exponential(alpha, n)
        _maybe_update(a)

    # Stretched exponential (α,β)
    alphas = np.logspace(-5, 0.7, 15)
    betas = np.linspace(0.2, 4.0, 15)
    for alpha in alphas:
        for beta in betas:
            a = _candidate_stretched(alpha, beta, n)
            _maybe_update(a)

    # Beta distributions – explore a much wider grid
    a_pars = np.concatenate(
        [np.logspace(-3, -0.2, 10), np.linspace(0.2, 5.0, 12), np.logspace(0.7, 2.0, 8)]
    )
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            a = _candidate_beta(a_par, b_par, n)
            _maybe_update(a)

    # Double Gaussian (mirrored)
    for sig in np.logspace(-4, -0.05, 25):
        a = _candidate_double_gaussian(sig, n)
        _maybe_update(a)

    # Random spike patterns (different sparsities)
    for k in [16, 32, 64, 128, 256, 512]:
        for _ in range(3):
            a = _candidate_spike(k, n, random=True)
            _maybe_update(a)

    # Dirichlet draws with many concentration levels
    for conc in [0.01, 0.05, 0.1, 0.2, 0.5, 1.0, 2.0, 5.0]:
        a = _candidate_dirichlet(conc, n)
        _maybe_update(a)

    # Warm start from a previously stored sequence (if any)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w /= w.sum()
            _maybe_update(w)

    if elite:
        best_val, best_seq = elite[0]
        print(f"[Sweep] best after exploration: {best_val:.6f}")
    else:
        # fallback – uniform distribution
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)
        best_val = _objective(best_seq, n)
        print("[Sweep] no candidates found, using uniform")

    # ------------------------------------------------------------------
    # 2) Adam continuation (smooth‑max) – allocate ~30 % of remaining budget
    # ------------------------------------------------------------------
    remaining = deadline - time.time()
    if remaining > 30.0:
        adam_deadline = time.time() + min(30.0, 0.3 * remaining)
        print("[Phase‑Adam] continuation Adam optimisation")
        # Run on the top 3 elite starts
        for idx, (_, a_start) in enumerate(elite[:3]):
            if time.time() >= adam_deadline:
                break
            a_adam, val_adam = _adam_continuation_optimize(
                a_start,
                n,
                conv_fft_len,
                corr_fft_len,
                adam_deadline,
                max_iters=400_000,
                lam_start=0.02,
                lam_factor=1.07,
                lam_max=5e5,
            )
            if val_adam < best_val:
                best_val = val_adam
                best_seq = a_adam.copy()
                print(f"[Phase‑Adam] start {idx} improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 3) Long exact‑max sub‑gradient descent (exponentiated gradient)
    # ------------------------------------------------------------------
    if time.time() < deadline - 5.0:
        print("[Phase‑Subgrad] long exact‑max sub‑gradient optimisation")
        sub_deadline = deadline - 4.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning (if any time left)
    # ------------------------------------------------------------------
    if time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Final sanity handling
    # ------------------------------------------------------------------
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final public check (if the evaluator is present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:  # pragma: no cover
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
