def _candidate_random_phase(n):
    # Uniform magnitude across frequencies
    M = 1.0 / np.sqrt(2 * n - 1)   # sqrt of C
    # For rfft length
    spec_len = n // 2 + 1
    phases = np.random.uniform(0, 2*np.pi, spec_len)
    # Ensure real at 0 frequency and Nyquist if needed.
    # For 0 frequency, we can set phase=0 because it's real.
    phases[0] = 0.0
    if n % 2 == 0:
        phases[-1] = 0.0  # Nyquist must be real
    # Build complex spec
    spec = M * np.exp(1j * phases)
    # Inverse FFT
    a = np.fft.irfft(spec, n)
    # Take absolute value to ensure nonnegativity (or take square)
    a = np.abs(a)
    # Ensure sum >0
    s = a.sum()
    if s == 0 or np.isnan(s):
        a = np.full(n, 1.0/n)
    else:
        a /= s
    return a

"""
step_function_search.py

A robust 100‑second‑budget optimiser for the “max‑convolution / (sum)²” score.

Algorithm
---------

1. **Fast parametric sweep** – generate many candidates from a wide
   family of analytic shapes (power‑law, exponential, stretched‑exponential,
   beta, double‑Gaussian, sine, random Dirichlet spikes, random‑phase
   constant‑magnitude spectra).  The best few are kept.

2. **Mixture‑exponential global search** – a symmetric mixture of a uniform
   base and a few mirrored exponentials is optimised with SciPy’s
   ``differential_evolution`` (purely black‑box, no gradients).  The
   dimensionality is tiny (≈9), allowing several thousand exact
   objective evaluations.

3. **Long Adam refinement** – the current best vector is refined with
   Adam on a smooth‑max surrogate of the true max‑norm.  The soft‑max
   parametrisation guarantees the simplex constraints.

4. **Exact‑max exponentiated‑gradient sub‑gradient** – finally a
   sub‑gradient descent on the true max‑norm is performed (mirror‑descent
   on the simplex) to squeeze out the last bits of improvement.

5. **Optional annealing fine‑tune** – if time permits a tiny simulated‑
   annealing pass transfers a small fraction of mass between two indices.

All stages respect the overall time budget (default 1000 s) and always
return a non‑negative list that is valid for ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective -------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_sine(power: float, n: int) -> np.ndarray:
    i = np.arange(n, dtype=np.float64)
    t = (i + 0.5) / n
    a = np.sin(np.pi * t) ** power
    a = np.clip(a, 0.0, None)
    a_sum = a.sum()
    if a_sum == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a_sum
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a_sum = a.sum()
    a /= a_sum
    return a


def _candidate_random_phase(n: int) -> np.ndarray:
    """
    Generate a sequence whose DFT has (approximately) constant magnitude.
    The resulting real vector is made non‑negative by taking absolute values.
    """
    # constant magnitude in the frequency domain
    const_mag = 1.0 / np.sqrt(2 * n - 1)
    spec_len = n // 2 + 1
    phases = np.random.uniform(0.0, 2.0 * np.pi, spec_len)
    phases[0] = 0.0                     # zero‑frequency must be real
    if n % 2 == 0:                     # Nyquist term (real for even n)
        phases[-1] = 0.0
    spec = const_mag * np.exp(1j * phases)
    a = np.fft.irfft(spec, n)
    a = np.abs(a)                      # force non‑negativity
    s = a.sum()
    if s == 0 or np.isnan(s):
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


# ----------------------------------------------------------------------
# Adam optimiser (smooth‑max + L₂ regularisation) ---------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 600_000,
    lam_start: float = 0.08,
    lam_factor: float = 1.008,
    l2_weight: float = 0.13,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate:
        smooth‑max(conv) + λ·‖conv – uniform‖₂²
    Returns the best vector seen and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.07

    uniform_target = np.full(2 * n - 1, 1.0 / (2 * n - 1), dtype=np.float64)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    lam = lam_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max (mirrored descent)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact convolution for tracking
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 10_000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # L₂ part
        residual = conv - uniform_target
        grad_l2 = 2.0 * _cross_correlation(residual, a, fft_len)

        grad = grad_smooth + l2_weight * grad_l2

        # chain rule through soft‑max
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam step
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)
        if it % 100_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max sub‑gradient (exponentiated gradient) ---------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad)

        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simulated‑annealing fine‑tuning (optional) ---------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 180_000,
) -> tuple[np.ndarray, float]:
    """Simple annealing that moves a fraction of mass between two positions."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)

    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Mixture‑exponential candidate and DE optimiser -----------------------

def _candidate_mixture_exponential(params: np.ndarray, n: int) -> np.ndarray:
    """
    Build a symmetric mixture:
        a = w0·1 + Σ_{k=1..K} wk·(exp(-αk·i) + exp(-αk·(n‑1‑i)))
    where ``params`` holds log‑weights (K+1 values) followed by log‑alphas (K values).
    The result is normalised to sum = 1.
    """
    K = (len(params) - 1) // 2          # inferred number of exponential components
    log_weights = params[: K + 1]
    log_alphas = params[K + 1 :]

    weights = np.exp(log_weights)
    alphas = np.exp(log_alphas)

    i = np.arange(n, dtype=np.float64)
    a = weights[0] * np.ones(n, dtype=np.float64)            # uniform base
    for k in range(K):
        w = weights[k + 1]
        a += w * (np.exp(-alphas[k] * i) + np.exp(-alphas[k] * (n - 1 - i)))
    total = a.sum()
    if total == 0 or np.isnan(total):
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _de_optimize_mixture(n: int, deadline: float) -> tuple[np.ndarray, float]:
    """
    Differential‑evolution optimisation of the mixture‑exponential family.
    Uses a modest mixture (K=4 components → 9 parameters).
    Returns the best sequence and its exact objective.
    """
    K = 4                              # number of exponentials (plus uniform)
    dim = (K + 1) + K                  # log‑weights + log‑alphas
    # reasonable search ranges:
    #   log‑weights ∈ [‑8, 8]   → weights from ≈3e‑4 to ≈3 000
    #   log‑alphas ∈ [‑8, 2]    → alphas from ≈3e‑4 to 7.4
    bounds = [(-8.0, 8.0)] * (K + 1) + [(-8.0, 2.0)] * K

    # objective wrapper (to be minimised)
    def wrapper(p):
        a = _candidate_mixture_exponential(p, n)
        return _objective(a, n)

    # DE with early‑stop callback
    def stop_callback(xk, convergence):
        return time.time() > deadline

    # allocate a sensible budget: at most a few tens of seconds
    result = differential_evolution(
        wrapper,
        bounds,
        maxiter=200,
        popsize=15,
        updating='deferred',
        polish=False,
        callback=stop_callback,
        seed=0,
        workers=1,
    )
    best_params = result.x
    best_seq = _candidate_mixture_exponential(best_params, n)
    best_val = _objective(best_seq, n)
    return best_seq, best_val


# ----------------------------------------------------------------------
# Main search routine ---------------------------------------------------

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget (seconds) and always returns
    the best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5          # tiny safety margin

    n = 4096
    conv_len = 2 * n - 1
    fft_len = _next_pow2(conv_len)

    best_seq = None
    best_val = float("inf")

    # ------------------------------------------------------------------
    # 0) Warm start from a previously stored sequence (if any)
    if "height_sequence_1" in globals():
        w = np.asarray(globals()["height_sequence_1"], dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.clip(w, 0.0, None)
        s = w.sum()
        if s >= 0.01:
            w = w / s
            val = _objective(w, n)
            if val < best_val:
                best_val = val
                best_seq = w.copy()
                print(f"[Warm ] new best {val:.6f}")

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep (lightweight)
    def _evaluate_and_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # power‑law
    for p in np.linspace(-5.0, -0.05, 30):
        _evaluate_and_update(_candidate_power(p, n))

    # exponential
    for alpha in np.logspace(-6, 1.2, 30):
        _evaluate_and_update(_candidate_exponential(alpha, n))

    # stretched exponential
    for a_alpha in np.logspace(-4, 0.6, 12):
        for b_beta in np.linspace(0.2, 3.5, 9):
            _evaluate_and_update(_candidate_stretched(a_alpha, b_beta, n))

    # beta distributions (random draws)
    for _ in range(80):
        a_par = 10 ** np.random.uniform(-2.0, 1.5)
        b_par = 10 ** np.random.uniform(-2.0, 1.5)
        _evaluate_and_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-3.0, -0.1, 15):
        _evaluate_and_update(_candidate_double_gaussian(sig, n))

    # left‑exponential
    for alpha in np.logspace(-6, 1.5, 20):
        _evaluate_and_update(_candidate_left_exponential(alpha, n))

    # left‑power
    for p in np.linspace(-4.0, 1.0, 20):
        _evaluate_and_update(_candidate_left_power(p, n))

    # sine‑shaped bump
    for pow_ in np.linspace(0.5, 4.5, 12):
        _evaluate_and_update(_candidate_sine(pow_, n))

    # random spikes (different sparsities)
    for k in [32, 64, 128, 256, 512]:
        for _ in range(4):
            _evaluate_and_update(_candidate_spike(k, n, random=True))

    # random‑phase constant‑magnitude candidates
    for _ in range(150):
        _evaluate_and_update(_candidate_random_phase(n))

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # ------------------------------------------------------------------
    # 2) Global search in the mixture‑exponential family (DE)
    if time.time() < deadline - 30:
        print("[Phase‑DE] mixture‑exponential differential evolution")
        de_seq, de_val = _de_optimize_mixture(n, deadline - 5.0)
        if de_val < best_val:
            best_val = de_val
            best_seq = de_seq.copy()
            print(f"[Phase‑DE] new best {de_val:.6f}")

    # ------------------------------------------------------------------
    # 3) Adam refinement (smooth‑max + L₂ regularisation)
    if best_seq is not None and time.time() < deadline - 30:
        print("[Phase‑Adam] long Adam refinement")
        adam_deadline = min(deadline - 5, time.time() + 0.35 * (deadline - time.time()))
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            fft_len,
            adam_deadline,
            max_iters=800_000,
            lam_start=0.07,
            lam_factor=1.005,
            l2_weight=0.12,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # ------------------------------------------------------------------
    # 4) Exact‑max sub‑gradient optimisation (exponentiated‑gradient)
    if best_seq is not None and time.time() < deadline - 10:
        print("[Phase‑Subgrad] exhaustive sub‑gradient optimisation")
        sub_deadline = deadline - 2.0
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            fft_len,
            sub_deadline,
            max_iters=12_000_000,
            eta0=1.0,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # 5) Optional simulated‑annealing fine‑tune (if time permits)
    if best_seq is not None and time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing fine‑tune")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Safeguard & final evaluation
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception as exc:
        print("Final evaluation failed:", exc)

    return best_seq.tolist()
