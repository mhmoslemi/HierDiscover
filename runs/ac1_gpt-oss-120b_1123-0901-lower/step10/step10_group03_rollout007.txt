"""
Step‑function optimizer – improved version.

The algorithm consists of four stages:

1. **Broad parametric sweep** – many analytic families (exponential,
   power‑law, stretched‑exponential, Gamma‑like, Beta, double‑Gaussian,
   random spikes) are generated and evaluated with a fast FFT‐based
   convolution. The best `K` candidates are retained.

2. **Multi‑restart exact‑max sub‑gradient descent** – for each retained
   candidate we run a fast sub‑gradient optimizer that uses the exact
   max‑convolution (no smooth approximation).  The gradient of the
   max‑convolution is computed analytically by a single shift, thus each
   iteration needs only one FFT (instead of two).  This allows many more
   iterations within the time budget.

3. **Optional Adam‑refinement on the smooth‑max surrogate** – a short
   Adam run on the log‑sum‑exp surrogate is performed from the best vector
   found so far, to squeeze out a few extra percent.

4. **Simulated‑annealing fine‑tuner** – a cheap local search that swaps
   small fractions of mass between coordinates, used if time permits.

The routine respects the supplied time budget (`budget_s` seconds) and
always returns a valid non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (full length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


# ----------------------------------------------------------------------
# Exact objective (no clipping) -----------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """2·n·max(a*a)/(sum a)² (exact, no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:                     # should never happen
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators ---------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential a_i ∝ exp(-α·t^β)+exp(-α·(1‑t)^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p (p may be negative)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Monotone decreasing stretched‑exponential a_i ∝ exp(-α·(i/(n‑1))^β)."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta))
    a /= a.sum()
    return a


def _candidate_gamma(shape: float, scale: float, n: int) -> np.ndarray:
    """Gamma‑like a_i ∝ (i+1)^{shape‑1}·exp(-scale·(i+1))."""
    i = np.arange(1, n + 1, dtype=np.float64)
    a = (i ** (shape - 1)) * np.exp(-scale * i)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta distribution on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    total = a.sum()
    if total == 0.0:
        a = np.full(n, 1.0 / n)
    else:
        a /= total
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussian bumps anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_mix_two_expon(alpha1: float, alpha2: float, w1: float, n: int) -> np.ndarray:
    """Mixture of two left‑exponential components."""
    a1 = _candidate_left_exponential(alpha1, n)
    a2 = _candidate_left_exponential(alpha2, n)
    a = w1 * a1 + (1.0 - w1) * a2
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers -----------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max (log‑sum‑exp) surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.1
    lam_factor = 1.02

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of the surrogate (cross‑correlation)
        rev_a = a[::-1]
        fw = np.fft.rfft(w, corr_fft_len)
        fr = np.fft.rfft(rev_a, corr_fft_len)
        corr = np.fft.irfft(fw * fr, corr_fft_len)
        start = a.shape[0] - 1
        grad_a = 2.0 * corr[start:start + a.shape[0]]

        # mirror‑descent on logits
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 100_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_fast(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 10_000_000,
    eta0: float = 2.0,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient optimiser using exponentiated‑gradient updates.
    The gradient of the max‑convolution is computed analytically by a
    single shift, requiring only one FFT per iteration.
    """
    a = np.maximum(a0, 0.0)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
        s = a.sum()
    a /= s

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c       # sum(a) == 1 by construction
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[FastSubgrad it={it}] new best {val:.6f}")

        # active index (first occurrence is fine as a sub‑gradient)
        max_idx = int(conv.argmax())

        # gradient = 2 * shifted copy of a (zero‑padded outside the range)
        grad = np.zeros_like(a)
        start = max_idx - (n - 1)
        if start < 0:
            start = 0
        end = max_idx
        if end > n - 1:
            end = n - 1
        t = np.arange(start, end + 1)
        grad[t] = 2.0 * a[max_idx - t]

        # exponentiated‑gradient update
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)

        # renormalise (protect against under‑flow)
        s = a.sum()
        if s <= 0 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Very cheap simulated‑annealing fine‑tuner."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000.0) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5    # small safety margin

    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)     # needed only for Adam

    # ------------------------------------------------------------------
    # 1) Broad parametric sweep – keep the K best candidates
    K = 5
    best_candidates = []                     # list of (value, vector)

    def _maybe_push(a: np.ndarray):
        """Update the list of top‑K candidates."""
        nonlocal best_candidates
        if time.time() >= deadline:
            return
        val = _objective(a, n)
        if len(best_candidates) < K:
            best_candidates.append((val, a.copy()))
            best_candidates.sort(key=lambda x: x[0])
        else:
            if val < best_candidates[-1][0]:
                best_candidates[-1] = (val, a.copy())
                best_candidates.sort(key=lambda x: x[0])

    # ----- symmetric families ------------------------------------------------
    for p in np.linspace(-6.0, -0.1, 80):
        _maybe_push(_candidate_power(p, n))
    for alpha in np.logspace(-4, 2.5, 70):
        _maybe_push(_candidate_exponential(alpha, n))
    alphas = np.logspace(-3, 1.5, 30)
    betas = np.linspace(0.3, 4.0, 20)
    for alpha in alphas:
        for beta in betas:
            _maybe_push(_candidate_stretched(alpha, beta, n))

    # ----- monotone families -------------------------------------------------
    for alpha in np.logspace(-4, 2.0, 70):
        _maybe_push(_candidate_left_exponential(alpha, n))
    for p in np.linspace(-6.0, 2.0, 80):
        _maybe_push(_candidate_left_power(p, n))
    for beta in np.linspace(0.5, 4.0, 16):
        for alpha in np.logspace(-4, 1.5, 40):
            _maybe_push(_candidate_left_stretched(alpha, beta, n))

    # ----- Gamma‑like --------------------------------------------------------
    shapes = np.linspace(0.2, 4.0, 15)
    scales = np.logspace(-4, 0.5, 15)
    for shape in shapes:
        for scale in scales:
            _maybe_push(_candidate_gamma(shape, scale, n))

    # ----- Beta‑distribution -------------------------------------------------
    a_pars = np.logspace(-2, 0.5, 15)
    b_pars = np.logspace(0, 1.0, 15)
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_push(_candidate_beta(a_par, b_par, n))

    # ----- double Gaussian ---------------------------------------------------
    for sig in np.logspace(-3, -0.2, 20):
        _maybe_push(_candidate_double_gaussian(sig, n))

    # ----- random spikes ------------------------------------------------------
    for k in [8, 16, 32, 64, 128, 256]:
        for _ in range(5):
            _maybe_push(_candidate_spike(k, n, random=True))

    # ----- mixtures of two exponentials (random) ------------------------------
    for _ in range(300):
        alpha1 = 10 ** np.random.uniform(-4, 2)
        alpha2 = 10 ** np.random.uniform(-4, 2)
        w1 = np.random.rand()
        a_mix = _candidate_mix_two_expon(alpha1, alpha2, w1, n)
        _maybe_push(a_mix)

    # ----- Dirichlet random vectors (various concentrations) -----------------
    for conc in [0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0]:
        a = np.random.dirichlet(np.full(n, conc))
        _maybe_push(a)

    # ----- warm‑start from a previously stored global variable ---------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_push(w)

    print(f"[Sweep] best after exploration (top {K}):")
    for idx, (val, _) in enumerate(best_candidates):
        print(f"  #{idx+1}: {val:.6f}")

    # ------------------------------------------------------------------
    # 2) Multi‑restart exact‑max sub‑gradient descent
    # Allocate remaining time roughly equally among the retained candidates.
    remaining = deadline - time.time()
    if remaining <= 0:
        # time exhausted – return the best from sweep
        best_seq = min(best_candidates, key=lambda x: x[0])[1]
        best_seq = np.clip(best_seq, 0.0, 1000.0)
        return best_seq.tolist()

    print("[Phase‑FastSubgrad] running exact‑max sub‑gradient on top candidates")
    best_seq = None
    best_val = np.inf
    for rank, (init_val, a0) in enumerate(best_candidates):
        now = time.time()
        if now >= deadline:
            break
        # give each candidate a fair share of the remaining budget
        per_candidate_deadline = now + max(0.0, (deadline - now) / (len(best_candidates) - rank))
        a_opt, val_opt = _subgradient_optimize_fast(
            a0,
            n,
            conv_fft_len,
            per_candidate_deadline,
            max_iters=10_000_000,
            eta0=2.0,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase‑FastSubgrad] new global best {best_val:.6f} (from candidate #{rank+1})")

    # ------------------------------------------------------------------
    # 3) Short Adam refinement on smooth‑max (if time permits)
    if best_seq is not None and time.time() < deadline - 5.0:
        print("[Phase‑Adam] short Adam refinement")
        adam_deadline = min(deadline - 1.0, time.time() + 30.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            adam_deadline,
            max_iters=300_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # 4) Optional simulated‑annealing fine‑tuning
    if best_seq is not None and time.time() < deadline - 1.0:
        print("[Phase‑Anneal] simulated‑annealing fine tuning")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=200_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved global best to {best_val:.6f}")

    # ------------------------------------------------------------------
    # Final sanitisation and return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # final check with the public evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Could not compute final score (evaluate_sequence not available).")

    return best_seq.tolist()
