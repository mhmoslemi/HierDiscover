"""
Propose a step‑function sequence with a low evaluation score.

The algorithm builds a diverse pool of initial vectors (uniform,
Dirichlet draws, exponentials, Gaussians, power‑laws, deterministic and
random spikes).  Each candidate is first refined with Adam on a smooth
approximation of the max‑convolution, then with a projected‑gradient
descent that performs a tiny line‑search at every iteration, and finally
with the exponentiated‑gradient sub‑gradient method.  The best vector
found is further polished by a short jitter‑and‑refine loop.

All optimisation runs respect the global time budget (≈ 1000 s) and the
result is returned as a plain list of non‑negative floats.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# FFT‑based convolution utilities

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a (length 2·len(a)‑1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a) needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start:start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex."""
    if np.all(v >= 0) and np.isclose(v.sum(), 1.0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
# Objective helpers (internal – avoids the heavy evaluate_sequence call)

def _obj_value(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """2·n·max(conv) for a normalized (sum=1) vector."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())

# Optimisers

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the log‑sum‑exp (smooth‑max) surrogate.
    Returns the best vector encountered and its exact objective.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parameterisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.5                # start temperature
    lam_factor = 1.015

    best_a = a0.copy()
    best_val = _obj_value(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain probabilities
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # smooth‑max weights over convolution entries
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of smooth‑max (via cross‑correlation)
        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # evaluate exact objective
        val = _obj_value(a, n, conv_fft_len)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.93

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 400_000,
    eta0: float = 0.8,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _obj_value(a, n, conv_fft_len)
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[ExpGrad it={it:06d}] new best {val:.6f}")

        # active set (exact max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad_a)     # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _line_search_pg_opt(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 120_000,
) -> tuple[np.ndarray, float]:
    """
    Projected gradient descent on the exact max‑convolution with a tiny
    line‑search (testing a few step‑sizes each iteration).
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _obj_value(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)

        # sub‑gradient (average over all active entries)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # base step that decays like 1/√t
        base_step = 0.25 / np.sqrt(it + 1.0)
        # try a few multipliers around the base step
        candidates = (base_step, base_step * 0.5, base_step * 0.25,
                      base_step * 2.0, 0.0)

        local_best_a = a
        local_best_val = best_val

        for step in candidates:
            a_cand = a - step * grad
            a_cand = np.maximum(a_cand, 0.0)
            a_cand = _project_simplex(a_cand)
            val = _obj_value(a_cand, n, conv_fft_len)
            if val < local_best_val:
                local_best_val = val
                local_best_a = a_cand

        a = local_best_a
        if local_best_val < best_val:
            best_val = local_best_val
            best_a = local_best_a.copy()
            print(f"[PG it={it:06d}] new best {best_val:.6f}")

        it += 1

    return best_a, best_val

# Candidate factories (simple families)

def _make_uniform(n):
    return np.full(n, 1.0 / n, dtype=np.float64)


def _make_dirichlet(n, alpha):
    a = np.random.dirichlet(alpha * np.ones(n))
    return a.astype(np.float64)


def _make_exponential(n, gamma):
    i = np.arange(n, dtype=np.float64)
    left = np.exp(-gamma * i)
    right = left[::-1]
    a = left + right
    a /= a.sum()
    return a


def _make_gaussian(n, sigma):
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-0.5 * ((i - (n - 1) / 2) / (sigma * n)) ** 2)
    a /= a.sum()
    return a


def _make_triangle(n):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i * (n - i + 1)
    a /= a.sum()
    return a


def _make_powerlaw(n, expo):
    i = np.arange(1, n + 1, dtype=np.float64)
    a = i ** expo
    a = np.where(a > 0, a, 0.0)
    a /= a.sum()
    return a


def _make_spikes(n, k):
    """Equally spaced spikes of equal weight."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        a[pos] = 1.0 / k
    a /= a.sum()
    return a


def _make_random_spikes(n, k):
    """k spikes on random positions with Dirichlet weights."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0 and k <= n:
        pos = np.random.choice(n, size=k, replace=False)
        w = np.random.dirichlet(np.ones(k))
        a[pos] = w
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Main search routine

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    Time budget ≈ budget_s seconds (default 1000 s).  Returns the best
    sequence (list of floats) found.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # problem size – 4096 works well and fits the memory budget
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm‑start handling (previous best if available)
    warm = globals().get("height_sequence_1")
    if warm is not None:
        warm = np.asarray(warm, dtype=np.float64)
        if warm.shape[0] > n:
            warm = warm[:n]
        elif warm.shape[0] < n:
            warm = np.pad(warm, (0, n - warm.shape[0]))
        warm = np.maximum(warm, 0.0)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build an enriched pool of diverse initial vectors
    pool = []

    if warm is not None:
        pool.append(warm.copy())
        # a few cyclic shifts of the warm start
        for shift in (n // 4, n // 2, 3 * n // 4):
            pool.append(np.roll(warm, shift))

    # Dirichlet draws with a wide range of concentrations
    for alpha in (0.01, 0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        pool.append(_make_dirichlet(n, alpha))

    # Exponential families (symmetric tails)
    for gamma in (0.1, 0.3, 0.6, 1.0, 2.0):
        pool.append(_make_exponential(n, gamma))

    # Double‑Gaussian families
    for sigma in (0.05, 0.15, 0.3, 0.6):
        pool.append(_make_gaussian(n, sigma))

    # Power‑law families (negative exponents give decreasing tails)
    for expo in (-2.0, -1.5, -1.0, -0.5):
        pool.append(_make_powerlaw(n, expo))

    # Simple triangle (convex) shape
    pool.append(_make_triangle(n))

    # Uniform
    pool.append(_make_uniform(n))

    # Random sparse spike patterns
    for k in (32, 64, 128):
        pool.append(_make_spikes(n, k))

    # Random positions with random weights
    for k in (32, 64, 128):
        pool.append(_make_random_spikes(n, k))

    # A few completely random Dirichlet draws
    for _ in range(12):
        alpha = 10 ** np.random.uniform(-3, 1)  # between 1e‑3 and 10
        pool.append(_make_dirichlet(n, alpha))

    # Shuffle to avoid ordering bias
    np.random.shuffle(pool)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = 15   # number of distinct starts we will treat seriously

    # ------------------------------------------------------------------
    # Phase I – intensive optimisation of each start
    for idx, a0 in enumerate(pool):
        if time.time() > deadline or idx >= max_candidates:
            break

        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # ----- Adam (smooth‑max) ------------------------------------
        sub_dead = min(deadline, time.time() + 18.0)   # ~18 s
        a_adam, val_adam = _adam_optimize(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=250_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # ----- Projected gradient with line‑search ------------------
        if time.time() < deadline - 5:
            sub_dead = min(deadline, time.time() + 16.0)   # ~16 s
            a_pg, val_pg = _line_search_pg_opt(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=120_000,
            )
            if val_pg < best_val:
                best_val = val_pg
                best_seq = a_pg.copy()
                print(f"[Explore {idx}] PG improved to {val_pg:.6f}")

        # ----- Exponentiated‑gradient sub‑gradient --------------------
        if time.time() < deadline - 5:
            sub_dead = min(deadline, time.time() + 22.0)   # ~22 s
            a_sub, val_sub = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=500_000, eta0=0.9,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] ExpGrad improved to {val_sub:.6f}")

    # ------------------------------------------------------------------
    # Phase II – intensive polishing of the current best
    if best_seq is not None and time.time() < deadline - 20:
        print("[Polish] Long PG refinement")
        sub_dead = min(deadline, time.time() + 30.0)
        a_pg2, val_pg2 = _line_search_pg_opt(
            best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
            max_iters=200_000,
        )
        if val_pg2 < best_val:
            best_val, best_seq = val_pg2, a_pg2.copy()
            print(f"[Polish] PG improved to {val_pg2:.6f}")

        # a final short exponentiated‑gradient sweep
        if time.time() < deadline - 5:
            print("[Polish] ExpGrad final sweep")
            sub_dead = min(deadline, time.time() + 15.0)
            a_sub2, val_sub2 = _subgradient_optimize_expgrad(
                best_seq, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=300_000, eta0=0.7,
            )
            if val_sub2 < best_val:
                best_val, best_seq = val_sub2, a_sub2.copy()
                print(f"[Polish] ExpGrad improved to {val_sub2:.6f}")

    # ------------------------------------------------------------------
    # Phase III – jitter‑and‑refine loop (helps escape shallow minima)
    if best_seq is not None and time.time() < deadline - 5:
        print("[Jitter] Random perturbations + quick PG")
        while time.time() < deadline - 5:
            jitter = np.random.normal(scale=0.02, size=n)   # 2 % noise
            cand = best_seq + jitter
            cand = np.maximum(cand, 0.0)
            cand = _project_simplex(cand)

            sub_dead = min(deadline, time.time() + 4.0)     # short budget
            a_jit, val_jit = _line_search_pg_opt(
                cand, n, conv_fft_len, corr_fft_len, sub_dead,
                max_iters=30_000,
            )
            if val_jit < best_val:
                best_val, best_seq = val_jit, a_jit.copy()
                print(f"[Jitter] got improvement {val_jit:.6f}")

    # ------------------------------------------------------------------
    # Fallback – guarantee a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range (non‑negative, ≤ 1000)
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
