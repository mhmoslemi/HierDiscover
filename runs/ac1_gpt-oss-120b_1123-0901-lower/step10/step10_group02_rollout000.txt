def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5
    # Choose length n = 4096 (could be changed)
    n = 4096
    conv_len = _next_pow2(2*n-1)
    corr_len = _next_pow2(3*n-2)

    # Warm start
    warm = globals().get("height_sequence_1")
    if warm is not None:
        a0 = np.asarray(warm, dtype=float)
        if a0.size > n:
            a0 = a0[:n]
        elif a0.size < n:
            a0 = np.pad(a0, (0, n-a0.size), constant_values=0.0)
    else:
        a0 = np.random.dirichlet(np.ones(n))

    # Ensure sum=1
    a0 = np.maximum(a0, 0)
    a0 /= a0.sum()

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_len)
    print("[Init] objective", best_val)

    # Define helper adaptation loops

    # Stage: multiple runs of exponentiated gradient with restarts
    while time.time() < deadline:
        # run a sub-phase of expgrad from current best or random start
        # Determine time to allocate for each run: e.g., 30 sec per run
        run_deadline = min(deadline, time.time() + 30.0)
        a_start = best_a.copy()
        # maybe occasionally random restart with prob 0.1
        if np.random.rand() < 0.1:
            a_start = np.random.dirichlet(np.ones(n))
        a_opt, val_opt = _subgradient_optimize_expgrad(a_start, n, conv_len, corr_len, run_deadline, max_iters=2_000_000, eta0=0.8)
        if val_opt < best_val:
            best_val = val_opt
            best_a = a_opt.copy()
            print("[ExpGrad restart] improved to", best_val)
        # possibly a short Adam after each
        if time.time() < deadline:
            adam_dead = min(deadline, time.time() + 10.0)
            a_ad, val_ad = _adam_optimize(best_a, n, conv_len, corr_len, adam_dead, max_iters=100_000)
            if val_ad < best_val:
                best_val = val_ad
                best_a = a_ad.copy()
                print("[Adam after] improved to", best_val)
        # small random perturbation local search
        # perform a few steps of random tweaking per iteration
        for _ in range(5):
            if time.time() >= deadline: break
            # pick two indices i != j
            i, j = np.random.choice(n, 2, replace=False)
            delta = (np.random.rand() - 0.5) * 0.01  # small transfer proportion
            # ensure positivity
            if best_a[i] - delta < 0:
                delta = best_a[i] * 0.5
            new_a = best_a.copy()
            new_a[i] -= delta
            new_a[j] += delta
            new_a = _project_simplex(new_a) # maintain sum=1
            val = _objective(new_a, n, conv_len)
            if val < best_val:
                best_val = val
                best_a = new_a
                print("[Perturb] improved to", best_val)
    # final cyclic shifts
    for sh in [n//8, n//4, 3*n//8, n//2, 5*n//8, 3*n//4, 7*n//8]:
        a_shift = np.roll(best_a, sh)
        val = _objective(a_shift, n, conv_len)
        if val < best_val:
            best_val = val
            best_a = a_shift
            print("[Shift] improved to", best_val)

    # final sanity: clip, ensure sum>0, evaluate
    best_a = np.clip(best_a, 0.0, 1000.0)
    if best_a.sum() < 0.01:
        best_a += 1e-6
        best_a /= best_a.sum()
    final_val = evaluate_sequence(best_a.tolist())
    print("Final evaluation", final_val)
    return best_a.tolist()

"""
Hybrid search for a low‑value step‑function.

The algorithm mixes three ingredients:

1. **Warm start** – uses the global variable ``height_sequence_1`` if it
   exists, otherwise starts from a random Dirichlet vector on the simplex.

2. **Long exponentiated‑gradient (EG) phase** – EG is the natural mirror‑descent
   algorithm for the exact max‑convolution objective.  It is run repeatedly
   from the current best solution (occasionally from a fresh random start) until
   the time budget runs out.  The EG step is
        a ← a ⊙ exp(‑η·∇)    (⊙ = elementwise product)
   followed by Euclidean projection onto the probability simplex.

3. **Short refinement passes** – a brief Adam optimisation on a smooth‑max
   surrogate and a few cheap random perturbations are performed after each EG
   block.  Finally a set of deterministic cyclic shifts is examined.

The routine respects a 1000 s wall‑clock limit, always returns a valid
non‑negative sequence and prints progress information.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using an FFT of length ``fft_len``."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    # keep the first 2·len(a)‑1 entries (the linear convolution)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear cross‑correlation (w ★ a).
    ``w`` has length 2·len(a)‑1, ``a`` length n.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start : start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of ``v`` onto the probability simplex."""
    if v.sum() == 1.0 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray, n: int, conv_fft_len: int) -> float:
    """Exact evaluation 2·n·max(conv) for a probability vector ``a``."""
    conv = _full_convolution(a, conv_fft_len)
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _adam_optimize(a0: np.ndarray, n: int, conv_fft_len: int,
                   corr_fft_len: int, deadline: float,
                   max_iters: int = 500_000) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate of the max‑convolution.
    The parameters are tuned for a slower “temperature” increase than the
    reference implementation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)          # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.04
    lam = 0.01                # start with a very smooth surrogate
    lam_factor = 1.01         # increase temperature slowly

    best_a = a0.copy()
    best_val = _objective(best_a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain a probability vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights (soft assignment to the active set)
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # chain rule through the soft‑max (gradient w.r.t. logits)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        # occasional learning‑rate decay
        if it % 100_000 == 0 and it > 0:
            lr *= 0.9

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(a0: np.ndarray, n: int,
                                 conv_fft_len: int, corr_fft_len: int,
                                 deadline: float,
                                 max_iters: int = 10_000_000,
                                 eta0: float = 1.0) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient phase using exponentiated‑gradient updates.
    The step‑size schedule η = η₀ / √t is standard for EG.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set (indices attaining the max)
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad_a)

        # re‑normalise onto the simplex
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _max_conv_descent(a0: np.ndarray, n: int,
                     conv_fft_len: int, corr_fft_len: int,
                     deadline: float,
                     max_iters: int = 300_000) -> tuple[np.ndarray, float]:
    """
    Plain projected gradient descent on the exact max‑convolution.
    Used as a short polishing pass.
    """
    a = a0.copy()
    best_a = a.copy()
    best_val = _objective(a, n, conv_fft_len)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = float(2.0 * n * max_c)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.08 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        it += 1

    return best_a, best_val
