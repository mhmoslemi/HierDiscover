def _candidate_piecewise_linear(values: np.ndarray, n: int) -> np.ndarray:
    # values length K >=2.
    K = len(values)
    positions = np.linspace(0, 1, K)
    t = np.arange(n, dtype=np.float64) / (n-1)
    a = np.interp(t, positions, values)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a
```

"""
Propose a low‑score step‑function for the given evaluation.
The algorithm combines

1. a dense parametric sweep (including several new asymmetric families);
2. a set of random Dirichlet restarts refined by exponentiated‑gradient
   (sub‑gradient) descent;
3. a longer Adam‑style optimisation on a smooth‑max surrogate;
4. a short simulated‑annealing polishing phase.

All phases respect the time budget (default 1000 s) and always return a
non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np
from scipy.optimize import differential_evolution

# ----------------------------------------------------------------------
# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a (length 2·len(a)-1) using FFT."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.ifft(fa * fa, fft_len).real
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the sub‑gradient.
    ``w`` has length 2·n‑1, ``a`` length n.
    The result has length n and corresponds to
    ∂max(w·(a*a))/∂a.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.ifft(fw * fr, fft_len).real
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective 2·n·max(a*a)/(sum a)² (no clipping).
    ``a`` is assumed to be non‑negative and already normalised to sum=1.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    return float(2.0 * n * max_c)


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n-i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_two_exponential(alpha_left: float, alpha_right: float, n: int) -> np.ndarray:
    """Asymmetric exponential: left side uses α_left, right side α_right."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha_left * i) + np.exp(-alpha_right * (n - 1 - i))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_two_power(p_left: float, p_right: float, n: int) -> np.ndarray:
    """Asymmetric power‑law (different exponents on each side)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p_left + (n - i) ** p_right
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (random positions if ``random``)."""
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


def _candidate_piecewise_linear(values: np.ndarray, n: int) -> np.ndarray:
    """
    Piecewise linear shape defined by ``values`` at equally spaced break‑points.
    ``values`` must contain at least two positive numbers.
    """
    K = len(values)
    positions = np.linspace(0.0, 1.0, K)
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.interp(t, positions, values)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimisers ------------------------------------------------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 1_000_000,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the smooth‑max surrogate (log‑sum‑exp) while
    staying on the simplex via a soft‑max parametrisation.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lr = 0.05
    lam = 0.2           # temperature for smooth‑max
    lam_factor = 1.4

    best_a = a0.copy()
    best_val = _objective(a0, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        # Soft‑max → a on the simplex
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # exact objective for logging
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:07d}] new best {val:.6f}")

        # smooth‑max surrogate weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        # gradient of surrogate (w.r.t. a)
        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        # Chain rule through soft‑max (mirror‑descent form)
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # temperature schedule
        lam = min(lam * lam_factor, 1e7)

        # occasional LR decay
        if it % 200_000 == 0 and it > 0:
            lr *= 0.85

        it += 1

    return best_a, best_val


def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    corr_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient (mirror descent on simplex) using the exact
    sub‑gradient of the max‑convolution.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_len)
        max_c = conv.max()
        val = 2.0 * n * max_c

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 250_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set where convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0)
        w = np.zeros_like(conv)
        if active.any():
            w[active] = 1.0 / active.sum()

        grad_a = 2.0 * _cross_correlation(w, a, corr_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)

        # renormalise
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing that transfers a random fraction of mass
    between two coordinates.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.25
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the supplied time budget and always returns the
    best sequence found.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.4  # small safety margin

    # --------------------------------------------------------------
    # problem dimensions
    n = 4096
    conv_len = _next_pow2(2 * n - 1)
    corr_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # bookkeeping
    best_seq = None
    best_val = np.inf

    def _maybe_update(a: np.ndarray):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[Sweep] new best {val:.6f}")

    # --------------------------------------------------------------
    # 1) dense parametric sweep – include a few new families

    # symmetric power‑law
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_power(p, n))

    # symmetric exponential
    for alpha in np.logspace(-5, 1.0, 110):
        _maybe_update(_candidate_exponential(alpha, n))

    # asymmetric exponential (two alphas)
    alphas_left = np.logspace(-5, 1.0, 30)
    alphas_right = np.logspace(-5, 1.0, 30)
    for al in alphas_left:
        for ar in alphas_right[::3]:   # a subset to keep runtime reasonable
            _maybe_update(_candidate_two_exponential(al, ar, n))

    # stretched exponential (symmetric)
    for alpha in np.logspace(-4, 0.7, 20):
        for beta in np.linspace(0.2, 3.0, 15):
            _maybe_update(_candidate_stretched(alpha, beta, n))

    # beta distributions (skew allowed)
    a_pars = [0.1, 0.3, 0.6, 1.0, 2.0, 5.0, 10.0]
    b_pars = a_pars
    for a_par in a_pars:
        for b_par in b_pars:
            _maybe_update(_candidate_beta(a_par, b_par, n))

    # double Gaussian
    for sig in np.logspace(-3.0, -0.2, 20):
        _maybe_update(_candidate_double_gaussian(sig, n))

    # monotone left‑exponential
    for alpha in np.logspace(-5, 1.2, 100):
        _maybe_update(_candidate_left_exponential(alpha, n))

    # monotone left‑power
    for p in np.linspace(-4.0, -0.01, 80):
        _maybe_update(_candidate_left_power(p, n))

    # monotone left‑logistic
    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 60):
        _maybe_update(_candidate_left_logistic(beta, mu, n))

    # monotone linear decreasing
    for slope in np.linspace(0.1, 3.0, 60):
        _maybe_update(_candidate_left_linear(slope, n))

    # random spikes
    for k in [32, 64, 128, 256]:
        for _ in range(6):
            _maybe_update(_candidate_spike(k, n, random=True))

    # piecewise linear with random break‑point values
    K = 8
    for _ in range(300):
        vals = np.random.rand(K) + 0.01   # ensure positivity
        _maybe_update(_candidate_piecewise_linear(vals, n))

    # a few pure random Dirichlet samples (dense)
    for _ in range(30):
        a_rand = np.random.dirichlet(np.ones(n))
        _maybe_update(a_rand)

    # Warm start from a global variable if present
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w)

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Adam polishing (allow more time if still available)
    if time.time() < deadline - 40:
        print("[Phase‑Adam] polishing")
        adam_deadline = min(deadline - 5, time.time() + 25.0)
        a_adam, val_adam = _adam_optimize(
            best_seq,
            n,
            conv_len,
            corr_len,
            adam_deadline,
            max_iters=800_000,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Phase‑Adam] improved to {val_adam:.6f}")

    # --------------------------------------------------------------
    # 3) Stochastic Dirichlet restarts + sub‑gradient refinement
    remaining = deadline - time.time()
    max_restarts = 6
    for r in range(max_restarts):
        if time.time() >= deadline - 8:
            break
        concentration = float(np.random.choice([0.05, 0.1, 0.3, 0.7, 1.0, 2.0, 5.0]))
        a0 = np.random.dirichlet(np.full(n, concentration))
        slice_deadline = time.time() + min(remaining / (max_restarts - r), 22.0)
        a_sub, val_sub = _subgradient_optimize_expgrad(
            a0,
            n,
            conv_len,
            corr_len,
            slice_deadline,
            max_iters=2_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Restart {r}] new best {val_sub:.6f}")
        remaining = deadline - time.time()

    # --------------------------------------------------------------
    # 4) Long sub‑gradient run on current best
    if time.time() < deadline - 6:
        print("[Phase‑Subgrad] final long run")
        a_sub, val_sub = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_len,
            corr_len,
            deadline,
            max_iters=12_000_000,
            eta0=0.9,
        )
        if val_sub < best_val:
            best_val = val_sub
            best_seq = a_sub.copy()
            print(f"[Phase‑Subgrad] improved to {val_sub:.6f}")

    # --------------------------------------------------------------
    # 5) Simulated‑annealing fine‑tune (short)
    if time.time() < deadline - 1:
        print("[Phase‑Anneal] final anneal")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_len,
            deadline,
            max_moves=350_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # final sanity checks
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    best_seq = np.clip(best_seq, 0.0, 1000.0)

    # Verify with the provided evaluator (if available)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score = {final_score:.6f}")
    except Exception:
        print("Could not evaluate final score (evaluator not present).")

    return best_seq.tolist()
