"""
Step‑function optimizer (improved).

The search proceeds in three phases:

1. **Dense parametric sweep + uniform mixing** – many analytic families
   (exponential, power‑law, beta, stretched‑exponential, logistic) are
   sampled on a fine grid.  Each candidate is also linearly blended with
   the uniform distribution (which is an exact optimum for the L₂‑norm)
   for several mixing ratios; this yields a richer collection of shapes.

2. **Long exponentiated‑gradient refinement** – starting from the best
   candidate found above, a mirror‑descent (exponentiated‑gradient) scheme
   minimises the exact objective `2·n·max(conv)/(sum²)`.  The step‑size
   decays as ηₜ = η₀/√(t+1) (η₀=0.5 works well).  The routine runs until
   the time budget expires, continuously tracking the best iterate.

3. **Short simulated‑annealing polishing** – a final local‑search moves a
   random fraction of mass between two coordinates, accepting deteriorations
   with a slowly decreasing temperature.  This helps to escape shallow
   plateaux left by the gradient method.

All heavy work (convolution, cross‑correlation) is performed with FFTs.
The implementation respects the 1000 s budget, prints progress, and always
returns a non‑negative list that can be fed to ``evaluate_sequence``.
"""

import time
import numpy as np

# FFT utilities ---------------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Linear convolution a*a (length 2·len(a)‑1) using FFT.
    """
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for the gradient of max‑convolution.
    The result has length len(a) and corresponds to the sub‑gradient of
    max( a ★ a ) w.r.t. a when w is an indicator of the active set.
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Objective ------------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """
    Exact objective: 2·n·max(a*a) / (sum a)².
    ``a`` must be non‑negative and is assumed to be on the simplex.
    """
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate generators --------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law: a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential: a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution (skewed allowed)."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.maximum(a, 0.0)
    if a.sum() == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= a.sum()
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing exponential (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing power‑law: a_i ∝ (i+1)^p (p<0)."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_logistic(beta: float, mu: float, n: int) -> np.ndarray:
    """Sigmoid decreasing shape."""
    i = np.arange(n, dtype=np.float64)
    a = 1.0 / (1.0 + np.exp(beta * (i - mu)))
    a = np.maximum(a, 0.0)
    a /= a.sum()
    return a


def _candidate_left_linear(slope: float, n: int) -> np.ndarray:
    """Linear decreasing shape clipped at zero."""
    i = np.arange(n, dtype=np.float64)
    a = np.maximum(0.0, 1.0 - slope * i / (n - 1))
    a /= a.sum()
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """
    k spikes (random positions if ``random``).  The masses are drawn from a
    Dirichlet distribution so the total sum is 1.
    """
    a = np.zeros(n, dtype=np.float64)
    if random:
        pos = np.random.choice(n, size=k, replace=False)
    else:
        pos = np.round(np.linspace(0, n - 1, k)).astype(int)
    if k > 0:
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Optimiser ------------------------------------------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 2_000_000,
    eta0: float = 0.5,
) -> tuple[np.ndarray, float]:
    """
    Exact‑max sub‑gradient (exponentiated‑gradient / mirror descent on the simplex).
    Step size ηₜ = η₀ / √(t+1).  Stops when the deadline is reached.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c / (a.sum() ** 2)

        # keep the best ever seen
        if val < best_val:
            best_val = val
            best_a = a.copy()
            # occasional progress report
            if it % 100_000 == 0:
                print(f"[SubGrad] it={it:07d} new best {val:.6f}")

        # active set where the convolution attains the maximum
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update (mirror descent on the simplex)
        a = a * np.exp(-eta * grad_a)

        # renormalise, guard against numerical issues
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 300_000,
) -> tuple[np.ndarray, float]:
    """
    Simple simulated‑annealing local search: repeatedly move a random
    fraction of mass from one coordinate to another.
    """
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max() / (a.sum() ** 2)
    best_val = val
    best_a = a.copy()

    T_start = 0.1
    T_end = 1e-4
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        # transfer a random fraction (≤30 %) of source mass
        frac = np.random.rand() * 0.3
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max() / (a2.sum() ** 2)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Main entry point ------------------------------------------------------


def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects the given time budget and always returns the best
    sequence found.  It combines:
        – a dense parametric sweep (including mixtures with the uniform)
        – a long exponentiated‑gradient refinement
        – a short simulated‑annealing polishing.
    """
    np.random.seed(seed)
    start_time = time.time()
    deadline = start_time + budget_s - 0.5   # safety margin

    # --------------------------------------------------------------
    # problem size
    n = 4096
    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # --------------------------------------------------------------
    # bookkeeping
    best_seq = np.full(n, 1.0 / n, dtype=np.float64)   # uniform start
    best_val = _objective(best_seq, n)
    print(f"[Init] uniform value = {best_val:.6f}")

    def _maybe_update(a: np.ndarray, tag: str = ""):
        nonlocal best_seq, best_val
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_seq = a.copy()
            print(f"[{tag}] new best {val:.6f}")

    # --------------------------------------------------------------
    # 1) dense parametric sweep + mixing with uniform
    uniform = best_seq.copy()

    # ----- left‑exponential -------------------------------------------------
    for alpha in np.logspace(-5, 0.6, 140):
        cand = _candidate_left_exponential(alpha, n)
        _maybe_update(cand, "Exp")
        # mix with uniform (λ up to 0.4)
        for lam in (0.1, 0.2, 0.3, 0.4):
            mix = (1.0 - lam) * uniform + lam * cand
            _maybe_update(mix, f"ExpMix{lam}")

    # ----- left‑power -------------------------------------------------------
    for p in np.linspace(-4.0, -0.01, 140):
        cand = _candidate_left_power(p, n)
        _maybe_update(cand, "Pow")
        for lam in (0.1, 0.2, 0.3):
            mix = (1.0 - lam) * uniform + lam * cand
            _maybe_update(mix, f"PowMix{lam}")

    # ----- beta distributions (randomised) ---------------------------------
    for _ in range(120):
        a_par = 10 ** np.random.uniform(-2.0, 1.0)   # 0.01 … 10
        b_par = 10 ** np.random.uniform(-2.0, 1.0)
        cand = _candidate_beta(a_par, b_par, n)
        _maybe_update(cand, "Beta")
        # light mixing
        lam = 0.15
        mix = (1.0 - lam) * uniform + lam * cand
        _maybe_update(mix, "BetaMix")

    # ----- stretched‑exponential --------------------------------------------
    alphas = np.logspace(-4, 0.6, 30)
    betas = np.linspace(0.3, 3.0, 20)
    for alpha in alphas:
        for beta in betas:
            cand = _candidate_stretched(alpha, beta, n)
            _maybe_update(cand, "Stretched")

    # ----- logistic (sigmoid) ------------------------------------------------
    mu = n / 2.0
    for beta in np.logspace(-3, 1.5, 60):
        cand = _candidate_left_logistic(beta, mu, n)
        _maybe_update(cand, "Logi")

    # ----- linear decreasing -------------------------------------------------
    for slope in np.linspace(0.2, 3.0, 40):
        cand = _candidate_left_linear(slope, n)
        _maybe_update(cand, "Linear")

    # ----- double Gaussian ---------------------------------------------------
    for sig in np.logspace(-3.0, -0.2, 30):
        cand = _candidate_double_gaussian(sig, n)
        _maybe_update(cand, "DGaus")

    # ----- random spikes (many) ---------------------------------------------
    for k in (32, 64, 128, 256):
        for _ in range(6):
            cand = _candidate_spike(k, n, random=True)
            _maybe_update(cand, f"Spike{k}")

    # ----- warm start from a previous global variable, if any ---------------
    warm = globals().get("height_sequence_1")
    if warm is not None:
        w = np.asarray(warm, dtype=np.float64)
        if w.shape[0] > n:
            w = w[:n]
        elif w.shape[0] < n:
            w = np.pad(w, (0, n - w.shape[0]))
        w = np.maximum(w, 0.0)
        if w.sum() >= 0.01:
            w = w / w.sum()
            _maybe_update(w, "Warm")

    print(f"[Sweep] best after exploration: {best_val:.6f}")

    # --------------------------------------------------------------
    # 2) Long exponentiated‑gradient refinement (starting from best)
    if time.time() < deadline - 30:
        print("[Phase‑Grad] long sub‑gradient refinement")
        a_opt, val_opt = _subgradient_optimize_expgrad(
            best_seq,
            n,
            conv_fft_len,
            corr_fft_len,
            deadline,
            max_iters=10_000_000,
            eta0=0.5,
        )
        if val_opt < best_val:
            best_val = val_opt
            best_seq = a_opt.copy()
            print(f"[Phase‑Grad] improved to {val_opt:.6f}")

    # --------------------------------------------------------------
    # 3) Optional simulated‑annealing fine‑tuning (short if time permits)
    if time.time() < deadline - 1:
        print("[Phase‑Anneal] final annealing")
        a_ann, val_ann = _anneal_optimize(
            best_seq,
            n,
            conv_fft_len,
            deadline,
            max_moves=400_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Phase‑Anneal] improved to {val_ann:.6f}")

    # --------------------------------------------------------------
    # final sanity checks / clipping
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        print("Could not compute final score (evaluator missing).")
        final_score = None

    return best_seq.tolist()
