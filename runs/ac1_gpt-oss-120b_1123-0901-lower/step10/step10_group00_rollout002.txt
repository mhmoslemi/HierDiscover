"""
Hybrid multi‑start optimizer for the max‑autocorrelation problem.

The routine builds a diverse pool of start vectors (warm‑start, Dirichlet,
spikes, piece‑wise constant blocks, sinusoid, etc.).  For each start a
randomised configuration of hyper‑parameters is chosen and a cascade of
optimisation phases is run, always respecting the global deadline:

    1. Adam on a smooth‑max surrogate of the max‑convolution.
    2. Exact‑max sub‑gradient (exponentiated‑gradient) refinement.
    3. Spectral‑flattening: force the FFT magnitude to be (almost) constant.
    4. Projected gradient descent on the exact max‑convolution.
    5. Simulated‑annealing local search.

The best sequence seen over all runs is kept, finally clipped to the
evaluator’s allowed range and returned.  Randomised hyper‑parameters
(learning rates, smooth‑max temperature schedule, etc.) increase the
exploration of the non‑convex landscape and usually improve the bound
below the previous best (~1.5077) within the 1000 s budget.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a * a using FFT (length 2·len(a)‑1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear cross‑correlation (w ★ a) needed for gradients."""
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Euclidean projection of v onto the probability simplex {x≥0, Σx=1}."""
    if v.sum() == 1 and np.all(v >= 0):
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(len(v)) + 1
    cond = u - cssv / ind > 0
    if not np.any(cond):
        return np.full_like(v, 1.0 / len(v))
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    return np.maximum(v - theta, 0.0)


# ----------------------------------------------------------------------
def _adam_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    lr: float = 0.05,
    lam_init: float = 0.1,
    lam_factor: float = 1.02,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Adam optimiser on the smooth‑max surrogate."""
    eps = 1e-15
    logits = np.log(a0 + eps)                # soft‑max parametrisation
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    lam = lam_init

    best_a = a0.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        # soft‑max to obtain current vector
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Adam it={it:06d}] new best {val:.6f}")

        # smooth‑max weights
        w = np.exp(lam * (conv - max_c))
        w_sum = w.sum()
        if w_sum == 0 or np.isnan(w_sum):
            w = np.full_like(w, 1.0 / w.size)
        else:
            w /= w_sum

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        # back‑prop through soft‑max
        avg = np.dot(a, grad_a)
        grad_log = a * (grad_a - avg)

        # Adam update
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        lam = min(lam * lam_factor, 1e7)

        if it % 50_000 == 0 and it > 0:
            lr *= 0.95

        it += 1

    return best_a, best_val


def _subgradient_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    eta0: float = 1.0,
    max_iters: int = 300_000,
) -> tuple[np.ndarray, float]:
    """Exact‑max sub‑gradient optimisation via exponentiated‑gradient."""
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a = a / a.sum()

    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 20_000 == 0:
                print(f"[Subgrad it={it:06d}] new best {val:.6f}")

        # active set of indices attaining the current max
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad_a = 2.0 * _cross_correlation(w, a, corr_fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        a = a * np.exp(-eta * grad_a)   # exponentiated‑gradient step
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


def _spectral_flatten_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_iters: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Force FFT magnitude to be almost constant while staying on the simplex."""
    a = a0.copy()
    best_a = a.copy()
    best_val = float("inf")
    it = 0

    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[Spectral it={it:06d}] new best {val:.6f}")

        A = np.fft.fft(a)               # full complex FFT
        energy = np.dot(a, a)           # ‖a‖₂²

        if n > 1:
            rem_energy = max(0.0, n * energy - (np.abs(A[0]) ** 2))
            const_mag = np.sqrt(rem_energy / (n - 1))
        else:
            const_mag = np.sqrt(energy)

        A_new = np.empty_like(A)
        A_new[0] = A[0]                     # keep DC unchanged
        if n > 1:
            A_new[1:] = const_mag * np.exp(1j * np.angle(A[1:]))

        a_new = np.real(np.fft.ifft(A_new))

        a_new = np.maximum(a_new, 0.0)
        s = a_new.sum()
        if s < 1e-12 or np.isnan(s):
            a_new = np.random.dirichlet(np.ones(n))
            s = a_new.sum()
        a = a_new / s

        it += 1

    return best_a, best_val


def _max_conv_descent_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    corr_fft_len: int,
    deadline: float,
    max_iters: int = 150_000,
) -> tuple[np.ndarray, float]:
    """Projected gradient descent on the exact max‑convolution."""
    a = a0.copy()
    best_a = a.copy()
    conv = _full_convolution(a, conv_fft_len)
    best_val = 2.0 * n * conv.max()

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, conv_fft_len)
        max_c = conv.max()
        active = np.isclose(conv, max_c, atol=1e-14, rtol=0)
        w = np.zeros_like(conv)
        if np.any(active):
            w[active] = 1.0 / active.sum()

        grad = 2.0 * _cross_correlation(w, a, corr_fft_len)

        step = 0.1 / np.sqrt(it + 1.0)
        a = a - step * grad
        a = np.maximum(a, 0.0)
        a = _project_simplex(a)

        val = 2.0 * n * max_c
        if val < best_val:
            best_val = val
            best_a = a.copy()
            print(f"[GD it={it:06d}] new best {val:.6f}")

        it += 1

    return best_a, best_val


def _anneal_optimize_custom(
    a0: np.ndarray,
    n: int,
    conv_fft_len: int,
    deadline: float,
    max_moves: int = 60_000,
    T_start: float = 0.08,
    T_end: float = 1e-4,
) -> tuple[np.ndarray, float]:
    """Simulated‑annealing local search on the simplex."""
    a = a0.copy()
    conv = _full_convolution(a, conv_fft_len)
    val = 2.0 * n * conv.max()
    best_val = val
    best_a = a.copy()

    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-14:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.4
        delta = a[i] * frac

        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, conv_fft_len)
        val2 = 2.0 * n * conv2.max()

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                val = val2

        it += 1

    return best_a, best_val

def _initial_candidates(n: int, warm: np.ndarray | None) -> list[np.ndarray]:
    """Create a modest but diverse pool of start vectors."""
    cand = []

    if warm is not None:
        cand.append(warm.copy())
        for shift in (n // 4, n // 2, 3 * n // 4):
            cand.append(np.roll(warm, shift))

    # Dirichlet draws with various concentrations
    for alpha in (0.05, 0.2, 0.5, 1.0, 2.0, 5.0):
        cand.append(np.random.dirichlet(alpha * np.ones(n)))

    # Edge‑exponential families
    for gamma in (0.2, 0.5, 1.0, 2.0):
        left = np.exp(-gamma * np.arange(n, dtype=np.float64))
        right = left[::-1]
        shape = left + right
        shape /= shape.sum()
        cand.append(shape)

    # Power‑law families
    idx = np.arange(1, n + 1, dtype=np.float64)
    prod = idx * (n - idx + 1.0)
    for alpha in np.linspace(-2.0, -0.5, 5):
        shape = np.where(prod > 0, prod ** alpha, 0.0)
        if shape.sum() > 0:
            shape /= shape.sum()
            cand.append(shape)

    # Double‑Gaussian
    i = np.arange(n, dtype=np.float64)
    for sigma in (0.1, 0.3, 0.6):
        shape = np.exp(-0.5 * ((i) / (sigma * n)) ** 2) \
                + np.exp(-0.5 * ((i - (n - 1)) / (sigma * n)) ** 2)
        shape /= shape.sum()
        cand.append(shape)

    # Half‑wave sinusoid
    shape = np.sin(np.pi * (np.arange(n) + 0.5) / n)
    shape = np.maximum(shape, 0.0)
    shape /= shape.sum()
    cand.append(shape)

    # Uniform
    cand.append(np.full(n, 1.0 / n))

    return cand


def _spike_candidates(n: int) -> list[np.ndarray]:
    """Sparse candidates composed of a handful of equally spaced spikes."""
    out = []
    for k in (16, 32, 64, 128, 256, 512):
        a = np.zeros(n, dtype=np.float64)
        if k > 0:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
            a[pos] = 1.0 / k
        out.append(a)

    for k in (32, 64, 128):
        for _ in range(3):
            a = np.zeros(n, dtype=np.float64)
            pos = np.random.choice(n, size=k, replace=False)
            masses = np.random.dirichlet(np.ones(k))
            a[pos] = masses
            out.append(a)
    return out

def propose_candidate(seed: int = 0, budget_s: float = 1000) -> list:
    """
    Search for a non‑negative step‑function with a low evaluation score.
    The routine respects a total time budget (≈ 1000 s) and always returns
    the best sequence found before the deadline.
    """
    np.random.seed(seed)
    rng = np.random.default_rng(seed)
    deadline = time.time() + budget_s - 0.5   # safety margin

    # ------------------------------------------------------------------
    # Length of the vector – default 4096, but honour a warm‑start length.
    warm_global = globals().get("height_sequence_1")
    if warm_global is not None:
        warm_arr = np.asarray(warm_global, dtype=np.float64)
        n = warm_arr.shape[0]
    else:
        n = 4096
        warm_arr = None

    conv_fft_len = _next_pow2(2 * n - 1)
    corr_fft_len = _next_pow2(3 * n - 2)

    # ------------------------------------------------------------------
    # Warm start processing
    warm = None
    if warm_arr is not None:
        warm = np.clip(warm_arr, 0.0, None)
        if warm.sum() >= 0.01:
            warm = warm / warm.sum()
        else:
            warm = None

    # ------------------------------------------------------------------
    # Build a diverse pool of start vectors
    pool = _initial_candidates(n, warm) + _spike_candidates(n)

    # Additional random Dirichlet draws with shuffled order
    for alpha in (0.1, 0.5, 1.0, 2.0):
        a = np.random.dirichlet(alpha * np.ones(n))
        a = a[np.random.permutation(n)]
        pool.append(a)

    # Random piece‑wise constant blocks
    for blocks in (8, 16, 32):
        block_len = n // blocks
        a = np.zeros(n, dtype=np.float64)
        for b in range(blocks):
            mass = np.random.rand()
            a[b * block_len:(b + 1) * block_len] = mass
        if a.sum() > 0:
            a = a / a.sum()
        pool.append(a)

    best_seq: np.ndarray | None = None
    best_val = float("inf")
    max_candidates = min(len(pool), 12)   # a handful of promising starts

    # ------------------------------------------------------------------
    # Phase I – explore a few seeds with random hyper‑parameters
    for idx in range(max_candidates):
        now = time.time()
        if now >= deadline:
            break

        # Allocate a slice of the remaining budget (≈ 30 s per seed)
        remaining = deadline - now
        seeds_left = max_candidates - idx
        per_seed = max(20.0, remaining / seeds_left)   # at least 20 s
        sub_dead = now + per_seed

        a0 = pool[idx]
        a0 = np.maximum(a0, 0.0)
        if a0.sum() < 0.01:
            a0 = np.random.dirichlet(np.ones(n))
        a0 = a0 / a0.sum()

        # Randomised hyper‑parameters for Adam
        lr = float(rng.uniform(0.03, 0.07))
        lam_init = float(rng.uniform(0.005, 0.2))
        lam_factor = float(rng.uniform(1.01, 1.04))
        max_it_adam = int(350_000 * rng.uniform(0.7, 1.3))

        a_adam, val_adam = _adam_optimize_custom(
            a0, n, conv_fft_len, corr_fft_len, sub_dead,
            lr=lr, lam_init=lam_init, lam_factor=lam_factor,
            max_iters=max_it_adam,
        )
        if val_adam < best_val:
            best_val = val_adam
            best_seq = a_adam.copy()
            print(f"[Explore {idx}] Adam improved to {val_adam:.6f}")

        # Sub‑gradient phase (if time permits)
        now = time.time()
        if now < deadline - 5:
            eta0 = float(rng.uniform(0.5, 2.0))
            max_it_sub = int(250_000 * rng.uniform(0.6, 1.2))
            a_sub, val_sub = _subgradient_optimize_custom(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                eta0=eta0, max_iters=max_it_sub,
            )
            if val_sub < best_val:
                best_val = val_sub
                best_seq = a_sub.copy()
                print(f"[Explore {idx}] Subgrad improved to {val_sub:.6f}")

        # Spectral‑flattening stage
        now = time.time()
        if now < deadline - 5:
            max_it_spec = int(200_000 * rng.uniform(0.5, 1.0))
            a_spec, val_spec = _spectral_flatten_optimize_custom(
                best_seq, n, conv_fft_len, deadline, max_iters=max_it_spec,
            )
            if val_spec < best_val:
                best_val = val_spec
                best_seq = a_spec.copy()
                print(f"[Explore {idx}] Spectral improved to {val_spec:.6f}")

        # Projected GD
        now = time.time()
        if now < deadline - 10:
            max_it_gd = int(150_000 * rng.uniform(0.5, 1.0))
            a_gd, val_gd = _max_conv_descent_custom(
                best_seq, n, conv_fft_len, corr_fft_len, deadline,
                max_iters=max_it_gd,
            )
            if val_gd < best_val:
                best_val = val_gd
                best_seq = a_gd.copy()
                print(f"[Explore {idx}] GD improved to {val_gd:.6f}")

        # Simulated annealing refinement
        now = time.time()
        if now < deadline - 5:
            max_moves = int(80_000 * rng.uniform(0.5, 1.0))
            a_ann, val_ann = _anneal_optimize_custom(
                best_seq, n, conv_fft_len, deadline, max_moves=max_moves,
            )
            if val_ann < best_val:
                best_val = val_ann
                best_seq = a_ann.copy()
                print(f"[Explore {idx}] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Phase II – final long‑run refinement on the current best
    now = time.time()
    if best_seq is not None and now < deadline - 30:
        print("[Refine] Starting prolonged refinement phase")
        # Long Adam run
        a_ref, val_ref = _adam_optimize_custom(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            lr=0.045, lam_init=0.015, lam_factor=1.02,
            max_iters=800_000,
        )
        if val_ref < best_val:
            best_val = val_ref
            best_seq = a_ref.copy()
            print(f"[Refine] Adam improved to {val_ref:.6f}")

        # Longer spectral flattening
        a_spec, val_spec = _spectral_flatten_optimize_custom(
            best_seq, n, conv_fft_len, deadline, max_iters=400_000,
        )
        if val_spec < best_val:
            best_val = val_spec
            best_seq = a_spec.copy()
            print(f"[Refine] Spectral improved to {val_spec:.6f}")

        # Short but focused GD
        a_gd, val_gd = _max_conv_descent_custom(
            best_seq, n, conv_fft_len, corr_fft_len, deadline,
            max_iters=200_000,
        )
        if val_gd < best_val:
            best_val = val_gd
            best_seq = a_gd.copy()
            print(f"[Refine] GD improved to {val_gd:.6f}")

        # Final annealing
        a_ann, val_ann = _anneal_optimize_custom(
            best_seq, n, conv_fft_len, deadline, max_moves=120_000,
        )
        if val_ann < best_val:
            best_val = val_ann
            best_seq = a_ann.copy()
            print(f"[Refine] Anneal improved to {val_ann:.6f}")

    # ------------------------------------------------------------------
    # Fallback – ensure a non‑empty return
    if best_seq is None:
        best_seq = np.full(n, 1.0 / n, dtype=np.float64)

    # Clip to the evaluator’s allowed range
    best_seq = np.clip(best_seq, 0.0, 1000.0)

    print(f"Final best value {best_val:.6f} (length {len(best_seq)})")
    return best_seq.tolist()
