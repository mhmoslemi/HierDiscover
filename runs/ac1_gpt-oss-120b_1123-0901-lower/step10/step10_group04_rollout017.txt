def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:

    rev_a = a[::-1]

    fw = np.fft.rfft(w, fft_len)

    fr = np.fft.rfft(rev_a, fft_len)

    corr = np.fft.irfft(fw * fr, fft_len)

    start = a.shape[0] - 1

    return corr[start:start + a.shape[0]]

def _adam_optimize(a0, n, fft_len, deadline, max_iters=500_000, lr=0.07, beta_start=0.5, beta_factor=1.005, beta_max=5e5, lambda_l2=0.15, decay_lr_every=None, decay_factor=0.9):

  eps=1e-15

  logits = np.log(a0 + eps)  # shape n

  m = np.zeros_like(logits)

  v = np.zeros_like(logits)

  beta1, beta2, eps_adam = 0.9, 0.999, 1e-8

  # target average conv
  avg_conv = 1.0/(2 * n - 1)

  best_a = a0.copy()
  best_val = _objective(a0, n)

  beta = beta_start

  it = 0

  while time.time() < deadline and it < max_iters:

        # compute a = softmax(logits)
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # conv
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        # Evaluate objective for tracking
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it}] new best {val:.6f}")

        # compute smoothmax weights
        # to avoid overflow, subtract max_c
        exp_term = np.exp(beta * (conv - max_c))
        w_sum = exp_term.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0/conv.size)
        else:
            w = exp_term / w_sum

        # gradient parts
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)
        diff = conv - avg_conv
        grad_l2 = 2.0 * _cross_correlation(diff, a, fft_len)
        grad = grad_smooth + lambda_l2 * grad_l2

        # compute gradient w.r.t. logits (mirror descent)
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # Adam update
        m = beta1 * m + (1 - beta1) * grad_log
        v = beta2 * v + (1 - beta2) * (grad_log ** 2)
        m_hat = m / (1 - beta1 ** (it + 1))
        v_hat = v / (1 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # update beta
        if beta < beta_max:
            beta *= beta_factor

        # optional learning rate decay
        if decay_lr_every and it % decay_lr_every == 0 and it > 0:
            lr *= decay_factor

        it += 1

  return best_a, best_val

def _subgradient_optimize_expgrad(a0, n, fft_len, deadline, max_iters=2_000_000, eta0=1.0):

    a = np.clip(a0, 0.0, None)
    s = a.sum()
    if s < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    else:
        a = a / s

    best_a = a.copy()
    best_val = _objective(a, n)
    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it}] new best {val:.6f}")

        # active indices
        # using tolerance maybe 1e-12 relative tolerance not needed because conv values small; use absolute tolerance.
        # Accept maybe top indices: we can select indices where conv >= max_c - tolerance.
        tolerance = 1e-12
        active = np.isclose(conv, max_c, atol=tolerance)
        if not active.any():
            # if none (should not happen) fallback to uniform weight
            w = np.zeros_like(conv)
        else:
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        grad = 2.0 * _cross_correlation(w, a, fft_len)
        # adaptive step size
        eta = eta0 / np.sqrt(it + 1.0)
        a = a * np.exp(-eta * grad)
        # renorm
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s
        it += 1

    return best_a, best_val

def _anneal_optimize(a0,n,fft_len,deadline,max_moves=200_000):

   a = np.clip(a0,0.0,None)
   s = a.sum()
   if s < 1e-12:
       a = np.full(n,1.0/n)
   else:
       a = a / s

   conv = _full_convolution(a, fft_len)
   val = _objective(a,n)

   best_a = a.copy()
   best_val = val

   T_start=0.12; T_end=5e-5
   it=0
   while time.time()<deadline and it<max_moves:
       T = max(T_end, T_start * (1 - it/float(max_moves)))
       i=np.random.randint(n)
       if a[i] < 1e-16:
           it+=1
           continue
       j=np.random.randint(n)
       if i==j:
           it+=1
           continue
       frac=np.random.rand()*0.35
       delta = a[i]*frac
       a2=a.copy()
       a2[i]-=delta
       a2[j]+=delta
       conv2=_full_convolution(a2,fft_len)
       val2=_objective(a2,n)
       if val2<best_val:
           best_val=val2
           best_a=a2.copy()
           a=a2
           conv=conv2
           val=val2
       else:
           if np.random.rand()<np.exp(-(val2-val)/T):
               a=a2
               conv=conv2
               val=val2
       it+=1
   return best_a,best_val

"""
Optimiser for a non‑negative step‑function sequence.

The routine builds a small pool of diverse start points (uniform,
parameteric families, random Dirichlet draws and the optional global
`height_sequence_1`).  From the best start it runs a long Adam optimiser
on a smooth‑max + L2‑flattening surrogate, then refines the result with a
pure exponentiated‑gradient sub‑gradient descent on the true max‑norm.
If time is left it performs several cheap perturbation‑subgradient
cycles and a final simulated‑annealing fine‑tune.  All steps respect the
overall 1000 s budget and the function always returns a valid
non‑negative list of coefficients.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
# Basic FFT utilities ---------------------------------------------------

def _next_pow2(x: int) -> int:
    """Smallest power of two ≥ x."""
    return 1 << (x - 1).bit_length()


def _full_convolution(a: np.ndarray, fft_len: int) -> np.ndarray:
    """Linear convolution a*a using FFT (length 2·len(a)-1)."""
    fa = np.fft.rfft(a, fft_len)
    conv = np.fft.irfft(fa * fa, fft_len)
    return conv[: 2 * a.shape[0] - 1]


def _cross_correlation(w: np.ndarray, a: np.ndarray, fft_len: int) -> np.ndarray:
    """
    Cross‑correlation w ★ a needed for gradients.
    Returns a vector of length len(a).
    """
    rev_a = a[::-1]
    fw = np.fft.rfft(w, fft_len)
    fr = np.fft.rfft(rev_a, fft_len)
    corr = np.fft.irfft(fw * fr, fft_len)
    start = a.shape[0] - 1
    return corr[start: start + a.shape[0]]


# ----------------------------------------------------------------------
# Exact objective --------------------------------------------------------

def _objective(a: np.ndarray, n: int) -> float:
    """Exact evaluation 2·n·max(conv)/(sum a)² (no clipping)."""
    conv = _full_convolution(a, _next_pow2(2 * n - 1))
    max_c = conv.max()
    s = a.sum()
    if s < 1e-12:
        return np.inf
    return float(2.0 * n * max_c / (s * s))


# ----------------------------------------------------------------------
# Candidate families -----------------------------------------------------

def _candidate_power(p: float, n: int) -> np.ndarray:
    """Symmetric power‑law shape a_i ∝ (i+1)^p + (n‑i)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p + (n - i) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_exponential(alpha: float, n: int) -> np.ndarray:
    """Symmetric exponential a_i ∝ exp(-α·i) + exp(-α·(n‑1‑i))."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i) + np.exp(-alpha * (n - 1 - i))
    a /= a.sum()
    return a


def _candidate_stretched(alpha: float, beta: float, n: int) -> np.ndarray:
    """Symmetric stretched‑exponential."""
    i = np.arange(n, dtype=np.float64)
    t = i / (n - 1)
    a = np.exp(-alpha * (t ** beta)) + np.exp(-alpha * ((1 - t) ** beta))
    a /= a.sum()
    return a


def _candidate_beta(a_par: float, b_par: float, n: int) -> np.ndarray:
    """Beta‑distribution shape on [0,1]; a_par,b_par>0."""
    x = (np.arange(n, dtype=np.float64) + 0.5) / n
    a = np.where(x > 0, x ** (a_par - 1), 0.0)
    b = np.where(1 - x > 0, (1 - x) ** (b_par - 1), 0.0)
    a = a * b
    a = np.nan_to_num(a, nan=0.0, posinf=0.0, neginf=0.0)
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_double_gaussian(sig: float, n: int) -> np.ndarray:
    """Two mirrored Gaussians anchored at the ends."""
    t = np.arange(n, dtype=np.float64) / (n - 1)
    a = np.exp(-0.5 * (t / sig) ** 2) + np.exp(-0.5 * ((1 - t) / sig) ** 2)
    a /= a.sum()
    return a


def _candidate_left_exponential(alpha: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ exp(-α·i) (no mirroring)."""
    i = np.arange(n, dtype=np.float64)
    a = np.exp(-alpha * i)
    a /= a.sum()
    return a


def _candidate_left_power(p: float, n: int) -> np.ndarray:
    """Monotone decreasing a_i ∝ (i+1)^p."""
    i = np.arange(n, dtype=np.float64)
    a = (i + 1) ** p
    a = np.clip(a, 0.0, None)
    s = a.sum()
    if s == 0:
        a = np.full(n, 1.0 / n)
    else:
        a /= s
    return a


def _candidate_spike(k: int, n: int, random: bool = True) -> np.ndarray:
    """k spikes (equal mass or Dirichlet) on a grid of length n."""
    a = np.zeros(n, dtype=np.float64)
    if k > 0:
        if random:
            pos = np.random.choice(n, size=k, replace=False)
        else:
            pos = np.round(np.linspace(0, n - 1, k)).astype(int)
        masses = np.random.dirichlet(np.ones(k))
        a[pos] = masses
    else:
        a[:] = 1.0 / n
    a /= a.sum()
    return a


# ----------------------------------------------------------------------
# Adam optimiser on smooth‑max + L2 regularisation ---------------------

def _adam_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 800_000,
    lr: float = 0.07,
    beta_start: float = 0.3,
    beta_factor: float = 1.007,
    beta_max: float = 5e5,
    lambda_l2: float = 0.2,
    decay_lr_every: int = None,
    decay_factor: float = 0.9,
) -> tuple[np.ndarray, float]:
    """
    Adam optimiser on the surrogate
        smooth‑max(conv) + λ·‖conv−avg‖₂² .
    Parameters are chosen for a fast yet reliable descent.
    Returns the best vector encountered (simplex‑feasible) and its
    exact objective value.
    """
    eps = 1e-15
    logits = np.log(a0 + eps)               # soft‑max parameters
    m = np.zeros_like(logits)
    v = np.zeros_like(logits)

    beta1, beta2, eps_adam = 0.9, 0.999, 1e-8
    avg_conv = 1.0 / (2 * n - 1)

    best_a = a0.copy()
    best_val = _objective(a0, n)

    beta = beta_start
    it = 0
    while time.time() < deadline and it < max_iters:
        # ----- simplex projection via soft‑max -----
        mx = logits.max()
        exp_log = np.exp(logits - mx)
        a = exp_log / exp_log.sum()

        # ----- exact convolution (for tracking) -----
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = _objective(a, n)
        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 5000 == 0:
                print(f"[Adam it={it:06d}] new best {val:.6f}")

        # ----- smooth‑max weights (stable formulation) -----
        # w_i ∝ exp(β·(conv_i−max_c))
        exp_term = np.exp(beta * (conv - max_c))
        w_sum = exp_term.sum()
        if w_sum == 0.0 or np.isnan(w_sum):
            w = np.full_like(conv, 1.0 / conv.size)
        else:
            w = exp_term / w_sum

        # ----- gradient of smooth‑max part -----
        grad_smooth = 2.0 * _cross_correlation(w, a, fft_len)

        # ----- L₂ regularisation towards the uniform average -----
        diff = conv - avg_conv
        grad_l2 = 2.0 * _cross_correlation(diff, a, fft_len)

        # combined gradient w.r.t. a
        grad = grad_smooth + lambda_l2 * grad_l2

        # ----- mirror‑descent (soft‑max) chain rule -----
        avg = np.dot(a, grad)
        grad_log = a * (grad - avg)

        # ----- Adam update on logits -----
        m = beta1 * m + (1.0 - beta1) * grad_log
        v = beta2 * v + (1.0 - beta2) * (grad_log ** 2)
        m_hat = m / (1.0 - beta1 ** (it + 1))
        v_hat = v / (1.0 - beta2 ** (it + 1))

        step = lr * m_hat / (np.sqrt(v_hat) + eps_adam)
        logits -= step

        # ----- temperature schedule -----
        if beta < beta_max:
            beta *= beta_factor

        # optional learning‑rate decay
        if decay_lr_every and it % decay_lr_every == 0 and it > 0:
            lr *= decay_factor

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Exact‑max exponentiated‑gradient sub‑gradient -------------------------

def _subgradient_optimize_expgrad(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_iters: int = 5_000_000,
    eta0: float = 1.0,
) -> tuple[np.ndarray, float]:
    """
    Exponentiated‑gradient sub‑gradient descent on the true max‑norm.
    Returns the best vector encountered and its exact objective.
    """
    a = np.maximum(a0, 0.0)
    if a.sum() < 1e-12:
        a = np.random.dirichlet(np.ones(n))
    a /= a.sum()

    best_a = a.copy()
    best_val = _objective(a, n)

    it = 0
    while time.time() < deadline and it < max_iters:
        conv = _full_convolution(a, fft_len)
        max_c = conv.max()
        val = _objective(a, n)

        if val < best_val:
            best_val = val
            best_a = a.copy()
            if it % 200_000 == 0:
                print(f"[Subgrad it={it:07d}] new best {val:.6f}")

        # active set of maximal indices (tight constraints)
        active = np.isclose(conv, max_c, atol=1e-12, rtol=0.0)
        if active.any():
            w = np.zeros_like(conv)
            w[active] = 1.0 / active.sum()
        else:
            w = np.zeros_like(conv)

        grad = 2.0 * _cross_correlation(w, a, fft_len)

        eta = eta0 / np.sqrt(it + 1.0)

        # exponentiated‑gradient update
        a = a * np.exp(-eta * grad)

        # renormalise (guard against drift)
        s = a.sum()
        if s < 1e-12 or np.isnan(s):
            a = np.random.dirichlet(np.ones(n))
            s = a.sum()
        a /= s

        it += 1

    return best_a, best_val


# ----------------------------------------------------------------------
# Simple simulated‑annealing fine‑tuner ----------------------------------

def _anneal_optimize(
    a0: np.ndarray,
    n: int,
    fft_len: int,
    deadline: float,
    max_moves: int = 200_000,
) -> tuple[np.ndarray, float]:
    """Cheap annealing that moves a fraction of mass between two positions."""
    a = a0.copy()
    conv = _full_convolution(a, fft_len)
    val = _objective(a, n)

    best_a = a.copy()
    best_val = val

    T_start = 0.12
    T_end = 5e-5
    it = 0
    while time.time() < deadline and it < max_moves:
        T = max(T_end, T_start * (1.0 - it / float(max_moves)))

        i = np.random.randint(n)
        if a[i] < 1e-16:
            it += 1
            continue
        j = np.random.randint(n)
        if i == j:
            it += 1
            continue

        frac = np.random.rand() * 0.35   # up to 35 % transfer
        delta = a[i] * frac
        a2 = a.copy()
        a2[i] -= delta
        a2[j] += delta

        conv2 = _full_convolution(a2, fft_len)
        val2 = _objective(a2, n)

        if val2 < best_val:
            best_val = val2
            best_a = a2.copy()
            a = a2
            conv = conv2
            val = val2
        else:
            if np.random.rand() < np.exp(-(val2 - val) / T):
                a = a2
                conv = conv2
                val = val2
        it += 1

    return best_a, best_val

        print(f"Final evaluated score (evaluate_sequence) = {final_score:.6f}")
    except Exception:
        final_score = None
        print("Final evaluated score could not be computed (missing evaluator).")

    return best_seq.tolist()
