import numpy as np

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a feasible packing of 26 circles inside the unit square.
    25 circles lie on a 5×5 regular grid with radius 0.1.
    One additional circle is placed at (0.2, 0.2) with the largest
    radius that does not intersect any grid circle.
    """
    # 5×5 grid points (0.1, 0.3, 0.5, 0.7, 0.9) in both directions
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    xv, yv = np.meshgrid(xs, ys)
    grid_centers = np.column_stack((xv.ravel(), yv.ravel()))   # shape (25, 2)

    # All grid circles can have radius 0.1
    grid_radii = np.full(grid_centers.shape[0], 0.1, dtype=np.float64)

    # Extra circle placed in the centre of the lower‑left cell of the grid
    extra_center = np.array([[0.2, 0.2]], dtype=np.float64)

    # Minimum distance from the extra centre to the four surrounding grid centres
    # (0.1,0.1), (0.3,0.1), (0.1,0.3), (0.3,0.3) is sqrt(0.1²+0.1²) = sqrt(0.02)
    min_dist = np.sqrt(0.02)
    extra_radius = max(min_dist - 0.1, 0.0)   # 0.1 is the radius of the neighbouring circles

    # Assemble the final arrays
    centers = np.vstack((grid_centers, extra_center))                 # (26, 2)
    radii = np.concatenate((grid_radii, np.array([extra_radius])))   # (26,)

    total_radius = float(np.sum(radii))

    return centers, radii, total_radius
