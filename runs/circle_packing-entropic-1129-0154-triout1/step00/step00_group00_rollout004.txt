```
def calc_optimal_radii(centers):
    n = centers.shape[0]
    # Number of constraints: 4n boundary + n*(n-1)/2 pair constraints
    rows = []
    vals = []
    b = []
    # Pair constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(np.linalg.norm(centers[i] - centers[j]))
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # x - r >= 0 => -r <= -x ??? Actually we want r <= x. So r <= x.
        # So row[i] = 1 <= x
        row = np.zeros(n); row[i] = 1
        rows.append(row); b.append(x)
        row = np.zeros(n); row[i] = 1
        rows.append(row); b.append(1 - x)
        row = np.zeros(n); row[i] = 1
        rows.append(row); b.append(y)
        row = np.zeros(n); row[i] = 1
        rows.append(row); b.append(1 - y)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)  # maximize sum r = minimize -sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: set radii to zeros
        return np.zeros(n), -np.inf
    radii = res.x
    sum_r = radii.sum()
    return radii, sum_r
```

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    # Build constraints
    # number of pair constraints
    num_pairs = n * (n - 1) // 2
    # total constraints = num_pairs + 4*n
    total_constraints = num_pairs + 4 * n
    A = np.zeros((total_constraints, n))
    b = np.zeros(total_constraints)
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1; b[row] = x; row += 1
        A[row, i] = 1; b[row] = 1 - x; row += 1
        A[row, i] = 1; b[row] = y; row += 1
        A[row, i] = 1; b[row] = 1 - y; row += 1
    # Solve LP
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = radii.sum()
    return radii, sum_r

```python
    n = centers.shape[0]
    # Pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    num_pairs = n * (n - 1) // 2
    total_constraints = num_pairs + 4 * n
    A = np.zeros((total_constraints, n), dtype=float)
    b = np.zeros(total_constraints, dtype=float)

    row = 0
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1.0
        b[row] = x
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        A[row, i] = 1.0
        b[row] = y
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    c = -np.ones(n, dtype=float)  # maximize sum r_i  → minimize -sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n, method='highs')
    if not res.success:
        # fallback: all radii zero
        radii = np.zeros(n, dtype=float)
    else:
        radii = res.x

    return radii, float(radii.sum())
```

The problem is a convex optimisation: for a fixed set of centre points the maximal feasible radii are obtained by the linear programme  

\[
\max\; \sum_i r_i\qquad
\text{s.t.}\quad 
\begin{cases}
r_i\ge 0\\[2pt]
r_i \le x_i,\; r_i \le 1-x_i,\; r_i \le y_i,\; r_i \le 1-y_i\\[2pt]
r_i+r_j \le \|c_i-c_j\|_2\qquad(i<j)
\end{cases}
\]

If the centre positions are well‑spread, the LP yields large radii and therefore a large total sum.  
Hence we:

1. **Choose an initial set of 26 points** – a hexagonal lattice (good for dense packing) together with a few purely random starts.  
2. **Solve the LP** for those points (SciPy’s `linprog` with the “highs’’ solver). The LP returns the optimal radii for the given centres.  
3. **Improve the point layout** by a simple hill‑climbing (randomly move one centre a little, resolve the LP, keep the move if the total radius sum grows).  
   The step size is gradually reduced; the process is repeated from several different starts.  
4. Keep the best layout ever seen, recompute its radii one last time and return the result.

The algorithm is deterministic (the random number generator is seeded) and uses only `numpy` and `scipy`, respecting the constraints of the task.

```python
import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers: np.ndarray):
    """
    For a fixed set of centre points solve the linear programme that maximises
    the sum of the radii while respecting non‑overlap and boundary constraints.
    Returns the radii array and its total sum.
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of constraints: one for each pair plus four per centre (the walls)
    n_pairs = n * (n - 1) // 2
    n_cons = n_pairs + 4 * n
    A = np.zeros((n_cons, n), dtype=float)
    b = np.zeros(n_cons, dtype=float)

    row = 0
    # r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # wall constraints: r_i <= x_i, 1 - x_i, y_i, 1 - y_i
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1.0
        b[row] = x
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        A[row, i] = 1.0
        b[row] = y
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=float)

    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        # a feasible (zero) solution always exists
        radii = np.zeros(n, dtype=float)
    else:
        radii = res.x

    return radii, float(radii.sum())


def generate_hex_grid(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a rough hexagonal lattice inside the unit square.
    The spacing is deliberately chosen to be a little smaller than the target
    distance, then the layout is refined by the hill‑climb.
    """
    # horizontal spacing – about twice the expected radius (≈0.09 → spacing ≈0.18)
    s = 0.18
    vert = np.sqrt(3.0) / 2.0 * s

    pts = []
    y = s / 2.0
    row = 0
    while y <= 1.0 - s / 2.0:
        offset = 0.0 if (row % 2 == 0) else s / 2.0
        x = s / 2.0 + offset
        while x <= 1.0 - s / 2.0:
            pts.append([x, y])
            x += s
        y += vert
        row += 1

    pts = np.asarray(pts, dtype=float)
    if pts.shape[0] < n:
        extra = rng.uniform(0.0, 1.0, size=(n - pts.shape[0], 2))
        pts = np.vstack([pts, extra])

    rng.shuffle(pts)
    return pts[:n].copy()


def generate_random_points(n: int, rng: np.random.Generator,
                           margin: float = 0.0) -> np.ndarray:
    """Uniform random points, optionally keeping a margin from the walls."""
    if margin <= 0.0:
        return rng.uniform(0.0, 1.0, size=(n, 2))
    return rng.uniform(margin, 1.0 - margin, size=(n, 2))


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main entry point required by the task.
    Returns (centres, radii, total_sum_of_radii) for 26 circles.
    """
    rng = np.random.default_rng(0)          # deterministic randomness
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ------------------------------------------------------------------
    #   1. generate a handful of diverse starting configurations
    # ------------------------------------------------------------------
    starts = []
    starts.append(generate_hex_grid(n, rng))
    for _ in range(4):
        starts.append(generate_random_points(n, rng))

    # ------------------------------------------------------------------
    #   2. hill‑climbing optimisation for each start
    # ------------------------------------------------------------------
    for start in starts:
        # a small jitter avoids perfectly symmetric configurations
        centres = start + rng.uniform(-0.02, 0.02, size=start.shape)
        centres = np.clip(centres, 0.0, 1.0)

        radii, cur_sum = compute_optimal_radii(centres)
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centres.copy()
            best_radii = radii.copy()

        step = 0.10                                   # initial move size
        for it in range(1500):
            # pick a circle at random and propose a small move
            idx = rng.integers(n)
            new_c = best_centers[idx] + rng.uniform(-step, step, size=2)
            new_c = np.clip(new_c, 0.0, 1.0)

            trial_centres = best_centers.copy()
            trial_centres[idx] = new_c

            trial_radii, trial_sum = compute_optimal_radii(trial_centres)

            # keep the move only if it improves the total radius sum
            if trial_sum > best_sum:
                best_sum = trial_sum
                best_centers = trial_centres
                best_radii = trial_radii

            # slowly shrink the step to fine‑tune the layout
            if (it + 1) % 300 == 0:
                step *= 0.7

    # ------------------------------------------------------------------
    #   3. final LP evaluation for the best geometry found
    # ------------------------------------------------------------------
    final_radii, final_sum = compute_optimal_radii(best_centers)

    # ensure the returned objects have the exact required shapes / dtype
    best_centers = np.asarray(best_centers, dtype=float)
    final_radii = np.asarray(final_radii, dtype=float)

    return best_centers, final_radii, final_sum
```
