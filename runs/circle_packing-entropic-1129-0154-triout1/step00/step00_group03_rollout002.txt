def solve_lp(centers):
    n = centers.shape[0]
    # Compute boundary constraints
    x = centers[:,0]
    y = centers[:,1]
    b_boundary = np.minimum.reduce([x, 1-x, y, 1-y])
    # Build A_ub and b_ub
    # Number of constraints: n (boundary) + n*(n-1)//2 (pairs)
    m = n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    # Fill boundary constraints
    for i in range(n):
        A[i, i] = 1.0
        b[i] = b_boundary[i]
    # Pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            # compute Euclidean distance
            dx = centers[i,0] - centers[j,0]
            dy = centers[i,1] - centers[j,1]
            d = np.hypot(dx, dy)
            b[idx] = d
            idx += 1
    c = -np.ones(n)  # maximize sum => minimize negative sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: set radii to zeros
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = np.maximum(res.x, 0.0)  # clip negative
        sum_r = np.sum(radii)
    return radii, sum_r
```

    from scipy.optimize import linprog
    from scipy.spatial.distance import cdist
    n = 26

        # compute boundary limits
        x = centers[:,0]
        y = centers[:,1]
        boundary = np.minimum.reduce([x, 1-x, y, 1-y])
        # distances
        dmat = cdist(centers, centers, metric='euclidean')
        # pair constraints
        n = centers.shape[0]
        m = n + n*(n-1)//2
        A = np.zeros((m, n))

        A[np.arange(n), np.arange(n)] = 1.0

        res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
        if not res.success:
            return None, -np.inf
        radii = np.maximum(res.x, 0.0)
        # clip radii to boundary just in case
        radii = np.minimum(radii, boundary)
        total = np.sum(radii)

def solve_radii(centers):
    n = centers.shape[0]
    # boundary limits
    x = centers[:,0]
    y = centers[:,1]
    boundary = np.minimum.reduce([x, 1-x, y, 1-y])
    # distances
    dmat = cdist(centers, centers, metric='euclidean')
    # number constraints
    n_boundary = n
    n_pairs = n*(n-1)//2
    m = n_boundary + n_pairs
    # allocate A_ub and b_ub
    A = np.zeros((m, n))
    b = np.zeros(m)
    # boundary constraints
    A[np.arange(n), np.arange(n)] = 1.0
    b[:n] = boundary
    # pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dmat[i, j]
            idx += 1
    # objective
    c = -np.ones(n)
    # call linprog
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
        return radii, 0.0
    radii = np.maximum(res.x, 0.0)
    # Clip to boundary just to be safe
    radii = np.minimum(radii, boundary)
    total = float(np.sum(radii))
    return radii, total
```

def generate_grid(nx, ny, margin):
    if nx == 1:
        xs = np.array([0.5])
    else:
        xs = margin + np.linspace(0, 1 - 2*margin, nx)
    if ny == 1:
        ys = np.array([0.5])
    else:
        ys = margin + np.linspace(0, 1 - 2*margin, ny)
    xs_grid, ys_grid = np.meshgrid(xs, ys)
    centers = np.vstack([xs_grid.ravel(), ys_grid.ravel()]).T
    return centers

def generate_hex_grid(n_points, margin):
    import numpy as np
    # tentative spacing: we aim for about 0.17-0.2 to be able to fit many points
    # We'll use a base spacing that yields enough points
    spacing = 0.18
    dx = spacing
    dy = spacing * np.sqrt(3) / 2.0
    points = []
    y = margin
    row = 0
    while y <= 1 - margin + 1e-12:
        offset = (dx / 2.0) if (row % 2 == 1) else 0.0
        x = margin + offset
        while x <= 1 - margin + 1e-12:
            points.append((x, y))
            x += dx
        row += 1
        y += dy
    points = np.array(points)
    if points.shape[0] >= n_points:
        # random subset
        idx = np.random.choice(points.shape[0], n_points, replace=False)
        return points[idx]
    else:
        # if not enough, fill with random points
        extra_needed = n_points - points.shape[0]
        extra = np.random.rand(extra_needed, 2) * (1 - 2*margin) + margin
        return np.vstack([points, extra])

import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

# ------------------------------------------------------------
# Linear programming: given centre positions, compute optimal radii
# ------------------------------------------------------------
def solve_radii(centers: np.ndarray):
    """
    Solve the LP:
        max  sum r_i
        s.t. 0 <= r_i
             r_i <= distance to each side of the unit square
             r_i + r_j <= distance(i,j)   for all i < j
    Returns the radii (np.ndarray) and the sum of radii (float).
    """
    n = centers.shape[0]
    # distance to the four sides
    x = centers[:, 0]
    y = centers[:, 1]
    boundary = np.minimum.reduce([x, 1 - x, y, 1 - y])

    # pairwise distances
    dmat = cdist(centers, centers, metric='euclidean')

    # Build inequality matrix A_ub * r <= b_ub
    n_pair = n * (n - 1) // 2
    m = n + n_pair                     # total number of inequalities
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    # boundary constraints r_i <= boundary_i
    A[np.arange(n), np.arange(n)] = 1.0
    b[:n] = boundary

    # pair constraints r_i + r_j <= distance_ij
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dmat[i, j]
            idx += 1

    # Objective: maximize sum r_i  <=>  minimize -sum r_i
    c = -np.ones(n, dtype=np.float64)

    # Solve with the HiGHS interior‑point method (fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)] * n,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback: zero radii (should rarely happen)
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)               # enforce non‑negativity
        radii = np.minimum(radii, boundary)         # safety clip to boundaries
        total = float(np.sum(radii))

    return radii, total

def generate_grid(nx: int, ny: int, margin: float) -> np.ndarray:
    """Regular nx × ny grid with a margin from the square border."""
    if nx == 1:
        xs = np.array([0.5])
    else:
        xs = margin + np.linspace(0.0, 1.0 - 2.0 * margin, nx)
    if ny == 1:
        ys = np.array([0.5])
    else:
        ys = margin + np.linspace(0.0, 1.0 - 2.0 * margin, ny)

    xs_grid, ys_grid = np.meshgrid(xs, ys)
    points = np.vstack([xs_grid.ravel(), ys_grid.ravel()]).T
    return points


def generate_hex_grid(num_points: int, margin: float) -> np.ndarray:
    """
    Produce a (potentially larger) set of points on a hexagonal lattice,
    then keep a random subset of exactly `num_points` points.
    """
    # a spacing that usually yields enough points inside the square
    spacing = 0.18
    dx = spacing
    dy = spacing * np.sqrt(3.0) / 2.0

    pts = []
    y = margin
    row = 0
    while y <= 1.0 - margin + 1e-12:
        offset = (dx / 2.0) if (row % 2 == 1) else 0.0
        x = margin + offset
        while x <= 1.0 - margin + 1e-12:
            pts.append((x, y))
            x += dx
        row += 1
        y += dy

    pts = np.asarray(pts, dtype=np.float64)

    if pts.shape[0] >= num_points:
        # random subset without replacement
        chosen = np.random.choice(pts.shape[0], num_points, replace=False)
        return pts[chosen]
    else:
        # not enough points – fill the rest with uniform random points
        extra_needed = num_points - pts.shape[0]
        extra = np.random.rand(extra_needed, 2) * (1.0 - 2.0 * margin) + margin
        return np.vstack([pts, extra])
