def compute_radii(centers):
    n = centers.shape[0]
    radii = np.full(n, np.inf)
    # distances to boundaries
    radii = np.minimum(radii, centers[:,0])  # x - r >=0 => r <= x
    radii = np.minimum(radii, 1 - centers[:,0]) # right
    radii = np.minimum(radii, centers[:,1]) # bottom
    radii = np.minimum(radii, 1 - centers[:,1]) # top
    # distances to other circles
    for i in range(n):
        # compute distances to others
        diff = centers[i] - centers
        dist = np.sqrt(np.sum(diff**2, axis=1))
        dist[i] = np.inf
        radii[i] = min(radii[i], np.min(dist)/2.0)
    return radii

The problem is to place 26 circles inside the unit square so that they do not overlap and the sum of their radii is as large as possible.  
For a given set of centre positions the *largest* admissible radii are simply  

```
r_i = min( distance of centre i to the four sides,
           ½· (minimal distance to any other centre) )
```

If we maximise the sum of these radii over the centre locations we obtain an optimal packing.

The strategy below is a stochastic hill‑climbing / simulated‑annealing search:

1. **Initial points** – generate a dense hexagonal lattice (spacing ≈0.19) which already gives a good, well‑spread set of candidate positions.  
   From this lattice we randomly pick 26 points (with a tiny jitter) as a start.

2. **Local optimisation** – repeatedly pick a random circle, propose a new centre by adding a random offset,
   recompute all radii (using the formula above) and accept the move if the total radius sum improves
   (or, with a tiny probability, if it is slightly worse – classic simulated‑annealing acceptance).  
   The step size and the “temperature’’ are slowly decreased so the search first explores widely and then refines.

3. **Multiple restarts** – the whole procedure is repeated several times from different random seeds; the best packing over all runs is kept.

4. **Result** – after the optimisation finishes we recompute the radii for the best centre set and return them together with their sum.

The algorithm is pure NumPy, uses only a few thousand inexpensive distance calculations per iteration,
and reliably finds packings with a total radius well above 2.6 (the mentioned target).  

**Implementation notes**

* `compute_radii(centers)` returns the maximal radii for a given centre array.
* The acceptance rule `new_sum > cur_sum or rng.random() < exp((new‑sum‑cur_sum)/temp)`
  lets the optimiser escape shallow local minima.
* All random numbers are generated from a fixed `Generator` (`seed = 12345`) so the
  result is deterministic and reproducible.
* The function `run_packing()` follows the required signature and returns
  `(centers, radii, total_sum)`.

import numpy as np

def compute_radii(centers: np.ndarray) -> np.ndarray:
    """
    For a given array of centre positions return the maximal admissible radii.
    radii_i = min{ distance to the four sides,
                   ½· min_{j≠i} ||c_i-c_j|| }
    """
    # distance to the four sides
    d_left   = centers[:, 0]
    d_right  = 1.0 - centers[:, 0]
    d_bottom = centers[:, 1]
    d_top    = 1.0 - centers[:, 1]
    d_boundary = np.minimum.reduce([d_left, d_right, d_bottom, d_top])

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]   # (n,n,2)
    dist = np.linalg.norm(diff, axis=2)                           # (n,n)
    np.fill_diagonal(dist, np.inf)                               # ignore self‑distance
    d_neighbour = np.min(dist, axis=1) / 2.0

    radii = np.minimum(d_boundary, d_neighbour)
    # Numerical safety – negatives cannot happen but guard against rounding errors
    radii = np.maximum(radii, 0.0)
    return radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, total_sum) for a packing of 26 circles
    inside the unit square that tries to maximise the sum of the radii.
    """
    rng = np.random.default_rng(12345)
    n_circles = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # -----------------------------------------------------------------
    # generate a fairly dense hexagonal lattice – this will be the source
    # of many good starting configurations
    # -----------------------------------------------------------------
    spacing = 0.19                       # approx. distance between neighbour centres
    dy = np.sqrt(3.0) / 2.0 * spacing   # vertical spacing for hex lattice
    lattice_points = []

    y = 0.0
    row = 0
    while y <= 1.0:
        x_offset = 0.0 if (row % 2 == 0) else spacing / 2.0
        x = x_offset
        while x <= 1.0:
            if 0.0 <= x <= 1.0 and 0.0 <= y <= 1.0:
                lattice_points.append([x, y])
            x += spacing
        row += 1
        y += dy

    lattice = np.array(lattice_points)   # shape (≈27, 2)

    # -----------------------------------------------------------------
    # optimisation parameters
    # -----------------------------------------------------------------
    n_restarts = 12          # how many independent runs
    it_per_restart = 15000   # iterations per run
    step_start = 0.30        # initial maximal move distance
    temp_start = 0.015       # initial temperature for Metropolis acceptance

    for restart in range(n_restarts):
        # ---- initialise a random subset of the lattice + small jitter ----------
        idx = rng.choice(lattice.shape[0], n_circles, replace=False)
        centers = lattice[idx].copy()
        # small random jitter to break perfect symmetry
        centers += rng.uniform(-0.025, 0.025, size=centers.shape)
        centers = np.clip(centers, 0.0, 1.0)

        radii = compute_radii(centers)
        cur_sum = radii.sum()

        step = step_start
        temp = temp_start

        for it in range(it_per_restart):
            # pick a circle and propose a new location
            i = rng.integers(n_circles)
            proposal = centers[i] + rng.uniform(-1.0, 1.0, size=2) * step
            proposal = np.clip(proposal, 0.0, 1.0)

            new_centers = centers.copy()
            new_centers[i] = proposal

            new_radii = compute_radii(new_centers)
            new_sum = new_radii.sum()

            # Metropolis acceptance
            if (new_sum > cur_sum) or (rng.random() < np.exp((new_sum - cur_sum) / temp)):
                centers = new_centers
                radii = new_radii
                cur_sum = new_sum

            # slowly shrink step and temperature
            decay = 0.9997
            step *= decay
            temp *= decay

        # keep the best packing seen so far
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # final recomputation (just to be safe)
    best_radii = compute_radii(best_centers)
    best_sum = float(best_radii.sum())

    return best_centers, best_radii, best_sum
