def generate_hex_grid(d):
    points = []
    dy = d * math.sqrt(3) / 2
    y = 0.0
    row = 0
    while y <= 1.0:
        offset = 0.0 if row % 2 == 0 else d/2
        x = offset
        while x <= 1.0:
            points.append((x, y))
            x += d
        row += 1
        y += dy
    return np.array(points)

import numpy as np
import math
from scipy.optimize import linprog
from typing import Tuple

# ------------------- Linear programming for radii ------------------- #
def solve_lp(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises the sum of radii for a fixed set of centres.

    Returns an array of radii (length = number of centres).
    """
    n = centers.shape[0]

    # Objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    # Bounds for each radius: r_i >= 0, no explicit upper bound (handled by constraints)
    bounds = [(0.0, None) for _ in range(n)]

    # Build inequality matrix A_ub * r <= b_ub
    rows = []
    rhs = []

    # Boundary constraints: r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        x, y = centers[i]
        e = np.zeros(n)
        e[i] = 1.0
        rows.append(e.copy()); rhs.append(x)          # r_i <= x_i
        rows.append(e.copy()); rhs.append(1.0 - x)    # r_i <= 1 - x_i
        rows.append(e.copy()); rhs.append(y)          # r_i <= y_i
        rows.append(e.copy()); rhs.append(1.0 - y)    # r_i <= 1 - y_i

    # Pairwise non‑overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=np.float64)

    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero (should never happen)
        return np.zeros(n)
    return res.x

def generate_grid(m: int = 9) -> np.ndarray:
    """
    Regular m×m grid inside the open square (0,1)².
    Points are placed at (i+1)/(m+2) to keep a small margin from the border.
    """
    coords = [(i + 1) / (m + 2) for i in range(m)]
    pts = [(x, y) for x in coords for y in coords]
    return np.array(pts, dtype=np.float64)


def generate_random(n: int = 200, eps: float = 1e-3) -> np.ndarray:
    """
    Uniform random points inside [eps, 1‑eps]².
    """
    rng = np.random.default_rng(0)  # deterministic seed
    pts = rng.random((n, 2)) * (1.0 - 2 * eps) + eps
    return pts


# ------------------- Farthest‑point sampling ------------------- #
def farthest_point_sampling(pool: np.ndarray, k: int) -> np.ndarray:
    """
    Choose k points from *pool* by greedy farthest‑point sampling.
    Starts from the point closest to the centre (0.5,0.5).
    """
    n_pool = pool.shape[0]
    selected_idx = []

    # first point – the one nearest to the centre
    centre = np.array([0.5, 0.5])
    d0 = np.linalg.norm(pool - centre, axis=1)
    first = int(np.argmin(d0))
    selected_idx.append(first)

    # remaining points
    remaining = set(range(n_pool))
    remaining.remove(first)

    while len(selected_idx) < k:
        # for each remaining candidate compute its distance to the nearest selected point
        best_idx = None
        best_dist = -1.0
        for idx in remaining:
            d = np.min(np.linalg.norm(pool[idx] - pool[selected_idx], axis=1))
            if d > best_dist:
                best_dist = d
                best_idx = idx
        selected_idx.append(best_idx)
        remaining.remove(best_idx)

    return pool[selected_idx, :]


# ------------------- Greedy pruning to exactly 26 circles ------------------- #
def prune_to_k(centers: np.ndarray, k: int) -> Tuple[np.ndarray, np.ndarray]:
    """
    Repeatedly solve the LP and drop the circle with the smallest radius
    until exactly k circles remain.
    """
    cur_centers = centers.copy()
    while cur_centers.shape[0] > k:
        radii = solve_lp(cur_centers)
        # index of the smallest radius (ties broken arbitrarily)
        remove_idx = int(np.argmin(radii))
        mask = np.arange(cur_centers.shape[0]) != remove_idx
        cur_centers = cur_centers[mask, :]
    final_radii = solve_lp(cur_centers)
    return cur_centers, final_radii


# ------------------- Local swap improvement ------------------- #
def improve_by_swaps(
    current_centers: np.ndarray,
    current_radii: np.ndarray,
    pool: np.ndarray,
    max_iter: int = 80,
) -> Tuple[np.ndarray, np.ndarray]:
    """
    Try to improve the solution by swapping one selected centre with a
    point from the original pool. Keep the swap if the total radius grows.
    """
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = best_radii.sum()

    # keep a set of indices that are currently used for fast membership tests
    used = {tuple(p) for p in best_centers}
    pool_list = [tuple(p) for p in pool]

    rng = np.random.default_rng(1)  # deterministic but different from the first generator
    for _ in range(max_iter):
        # choose a random point to remove
        out_idx = rng.integers(len(best_centers))
        out_point = tuple(best_centers[out_idx])

        # pick a random candidate not already used
        cand_idx = rng.integers(len(pool_list))
        in_point = pool_list[cand_idx]
        if in_point in used:

        new_points = best_centers.tolist()
        new_points[out_idx] = np.array(in_point)
        new_points_arr = np.array(new_points, dtype=np.float64)

        new_radii = solve_lp(new_points_arr)
        new_sum = new_radii.sum()

        if new_sum > best_sum + 1e-8:  # accept only clear improvements
            best_sum = new_sum
            best_centers = new_points_arr
            best_radii = new_radii
            used.remove(out_point)
            used.add(in_point)

    return best_centers, best_radii

    centres_26, radii_26 = improve_by_swaps(centres_26, radii_26, pool, max_iter=120)
