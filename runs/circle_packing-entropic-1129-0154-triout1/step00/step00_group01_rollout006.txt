def solve_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.linalg.norm(diff, axis=2)  # shape (n,n)
    # set up constraints:

    m_pair = n*(n-1)//2
    m_boundary = 4*n
    m = m_pair + m_boundary

    A = np.zeros((m, n))
    b = np.zeros(m)

    row = 0
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = d[i, j]
            row += 1
    # boundary constraints
    for i in range(n):
        x = centers[i,0]
        y = centers[i,1]
        # left: r_i <= x
        A[row, i] = 1
        b[row] = x
        row+=1
        # right: r_i <= 1-x
        A[row, i] = 1
        b[row] = 1-x
        row+=1
        # bottom: r_i <= y
        A[row, i] = 1
        b[row] = y
        row+=1
        # top: r_i <= 1-y
        A[row, i] = 1
        b[row] = 1-y
        row+=1

    c = -np.ones(n) # maximize sum; minimise -sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success: return None maybe.

    radii = res.x
    sum_r = np.sum(radii)

    return radii, sum_r

def solve_radii(centers):
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub
    m_pair = n*(n-1)//2
    m_boundary = 4*n
    total_constraints = m_pair + m_boundary
    A = np.zeros((total_constraints, n))
    b = np.empty(total_constraints)
    row = 0
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = d[i, j]
            row += 1
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left
        A[row, i] = 1
        b[row] = x
        row+=1
        # right
        A[row, i] = 1
        b[row] = 1 - x
        row+=1
        # bottom
        A[row, i] = 1
        b[row] = y
        row+=1
        # top
        A[row, i] = 1
        b[row] = 1 - y
        row+=1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
    else:
        radii = res.x
    # Apply small safety margin
    radii = np.maximum(radii - 1e-10, 0.0)
    sum_r = radii.sum()
    return radii, sum_r

import numpy as np
from scipy.optimize import linprog

def _solve_radii(centers):
    """Solve the linear program for maximum sum of radii for given centers.

    Returns radii (np.ndarray) and sum.
    """
    n = centers.shape[0]
    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff * diff, axis=2))
    # Build constraints
    m_pair = n * (n - 1) // 2
    m_boundary = 4 * n
    total_constraints = m_pair + m_boundary
    A = np.zeros((total_constraints, n), dtype=float)
    b = np.empty(total_constraints, dtype=float)
    row = 0
    # pair constraints r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d[i, j]
            row += 1
    # boundary constraints r_i <= x_i, 1 - x_i, y_i, 1 - y_i
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1.0
        b[row] = x            # left
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x      # right
        row += 1
        A[row, i] = 1.0
        b[row] = y            # bottom
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y      # top
        row += 1
    # objective: maximize sum r_i -> minimize -sum r_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n
    # Solve linear program with HiGHS (fast, reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii (should not happen)
        radii = np.zeros(n, dtype=float)
    else:
        radii = res.x
    # tiny safety margin to respect strict inequalities in validator
    radii = np.maximum(radii - 1e-10, 0.0)
    return radii, float(radii.sum())

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Search for a packing of 26 circles inside the unit square that maximises
    the sum of radii.  The routine generates many random point sets,
    solves a linear program for each set and keeps the best feasible solution.
    The returned triple is (centers, radii, sum_of_radii).
    """

    target = 2.636
    rng = np.random.default_rng(20231129)   # deterministic seed
    best_sum = -1.0
    best_centers = None
    best_radii = None

        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()
            if best_sum >= target:

* inside the square                                                        0 ≤ *c*ᵢˣ ± *r*ᵢ ≤ 1, 0 ≤ *c*ᵢʸ ± *r*ᵢ ≤ 1

If the centre positions are fixed, the radii appear only in linear
inequalities – therefore for a fixed set of centres the optimal radii are
the solution of a **linear program** (LP).  
The idea is:

1. generate many different centre configurations,
2. solve the LP for each configuration,
3. keep the configuration with the largest sum of radii.

Because the LP is tiny (26 variables, ≤ 429 constraints) it can be solved

The LP is solved with SciPy’s HiGHS backend – fast and numerically

solution is safely inside the tolerance used by the validator.

The algorithm stops as soon as a packing with a sum of radii at least the
target value `2.636` is found; otherwise it returns the best packing after
the allotted number of trials.

import numpy as np
from scipy.optimize import linprog

def _solve_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the LP that maximises the sum of radii for the given centre
    positions.  Returns the optimal radii and their sum.
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff * diff, axis=2))

    # number of constraints
    m_pair = n * (n - 1) // 2          # r_i + r_j <= d_ij
    m_boundary = 4 * n                # distance to the four sides
    m = m_pair + m_boundary

    A = np.zeros((m, n), dtype=float)
    b = np.empty(m, dtype=float)

    # ---- pairwise constraints ------------------------------------------------
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d[i, j]
            row += 1

    # ---- boundary constraints -------------------------------------------------
    for i in range(n):
        x, y = centers[i]
        # left side   : r_i <= x
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # right side  : r_i <= 1 - x
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # bottom side : r_i <= y
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # top side    : r_i <= 1 - y
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None)] * n

    # solve with HiGHS (fast, reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # a feasible solution always exists (all radii = 0)
        radii = np.zeros(n, dtype=float)
    else:
        radii = res.x

    # safety margin – guarantees strict inequality for the validator
    radii = np.maximum(radii - 1e-10, 0.0)

    return radii, float(radii.sum())


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Search for a packing of 26 circles that maximises the sum of radii.
    The routine draws many random centre configurations, solves the LP
    for each, and returns the best feasible packing found.
    """
    n = 26
    target = 2.636                     # early‑stop threshold
    rng = np.random.default_rng(20231129)  # deterministic seed

    best_sum = -1.0
    best_centers = None
    best_radii = None

    max_iterations = 3000
    for _ in range(max_iterations):
        # uniformly random points inside the unit square
        centres = rng.random((n, 2))

        radii, s = _solve_radii(centres)

        if s > best_sum:
            best_sum = s
            best_centers = centres.copy()
            best_radii = radii.copy()
            if best_sum >= target:
                break   # sufficient quality reached

    # In the extremely unlikely event that no trial succeeded,
    # fall back to a simple 5×5 grid (25 circles) plus a tiny extra one.
    if best_centers is None:
        grid = np.array([[0.1 + 0.2 * i, 0.1 + 0.2 * j]
                         for i in range(5) for j in range(5)])
        extra = np.array([[0.5, 0.5]])          # will be shrunk to a very small circle
        best_centers = np.vstack([grid, extra])
        best_radii, best_sum = _solve_radii(best_centers)

    return best_centers, best_radii, float(best_sum)
