def prune_to_n(centers, target_n):
    cur_centers = centers.copy()
    while cur_centers.shape[0] > target_n:
        radii = solve_lp(cur_centers)
        # find smallest radius

def hill_climb(centers, radii, rng, max_iter=500):
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = radii.sum()
    n = centers.shape[0]
    step = 0.05
    for _ in range(max_iter):
        i = rng.integers(n)
        delta = (rng.random(2) - 0.5) * 2 * step
        new_center = best_centers[i] + delta
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not identical to another center to avoid redundancy (optional)
        new_centers = best_centers.copy()
        new_centers[i] = new_center
        new_radii = solve_lp(new_centers)
        new_sum = new_radii.sum()
        if new_sum > best_sum + 1e-8:
            best_sum = new_sum
            best_centers = new_centers
            best_radii = new_radii
    return best_centers, best_radii, best_sum

import numpy as np
from scipy.optimize import linprog


def _solve_lp(centers: np.ndarray) -> np.ndarray:
    """
    For a given set of centres solve the linear programme that maximises the sum
    of radii under the non‑overlap and boundary constraints.
    Returns the optimal radii (non‑negative, may contain zeros).
    """
    n = centers.shape[0]
    if n == 0:
        return np.empty((0,))

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # Build A_ub * r <= b_ub  for all i < j : r_i + r_j <= d_ij
    m = n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1

    # bounds for each radius: 0 <= r_i <= distance to closest side
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max_r))

    # maximise sum r_i  ->  minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fall back to zero radii – should never happen with a feasible set
        return np.zeros(n, dtype=np.float64)

    radii = res.x
    # Clip tiny negative values caused by numerical noise
    radii[radii < 0.0] = 0.0
    return radii


def _generate_candidates(rng: np.random.RandomState, total: int = 80) -> np.ndarray:
    """
    Create a pool of candidate centres.
    Four deterministic corner circles are always present;
    the remaining points are random inside the square.
    """
    # four large corner circles (they can reach radius 0.25)
    corner_pts = np.array(
        [[0.25, 0.25], [0.75, 0.25], [0.25, 0.75], [0.75, 0.75]], dtype=np.float64
    )
    need = max(total - corner_pts.shape[0], 0)
    random_pts = rng.uniform(0.0, 1.0, size=(need, 2))
    return np.vstack([corner_pts, random_pts])


def _prune_to_n(centers: np.ndarray, target_n: int = 26) -> (np.ndarray, np.ndarray):
    """
    Greedy removal of the circle with the smallest radius until exactly
    target_n centres remain.
    """
    cur = centers.copy()
    while cur.shape[0] > target_n:
        radii = _solve_lp(cur)
        # index of the smallest radius (if several are equal, remove the first)
        worst = int(np.argmin(radii))
        cur = np.delete(cur, worst, axis=0)
    final_radii = _solve_lp(cur)
    return cur, final_radii


def _hill_climb(
    centers: np.ndarray,
    radii: np.ndarray,
    rng: np.random.RandomState,
    max_iter: int = 2000,
    step: float = 0.05,
) -> (np.ndarray, np.ndarray, float):
    """
    Simple deterministic hill‑climbing: move a random centre by a small
    amount and keep the move if the total sum of radii improves.
    """
    best_c = centers.copy()
    best_r = radii.copy()
    best_sum = best_r.sum()
    n = best_c.shape[0]

    for _ in range(max_iter):
        i = rng.integers(n)  # which centre to perturb
        delta = (rng.random(2) - 0.5) * 2.0 * step
        new_pos = best_c[i] + delta
        new_pos = np.clip(new_pos, 0.0, 1.0)

        # avoid creating an exact duplicate (which would force both radii to zero)
        if np.any(np.all(np.isclose(best_c, new_pos, atol=1e-12), axis=1)):
            continue

        trial_c = best_c.copy()
        trial_c[i] = new_pos
        trial_r = _solve_lp(trial_c)
        trial_sum = trial_r.sum()

        if trial_sum > best_sum + 1e-9:
            best_c = trial_c
            best_r = trial_r
            best_sum = trial_sum

    return best_c, best_r, best_sum


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a packing of exactly 26 circles inside the unit square
    that maximises the sum of radii.
    Returns (centres, radii, sum_of_radii).
    """
    rng = np.random.RandomState(123456)  # deterministic seed
    target_n = 26
    best_total = -1.0
    best_centres = None
    best_radii = None

    # Several independent trials increase robustness
    for trial in range(8):
        # 1) generate a generous candidate pool
        cand = _generate_candidates(rng, total=80)

        # 2) greedy pruning to exactly 26 circles
        pruned_c, pruned_r = _prune_to_n(cand, target_n=target_n)

        # 3) optional local optimisation
        opt_c, opt_r, opt_sum = _hill_climb(pruned_c, pruned_r, rng, max_iter=1500)

        if opt_sum > best_total:
            best_total = opt_sum
            best_centres = opt_c
            best_radii = opt_r

    # Final safety – make sure radii are non‑negative and slightly shrink
    # them to stay clear of the 1e‑12 tolerance used in validation.
    best_radii = np.maximum(best_radii - 1e-10, 0.0)
    best_total = float(best_radii.sum())

    return best_centres, best_radii, best_total
