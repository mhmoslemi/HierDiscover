def solve_radii(centers):
    N = len(centers)
    # Build A_ub and b_ub
    constraints = []
    b = []
    # boundary
    for i in range(N):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(N)
        row[i] = 1
        constraints.append(row)
        b.append(x)
        # r_i <= 1 - x
        row = np.zeros(N)
        row[i] = 1
        constraints.append(row)
        b.append(1 - x)
        # r_i <= y
        row = np.zeros(N)
        row[i] = 1
        constraints.append(row)
        b.append(y)
        # r_i <= 1 - y
        row = np.zeros(N)
        row[i] = 1
        constraints.append(row)
        b.append(1 - y)

    # overlap constraints
    for i in range(N):
        for j in range(i+1, N):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(N)
            row[i] = 1
            row[j] = 1
            constraints.append(row)
            b.append(d)

    A_ub = np.vstack(constraints)
    b_ub = np.array(b)
    c = -np.ones(N)  # maximize sum => minimize -sum

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if not res.success:
        # fallback: assign zero radii
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

import numpy as np
from scipy.optimize import linprog

def generate_hex_lattice(r):
    """
    Generate a hexagonal lattice of points inside the unit square.
    The lattice uses spacing 2*r horizontally and sqrt(3)*r vertically.
    """
    points = []
    row = 0
    y = r
    while y <= 1.0 - r + 1e-12:
        if row % 2 == 0:
            x_start = r
        else:
            x_start = 2.0 * r
        x = x_start
        while x <= 1.0 - r + 1e-12:
            points.append([x, y])
            x += 2.0 * r
        row += 1
        y = r + row * np.sqrt(3.0) * r
    return np.array(points)


def solve_radii(centers):
    """
    For a fixed set of centres solve the linear program that maximises
    the sum of radii under the non‑overlap and boundary constraints.
    Returns an array of radii (may contain zeros).
    """
    n = centers.shape[0]
    # Build inequality matrix A_ub * r <= b_ub
    rows = []
    bounds_rhs = []

    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        for b in (x, 1.0 - x, y, 1.0 - y):
            a = np.zeros(n)
            a[i] = 1.0
            rows.append(a)
            bounds_rhs.append(b)

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            a = np.zeros(n)
            a[i] = 1.0
            a[j] = 1.0
            rows.append(a)
            bounds_rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(bounds_rhs)

    # maximise sum(r)  ==> minimise -sum(r)
    c = -np.ones(n)

    # radii must be non‑negative
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fallback – all radii zero (still a valid packing)
        return np.zeros(n)
    return res.x


def hill_climb(initial_centers, max_iter=1500, seed=0):
    """
    Simple stochastic hill‑climber that repeatedly perturbs a random centre,
    resolves the LP for the radii and keeps the move if it improves the
    total sum of radii.
    """
    rng = np.random.default_rng(seed)
    centers = initial_centers.copy()
    radii = solve_radii(centers)
    best_sum = radii.sum()
    best_centers = centers.copy()
    best_radii = radii.copy()

    n = centers.shape[0]
    step = 0.12  # initial maximal perturbation

    for it in range(max_iter):
        # gradually shrink the step size
        cur_step = step * (0.99 ** (it // 100))

        i = rng.integers(n)
        new_center = centers[i] + (rng.random(2) - 0.5) * 2.0 * cur_step
        new_center = np.clip(new_center, 0.0, 1.0)

        old = centers[i].copy()
        centers[i] = new_center

        new_radii = solve_radii(centers)
        new_sum = new_radii.sum()

        if new_sum > best_sum + 1e-9:
            best_sum = new_sum
            best_centers = centers.copy()
            best_radii = new_radii.copy()
        else:
            # revert the move
            centers[i] = old

    return best_centers, best_radii, float(best_sum)


def initialise_26_points():
    """
    Produce a deterministic set of 26 starting points.
    We start from a hexagonal lattice with spacing that yields
    at least 26 points and then keep a random subset of 26.
    """
    # spacing chosen so that the lattice contain ~30 points

    # ensure we have more than 26 candidates
    if cand.shape[0] < 26:
        # fallback to a simple regular grid
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        grid = np.array([[x, y] for y in ys for x in xs])
        cand = grid

    rng = np.random.default_rng(42)
    idx = rng.choice(cand.shape[0], size=26, replace=False)
    return cand[idx]

    # try a few independent hill‑climbs and keep the best result
    best_total = -1.0
    best_centers = None
    best_radii = None

    for run in range(3):
        centers, radii, total = hill_climb(init_centers, max_iter=1500, seed=run)

            best_centers = centers

    # final sanity – ensure the radii are non‑negative (tiny negatives may appear due
    # to numerical noise)
    best_radii = np.maximum(best_radii, 0.0)

    return best_centers, best_radii, float(best_total)
