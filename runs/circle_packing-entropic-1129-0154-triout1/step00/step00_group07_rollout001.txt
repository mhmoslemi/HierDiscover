import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # number of constraints
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    for i in range(n):
        xi, yi = centers[i]
        A[row, i] = 1
        b[row] = xi
        row += 1
        A[row, i] = 1
        b[row] = 1 - xi
        row += 1
        A[row, i] = 1
        b[row] = yi
        row += 1
        A[row, i] = 1
        b[row] = 1 - yi
        row += 1
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback to zero radii
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii

def run_packing():
    np.random.seed(0)
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None
    # Pre-generate many candidate sets

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # number of constraints
    m = 4 * n + n*(n-1)//2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)
    row = 0
    # boundary constraints
    for i in range(n):
        xi, yi = centers[i]
        # left
        A[row, i] = 1.0
        b[row] = xi
        row += 1
        # right
        A[row, i] = 1.0
        b[row] = 1.0 - xi
        row += 1
        # bottom
        A[row, i] = 1.0
        b[row] = yi
        row += 1
        # top
        A[row, i] = 1.0
        b[row] = 1.0 - yi
        row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1
    c = -np.ones(n, dtype=np.float64)  # maximize sum -> minimize negative sum
    bounds = [(0.0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: assign zero radii
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x
        # Clip to avoid tiny numerical violations
        radii = np.maximum(radii, 0.0)
        # Ensure within boundary limits
        rad_bound = np.minimum(centers[:,0], 1.0-centers[:,0])
        rad_bound = np.minimum(rad_bound, np.minimum(centers[:,1], 1.0-centers[:,1]))
        radii = np.minimum(radii, rad_bound)
    return radii

import numpy as np
from scipy.optimize import linprog

def _compute_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme that gives the maximal radii for a
    fixed set of centre positions.
    """
    n = centers.shape[0]
    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # total number of linear constraints:
    #   4 per circle (border) + one for each unordered pair
    m = 4 * n + n * (n - 1) // 2

    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    row = 0
    # border constraints
    for i in range(n):
        xi, yi = centers[i]
        # left
        A[row, i] = 1.0
        b[row] = xi
        row += 1
        # right
        A[row, i] = 1.0
        b[row] = 1.0 - xi
        row += 1
        # bottom
        A[row, i] = 1.0
        b[row] = yi
        row += 1
        # top
        A[row, i] = 1.0
        b[row] = 1.0 - yi
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  →  minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use the modern HiGHS solver (fast and robust)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero (still a feasible packing)
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x
        # Guard against tiny numerical violations
        radii = np.maximum(radii, 0.0)
        # also enforce the obvious boundary limits explicitly
        max_by_border = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
        max_by_border = np.minimum(max_by_border,
                                   np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
        radii = np.minimum(radii, max_by_border)

    return radii
