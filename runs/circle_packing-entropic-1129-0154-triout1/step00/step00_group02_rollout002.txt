import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # compute distances
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Upper triangular distances
    # Setup constraints
    # Each r_i <= x_i
    # each r_i <= 1 - x_i
    # similarly for y
    # Pair constraints
    A_ub = []
    b_ub = []
    for i in range(n):
        # r_i <= x_i
        a = np.zeros(n)
        a[i] = 1
        A_ub.append(a)
        b_ub.append(centers[i,0])
        # r_i <= 1 - x_i
        a = np.zeros(n)
        a[i] = 1
        A_ub.append(a)
        b_ub.append(1 - centers[i,0])
        # r_i <= y_i
        a = np.zeros(n)
        a[i] = 1
        A_ub.append(a)
        b_ub.append(centers[i,1])
        # r_i <= 1 - y_i
        a = np.zeros(n)
        a[i] = 1
        A_ub.append(a)
        b_ub.append(1 - centers[i,1])
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            a = np.zeros(n)
            a[i] = 1
            a[j] = 1
            A_ub.append(a)
            b_ub.append(dists[i,j])
    A = np.array(A_ub)
    b = np.array(b_ub)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # Return something
        raise RuntimeError("LP failed")
    radii = res.x
    sum_r = -res.fun
    return radii, sum_r

def compute_radii_lp(centers: np.ndarray) -> tuple[np.ndarray, float]:
    n = centers.shape[0]
    # preallocate A_ub and b_ub using lists
    # boundary constraints
    # pair constraints
    # Use numpy for speed?
    # We'll allocate potentially large arrays. Use list.

    # boundary constraints
    # For each i:
    # constraint r_i <= x_i
    # constraint r_i <= 1 - x_i
    # constraint r_i <= y_i
    # constraint r_i <= 1 - y_i

    # pair constraints:
    # for i<j: r_i + r_j <= d_ij

    # Build A_ub row by row
    A_rows = []
    b = []

    # boundary constraints
    x = centers[:,0]
    y = centers[:,1]
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        # r_i <= x_i
        A_rows.append(e.copy())
        b.append(x[i])
        # r_i <= 1 - x_i
        A_rows.append(e.copy())
        b.append(1.0 - x[i])
        # r_i <= y_i
        A_rows.append(e.copy())
        b.append(y[i])
        # r_i <= 1 - y_i
        A_rows.append(e.copy())
        b.append(1.0 - y[i])

    # distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            A_rows.append(e)
            b.append(dists[i,j])

    A_ub = np.vstack(A_rows)
    b_ub = np.array(b)

    c = -np.ones(n)  # minimize -sum = maximize sum

    bounds = [(0, None) for _ in range(n)]

    # Solve LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    radii = res.x
    sum_r = -res.fun
    return radii, sum_r

# Helper: compute optimal radii for a *fixed* set of centres

def compute_optimal_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the linear programme:
        maximise   sum r_i
        s.t.       r_i >= 0
                   r_i <= x_i,   r_i <= 1 - x_i
                   r_i <= y_i,   r_i <= 1 - y_i
                   r_i + r_j <= d_ij   for all i < j
    Returns the radii vector and the sum of radii.
    """
    n = centers.shape[0]
    x = centers[:, 0]
    y = centers[:, 1]

    # ----- boundary constraints (4 per circle) -----
    A_rows = []
    b = []

    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        # r_i <= x_i
        A_rows.append(e.copy())
        b.append(x[i])
        # r_i <= 1 - x_i
        A_rows.append(e.copy())
        b.append(1.0 - x[i])
        # r_i <= y_i
        A_rows.append(e.copy())
        b.append(y[i])
        # r_i <= 1 - y_i
        A_rows.append(e.copy())
        b.append(1.0 - y[i])

    # ----- pairwise non‑overlap constraints -----
    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    for i in range(n):
        for j in range(i + 1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            A_rows.append(e)
            b.append(dists[i, j])

    A_ub = np.vstack(A_rows)
    b_ub = np.array(b)

    # objective: maximise sum r_i  →  minimise -sum r_i
    c = -np.ones(n)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError(f"LP failed: {res.message}")

    radii = res.x
    sum_r = -res.fun
    return radii, float(sum_r)

# Helper: generate a set of points using farthest‑point sampling

def farthest_point_set(num_points: int,
                       candidates_per_step: int = 300,
                       rng: np.random.Generator = np.random.default_rng()) -> np.ndarray:
    """
    Build a set of `num_points` points inside [0,1]^2.
    At each step a batch of random candidates is drawn.
    The candidate with the largest
        score = min( distance to nearest existing point ,
                     distance to closest edge )
    is added.
    """
    pts = [rng.random(2)]                     # start with one random point
    while len(pts) < num_points:
        existing = np.array(pts)               # shape (k,2)
        # draw a batch of candidates
        cand = rng.random((candidates_per_step, 2))
        # distance to the nearest existing centre
        if existing.shape[0] == 0:
            min_dist = np.full(candidates_per_step, np.inf)
        else:
            d = np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2)
            min_dist = d.min(axis=1)           # (candidates_per_step,)
        # distance to the closest square edge
        edge_dist = np.minimum(np.minimum(cand[:, 0], 1.0 - cand[:, 0]),
                               np.minimum(cand[:, 1], 1.0 - cand[:, 1]))
        # score = min(min_dist, edge_dist)
        score = np.minimum(min_dist, edge_dist)
        best_idx = np.argmax(score)
        pts.append(cand[best_idx])
    return np.array(pts)
