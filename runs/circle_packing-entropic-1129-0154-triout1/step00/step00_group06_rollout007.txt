import numpy as np
from scipy.optimize import minimize

# Global number of circles – used by the constraint routine
N_GLOBAL = 26


def objective(v: np.ndarray) -> float:
    """
    Objective for the optimiser:  -sum(radii)
    (minimise the negative sum to obtain a maximisation)
    """
    n = N_GLOBAL
    radii = v[2 * n :]
    return -float(np.sum(radii))


def inequality_constraints(v: np.ndarray) -> np.ndarray:
    """
    Returns a 1‑D array containing all inequality constraints.
    For SLSQP a constraint is satisfied when g(v) ≥ 0.
    """
    n = N_GLOBAL
    xs = v[:n]
    ys = v[n : 2 * n]
    rs = v[2 * n :]

    cons = []

    # stay inside the square
    cons.append(xs - rs)          # x - r ≥ 0
    cons.append(1.0 - xs - rs)    # 1 - x - r ≥ 0
    cons.append(ys - rs)          # y - r ≥ 0
    cons.append(1.0 - ys - rs)    # 1 - y - r ≥ 0

    # non‑overlap: (dx^2+dy^2) - (ri+rj)^2 ≥ 0
    for i in range(n):
        for j in range(i + 1, n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            cons.append(dx * dx + dy * dy - (rs[i] + rs[j]) ** 2)

    return np.concatenate(cons)


def _initial_grid(slack: float = 0.08) -> tuple[np.ndarray, np.ndarray]:
    """
    Build a 5×5 grid (25 circles) plus one centre circle.
    All circles start with a uniform small radius (slack).
    """
    centers = []
    radii = []

    # 5×5 grid
    for i in range(5):
        for j in range(5):
            centers.append([0.1 + i * 0.2, 0.1 + j * 0.2])
            radii.append(slack)

    # 26th circle roughly in the centre
    centers.append([0.5, 0.5])
    radii.append(slack / 2.0)

    centers_arr = np.array(centers, dtype=float)
    radii_arr = np.array(radii, dtype=float)
    return centers_arr, radii_arr


def _initial_random(r_small: float = 0.02, seed: int = 0) -> tuple[np.ndarray, np.ndarray]:
    """
    Produce a completely random feasible layout with a very small
    uniform radius, guaranteeing a feasible start for the optimiser.
    """
    rng = np.random.default_rng(seed)
    xs = rng.random(N_GLOBAL)
    ys = rng.random(N_GLOBAL)
    centers = np.column_stack((xs, ys))
    radii = np.full(N_GLOBAL, r_small, dtype=float)
    return centers, radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Optimises the placement of 26 circles in the unit square
    with the goal of maximising the sum of the radii.
    Returns (centers, radii, total_sum_of_radii).
    """
    global N_GLOBAL
    n = N_GLOBAL

    start_points = []

    start_points.append(_initial_grid(slack=0.08))

    # Optimisation settings (common for all starts)

    bounds = [(0.0, 1.0)] * (2 * n) + [(0.0, 0.5)] * n

    best_centers = None
    best_radii = None
