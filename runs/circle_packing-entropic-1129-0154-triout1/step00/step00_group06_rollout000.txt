```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    n = 26
    rng = np.random.default_rng(42)

    best_sum = -1.0
    best_centers = None
    best_radii = None

    def solve_lp(centers):
        n = centers.shape[0]
        # boundary distances
        b = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
        # distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        # build constraints
        rows = []
        bounds = []
        # boundary constraints
        for i in range(n):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            bounds.append(b[i])
        # pairwise constraints
        for i in range(n):
            for j in range(i+1, n):
                row = np.zeros(n)
                row[i] = 1.0
                row[j] = 1.0
                rows.append(row)
                bounds.append(dists[i,j])
        A = np.vstack(rows)
        ub = np.array(bounds)
        c = -np.ones(n)
        var_bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=ub, bounds=var_bounds, method='highs')
        if not res.success:
            # in case of numeric issues, return zeros
            return np.zeros(n), 0.0
        radii = res.x
        return radii, radii.sum()
    # initial positions: maybe hex lattice pattern
    def init_positions():
        # We'll use hex pattern with spacing ~0.19 (slightly less than 0.2)
        # Determine #rows, columns to get 26 points.
        s = 0.19
        v = s * np.sqrt(3)/2
        positions = []
        y = 0.5 # will adjust later
        # We'll try to fill rows from bottom.
        y0 = s/2  # start near boundary: need margin to accommodate radii later.
        # Let's generate rows until we have enough points.
        rows = []
        y = y0
        row_idx = 0
        while len(positions) < n and y <= 1 - s/2:
            if row_idx % 2 == 0:
                # even row: start at x = s/2
                x_start = s/2
                num_cols = int(np.floor((1 - s) / s)) + 1
            else:
                x_start = s
                num_cols = int(np.floor((1 - 2*s) / s)) + 1
            for col in range(num_cols):
                x = x_start + col * s
                if x <= 1 - s/2 + 1e-9:
                    positions.append([x, y])
                if len(positions) >= n:
                    break
            row_idx += 1
            y += v
        # If not enough, fill random points
        while len(positions) < n:
            positions.append(rng.uniform(size=2))
        return np.array(positions[:n])
    # start with initial layout
    centers = init_positions()
    # maybe also random perturb starting points
    # set global best as solution of initial layout
    radii, sum_r = solve_lp(centers)
    best_sum = sum_r
    best_centers = centers.copy()
    best_radii = radii.copy()
    # hill climbing
    max_iters = 4000
    temperature0 = 0.1
    for it in range(max_iters):
        # temperature schedule
        temp = temperature0 * (1 - it / max_iters)
        # propose new centers
        new_centers = best_centers.copy()
        # pick random index
        i = rng.integers(n)
        # propose perturbation
        sigma = 0.05 * (1 - it / max_iters) + 0.005
        new_center = new_centers[i] + rng.normal(scale=sigma, size=2)
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        new_centers[i] = new_center
        # check minimal distance from other centers to avoid extremely close points (optional)
        # compute sum via LP
        radii_new, sum_new = solve_lp(new_centers)
        # acceptance
        if sum_new > best_sum:
            best_sum = sum_new
            best_centers = new_centers
            best_radii = radii_new
        else:

                prob = np.exp((sum_new - best_sum) / temp)
                if rng.random() < prob:
                    best_centers = new_centers
                    best_radii = radii_new
                    best_sum = sum_new
    # final compute radii for best centers (should already have)
    # Ensure they are feasible
    # Return
    return best_centers, best_radii, float(best_sum)
```

```
def build_constraints(centers):
    n = centers.shape[0]
    # boundary distances
    b = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # number of constraints
    m = n + n*(n-1)//2
    A = np.zeros((m, n))
    ub = np.empty(m)
    row = 0
    # boundary
    for i in range(n):
        A[row,i] = 1.0
        ub[row] = b[i]
        row += 1
    # pairwise

            A[row,i] = 1.0
            A[row,j] = 1.0
            ub[row] = np.linalg.norm(centers[i]-centers[j])
            row += 1
    return A, ub
```

```
def solve_lp_full(centers):
    A, ub = build_constraints(centers)
    n = centers.shape[0]
    c = -np.ones(n)
    bounds = [(0, None)]*n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # maybe fallback to zero radii
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = radii.sum()
    return radii, sum_r
```

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple

def _build_constraints(centers: np.ndarray):
    n = centers.shape[0]
    # distance to the four sides of the square
    left = centers[:, 0]
    right = 1.0 - centers[:, 0]
    bottom = centers[:, 1]
    top = 1.0 - centers[:, 1]
    bound_dist = np.minimum(np.minimum(left, right), np.minimum(bottom, top))

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    m = n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    ub = np.empty(m, dtype=np.float64)

    row = 0
    # boundary constraints: r_i <= bound_dist_i
    for i in range(n):
        A[row, i] = 1.0
        ub[row] = bound_dist[i]
        row += 1

    # non‑overlap constraints: r_i + r_j <= dist(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            ub[row] = dists[i, j]
            row += 1

    return A, ub

def _solve_for_radii(centers: np.ndarray):
    """Linear program that maximises the sum of the radii for a fixed set of centres."""
    n = centers.shape[0]
    A, ub = _build_constraints(centers)
    c = -np.ones(n)               # maximise sum(r)  <=> minimise -sum(r)
    bounds = [(0.0, None)] * n    # radii are non‑negative
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # if the solver fails we fall back to a zero vector – the caller will reject it
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x
    return radii, float(radii.sum())

def _hexagonal_initial(rng: np.random.Generator, n: int):
    """Return n points placed on a slightly compressed hexagonal lattice."""
    # spacing that would allow a regular 5x5 grid; we start with a little smaller value
    s = 0.19
    v = s * np.sqrt(3.0) / 2.0
    points = []
    y = s / 2.0
    row_id = 0
    while len(points) < n and y <= 1.0 - s / 2.0:
        if row_id % 2 == 0:
            x_start = s / 2.0
            max_x = 1.0 - s / 2.0
        else:
            x_start = s
            max_x = 1.0 - s
        x = x_start
        while x <= max_x + 1e-12:
            points.append([x, y])
            if len(points) >= n:
                break
            x += s
        row_id += 1
        y += v
    # if we still need points (because the lattice is too sparse) fill with random ones
    while len(points) < n:
        points.append(rng.uniform(size=2))
    pts = np.array(points[:n], dtype=np.float64)
    # add a tiny jitter so the optimiser does not start on a perfectly symmetric configuration
    pts += rng.uniform(-0.008, 0.008, size=pts.shape)
    np.clip(pts, 0.0, 1.0, out=pts)
    return pts

def _random_initial(rng: np.random.Generator, n: int):
    """Simple uniform random start."""
    pts = rng.uniform(size=(n, 2))
    return pts

def _anneal(initial: np.ndarray, rng: np.random.Generator,
            iters: int = 1200, temperature0: float = 0.2) -> Tuple[np.ndarray, np.ndarray, float]:
    """Simulated‑annealing/greedy search on the centre positions."""
    n = initial.shape[0]
    # current state
    curr_centers = initial.copy()
    curr_radii, curr_sum = _solve_for_radii(curr_centers)

    best_centers = curr_centers.copy()
    best_radii = curr_radii.copy()
    best_sum = curr_sum

    for it in range(iters):
        # linear cooling schedule for temperature and step size
        temp = temperature0 * (1.0 - it / iters)
        sigma = 0.2 * (1.0 - it / iters) + 0.005

        # propose moving a single circle
        i = rng.integers(n)
        proposal = curr_centers[i] + rng.normal(scale=sigma, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)
        new_centers = curr_centers.copy()
        new_centers[i] = proposal

        # evaluate the new configuration
        new_radii, new_sum = _solve_for_radii(new_centers)

        # Metropolis acceptance
        accept = False
        if new_sum > curr_sum:
            accept = True
        elif temp > 0.0:
            prob = np.exp((new_sum - curr_sum) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            curr_centers = new_centers
            curr_radii = new_radii
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

    return best_centers, best_radii, best_sum

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a packing of 26 circles inside the unit square.
    The function returns (centres, radii, sum_of_radii).
    """
    rng = np.random.default_rng(0)
    n = 26
    global_best_sum = -1.0
    global_best_centers = None
    global_best_radii = None

    # two different kinds of starts – one structured, the other random
    starts = [
        _hexagonal_initial(rng, n),
        _random_initial(rng, n)
    ]

    for start in starts:
        centres, radii, total = _anneal(start, rng, iters=1500, temperature0=0.3)
        if total > global_best_sum:
            global_best_sum = total
            global_best_centers = centres
            global_best_radii = radii

    # a last tiny refinement: run a short greedy pass from the current best configuration
    centres, radii, total = _anneal(global_best_centers, rng, iters=800, temperature0=0.1)
    if total > global_best_sum:
        global_best_sum = total
        global_best_centers = centres
        global_best_radii = radii

    # shrink radii a little to guarantee a strict margin for the validator (1e-12)
    eps = 1e-8
    final_radii = np.maximum(global_best_radii - eps, 0.0)
    final_sum = float(final_radii.sum())

    return np.asarray(global_best_centers, dtype=np.float64), final_radii, final_sum
```

**Idea**  
For a *fixed* set of centre positions the maximal radii can be found by a **linear program**:  

* variables – the radii `r_i` (i = 0 … 25)  
* objective – maximise `∑ r_i` (equivalently minimise `‑∑ r_i`)  
* constraints  
   * `r_i ≥ 0`  
   * `r_i ≤ distance from centre i to the nearest square side` (boundary constraint)  
   * `r_i + r_j ≤ Euclidean distance between centre i and centre j` (non‑overlap)  

All constraints are linear in the radii, therefore for any arrangement of the 26 centres we can compute the **optimal radii** instantly with `scipy.optimize.linprog`.

The remaining freedom is the placement of the centres.  
This is a non‑convex optimisation problem, but a simple stochastic hill‑climbing / simulated‑annealing scheme works well:

1. **Initial layout** – start from a slightly compressed hexagonal lattice (good spread) and also from a completely random layout.  
2. **Iterative move** – pick one circle at random, perturb its centre with a Gaussian step whose size shrinks during the run.  
3. **Evaluation** – solve the LP for the new centre set, obtaining a new total radius sum.  
4. **Acceptance** – always accept an improvement; otherwise accept with the Metropolis probability `exp((Δ)/T)` where the temperature `T` is linearly cooled.  

The algorithm is run several independent restarts; the best configuration over all restarts is kept.

Finally, to satisfy the validator’s strict “no‑overlap by 1 e‑12” rule we **shrink every radius by a tiny ε = 1 e‑8** – this guarantees a safe margin while barely affecting the objective value.

The whole procedure is deterministic (fixed RNG seed) and uses only top‑level helper functions, no lambdas, no I/O, and complies with the required signature `run_packing() → (centres, radii, sum)`.

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _build_constraints(centers: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
    """
    Construct the linear‑program constraints for a given set of centres.
    Returns the matrix A (<=) and the right‑hand side vector ub.
    """
    n = centers.shape[0]

    # distance to each of the four sides of the unit square
    left = centers[:, 0]
    right = 1.0 - centers[:, 0]
    bottom = centers[:, 1]
    top = 1.0 - centers[:, 1]
    bound_dist = np.minimum(np.minimum(left, right), np.minimum(bottom, top))

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    m = n + n * (n - 1) // 2          # total number of constraints
    A = np.zeros((m, n), dtype=np.float64)
    ub = np.empty(m, dtype=np.float64)

    row = 0
    # boundary constraints: r_i <= bound_dist_i
    for i in range(n):
        A[row, i] = 1.0
        ub[row] = bound_dist[i]
        row += 1

    # non‑overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            ub[row] = dists[i, j]
            row += 1

    return A, ub


def _solve_for_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the LP that maximises the sum of radii for the supplied centres.
    Returns the radii vector and the associated sum.
    """
    n = centers.shape[0]
    A, ub = _build_constraints(centers)
    c = -np.ones(n)                     # minimise -sum(r_i)
    bounds = [(0.0, None)] * n          # r_i ≥ 0
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback – should rarely happen
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x
    return radii, float(radii.sum())


def _hexagonal_initial(rng: np.random.Generator, n: int) -> np.ndarray:
    """
    Produce a slightly compressed hexagonal lattice with `n` points.
    A small random jitter is added to avoid perfect symmetry.
    """
    s = 0.19                                   # horizontal spacing
    v = s * np.sqrt(3.0) / 2.0                # vertical spacing
    points = []
    y = s / 2.0
    row_id = 0
    while len(points) < n and y <= 1.0 - s / 2.0:
        if row_id % 2 == 0:
            x_start = s / 2.0
            max_x = 1.0 - s / 2.0
        else:
            x_start = s
            max_x = 1.0 - s
        x = x_start
        while x <= max_x + 1e-12:
            points.append([x, y])
            if len(points) >= n:
                break
            x += s
        row_id += 1
        y += v

    while len(points) < n:                     # safety net
        points.append(rng.uniform(size=2))

    pts = np.array(points[:n], dtype=np.float64)

    # tiny jitter to break symmetry
    pts += rng.uniform(-0.008, 0.008, size=pts.shape)
    np.clip(pts, 0.0, 1.0, out=pts)
    return pts


def _random_initial(rng: np.random.Generator, n: int) -> np.ndarray:
    """Uniform random start."""
    return rng.uniform(size=(n, 2))


def _anneal(initial: np.ndarray,
            rng: np.random.Generator,
            iters: int = 1200,
            temperature0: float = 0.2) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Simulated‑annealing / greedy search on the centre positions.
    Returns the best centres, radii and the corresponding sum of radii.
    """
    n = initial.shape[0]

    curr_centers = initial.copy()
    curr_radii, curr_sum = _solve_for_radii(curr_centers)

    best_centers = curr_centers.copy()
    best_radii = curr_radii.copy()
    best_sum = curr_sum

    for it in range(iters):
        # linear cooling for temperature and step size
        temp = temperature0 * (1.0 - it / iters)
        sigma = 0.2 * (1.0 - it / iters) + 0.005

        # pick a circle and propose a move
        i = rng.integers(n)
        proposal = curr_centers[i] + rng.normal(scale=sigma, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centers = curr_centers.copy()
        new_centers[i] = proposal

        new_radii, new_sum = _solve_for_radii(new_centers)

        # Metropolis acceptance
        accept = False
        if new_sum > curr_sum:
            accept = True
        elif temp > 0.0:
            prob = np.exp((new_sum - curr_sum) / temp)
            if rng.random() < prob:
                accept = True

        if accept:
            curr_centers = new_centers
            curr_radii = new_radii
            curr_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

    return best_centers, best_radii, best_sum

    Returns (centres, radii, sum_of_radii).

    n = 26

    # store the globally best solution found
    global_best_sum = -1.0
    global_best_centers = None
    global_best_radii = None

    # two different starting strategies
    starts = [
        _hexagonal_initial(rng, n),
        _random_initial(rng, n)
    ]

    for start in starts:
        centres, radii, total = _anneal(start, rng, iters=1500, temperature0=0.3)
        if total > global_best_sum:
            global_best_sum = total
            global_best_centers = centres
            global_best_radii = radii

    # final short refinement from the best configuration obtained so far
    centres, radii, total = _anneal(global_best_centers, rng,
                                    iters=800, temperature0=0.1)
    if total > global_best_sum:
        global_best_sum = total
        global_best_centers = centres
        global_best_radii = radii

    # safety shrink to guarantee a strict non‑overlap margin for the validator
    eps = 1e-8
    final_radii = np.maximum(global_best_radii - eps, 0.0)
    final_sum = float(final_radii.sum())

    return (np.asarray(global_best_centers, dtype=np.float64),
            final_radii,
            final_sum)
```
