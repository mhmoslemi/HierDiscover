```
def solve_lp(pts):
    n = pts.shape[0]
    # distances
    d = np.linalg.norm(pts[:, None, :] - pts[None, :, :], axis=2)
    # Build A_ub rows:
    rows = []
    bounds = []
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            bounds.append(d[i, j])
    # boundary constraints
    x = pts[:,0]; y = pts[:,1]
    for i in range(n):
        # x_i - r_i >= 0  -> -x_i + r_i <= 0? Actually r_i <= x_i.
        row = np.zeros(n); row[i]=1
        rows.append(row); bounds.append(x[i])
        rows.append(row); bounds.append(1 - x[i]) # same row? Need separate rows; we must create separate rows.
        rows.append(row); bounds.append(y[i])
        rows.append(row); bounds.append(1 - y[i])
    A_ub = np.vstack(rows)
    b_ub = np.array(bounds)
    c = -np.ones(n)
    # bounds for variables
    var_bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=var_bounds, method='highs')
    if not res.success:
        # fallback: all zeros
        return np.zeros(n)
    return res.x
```

```
import numpy as np

from scipy.optimize import linprog

TARGET = 2.636

def solve_lp(pts):
    n = pts.shape[0]
    x = pts[:,0]; y = pts[:,1]
    # pairwise distances
    diff = pts[:, None, :] - pts[None, :, :]
    dist = np.sqrt(np.sum(diff**2, axis=2))
    # build constraints
    rows = []
    ub = []
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            ub.append(dist[i, j])
    # boundary constraints
    for i in range(n):
        row = np.zeros(n); row[i] = 1.0
        rows.append(row); ub.append(x[i])          # r <= x
        rows.append(row); ub.append(1.0 - x[i])    # r <= 1-x
        rows.append(row); ub.append(y[i])          # r <= y
        rows.append(row); ub.append(1.0 - y[i])    # r <= 1-y
    A_ub = np.vstack(rows)
    b_ub = np.array(ub)
    c = -np.ones(n)  # minimize -sum(r) => maximize sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback zeros
        return np.zeros(n)
    return res.x

                best_radii = radii.copy()
```

The task is to place 26 circles inside the unit square `[0,1]×[0,1]` and maximise the sum of their radii.
The constraints are linear in the radii *if the circle centres are fixed*:

* each radius is bounded by the distance to the four sides,
* for every pair of circles `i , j` we must have `r_i + r_j ≤ dist(i,j)`.

For a fixed set of centres the problem is therefore a **linear programme** (LP):
```
maximise   Σ r_i
subject to r_i ≥ 0
           r_i ≤ x_i ,  r_i ≤ 1−x_i
           r_i ≤ y_i ,  r_i ≤ 1−y_i
           r_i + r_j ≤ dist(i,j)   for all i<j
```
If we can find a set of 26 centre positions that allows large radii, solving the LP will give the optimal radii for those centres.

The algorithm below builds a set of 26 centre points, continuously improves them by a very small stochastic hill‑climber, and solves the LP after each move.  
The LP is solved with `scipy.optimize.linprog` (method *highs* – very fast for the problem size).  
Whenever a move improves the total sum of radii we keep it; otherwise we accept it with a tiny probability (simulated‑annealing style) to escape local minima.

The procedure is repeated with several random restarts until the time limit is reached or a sum larger than the required target `2.636` is obtained.

All helper functions are top‑level, there are no closures or filesystem I/O, and the returned arrays satisfy the validator exactly.

```python
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme for a *fixed* set of centre points
# ----------------------------------------------------------------------
def _solve_lp(pts: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises Σr_i for the given centre positions.

    Parameters
    ----------
    pts : np.ndarray (n,2)
        Circle centre coordinates.

    Returns
    -------
    np.ndarray (n,)
        Optimal radii (non‑negative, respecting all constraints).
    """
    n = pts.shape[0]
    x = pts[:, 0]
    y = pts[:, 1]

    # pairwise Euclidean distances
    diff = pts[:, None, :] - pts[None, :, :]
    d = np.sqrt(np.sum(diff * diff, axis=2))

    rows = []
    ub = []

    # pairwise non‑overlap constraints  r_i + r_j ≤ d_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            ub.append(d[i, j])

    # distance to the four sides  r_i ≤ x_i , 1‑x_i , y_i , 1‑y_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row); ub.append(x[i])          # ≤ left side
        rows.append(row); ub.append(1.0 - x[i])    # ≤ right side
        rows.append(row); ub.append(y[i])          # ≤ bottom side
        rows.append(row); ub.append(1.0 - y[i])    # ≤ top side

    A_ub = np.vstack(rows)
    b_ub = np.array(ub)

    # maximise Σr_i  ⇔  minimise -Σr_i
    c = -np.ones(n)

    # radii are non‑negative, no explicit upper bound (handled by A_ub)
    var_bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=var_bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – all radii zero (still a feasible solution)
        return np.zeros(n, dtype=np.float64)

    # numerical cleanup
    rad = np.maximum(res.x, 0.0)
    return rad


# ----------------------------------------------------------------------
# Hill‑climbing / simulated annealing on the centre positions
# ----------------------------------------------------------------------
def _hill_climb(initial_pts: np.ndarray,
                max_iter: int = 1500,
                step: float = 0.07) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Stochastic local optimisation of the centre positions.

    Parameters
    ----------
    initial_pts : np.ndarray (26,2)
        Starting centre coordinates.
    max_iter : int
        Number of optimisation steps.
    step : float
        Initial maximum translation applied to a randomly chosen centre.

    Returns
    -------
    centres, radii, total_sum
        The best configuration discovered during the climb.
    """
    n = initial_pts.shape[0]
    pts = initial_pts.copy()
    radii = _solve_lp(pts)
    cur_sum = float(radii.sum())

    best_pts = pts.copy()
    best_rad = radii.copy()
    best_sum = cur_sum

    temperature = 0.01
    for it in range(max_iter):
        # pick one circle and propose a small random move
        idx = np.random.randint(n)
        proposal = pts[idx] + np.random.uniform(-step, step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_pts = pts.copy()
        new_pts[idx] = proposal
        new_rad = _solve_lp(new_pts)
        new_sum = float(new_rad.sum())

        # Accept if better, otherwise accept with a tiny probability
        accept = False
        if new_sum > cur_sum:
            accept = True
        else:
            # simulated‑annealing acceptance
            prob = np.exp((new_sum - cur_sum) / max(temperature, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            pts = new_pts
            radii = new_rad
            cur_sum = new_sum
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_pts = pts.copy()
                best_rad = radii.copy()

        # annealing schedule
        temperature *= 0.995
        # slowly shrink the step size
        if (it + 1) % 200 == 0 and step > 0.005:
            step *= 0.9

    return best_pts, best_rad, best_sum


# ----------------------------------------------------------------------
# Generation of a reasonable starting configuration
# ----------------------------------------------------------------------
def _initial_configuration() -> np.ndarray:
    """
    Create an initial set of 26 points.
    25 points form a jittered 5×5 grid, the 26‑th point is placed at random.
    """
    pts = []
    for i in range(5):
        for j in range(5):
            # base grid point
            x = 0.1 + 0.2 * i
            y = 0.1 + 0.2 * j
            # add a small random jitter (kept inside the unit square)
            x += np.random.uniform(-0.03, 0.03)
            y += np.random.uniform(-0.03, 0.03)
            x = np.clip(x, 0.0, 1.0)
            y = np.clip(y, 0.0, 1.0)
            pts.append([x, y])
    # one extra random point
    pts.append([np.random.rand(), np.random.rand()])
    return np.array(pts, dtype=np.float64)


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Find a placement of 26 circles in the unit square that maximises
    the sum of their radii.

    Returns
    -------
    centres : np.ndarray (26,2)
        Circle centre coordinates.
    radii   : np.ndarray (26,)
        Radii of the circles (non‑negative, respect all constraints).
    total   : float
        Σ radii – the objective value.
    """
    TARGET = 2.636          # early‑stop target
    best_sum = -1.0
    best_centres = None
    best_radii = None

    start_time = time.time()
    TIME_LIMIT = 9.0        # seconds – enough for the optimiser, safe for the judge

    # Repeated random restarts until time runs out or target is reached
    while time.time() - start_time < TIME_LIMIT:
        init_pts = _initial_configuration()
        centres, radii, s = _hill_climb(init_pts, max_iter=1500, step=0.07)

        if s > best_sum:
            best_sum = s
            best_centres = centres
            best_radii = radii

        if best_sum >= TARGET:
            break

    # Final sanity check – tiny negative values are clamped to zero
    best_radii = np.maximum(best_radii, 0.0)

    total = float(best_radii.sum())
    return best_centres, best_radii, total
```
