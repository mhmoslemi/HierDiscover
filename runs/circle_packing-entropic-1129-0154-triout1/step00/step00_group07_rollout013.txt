def compute_optimal_radii(centers):
    n = centers.shape[0]
    # objective: minimize -sum(r)
    c = -np.ones(n)
    # constraints list
    A = []
    b = []
    # Wall constraints
    for i in range(n):
        x, y = centers[i]
        A.append(np.eye(1, n, i).reshape(-1))
        b.append(x)
        A.append(np.eye(1, n, i).reshape(-1))
        b.append(1 - x)
        A.append(np.eye(1, n, i).reshape(-1))
        b.append(y)
        A.append(np.eye(1, n, i).reshape(-1))
        b.append(1 - y)
    # neighbor constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(d)
    A = np.array(A)
    b = np.array(b)
    # bounds for each variable i: (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # if LP fails, fallback to zero radii
        radii=np.zeros(n)
    else:
        radii = res.x
    # optionally clip small negatives
    radii = np.maximum(radii, 0.0)
    return radii, np.sum(radii)
```

def optimize(initial_centers, max_iter=2000, init_temp=0.1, final_temp=1e-5):
    centers = initial_centers.copy()
    radii, current_sum = compute_optimal_radii(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = current_sum
    n = centers.shape[0]
    for t in range(max_iter):
        # temperature schedule linear or exponential
        temp = init_temp * ((final_temp / init_temp) ** (t / max_iter))
        i = np.random.randint(n)
        # propose move
        delta = np.random.normal(scale=0.05, size=2)
        new_center = centers[i] + delta
        new_center = np.clip(new_center, 0.0, 1.0)
        old_center = centers[i].copy()
        centers[i] = new_center
        new_radii, new_sum = compute_optimal_radii(centers)
        # delta in sum:
        delta_sum = new_sum - current_sum
        if delta_sum > 0 or np.exp(delta_sum / (temp+1e-12)) > np.random.rand():
            # accept
            radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # revert
            centers[i] = old_center
    return best_centers, best_radii, best_sum
```

def hex_initial(n=26, dx=0.18):
    dy = np.sqrt(3)/2 * dx
    centers = []
    row = 0
    y = 0.0
    while y <= 1 and len(centers) < n*2:  # generate more than needed
        offset = (row % 2) * dx/2
        x = 0.0
        while x <= 1:
            cx = x + offset
            if cx <= 1:
                centers.append([cx, y])
            x += dx
        row += 1
        y += dy
    arr = np.array(centers)
    # Filter points inside [0,1]
    arr = arr[(arr[:,0] >= 0) & (arr[:,0] <= 1) & (arr[:,1] >= 0) & (arr[:,1] <= 1)]
    # Randomly choose n points
    idx = np.random.choice(len(arr), size=n, replace=False)
    return arr[idx]

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear program: given centres, compute optimal radii that maximise sum(r)
# ----------------------------------------------------------------------
def compute_optimal_radii(centres: np.ndarray):
    """
    Solve the LP:
        maximise   sum_i r_i
        subject to r_i >= 0,
                   r_i <= x_i, 1-x_i, y_i, 1-y_i,
                   r_i + r_j <= dist(i,j)   for all i<j
    Returns radii (np.ndarray) and the sum.
    """
    n = centres.shape[0]
    # Objective: minimise -sum(r_i)
    c = -np.ones(n)

    # Build inequality matrix A_ub @ r <= b_ub
    rows = []
    rhs = []

    # Wall constraints
    for i in range(n):
        x, y = centres[i]
        # r_i <= x_i   (left wall)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(x)

        # r_i <= 1 - x_i   (right wall)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - x)

        # r_i <= y_i   (bottom wall)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(y)

        # r_i <= 1 - y_i   (top wall)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - y)

    # Pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    bounds = [(0.0, None) for _ in range(n)]

    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # Fallback: all radii zero
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)
        total = float(radii.sum())

    # Subtract a tiny epsilon to stay safely inside the validator tolerances
    eps = 1e-8
    radii = np.maximum(radii - eps, 0.0)
    total = float(radii.sum())
    return radii, total


# ----------------------------------------------------------------------
# Helpers to generate starting configurations
# ----------------------------------------------------------------------
def generate_hex_initial(n_points: int, dx: float = 0.18) -> np.ndarray:
    """
    Produce a set of points on a (truncated) hexagonal lattice.
    More points than needed are generated and a random subset of size n_points
    is returned.
    """
    dy = np.sqrt(3.0) / 2.0 * dx
    pts = []
    row = 0
    y = 0.0
    while y <= 1.0:
        offset = (dx / 2.0) if (row % 2 == 1) else 0.0
        x = 0.0
        while x <= 1.0:
            cx = x + offset
            if cx <= 1.0:
                pts.append([cx, y])
            x += dx
        row += 1
        y += dy
    pts = np.array(pts)
    # keep only points strictly inside the square (excluding the exact walls)
    mask = (pts[:, 0] > 0) & (pts[:, 0] < 1) & (pts[:, 1] > 0) & (pts[:, 1] < 1)
    pts = pts[mask]
    if len(pts) < n_points:
        # fallback: fill the missing slots with random points
        extra = np.random.rand(n_points - len(pts), 2)
        pts = np.vstack([pts, extra])
    idx = np.random.choice(len(pts), size=n_points, replace=False)
    return pts[idx].astype(float)


def generate_random_initial(n_points: int) -> np.ndarray:
    """Uniform random points inside the unit square."""
    return np.random.rand(n_points, 2)


# ----------------------------------------------------------------------
# Local optimisation (simulated‑annealing style)
# ----------------------------------------------------------------------
def optimise_from_initial(initial_centres: np.ndarray,
                         max_iter: int = 1200,
                         temp_start: float = 0.12,
                         temp_end: float = 1e-5) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Hill‑climb / simulated‑annealing on the centre positions.
    Returns the best centres, radii and sum found.
    """
    n = initial_centres.shape[0]
    centres = initial_centres.copy()
    radii, current_sum = compute_optimal_radii(centres)
    best_c = centres.copy()
    best_r = radii.copy()
    best_sum = current_sum

    for it in range(max_iter):
        # linear temperature schedule
        t = it / max_iter
        temp = temp_start * ((temp_end / temp_start) ** t)

        # pick a circle and propose a move
        i = np.random.randint(n)
        old_pos = centres[i].copy()
        # step size decays with temperature
        sigma = 0.08 * (1.0 - t) + 0.02 * t
        proposal = old_pos + np.random.normal(scale=sigma, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)
        centres[i] = proposal

        new_radii, new_sum = compute_optimal_radii(centres)
        delta = new_sum - current_sum

        # Metropolis acceptance
        if delta > 0.0 or np.exp(delta / (temp + 1e-12)) > np.random.rand():
            # accept
            radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_c = centres.copy()
                best_r = new_radii.copy()
        else:
            # reject – restore old centre
            centres[i] = old_pos

    return best_c, best_r, best_sum


# ----------------------------------------------------------------------
# Main entry point required by the grader
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles.
    The routine tries a few random restarts and keeps the best layout.
    """
    np.random.seed(0)                # deterministic behaviour
    n_circles = 26
    best_overall_sum = -1.0
    best_overall_centres = None
    best_overall_radii = None

    # 1) Hexagonal start
    init_hex = generate_hex_initial(n_circles, dx=0.18)
    c, r, s = optimise_from_initial(init_hex, max_iter=1500)
    if s > best_overall_sum:
        best_overall_sum = s
        best_overall_centres = c
        best_overall_radii = r

    # 2) Several pure random starts
    for _ in range(3):
        init_rand = generate_random_initial(n_circles)
        c, r, s = optimise_from_initial(init_rand, max_iter=1200)
        if s > best_overall_sum:
            best_overall_sum = s
            best_overall_centres = c
            best_overall_radii = r

    # Safety: ensure radii are truly feasible (run a final LP)
    final_radii, final_sum = compute_optimal_radii(best_overall_centres)
    # tiny correction for numerical safety
    final_radii = np.maximum(final_radii - 1e-8, 0.0)
    final_sum = float(final_radii.sum())

    return best_overall_centres, final_radii, final_sum
