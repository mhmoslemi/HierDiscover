def anneal_placement(num_circles=26, max_iter=50000, temp_start=0.1, temp_end=1e-4):
    np.random.seed(0)
    # Initialize centers randomly in [0,1]^2
    centers = np.random.rand(num_circles, 2)
    # Could also initialize using hex grid to start near optimum
    sum_r, radii = solve_lp(centers)
    best_sum = sum_r
    best_centers = centers.copy()
    best_radii = radii.copy()
    for iter in range(max_iter):
        # temperature schedule: e.g., exponential cooling
        t = temp_start * (temp_end / temp_start) ** (iter / max_iter)
        # propose move: pick random index i, add Gaussian perturbation sigma maybe 0.05
        i = np.random.randint(num_circles)
        new_center = centers[i] + np.random.normal(scale=0.05, size=2)
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # Save old center
        old_center = centers[i].copy()
        centers[i] = new_center
        # compute new sum radii
        sum_r_new, radii_new = solve_lp(centers)
        # acceptance criterion
        if sum_r_new > sum_r or np.random.rand() < np.exp((sum_r_new - sum_r)/t):
            sum_r = sum_r_new
            radii = radii_new
            if sum_r_new > best_sum:
                best_sum = sum_r_new
                best_centers = centers.copy()
                best_radii = radii.copy()
        else:
            # revert
            centers[i] = old_center
    return best_centers, best_radii, best_sum

Pseudo:

def hill_climb(initial_centers, init_temp=0.1, final_temp=1e-4, iterations=3000):
    centers = initial_centers.copy()
    cur_sum, cur_radii = solve_lp(centers)
    best_sum = cur_sum
    best_centers = centers.copy()
    best_radii = cur_radii.copy()
    n = centers.shape[0]
    for t in range(iterations):
        # linear temperature schedule
        T = init_temp + (final_temp - init_temp) * (t / iterations)
        # propose
        i = np.random.randint(n)
        # candidate move: gaussian offset; step sigma ~0.05
        new_center = centers[i] + np.random.normal(scale=0.05, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        old_center = centers[i].copy()
        centers[i] = new_center
        new_sum, new_radii = solve_lp(centers)
        delta = new_sum - cur_sum
        if delta > 0 or np.random.rand() < np.exp(delta / T):
            # accept
            cur_sum = new_sum
            cur_radii = new_radii
            # update best
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = centers.copy()
                best_radii = new_radii.copy()
        else:
            # revert
            centers[i] = old_center
    return best_centers, best_radii, best_sum

assistantfinal**Solution Explanation**

The task is to place 26 circles in a unit square `[0,1]×[0,1]`.
For a given set of centres `c[i] = (x_i , y_i)` the admissible radii are
the solution of the linear program  

```
 maximise   Σ r_i
 subject to r_i ≥ 0
            r_i ≤ min(x_i, 1‑x_i, y_i, 1‑y_i)                (inside the square)
            r_i + r_j ≤ dist(c[i],c[j])   for all i<j        (no overlap)
```

For fixed centres the program is trivial – it has only linear constraints
and a linear objective.
Hence the whole optimisation problem can be tackled in two steps

1. **Given a set of centres → compute the optimal radii**
   (`solve_lp` – a tiny wrapper around `scipy.optimize.linprog`).

2. **Move the centres so that the value of the linear program becomes larger**
   (simulated annealing / stochastic hill climbing).

The annealing loop works on the *centres only*.
After each move the linear program is solved, the obtained sum of radii is
the “energy” of the configuration.
A Metropolis criterion with a slowly decreasing temperature is used;
this lets the algorithm escape shallow local optima while still
converging to a high‑quality solution.

The whole procedure is deterministic because a fixed random seed is used.
The final radii are obtained a last time from the best set of centres
found during the search, guaranteeing that the returned circles satisfy
all constraints checked by the validator.

The implementation contains only top‑level functions, no file or network
access, and uses only `numpy` and `scipy`.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programme – optimal radii for a fixed set of centres
# ------------------------------------------------------------
def solve_lp(centres: np.ndarray):
    """
    Compute the maximal sum of radii for the given centre positions.
    Returns (sum_of_radii, radii_vector).
    """
    n = centres.shape[0]

    # distance to the four sides of the square
    d_boundary = np.minimum.reduce(
        [centres[:, 0], 1.0 - centres[:, 0], centres[:, 1], 1.0 - centres[:, 1]]
    )                     # shape (n,)

    # pairwise Euclidean distances
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diff * diff, axis=2))

    # build the inequality matrix A_ub * r <= b_ub
    n_constraints = n + n * (n - 1) // 2
    A = np.zeros((n_constraints, n), dtype=np.float64)
    b = np.empty(n_constraints, dtype=np.float64)

    # boundary constraints  r_i <= d_boundary_i
    rows = np.arange(n)
    A[rows, rows] = 1.0
    b[rows] = d_boundary

    # pair constraints  r_i + r_j <= dist(i,j)
    idx = n
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = d_pair[i, j]
            idx += 1

    # linear programme: maximise Σ r_i  <=> minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # In the (very unlikely) case of a failure return zero radii
        radii = np.zeros(n, dtype=np.float64)
        return 0.0, radii

    radii = np.maximum(res.x, 0.0)           # guard against tiny negatives
    return float(radii.sum()), radii


# ------------------------------------------------------------
# Simulated annealing on the centre positions
# ------------------------------------------------------------
def optimise_centres(initial: np.ndarray,
                    rng: np.random.Generator,
                    max_iter: int = 5000,
                    t_start: float = 0.02,
                    t_end: float = 1e-5,
                    sigma_start: float = 0.20,
                    sigma_end: float = 0.001):
    """
    Stochastic optimisation of the centre positions.
    Returns (best_centres, best_radii, best_sum).
    """
    centres = initial.copy()
    cur_sum, cur_radii = solve_lp(centres)

    best_sum = cur_sum
    best_centres = centres.copy()
    best_radii = cur_radii.copy()

    n = centres.shape[0]

    for it in range(max_iter):
        # geometric cooling for temperature and step size
        frac = it / max_iter
        T = t_start * (t_end / t_start) ** frac
        sigma = sigma_start * (sigma_end / sigma_start) ** frac

        # propose a new configuration (move all centres a little)
        cand = centres + rng.normal(scale=sigma, size=(n, 2))
        np.clip(cand, 0.0, 1.0, out=cand)

        cand_sum, cand_radii = solve_lp(cand)
        delta = cand_sum - cur_sum

        # Metropolis acceptance
        if delta > 0.0 or rng.random() < np.exp(delta / T):
            centres = cand
            cur_sum = cand_sum
            cur_radii = cand_radii

            if cand_sum > best_sum:
                best_sum = cand_sum
                best_centres = cand.copy()
                best_radii = cand_radii.copy()

    return best_centres, best_radii, best_sum


# ------------------------------------------------------------
# Helper – initial configurations
# ------------------------------------------------------------
def initial_grid():
    """5×5 regular grid (25 points) + one point in the centre."""
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    grid = np.array(np.meshgrid(xs, ys)).T.reshape(-1, 2)   # 25×2
    centre = np.array([[0.5, 0.5]])                        # extra point
    return np.vstack([grid, centre])


def initial_hex():
    """Hexagonal pattern (23 points) + three random points."""
    rng = np.random.default_rng(0)
    d = 0.20                     # target centre‑to‑centre distance
    dy = np.sqrt(3) / 2 * d
    rows = []
    for k in range(5):
        y = 0.1 + k * dy
        offset = 0.0 if k % 2 == 0 else d / 2
        xs = np.arange(0.1 + offset, 0.9 + 1e-12, d)
        rows.append(np.column_stack([xs, np.full_like(xs, y)]))
    hex_pts = np.vstack(rows)    # 23 points
    extra = rng.uniform(low=0.1, high=0.9, size=(3, 2))
    return np.vstack([hex_pts, extra])


def initial_random():
    """Purely random initialisation (26 points)."""
    rng = np.random.default_rng(0)
    return rng.uniform(size=(26, 2))

def run_packing():
    """
    Computes a packing of 26 circles in the unit square.
    Returns (centres, radii, sum_of_radii).
    """
    rng = np.random.default_rng(42)          # deterministic randomness

    # a few different start configurations – the best will be kept
    starters = [
        initial_grid(),
        initial_hex(),
        initial_random(),
    ]

    best_sum = -1.0
    best_centres = None
    best_radii = None

    for start in starters:
        # jitter the start a little so that the optimiser does not sit on a perfectly
        # symmetric configuration (which can be a shallow local optimum)
        jitter = rng.normal(scale=0.02, size=start.shape)
        start_jit = np.clip(start + jitter, 0.0, 1.0)

        c, r, s = optimise_centres(
            start_jit,
            rng,
            max_iter=4000,          # enough iterations for a good solution
            t_start=0.02,
            t_end=1e-5,
            sigma_start=0.18,
            sigma_end=0.001,
        )

        if s > best_sum:
            best_sum = s
            best_centres = c
            best_radii = r

    # final sanity check – recompute the radii for the best centres
    final_sum, final_radii = solve_lp(best_centres)

    # return exactly what the checker expects
    return best_centres, final_radii, final_sum
```
