import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    # list constraints
    # Pairwise
    rows = []
    cols = []
    data = []
    b = []
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(dists[i,j])
    # boundary constraints
    for i in range(n):
        x = centers[i,0]
        y = centers[i,1]
        # r_i <= x
        row = np.zeros(n); row[i]=1
        rows.append(row); b.append(x)
        # r_i <= 1-x
        row = np.zeros(n); row[i]=1
        rows.append(row); b.append(1-x)
        # r_i <= y
        row = np.zeros(n); row[i]=1
        rows.append(row); b.append(y)
        # r_i <= 1-y
        row = np.zeros(n); row[i]=1
        rows.append(row); b.append(1-y)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback
        radii = np.zeros(n)
        total = 0
    else:
        radii = res.x
        total = radii.sum()
    return radii, total

def farthest_point_sampling(n, candidates_per_iter=5000, rng=None):
    if rng is None: rng=np.random.default_rng()
    pts = []
    for k in range(n):
        # generate candidates
        cand = rng.random((candidates_per_iter, 2))
        # compute distances to walls: d_wall = np.minimum.reduce([cand[:,0], 1-cand[:,0], cand[:,1], 1-cand[:,1]])
        d_wall = np.minimum(np.minimum(cand[:,0], 1 - cand[:,0]), np.minimum(cand[:,1], 1 - cand[:,1]))
        if len(pts)==0:
            d = d_wall
        else:
            existing = np.array(pts)
            # compute distances to existing points
            diff = cand[:, None, :] - existing[None, :, :]
            d_pts = np.linalg.norm(diff, axis=2).min(axis=1)
            d = np.minimum(d_wall, d_pts)
        # choose candidate with max d
        idx = np.argmax(d)
        pts.append(cand[idx])
    return np.array(pts)

def local_optimize(centers, radii, best_sum, iterations=5000, step_scale=0.05, rng=None):
    if rng is None: rng=np.random.default_rng()
    n = centers.shape[0]
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_val = best_sum
    for it in range(iterations):
        i = rng.integers(n)
        # propose new center by adding noise
        new_center = best_centers[i] + rng.normal(scale=step_scale, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        # generate new centers copy
        candidate = best_centers.copy()
        candidate[i] = new_center
        # recompute radii
        cand_r, cand_sum = compute_optimal_radii(candidate)
        if cand_sum > best_val:
            best_val = cand_sum
            best_centers = candidate
            best_radii = cand_r
        # optional annealing: reduce step scale gradually

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)
    # Build A_ub and b_ub
    m = n * (n - 1) // 2 + 4 * n
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1
    # boundary
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        A[row, i] = 1.0
        b[row] = x
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        A[row, i] = 1.0
        b[row] = y
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1
    # objective
    c = -np.ones(n)

def farthest_point_sampling(n, candidates_per_iter=2000, rng=None):
    if rng is None: rng = np.random.default_rng()
    points = []
    for _ in range(n):
        cand = rng.random((candidates_per_iter, 2))
        # distance to walls
        d_wall = np.minimum(np.minimum(cand[:,0], 1 - cand[:,0]), np.minimum(cand[:,1], 1 - cand[:,1]))
        if points:
            existing = np.array(points)
            # compute distances to existing points
            diff = cand[:, None, :] - existing[None, :, :]
            d_pts = np.linalg.norm(diff, axis=2).min(axis=1)
            d = np.minimum(d_wall, d_pts)
        else:
            d = d_wall
        idx = np.argmax(d)
        points.append(cand[idx])
    return np.array(points)

import numpy as np
from scipy.optimize import linprog

TARGET = 2.636  # the score we must beat


def compute_optimal_radii(centers: np.ndarray):
    """
    Solve the linear programme that gives the largest possible radii for
    the supplied centre positions.

    Returns (radii, total_sum).
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)

    # Number of constraints: n*(n-1)/2 pairwise + 4*n boundary
    m = n * (n - 1) // 2 + 4 * n
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # r_i <= 1 - x
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # r_i <= y
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # r_i <= 1 - y
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None) for _ in range(n)],
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # should never happen, but return a safe fallback
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        # tiny safety margin against the 1e-12 tolerance used in the validator
        radii = np.maximum(res.x, 0.0) * 0.9995
        total = float(radii.sum())

    return radii, total


def farthest_point_sampling(num_points: int,
                            candidates_per_iter: int = 3000,
                            rng: np.random.Generator = None) -> np.ndarray:
    """
    Choose `num_points` locations inside the unit square.
    At each step a large set of random candidates is generated and
    the candidate whose *minimum* distance to the existing points
    **and** to the square’s sides is maximal is kept.
    """
    if rng is None:
        rng = np.random.default_rng()

    pts = []  # list of already chosen points
    for _ in range(num_points):
        cand = rng.random((candidates_per_iter, 2))

        # distance to the four sides
        d_wall = np.minimum.reduce([cand[:, 0], 1.0 - cand[:, 0],
                                    cand[:, 1], 1.0 - cand[:, 1]])

        if pts:
            existing = np.asarray(pts)
            # Euclidean distance to already placed points
            diff = cand[:, np.newaxis, :] - existing[np.newaxis, :, :]
            d_pts = np.linalg.norm(diff, axis=2).min(axis=1)
            metric = np.minimum(d_wall, d_pts)
        else:
            metric = d_wall

        best_idx = np.argmax(metric)
        pts.append(cand[best_idx])

    return np.asarray(pts)


def hill_climb(centers: np.ndarray,
               radii: np.ndarray,
               total: float,
               rng: np.random.Generator,
               steps: int = 800,
               step_scale: float = 0.07) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simple hill‑climbing: repeatedly move a random circle a little,
    re‑optimise the radii and keep the move when the total sum grows.
    """
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_val = total
    n = centers.shape[0]

    for it in range(steps):
        # occasionally replace a centre with a completely new random location
        if rng.random() < 0.02:
            i = rng.integers(n)
            new_c = rng.random(2)
        else:
            i = rng.integers(n)
            # Gaussian perturbation whose amplitude slowly shrinks
            sigma = step_scale * (1.0 - it / steps)
            new_c = best_centers[i] + rng.normal(scale=sigma, size=2)

        new_c = np.clip(new_c, 0.0, 1.0)

        cand_centers = best_centers.copy()
        cand_centers[i] = new_c

        cand_radii, cand_sum = compute_optimal_radii(cand_centers)

        if cand_sum > best_val + 1e-12:  # strict improvement
            best_val = cand_sum
            best_centers = cand_centers
            best_radii = cand_radii

        # early stop if we already beat the target
        if best_val >= TARGET:
            break

    return best_centers, best_radii, best_val


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main entry point required by the grader.
    Returns (centres, radii, sum_of_radii).
    """
    n = 26
    rng = np.random.default_rng(123456)  # deterministic seed

    best_total = -1.0
    best_centers = None
    best_radii = None

    # several independent attempts – cheap because each LP is tiny
    for trial in range(6):
        # 1) generate a well‑spread initial set of points
        centres = farthest_point_sampling(n,
                                          candidates_per_iter=3000,
                                          rng=rng)

        # 2) compute the optimal radii for that set
        radii, total = compute_optimal_radii(centres)

        # 3) improve by hill climbing
        centres, radii, total = hill_climb(centres,
                                           radii,
                                           total,
                                           rng,
                                           steps=1200,
                                           step_scale=0.08)

        if total > best_total:
            best_total = total
            best_centers = centres
            best_radii = radii

        if best_total >= TARGET:
            break  # no need to keep searching

    # final safety check – the validator is very strict, but our radii
    # were already shrunken by 0.5 % to stay safely inside all constraints.
    return best_centers, best_radii, float(best_total)
