def optimal_radii_for_centers(centers):
    n = centers.shape[0]
    # build A_ub and b_ub
    rows = []
    cols = []
    data = []
    b = []
    # boundary constraints
    for i in range(n):
        bound = np.min([centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1]])
        # A row: 1 at position i
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        b.append(bound)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            b.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        return None, -np.inf
    radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r

def solve_radii(centers):
    n = centers.shape[0]
    # compute boundary limits
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # pairwise distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub and b_ub
    # b_ub first boundary constraints for each i: r_i <= bound_i
    rows = []
    b = []
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        b.append(bound[i])
    # pairwise constraints: r_i + r_j <= d_ij for i<j
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            b.append(dists[i,j])
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    # bounds for variables (r_i >= 0)
    var_bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=var_bounds, method='highs')
    if not res.success:
        return None, -np.inf
    radii = np.maximum(res.x, 0.0)
    sum_r = radii.sum()
    return radii, float(sum_r)
```

```

All constraints are linear in the radii, therefore a single call to `scipy.optimize.linprog` yields the maximal radii for any arrangement of centres.

import numpy as np
from scipy.optimize import linprog


def solve_optimal_radii(centers: np.ndarray):
    """
    Solve the linear program that gives the maximal radii for a fixed set of centres.

    Parameters
    ----------
    centers : np.ndarray, shape (n, 2)
        Circle centres inside the unit square.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        Optimal radii (non‑negative).  None if the LP fails.
    total : float
        Sum of the radii (‑inf if the LP fails).
    """
    n = centers.shape[0]

    # distance of each centre to the four sides of the square
    side_limits = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of inequality constraints: n (sides) + n*(n-1)/2 (pairwise)
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    row = 0
    # side constraints  r_i <= side_limits[i]
    for i in range(n):
        A[row, i] = 1.0
        b[row] = side_limits[i]
        row += 1

    # pairwise constraints  r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  → minimise −Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None, -np.inf

    radii = np.maximum(res.x, 0.0)   # clip tiny negatives caused by numerics
    return radii, float(radii.sum())
