import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # bounds for each radius: (0, None)
    bounds = [(0, None) for _ in range(n)]

    # Build list of rows and b
    rows = []
    b = []

    # boundary constraints
    xs = centers[:,0]
    ys = centers[:,1]
    for i in range(n):
        # r_i <= x_i
        e = np.zeros(n)
        e[i] = 1
        rows.append(e)
        b.append(xs[i])
        # r_i <= 1 - x_i
        rows.append(e)
        b.append(1 - xs[i])
        # r_i <= y_i
        rows.append(e)
        b.append(ys[i])
        # r_i <= 1 - y_i
        rows.append(e)
        b.append(1 - ys[i])
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            e = np.zeros(n)
            e[i] = 1
            e[j] = 1
            rows.append(e)
            b.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)

    c = -np.ones(n)  # maximize sum => min -sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # something wrong, maybe infeasible; return zeros
        return np.zeros(n)
    return res.x

def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    n = 26
    # Initialize positions random
    rng = np.random.default_rng(0)    # deterministic seed for reproducibility
    centers = rng.uniform(0.0, 1.0, (n, 2))

    # maybe trying to keep them away from each other: repulsive
    # use a few steps to spread them
    for _ in range(200):
        # compute pairwise distances
        d = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=-1)
        # compute min distances to others
        min_d = np.min(np.where(d > 0, d, np.inf), axis=1)
        # move each point away from nearest neighbor
        for i in range(n):
            j = np.argmin(d[i] + np.eye(n)*np.inf) # nearest neighbor index
        # but simpler: just jitter
        centers += rng.normal(0, 0.02, centers.shape)
        centers = np.clip(centers, 0, 1)
    # now we have start.

    best_centers = centers.copy()
    best_radii = compute_radii(best_centers)
    best_sum = best_radii.sum()

    # Simulated annealing
    T0 = 0.1
    T_end = 1e-4
    steps = 5000
    for step in range(steps):
        T = T0 * (T_end/T0) ** (step/steps) # exponential cooling
        # propose move: pick random circle i, propose new center within radius sigma
        i = rng.integers(n)
        sigma = 0.1 * (1 - step/steps) + 0.01  # gradually reduce
        new_center = best_centers[i] + rng.normal(0, sigma, 2)
        new_center = np.clip(new_center, 0, 1)
        # keep other centers same
        trial_centers = best_centers.copy()
        trial_centers[i] = new_center
        trial_radii = compute_radii(trial_centers)
        trial_sum = trial_radii.sum()
        if trial_sum > best_sum or rng.random() < np.exp((trial_sum - best_sum)/max(T, 1e-12)):
            best_centers = trial_centers
            best_radii = trial_radii
            best_sum = trial_sum

    return best_centers, best_radii, float(best_sum)

```
def compute_radii(centers):
    n = centers.shape[0]
    # Build constraint matrix and RHS
    A = []
    b = []
    # boundary constraints
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1
        A.append(e)
        b.append(centers[i,0])   # x_i
        A.append(e)
        b.append(1 - centers[i,0]) # 1-x_i
        A.append(e)
        b.append(centers[i,1])   # y_i
        A.append(e)
        b.append(1 - centers[i,1]) # 1-y_i

    # pairwise constraints
    for i in range(n):
        for j in range(i+1,n):
            d = np.linalg.norm(centers[i] - centers[j])
            e = np.zeros(n)
            e[i] = 1
            e[j] = 1
            A.append(e)
            b.append(d)

    A_ub = np.array(A)
    b_ub = np.array(b)

    bounds = [(0, None)] * n

    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: all zeros
        return np.zeros(n)
    return res.x
```

def compute_radii(centers):
    n = centers.shape[0]
    # boundary constraints
    bound = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # compute boundary bound
    bound = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # A_ub start with identity for boundary constraints
    A_ub = np.eye(n)
    b_ub = bound.copy()
    # pairwise constraints
    if n > 1:
        # number of pairs
        pair_count = n * (n - 1) // 2
        A_pair = np.zeros((pair_count, n))
        b_pair = np.empty(pair_count)
        idx = 0
        for i in range(n):
            for j in range(i+1, n):
                A_pair[idx, i] = 1.0
                A_pair[idx, j] = 1.0
                b_pair[idx] = np.linalg.norm(centers[i] - centers[j])
                idx += 1
        A_ub = np.vstack([A_ub, A_pair])
        b_ub = np.concatenate([b_ub, b_pair])
    # objective - maximize sum
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # return zeros if failed
        return np.zeros(n)
    # Clip tiny negatives
    radii = np.maximum(res.x, 0.0)
    return radii

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers: np.ndarray) -> np.ndarray:
    """
    Given centre positions, compute the optimal radii by linear programming.
    Returns an array of radii (non‑negative) that satisfy all constraints and
    maximise the sum of radii.
    """
    n = centers.shape[0]

    # maximal radius allowed by the four walls for each centre
    bound = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )

    # Boundary constraints: r_i <= bound_i
    A_ub = np.eye(n)
    b_ub = bound.copy()

    # Pairwise non‑overlap constraints: r_i + r_j <= distance(i,j)
    if n > 1:
        pair_cnt = n * (n - 1) // 2
        A_pair = np.zeros((pair_cnt, n))
        b_pair = np.empty(pair_cnt)
        idx = 0
        for i in range(n):
            for j in range(i + 1, n):
                A_pair[idx, i] = 1.0
                A_pair[idx, j] = 1.0
                b_pair[idx] = np.linalg.norm(centers[i] - centers[j])
                idx += 1
        A_ub = np.vstack([A_ub, A_pair])
        b_ub = np.concatenate([b_ub, b_pair])

    # Objective: maximise sum(r)  → minimise -sum(r)
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    # Solve the LP
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # In the very unlikely case of failure return zeros
        return np.zeros(n, dtype=float)

    # Clip very small negative values that can appear due to numerical noise
    radii = np.maximum(res.x, 0.0)
    return radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a stochastic optimisation to place 26 circles inside [0,1]².
    Returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(42)   # deterministic seed
    n = 26

    # ---- initial layout -------------------------------------------------
    # 5×5 regular grid (25 points)
    grid_vals = np.linspace(0.1, 0.9, 5)          # 0.1,0.3,…,0.9
    xv, yv = np.meshgrid(grid_vals, grid_vals)
    init_grid = np.column_stack([xv.ravel(), yv.ravel()])

    # one additional point placed away from the grid points
    extra = rng.uniform(0.2, 0.8, size=(1, 2))
    centers = np.vstack([init_grid, extra])      # shape (26,2)

    # initial radii and score
    radii = compute_radii(centers)
    best_sum = float(radii.sum())
    best_centers = centers.copy()
    best_radii = radii.copy()

    # ---- simulated‑annealing style optimisation --------------------------
    steps = 8000                     # total number of proposals
    T_start = 0.30                   # initial temperature
    T_end   = 1e-4                   # final temperature

    for step in range(steps):
        # exponential cooling schedule
        T = T_start * (T_end / T_start) ** (step / steps)

        # pick a circle to move
        i = rng.integers(n)

        # occasionally make a large jump, otherwise a small Gaussian step
        if rng.random() < 0.01:
            new_pos = rng.uniform(0.0, 1.0, size=2)
        else:
            sigma = 0.08 * (1.0 - step / steps) + 0.005
            new_pos = best_centers[i] + rng.normal(scale=sigma, size=2)

        # keep inside the square
        new_pos = np.clip(new_pos, 0.0, 1.0)

        # evaluate the proposal
        trial_centers = best_centers.copy()
        trial_centers[i] = new_pos
        trial_radii = compute_radii(trial_centers)
        trial_sum = float(trial_radii.sum())

        delta = trial_sum - best_sum
        accept = False
        if delta > 0:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp(delta / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            best_centers = trial_centers
            best_radii = trial_radii
            best_sum = trial_sum

    return best_centers, best_radii, best_sum
