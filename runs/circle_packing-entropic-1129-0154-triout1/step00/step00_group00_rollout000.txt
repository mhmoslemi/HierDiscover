def run_packing():
    import numpy as np
    from scipy.optimize import linprog
    np.random.seed(0)

    n=26
    # initial centers: hex grid with small step
    s=0.09
    centers = []
    row = 0
    while True:
        y = s + row*np.sqrt(3)*s
        if y > 1 - s:
            break
        if row%2 == 0:
            x_start = s
        else:
            x_start = 2*s
        col=0
        while True:
            x = x_start + col*2*s
            if x > 1 - s:
                break
            centers.append([x,y])
            col+=1
        row+=1
    centers = np.array(centers)
    if centers.shape[0] < n:
        # if less than needed, generate random points to fill
        extra_needed = n - centers.shape[0]
        extra = np.random.rand(extra_needed,2)
        centers = np.vstack([centers, extra])
    else:
        # if more, randomly select n
        idx = np.random.choice(centers.shape[0], n, replace=False)
        centers = centers[idx]

    def solve(centers):
        n = centers.shape[0]
        # compute distances
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        D = np.sqrt(np.sum(diff**2, axis=2))
        # Build A_ub
        npair = n*(n-1)//2
        m = 4*n + npair
        A_ub = np.zeros((m, n))
        b_ub = np.zeros(m)
        row = 0
        # boundary constraints
        for i in range(n):
            A_ub[row, i] = 1; b_ub[row] = centers[i,0]; row+=1
            A_ub[row, i] = 1; b_ub[row] = 1 - centers[i,0]; row+=1
            A_ub[row, i] = 1; b_ub[row] = centers[i,1]; row+=1
            A_ub[row, i] = 1; b_ub[row] = 1 - centers[i,1]; row+=1
        # pair constraints
        for i in range(n):
            for j in range(i+1, n):
                A_ub[row, i] = 1; A_ub[row, j] = 1; b_ub[row] = D[i,j]
                row+=1
        c = -np.ones(n)
        bounds = [(0, None)]*n
        res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
        if res.success:
            r = res.x
        else:
            r = np.zeros(n)
        return r, r.sum()

    radii, cur_sum = solve(centers)

    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = cur_sum

    # local search
    delta = 0.1
    for iter in range(3000):
        # pick random index
        i = np.random.randint(n)
        # propose new position for i
        new_center_i = best_centers[i] + np.random.uniform(-delta, delta, size=2)
        new_center_i = np.clip(new_center_i, 0.0, 1.0)
        new_centers = best_centers.copy()
        new_centers[i] = new_center_i
        # solve radii at new positions
        radii_new, sum_new = solve(new_centers)
        if sum_new > best_sum + 1e-6:
            best_sum = sum_new
            best_centers = new_centers
            best_radii = radii_new
        # gradually reduce delta
        delta *= 0.9995
        if delta < 1e-4:
            delta = 0.1

    return best_centers, best_radii, float(best_sum)

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    # construct constraints
    npair = n * (n - 1) // 2
    m = 4 * n + npair
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    row = 0
    # boundary constraints
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = centers[i, 0]          # r_i <= x_i
        row += 1
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - centers[i, 0]    # r_i <= 1 - x_i
        row += 1
        A_ub[row, i] = 1.0
        b_ub[row] = centers[i, 1]          # r_i <= y_i
        row += 1
        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - centers[i, 1]    # r_i <= 1 - y_i
        row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = D[i, j]            # r_i + r_j <= dist(i,j)
            row += 1
    c = -np.ones(n)  # maximize sum of radii -> minimize -sum
    bounds = [(0.0, None)] * n
    # solve linear program
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        radii = res.x
    else:
        radii = np.zeros(n)
    return radii, float(radii.sum())

assistantfinal**<strategy>**  
The problem can be written as a linear program **once the circle centres are fixed**.  
For a given set of centres `c_i = (x_i, y_i)` the radii `r_i` are limited by

* the four borders of the unit square  

  `r_i ≤ x_i , r_i ≤ 1‑x_i , r_i ≤ y_i , r_i ≤ 1‑y_i`

* every pair of circles  

  `r_i + r_j ≤ ‖c_i‑c_j‖`

All constraints are linear in the radii, therefore for fixed centres the maximal
sum of radii is obtained by solving the LP  

```
max Σ r_i      ⇔   min –Σ r_i
s.t.  A_ub·r ≤ b_ub ,   r ≥ 0
```

The overall optimisation is then to **choose the 26 centre positions** so that
the LP optimum is as large as possible.  
A deterministic, fast heuristic is used:

1. **Initial placement – furthest‑point sampling**  
   Starting from one random point, repeatedly draw a large batch of random
   candidates and keep the point whose minimum distance to the already placed
   points is maximal. This yields a set of points that are as far apart as
   possible, which in turn gives large radii after the LP is solved.

2. **Local refinement**  
   Repeatedly pick a single circle, perturb its centre by a small random
   offset (still inside the square) and resolve the LP.  
   If the new sum of radii is larger, the move is accepted.  
   The step size `δ` is slowly reduced, allowing coarse moves at the start
   and fine tuning later on.

3. **Multiple restarts**  
   The whole process (steps 1‑2) is performed several times with different
   random seeds; the best result among the restarts is kept.

All heavy work (building the LP and solving it with `scipy.optimize.linprog`
using the efficient “highs” solver) is performed in a helper function
`solve_radii`.  
The final routine `run_packing` returns the best centres, the corresponding
optimal radii, and their sum.  The solution is guaranteed to satisfy the
validation function because the LP enforces the non‑overlap and boundary
constraints exactly.

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

def solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a fixed set of centres solve the linear program that maximises the
    sum of radii under the border and non‑overlap constraints.
    Returns the optimal radii and their sum.
    """
    n = centers.shape[0]

    # pairwise distances (constant for the LP)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    D = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of linear constraints
    npair = n * (n - 1) // 2
    m = 4 * n + npair

    A_ub = np.zeros((m, n), dtype=np.float64)
    b_ub = np.zeros(m, dtype=np.float64)

    row = 0
    # border constraints r_i <= x_i, 1‑x_i, y_i, 1‑y_i
    for i in range(n):
        A_ub[row, i] = 1.0
        b_ub[row] = centers[i, 0]
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - centers[i, 0]
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = centers[i, 1]
        row += 1

        A_ub[row, i] = 1.0
        b_ub[row] = 1.0 - centers[i, 1]
        row += 1

    # pairwise constraints r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A_ub[row, i] = 1.0
            A_ub[row, j] = 1.0
            b_ub[row] = D[i, j]
            row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if res.success:
        radii = res.x
    else:
        # fallback – should never happen on a feasible instance
        radii = np.zeros(n, dtype=np.float64)

    return radii, float(radii.sum())


def generate_initial_positions(n: int) -> np.ndarray:
    """
    Furthest‑point sampling: picks n points that are as far from each other as
    possible by repeatedly sampling a large batch of candidates.
    """
    np.random.seed()                      # use global RNG state
    centers = [np.random.rand(2)]
    while len(centers) < n:
        # draw a batch of candidates
        cand = np.random.rand(5_000, 2)
        existing = np.asarray(centers)          # shape (k,2)
        # compute distance to the nearest existing point for each candidate
        d = np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2)
        min_d = d.min(axis=1)
        best = np.argmax(min_d)
        centers.append(cand[best])
    return np.asarray(centers)


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_of_radii) for a packing of 26 circles
    in the unit square that (almost) maximises the sum of the radii.
    """
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # several independent restarts improve the chance to reach a high value
    for restart in range(5):
        # ---------- initial layout ----------
        centers = generate_initial_positions(n)
        radii, cur_sum = solve_radii(centers)

        # ---------- local optimisation ----------
        delta = 0.18                     # maximal trial displacement
        for it in range(3000):
            i = np.random.randint(n)

            # propose a random move for circle i
            new_center = centers[i] + np.random.uniform(-delta, delta, size=2)
            new_center = np.clip(new_center, 0.0, 1.0)   # stay inside the square
            new_centers = centers.copy()
            new_centers[i] = new_center

            new_radii, new_sum = solve_radii(new_centers)

            if new_sum > cur_sum + 1e-9:
                # accept improvement
                centers = new_centers
                radii = new_radii
                cur_sum = new_sum

            # slowly shrink the step size
            delta *= 0.9995
            if delta < 5e-4:
                delta = 0.18

        # keep the best result among restarts
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centers.copy()
            best_radii = radii.copy()

    # final sanity check – ensure shapes are correct
    best_centers = np.asarray(best_centers).reshape(n, 2)
    best_radii = np.asarray(best_radii).reshape(n)

    return best_centers, best_radii, float(best_sum)
