def compute_optimal_radii(centers):
    n = len(centers)
    # Build A_ub matrix as list of rows
    rows = []
    ub = []
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1 - x, y, 1 - y)
        # r_i <= bound
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        ub.append(bound)
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            ub.append(d)
    A_ub = np.array(rows)
    b_ub = np.array(ub)
    c = -np.ones(n)  # maximize sum, minimize negative
    result = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=(0, None), method='highs')
    if not result.success:
        raise RuntimeError('LP failed')
    radii = result.x
    sum_r = radii.sum()
    return radii, sum_r

def solve_lp(centers):
    n = centers.shape[0]
    r = cvxpy.Variable(n)
    constraints = [r >= 0]
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1 - x, y, 1 - y)
        constraints.append(r[i] <= bound + 1e-9) # add small tolerance
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d + 1e-9)
    prob = cvxpy.Problem(cvxpy.Maximize(cvxpy.sum(r)), constraints)
    prob.solve(solver=cvxpy.ECOS, warm_start=True, max_iters=5000) # or default.
    radii = np.maximum(r.value, 0)
    return radii, radii.sum()

import scipy.optimize

def solve_lp(centers):
    n = centers.shape[0]
    # build A_ub
    rows = []
    ub = []
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        ub.append(bound)
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            ub.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(ub)
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback to zero radii
        return np.zeros(n), 0.0
    radii = res.x
    return radii, radii.sum()

import numpy as np
import scipy.optimize as opt
import time

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # Build A_ub, b_ub
    rows = []
    ub = []
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1 - x, y, 1 - y)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        ub.append(bound)
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            ub.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(ub)
    c = -np.ones(n)
    res = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros(n), 0.0
    radii = np.maximum(res.x, 0.0)
    return radii, radii.sum()

Now hill climbing:

def improve_positions(centers):
    best_centers = centers.copy()
    best_radii, best_sum = compute_optimal_radii(best_centers)
    n = len(centers)
    # parameters
    start_time = time.time()
    max_time = 1.5
    temp = 0.1
    step = 0.2
    while time.time() - start_time < max_time:
        i = np.random.randint(n)
        # propose move
        new_center = best_centers[i].copy()
        new_center += np.random.normal(scale=step, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not too close to boundary? Not required; radii will be limited.
        proposal = best_centers.copy()
        proposal[i] = new_center
        # compute
        radii, sum_r = compute_optimal_radii(proposal)
        # acceptance
        if sum_r > best_sum + 1e-6:
            best_centers = proposal
            best_radii = radii
            best_sum = sum_r
        # maybe reduce step gradually
        step *= 0.9995
    return best_centers, best_radii, best_sum

                ys.append(y)

            arr = arr[:n]
        # if less, add random points
        while arr.shape[0] < n:

            arr = np.vstack([arr, pt])

    best_centers = generate_initial_positions()
    best_radii, best_sum = solve_lp(best_centers)

    # hill climb
    start = time.time()
    max_time = 2.0
    step = 0.2
    while time.time() - start < max_time:
        # propose move
        i = np.random.randint(n)
        new_center = best_centers[i].copy()
        # random perturb
        new_center += np.random.normal(scale=step, size=2)

    best_radii = np.maximum(best_radii - 1e-8, 0.0)

import numpy as np
import scipy.optimize
import time
import math

def _solve_lp(centers: np.ndarray):
    """
    Solve a linear programme to obtain the largest possible radii for a fixed set
    of centres inside the unit square.
    Returns (radii, sum_of_radii).
    """
    n = centers.shape[0]
    # build the constraints  r_i <= distance to the four edges
    rows = []
    ub = []
    for i in range(n):
        x, y = centers[i]
        bound = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        ub.append(bound)

    # pairwise non‑overlap constraints  r_i + r_j <= distance(centres_i,centres_j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            # a tiny epsilon prevents numerical trouble when points are extremely close
            if d < 1e-12:
                d = 0.0
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            ub.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(ub)

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n)

    res = scipy.optimize.linprog(c,
                                 A_ub=A_ub,
                                 b_ub=b_ub,
                                 bounds=[(0.0, None)] * n,
                                 method='highs',
                                 options={'presolve': True})
    if not res.success:
        # fallback – all zero radii (should never happen)
        return np.zeros(n), 0.0
    radii = np.maximum(res.x, 0.0)
    return radii, float(radii.sum())

def _initial_centres():
    """
    Build an initial set of 26 centres:
    three moderately large circles near corners and the rest random.
    """
    np.random.seed(0)   # deterministic start
    centres = []

    # three large circles (they can grow to ~0.2)
    centres.append([0.2, 0.2])
    centres.append([0.8, 0.2])
    centres.append([0.5, 0.8])

    # the remaining 23 points – uniform random, later jittered by the optimiser
    while len(centres) < 26:
        pt = np.random.rand(2)
        centres.append(pt.tolist())
    return np.array(centres)

def run_packing():
    """
    Returns (centers, radii, total_sum) for 26 circles inside [0,1]².
    The implementation uses a small stochastic optimisation loop that repeatedly
    solves a linear programme to obtain the optimal radii for the current
    centre positions.
    """
    np.random.seed(42)           # full reproducibility
    start_time = time.time()
    TIME_LIMIT = 2.0             # seconds allotted for the optimiser
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Helper to perform a short hill‑climbing optimisation starting from `cent`
    def _hill_climb(cent):
        radii, cur_sum = _solve_lp(cent)
        step = 0.15               # initial perturbation size
        # a few hundred local moves – cheap for n=26
        for _ in range(200):
            i = np.random.randint(n)
            new_pt = cent[i].copy()
            new_pt += np.random.normal(scale=step, size=2)
            new_pt = np.clip(new_pt, 0.0, 1.0)

            trial = cent.copy()
            trial[i] = new_pt
            trial_r, trial_sum = _solve_lp(trial)

            if trial_sum > cur_sum + 1e-7:
                cent = trial
                radii = trial_r
                cur_sum = trial_sum
            # slowly shrink the step size – a simple annealing schedule
            step *= 0.997
        return cent, radii, cur_sum

    # Main loop – several restarts until the time budget is exhausted
    while time.time() - start_time < TIME_LIMIT:
        # start from the deterministic pattern plus a tiny random jitter
        centres = _initial_centres()
        jitter = np.random.uniform(-0.02, 0.02, size=centres.shape)
        centres = np.clip(centres + jitter, 0.0, 1.0)

        centres, radii, cur_sum = _hill_climb(centres)

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centres
            best_radii = radii

    # safety margin – move each radius a little inside the feasible region
    best_radii = np.maximum(best_radii - 1e-8, 0.0)
    best_sum = float(best_radii.sum())

    # The validator expects numpy arrays of the exact shape
    return best_centers, best_radii, best_sum
