def solve_radii(centers):
    N = centers.shape[0]
    # Compute distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # linear programming
    c = -np.ones(N)  # minimize -sum => maximize sum
    # Build A_ub, b_ub
    # boundary constraints: for each i, we have four
    A = []
    b = []
    for i in range(N):
        xi, yi = centers[i]
        # r_i <= xi
        row = np.zeros(N)
        row[i] = 1.0
        A.append(row)
        b.append(xi)
        # r_i <= 1 - xi
        row = np.zeros(N)
        row[i] = 1.0
        A.append(row)
        b.append(1.0 - xi)
        # r_i <= yi
        row = np.zeros(N)
        row[i] = 1.0
        A.append(row)
        b.append(yi)
        # r_i <= 1 - yi
        row = np.zeros(N)
        row[i] = 1.0
        A.append(row)
        b.append(1.0 - yi)
    # pairwise constraints
    for i in range(N):
        for j in range(i+1, N):
            dij = dists[i, j]
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(dij)
    A = np.array(A)
    b = np.array(b)
    bounds = [(0, None)] * N
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        r = res.x
    else:
        # if fails, fallback to zeros
        r = np.zeros(N)
    return r, float(np.sum(r)), res

def farthest_point_sampling(N, rng, n_candidates=5000):
    points = []
    # start with a random point
    p0 = rng.uniform(0,1,size=2)
    points.append(p0)
    while len(points) < N:
        # generate candidates
        cand = rng.uniform(0,1,size=(n_candidates,2))
        # compute distances to existing points
        pts_arr = np.array(points)
        # compute for each candidate the minimal Euclidean distance to any existing point
        # also compute min distance to walls

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

# --------------------------------------------------------------
# Helper: linear program that, for fixed centres, gives the
# maximal radii and the corresponding sum of radii.
# --------------------------------------------------------------
def solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    centres : (n,2) array, coordinates in [0,1]^2
    returns : radii (n,) and sum of radii
    """
    n = centers.shape[0]
    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # ----- build linear constraints  A_ub @ r <= b_ub -----
    # each circle has four wall constraints
    rows = []
    bounds_vec = []

    xs = centers[:, 0]
    ys = centers[:, 1]

    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        # left wall
        rows.append(e.copy())
        bounds_vec.append(xs[i])
        # right wall
        rows.append(e.copy())
        bounds_vec.append(1.0 - xs[i])
        # bottom wall
        rows.append(e.copy())
        bounds_vec.append(ys[i])
        # top wall
        rows.append(e.copy())
        bounds_vec.append(1.0 - ys[i])

    # pairwise non‑overlap constraints r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            bounds_vec.append(dists[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(bounds_vec)

    # objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    # radii are non‑negative
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – all radii zero (should never happen)
        radii = np.zeros(n)
    else:
        radii = res.x

    total = float(np.sum(radii))
    return radii, total


# --------------------------------------------------------------
# Helper: produce a set of points that are as far apart as possible.
# Simple farthest‑point sampling with many random candidates.
# --------------------------------------------------------------
def farthest_point_sampling(n: int,
                            rng: np.random.Generator,
                            n_candidates: int = 5000) -> np.ndarray:
    """Return (n,2) array of points inside the unit square."""
    pts = []
    # start with a random seed point
    pts.append(rng.uniform(0.0, 1.0, size=2))

    while len(pts) < n:
        # generate candidates
        cand = rng.uniform(0.0, 1.0, size=(n_candidates, 2))

        # distance to already placed points
        existing = np.asarray(pts)
        d_to_pts = np.min(
            np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2),
            axis=1)

        # distance to the four walls (this also limits the possible radius)
        d_to_walls = np.minimum.reduce(
            [cand[:, 0], 1.0 - cand[:, 0], cand[:, 1], 1.0 - cand[:, 1]])

        # we want the point whose *minimal* limiting distance is maximal
        score = np.minimum(d_to_pts, d_to_walls)
        best = np.argmax(score)
        pts.append(cand[best])
    return np.asarray(pts)


# --------------------------------------------------------------
# Main routine required by the task
# --------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns
    -------
    centers : (26,2) array of (x,y) coordinates
    radii   : (26,) array of radii
    sum_r   : float, sum of the radii
    """
    N = 26
    rng = np.random.default_rng(42)        # deterministic randomness

    # ----- outer optimisation – several independent restarts -----
    best_sum = -1.0
    best_centers = None
    best_radii = None

    for restart in range(4):
        # initialise a well‑spaced set of points
        centres = farthest_point_sampling(N, rng, n_candidates=4000)

        # a small jitter helps the optimiser escape exact symmetries
        centres += rng.uniform(-0.015, 0.015, size=centres.shape)
        centres = np.clip(centres, 0.0, 1.0)

        # optimal radii for the current layout
        radii, cur_sum = solve_radii(centres)

        # ----- local stochastic improvement (hill‑climbing + annealing) -----
        max_iter = 1500
        for it in range(max_iter):
            # pick a circle to move
            i = rng.integers(N)

            # step size slowly shrinks
            step = 0.12 * (1.0 - it / max_iter) + 0.01
            new_pos = centres[i] + rng.normal(scale=step, size=2)
            new_pos = np.clip(new_pos, 0.0, 1.0)

            # reject moves that place two centres on top of each other
            dmin = np.min(np.linalg.norm(centres - new_pos, axis=1) + 1e-12)
            if dmin < 1e-6:
                continue

            new_centres = centres.copy()
            new_centres[i] = new_pos

            new_radii, new_sum = solve_radii(new_centres)

            # acceptance rule
            if new_sum > cur_sum:
                centres, radii, cur_sum = new_centres, new_radii, new_sum
            else:
                # simulated‑annealing style acceptance
                T = 0.015 * (1.0 - it / max_iter) + 1e-5
                prob = np.exp((new_sum - cur_sum) / T)
                if rng.random() < prob:
                    centres, radii, cur_sum = new_centres, new_radii, new_sum

        # keep the best result over all restarts
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centres.copy()
            best_radii = radii.copy()

    # ----------------------------------------------------------
    # Tiny safety margin – shrink all radii by a negligible epsilon
    # ----------------------------------------------------------
    eps = 1e-9
    best_radii = np.maximum(best_radii - eps, 0.0)
    best_sum = float(np.sum(best_radii))

    return best_centers, best_radii, best_sum
