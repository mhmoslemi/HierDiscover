```python
import numpy as np
from scipy.optimize import linprog

def optimal_radii(centers):
    n = centers.shape[0]
    # compute pairwise distances
    # Use broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]  # shape (n,n,2)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub
    # for each pair i<j: r_i + r_j <= d_ij
    m = n*(n-1)//2 + 4*n
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    row = 0
    for i in range(n):
        for j in range(i+1, n):
            A_ub[row, i] = 1
            A_ub[row, j] = 1
            b_ub[row] = dists[i, j]
            row += 1
    # boundary constraints: r_i <= x_i, etc.
    for i in range(n):
        A_ub[row, i] = 1
        b_ub[row] = centers[i,0]  # x_i
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = centers[i,1]  # y_i
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = 1.0 - centers[i,0]
        row += 1
        A_ub[row, i] = 1
        b_ub[row] = 1.0 - centers[i,1]
        row += 1
    # objective: maximize sum of radii => minimize -sum(r)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # In case of numerical issues, fallback to zeros
        radii = np.zeros(n)
    else:
        radii = res.x
        # Clip small negative values
        radii = np.maximum(radii, 0.0)
    return radii
```

The problem can be written as a linear program for a *fixed* set of circle centres \((x_i,y_i)\).  
For each circle we introduce a radius \(r_i\ge 0\). The constraints are  

* **boundary**  \(r_i \le x_i,\; r_i \le y_i,\; r_i \le 1-x_i,\; r_i \le 1-y_i\)  
* **non‑overlap** \(r_i+r_j \le d_{ij}\) with \(d_{ij}= \|c_i-c_j\|_2\).

For a given centre configuration the LP  

```
max Σ r_i
s.t.  A_ub · r ≤ b_ub ,   r ≥ 0
```

yields the *largest possible* sum of radii for that configuration – it is exact because the
objective and all constraints are linear in the radii.

Thus the whole packing problem becomes a *continuous optimisation* over the centre
coordinates only.  The objective is the optimum value of the LP described above.
We solve it with a simple simulated‑annealing / hill‑climbing scheme:

1. **Pre‑compute the linear part** – the matrix `A_ub` does not depend on the centres,
   only on the pair indices.  It is built once for \(N=26\).

2. **LP sub‑routine** – given a centre array we compute all pairwise Euclidean distances,
   fill the right‑hand side `b_ub` (pair distances + distances to the four square sides)  
   and call `scipy.optimize.linprog` (HiGHS) to obtain the optimal radii and their sum.

3. **Initial guesses** – a Sobol low‑discrepancy sequence gives a well‑spread start,
   followed by several random restarts.

4. **Neighbour move** – pick a random circle, perturb its centre by a Gaussian step,
   evaluate the LP.  Accept the move if the sum of radii improves, otherwise accept it
   with a Metropolis probability \(\exp((\Delta)/T)\) where the temperature \(T\) is
   slowly cooled.

5. **Exploratory move** – with a smaller probability replace a centre by a completely
   new random point; this helps escape local minima.

6. **Iterate** – a few thousand LP evaluations per restart are cheap (the LP has only

7. **Return** – after the search the optimal radii for the best centre set are
   recomputed once more (to avoid any floating‑point drift) and the triple
   *(centres, radii, sum_of_radii)* is returned.

N = 26                         # number of circles
pair_i, pair_j = np.triu_indices(N, k=1)   # all i < j pairs
NUM_PAIRS = len(pair_i)                     # 325 for N=26
M = NUM_PAIRS + 4 * N                       # total number of inequality rows

# Build the constant matrix A_ub (shape M x N)
A_ub = np.zeros((M, N), dtype=np.float64)
row = 0
# pair constraints: r_i + r_j <= d_ij
for i, j in zip(pair_i, pair_j):
    A_ub[row, i] = 1.0
    A_ub[row, j] = 1.0
    row += 1
# boundary constraints: r_i <= x_i, r_i <= y_i, r_i <= 1-x_i, r_i <= 1-y_i
for i in range(N):
    A_ub[row, i] = 1.0   # x_i
    row += 1
    A_ub[row, i] = 1.0   # y_i
    row += 1
    A_ub[row, i] = 1.0   # 1 - x_i
    row += 1
    A_ub[row, i] = 1.0   # 1 - y_i
    row += 1

# Objective vector (we maximise Σ r_i → minimise -Σ r_i)
c_vec = -np.ones(N, dtype=np.float64)

# ----------------------------------------------------------------------
def optimal_radii(centres: np.ndarray) -> np.ndarray:
    """
    Solve the linear program for a fixed set of centre coordinates.
    Returns the optimal radii (length N).
    """
    # pairwise Euclidean distances
    diff = centres[:, None, :] - centres[None, :, :]          # (N,N,2)
    dists = np.sqrt(np.sum(diff * diff, axis=2))            # (N,N)

    # right‑hand side vector b_ub
    b = np.empty(M, dtype=np.float64)
    # pair distances first
    b[:NUM_PAIRS] = dists[pair_i, pair_j]
    # then four boundary vectors (x, y, 1‑x, 1‑y)
    start = NUM_PAIRS
    b[start:start + N] = centres[:, 0]               # x_i
    start += N
    b[start:start + N] = centres[:, 1]               # y_i
    start += N
    b[start:start + N] = 1.0 - centres[:, 0]         # 1 - x_i
    start += N
    b[start:start + N] = 1.0 - centres[:, 1]         # 1 - y_i

    # Solve the LP
    res = linprog(c_vec, A_ub=A_ub, b_ub=b,
                  bounds=[(0.0, None)] * N,
                  method='highs',
                  options={'presolve': True, 'disp': False})

    if not res.success:
        # Should never happen for a feasible set of centres, but guard anyway
        return np.zeros(N, dtype=np.float64)

    # Numerical noise can give tiny negatives – clip them
    radii = np.maximum(res.x, 0.0)
    return radii


def sum_and_radii(centres: np.ndarray):
    """
    Return (sum_of_radii, radii_array) for the given centre positions.
    """
    r = optimal_radii(centres)
    return float(r.sum()), r


# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Main routine: returns (centers, radii, sum_of_radii) for 26 circles.
    The algorithm is a simulated‑annealing hill‑climber that repeatedly
    solves the LP defined above.
    """
    rng = np.random.default_rng(20231129)

    # ------------------------------------------------------------------
    # 1) First deterministic start: Sobol low‑discrepancy points (if available)
    try:
        from scipy.stats import qmc
        sobol = qmc.Sobol(d=2, scramble=True, seed=12345)
        pts = sobol.random_base2(m=5)                 # 2^5 = 32 points
        start_centres = pts[:N].copy()
    except Exception:
        # Fallback: plain uniform random points
        start_centres = rng.random((N, 2))

    # Keep all points away from the exact border to avoid zero radii initially
    start_centres = np.clip(start_centres, 0.01, 0.99)

    # Evaluate the first solution
    best_sum, best_radii = sum_and_radii(start_centres)
    best_centres = start_centres.copy()

    # ------------------------------------------------------------------
    # 2) Meta‑heuristic parameters
    RESTARTS = 5
    ITER_PER_RESTART = 2000
    T_INITIAL = 0.05          # initial temperature for Metropolis acceptance

    for restart in range(RESTARTS):
        # Random fresh start for this restart
        centres = rng.random((N, 2))
        centres = np.clip(centres, 0.01, 0.99)

        cur_sum, cur_radii = sum_and_radii(centres)
        local_best_sum = cur_sum
        local_best_centres = centres.copy()

        for it in range(ITER_PER_RESTART):
            # Linear cooling schedule
            T = T_INITIAL * (1.0 - it / ITER_PER_RESTART)

            # Choose the type of move
            if rng.random() < 0.75:
                # Small Gaussian perturbation of a single centre
                i = rng.integers(N)
                old = centres[i].copy()
                sigma = 0.03 * (1.0 - it / ITER_PER_RESTART) + 0.005
                new = old + rng.normal(scale=sigma, size=2)
                new = np.clip(new, 0.0, 1.0)
                centres[i] = new
                new_sum, _ = sum_and_radii(centres)

                accept = (new_sum > cur_sum or
                          rng.random() < math.exp((new_sum - cur_sum) / (T + 1e-12)))
                if accept:
                    cur_sum = new_sum
                    if cur_sum > local_best_sum:
                        local_best_sum = cur_sum
                        local_best_centres = centres.copy()
                else:
                    centres[i] = old
            else:
                # Replace a random centre by a brand‑new location (global jump)
                i = rng.integers(N)
                old = centres[i].copy()
                centres[i] = rng.random(2)               # completely new point
                new_sum, _ = sum_and_radii(centres)

                accept = (new_sum > cur_sum or
                          rng.random() < math.exp((new_sum - cur_sum) / (T + 1e-12)))
                if accept:
                    cur_sum = new_sum
                    if cur_sum > local_best_sum:
                        local_best_sum = cur_sum
                        local_best_centres = centres.copy()
                else:
                    centres[i] = old

        # End of this restart – keep the best configuration seen
        if local_best_sum > best_sum:
            best_sum = local_best_sum
            best_centres = local_best_centres.copy()
            best_radii = optimal_radii(best_centres)

    # One final clean evaluation (removes any floating‑point drift)
    best_radii = optimal_radii(best_centres)
    best_sum = float(best_radii.sum())

    return best_centres, best_radii, best_sum
