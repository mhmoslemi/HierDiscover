```
def compute_radii(centers):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r >= 0]
    # boundary constraints
    constraints += [r <= centers[:,0]]
    constraints += [r <= 1 - centers[:,0]]
    constraints += [r <= centers[:,1]]
    constraints += [r <= 1 - centers[:,1]]
    # pairwise constraints
    for i in range(n):
        for j in range(i+1,n):
            d = np.linalg.norm(centers[i]-centers[j])
            constraints.append(r[i] + r[j] <= d)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    radii = np.maximum(r.value, 0)  # clip negative
    # enforce small negative values to zero
    return radii, radii.sum()
```

```
def compute_radii(centers):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r >= 0]
    # boundary
    constraints += [r <= centers[:,0]]
    constraints += [r <= 1 - centers[:,0]]
    constraints += [r <= centers[:,1]]
    constraints += [r <= 1 - centers[:,1]]
    # pairwise
    # compute distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # only i<j
    for i in range(n):
        for j in range(i+1, n):
            constraints.append(r[i] + r[j] <= dists[i, j])
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, max_iters=500)
    radii = np.maximum(r.value, 0)
    sum_r = float(np.sum(radii))
    return radii, sum_r
```

```
def run_packing():
    import numpy as np, math
    # grid of 5x5
    spacing = 0.2
    offset = 0.1
    centers = []
    radii = []
    for i in range(5):
        for j in range(5):
            centers.append([offset + spacing*j, offset + spacing*i])
            radii.append(0.1)
    # add extra small circle at (0.2,0.2)
    extra_center = [0.2, 0.2]
    extra_radius = math.sqrt(2) * 0.1 - 0.1 - 1e-6  # safe
    centers.append(extra_center)
    radii.append(extra_radius)
    centers = np.array(centers, dtype=float)
    radii = np.array(radii, dtype=float)
    sum_r = float(radii.sum())
    return centers, radii, sum_r
```

```python
import numpy as np
import math

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a feasible packing of 26 circles inside the unit square.
    The construction is a 5×5 regular grid of circles with radius 0.1,
    plus one small extra circle placed in the lower‑left vacant cell.
    """
    # parameters for the regular grid
    grid_spacing = 0.2          # distance between neighbouring centres
    margin = 0.1                # distance of the outermost centres from the square sides
    radius_grid = 0.1           # radius of the grid circles

    # generate the 5×5 grid points
    centers = []
    radii = []
    for i in range(5):
        y = margin + i * grid_spacing
        for j in range(5):
            x = margin + j * grid_spacing
            centers.append([x, y])
            radii.append(radius_grid)

    # extra circle in the centre of the lower‑left cell
    extra_center = [margin + grid_spacing / 2, margin + grid_spacing / 2]  # (0.2, 0.2)
    # maximal radius limited by the four neighbouring grid circles
    # distance to each neighbour = sqrt(0.1^2 + 0.1^2) = sqrt(2)*0.1
    # so r_extra = sqrt(2)*0.1 - 0.1, we subtract a tiny epsilon for safety
    epsilon = 1e-6
    r_extra = math.sqrt(2) * radius_grid - radius_grid - epsilon

    centers.append(extra_center)
    radii.append(r_extra)

    # convert to numpy arrays
    centers = np.array(centers, dtype=float)
    radii = np.array(radii, dtype=float)

    # total sum of radii
    sum_radii = float(np.sum(radii))

    return centers, radii, sum_radii
```
