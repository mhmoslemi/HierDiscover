```python
import cvxpy as cp
r = cp.Variable(N)
constraints = []
# Non-negative radii
constraints += [r >= 0]
# Boundary constraints
constraints += [centers[:,0] - r >= 0]
constraints += [centers[:,0] + r <= 1]
constraints += [centers[:,1] - r >= 0]
constraints += [centers[:,1] + r <= 1]
# Pairwise non-overlap
for i in range(N):
    for j in range(i+1,N):
        d = np.linalg.norm(centers[i] - centers[j])
        constraints += [r[i] + r[j] <= d]
objective = cp.Maximize(cp.sum(r))
prob = cp.Problem(objective, constraints)
prob.solve(solver=cp.ECOS, warm_start=True)
r_opt = r.value
```

```
def solve_lp(centers, delta=1e-7):
    N = centers.shape[0]
    r = cp.Variable(N)
    constraints = [r >= 0]
    # boundaries
    constraints += [centers[:,0] - r >= delta]  # x - r >= delta
    constraints += [centers[:,0] + r <= 1 - delta]
    constraints += [centers[:,1] - r >= delta]
    constraints += [centers[:,1] + r <= 1 - delta]
    # pairwise
    for i in range(N):
        for j in range(i+1,N):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d - delta)
    # objective
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    radii = r.value
    if radii is None:
        # infeasible; return zeros
        radii = np.zeros(N)
    else:
        radii = np.maximum(radii, 0) # clip negative
    return radii
```

```
def compute_radii(centers, delta):
    N = len(centers)
    # Build A_ub, b_ub
    # Each constraint row: a.dot(r) <= b.
    # Number of constraints: boundaries: 4*N
    # Pairwise: N*(N-1)//2
    M = 4*N + N*(N-1)//2
    A_rows = []
    b = []
    # bound constraints
    for i in range(N):
        xi, yi = centers[i]
        # left bound: r_i <= xi - delta
        row = np.zeros(N)
        row[i] = 1
        A_rows.append(row)
        b.append(xi - delta)
        # right bound: r_i <= 1 - xi - delta
        row = np.zeros(N)
        row[i] = 1
        A_rows.append(row)
        b.append(1 - xi - delta)
        # bottom bound: r_i <= yi - delta
        row = np.zeros(N)
        row[i] = 1
        A_rows.append(row)
        b.append(yi - delta)
        # top bound: r_i <= 1 - yi - delta
        row = np.zeros(N)
        row[i] = 1
        A_rows.append(row)
        b.append(1 - yi - delta)
    # pairwise constraints
    for i in range(N):
        for j in range(i+1, N):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(N)
            row[i] = 1
            row[j] = 1
            A_rows.append(row)
            b.append(d - delta)
    A_ub = np.vstack(A_rows)
    b_ub = np.array(b)
    c = -np.ones(N)  # maximize sum

    # bounds for each variable r_i >= 0
    bounds = [(0, None) for _ in range(N)]
    # Use linprog
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        radii = np.zeros(N)
    else:
        radii = res.x
        # Clip small negatives due to precision: ensure >=0
        radii = np.maximum(radii, 0)
    return radii
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    N = 26
    delta = 1e-6
    np.random.seed(0)

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # precomputed grid positions
    grid_x = np.linspace(0.1, 0.9, 5)  # 5 positions
    grid_y = np.linspace(0.1, 0.9, 5)
    grid = np.array(np.meshgrid(grid_x, grid_y)).T.reshape(-1,2) # 25 points

    # We will use this base grid, plus one additional random point
    for restart in range(10):
        # initial centers: start from grid plus a random point
        if restart == 0:
            centers = np.vstack([grid, np.array([[0.5, 0.5]])])  # use center as extra
        else:
            # random perturbations of grid
            perturb = np.random.uniform(-0.02, 0.02, size=grid.shape)
            base = np.clip(grid + perturb, 0.0, 1.0)
            extra = np.random.rand(1,2)
            centers = np.vstack([base, extra])
        # ensure shape (N,2)
        assert centers.shape == (N,2)

        # compute radii
        radii = compute_radii(centers, delta)
        cur_sum = np.sum(radii)
        cur_centers = centers.copy()
        cur_radii = radii.copy()
        # hill climbing
        step = 0.05
        for it in range(500):
            # choose random circle to move
            idx = np.random.randint(N)
            # propose new position
            new_center = cur_centers[idx] + np.random.normal(scale=step, size=2)
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = cur_centers.copy()
            new_centers[idx] = new_center
            # compute radii
            new_radii = compute_radii(new_centers, delta)
            new_sum = np.sum(new_radii)
            if new_sum > cur_sum + 1e-9:
                cur_centers = new_centers
                cur_radii = new_radii
                cur_sum = new_sum
                # update best
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centers.copy()
                    best_radii = new_radii.copy()
            # anneal step size
            step *= 0.9995
        # after inner loop maybe update best if current better
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()
    # final compute radii for best_centers (to ensure margin)
    best_radii = compute_radii(best_centers, delta)
    best_sum = np.sum(best_radii)
    return best_centers, best_radii, float(best_sum)
```

```
def compute_radii(centers, delta=1e-6):
    N = centers.shape[0]
    # number of constraints
    n_constraints = 4 * N + N * (N - 1) // 2
    A = np.zeros((n_constraints, N))
    b = np.zeros(n_constraints)
    row = 0
    # boundary constraints
    for i in range(N):
        x, y = centers[i]
        # left
        A[row, i] = 1
        b[row] = x - delta
        row += 1
        # right
        A[row, i] = 1
        b[row] = 1 - x - delta
        row += 1
        # bottom
        A[row, i] = 1
        b[row] = y - delta
        row += 1
        # top
        A[row, i] = 1
        b[row] = 1 - y - delta
        row += 1
    # pairwise constraints
    for i in range(N):
        for j in range(i+1, N):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row, i] = 1
            A[row, j] = 1
            b[row] = d - delta
            row += 1

    # objective: maximize sum r -> minimize -sum r
    c = -np.ones(N)
    bounds = [(0, None) for _ in range(N)]
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # if fail, return zeros
        return np.zeros(N)
    radii = res.x
    # Clip to non-negative
    radii = np.maximum(radii, 0.0)
    return radii
```

```python
import numpy as np
from scipy.optimize import linprog

def _max_radii_for_centers(centers: np.ndarray, delta: float = 1e-6) -> np.ndarray:
    """
    Solve the linear programme that maximises the sum of radii for a given
    set of centre points.  A small safety margin `delta` is subtracted from
    every inequality so that the returned radii are safely feasible
    with respect to the verifier's 1e-12 tolerance.
    """
    n = centers.shape[0]

    # Number of inequality constraints:
    #   4 per circle (left, right, bottom, top)  +
    #   n*(n-1)/2 pairwise non‑overlap constraints.
    m = 4 * n + n * (n - 1) // 2

    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]

        # x - r >= delta   ->   r <= x - delta
        A[row, i] = 1.0
        b[row] = x - delta
        row += 1

        # x + r <= 1 - delta   ->   r <= 1 - x - delta
        A[row, i] = 1.0
        b[row] = 1.0 - x - delta
        row += 1

        # y - r >= delta   ->   r <= y - delta
        A[row, i] = 1.0
        b[row] = y - delta
        row += 1

        # y + r <= 1 - delta   ->   r <= 1 - y - delta
        A[row, i] = 1.0
        b[row] = 1.0 - y - delta
        row += 1

    # Pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d - delta
            row += 1

    # Objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # Use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # In the unlikely event of numerical failure, fall back to zeros.
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(res.x, 0.0)   # eliminate tiny negative entries
    return radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, total_sum) for 26 circles packed inside the unit
    square.  The algorithm starts from a regular grid, repeatedly moves a
    randomly chosen centre, re‑optimises the radii by linear programming and
    keeps any improving move.  Several independent restarts are performed.
    """
    np.random.seed(0)          # deterministic result for reproducibility
    N = 26
    delta = 1e-6               # safety margin for the LP

    # ----- build a regular 5×5 grid (25 points) -------------------------
    grid_vals = np.linspace(0.1, 0.9, 5)          # 0.1,0.3,0.5,0.7,0.9
    gx, gy = np.meshgrid(grid_vals, grid_vals)
    base_grid = np.column_stack([gx.ravel(), gy.ravel()])   # shape (25,2)

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ----- optimisation parameters ---------------------------------------
    restarts = 8               # independent starts
    inner_iters = 1200        # hill‑climbing steps per start
    step_initial = 0.05       # initial Gaussian move scale

    for restart in range(restarts):
        # ---- initialise centres ----------------------------------------
        if restart == 0:
            # first start: exact grid + a point in the centre
            centres = np.vstack([base_grid, np.array([[0.5, 0.5]])])
        else:
            # later starts: perturb the grid a little and add one random point
            perturb = np.random.uniform(-0.04, 0.04, size=base_grid.shape)
            grid_perturbed = np.clip(base_grid + perturb, 0.0, 1.0)
            extra = np.random.rand(1, 2)
            centres = np.vstack([grid_perturbed, extra])

        # ---- solve LP for the initial layout ---------------------------
        radii = _max_radii_for_centers(centres, delta=delta)
        cur_sum = float(np.sum(radii))
        cur_centres = centres.copy()
        cur_radii = radii.copy()

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centres.copy()
            best_radii = cur_radii.copy()

        # ---- hill climbing ---------------------------------------------
        step = step_initial
        for it in range(inner_iters):
            i = np.random.randint(N)          # pick one circle to move
            proposal = cur_centres[i] + np.random.normal(scale=step, size=2)
            proposal = np.clip(proposal, 0.0, 1.0)   # keep it inside the square

            new_centres = cur_centres.copy()
            new_centres[i] = proposal

            new_radii = _max_radii_for_centers(new_centres, delta=delta)
            new_sum = float(np.sum(new_radii))

            # accept only genuine improvements (tiny tolerance avoids noise)
            if new_sum > cur_sum + 1e-9:
                cur_centres = new_centres
                cur_radii = new_radii
                cur_sum = new_sum

                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centres.copy()
                    best_radii = new_radii.copy()

            # gradually shrink the step size (simulated‑annealing style)
            step *= 0.9995

    # Final safety recomputation (ensures the radii correspond exactly to the
    # finally stored centres)
    best_radii = _max_radii_for_centers(best_centers, delta=delta)
    best_sum = float(np.sum(best_radii))

    return best_centers, best_radii, best_sum
```
