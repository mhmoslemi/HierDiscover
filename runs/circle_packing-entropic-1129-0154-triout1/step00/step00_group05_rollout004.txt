def solve_for_centers(centers):
    n = len(centers)
    w = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # can compute w_i minimal distance to any boundary
    # Build constraint matrix A_ub (list of rows)
    rows = []
    b = []
    # wall constraints
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(w[i] - 1e-9)
    # pair constraints
    for i in range(n):
        for j in range(i+1,n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(d - 1e-9)
    A_ub = np.vstack(rows)
    b = np.array(b)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None) for _ in range(n)]  # but we still have w_i constraints, so we can set upper bound large
    # Actually we already have explicit constraints for walls, so upper bound as None.
    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: all zeros
        return np.zeros(n)
    radii = res.x
    # Ensure nonnegative and not exceed w_i (in case of numerical)
    radii = np.maximum(radii, 0)
    radii = np.minimum(radii, w)
    return radii

```
def max_radii_given_centers(centers):
    n = centers.shape[0]
    w = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # number of constraints
    nwall = n
    npair = n*(n-1)//2
    m = nwall + npair
    A = np.zeros((m, n))
    b = np.empty(m)
    # wall constraints
    for i in range(n):
        A[i,i] = 1.0
        b[i] = w[i] - 1e-9
    # pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx,i] = 1.0
            A[idx,j] = 1.0
            d = np.linalg.norm(centers[i] - centers[j])
            b[idx] = d - 1e-9
            idx += 1
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: small radii
        return np.zeros(n)
    radii = res.x
    # Ensure non-negative
    radii = np.maximum(radii, 0.0)
    return radii
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog
    import time
    from math import sqrt

    n = 26
    eps = 1e-9
    rng = np.random.default_rng()

    def solve_lp(centers):
        n = centers.shape[0]
        w = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
        m = n + n*(n-1)//2
        A = np.zeros((m, n))
        b = np.empty(m)
        # walls
        for i in range(n):
            A[i,i] = 1.0
            b[i] = w[i] - eps
        # pairs
        row = n
        for i in range(n):
            for j in range(i+1, n):
                A[row,i] = 1.0
                A[row,j] = 1.0
                d = np.linalg.norm(centers[i] - centers[j])
                b[row] = d - eps
                row += 1
        c = -np.ones(n)
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # return minimal radii (zero)
            return np.zeros(n)
        radii = res.x
        radii = np.maximum(radii, 0.0)
        return radii

    # create initial candidates
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # number of random initial tries
    for _ in range(30):
        # generate via Halton if available else random
        try:
            from scipy.stats import qmc
            sampler = qmc.Halton(d=2, scramble=False)
            centers = sampler.random(n)
        except Exception:
            centers = rng.random((n,2))
        # optional small jitter to avoid alignment
        centers += (rng.random((n,2))-0.5) * 0.02
        centers = np.clip(centers, 0.0, 1.0)
        radii = solve_lp(centers)
        s = radii.sum()
        if s > best_sum:
            best_sum = s
            best_centers = centers.copy()
            best_radii = radii.copy()

    # hill climbing refinement
    start_time = time.time()
    time_limit = 1.8  # seconds
    while time.time() - start_time < time_limit:
        # Randomly select a circle to move
        i = rng.integers(n)
        # propose new location
        step = 0.05
        new_center = best_centers[i] + rng.normal(scale=step, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        centers_try = best_centers.copy()
        centers_try[i] = new_center
        radii_try = solve_lp(centers_try)
        sum_try = radii_try.sum()
        if sum_try > best_sum:
            best_sum = sum_try
            best_centers = centers_try
            best_radii = radii_try
        # optional: occasionally restart with new random centers
        if rng.random() < 0.001:
            # random restart
            centers = rng.random((n,2))
            radii = solve_lp(centers)
            s = radii.sum()
            if s > best_sum:
                best_sum = s
                best_centers = centers
                best_radii = radii

    # final slight shrink to be safe
    best_radii = np.maximum(best_radii - 1e-10, 0.0)
    return best_centers, best_radii, float(best_sum)
```

```python
import numpy as np
import time
from scipy.optimize import linprog
from typing import Tuple

# ----------------------------------------------------------------------
# Helper: solve the linear program for a fixed set of centres
# ----------------------------------------------------------------------
def solve_radii_for_centers(centers: np.ndarray, eps: float = 1e-9) -> np.ndarray:
    """
    Given centre positions (n,2) return the maximal radii (n,)
    that satisfy non‑overlap and square‑boundary constraints.
    A tiny epsilon is subtracted from every bound to keep a safety margin.
    """
    n = centers.shape[0]
    # distances to the four sides of the unit square
    wall_dist = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0],
         centers[:, 1], 1.0 - centers[:, 1]]
    )
    # number of inequality constraints: n (walls) + n*(n‑1)/2 (pairs)
    n_pairs = n * (n - 1) // 2
    m = n + n_pairs

    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    # wall constraints: r_i <= wall_dist_i - eps
    for i in range(n):
        A[i, i] = 1.0
        b[i] = wall_dist[i] - eps

    # pairwise non‑overlap constraints: r_i + r_j <= dist_ij - eps
    row = n
    for i in range(n):
        ci = centers[i]
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            d = np.linalg.norm(ci - centers[j])
            b[row] = d - eps
            row += 1

    # objective: maximise sum(r)  ==> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    # variable bounds: r_i >= 0, no explicit upper bound (walls already enforced)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the very unlikely case of infeasibility fall back to zeros
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(res.x, 0.0)          # enforce non‑negativity
    # Clip to wall distances for extra safety (numerical noise)
    radii = np.minimum(radii, wall_dist)
    return radii


# ----------------------------------------------------------------------
# Helper: generate several well‑spaced initial centre sets
# ----------------------------------------------------------------------
def generate_initial_sets(rng: np.random.Generator, n: int, count: int):
    """Yield up to *count* centre arrays using different low‑discrepancy methods."""
    produced = 0

    # 1. Halton sequence (if SciPy provides it)
    try:
        from scipy.stats import qmc
        halton = qmc.Halton(d=2, scramble=False)
        for _ in range(count // 3):
            pts = halton.random(n)
            # small jitter to avoid exact collinearity with the walls
            pts += (rng.random((n, 2)) - 0.5) * 0.02
            pts = np.clip(pts, 0.0, 1.0)
            yield pts
            produced += 1
    except Exception:
        pass

    # 2. Latin‑Hypercube (also from scipy.stats.qmc)
    try:
        from scipy.stats import qmc
        lhs = qmc.LatinHypercube(d=2)
        for _ in range(count // 3):
            pts = lhs.random(n)
            pts += (rng.random((n, 2)) - 0.5) * 0.02
            pts = np.clip(pts, 0.0, 1.0)
            yield pts
            produced += 1
    except Exception:
        pass

    # 3. Plain uniform random draws (fill the rest)
    while produced < count:
        pts = rng.random((n, 2))
        pts += (rng.random((n, 2)) - 0.5) * 0.02
        pts = np.clip(pts, 0.0, 1.0)
        yield pts
        produced += 1

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Pack 26 circles inside the unit square, maximising the sum of radii.
    Returns (centres, radii, total_radius_sum).
    """
    n_circles = 26
    rng = np.random.default_rng(2023)          # deterministic but good enough
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # --------------------------------------------------------------
    # Stage 1 – evaluate a handful of well‑spread initial configurations
    # --------------------------------------------------------------
    for centres in generate_initial_sets(rng, n_circles, count=30):
        radii = solve_radii_for_centers(centres)
        cur_sum = radii.sum()
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centres = centres.copy()
            best_radii = radii.copy()

    # --------------------------------------------------------------
    # Stage 2 – simple stochastic hill‑climbing to improve the layout
    # --------------------------------------------------------------
    start = time.time()
    time_limit = 1.8                     # seconds of optimisation
    step_scale = 0.06                    # typical move size

    while time.time() - start < time_limit:
        # pick a circle and propose a small move
        i = rng.integers(n_circles)
        trial = best_centres.copy()
        move = rng.normal(scale=step_scale, size=2)
        trial[i] = np.clip(trial[i] + move, 0.0, 1.0)

        # solve the LP for the perturbed layout
        radii_trial = solve_radii_for_centers(trial)
        sum_trial = radii_trial.sum()

        # accept only improving moves
        if sum_trial > best_sum:
            best_sum = sum_trial
            best_centres = trial
            best_radii = radii_trial

        # occasional random restart (tiny probability) to escape very bad plateaus
        if rng.random() < 0.001:
            trial = rng.random((n_circles, 2))
            trial = np.clip(trial, 0.0, 1.0)

            if sum_trial > best_sum:

                best_centres = trial
                best_radii = radii_trial

    # --------------------------------------------------------------

    # --------------------------------------------------------------
    best_radii = np.maximum(best_radii - 1e-10, 0.0)
```
