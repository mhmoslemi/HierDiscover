def solve_lp(centers):
    import numpy as np
    from scipy.optimize import linprog

    n = centers.shape[0]
    # objective: maximize sum(r) -> minimize -sum(r)
    c = -np.ones(n)

    # A_ub list and b_ub list
    A_ub = []
    b_ub = []

    # boundary constraints: r_i <= bound_i
    bounds = []

    # compute bound_i = min(x, 1-x, y, 1-y)
    for i in range(n):
        x, y = centers[i]
        bound_i = min(x, 1 - x, y, 1 - y)
        # Add constraint r_i <= bound_i => A row with 1 at i
        A_ub.append(np.eye(1, n, i).flatten()) # not efficient; can just append row array.
        b_ub.append(bound_i)
        bounds.append((0, None))

    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A_ub.append(row)
            b_ub.append(d)

    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r

import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n=centers.shape[0]
    c = -np.ones(n)
    # pairwise constraints
    # compute distances matrix
    dists = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)
    # construct A_ub rows
    rows = []
    b = []
    # boundary constraints
    for i in range(n):
        bound = min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(bound)
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(dists[i,j])
    A_ub = np.array(rows)
    b_ub = np.array(b)

    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = radii.sum()
    return radii, sum_r

import numpy as np
from typing import Tuple

# ----------------------------------------------------------------------
# Linear‑programming solver for a fixed set of centres.
# ----------------------------------------------------------------------
def _solve_lp(centres: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the LP:
        maximise   sum r_i
        subject to r_i >= 0,
                   r_i <= distance to the four sides,
                   r_i + r_j <= distance(centre_i, centre_j)

    Returns the optimal radii and the sum of radii.
    """
    from scipy.optimize import linprog

    n = centres.shape[0]

    # ---- objective ( maximise sum r_i  == minimise -sum r_i )
    c = -np.ones(n)

    # ---- pairwise distance constraints  r_i + r_j <= d_ij
    # compute the full distance matrix once
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)

    # ---- construct A_ub and b_ub
    rows = []
    rhs = []

    # boundary constraints: r_i <= min(x,1-x,y,1-y)
    for i in range(n):
        xi, yi = centres[i]
        bound = min(xi, 1.0 - xi, yi, 1.0 - yi)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(bound)

    # non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # variable bounds (r_i >= 0, no explicit upper bound – already in A_ub)
    bounds = [(0.0, None) for _ in range(n)]

    # solve
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # fall back to zero radii (should never happen)
        radii = np.zeros(n)
    else:
        radii = res.x

    total = float(radii.sum())
    return radii, total


# ----------------------------------------------------------------------
# Main optimisation routine
# ----------------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns
    -------
    centres : (26,2) array of (x,y) positions
    radii   : (26,)   array of radii
    total   : float   sum of radii
    The arrangement satisfies the validator requirements.
    """
    np.random.seed(0)        # deterministic output

    # ---- start from a regular 5×5 grid (25 points)
    step = 0.2
    grid_coords = np.arange(0.1, 1.0, step)          # 0.1,0.3,0.5,0.7,0.9
    grid = np.array(
        [(x, y) for x in grid_coords for y in grid_coords]
    )                                                # shape (25,2)

    # ---- add a 26‑th point (initially random, but not on the grid)
    extra = np.random.rand(1, 2) * 0.8 + 0.1          # keep away from the very border
    centres = np.vstack([grid, extra])               # (26,2)

    # initial LP solution
    radii, best_total = _solve_lp(centres)

    # ---- simple stochastic local search
    max_iter = 800
    step_size = 0.07
    for it in range(max_iter):
        idx = np.random.randint(0, 26)

        # occasional large jump (10 % of the time)
        if np.random.rand() < 0.1:
            new_pos = np.random.rand(2)               # anywhere in the square
        else:
            delta = np.random.uniform(-step_size, step_size, size=2)
            new_pos = centres[idx] + delta
            new_pos = np.clip(new_pos, 0.0, 1.0)

        new_centres = centres.copy()
        new_centres[idx] = new_pos

        new_radii, new_total = _solve_lp(new_centres)

        # keep the change only if it improves the total radius sum
        if new_total > best_total + 1e-12:
            centres = new_centres
            radii = new_radii
            best_total = new_total

        # slowly shrink the step size
        step_size *= 0.998

    return centres, radii, best_total
