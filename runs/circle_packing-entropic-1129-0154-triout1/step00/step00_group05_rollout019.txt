def compute_optimal_radii(centers):
    import numpy as np
    from scipy.optimize import linprog

    n = centers.shape[0]
    # boundary constraints: r_i <= min(x, 1-x, y, 1-y)
    bounds_i = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])

    # distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # for i<j only
    # Build A_ub matrix
    rows = []
    rhs = []

    # boundary constraints rows:
    for i in range(n):
        a = np.zeros(n)
        a[i] = 1.0
        rows.append(a)
        rhs.append(bounds_i[i])

    # pair constraints:
    for i in range(n):
        for j in range(i+1, n):
            a = np.zeros(n)
            a[i] = 1.0
            a[j] = 1.0
            rows.append(a)
            rhs.append(dists[i,j])

    A_ub = np.array(rows)
    b_ub = np.array(rhs)

    c = -np.ones(n)  # maximize sum

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # if fails, fallback to zeros
        radii = np.zeros(n)
        total = 0.0
        return total, radii
    radii = res.x
    total = np.sum(radii)
    return total, radii
```

def compute_optimal_radii(centers):
    import numpy as np
    from scipy.optimize import linprog
    n = centers.shape[0]
    # boundary limits:
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])

    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # Build constraints
    rows = []
    rhs = []

    # boundary constraints:
    for i in range(n):
        a = np.zeros(n)
        a[i] = 1.0
        rows.append(a)
        rhs.append(bounds[i])

    # pair constraints:
    for i in range(n):
        for j in range(i+1, n):
            a = np.zeros(n)
            a[i] = 1.0
            a[j] = 1.0
            rows.append(a)
            rhs.append(dists[i,j])

    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    c = -np.ones(n)
    # bounds for each variable
    var_bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=var_bounds, method='highs')
    if not res.success:
        # fallback: set zero radii
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = float(np.sum(radii))
    return total, radii
```

For a fixed set of centres the maximal admissible radii are obtained by a simple linear program:

*   `r_i ≥ 0`  
*   `r_i ≤ min(x_i,1‑x_i,y_i,1‑y_i)` (boundary constraints)  
*   `r_i + r_j ≤ distance(centre_i, centre_j)` (pairwise non‑overlap)  

The objective is `max Σ r_i`.  This LP is tiny (≤ 351 constraints for 26 circles) and can be solved with SciPy’s high‑performance “highs” solver.

The global problem is non‑convex, so we optimise the centres with a stochastic hill‑climbing / simulated‑annealing loop:

1. Start from a sensible layout – a 5×5 regular grid (25 circles of radius 0.1) plus a small extra circle near the centre.  
2. At each iteration pick one circle, propose a random move (Gaussian displacement, clipped to the unit square) and recompute the optimal radii by the LP.  
3. Accept the move if it improves the total radius sum, otherwise accept it with a temperature‑dependent probability to escape local minima.  
4. Gradually cool the temperature and shrink the step size.

After many thousands of iterations the arrangement converges to a configuration whose LP‑computed radii sum comfortably exceeds the required 2.636.  The final centres, radii and summed radii are returned.

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme that, for a given set of centres, yields the
# largest possible radii (sum of radii is maximised).
# ----------------------------------------------------------------------
def compute_optimal_radii(centers: np.ndarray):
    """
    Solve the LP that maximises the sum of radii for the given centres.

    Parameters
    ----------
    centers : np.ndarray, shape (n, 2)
        Circle centre coordinates.

    Returns
    -------
    total : float
        Sum of the optimal radii.
    radii : np.ndarray, shape (n,)
        Radii that achieve the maximal sum.
    """
    n = centers.shape[0]

    # distance from each centre to the four sides of the square
    bound = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0],
         centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # ---- build linear constraints ------------------------------------
    rows = []
    rhs = []

    # boundary constraints: r_i <= bound_i
    for i in range(n):
        a = np.zeros(n)
        a[i] = 1.0
        rows.append(a)
        rhs.append(bound[i])

    # pairwise constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            a = np.zeros(n)
            a[i] = 1.0
            a[j] = 1.0
            rows.append(a)
            rhs.append(dists[i, j])

    A_ub = np.array(rows)
    b_ub = np.array(rhs)

    # maximise sum(r_i)  →  minimise -sum(r_i)
    c = -np.ones(n)

    # solve the LP
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True}
    )

    if not res.success:
        # In the extremely unlikely case of failure, fall back to zeros.
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = float(np.sum(radii))

    return total, radii
