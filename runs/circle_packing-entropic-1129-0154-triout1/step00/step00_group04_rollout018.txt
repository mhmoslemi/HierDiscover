import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    # Upper bounds: each radius <= min distance to boundaries
    ub = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Pair constraints
    rows = []
    rhs = []
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            rhs.append(dists[i,j])
    # Linear program: maximize sum r_i -> minimize -sum r_i
    c = -np.ones(n)
    # Add bound constraints as inequalities as well?
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    # Also we include bound constraints via bounds parameter:
    bounds = [(0, ub_i) for ub_i in ub]
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # In case of failure maybe return zeros
        return np.zeros(n)
    radii = res.x
    return radii

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    # upper bounds based on boundary
    ub = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Build constraint matrix
    num_pairs = n*(n-1)//2
    A = np.zeros((num_pairs + 4*n, n))
    b = np.zeros(num_pairs + 4*n)
    # pair constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = d[i, j]
            idx += 1
    # boundary constraints
    for i in range(n):
        # x_i <= ub_i
        # actually ub already bound, but we also need r_i <= distance to left side (x_i) and right side (1-x_i), etc.
        # we will set bounds via bounds parameter = (0, ub_i)
        pass
    # solve LP
    c = -np.ones(n)
    bounds = [(0, ub_i) for ub_i in ub]
    res = linprog(c, A_ub=A[:num_pairs], b_ub=b[:num_pairs], bounds=bounds, method='highs')
    if not res.success:
        # fallback to zero radii
        return np.zeros(n)
    radii = res.x
    # ensure non-negative and small epsilon reduction
    radii = np.maximum(radii, 0)
    radii = radii - 1e-9
    radii[radii < 0] = 0
    return radii
