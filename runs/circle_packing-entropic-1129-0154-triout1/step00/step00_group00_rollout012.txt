```
def solve_lp(centers):
    N = centers.shape[0]
    # boundaries
    bnds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # For i<j
    rows = []
    ub = []
    for i in range(N):
        for j in range(i+1, N):
            rows.append((i,j))
    # Build A_ub with shape (num_constraints, N)
    num_pairs = len(rows)
    A_ub = np.zeros((num_pairs + N, N))
    b_ub = np.zeros(num_pairs + N)
    # Pair constraints
    for idx, (i,j) in enumerate(rows):
        A_ub[idx, i] = 1.0
        A_ub[idx, j] = 1.0
        b_ub[idx] = dists[i,j]
    # Boundary constraints
    for i in range(N):
        A_ub[num_pairs + i, i] = 1.0
        b_ub[num_pairs + i] = bnds[i]
    # Objective
    c = -np.ones(N)
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if not res.success:
        # In case of failure, fallback to zeros radii
        radii = np.zeros(N)
        total = 0.0
    else:
        radii = res.x
        total = np.sum(radii)
    return radii, total
```

```
def compute_optimal_radii(centers):
    N = centers.shape[0]
    # boundaries distances
    bnd = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # constraints as described
    # build matrix
    idx_i, idx_j = np.triu_indices(N, k=1)  # returns upper triangle indices
    num_pairs = len(idx_i)
    A_ub = np.zeros((num_pairs + N, N), dtype=np.float64)
    b_ub = np.empty(num_pairs + N, dtype=np.float64)

    # pairs
    A_ub[np.arange(num_pairs), idx_i] = 1.0
    A_ub[np.arange(num_pairs), idx_j] = 1.0
    b_ub[:num_pairs] = dists[idx_i, idx_j]

    # boundary constraints
    A_ub[num_pairs + np.arange(N), np.arange(N)] = 1.0
    b_ub[num_pairs:] = bnd

    # objective
    c = -np.ones(N)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*N, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(N)
        total = 0.0
    else:
        radii = res.x
        total = radii.sum()
    return radii, total
```

Implementation:

```
def half_radii(centers):
    N = centers.shape[0]
    # boundary distances
    b = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # set diagonal to large number
    np.fill_diagonal(dists, np.inf)
    # min distance to other centers
    min_dist = np.min(dists, axis=1)
    radii = np.minimum(b, min_dist / 2.0)
    return radii, radii.sum()
```

Better to set diagonal to large to exclude self distances.

Now define search:

```
def improve_positions(initial_centers, rng, iterations=2000):
    best_centers = initial_centers.copy()
    radii, best_sum = half_radii(best_centers)
    N = best_centers.shape[0]
    sigma = 0.1
    for it in range(iterations):
        i = rng.integers(N)
        # propose new center
        new_center = best_centers[i] + rng.normal(scale=sigma, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not too close to boundaries? We'll let radii handle.
        new_centers = best_centers.copy()
        new_centers[i] = new_center
        _, new_sum = half_radii(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
            # optionally reduce sigma for next maybe?
        # decay sigma
        sigma *= 0.9999
        if sigma < 0.001:
            sigma = 0.001
    return best_centers, best_sum
```

```
def solve_lp(centers):
    N = centers.shape[0]
    # boundary distances
    b = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # get indices of pairs
    i_idx, j_idx = np.triu_indices(N, k=1)
    num_pairs = len(i_idx)
    A_ub = np.zeros((num_pairs + N, N), dtype=np.float64)
    b_ub = np.empty(num_pairs + N, dtype=np.float64)

    # pair constraints: r_i + r_j <= d_ij
    A_ub[np.arange(num_pairs), i_idx] = 1.0
    A_ub[np.arange(num_pairs), j_idx] = 1.0
    b_ub[:num_pairs] = dists[i_idx, j_idx]

    # boundary constraints: r_i <= b_i
    A_ub[num_pairs + np.arange(N), np.arange(N)] = 1.0
    b_ub[num_pairs:] = b

    c = -np.ones(N, dtype=np.float64)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)] * N, method='highs')
    if not res.success:
        # fallback to half radii if fails
        radii, s = half_radii(centers)
        return radii, s
    radii = res.x
    total = radii.sum()
    return radii, total
```

```
def local_search(centers, rng, iterations=2000):
    N = centers.shape[0]
    best_centers = centers.copy()
    radii, best_sum = half_radii(best_centers)
    sigma = 0.2
    for it in range(iterations):
        i = rng.integers(N)
        # propose new center
        new_center = best_centers[i] + rng.normal(scale=sigma, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        new_centers = best_centers.copy()
        new_centers[i] = new_center
        _, new_sum = half_radii(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
        # decay sigma
        sigma *= 0.9995
        if sigma < 0.001:
            sigma = 0.001
    return best_centers, best_sum
```

```python
import numpy as np
from scipy.optimize import linprog

def half_radii(centers: np.ndarray):
    """Fast lower‑bound radii: min(boundary distance, half nearest neighbour distance)."""
    n = centers.shape[0]
    # distance to the four sides
    side = np.minimum.reduce([centers[:, 0],
                              1.0 - centers[:, 0],
                              centers[:, 1],
                              1.0 - centers[:, 1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    # ignore self‑distance
    np.fill_diagonal(dists, np.inf)
    nearest = np.min(dists, axis=1) / 2.0
    radii = np.minimum(side, nearest)
    return radii, np.sum(radii)

def solve_lp(centers: np.ndarray):
    """Exact maximal radii for a fixed set of centres (LP)."""
    n = centers.shape[0]

    # boundary limits
    bnd = np.minimum.reduce([centers[:, 0],
                             1.0 - centers[:, 0],
                             centers[:, 1],
                             1.0 - centers[:, 1]])

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff ** 2, axis=2))

    # indices of all unordered pairs i<j
    i_idx, j_idx = np.triu_indices(n, k=1)
    n_pairs = i_idx.shape[0]

    # build A_ub * r <= b_ub
    A_ub = np.zeros((n_pairs + n, n), dtype=np.float64)
    b_ub = np.empty(n_pairs + n, dtype=np.float64)

    # r_i + r_j <= d_ij
    rows = np.arange(n_pairs)
    A_ub[rows, i_idx] = 1.0
    A_ub[rows, j_idx] = 1.0
    b_ub[rows] = d[i_idx, j_idx]

    # r_i <= boundary_i
    A_ub[n_pairs + np.arange(n), np.arange(n)] = 1.0
    b_ub[n_pairs:] = bnd

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub,
                  bounds=[(0.0, None)] * n,
                  method='highs')
    if not res.success:
        # fall back to the lower bound if LP fails for any reason
        rad, total = half_radii(centers)
        return rad, total
    rad = res.x
    return rad, np.sum(rad)

def improve_layout(initial_centers: np.ndarray,
                  rng: np.random.Generator,
                  iterations: int = 3000):
    """Hill‑climbing on the lower‑bound sum of radii."""
    best_centers = initial_centers.copy()
    _, best_score = half_radii(best_centers)
    n = best_centers.shape[0]
    sigma = 0.15  # initial perturbation size

    for it in range(iterations):
        i = rng.integers(n)               # which circle to move
        proposal = best_centers[i] + rng.normal(scale=sigma, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centers = best_centers.copy()
        new_centers[i] = proposal

        _, new_score = half_radii(new_centers)

        if new_score > best_score:
            best_score = new_score
            best_centers = new_centers

        # slowly shrink the step size
        sigma *= 0.9995
        if sigma < 0.001:
            sigma = 0.001

    return best_centers, best_score

def run_packing():
    """
    Returns:
        centers (np.ndarray shape (26,2))
        radii   (np.ndarray shape (26,))
        sum_radii (float)
    """
    rng = np.random.default_rng(123456789)

    # 1. Fixed corner circles (they naturally become large)
    corners = np.array([[0.125, 0.125],
                        [0.875, 0.125],
                        [0.125, 0.875],
                        [0.875, 0.875]])

    # 2. Fixed side‑mid points
    mids = np.array([[0.5, 0.1],
                     [0.5, 0.9],
                     [0.1, 0.5],
                     [0.9, 0.5]])

    # 3. Remaining 18 circles – initialise uniformly in the inner square [0.2,0.8]^2
    remaining = rng.uniform(0.2, 0.8, size=(26 - len(corners) - len(mids), 2))

    init_centers = np.vstack([corners, mids, remaining])

    # 4. Stochastic improvement of the centre positions
    best_centers, _ = improve_layout(init_centers, rng, iterations=4000)

    # 5. Exact LP optimisation of radii for the final centre set
    final_radii, total = solve_lp(best_centers)

    # Return as required
    return best_centers, final_radii, float(total)
```
