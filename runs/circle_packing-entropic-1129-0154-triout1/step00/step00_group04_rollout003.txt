import numpy as np
from scipy.optimize import linprog
from scipy.stats import qmc
import math

# ----------------------------------------------------------------------
# Linear programming: given centre coordinates, compute the maximal
# admissible radii that satisfy the square and non‑overlap constraints.
# ----------------------------------------------------------------------
def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises the sum of radii for a fixed set of centres.

    Parameters
    ----------
    centers : np.ndarray, shape (n, 2)
        (x, y) coordinates of the circle centres.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        The radii that maximise the total sum while respecting all constraints.
        If the LP fails, an array of zeros is returned.
    """
    n = centers.shape[0]

    # Number of inequality rows:
    #   4 per circle (left, right, bottom, top) + one per unordered pair.
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # Boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left wall:   r_i <= x
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # right wall:  r_i <= 1 - x
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # bottom wall: r_i <= y
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # top wall:    r_i <= 1 - y
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # Pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = np.linalg.norm(centers[i] - centers[j])
            row += 1

    # Objective: maximise sum(r_i)  → minimise -sum(r_i)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # Use the HiGHS solver (fast and reliable for small LPs)
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # In the extremely unlikely case of failure, fall back to zeros.
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(result.x, 0.0)  # Clip tiny negatives to zero
    return radii

def hexagonal_grid_centres(r: float = 0.10) -> np.ndarray:
    """
    Produce a hexagonal lattice inside the unit square with spacing
    2r (horizontal) and sqrt(3)*r (vertical).  This yields 23 points for r≈0.10.
    """
    step_x = 2.0 * r
    step_y = math.sqrt(3.0) * r

    # y coordinate runs from r to 1 - r (inclusive)
    y_vals = []
    y = r
    while y <= 1.0 - r + 1e-12:
        y_vals.append(y)
        y += step_y

    points = []
    for idx, y in enumerate(y_vals):
        # offset every other row by r (i.e. half‑step)
        offset = r if (idx % 2 == 1) else 0.0
        x = r + offset
        while x <= 1.0 - r + 1e-12:
            points.append((x, y))
            x += step_x

    return np.array(points, dtype=np.float64)


def sobol_centres(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Generate n points using a scrambled Sobol low‑discrepancy sequence.
    The resulting points are well spread over the unit square.
    """
    sampler = qmc.Sobol(d=2, scramble=True, seed=rng.integers(1 << 30))
    pts = sampler.random_base2(m=int(np.ceil(np.log2(n))))
    return pts[:n]


def random_centres(n: int, rng: np.random.Generator) -> np.ndarray:
    """Uniform random points."""
    return rng.random((n, 2))


def corner_biased_centres(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a set of centres where a few are placed close to the four corners
    (they tend to get relatively large radii) and the rest are uniform.

    # Number of corner‑biased points – at most four (one per corner)

        centres[i] = (x, y)

    if n > n_corner:
