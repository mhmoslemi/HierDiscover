def solve_lp(centers):
    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]  # shape (n,n,2)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Flatten constraints
    # A_ub rows count: N*(N-1)/2 + 4*N
    m = n*(n-1)//2 + 4*n
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i,j]
            row += 1
    # boundary constraints
    for i in range(n):
        # x_i - r_i >= 0 => r_i <= x_i
        A[row,i] = 1; b[row] = centers[i,0]; row+=1
        # 1 - x_i - r_i >=0 => r_i <= 1 - x_i
        A[row,i] = 1; b[row] = 1 - centers[i,0]; row+=1
        # y_i - r_i >=0 => r_i <= y_i
        A[row,i] = 1; b[row] = centers[i,1]; row+=1
        # 1 - y_i - r_i >=0 => r_i <= 1 - y_i
        A[row,i] = 1; b[row] = 1 - centers[i,1]; row+=1
    # Objective: maximize sum r_i => minimize -sum r_i
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: assign zeros
        return np.zeros(n)
    r = res.x
    # Ensure small epsilon removal
    eps = 1e-9
    r = np.maximum(r - eps, 0.0)
    return r

def hill_climb(initial_centers, steps=2000, seed=0):
    rng = np.random.default_rng(seed)
    n = initial_centers.shape[0]
    centers = initial_centers.copy()
    radii = solve_lp(centers)
    best_sum = radii.sum()
    best_centers = centers.copy()
    best_radii = radii.copy()
    step_scale = 0.1  # initial max move
    for it in range(steps):
        # gradually reduce step size
        scale = step_scale * (1 - it/steps)
        i = rng.integers(n)
        new_center = centers[i] + rng.uniform(-scale, scale, size=2)
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not too close to borders such that radius may be zero: we can keep as is
        temp_centers = centers.copy()
        temp_centers[i] = new_center
        temp_radii = solve_lp(temp_centers)
        temp_sum = temp_radii.sum()
        if temp_sum > best_sum + 1e-9:  # accept improvement
            centers[i] = new_center
            radii = temp_radii
            best_sum = temp_sum
            best_centers = centers.copy()
            best_radii = radii.copy()
    return best_centers, best_radii, best_sum

def simulated_annealing(initial_centers, steps=5000, seed=0):
    rng = np.random.default_rng(seed)
    centers = initial_centers.copy()
    radii = solve_lp(centers)
    best_sum = radii.sum()
    best = (centers.copy(), radii.copy())
    T0 = 0.01  # initial temperature
    for it in range(steps):
        T = T0 * (1 - it/steps)  # linear cooling
        # randomly choose one circle to move
        i = rng.integers(26)
        # propose new location via Gaussian displacement
        delta = rng.normal(scale=0.05, size=2)
        new_center = centers[i] + delta
        new_center = np.clip(new_center, 0.0, 1.0)
        new_centers = centers.copy()
        new_centers[i] = new_center
        new_radii = solve_lp(new_centers)
        new_sum = new_radii.sum()
        delta_sum = new_sum - best_sum
        if delta_sum > 0 or rng.random() < np.exp(delta_sum / max(T,1e-12)):
            centers = new_centers
            radii = new_radii
            if new_sum > best_sum:
                best_sum = new_sum
                best = (centers.copy(), radii.copy())
    return best[0], best[1], best_sum

def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    def solve_lp(centers):
        n = centers.shape[0]
        # compute distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.linalg.norm(diff, axis=2)
        # Build constraints
        m = n*(n-1)//2 + 4*n
        A = np.zeros((m, n))
        b = np.zeros(m)
        row = 0
        # pairwise
        for i in range(n):
            for j in range(i+1, n):
                A[row, i] = 1
                A[row, j] = 1
                b[row] = dists[i,j]
                row += 1
        # boundary
        for i in range(n):
            # left
            A[row, i] = 1
            b[row] = centers[i,0]
            row += 1
            # right
            A[row, i] = 1
            b[row] = 1 - centers[i,0]
            row += 1
            # bottom
            A[row, i] = 1
            b[row] = centers[i,1]
            row += 1
            # top
            A[row, i] = 1
            b[row] = 1 - centers[i,1]
            row += 1
        # objective
        c = -np.ones(n)
        bounds = [(0, None)] * n
        # use method high
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # fallback
            return np.zeros(n)
        radii = res.x
        # subtract epsilon to satisfy constraints strictly
        eps = 1e-9
        radii = np.maximum(radii - eps, 0.0)
        return radii

    # hill climbing / simulated annealing function

    # Initial positions: jittered grid + extra point
    rng = np.random.default_rng(12345)
    # create jittered grid

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programming: given centre positions compute the optimal radii
# ----------------------------------------------------------------------
def solve_lp(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises the sum of radii for a fixed set of centres.
    Returns an array of radii (shape (n,)).
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of inequality constraints:
    #   one for each pair (i<j)  + 4 per circle (boundary)
    m = n * (n - 1) // 2 + 4 * n
    A = np.zeros((m, n), dtype=float)
    b = np.zeros(m, dtype=float)

    row = 0
    # pairwise constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        # left side
        A[row, i] = 1.0
        b[row] = centers[i, 0]
        row += 1
        # right side
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 0]
        row += 1
        # bottom side
        A[row, i] = 1.0
        b[row] = centers[i, 1]
        row += 1
        # top side
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 1]
        row += 1

    # maximise sum(r)  == minimise -sum(r)
    c = -np.ones(n, dtype=float)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback – every radius zero (should never happen)
        return np.zeros(n, dtype=float)

    # push radii a tiny bit inside the feasible region to avoid 1e‑12 tolerance issues
    eps = 1e-9
    radii = np.maximum(res.x - eps, 0.0)
    return radii


# ----------------------------------------------------------------------
# Build an initial set of centres (jittered 5×5 grid + one random point)
# ----------------------------------------------------------------------
def initial_centres(rng: np.random.Generator) -> np.ndarray:
    base = np.linspace(0.1, 0.9, 5)          # ideal grid positions
    pts = []
    for x in base:
        for y in base:
            # small jitter (±0.02) keeps the points inside the square
            jitter = rng.uniform(-0.02, 0.02, size=2)
            cx = np.clip(x + jitter[0], 0.0, 1.0)
            cy = np.clip(y + jitter[1], 0.0, 1.0)
            pts.append([cx, cy])
    pts = np.array(pts)                     # 25 points
    # one extra point placed away from the grid centre to avoid complete overlap
    extra = rng.uniform(0.2, 0.8, size=2)
    pts = np.vstack([pts, extra])
    return pts                               # shape (26,2)


# ----------------------------------------------------------------------
# Simple stochastic hill‑climbing / simulated annealing
# ----------------------------------------------------------------------
def optimise(centres: np.ndarray,
             rng: np.random.Generator,
             steps: int = 2000,
             max_step: float = 0.15) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a greedy stochastic search on the centre positions.
    Returns the best (centres, radii, total_radius) found.
    """
    n = centres.shape[0]
    best_c = centres.copy()
    best_r = solve_lp(best_c)
    best_sum = best_r.sum()

    cur_c = best_c.copy()
    cur_r = best_r.copy()
    cur_sum = best_sum

    for it in range(steps):
        # linear cooling schedule for the step size
        step = max_step * (1.0 - it / steps)

        # pick a circle and propose a new position
        i = rng.integers(n)
        proposal = cur_c[i] + rng.uniform(-step, step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_c = cur_c.copy()
        new_c[i] = proposal
        new_r = solve_lp(new_c)
        new_sum = new_r.sum()

        # accept if the total radius improves (tiny tolerance used)
        if new_sum > cur_sum + 1e-12:
            cur_c, cur_r, cur_sum = new_c, new_r, new_sum
            if new_sum > best_sum + 1e-12:
                best_c, best_r, best_sum = new_c.copy(), new_r.copy(), new_sum

    return best_c, best_r, best_sum


# ----------------------------------------------------------------------
# Main entry point required by the platform
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, total_radius) for 26 circles.
    The algorithm uses a jittered grid as a starting point and
    improves it with a short stochastic optimisation.
    """
    rng = np.random.default_rng(123456)   # deterministic seed

    # start from a jittered regular grid
    start = initial_centres(rng)

    # a few independent optimisation runs – keep the best result
    best_centers, best_radii, best_total = optimise(start, rng, steps=2500, max_step=0.14)

    # second restart (different seed) – useful in case the first run gets stuck
    rng2 = np.random.default_rng(987654)
    start2 = initial_centres(rng2)
    c2, r2, s2 = optimise(start2, rng2, steps=2500, max_step=0.14)

    if s2 > best_total:
        best_centers, best_radii, best_total = c2, r2, s2

    # final safety check: make sure radii are non‑negative and inside the square
    # (a tiny epsilon is subtracted inside solve_lp already)
    best_total = float(best_radii.sum())
    return best_centers, best_radii, best_total
