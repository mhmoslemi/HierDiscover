```
def compute_optimal_radii(centers):
    n = centers.shape[0]
    # compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    # doping
    M = 4 * n + n * (n - 1) // 2
    A = np.zeros((M, n))
    b = np.zeros(M)
    idx = 0
    for i in range(n):
        x, y = centers[i]
        A[idx, i] = 1
        b[idx] = x
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - x
        idx += 1
        A[idx, i] = 1
        b[idx] = y
        idx += 1
        A[idx, i] = 1
        b[idx] = 1 - y
        idx += 1
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = dists[i, j]
            idx += 1
    c = -np.ones(n)  # minimize negative sum => maximize sum
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: radii = zeros
        return np.zeros(n), -np.inf
    radii = res.x
    sum_r = -res.fun  # because min of -sum = -sum
    return radii, sum_r
```

```
def compute_optimal_radii(centers):
    import numpy as np
    from scipy.optimize import linprog
    n = centers.shape[0]
    # compute distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # number of constraints
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    idx = 0
    for i in range(n):
        x = centers[i, 0]
        y = centers[i, 1]
        # x_i
        A[idx, i] = 1
        b[idx] = x
        idx += 1
        # 1 - x_i
        A[idx, i] = 1
        b[idx] = 1 - x
        idx += 1
        # y_i
        A[idx, i] = 1
        b[idx] = y
        idx += 1
        # 1 - y_i
        A[idx, i] = 1
        b[idx] = 1 - y
        idx += 1
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = dists[i, j]
            idx += 1
    c = -np.ones(n)  # maximize sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:

        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = -res.fun  # because we minimized -sum
    return radii, sum_r
```

**Idea**

The radii are linear variables that are only limited by linear inequalities:

* `r_i ≤ x_i ,  r_i ≤ 1‑x_i ,  r_i ≤ y_i ,  r_i ≤ 1‑y_i` (boundary)
* `r_i + r_j ≤ distance(i , j)` (no overlap)

For a *fixed* set of centre coordinates the optimal radii (maximising the sum of radii) are obtained by a tiny linear programme:
```
max   Σ r_i
s.t.  the inequalities above
      r_i ≥ 0
```
Hence the whole problem can be tackled as a **two‑level optimisation**:

1. **inner level** – for any placement of the 26 centres solve the LP and obtain the optimal radii and the sum of radii;
2. **outer level** – move the centre points so that the LP‑sum becomes as large as possible.

The inner LP is tiny (26 variables, ≤429 constraints) and can be solved in a few milliseconds with `scipy.optimize.linprog` (HiGHS solver).  
The outer optimisation is performed with a simple simulated‑annealing / hill‑climbing loop:

* start from a random placement (several independent restarts);
* at each step propose a new placement by adding Gaussian noise;
* solve the LP for the proposal, obtain the new sum of radii;
* accept the move if it improves the sum, otherwise accept with a probability that slowly decreases with temperature (standard SA).

The best placement found during the whole run is returned together with its optimal radii.  
All circles are guaranteed to satisfy the non‑overlap and boundary constraints because the radii come from the LP solution.

**Why it works**

* The LP gives the *exact* maximal radii for any set of centre points, therefore the outer search never wastes time on infeasible radius choices.
* Simulated annealing with a gradually shrinking step‑size and temperature explores the 52‑dimensional centre space efficiently and is able to escape local plateaus.
* Multiple random restarts increase the chance of finding a high‑quality configuration.
* The method is completely deterministic apart from the seeded random generator, uses only NumPy and SciPy and respects all rules (no closures, no I/O).

The final routine `run_packing()` returns a tuple `(centers, radii, sum_radii)` ready for the provided validator.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# inner optimisation – linear programme for radii
# ------------------------------------------------------------
def _optimal_radii(centers: np.ndarray):
    """
    Solve the LP:
        max sum(r_i)
        subject to:
            r_i <= x_i, 1-x_i, y_i, 1-y_i
            r_i + r_j <= dist(i,j)
            r_i >= 0
    Returns radii (np.ndarray) and the sum of radii (float).
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of inequality constraints
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    for i in range(n):
        x, y = centers[i]
        # boundary constraints
        A[row, i] = 1.0
        b[row] = x
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        A[row, i] = 1.0
        b[row] = y
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise sum(r)  → minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    # bounds for each r_i
    bounds = [(0.0, None)] * n

    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = res.x
        total = -res.fun  # because we minimised the negative sum
    return radii, total


# ------------------------------------------------------------
# outer optimisation – simulated annealing on centre locations
# ------------------------------------------------------------
def run_packing():
    """
    Returns (centers, radii, sum_of_radii) for 26 circles
    packed inside the unit square.  The arrangement is obtained
    by simulated‑annealing on the centre positions together with
    an inner linear programme that yields the optimal radii for
    a given placement.
    """
    np.random.seed(0)               # reproducibility
    N = 26

    # SA hyper‑parameters
    restarts = 5
    it_per_restart = 500            # total LP solves ≈ restarts·it_per_restart
    sigma_start = 0.20              # initial positional jitter size
    temp_start = 0.12               # initial temperature

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    for _ in range(restarts):
        # start from a random (well‑spread) configuration
        cur_centers = np.random.rand(N, 2)

        # initial LP evaluation
        cur_radii, cur_sum = _optimal_radii(cur_centers)

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()

        for it in range(it_per_restart):
            # linear schedules for step size and temperature
            frac = it / it_per_restart
            sigma = sigma_start * (1.0 - frac)
            temp = temp_start * (1.0 - frac)

            # propose a new placement
            proposal = cur_centers + np.random.normal(scale=sigma,
                                                      size=cur_centers.shape)
            proposal = np.clip(proposal, 0.0, 1.0)

            # evaluate the LP for the proposal
            prop_radii, prop_sum = _optimal_radii(proposal)

            delta = prop_sum - cur_sum
            # accept if better, otherwise accept with Metropolis probability
            if delta > 0.0 or np.random.rand() < np.exp(delta / (temp + 1e-12)):
                cur_centers = proposal
                cur_radii = prop_radii
                cur_sum = prop_sum

                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers.copy()
                    best_radii = cur_radii.copy()

    # final LP on the best centre set (to clean possible numerical slack)
    best_radii, best_sum = _optimal_radii(best_centers)

    # Ensure the returned sum is a plain Python float
    return best_centers, best_radii, float(best_sum)
```
