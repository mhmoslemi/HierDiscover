def radii_from_centers(centers):
    n = centers.shape[0]
    A = []
    b = []
    # boundary constraints
    for i in range(n):
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row); b.append(centers[i,0])
        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row); b.append(1 - centers[i,0])
        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row); b.append(centers[i,1])
        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row); b.append(1 - centers[i,1])
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            d = np.linalg.norm(centers[i] - centers[j])
            b.append(d)
    A = np.array(A)
    b = np.array(b)
    # Bounds r_i >= 0
    bounds = [(0, None) for _ in range(n)]
    # objective
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    radii = res.x
    return radii, np.sum(radii)

def solve_lp(centers):
    n = centers.shape[0]
    # compute distances matrix (maybe using broadcasting)
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # determine number of constraints
    n_pairs = n * (n - 1) // 2
    n_constraints = 4 * n + n_pairs
    A = np.zeros((n_constraints, n))
    b = np.zeros(n_constraints)
    # fill boundaries
    row = 0
    for i in range(n):
        A[row, i] = 1.0; b[row] = centers[i, 0]; row += 1
        A[row, i] = 1.0; b[row] = 1.0 - centers[i, 0]; row += 1
        A[row, i] = 1.0; b[row] = centers[i, 1]; row += 1
        A[row, i] = 1.0; b[row] = 1.0 - centers[i, 1]; row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1
    # objective: maximize sum radii -> minimize -sum radii
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    # solve
    # Method HIGHS is default; choose options to maybe speed
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed: " + res.message)
    radii = res.x
    total = np.sum(radii)
    return radii, total

1. **Linear programming for fixed centers**  
   For a given set of circle centres the optimal radii are obtained by the linear program  

   \[
   \max \sum_i r_i\quad\text{s.t.}\quad
   \begin{cases}
   r_i \le x_i,\; r_i \le 1-x_i,\; r_i \le y_i,\; r_i \le 1-y_i,\\
   r_i+r_j \le \|c_i-c_j\|_2\;(i<j),\\
   r_i\ge 0 .
   \end{cases}
   \]

   The constraints are linear in the radii, therefore `scipy.optimize.linprog` can solve them efficiently.

2. **Stochastic optimisation of the centres**  
   The positions of the 26 centres are the non‑convex part of the problem.  
   We use a simple simulated‑annealing / hill‑climbing loop:
   * start from several random initial configurations,
   * at each iteration pick a circle and move its centre by a small random
     displacement (clipped to the unit square),
   * solve the LP for the new configuration,
   * accept the move if it improves the total radius, otherwise accept it
     with a temperature‑controlled probability,
   * the step size and temperature are decreased linearly so the search
     gradually focuses on fine adjustments.

   The best configuration found over a few restarts is returned.

3. **Determinism**  
   The random seed is fixed, so the function is deterministic and reproducible.
