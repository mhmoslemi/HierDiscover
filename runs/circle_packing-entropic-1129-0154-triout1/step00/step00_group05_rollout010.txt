import numpy as np
from scipy.optimize import linprog

def _solve_linear_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    For a fixed set of centre positions solve the linear program:
        maximise   sum_i r_i
        subject to r_i >= 0
                   r_i <= x_i,  r_i <= 1-x_i
                   r_i <= y_i,  r_i <= 1-y_i
                   r_i + r_j <= dist(i,j)   for all i<j

    Returns the optimal radii and the sum of radii.
    """
    n = centers.shape[0]

    # objective: maximise sum(r)  -> minimise -sum(r)
    c = -np.ones(n)

    # inequality matrix A_ub * r <= b_ub
    rows = []
    rhs = []

    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row.copy())
        rhs.append(x)           # r_i <= x_i
        rows.append(row.copy())
        rhs.append(1.0 - x)     # r_i <= 1 - x_i
        rows.append(row.copy())
        rhs.append(y)           # r_i <= y_i
        rows.append(row.copy())
        rhs.append(1.0 - y)     # r_i <= 1 - y_i

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # Infeasible or numerical problem – return zeros (will be rejected later)
        return np.zeros(n), 0.0

    radii = np.maximum(res.x, 0.0)     # clip possible tiny negatives
    total = float(np.sum(radii))
    return radii, total
