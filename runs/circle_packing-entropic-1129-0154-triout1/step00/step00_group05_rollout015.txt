import numpy as np

def compute_radii(centers):
    n = centers.shape[0]
    # distances to walls:
    d_to_walls = np.minimum(centers, 1 - centers)  # shape (n,2)
    min_dist_to_walls = np.min(d_to_walls, axis=1) # radius bound from walls
    # pairwise distances
    radii = np.copy(min_dist_to_walls)
    # compute half the distance to each other center
    # compute pairwise distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # for each i, compute min over j!=i of dists[i, j]/2
    # avoid zero on diagonal
    np.fill_diagonal(dists, np.inf)
    min_dist_to_others = np.min(dists / 2.0, axis=1)
    radii = np.minimum(radii, min_dist_to_others)
    return radii
```

def hill_climb(num_iter=20000, step=0.1, seed=0):
    np.random.seed(seed)
    centers = np.random.rand(26,2)  # uniform
    radii = compute_radii(centers)
    sum_r = radii.sum()
    best = (centers.copy(), radii.copy(), sum_r)

    for t in range(num_iter):
        i = np.random.randint(26)
        # propose new position:
        proposal = centers[i].copy()
        proposal += (np.random.rand(2) - 0.5) * 2 * step
        proposal = np.clip(proposal, 0, 1)
        # temporarily update
        old_center = centers[i].copy()
        centers[i] = proposal
        radii_new = compute_radii(centers)
        sum_new = radii_new.sum()
        # accept if sum_new > sum_r (maybe also accept sometimes)
        if sum_new > sum_r:
            radii = radii_new
            sum_r = sum_new
            if sum_new > best[2]:
                best = (centers.copy(), radii.copy(), sum_new)
        else:
            # revert
            centers[i] = old_center
    return best

def maximal_radii(centers, max_iter=100, eps=1e-8):
    n = centers.shape[0]
    radii = np.zeros(n)
    # initial radii maybe based on walls only
    # but we will iterate
    for _ in range(max_iter):
        radii_old = radii.copy()
        # compute pairwise distances
        diff = centers[:, None, :] - centers[None, :, :]
        dists = np.sqrt(np.sum(diff**2, axis=2))
        # set diagonal large to avoid self
        np.fill_diagonal(dists, np.inf)
        # compute for each i allowed from other circles: min over j of dists[i,j] - radii[j]
        # note radii[j] are from previous iteration; alt using radii_j currently.
        allowed = np.minimum(centers[:,0], 1 - centers[:,0])
        allowed = np.minimum(allowed, np.minimum(centers[:,1], 1 - centers[:,1]))
        # compute min_{j!=i} (d_ij - radii_j)
        # for i, we want vector of dists[i] - radii (broadcast)
        # We can compute min across j
        limit_from_others = np.min(dists - radii[None, :], axis=1)  # shape (n,)
        # overall allowed radius
        radii = np.minimum(allowed, limit_from_others)
        radii = np.maximum(radii, 0.0)
        # check convergence
        if np.max(np.abs(radii - radii_old)) < eps:
            break
    return radii

def optimal_radii(centers):
    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diff**2).sum(axis=2))
    # Build constraints:
    # A_ub @ r <= b_ub
    # For each i: r_i <= boundary_i
    # For each pair i<j: r_i + r_j <= d_ij
    num_constraints = n + n*(n-1)//2
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    # bounds constraints
    for i in range(n):
        bound_i = min(centers[i,0],1-centers[i,0],centers[i,1],1-centers[i,1])
        A[i,i] = 1.0
        b[i] = bound_i
    # pair constraints
    row = n
    for i in range(n):
        for j in range(i+1,n):
            A[row,i] = 1.0
            A[row,j] = 1.0
            b[row] = dists[i,j]
            row += 1
    # Objective: maximize sum r => minimize -sum r
    c = -np.ones(n)
    # Using high's method
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # In case of failure, fallback maybe using zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii

def farthest_point_sampling(n, n_candidates=2000, margin=0.05, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    pts = []
    # start with a point at center
    pts.append(np.array([0.5,0.5]))
    while len(pts) < n:
        cand = rng.random((n_candidates,2))
        # margin
        cand = margin + (1-2*margin) * cand
        if pts:
            existing = np.array(pts)
            d = np.sqrt(((cand[:,None,:] - existing[None,:,:])**2).sum(axis=2))
            min_d = np.min(d, axis=1)
        else:
            min_d = np.full(n_candidates, np.inf)
        idx = np.argmax(min_d)
        pts.append(cand[idx])
    return np.vstack(pts)

def local_search(num_iters=5000, seed=0):
    rng = np.random.default_rng(seed)
    centers = farthest_point_sampling(26, n_candidates=3000, margin=0.05, rng=rng)
    radii = compute_optimal_radii(centers)
    best_sum = radii.sum()
    best_centers = centers.copy()
    best_radii = radii.copy()
    # step schedule
    max_step = 0.3
    min_step = 0.01
    for it in range(num_iters):
        # linear interpolation for step
        step = max_step - (max_step - min_step) * it / num_iters
        i = rng.integers(26)
        new_center = centers[i] + (rng.random(2)-0.5)*2*step
        new_center = np.clip(new_center, 0.0, 1.0)
        old_center = centers[i].copy()
        centers[i] = new_center
        new_radii = compute_optimal_radii(centers)
        new_sum = new_radii.sum()
        if new_sum > best_sum + 1e-9:
            best_sum = new_sum
            best_centers = centers.copy()
            best_radii = new_radii.copy()
            radii = new_radii
        else:
            # revert
            centers[i] = old_center
    return best_centers, best_radii, best_sum

def run_packing():
    best_overall_sum = -1.0
    best_overall_centers = None
    best_overall_radii = None
    # try a few restarts
    for seed in range(5):
        centers, radii, s = local_search(num_iters=3000, seed=seed)

            best_overall_centers = centers
            best_overall_radii = radii
        if best_overall_sum >= 2.636:
            break
    # ensure returned arrays are float64
    return best_overall_centers, best_overall_radii, float(best_overall_sum)

def compute_optimal_radii(centers):
    n = centers.shape[0]
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    # Build A_ub
    # number of constraints = n + n*(n-1)//2
    m = n + n*(n-1)//2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    # boundary constraints
    bounds = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]),
                        np.minimum(centers[:,1], 1-centers[:,1]))
    # fill
    for i in range(n):
        A[i, i] = 1.0
        b[i] = bounds[i]
    # pairwise constraints
    row = n
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i,j]
            row += 1
    # Solve LP
    res = linprog(-np.ones(n), A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # fallback: set radii to zero
        return np.zeros(n)
    return res.x
```

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

# ----------------------------------------------------------------------
# Helper: compute a feasible set of radii using the simple half‑distance rule.
# This gives a quick lower bound on the optimal radii for a fixed set of
# centers and is always feasible (no overlaps, circles stay inside the square).
# ----------------------------------------------------------------------
def estimated_radii(centers: np.ndarray) -> np.ndarray:
    """Return radii = min( distance to nearest wall ,
                           half of distance to the nearest other centre )."""
    n = centers.shape[0]
    # distance to the four walls
    wall_dist = np.minimum(centers, 1.0 - centers)          # shape (n,2)
    wall_min = np.min(wall_dist, axis=1)                   # shape (n,)

    # pairwise centre distances
    diff = centers[:, None, :] - centers[None, :, :]        # (n,n,2)
    dists = np.sqrt(np.sum(diff * diff, axis=2))           # (n,n)
    np.fill_diagonal(dists, np.inf)                       # ignore self–distance

    nearest = np.min(dists, axis=1) / 2.0                  # half of nearest distance
    radii = np.minimum(wall_min, nearest)
    radii = np.maximum(radii, 0.0)
    return radii


# ----------------------------------------------------------------------
# Helper: given centre positions, solve the exact linear programme
#   maximise sum r_i
#   s.t.   0 <= r_i <= distance to walls
#          r_i + r_j <= dist(i,j)   for all i<j
# ----------------------------------------------------------------------
def optimal_radii(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    # distances between centres
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of constraints: n (wall) + n*(n-1)/2 (pairwise)
    m = n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    # wall constraints  r_i <= min distance to a wall
    wall_dist = np.minimum(centers, 1.0 - centers)
    wall_min = np.min(wall_dist, axis=1)
    for i in range(n):
        A[i, i] = 1.0
        b[i] = wall_min[i]

    # pairwise constraints  r_i + r_j <= d_ij
    row = n
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # fall back to the safe estimate (should never happen for well‑posed data)
        return estimated_radii(centers)
    return res.x


# ----------------------------------------------------------------------
# Initialise centres.
# We start with four points a moderate distance from the walls
# (so they can become relatively large) and then fill the remaining
# spots with a greedy farthest‑point strategy.
# ----------------------------------------------------------------------
def initialise_centres(rng: np.random.Generator, n: int = 26) -> np.ndarray:
    # four fixed points, 0.2 away from each side
    fixed = np.array([[0.2, 0.2],
                      [0.8, 0.2],
                      [0.2, 0.8],
                      [0.8, 0.8]], dtype=np.float64)
    centres = [p for p in fixed]

    # greedy farthest point sampling for the remaining circles
    n_candidates = 3000
    while len(centres) < n:
        cand = rng.random((n_candidates, 2))          # uniform in the whole square
        existing = np.asarray(centres)                # (k,2)
        # distance from each candidate to the existing set
        diff = cand[:, None, :] - existing[None, :, :]   # (c,k,2)
        d2 = np.sum(diff * diff, axis=2)                # (c,k)
        min_dist = np.sqrt(np.min(d2, axis=1))          # (c,)
        idx = np.argmax(min_dist)
        centres.append(cand[idx])
    return np.vstack(centres)


# ----------------------------------------------------------------------
# Simple hill‑climbing optimiser that works on the cheap estimated radii.
# The estimator guarantees feasibility, so we can accept only improvements.
# ----------------------------------------------------------------------
def hill_climb(
    start_centres: np.ndarray,
    rng: np.random.Generator,
    iterations: int = 15000,
) -> Tuple[np.ndarray, np.ndarray, float]:
    centres = start_centres.copy()
    radii = estimated_radii(centres)
    best_sum = radii.sum()
    best_centres = centres.copy()
    best_radii = radii.copy()

    max_step = 0.38
    min_step = 0.015

    for it in range(iterations):
        step = max_step - (max_step - min_step) * (it / iterations)
        i = rng.integers(centres.shape[0])
        proposal = centres[i] + (rng.random(2) - 0.5) * 2 * step
        proposal = np.clip(proposal, 0.0, 1.0)          # stay inside the square

        old = centres[i].copy()
        centres[i] = proposal
        new_radii = estimated_radii(centres)
        new_sum = new_radii.sum()

        if new_sum > best_sum:
            best_sum = new_sum
            best_centres = centres.copy()
            best_radii = new_radii.copy()

        # keep the move only if it does not worsen the current estimate
        if new_sum >= radii.sum():
            radii = new_radii
        else:
            centres[i] = old  # revert

    return best_centres, best_radii, best_sum

# Main entry point required by the grader.
# ----------------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles.
    The radii are produced by the exact LP, guaranteeing feasibility.
    """
    # deterministic sequence of rng seeds – guarantees reproducibility
    seeds = [0, 13, 42, 101, 202, 303]
    overall_best_sum = -1.0
    overall_best_centres = None
    overall_best_radii = None

    for seed in seeds:
        rng = np.random.default_rng(seed)

        # 1) initialise a sensible configuration
        start = initialise_centres(rng, n=26)

        # 2) improve it using cheap hill climbing
        centres, _, _ = hill_climb(start, rng, iterations=15000)

        # 3) compute the exact optimal radii for this geometry
        radii = optimal_radii(centres)
        total = float(radii.sum())

        if total > overall_best_sum:
            overall_best_sum = total
            overall_best_centres = centres.copy()
            overall_best_radii = radii.copy()

        # Early exit if we have already passed the target bound
        if overall_best_sum >= 2.636:
            break

    # A tiny safety factor to keep us strictly inside the tolerance band
    epsilon = 1.0 - 1e-9
    overall_best_radii = overall_best_radii * epsilon
    overall_best_sum = float(overall_best_radii.sum())

    return overall_best_centres, overall_best_radii, overall_best_sum
