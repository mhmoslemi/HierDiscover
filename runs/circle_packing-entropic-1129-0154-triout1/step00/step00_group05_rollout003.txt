import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # list constraints rows
    rows = []
    b = []
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(x)
        # right
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(1 - x)
        # bottom
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(y)
        # top
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        b.append(1 - y)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(dists[i, j])
    A = np.vstack(rows)
    b = np.array(b)
    # objective: maximize sum r -> minimize -sum
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None, -np.inf
    radii = res.x
    sum_r = radii.sum()
    return radii, sum_r

        if points:

def solve_radii(centers):
    import numpy as np
    from scipy.optimize import linprog

    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # Number of constraints:
    # 4*n boundary, + n*(n-1)/2 pairwise
    total_constraints = 4*n + n*(n-1)//2

    A = np.zeros((total_constraints, n), dtype=float)
    b = np.zeros(total_constraints, dtype=float)
    c_idx = 0
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left
        A[c_idx, i] = 1.0
        b[c_idx] = x
        c_idx += 1
        # right
        A[c_idx, i] = 1.0
        b[c_idx] = 1.0 - x
        c_idx += 1
        # bottom
        A[c_idx, i] = 1.0
        b[c_idx] = y
        c_idx += 1
        # top
        A[c_idx, i] = 1.0
        b[c_idx] = 1.0 - y
        c_idx += 1

    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            A[c_idx, i] = 1.0
            A[c_idx, j] = 1.0
            b[c_idx] = dists[i, j]
            c_idx += 1

    c = -np.ones(n)  # maximize sum radii

    bounds = [(0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None, -np.inf
    radii = np.maximum(res.x, 0.0)  # ensure nonnegative
    return radii, radii.sum()

def generate_maximin_points(n, rng, candidates=2000):
    points = []
    # start with center
    points.append(np.array([0.5, 0.5]))
    for _ in range(1, n):
        cand = rng.random((candidates,2))
        # distance to edges
        edge_dists = np.minimum(np.minimum(cand[:,0], 1-cand[:,0]),
                                np.minimum(cand[:,1], 1-cand[:,1]))
        # distance to existing points
        if points:
            existing = np.array(points)
            dists = np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2)
            min_dists = np.minimum(edge_dists, dists.min(axis=1))
        else:
            min_dists = edge_dists
        idx = np.argmax(min_dists)
        points.append(cand[idx])
    return np.array(points)

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

def _solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """Solve the linear programme that gives the maximal radii for fixed centres.

    Returns (radii, sum_of_radii).  If the LP fails, returns (None, -inf).
    """
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of inequality constraints:
    # 4 per circle for the four sides, plus one per unordered pair
    pair_cnt = n * (n - 1) // 2
    total_cnt = 4 * n + pair_cnt

    A = np.zeros((total_cnt, n), dtype=np.float64)
    b = np.zeros(total_cnt, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left side  x - r >= 0  →  r ≤ x
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # right side  x + r ≤ 1  →  r ≤ 1‑x
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # bottom side  y - r >= 0  →  r ≤ y
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # top side  y + r ≤ 1  →  r ≤ 1‑y
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise sum(r)  ⇔  minimise –sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        return None, -np.inf
    radii = np.maximum(res.x, 0.0)
    return radii, float(radii.sum())

def _maximin_points(n: int, rng: np.random.Generator,
                    candidates: int = 2000) -> np.ndarray:
    """Generate n points by the max‑imin (farthest‑point) heuristic."""
    points = [np.array([0.5, 0.5])]
    for _ in range(1, n):
        cand = rng.random((candidates, 2))               # uniform in [0,1]^2
        # distance to the four sides
        edge = np.minimum.reduce([cand[:, 0],
                                  1.0 - cand[:, 0],
                                  cand[:, 1],
                                  1.0 - cand[:, 1]])
        # distance to already placed points
        if points:
            existing = np.stack(points)                 # shape (k,2)
            d = np.linalg.norm(cand[:, None, :] - existing[None, :, :], axis=2)
            min_to_pts = d.min(axis=1)
            score = np.minimum(edge, min_to_pts)
        else:
            score = edge
        best = cand[np.argmax(score)]
        points.append(best)
    return np.stack(points)

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Compute a packing of 26 circles inside the unit square that maximises the
    sum of radii.  Returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(0)          # deterministic seed
    n_circles = 26

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # ----- multiple independent restarts --------------------------------- #
    for restart in range(8):
        # 1) generate a well‑spread set of centres
        centres = _maximin_points(n_circles, rng, candidates=1500)

        # small random jitter to break perfect symmetry
        centres += (rng.random(centres.shape) - 0.5) * 0.01
        centres = np.clip(centres, 0.0, 1.0)

        # solve the LP for this configuration
        radii, total = _solve_radii(centres)
        if radii is None:
            continue
        if total > best_sum:
            best_sum = total
            best_centers = centres.copy()
            best_radii = radii.copy()

        # ----- simple hill‑climbing on the centres ----------------------- #
        step = 0.18
        for it in range(150):
            i = rng.integers(n_circles)                # pick a circle to move
            new_pt = best_centers[i] + (rng.random(2) - 0.5) * step
            new_pt = np.clip(new_pt, 0.0, 1.0)

            cand_centers = best_centers.copy()
            cand_centers[i] = new_pt

            cand_radii, cand_sum = _solve_radii(cand_centers)
            if cand_radii is None:
                continue
            if cand_sum > best_sum:
                best_sum = cand_sum
                best_centers = cand_centers
                best_radii = cand_radii

            # slowly shrink the step size
            step *= 0.997

    # safety margin – take a tiny bite off each radius so the validator’s
    # 1e‑12 tolerance is never breached
    eps = 1e-9
    final_radii = np.maximum(best_radii - eps, 0.0)
    final_sum = float(final_radii.sum())

    return best_centers, final_radii, final_sum
