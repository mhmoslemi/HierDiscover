```python
def generate_points(num_points, n_candidates=2000, rng=None):
    points = []
    # For first point, choose interior (0.5,0.5)
    points.append(np.array([0.5,0.5]))
    for k in range(1, num_points):
        # sample candidates uniformly in [0,1]^2
        cand = rng.uniform(0,1,size=(n_candidates,2))
        # compute for each candidate the min distance to existing points and to boundaries
        if points:
            existing = np.array(points) # shape (m,2)
            # distances to each existing point
            dists = np.sqrt(((cand[:,None,:] - existing[None,:,:])**2).sum(axis=2))  # shape (n_candidates,m)
            min_dist_to_points = dists.min(axis=1)
        else:
            min_dist_to_points = np.full(n_candidates, np.inf)
        # distance to boundary = min(x,1-x,y,1-y)
        boundary_dist = np.minimum.reduce([cand[:,0],1-cand[:,0],cand[:,1],1-cand[:,1]])
        # score = min(min_dist_to_points, boundary_dist)
        scores = np.minimum(min_dist_to_points, boundary_dist)
        # choose candidate with max score
        best_idx = np.argmax(scores)
        points.append(cand[best_idx])
    return np.array(points)
```

```
def compute_optimal_radii(centers):
    N = centers.shape[0]
    # compute distance matrix
    diff = centers[:, None, :] - centers[None, :, :]  # shape NxNx2
    D = np.sqrt(np.sum(diff**2, axis=2))
    # bounds
    bounds = [(0.0, min(centers[i,0], 1-centers[i,0], centers[i,1], 1-centers[i,1])) for i in range(N)]
    # constraints
    pair_indices = [(i,j) for i in range(N) for j in range(i+1,N)]
    A_ub = np.zeros((len(pair_indices), N))
    b_ub = np.empty(len(pair_indices))
    for idx, (i,j) in enumerate(pair_indices):
        A_ub[idx,i] = 1.0
        A_ub[idx,j] = 1.0
        b_ub[idx] = D[i,j]
    # solve
    c = -np.ones(N)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii = zeros
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii
```

```
def fps_points(N, rng, n_candidates=4000):
    points = []
    # first point at centre
    points.append(np.array([0.5,0.5]))
    for k in range(1,N):
        # sample candidates uniformly
        cand = rng.uniform(0,1,size=(n_candidates,2))
        # compute distances to existing points
        existing = np.array(points)
        dists = np.sqrt(((cand[:,None,:] - existing[None,:,:])**2).sum(axis=2))
        min_dist = dists.min(axis=1)
        # also boundary distance
        boundary = np.minimum.reduce([cand[:,0], 1-cand[:,0], cand[:,1], 1-cand[:,1]])
        # score = min(min_dist, boundary)
        scores = np.minimum(min_dist, boundary)
        best_idx = np.argmax(scores)
        points.append(cand[best_idx])
    return np.array(points)
```

```python
import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    N = centers.shape[0]
    # compute bounds
    bounds = [(0.0, min(centers[i,0], 1.0 - centers[i,0], centers[i,1], 1.0 - centers[i,1]))
              for i in range(N)]
    # distance matrix
    diff = centers[:, None, :] - centers[None, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    # pair constraints
    pair_idx = [(i, j) for i in range(N) for j in range(i+1, N)]
    M = len(pair_idx)
    A = np.zeros((M, N))
    b = np.empty(M)
    for k, (i, j) in enumerate(pair_idx):
        A[k, i] = 1.0
        A[k, j] = 1.0
        b[k] = D[i, j]
    c = -np.ones(N)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

def farthest_point_sampling(N, rng, n_candidates=2000):
    points = []
    points.append(np.array([0.5, 0.5]))
    for _ in range(1, N):
        cand = rng.uniform(0.0, 1.0, size=(n_candidates, 2))
        existing = np.array(points)
        # distance to existing points
        d = np.sqrt(((cand[:, None, :] - existing[None, :, :])**2).sum(axis=2))
        min_dist = d.min(axis=1)
        # distance to boundaries
        bound = np.minimum.reduce([cand[:, 0], 1.0 - cand[:, 0], cand[:, 1], 1.0 - cand[:, 1]])
        score = np.minimum(min_dist, bound)
        best_idx = np.argmax(score)
        points.append(cand[best_idx])
    return np.array(points)

def evaluate(centers):

    total = radii.sum()
    return total, radii
```

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programming: given centre coordinates, compute the
# maximal radii (sum of radii is maximised).
# ------------------------------------------------------------
def solve_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP for a fixed set of centres.

    Parameters
    ----------
    centers : np.ndarray, shape (n, 2)
        (x, y) coordinates of the circles.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        Optimal radii for the supplied centres.
    """
    n = centers.shape[0]

    # --- bounds from the four walls ---------------------------------
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max_r))

    # --- pairwise non‑overlap constraints ----------------------------
    # distance matrix
    diff = centers[:, None, :] - centers[None, :, :]          # (n, n, 2)
    dist = np.sqrt(np.sum(diff * diff, axis=2))              # (n, n)

    pair_idx = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pair_idx)

    A = np.zeros((m, n))
    b = np.empty(m)
    for k, (i, j) in enumerate(pair_idx):
        A[k, i] = 1.0
        A[k, j] = 1.0
        b[k] = dist[i, j]

    # --- objective (maximise sum of radii) ---------------------------
    c = -np.ones(n)          # linprog does minimisation

    # Solve with the high‑speed HiGHS backend
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – return zeros (the configuration will be rejected later)
        return np.zeros(n)
    return res.x


# ------------------------------------------------------------
# Farthest‑point‑sampling (max‑imin) to obtain a well‑distributed set
# of points.
# ------------------------------------------------------------
def farthest_point_sampling(num: int, rng: np.random.Generator,
                            n_candidates: int = 4000) -> np.ndarray:
    """
    Generate `num` points inside the unit square using farthest point sampling.
    """
    points = [np.array([0.5, 0.5])]
    for _ in range(1, num):
        # sample a batch of candidates
        cand = rng.uniform(0.0, 1.0, size=(n_candidates, 2))

        # distance to the already chosen points
        existing = np.asarray(points)                     # (k, 2)
        d = np.sqrt(((cand[:, None, :] - existing[None, :, :]) ** 2).sum(axis=2))
        min_dist_to_pts = d.min(axis=1)

        # distance to the boundary (the limiting factor for circles near the edge)
        bound = np.minimum.reduce([cand[:, 0], 1.0 - cand[:, 0],
                                   cand[:, 1], 1.0 - cand[:, 1]])

        # The score is the smallest of the two distances; we want it large.
        score = np.minimum(min_dist_to_pts, bound)
        best = np.argmax(score)
        points.append(cand[best])
    return np.asarray(points)


# ------------------------------------------------------------
# Helper that evaluates a centre set: solves the LP and returns
# the total radius sum together with the radii vector.
# ------------------------------------------------------------
def evaluate(centres: np.ndarray):
    radii = solve_radii(centres)
    total = float(radii.sum())
    return total, radii


# ------------------------------------------------------------
# Main entry point required by the task.
# ------------------------------------------------------------
def run_packing():
    rng = np.random.default_rng(12345)   # deterministic seed
    N = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # --------------------------------------------------------
    # 1) 5×5 regular grid + a tiny extra point (25+1 = 26)
    # --------------------------------------------------------
    grid_vals = np.linspace(0.1, 0.9, 5)                 # 0.1,0.3,…,0.9
    grid = np.array([[x, y] for y in grid_vals for x in grid_vals])
    extra = np.array([[0.5, 0.5 + 0.02]])                # a small offset to avoid exact coincidence
    centres_grid = np.vstack([grid, extra])
    total, radii = evaluate(centres_grid)
    if total > best_sum:
        best_sum = total
        best_centers = centres_grid.copy()
        best_radii = radii.copy()

    # --------------------------------------------------------
    # 2) Farthest‑point‑sampling initial set
    # --------------------------------------------------------
    centres_fps = farthest_point_sampling(N, rng, n_candidates=4000)
    total, radii = evaluate(centres_fps)
    if total > best_sum:
        best_sum = total
        best_centers = centres_fps.copy()
        best_radii = radii.copy()

    # --------------------------------------------------------
    # 3) Pure random draws (a quick stochastic search)
    # --------------------------------------------------------
    for _ in range(2000):
        centres_rand = rng.uniform(0.0, 1.0, size=(N, 2))
        total, radii = evaluate(centres_rand)
        if total > best_sum:
            best_sum = total
            best_centers = centres_rand.copy()
            best_radii = radii.copy()

    # --------------------------------------------------------
    # 4) Local improvement (tiny hill‑climbing) starting from the best set
    # --------------------------------------------------------
    centres = best_centers.copy()
    radii = best_radii.copy()
    current_sum = best_sum

    sigma = 0.05                     # scale of random moves
    for step in range(2000):
        i = rng.integers(N)         # choose a circle to move
        new_center = centres[i] + rng.normal(scale=sigma, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)   # keep inside the square

        trial_centres = centres.copy()
        trial_centres[i] = new_center
        trial_sum, trial_radii = evaluate(trial_centres)

        if trial_sum > current_sum + 1e-12:
            centres = trial_centres
            radii = trial_radii
            current_sum = float(trial_sum)
            if current_sum > best_sum:
                best_sum = current_sum
                best_centers = centres.copy()
                best_radii = radii.copy()

        # slowly shrink the step size
        if step % 500 == 0 and sigma > 0.005:
            sigma *= 0.7

    # --------------------------------------------------------
    # Return the best configuration found
    # --------------------------------------------------------
    return best_centers, best_radii, float(best_sum)
```
