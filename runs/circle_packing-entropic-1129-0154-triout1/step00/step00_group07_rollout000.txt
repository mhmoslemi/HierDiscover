def generate_grid(n: int) -> np.ndarray:
    """
    Produce a regular grid of points inside the unit square.
    The grid is a bit inset (margin 0.1) so that all points have a
    reasonable distance to the borders.
    """
    rows = int(np.ceil(np.sqrt(n))) + 1      # a little extra
    cols = rows
    xs = np.linspace(0.1, 0.9, cols)
    ys = np.linspace(0.1, 0.9, rows)
    pts = np.array([(x, y) for y in ys for x in xs])
    return pts[:n]


def generate_hexagonal(n: int) -> np.ndarray:
    """
    Produce a hexagonal (triangular) lattice inside the square.
    The spacing is chosen so that about 26 points fit.
    """
    spacing_x = 0.19               # approx 1/(5) – a bit smaller than 0.2
    spacing_y = np.sqrt(3) / 2 * spacing_x
    pts = []
    y = 0.12

    while y <= 0.88 and len(pts) < n:
        offset = 0.0 if row % 2 == 0 else spacing_x / 2
        x = 0.12 + offset

            x += spacing_x
        y += spacing_y
        row += 1
    return np.array(pts[:n])

def solve_radii_lp(centres: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a fixed set of centres solve the linear program:
        max sum r_i
        s.t.   0 <= r_i <= distance to each square side
               r_i + r_j <= distance(centre_i, centre_j)   (i < j)
    Returns the optimal radii and the corresponding sum.
    """
    n = centres.shape[0]

    # pairwise Euclidean distances
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # build inequality matrix A_ub * r <= b_ub   (pairwise constraints)
    m = n * (n - 1) // 2
    A = np.zeros((m, n), dtype=float)
    b = np.zeros(m, dtype=float)

    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]                     # exact distance; a tiny safety
            row += 1

    # bounds for every variable coming from the four walls
    bounds = []
    for i in range(n):
        x, y = centres[i]
        max_rad = min(x, 1.0 - x, y, 1.0 - y)
        bounds.append((0.0, max_rad))

    # linprog solves a minimisation problem → minimise -sum r_i
    c = -np.ones(n, dtype=float)

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # fallback: all radii zero (should never happen)
        return np.zeros(n), 0.0

    radii = np.maximum(result.x, 0.0)   # guard against tiny negative values
    total = radii.sum()
    return radii, total


def safety_scale(centres: np.ndarray, radii: np.ndarray) -> np.ndarray:
    """
    Compute a uniform scale ≤1 that makes all constraints strict.
    The final radii are multiplied by this factor.
    """
    n = centres.shape[0]
    scale = 1.0

    # boundary constraints
    for i in range(n):
        x, y = centres[i]
        max_bd = min(x, 1.0 - x, y, 1.0 - y)
        if radii[i] > max_bd:
            if radii[i] > 0:
                scale = min(scale, max_bd / radii[i])

    # pairwise constraints
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    for i in range(n):
        for j in range(i + 1, n):

            if radii[i] + radii[j] > dij:
                denom = radii[i] + radii[j]
                if denom > 0:
                    scale = min(scale, (dij - 1e-12) / denom)

    scale *= 0.999999
    return radii * scale
