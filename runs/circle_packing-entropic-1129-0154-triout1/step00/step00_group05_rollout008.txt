```
import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diff**2).sum(axis=2))
    # constraints:
    # pair constraints: r_i + r_j <= d_ij
    # boundary constraints: r_i <= x_i, r_i <= 1 - x_i, r_i <= y_i, r_i <= 1 - y_i
    # Build A_ub and b_ub
    m_pair = n*(n-1)//2
    m_boundary = 4*n
    m = m_pair + m_boundary
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1
    # boundary constraints
    x = centers[:, 0]
    y = centers[:, 1]
    # r_i <= x_i
    for i in range(n):
        A[row, i] = 1
        b[row] = x[i]
        row += 1
    # r_i <= 1 - x_i
    for i in range(n):
        A[row, i] = 1
        b[row] = 1 - x[i]
        row += 1
    # r_i <= y_i
    for i in range(n):
        A[row, i] = 1
        b[row] = y[i]
        row += 1
    # r_i <= 1 - y_i
    for i in range(n):
        A[row, i] = 1
        b[row] = 1 - y[i]
        row += 1
    # Objective: maximize sum => minimize -sum
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = radii.sum()
    return radii, sum_r
```

```
def run_packing():
    np.random.seed(0)
    n=26
    best_sum = -np.inf
    best_centers = None
    best_radii = None
    # number of restarts:
    for restart in range(5):
        # initial positions: random uniform
        centers = np.random.rand(n, 2)
        # optionally jitter to avoid overlapping too close
        radii, cur_sum = solve_radii(centers)
        # hill climb simulated annealing
        temp0 = 0.1
        stepsize = 0.2
        for iter in range(4000):
            # decrease temperature and step
            temp = temp0 * (1 - iter/4000)
            step = stepsize * (1 - iter/4000)
            # propose move
            i = np.random.randint(n)
            new_center = centers[i] + (np.random.rand(2)-0.5)*2*step
            # clip to [0,1]
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = centers.copy()
            new_centers[i] = new_center
            new_radii, new_sum = solve_radii(new_centers)
            delta = new_sum - cur_sum
            if delta > 1e-12 or np.exp(delta / max(temp, 1e-12)) > np.random.rand():
                centers = new_centers
                radii = new_radii
                cur_sum = new_sum
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()
        # end of iterations
    # after restarts
    if best_centers is None:
        # fallback trivial
        best_centers = np.zeros((n,2))
        best_radii = np.zeros(n)
        best_sum = 0.0
    return best_centers, best_radii, float(best_sum)
```

For every circle we denote  

* centre `c[i] = (x[i] , y[i])`  `0 ≤ x[i] , y[i] ≤ 1`  
* radius `r[i] ≥ 0`

The set is feasible iff

* the disc stays inside the unit square  

  `r[i] ≤ x[i] ,  r[i] ≤ 1‑x[i] ,  r[i] ≤ y[i] ,  r[i] ≤ 1‑y[i]`
* two different discs do not intersect  

  `r[i] + r[j] ≤ ‖c[i]‑c[j]‖` for all `i < j`

For a *fixed* placement of the centres the radii that maximise the
total sum `∑ r[i]` can be found by a **linear program**:

```
max   1·r[0] + … + 1·r[25]
s.t.  r[i]                     ≤ x[i]
      r[i]                     ≤ 1‑x[i]
      r[i]                     ≤ y[i]
      r[i]                     ≤ 1‑y[i]                (4·26 constraints)
      r[i] + r[j]              ≤ distance(i , j)       (26·25/2 constraints)
      r[i]                     ≥ 0
```

All constraints are linear in the radii, therefore a standard LP solver
returns the optimal radii for the given centre positions.

The difficult part of the original problem is to *choose* the centre
positions.  
The search space is 52‑dimensional (`26·2` coordinates) and non‑convex.
A simple stochastic hill‑climbing (simulated annealing) works very well:

```
start with a random placement of the 26 centres
repeat many times
      pick one centre at random
      move it by a small random vector (still inside the square)
      solve the LP → new radii, new total sum
      if the new sum is larger, keep the move
      otherwise accept it with a small probability (temperature)
```

The algorithm is deterministic because a fixed random seed is used.
A few independent restarts increase the chance of finding a good
configuration; the best packing over all restarts is returned.

The LP is solved with `scipy.optimize.linprog` (HiGHS interior‑point
method), which is extremely fast for the tiny problems (`n = 26`).

The returned packing satisfies all constraints, therefore
`validate_packing` will accept it.  
The algorithm constantly tries to increase the sum of the radii, so the
obtained value is usually well above the naïve grid packing (`2.5`).

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Given centre positions, compute the radii that maximise the sum of radii.
    Returns (radii, total_sum).
    """
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt((diff ** 2).sum(axis=2))

    # number of constraints
    n_pair = n * (n - 1) // 2
    n_bound = 4 * n
    m = n_pair + n_bound

    A = np.zeros((m, n))
    b = np.zeros(m)

    row = 0
    # r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    x = centers[:, 0]
    y = centers[:, 1]

    # boundary constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = x[i]               # r_i <= x_i
        row += 1
    for i in range(n):
        A[row, i] = 1.0
        b[row] = 1.0 - x[i]         # r_i <= 1 - x_i
        row += 1
    for i in range(n):
        A[row, i] = 1.0
        b[row] = y[i]               # r_i <= y_i
        row += 1
    for i in range(n):
        A[row, i] = 1.0
        b[row] = 1.0 - y[i]         # r_i <= 1 - y_i
        row += 1

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    res = linprog(
        c, A_ub=A, b_ub=b,
        bounds=[(0.0, None)] * n,
        method='highs',
        options={'presolve': True}
    )

    if not res.success:
        # In the unlikely case of failure, fall back to zero radii
        radii = np.zeros(n)
    else:
        # a tiny safety margin removes possible 1e-12 numerical violations
        radii = np.maximum(0.0, res.x - 1e-9)

    return radii, float(radii.sum())


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Searches for a good placement of 26 circles inside the unit square.
    Returns (centres, radii, sum_of_radii).
    """
    np.random.seed(0)          # deterministic behaviour
    n = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # parameters of the stochastic search
    n_restarts = 4
    n_iters = 2500
    init_step = 0.2
    init_temp = 0.12

    for _ in range(n_restarts):
        # start from a random configuration
        centers = np.random.rand(n, 2)

        # initial radii and sum
        radii, cur_sum = solve_radii(centers)

        for it in range(1, n_iters + 1):
            # linearly decreasing temperature & step size
            temp = init_temp * (1.0 - it / n_iters)
            step = init_step * (1.0 - it / n_iters)

            # propose a move of a single centre
            idx = np.random.randint(n)
            delta = (np.random.rand(2) - 0.5) * 2.0 * step
            new_center = centers[idx] + delta
            new_center = np.clip(new_center, 0.0, 1.0)

            new_centers = centers.copy()
            new_centers[idx] = new_center

            new_radii, new_sum = solve_radii(new_centers)
            gain = new_sum - cur_sum

            # Metropolis acceptance
            if gain > 0.0 or np.exp(gain / max(temp, 1e-12)) > np.random.rand():
                centers = new_centers
                radii = new_radii
                cur_sum = new_sum

                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = centers.copy()
                    best_radii = radii.copy()

    # safety: if for some reason no configuration was found
    if best_centers is None:
        best_centers = np.zeros((n, 2))
        best_radii = np.zeros(n)
        best_sum = 0.0

    return best_centers, best_radii, float(best_sum)
```
