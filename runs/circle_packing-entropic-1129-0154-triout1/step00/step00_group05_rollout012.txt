def greedy_initial_centers(num_total=26, initial_centers=None, initial_radii=None, candidates_per_iter=2000, random_state=np.random.RandomState(0)):

   centers = list(initial_centers) if not None else []
   radii = list(initial_radii) if not None else []
   while len(centers) < num_total:
       N = candidates_per_iter
       cand = random_state.rand(N,2)
       # compute min boundary distance for each candidate
       bound = np.minimum.reduce([cand[:,0], 1-cand[:,0], cand[:,1], 1-cand[:,1]])   # shape (N,)
       # compute distances to existing centers
       if len(centers) > 0:
           existing = np.array(centers)
           existing_r = np.array(radii)
           # compute distances (N, m)
           diff = cand[:, None, :] - existing[None, :, :]
           dists = np.sqrt(np.sum(diff**2, axis=2))
           # subtract existing radii
           clearance = dists - existing_r[None, :]
           min_clearance = np.min(clearance, axis=1)  # shape (N,)
           feasible = np.minimum(bound, min_clearance)
       else:
           feasible = bound
       # any negative feasible set to -inf to avoid selection
       feasible[feasible < 0] = -1.0
       # choose best index
       best_idx = np.argmax(feasible)
       best_r = feasible[best_idx]
       if best_r <= 1e-8:
           # can't place any more circles with positive radius; break early
           # fill remainder with tiny circles at random positions with radius 0
           for _ in range(num_total - len(centers)):
               centers.append(random_state.rand(2))
               radii.append(0.0)
           break
       best_center = cand[best_idx]
       centers.append(best_center)
       radii.append(best_r)

   return np.array(centers), np.array(radii)

def compute_optimal_radii(centers):
    n = centers.shape[0]
    # Build constraints
    num_constraints = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    row = 0
    # boundary constraints
    xs = centers[:,0]
    ys = centers[:,1]
    for i in range(n):
        A[row, i] = 1
        b[row] = xs[i]  # x_i >= r_i => r_i <= x_i
        row += 1

        A[row, i] = 1
        b[row] = 1 - xs[i]  # r_i <= 1-x_i
        row += 1

        A[row, i] = 1
        b[row] = ys[i]  # r_i <= y_i
        row += 1

        A[row, i] = 1
        b[row] = 1 - ys[i]  # r_i <= 1-y_i
        row += 1

    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            # Euclidean distance
            d = np.linalg.norm(centers[i] - centers[j])
            b[row] = d
            row += 1

    c = -np.ones(n)  # maximize sum
    bounds = [(0, None)] * n
    # Solve using high-performance solver 'highs'
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to less strict result maybe but still.
        raise RuntimeError('LP solver failed: ' + res.message)
    radii = res.x
    return radii

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: compute the optimal radii for a fixed set of centres
# ------------------------------------------------------------
def compute_optimal_radii(centres: np.ndarray) -> np.ndarray:
    """
    Solve the LP:
        maximise  sum r_i
        subject to
            r_i ≤ x_i, r_i ≤ 1-x_i, r_i ≤ y_i, r_i ≤ 1-y_i
            r_i + r_j ≤ dist(i,j)   for all i<j
            r_i ≥ 0
    Returns the optimal radii as a 1‑d array.
    """
    n = centres.shape[0]
    # number of constraints: 4 per centre + one per unordered pair
    m = 4 * n + n * (n - 1) // 2

    A = np.zeros((m, n))
    b = np.zeros(m, dtype=float)

    xs = centres[:, 0]
    ys = centres[:, 1]

    row = 0
    # boundary constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = xs[i]          # r_i ≤ x_i
        row += 1

        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]    # r_i ≤ 1 - x_i
        row += 1

        A[row, i] = 1.0
        b[row] = ys[i]          # r_i ≤ y_i
        row += 1

        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]    # r_i ≤ 1 - y_i
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            d = np.linalg.norm(centres[i] - centres[j])
            b[row] = d
            row += 1

    # maximise sum r_i  →  minimise -sum r_i
    c = -np.ones(n, dtype=float)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In the unlikely event of a failure raise; the caller will
        # simply ignore this attempt.
        raise RuntimeError(f"LP failed: {res.message}")

    return res.x


# ------------------------------------------------------------
# Local hill‑climbing refinement
# ------------------------------------------------------------
def refine_solution(centres: np.ndarray,
                    best_sum: float,
                    rng: np.random.Generator,
                    steps: int = 800,
                    sigma: float = 0.015) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simple stochastic hill‑climbing:
    repeatedly perturb one centre and keep the change if the total
    sum of radii (after re‑solving the LP) increases.
    """
    n = centres.shape[0]
    best_centres = centres.copy()
    best_radii = compute_optimal_radii(best_centres)
    best_sum_local = best_sum

    for _ in range(steps):
        i = rng.integers(n)                     # pick a circle
        proposal = best_centres[i] + rng.normal(scale=sigma, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centres = best_centres.copy()
        new_centres[i] = proposal

        try:
            new_radii = compute_optimal_radii(new_centres)
        except RuntimeError:
            continue                     # infeasible – ignore this move

        new_sum = float(new_radii.sum())
        if new_sum > best_sum_local + 1e-9:
            best_centres = new_centres
            best_radii = new_radii
            best_sum_local = new_sum

    return best_centres, best_radii, best_sum_local

            best_radii = radii

    # ------------------------------------------------------------------
    # 2.  Stochastic local improvement starting from the best attempt
    # ------------------------------------------------------------------
    if best_centres is not None:
        best_centres, best_radii, best_sum = refine_solution(
            best_centres, best_sum, rng, steps=800, sigma=0.015
        )

    # Return the final packing
    return best_centres, best_radii, best_sum
