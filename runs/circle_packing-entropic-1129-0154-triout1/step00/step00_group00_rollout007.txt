def con_fun(X):
    xs = X[:n]
    ys = X[n:2*n]
    rs = X[2*n:]
    # boundary constraints
    c = np.concatenate([xs - rs,
                        1 - xs - rs,
                        ys - rs,
                        1 - ys - rs])

    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    dist2 = dx**2 + dy**2
    rs_sum = rs[:, None] + rs[None, :]
    # need upper triangle (i<j)
    mask = np.triu(np.ones((n, n), dtype=bool), k=1)
    c = np.concatenate([c, dist2[mask] - rs_sum[mask]**2])
    return c

def _constraint_fun(x: np.ndarray) -> np.ndarray:
    """
    Returns a vector of all inequality constraint values g(x) ≥ 0.
    Order:
        x_i - r_i
        1 - x_i - r_i
        y_i - r_i

    c = np.concatenate(
