import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute boundary limits
    b = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    # pair constraints: r_i + r_j <= d_ij
    m = n*(n-1)//2 + n
    A = np.zeros((m, n))
    ub = np.zeros(m)
    row = 0
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            ub[row] = dists[i,j]
            row += 1
    # boundary constraints: r_i <= b_i
    for i in range(n):
        A[row, i] = 1
        ub[row] = b[i]
        row += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # fall back maybe zero solution
        return np.zeros(n), 0.0
    radii = res.x
    sum_r = radii.sum()
    return radii, sum_r
```

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear programme for a fixed set of centres
# ----------------------------------------------------------------------
def _solve_radii(centres: np.ndarray,
                 pair_i: np.ndarray,
                 pair_j: np.ndarray,
                 A: np.ndarray) -> tuple[np.ndarray, float]:
    """
    For a given (n,2) array of centres compute the radii that maximise the
    total sum while obeying the non‑overlap and boundary constraints.
    Returns (radii, sum_of_radii).
    """
    n = centres.shape[0]
    # pairwise Euclidean distances
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # right‑hand side of the inequality constraints
    ub = np.empty(A.shape[0], dtype=np.float64)
    ub[:pair_i.size] = dists[pair_i, pair_j]

    # distance to the four sides of the square
    side = np.minimum.reduce(
        [centres[:, 0], 1.0 - centres[:, 0],
         centres[:, 1], 1.0 - centres[:, 1]]
    )
    ub[pair_i.size:] = side

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use the HiGHS interior‑point method – fast for small dense LPs
    res = linprog(c, A_ub=A, b_ub=ub, bounds=bounds, method='highs')
    if not res.success:
        # In the very unlikely case of failure return a safe zero‑radius solution
        return np.zeros(n, dtype=np.float64), -1.0
    radii = res.x
    return radii, float(radii.sum())
