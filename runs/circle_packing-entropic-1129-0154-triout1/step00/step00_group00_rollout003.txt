def optimal_radii(centers):
    n = centers.shape[0]
    # boundary max radii:
    b = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]), np.minimum(centers[:,1], 1-centers[:,1]))
    # distances between centers
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # build constraints
    pair_indices = [(i, j) for i in range(n) for j in range(i+1, n)]
    m = len(pair_indices)
    A = np.zeros((m, n))
    b_ub = np.empty(m)
    for k, (i, j) in enumerate(pair_indices):
        A[k, i] = 1
        A[k, j] = 1
        b_ub[k] = dists[i, j]
    # objective
    c = -np.ones(n)
    bounds = [(0, b[i]) for i in range(n)]
    res = linprog(c, A_ub=A, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii

import numpy as np
from scipy.optimize import linprog
from scipy import sparse

N = 26
# Pairwise indices for unordered pairs
PAIR_I, PAIR_J = np.triu_indices(N, k=1)
M = len(PAIR_I)  # = 325

# Build the (fixed) matrix A_ub for constraints r_i + r_j <= d_ij
row_idx = np.repeat(np.arange(M), 2)
col_idx = np.concatenate([PAIR_I, PAIR_J])
data = np.ones(2 * M, dtype=np.float64)
A_UB = sparse.csr_matrix((data, (row_idx, col_idx)), shape=(M, N))

def optimal_radii(centers: np.ndarray) -> np.ndarray:
    """Given N=26 points (centers), solve the linear program that maximises
    the sum of radii under non‑overlap and boundary constraints.
    Returns an array of radii (shape (N,)).
    """
    # boundary limits for each circle
    b = np.minimum(centers[:, 0], 1.0 - centers[:, 0])
    b = np.minimum(b, centers[:, 1])
    b = np.minimum(b, 1.0 - centers[:, 1])

    # distances for pairwise constraints
    dx = centers[PAIR_I, 0] - centers[PAIR_J, 0]
    dy = centers[PAIR_I, 1] - centers[PAIR_J, 1]
    dists = np.sqrt(dx * dx + dy * dy)

    c = -np.ones(N, dtype=np.float64)          # maximise sum -> minimise -sum
    bounds = [(0.0, float(bi)) for bi in b]    # 0 <= r_i <= distance to the nearest side

    res = linprog(c, A_ub=A_UB, b_ub=dists, bounds=bounds, method='highs')
    if not res.success:
        # In the very unlikely case the LP fails, return zeros
        return np.zeros(N, dtype=np.float64)

    # A small safety snap‑back to avoid possible 1e‑12 violations in the checker
    eps = 1e-12
    radii = np.maximum(res.x - eps, 0.0)
    return radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Heuristic search for a placement of 26 circles that maximises the sum of radii.
    The function returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(42)

    n = N
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ---- Phase 1 : many random starts -------------------------------------------------
    for _ in range(250):
        cand_centers = rng.uniform(size=(n, 2))
        cand_radii = optimal_radii(cand_centers)
        s = cand_radii.sum()
        if s > best_sum:
            best_sum = s
            best_centers = cand_centers.copy()
            best_radii = cand_radii.copy()
            if best_sum >= 2.636:
                break

    # ---- Phase 2 : local hill climbing ------------------------------------------------
    step = 0.15  # initial movement amplitude
    for it in range(3000):
        # Create a copy of the current best layout and perturb a few points
        cand_centers = best_centers.copy()
        # number of points we move in this step
        for _ in range(3):
            idx = rng.integers(n)
            delta = rng.uniform(-step, step, size=2)
            cand_centers[idx] += delta
            # keep inside the unit square
            cand_centers[idx] = np.clip(cand_centers[idx], 0.0, 1.0)

        cand_radii = optimal_radii(cand_centers)
        s = cand_radii.sum()

        if s > best_sum:
            best_sum = s
            best_centers = cand_centers.copy()
            best_radii = cand_radii.copy()
            # Early exit if we have already passed the reference target
            if best_sum >= 2.636:
                break

        # Slowly shrink the step size to fine‑tune
        if (it + 1) % 600 == 0:
            step *= 0.8

    return best_centers, best_radii, float(best_sum)

&nbsp;&nbsp;`r_i + r_j ≤ Euclidean distance between centre i and centre j` for every pair `i < j`.

The LP is tiny (26 variables, 325 pairwise constraints) and can be solved instantly with
`scipy.optimize.linprog` (the “highs” interior‑point solver).  

Therefore the problem reduces to **choosing good centre positions**.  
A simple heuristic works very well:

1. **Random initial layouts** – generate many random sets of 26 points,
   solve the LP for each and keep the best result.
2. **Local improvement** – start from the current best layout and repeatedly
   perturb a few points by a small random offset, resolve the LP and accept the
   change if the total radius sum rises.  The step size is slowly reduced,
   providing a hill‑climbing (simulated‑annealing‑like) search.
3. Stop as soon as the measured sum exceeds the target 2.636 (or after a fixed

the radii to avoid the stricter tolerance of the validator.
