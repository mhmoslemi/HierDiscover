import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute boundary limits
    bound = np.minimum.reduce([centers[:,0], centers[:,1], 1-centers[:,0], 1-centers[:,1]])
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]   # shape (n,n,2)
    D = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    # For each i: ri <= bound[i] => 1*ri <= bound_i
    # For each i<j: ri + rj <= D[i,j]
    # A_ub size is n + n*(n-1)/2
    num_constraints = n + n*(n-1)//2
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    row = 0
    for i in range(n):
        A[row,i] = 1
        b[row] = bound[i]
        row += 1
    for i in range(n):
        for j in range(i+1, n):
            A[row,i] = 1
            A[row,j] = 1
            b[row] = D[i,j]
            row += 1
    c = -np.ones(n)  # maximize sum ri
    # Solve LP using high speed method
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        radii = res.x
        sum_radii = radii.sum()
    else:
        radii = np.zeros(n)
        sum_radii = 0.0
    return radii, sum_radii

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # boundary bounds
    bound = np.minimum.reduce([centers[:,0], centers[:,1], 1-centers[:,0], 1-centers[:,1]])
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub matrix and b_ub
    # Number of constraints:
    pair_cnt = n*(n-1)//2
    constraints = n + pair_cnt
    A = np.zeros((constraints, n))
    b = np.zeros(constraints)
    row = 0
    # boundary constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = bound[i]
        row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = D[i, j]
            row += 1
    c = -np.ones(n)
    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback to zero radii
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)
        total = float(radii.sum())
    return radii, total

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# 1. Solve the linear program for radii given fixed centres
# ------------------------------------------------------------
def solve_radii(centers: np.ndarray):
    """
    Returns optimal radii and their sum for the passed centre positions.
    """
    n = centers.shape[0]

    # maximal radius limited by the square borders
    border = np.minimum.reduce([centers[:, 0],
                                centers[:, 1],
                                1.0 - centers[:, 0],
                                1.0 - centers[:, 1]])

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dist = np.sqrt(np.sum(diff ** 2, axis=2))

    # Build linear constraints  ri <= border_i   and   ri + rj <= dist_ij
    pair_cnt = n * (n - 1) // 2
    n_constraints = n + pair_cnt
    A = np.zeros((n_constraints, n))
    b = np.zeros(n_constraints)

    row = 0
    # border constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = border[i]
        row += 1

    # distance constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dist[i, j]
            row += 1

    # maximise Σ ri  → minimise -Σ ri
    c = -np.ones(n)

    # HiGHS (method='highs') is fast and reliable for these small problems
    res = linprog(c, A_ub=A, b_ub=b,
                  bounds=[(0.0, None)] * n,
                  method='highs')

    if not res.success:
        # fallback – all radii zero (never happens in practice)
        radii = np.zeros(n)
    else:
        radii = np.maximum(res.x, 0.0)

    # tiny safety margin (the validator tolerates 1e-12, we keep 1e-9)
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)

    return radii, float(radii.sum())


# ------------------------------------------------------------
# 2. Generate a well‑spread set of points (farthest‑point sampling)
# ------------------------------------------------------------
def farthest_points(num_points: int, num_candidates: int = 8000) -> np.ndarray:
    """
    Returns `num_points` 2‑D points that are as far apart as possible.
    """
    candidates = np.random.rand(num_candidates, 2)  # uniform in the unit square

    selected_idx = np.empty(num_points, dtype=int)

    # first point – random
    selected_idx[0] = np.random.randint(num_candidates)

    # distances from every candidate to the first selected point
    min_dists = np.linalg.norm(candidates - candidates[selected_idx[0]], axis=1)
    min_dists[selected_idx[0]] = -np.inf

    for k in range(1, num_points):
        # pick the candidate maximising its distance to the current set
        nxt = np.argmax(min_dists)
        selected_idx[k] = nxt

        # update the minimum distance to the growing set
        d = np.linalg.norm(candidates - candidates[nxt], axis=1)
        min_dists = np.minimum(min_dists, d)
        # exclude already selected points
        min_dists[selected_idx[:k + 1]] = -np.inf

    return candidates[selected_idx]


# ------------------------------------------------------------
# 3. Local optimisation of the centre positions
# ------------------------------------------------------------
def local_optimize(init_centers: np.ndarray,
                   max_iter: int = 1500,
                   step_scale: float = 0.05,
                   target_sum: float = 2.636) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simple hill‑climbing: move one centre, re‑solve the LP,
    keep the move if the total radius sum improves.
    """
    centres = init_centers.copy()
    radii, best_sum = solve_radii(centres)

    n = centres.shape[0]
    for it in range(max_iter):
        # gradually shrink the step size
        sigma = step_scale * (1.0 - it / max_iter) + 1e-4

        i = np.random.randint(n)
        new_c = centres[i] + np.random.normal(scale=sigma, size=2)
        new_c = np.clip(new_c, 0.0, 1.0)        # stay inside the square

        trial_centres = centres.copy()
        trial_centres[i] = new_c
        trial_radii, trial_sum = solve_radii(trial_centres)

        if trial_sum > best_sum + 1e-9:
            centres = trial_centres
            radii = trial_radii
            best_sum = trial_sum
            if best_sum >= target_sum:
                break

    return centres, radii, best_sum


# ------------------------------------------------------------
# 4. Main entry point required by the grader
# ------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles
    that satisfy the validation constraints and aim for a large total radius.
    """
    np.random.seed(0)          # deterministic result
    TARGET = 2.636
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # a few independent restarts increase robustness
    for restart in range(7):
        # 1) generate an initial well‑spaced set
        centres0 = farthest_points(26, num_candidates=8000)

        # small random jitter – gives the optimiser freedom to move points
        centres0 += (np.random.rand(26, 2) - 0.5) * 0.02
        centres0 = np.clip(centres0, 0.0, 1.0)

        # 2) refine by local hill‑climbing
        centres_opt, radii_opt, sum_opt = local_optimize(
            centres0,
            max_iter=2000,
            step_scale=0.06,
            target_sum=TARGET
        )

        if sum_opt > best_sum:
            best_sum = sum_opt
            best_centres = centres_opt
            best_radii = radii_opt

        if best_sum >= TARGET:
            break

    # final safety: make sure radii are non‑negative
    best_radii = np.maximum(best_radii, 0.0)

    return best_centres, best_radii, float(best_sum)
