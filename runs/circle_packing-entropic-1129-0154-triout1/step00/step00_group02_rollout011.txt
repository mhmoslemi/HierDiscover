def max_radii_for_centers(centers):
    N = len(centers)
    # Compute distances:
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints matrix:
    # Number of constraints:
    M = 4*N + N*(N-1)//2
    A = np.zeros((M, N))
    b = np.zeros(M)
    row = 0
    # boundary constraints:
    # r_i - x_i <= 0
    for i in range(N):
        A[row, i] = 1
        b[row] = centers[i,0]  # r_i <= x_i => r_i - x_i <= 0 => left side 1*r_i - x_i <= 0 -> we set inequality as A @ r <= b, thus A[i]=1, b = x_i
        row +=1
        # r_i + x_i <= 1
        A[row, i] = 1
        b[row] = 1 - centers[i,0]
        row +=1
        # r_i - y_i <=0
        A[row, i] = 1
        b[row] = centers[i,1]
        row +=1
        # r_i + y_i <=1
        A[row, i] = 1
        b[row] = 1 - centers[i,1]
        row +=1
    # pairwise constraints:
    for i in range(N):
        for j in range(i+1, N):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row +=1
    # Now solve maximize sum(r) = minimize -sum(r)
    c = -np.ones(N)
    # bounds: (0, None) for each r_i
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros radii to avoid errors
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii, radii.sum()
```

import numpy as np
from scipy.optimize import linprog

N = 26

def max_radii_for_centers(centers):
    N = centers.shape[0]
    # distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))

    # number of constraints
    M = 4*N + N*(N-1)//2
    A = np.zeros((M, N), dtype=np.float64)
    b = np.zeros(M, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(N):
        # r_i <= x_i
        A[row, i] = 1.0
        b[row] = centers[i,0]
        row += 1
        # r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i,0]
        row += 1
        # r_i <= y_i
        A[row, i] = 1.0
        b[row] = centers[i,1]
        row += 1
        # r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i,1]
        row += 1

    # pairwise constraints
    for i in range(N):
        for j in range(i+1, N):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # objective: maximize sum r_i -> minimize -sum r_i
    c = -np.ones(N)
    bounds = [(0, None)] * N
    # Using high-s performance method
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # fallback plausible solution: zeros
        radii = np.zeros(N)
    else:
        radii = result.x

    # small safety margin to avoid numeric border violation
    radii = np.maximum(radii - 1e-9, 0.0)
    return radii, radii.sum()
```

def max_radii_for_centers(centers: np.ndarray):
    N = centers.shape[0]
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))
    M = 4 * N + N * (N - 1) // 2
    A = np.zeros((M, N), dtype=np.float64)
    b = np.zeros(M, dtype=np.float64)
    row = 0
    # boundary constraints
    for i in range(N):
        # r_i <= x_i
        A[row, i] = 1.0
        b[row] = centers[i, 0]
        row += 1
        # r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 0]
        row += 1
        # r_i <= y_i
        A[row, i] = 1.0
        b[row] = centers[i, 1]
        row += 1
        # r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 1]
        row += 1
    # pairwise constraints
    for i in range(N):
        for j in range(i + 1, N):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1
    c = -np.ones(N)  # maximize sum => minimize -sum
    bounds = [(0.0, None)] * N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:

    else:
        radii = res.x

    radii = np.maximum(radii - 1e-9, 0.0)

**Key observation** –  
For a *fixed* set of centre points the optimal radii are obtained by the linear program  

```
max   Σ r_i
s.t.  r_i ≤ x_i ,  r_i ≤ 1‑x_i ,  r_i ≤ y_i ,  r_i ≤ 1‑y_i
      r_i + r_j ≤ dist((x_i,y_i),(x_j,y_j))   for all i<j

```

All constraints are linear, the objective is linear, so the LP gives the *largest possible* radii for that configuration.

**Approach**

1. **Linear‑program solver** – a tiny helper `max_radii_for_centers` builds the LP for any set of centres and returns the optimal radii and their sum (using `scipy.optimize.linprog` with the HiGHS backend).

2. **Search in the space of centre points** – the sum of optimal radii is a well‑behaved function of the centres.  
   *Start* from a random placement of 26 points.  
   *Improve* the placement with a very cheap stochastic hill‑climber:
   - pick one circle,
   - move its centre by a small random offset (clipped to `[0,1]`),
   - recompute the LP,
   - keep the move only if the total radius sum increases.

4. **Safety margin** – after the LP the radii are decreased by a tiny `1e‑9` to stay safely inside the validator tolerances.

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Parameters
N = 26                     # number of circles
TARGET = 2.636             # required sum of radii

# ----------------------------------------------------------------------
def max_radii_for_centers(centers: np.ndarray):
    """
    Solve the linear programme that gives the largest radii for a fixed set of centres.
    Returns (radii, total_sum).
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of linear constraints: 4 per circle + one per unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # boundary constraints: r_i ≤ x_i , r_i ≤ 1‑x_i , r_i ≤ y_i , r_i ≤ 1‑y_i
    for i in range(n):
        A[row, i] = 1.0          # r_i ≤ x_i
        b[row] = centers[i, 0]
        row += 1

        A[row, i] = 1.0          # r_i ≤ 1‑x_i
        b[row] = 1.0 - centers[i, 0]
        row += 1

        A[row, i] = 1.0          # r_i ≤ y_i
        b[row] = centers[i, 1]
        row += 1

        A[row, i] = 1.0          # r_i ≤ 1‑y_i
        b[row] = 1.0 - centers[i, 1]
        row += 1

    # pairwise non‑overlap constraints: r_i + r_j ≤ distance_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  → minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # HiGHS is very fast for this size of LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x

    # tiny safety margin to avoid the 1e‑12 validator tolerance
    radii = np.maximum(radii - 1e-9, 0.0)

    return radii, float(radii.sum())

# ----------------------------------------------------------------------
def hill_climb(start_centers: np.ndarray, rng: np.random.Generator,
               max_iter: int = 2000):
    """
    Simple stochastic hill‑climber on the centre positions.
    Returns the best centre layout found and its radius sum.
    """
    centers = start_centers.copy()
    _, cur_sum = max_radii_for_centers(centers)

    best_centers = centers.copy()
    best_sum = cur_sum

    step = 0.2  # initial maximal move length

    for it in range(max_iter):
        i = rng.integers(N)                     # which circle to move
        proposal = centers[i] + rng.uniform(-step, step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)   # stay inside the square

        new_centers = centers.copy()
        new_centers[i] = proposal

        _, new_sum = max_radii_for_centers(new_centers)

        if new_sum > cur_sum + 1e-12:           # accept only improvements
            centers = new_centers
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                if best_sum >= TARGET:
                    break

        # slowly shrink the step size – helps to fine‑tune at the end
        step = max(step * 0.997, 0.0005)

    return best_centers, best_sum

# ----------------------------------------------------------------------
def run_packing():
    """
    Searches for a feasible packing of 26 circles whose total radius sum
    reaches the required threshold.
    Returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(42)   # deterministic but still pseudo‑random
    best_sum = -1.0
    best_centers = None

    # Perform a few independent restarts – each restart gets its own hill‑climb.
    for restart in range(8):

        if sum_opt > best_sum:

            best_centers = centers_opt.copy()

        if best_sum >= TARGET:
            break

    # Final radii for the best centre configuration we discovered
    final_radii, final_sum = max_radii_for_centers(best_centers)

    return best_centers, final_radii, final_sum
