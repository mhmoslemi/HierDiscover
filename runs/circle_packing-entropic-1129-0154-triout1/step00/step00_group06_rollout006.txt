import numpy as np
import itertools
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def _solve_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the linear programme that maximises the sum of radii for a
    fixed set of centres.

    Returns (radii, sum_of_radii).
    """
    n = centers.shape[0]

    # pairwise distances
    dists = cdist(centers, centers)

    # number of inequality constraints:
    # 4 per circle (left, right, bottom, top) + one per unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # side constraints
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1.0
        b[row] = x           # left side
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x     # right side
        row += 1
        A[row, i] = 1.0
        b[row] = y           # bottom side
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y     # top side
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # objective: maximise sum(r_i)  -> minimise -sum(r_i)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    # use the high‑performance HiGHS solver
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii (should never happen with a sensible centre set)
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x

    # tiny safety margin – removes possible 1e‑12 numerical violations
    radii = np.maximum(radii - 1e-9, 0.0)

    return radii, float(radii.sum())


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles packed in the unit square.
    The function performs a cheap stochastic search over centre configurations,
    solving a linear programme for each and keeping the best result.
    """
    rng = np.random.default_rng(123456)   # deterministic seed
    N = 26
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # ----------------------------------------------------------------------
    # 1. Start from the 5×5 regular grid (the optimal layout for 25 equal circles)
    # ----------------------------------------------------------------------
    grid_vals = np.linspace(0.1, 0.9, 5)          # positions that allow r = 0.1
    base_grid = np.array(list(itertools.product(grid_vals, grid_vals)))  # (25, 2)

    # Search parameters
    n_grid_trials = 2000
    jitter = 0.03               # maximal uniform jitter for the grid points

    for _ in range(n_grid_trials):
        # jitter the 25 grid points
        centres = base_grid + rng.uniform(-jitter, jitter, base_grid.shape)
        centres = np.clip(centres, 0.0, 1.0)

        # add one additional random point (the 26‑th circle)
        extra = rng.uniform(size=2)
        centres = np.vstack([centres, extra])

        radii, s = _solve_radii(centres)

        if s > best_sum:
            best_sum = s
            best_centres = centres.copy()
            best_radii = radii.copy()

    # ----------------------------------------------------------------------
    # 2. Purely random centre sets – occasional shots may beat the grid pattern
    # ----------------------------------------------------------------------
    n_random_trials = 500
    for _ in range(n_random_trials):
        centres = rng.uniform(size=(N, 2))
        radii, s = _solve_radii(centres)

        if s > best_sum:
            best_sum = s
            best_centres = centres.copy()
            best_radii = radii.copy()

    # ----------------------------------------------------------------------
    # Return the best packing found
    # ----------------------------------------------------------------------
    # Ensure output shapes are exactly (26,2) and (26,)
    best_centres = np.asarray(best_centres, dtype=np.float64).reshape(N, 2)
    best_radii = np.asarray(best_radii, dtype=np.float64).reshape(N)

    return best_centres, best_radii, float(best_radii.sum())
