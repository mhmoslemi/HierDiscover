def compute_optimal_radii(centers):
    N = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub and b_ub.
    # For each i: x_i, 1-x_i, y_i, 1-y_i
    rows = []
    b = []
    # boundaries
    for i, (x,y) in enumerate(centers):
        row = np.zeros(N); row[i]=1
        rows.append(row); b.append(x)
        # 1-x
        rows.append(row); b.append(1-x)
        # y
        rows.append(row); b.append(y)
        # 1-y
        rows.append(row); b.append(1-y)
    # pairwise
    for i in range(N):
        for j in range(i+1,N):
            dij = dists[i,j]
            row = np.zeros(N)
            row[i]=1; row[j]=1
            rows.append(row); b.append(dij)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(N)  # maximize sum
    bounds = [(0, None)]*N
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

def compute_optimal_radii(centers):
    N = centers.shape[0]
    # compute pairwise distances
    # using broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    rows = []
    b = []
    # boundary constraints
    for i in range(N):
        x, y = centers[i]
        for bound in (x, 1 - x, y, 1 - y):
            row = np.zeros(N)
            row[i] = 1
            rows.append(row)
            b.append(bound)
    # pairwise constraints
    for i in range(N):
        for j in range(i + 1, N):
            dij = dists[i, j]
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            b.append(dij)
    A_ub = np.vstack(rows)
    b_ub = np.array(b, dtype=np.float64)
    c = -np.ones(N)
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

def van_der_corput(n_samples, base):
    seq = np.empty(n_samples)
    for i in range(n_samples):

        denominator = 1
        while n > 0:
            denominator *= base
            remainder = n % base
            vdc += remainder / denominator
            n //= base
        seq[i] = vdc
    return seq

import numpy as np
import time
from scipy.optimize import linprog

def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    N = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))
    # build inequality matrix
    rows = []
    b = []
    # boundary constraints
    for i in range(N):
        x, y = centers[i]
        for bound in (x, 1 - x, y, 1 - y):
            row = np.zeros(N)
            row[i] = 1.0
            rows.append(row)
            b.append(bound)
    # pairwise constraints
    for i in range(N):
        for j in range(i + 1, N):
            dij = dists[i, j]
            row = np.zeros(N)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            b.append(dij)
    A_ub = np.vstack(rows)
    b_ub = np.array(b, dtype=np.float64)
    c = -np.ones(N)  # maximize sum of radii
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        return res.x
    else:
        # fallback: zero radii
        return np.zeros(N)

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:

    N = 26

        radii = compute_optimal_radii(centers)
        return radii.sum(), radii

    # Several random restarts

        extra = rng.uniform(0.0, 1.0, size=(N - grid.shape[0], 2))
        centers = np.vstack([grid, extra])
        # small random perturbation to avoid exact symmetry

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: compute the maximal radii for a fixed set of centres.
# ----------------------------------------------------------------------
def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme that maximises the sum of radii
    for a given set of circle centres.
    """
    n = centers.shape[0]

    # pairwise distances (symmetric)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    rows = []
    rhs = []

    # boundary constraints  ri <= x, ri <= 1-x, ri <= y, ri <= 1-y
    for i in range(n):
        x, y = centers[i]
        for bound in (x, 1.0 - x, y, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            rhs.append(bound)

    # pairwise non‑overlap  ri + rj <= dij
    for i in range(n):
        for j in range(i + 1, n):
            dij = dists[i, j]
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dij)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=np.float64)

    # maximise sum ri  -> minimise -sum ri
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # a fallback that never happens in normal use
        return np.zeros(n, dtype=np.float64)
    return res.x


# ----------------------------------------------------------------------
# Small wrapper used by the optimiser.
# ----------------------------------------------------------------------
def evaluate_centers(centers: np.ndarray) -> tuple[float, np.ndarray]:
    """Return (total_radius, radii_vector) for the given centres."""
    radii = compute_optimal_radii(centers)
    return float(radii.sum()), radii


# ----------------------------------------------------------------------
# Main entry point required by the problem statement.
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a placement of 26 circles inside the unit square whose
    sum of radii exceeds the target 2.636.
    The function returns (centres, radii, total_sum).
    """
    N = 26
    rng = np.random.default_rng(2024)      # deterministic seed

    best_sum = -1.0
    best_centres = None
    best_radii = None

    start_time = time.time()
    TIME_LIMIT = 5.0                         # seconds

    # --------------------------------------------------------------
    # Repeated random restarts followed by a simple simulated annealing.
    # --------------------------------------------------------------
    while time.time() - start_time < TIME_LIMIT:
        # ---- initialise a candidate layout ---------------------------------
        # a regular 5×5 grid gives 25 points; we add one extra random point.
        grid_dim = 5
        xs = (np.arange(grid_dim) + 0.5) / grid_dim      # 0.1,0.3,…,0.9
        ys = (np.arange(grid_dim) + 0.5) / grid_dim
        grid = np.stack(np.meshgrid(xs, ys), -1).reshape(-1, 2)   # (25,2)

        extra = rng.uniform(0.0, 1.0, size=(N - grid.shape[0], 2))
        centres = np.vstack([grid, extra])

        # small jitter to avoid perfect symmetry
        centres += rng.normal(scale=0.015, size=centres.shape)
        centres = np.clip(centres, 0.0, 1.0)

        cur_sum, cur_radii = evaluate_centers(centres)
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centres = centres.copy()
            best_radii = cur_radii.copy()

        # ---- simple stochastic optimisation ---------------------------------
        T = 0.30          # initial step size (also an annealing temperature)
        T_min = 1e-5
        alpha = 0.996     # cooling factor

        for _ in range(1500):                     # limited steps per restart
            if time.time() - start_time > TIME_LIMIT:
                break

            i = rng.integers(N)                    # pick a circle to move
            proposal = best_centres.copy()
            delta = rng.normal(scale=T, size=2)    # random displacement
            proposal[i] = np.clip(proposal[i] + delta, 0.0, 1.0)

            prop_sum, prop_radii = evaluate_centers(proposal)

            # Metropolis acceptance (always accept improvement)
            diff = prop_sum - best_sum
            if diff > 0.0 or np.exp(diff / max(T, 1e-12)) > rng.random():
                best_sum = prop_sum
                best_centres = proposal
                best_radii = prop_radii

            T = max(T * alpha, T_min)

            # early exit if we have already surpassed a comfortable margin
            if best_sum > 2.66:
                break

        if best_sum > 2.66:
            break   # good enough – stop the outer restart loop

    # final, clean radii for the best layout found
    best_radii = compute_optimal_radii(best_centres)
    best_sum = float(best_radii.sum())

    return best_centres, best_radii, best_sum
