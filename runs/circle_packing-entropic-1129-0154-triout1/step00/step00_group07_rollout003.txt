import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distance matrix
    # using broadcasting
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraint matrix
    m_pair = n*(n-1)//2
    m = m_pair + 4*n
    A = np.zeros((m, n))
    b = np.zeros(m)

    # pair constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1

    # boundary constraints
    xs = centers[:,0]
    ys = centers[:,1]
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = xs[i]
        idx += 1
        A[idx, i] = 1.0
        b[idx] = 1.0 - xs[i]
        idx += 1
        A[idx, i] = 1.0
        b[idx] = ys[i]
        idx += 1
        A[idx, i] = 1.0
        b[idx] = 1.0 - ys[i]
        idx += 1

    c = -np.ones(n)  # maximize sum radii => minimize negative sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: set all radii zero
        return np.zeros(n), -np.inf
    radii = res.x
    total = -res.fun
    return radii, total

def generate_random_centers(n):
    return np.random.rand(n, 2)

def generate_grid_centers():
    # 5x5 grid with margin 0.1
    margin = 0.1
    rows = cols = 5
    xs = np.linspace(margin, 1-margin, cols)
    ys = np.linspace(margin, 1-margin, rows)
    X, Y = np.meshgrid(xs, ys)
    points = np.column_stack([X.ravel(), Y.ravel()])  # 25 points
    # add extra point that is not on grid, e.g., (0.5, 0.2)
    extra = np.array([[0.5, 0.2]])
    points = np.vstack([points, extra])
    return points

def generate_hex_centers():
    s = 0.2
    dx = s
    dy = np.sqrt(3)/2 * s
    points = []
    y = s/2
    row = 0
    while y <= 1 - s/2 + 1e-12:
        offset = 0.0 if row % 2 == 0 else dx/2
        x_start = s/2 + offset
        xs = np.arange(x_start, 1 - s/2 + 1e-12, dx)
        for x in xs:
            points.append([x, y])
        y += dy
        row += 1
    points = np.array(points)
    if points.shape[0] >= 26:
        idx = np.random.choice(points.shape[0], 26, replace=False)
        return points[idx]
    else:
        extra_needed = 26 - points.shape[0]
        extra = np.random.rand(extra_needed, 2)
        return np.vstack([points, extra])

def evaluate_centers(centers):
    radii, total = solve_radii(centers)
    return radii, total

def hill_climb(centers, initial_radii, initial_sum, iterations=1000, sigma=0.02):
    n = centers.shape[0]
    best_centers = centers.copy()
    best_radii = initial_radii.copy()
    best_sum = initial_sum
    for _ in range(iterations):
        i = np.random.randint(n)
        new_center = best_centers[i] + np.random.normal(scale=sigma, size=2)
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        new_centers = best_centers.copy()
        new_centers[i] = new_center
        radii, total = solve_radii(new_centers)
        if total > best_sum + 1e-8:
            best_centers = new_centers
            best_radii = radii
            best_sum = total
    return best_centers, best_radii, best_sum

def run_packing():
    np.random.seed(0)
    n = 26
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # evaluate a few deterministic seeds first
    patterns = [generate_grid_centers, generate_hex_centers]
    for gen in patterns:
        centers = gen()
        radii, total = solve_radii(centers)
        if total > best_sum:
            best_sum = total
            best_centers = centers
            best_radii = radii

    # random search
    for _ in range(1200):
        centers = generate_random_centers(n)
        radii, total = solve_radii(centers)
        if total > best_sum:
            best_sum = total
            best_centers = centers
            best_radii = radii

    # local hill climb starting from best found
    if best_centers is not None:
        best_centers, best_radii, best_sum = hill_climb(best_centers, best_radii, best_sum, iterations=2000, sigma=0.03)

    # final sanity: ensure radii non-negative
    best_radii = np.maximum(best_radii, 0.0)

    return best_centers, best_radii, float(best_sum)

The problem is a non‑convex packing problem, but for a *fixed* set of circle centres the radii that maximise the total sum are the solution of a **linear program**:

```
max   Σ r_i

      r_i ≤ x_i ,  r_i ≤ 1‑x_i                     (left/right borders)
      r_i ≤ y_i ,  r_i ≤ 1‑y_i                     (bottom/top borders)

```

import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a fixed set of centres compute the radii that maximise the sum.
    Returns (radii, total_sum).
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of linear constraints:
    #   - one for each unordered pair (i,j)
    #   - four for each circle (left, right, bottom, top)
    m_pair = n * (n - 1) // 2
    m = m_pair + 4 * n

    A = np.zeros((m, n))
    b = np.zeros(m)

    # pairwise non‑overlap constraints
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints
    xs = centers[:, 0]
    ys = centers[:, 1]
    for i in range(n):
        # left side  r_i ≤ x_i
        A[row, i] = 1.0
        b[row] = xs[i]
        row += 1
        # right side r_i ≤ 1‑x_i
        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]
        row += 1
        # bottom      r_i ≤ y_i
        A[row, i] = 1.0
        b[row] = ys[i]
        row += 1
        # top         r_i ≤ 1‑y_i
        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]
        row += 1

    # maximise Σ r_i  ⇔ minimise –Σ r_i
    c = -np.ones(n)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # fallback – all radii zero (should never happen)
        return np.zeros(n), 0.0

    radii = res.x
    total = -res.fun
    return radii, float(total)


def generate_grid_centers() -> np.ndarray:
    """5×5 square grid (margin 0.1) plus one extra point not on the grid."""
    margin = 0.1
    rows = cols = 5
    xs = np.linspace(margin, 1.0 - margin, cols)
    ys = np.linspace(margin, 1.0 - margin, rows)
    X, Y = np.meshgrid(xs, ys)
    pts = np.column_stack([X.ravel(), Y.ravel()])   # 25 points
    extra = np.array([[0.7, 0.7]])                  # a safe extra point
    return np.vstack([pts, extra])


def generate_hex_centers() -> np.ndarray:
    """Simple hexagonal pattern, then trimmed / padded to 26 points."""
    s = 0.2                     # horizontal spacing ≈ 2·desired radius
    dx = s
    dy = np.sqrt(3.0) / 2.0 * s
    pts = []
    y = s / 2.0
    row = 0
    while y <= 1.0 - s / 2.0 + 1e-12:
        offset = 0.0 if row % 2 == 0 else dx / 2.0
        x_start = s / 2.0 + offset
        xs = np.arange(x_start, 1.0 - s / 2.0 + 1e-12, dx)
        for x in xs:
            pts.append([x, y])
        y += dy
        row += 1
    pts = np.array(pts)
    if pts.shape[0] >= 26:
        idx = np.random.choice(pts.shape[0], 26, replace=False)
        return pts[idx]
    else:
        extra = np.random.rand(26 - pts.shape[0], 2)
        return np.vstack([pts, extra])


def generate_random_centers(n: int = 26) -> np.ndarray:
    """Uniform random points inside the unit square."""
    return np.random.rand(n, 2)


def hill_climb(
    centers: np.ndarray,
    radii: np.ndarray,
    total: float,
    iterations: int = 2000,
    sigma_start: float = 0.04,
    sigma_end: float = 0.002,
) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Simple stochastic hill‑climb.
    At each step one centre is perturbed; the LP is recomputed.
    Only moves that improve the total radius are kept.
    """
    n = centers.shape[0]
    best_c = centers.copy()
    best_r = radii.copy()
    best_total = total

    for it in range(iterations):
        # linear schedule for perturbation size
        sigma = sigma_start * ((sigma_end / sigma_start) ** (it / iterations))

        i = np.random.randint(n)
        new_center = best_c[i] + np.random.normal(scale=sigma, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)

        trial_c = best_c.copy()
        trial_c[i] = new_center

        trial_r, trial_total = solve_radii(trial_c)
        if trial_total > best_total + 1e-9:
            best_c, best_r, best_total = trial_c, trial_r, trial_total

    return best_c, best_r, float(best_total)


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles.
    The algorithm first evaluates a few deterministic patterns,
    then performs a random search, and finally refines the best
    candidate with a short hill‑climb.
    """
    np.random.seed(0)          # reproducibility
    n_circles = 26

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ------------------------------------------------------------------
    # 1. test a couple of simple deterministic patterns
    # ------------------------------------------------------------------
    for gen in (generate_grid_centers, generate_hex_centers):
        c = gen()
        r, s = solve_radii(c)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    # ------------------------------------------------------------------
    # 2. random sampling
    # ------------------------------------------------------------------
    N_RANDOM = 1200
    for _ in range(N_RANDOM):
        c = generate_random_centers(n_circles)
        r, s = solve_radii(c)
        if s > best_sum:
            best_sum = s
            best_centers = c
            best_radii = r

    # ------------------------------------------------------------------
    # 3. local improvement by hill‑climbing
    # ------------------------------------------------------------------
    if best_centers is not None:
        best_centers, best_radii, best_sum = hill_climb(
            best_centers, best_radii, best_sum,
            iterations=2500,
            sigma_start=0.04,
            sigma_end=0.003,
        )

    # safety – radii must be non‑negative
    best_radii = np.maximum(best_radii, 0.0)

    return best_centers, best_radii, best_sum
