```
def solve_radii(centers):
    n = centers.shape[0]
    # compute bounds
    bound_i = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    bounds = [(0, bound_i[i]) for i in range(n)]

    # pairwise constraints
    pair_count = n*(n-1)//2
    A = np.zeros((pair_count, n))
    b = np.zeros(pair_count)
    idx = 0
    for i in range(n):
        ci = centers[i]
        for j in range(i+1, n):
            dist = np.linalg.norm(ci - centers[j])
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dist
            idx += 1
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback radii zero
        radii = np.zeros(n)
    else:
        radii = res.x
    # ensure non-negative numeric noise
    radii = np.maximum(radii, 0.0)
    total = radii.sum()
    return radii, float(total)
```

```python
import numpy as np
from scipy.optimize import linprog

def _solve_radii(centers: np.ndarray):
    """
    Solve the linear program that maximizes the sum of radii for a fixed set of centres.
    """
    n = centers.shape[0]
    # Upper bounds from the square borders
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0],
                                centers[:,1], 1-centers[:,1]])
    # Build pairwise constraints r_i + r_j <= dist_ij
    m = n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    idx = 0
    for i in range(n):
        ci = centers[i]
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.linalg.norm(ci - centers[j])
            idx += 1
    # Linear objective: maximize sum(r)  <=> minimize -sum(r)
    c = -np.ones(n, dtype=np.float64)

    # Variable bounds for each radius
    lin_bounds = [(0.0, float(bounds[i])) for i in range(n)]

    # Solve the LP with the high-performance HiGHS backend
    res = linprog(c, A_ub=A, b_ub=b, bounds=lin_bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # In the unlikely event the LP fails, fall back to zero radii
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = np.maximum(res.x, 0.0)   # clip tiny negatives
    return radii, float(radii.sum())
```

The task is to produce 26 circles that lie inside the unit square, do not overlap and give a large sum of radii.  
For a *fixed* set of centre points the optimal radii are obtained by the linear program  

```
max   Σ r_i
s.t.  0 ≤ r_i ≤ distance of centre i to the four sides
      r_i + r_j ≤ distance(centre_i , centre_j)   for every i < j
```

The linear program is cheap (26 variables, ≤ 350 constraints) and can be solved
with `scipy.optimize.linprog`.  
The overall strategy is a lightweight stochastic search:

1. **Baseline** – a regular 5×5 grid (25 circles) plus one extra centre in the centre.
2. **Random starts** – many uniformly random point sets.
3. **Local hill‑climbing** – repeatedly pick a circle, jitter its centre, resolve the LP and keep the move if the total radius grows.
4. Keep the best configuration ever found (stopping early if we reach the target
   `2.636`).

All helper functions are top‑level, there are no lambdas or closures, and the
routine returns the required `(centers, radii, sum_radii)` tuple.

```python
import numpy as np
from scipy.optimize import linprog

def _solve_radii(centers: np.ndarray):
    """
    Solve the LP that maximises Σ r_i for a given set of centres.
    Returns the radii array and their sum.
    """
    n = centers.shape[0]

    # Upper bounds from the four sides of the square
    side_limits = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0],
         centers[:, 1], 1.0 - centers[:, 1]]
    )
    bounds = [(0.0, float(side_limits[i])) for i in range(n)]

    # Pairwise constraints   r_i + r_j ≤ distance(i, j)
    m = n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    idx = 0
    for i in range(n):
        ci = centers[i]
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = np.linalg.norm(ci - centers[j])
            idx += 1

    # Objective: maximise Σ r_i  ↔ minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # In the very unlikely event the LP fails, fall back to zero radii
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = np.maximum(res.x, 0.0)       # clip tiny negatives caused by numerics

    return radii, float(radii.sum())

def _initial_grid():
    """
    Regular 5×5 grid (0.1‑0.9) plus a centre point.
    """
    grid = np.array(
        [(0.1 + 0.2 * i, 0.1 + 0.2 * j) for i in range(5) for j in range(5)],
        dtype=np.float64,
    )
    extra = np.array([[0.5, 0.5]], dtype=np.float64)
    return np.vstack([grid, extra])

def run_packing():
    """
    Returns (centers, radii, sum_radii) for a packing of 26 circles.
    The routine performs a modest stochastic optimisation, keeping the
    best feasible configuration discovered.
    """
    rng = np.random.default_rng(0xDEADBEEF)   # reproducible but “random”

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ---------------------------------------------------------
    # 1) deterministic baseline (grid + centre)
    # ---------------------------------------------------------
    centres_grid = _initial_grid()
    radii, total = _solve_radii(centres_grid)
    best_sum = total
    best_centers = centres_grid.copy()
    best_radii = radii.copy()

    # ---------------------------------------------------------
    # 2) stochastic search + tiny hill‑climbing
    # ---------------------------------------------------------
    for trial in range(100):                     # number of random starts
        # random points, kept a little away from the border
        centres = rng.uniform(low=0.05, high=0.95, size=(26, 2))
        radii, total = _solve_radii(centres)

        # short local improvement: move one centre at a time
        for _ in range(20):
            i = rng.integers(26)
            proposal = centres[i] + rng.normal(scale=0.06, size=2)
            proposal = np.clip(proposal, 0.0, 1.0)

            old = centres[i].copy()
            centres[i] = proposal
            new_radii, new_total = _solve_radii(centres)

            if new_total > total:               # keep only improving moves
                radii, total = new_radii, new_total
            else:
                centres[i] = old                 # revert

        if total > best_sum:
            best_sum = total
            best_centers = centres.copy()
            best_radii = radii.copy()
            if best_sum >= 2.636:                # reached the target → stop early
                break

    # ---------------------------------------------------------
    # 3) safety net (should never be needed)
    # ---------------------------------------------------------
    if best_centers is None:
        best_centers, best_radii, best_sum = centres_grid, radii, total

    return best_centers, best_radii, float(best_sum)
```
