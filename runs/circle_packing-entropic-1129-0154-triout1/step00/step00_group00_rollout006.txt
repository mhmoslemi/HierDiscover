import cvxpy as cp

def compute_optimal_radii(centers):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r >= 0]
    constraints += [r <= centers[:,0]]
    constraints += [r <= 1 - centers[:,0]]
    constraints += [r <= centers[:,1]]
    constraints += [r <= 1 - centers[:,1]]
    # Pairwise
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, verbose=False, max_iters=5000)
    # In case of failure or nonoptimal, maybe fallback.
    if r.value is None:
        # fallback: zero radii
        return np.zeros(n), 0.0
    radii = np.maximum(r.value, 0.0)
    return radii, np.sum(radii)

import numpy as np
from scipy.optimize import linprog

def compute_optimal_radii(centers):
    n = len(centers)
    # compute boundary limits
    bounds = []
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1 - x, y, 1 - y)
        bounds.append((0.0, max_r))
    # pairwise constraints
    rows = []
    rhs = []
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d)
    A_ub = np.array(rows)
    b_ub = np.array(rhs)
    c = -np.ones(n)  # maximize sum -> minimize negative sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        radii = np.maximum(res.x, 0.0)
        sum_r = radii.sum()
        return radii, sum_r
    else:
        # fallback: zeros
        radii = np.zeros(n)
        return radii, 0.0
```

```
def compute_optimal_radii(centers):
    n = centers.shape[0]
    # bounds for each variable r_i
    max_r = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    bounds = [(0.0, v) for v in max_r]
    # pairwise constraints
    pair_indices = [(i, j) for i in range(n) for j in range(i+1, n)]
    A = np.zeros((len(pair_indices), n))
    b = np.empty(len(pair_indices))
    for idx, (i, j) in enumerate(pair_indices):
        A[idx, i] = 1.0
        A[idx, j] = 1.0
        b[idx] = np.linalg.norm(centers[i] - centers[j])
    # objective to maximize sum r => minimize -sum r
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        radii = np.maximum(res.x, 0.0)
        return radii, radii.sum()
    else:
        # fallback
        return np.zeros(n), 0.0
```

def hill_climb(initial_centers, max_iter=2000, step_init=0.3, seed=None):
    if seed is not None:
        rng = np.random.default_rng(seed)
    else:
        rng = np.random.default_rng()
    centers = initial_centers.copy()
    radii, cur_sum = compute_optimal_radii(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    best_sum = cur_sum
    step = step_init
    n = len(centers)
    for it in range(max_iter):
        i = rng.integers(n)
        # propose move
        delta = (rng.random(2) - 0.5) * 2 * step  # uniform in [-step, step]
        new_center = centers[i] + delta
        # Clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # If after clipping center does not move (i.e., at boundary) maybe accept anyway
        new_centers = centers.copy()
        new_centers[i] = new_center
        new_radii, new_sum = compute_optimal_radii(new_centers)
        # Accept if improvement
        if new_sum > cur_sum + 1e-9:
            centers = new_centers
            radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
        # else reject
        step *= 0.9999  # slowly decay
    return best_centers, best_radii, best_sum
```

import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def compute_optimal_radii(centers: np.ndarray, eps: float = 1e-8) -> Tuple[np.ndarray, float]:
    """
    Solve the linear program for a fixed set of centres.
    Returns optimal radii and their sum.
    """
    n = centers.shape[0]

    # Upper bounds imposed by the square
    max_r = (
        np.minimum.reduce(
            [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
        )
        - eps
    )
    max_r = np.maximum(max_r, 0.0)
    bounds = [(0.0, max_r[i]) for i in range(n)]

    # Pairwise non‑overlap constraints: r_i + r_j <= distance(i,j) - eps
    pair_cnt = n * (n - 1) // 2
    A = np.zeros((pair_cnt, n))
    b = np.empty(pair_cnt)
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            d = np.linalg.norm(centers[i] - centers[j]) - eps
            b[idx] = max(d, 0.0)
            idx += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback – all radii zero
        return np.zeros(n), 0.0

    radii = np.maximum(res.x, 0.0)
    # enforce the tiny safety margin explicitly
    radii = np.minimum(radii, max_r)
    return radii, float(radii.sum())


def hill_climb(initial_centers: np.ndarray,
               max_iter: int = 2000,
               step_init: float = 0.15,
               eps: float = 1e-8) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Simple hill‑climbing optimisation on the centre positions.
    """
    rng = np.random.default_rng()
    centres = initial_centers.copy()
    radii, cur_sum = compute_optimal_radii(centres, eps=eps)

    best_centres = centres.copy()
    best_radii = radii.copy()
    best_sum = cur_sum

    step = step_init
    for _ in range(max_iter):
        i = rng.integers(centres.shape[0])

        # random displacement
        delta = (rng.random(2) - 0.5) * 2.0 * step
        new_centre = centres[i] + delta
        new_centre = np.clip(new_centre, 0.0, 1.0)

        new_centres = centres.copy()
        new_centres[i] = new_centre
        new_radii, new_sum = compute_optimal_radii(new_centres, eps=eps)

        # accept only improving moves
        if new_sum > cur_sum + 1e-9:
            centres = new_centres
            radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres.copy()
                best_radii = new_radii.copy()

        # slowly reduce the step size – a tiny amount of simulated annealing
        step *= 0.9995
        if step < 1e-4:
            step = 1e-4

    return best_centres, best_radii, best_sum


def init_grid() -> np.ndarray:

    grid = np.column_stack((xs.ravel(), ys.ravel()))  # 25×2

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """

    Returns centres (26×2), radii (26,), and the total sum of radii.
    """

    best_total = -1.0

        if total_opt > best_total:
            best_total = total_opt
            best_centres = centres_opt

    return best_centres, best_radii, float(best_total)
