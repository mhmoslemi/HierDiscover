```python
import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # distances to borders
    d_edge = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    # distances between centers
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diff**2, axis=2))
    # build constraints
    m = n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    # edge constraints
    for i in range(n):
        A[row, i] = 1
        b[row] = d_edge[i]
        row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = d_pair[i, j]
            row += 1
    c = -np.ones(n)  # maximize sum radii
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = np.sum(radii)
    return radii, sum_r
```

    from scipy.optimize import linprog, differential_evolution

    n = 26

    def compute_radii_and_sum(centers):
        # centers shape (n,2)
        # compute edge distances
        d_edge = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
        # pairwise distances
        diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
        d_pair = np.sqrt(np.sum(diff**2, axis=2))
        # constraints: A_ub * r <= b_ub
        m = n + n*(n-1)//2
        A = np.zeros((m, n))
        b = np.empty(m)
        row = 0
        # edge constraints
        for i in range(n):
            A[row, i] = 1
            b[row] = d_edge[i]
            row += 1

                A[row, j] = 1

import numpy as np
from scipy.optimize import linprog, differential_evolution

# ----------------------------------------------------------------------
# Global constant: number of circles
N_CIRCLES = 26
# ----------------------------------------------------------------------


def _max_radii_from_centers(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the linear program that yields the largest possible radii for a given
    set of centre positions.

    Parameters
    ----------
    centers : np.ndarray, shape (N_CIRCLES, 2)
        (x, y) coordinates of the circles.

    Returns
    -------
    radii : np.ndarray, shape (N_CIRCLES,)
        Optimal radii (non‑negative) satisfying all constraints.
    total : float
        Sum of the radii.
    """
    n = N_CIRCLES
    # distance to each side of the unit square
    d_edge = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d_pair = np.sqrt(np.sum(diff ** 2, axis=2))

    # Build the inequality matrix A_ub * r <= b_ub
    n_pair = n * (n - 1) // 2
    n_cons = n + n_pair
    A = np.zeros((n_cons, n), dtype=float)
    b = np.empty(n_cons, dtype=float)

    # Edge constraints: r_i <= d_edge_i
    row = 0
    for i in range(n):
        A[row, i] = 1.0
        b[row] = d_edge[i]
        row += 1

    # Pairwise constraints: r_i + r_j <= distance_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d_pair[i, j]
            row += 1

    # Linear objective: maximise sum(r_i)  -> minimise -sum(r_i)
    c = -np.ones(n, dtype=float)

    # Bounds for each radius (non‑negative, no explicit upper bound)
    bounds = [(0.0, None) for _ in range(n)]

    # Solve the LP using the high‑performance HiGHS backend
    lp_res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not lp_res.success:
        # In the unlikely case of failure, fall back to zero radii
        radii = np.zeros(n, dtype=float)
        total = 0.0
    else:
        radii = lp_res.x
        total = float(np.sum(radii))

    return radii, total


def _objective_flat(x: np.ndarray) -> float:
    """
    Objective for the optimiser: negative of the maximal total radius.
    The optimiser works on a flat array of length 2*N_CIRCLES.
    """
    centers = x.reshape((N_CIRCLES, 2))
    _, total = _max_radii_from_centers(centers)
    # differential_evolution minimises, so return negative total radius
    return -total


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Compute a packing of 26 circles inside the unit square that maximises the
    sum of their radii.

    Returns
    -------
    centers : np.ndarray, shape (26, 2)
        Coordinates of the circle centres.
    radii   : np.ndarray, shape (26,)
        Radii of the circles (maximal for the returned centres).
    sum_r   : float
        Sum of the radii.
    """
    # ------------------------------------------------------------------
    # Global optimisation with Differential Evolution
    # ------------------------------------------------------------------
    bounds = [(0.0, 1.0) for _ in range(2 * N_CIRCLES)]

    # The parameters below give a reasonably thorough search while staying
    # within typical execution limits.
    de_result = differential_evolution(
        _objective_flat,
        bounds,
        maxiter=250,      # number of generations
        popsize=20,       # population size (popsize * len(x) individuals)
        polish=True,      # final local refinement by SLSQP
        seed=42,          # deterministic runs
        updating="deferred",  # minimise Python overhead
    )

    best_centers = de_result.x.reshape((N_CIRCLES, 2))
    best_radii, best_total = _max_radii_from_centers(best_centers)

    # ------------------------------------------------------------------
    # Light local refinement (hill‑climbing) to squeeze a little more
    # ------------------------------------------------------------------
    centers = best_centers.copy()
    radii = best_radii.copy()
    total = best_total
    step = 0.15  # initial random move amplitude

    rng = np.random.default_rng(1234)

    for _ in range(2000):
        i = rng.integers(N_CIRCLES)
        new_center = centers[i] + rng.uniform(-step, step, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)

        trial_centers = centers.copy()
        trial_centers[i] = new_center

        trial_radii, trial_total = _max_radii_from_centers(trial_centers)

        if trial_total > total:
            centers = trial_centers
            radii = trial_radii
            total = trial_total

        # slowly shrink the move size
        step *= 0.9995
        if step < 1e-4:
            step = 1e-4

    return centers, radii, float(total)
