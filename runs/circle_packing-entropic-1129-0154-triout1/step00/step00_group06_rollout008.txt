```
def compute_optimal_radii(centers):
    n = centers.shape[0]
    # Build A_ub and b_ub
    rows = []
    b = []
    # boundary constraints
    xs = centers[:,0]; ys = centers[:,1]
    rows.extend(np.eye(n))
    b.extend(xs)
    rows.extend(np.eye(n))
    b.extend(1 - xs)
    rows.extend(np.eye(n))
    b.extend(ys)
    rows.extend(np.eye(n))
    b.extend(1 - ys)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            d = np.linalg.norm(centers[i] - centers[j])
            b.append(d)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback maybe return zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = float(radii.sum())
    return radii, sum_r
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    n=26
    # initial centers: 5x5 grid plus one extra random
    grid = np.linspace(0.1, 0.9, 5)
    centers = [(x,y) for x in grid for y in grid]
    centers = np.array(centers[:25])
    # add extra extra circle at random position about center 0.5,0.5 but offset a bit
    extra = np.array([0.5, 0.5]) + (np.random.rand(2)-0.5)*0.1
    extra = np.clip(extra, 0,1)
    centers = np.vstack([centers, extra])

    # local search
    best_centers = centers.copy()
    best_radii, best_sum = compute_optimal_radii(best_centers)
    # main loop
    rng = np.random.default_rng(0)
    temperature = 0.1
    for iteration in range(2000):
        # propose move: randomly select one circle to move
        i = rng.integers(n)
        # propose new position by adding gaussian offset scaled by temperature
        new_center = best_centers[i] + rng.normal(scale=temperature, size=2)
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # create new centers copy
        new_centers = best_centers.copy()
        new_centers[i] = new_center
        # compute radii
        new_radii, new_sum = compute_optimal_radii(new_centers)
        # accept if sum bigger or with probability exp((new_sum - best_sum)/temp)
        if new_sum > best_sum or rng.random() < np.exp((new_sum - best_sum)/temperature):
            best_centers = new_centers
            best_radii = new_radii
            best_sum = new_sum
        # cool down
        temperature *= 0.9995 # or linearly reduce
    # after loop, finalize with best solution
    return best_centers, best_radii, float(best_sum)
```

```
def run_packing():
```

```
        # initialize
        # start with grid + extra random
        grid_coords = np.linspace(0.1, 0.9, 5)
        X, Y = np.meshgrid(grid_coords, grid_coords)
        centers = np.column_stack([X.ravel(), Y.ravel()])[:25]
        # add extra random location not close to any existing (distance >0.05)
        extra = None
        while extra is None or any(np.linalg.norm(centers - extra, axis=1) < 0.08):
            extra = rng.random(2)
            extra = np.clip(extra, 0.05, 0.95)
        centers = np.vstack([centers, extra])
        # maybe shuffle
        rng.shuffle(centers)
        # compute initial radii
        radii, sum_r = compute_optimal_radii(centers)
        cur_best_sum = sum_r
        cur_best_centers = centers.copy()
        cur_best_radii = radii
        # local search
        step = 0.15
        for it in range(1000):
            i = rng.integers(n)
            new_center = cur_best_centers[i] + rng.normal(scale=step, size=2)
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = cur_best_centers.copy()
            new_centers[i] = new_center
            # compute LP
            new_radii, new_sum = compute_optimal_radii(new_centers)
            # accept if better
            if new_sum > cur_best_sum:
                cur_best_sum = new_sum
                cur_best_centers = new_centers
                cur_best_radii = new_radii
                # optionally increase step?
            # cooling
            step *= 0.999
        # keep global best
        if cur_best_sum > best_sum:
            best_sum = cur_best_sum
            best_centers = cur_best_centers
            best_radii = cur_best_radii

    return best_centers, best_radii, float(best_sum)
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    n = 26
    rng = np.random.default_rng(123)

    # initial centers: regular grid + random extra
    grid = np.linspace(0.1,0.9,5)
    X,Y = np.meshgrid(grid,grid)
    centers = np.column_stack([X.ravel(),Y.ravel()])[:25]
    # add extra
    extra = np.array([0.4,0.4]) + (rng.random(2)-0.5)*0.1
    extra = np.clip(extra,0.0,1.0)
    centers = np.vstack([centers, extra])
    # random shuffle
    rng.shuffle(centers)

    # compute radii
    radii, sum_r = compute_optimal_radii(centers)
    best_sum = sum_r
    best_centers = centers.copy()
    best_radii = radii.copy()

    # annealing parameters
    T = 0.05
    step = 0.2
    for it in range(3000):
        # propose new centers:
        # choose a random subset size s (1..5)
        s = rng.integers(1,6)
        idx = rng.choice(n, size=s, replace=False)
        new_centers = best_centers.copy()
        # add random offsets
        offsets = rng.normal(scale=step, size=(s,2))
        new_centers[idx] += offsets
        # clip
        new_centers = np.clip(new_centers, 0.0, 1.0)
        # evaluate
        new_radii, new_sum = compute_optimal_radii(new_centers)
        # acceptance
        delta = new_sum - best_sum
        if delta > 0 or rng.random() < np.exp(delta / T):
            best_sum = new_sum
            best_centers = new_centers
            best_radii = new_radii
        # cooling
        T *= 0.9995
        step *= 0.9995
        # occasional bigger step increase maybe
        if it % 500 == 0:
            # Reset step to larger number occasionally?
            step = max(step, 0.05)

    return best_centers, best_radii, float(best_sum)
```

```
    # boundaries
    xs = centers[:,0]

    num_constraints = 4*n + n*(n-1)//2

    b = np.zeros(num_constraints, dtype=float)

        A[row,i] = 1.0
        b[row] = xs[i]
        row += 1

    for i in range(n):
        A[row,i] = 1.0
        b[row] = 1.0 - xs[i]
```

The problem is a two‑stage optimisation:

1. **Radii optimisation for fixed centres** –  
   For a given set of centre positions the radii that maximise the total
   sum are the solution of a linear program  

   \[
   \max\; \sum_i r_i \quad\text{s.t.}\quad    
   r_i \le x_i,\; r_i \le 1-x_i,\; r_i \le y_i,\; r_i \le 1-y_i,
   \]
   \[
   r_i+r_j \le d_{ij}\;(i<j),\qquad r_i\ge0 .
   \]

   Here \(d_{ij}=||c_i-c_j||\) are the Euclidean distances between the
   centres.  This LP is solved with `scipy.optimize.linprog`
   (`method='highs'`).  For any arrangement of centres the solver yields the
   optimal radii and the corresponding sum.

2. **Search over centre positions** –  
   The sum of the optimal radii is a black‑box function of the 52
   coordinates.  A simple simulated‑annealing scheme is used:

   * start from a 5×5 regular grid (25 circles of radius 0.1) plus one extra
     circle placed in a grid gap;
   * repeatedly pick a random subset of circles (1‑5 circles) and move them
     by a small Gaussian perturbation;
   * evaluate the new sum of radii (by solving the LP);
   * accept the move if the sum increases, otherwise accept it with a
     probability \(\exp((\Delta)/T)\) where \(T\) is a slowly decreasing

     slightly below 1 each iteration so the search becomes finer.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear‑program solver: for a fixed set of centres returns the
# optimal radii (maximising the sum) and the sum itself.
# ------------------------------------------------------------
def optimal_radii(centres: np.ndarray):
    """
    Solve the LP that maximises sum of radii for the supplied centre
    positions.

    Parameters
    ----------
    centres : np.ndarray of shape (n,2)
        (x, y) coordinates of the circles, 0 ≤ x,y ≤ 1.

    Returns
    -------
    radii : np.ndarray of shape (n,)
        The optimal radii (non‑negative, satisfy all constraints).
    total : float
        Sum of the radii.
    """
    n = centres.shape[0]
    # boundary limits
    xs = centres[:, 0]
    ys = centres[:, 1]

    # number of linear constraints:
    # 4 per circle for the four sides + one for each unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # r_i ≤ x_i
    for i in range(n):
        A[row, i] = 1.0
        b[row] = xs[i]
        row += 1
    # r_i ≤ 1 - x_i
    for i in range(n):
        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]
        row += 1
    # r_i ≤ y_i
    for i in range(n):
        A[row, i] = 1.0
        b[row] = ys[i]
        row += 1
    # r_i ≤ 1 - y_i
    for i in range(n):
        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]
        row += 1
    # pairwise non‑overlap: r_i + r_j ≤ distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            # Euclidean distance
            d = np.hypot(centres[i, 0] - centres[j, 0],
                         centres[i, 1] - centres[j, 1])
            b[row] = d
            row += 1

    # maximise sum(r_i)  <=> minimise -sum(r_i)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # Use the HiGHS solver (fast and reliable)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback – return all zeros (still a feasible packing)
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x

    total = float(radii.sum())
    return radii, total


# ------------------------------------------------------------
# Simulated‑annealing optimiser over centre positions.
# ------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a packing of 26 circles inside the unit square that
    maximises the sum of the radii.

    Returns
    -------
    centres : np.ndarray, shape (26, 2)
        (x, y) coordinates of the circles.
    radii   : np.ndarray, shape (26,)
        Radii corresponding to the centres (optimal for the given layout).
    total   : float
        Sum of the radii.
    """
    rng = np.random.default_rng(12345)   # reproducible randomness
    n = 26

    # ---- initialise with a regular 5×5 grid + one extra circle ----
    grid_vals = np.linspace(0.1, 0.9, 5)               # 0.1,0.3,…,0.9
    X, Y = np.meshgrid(grid_vals, grid_vals)
    centres = np.column_stack([X.ravel(), Y.ravel()])[:25]

    # a genuine gap centre (the centre of one of the 4‑circle cells)
    extra = np.array([0.4, 0.4])
    centres = np.vstack([centres, extra])

    # random shuffle – does not affect the geometry but breaks any ordering bias
    rng.shuffle(centres)

    # evaluate the initial layout
    best_radii, best_total = optimal_radii(centres)
    best_centres = centres.copy()

    # ---- simulated‑annealing parameters ----
    temperature = 0.6          # initial temperature
    step_scale   = 0.3         # initial Gaussian step size
    cooling_rate = 0.9995      # exponential cooling
    min_step     = 0.005

    iters = 5000               # total number of proposals

    for it in range(iters):
        # How many circles to move this step? 1–5 (chosen uniformly)
        k = rng.integers(1, 6)
        idx = rng.choice(n, size=k, replace=False)

        # propose a move
        proposal = best_centres.copy()
        proposal[idx] += rng.normal(scale=step_scale, size=(k, 2))

        # keep inside the unit square
        np.clip(proposal, 0.0, 1.0, out=proposal)

        # evaluate the new configuration
        new_radii, new_total = optimal_radii(proposal)

        delta = new_total - best_total
        if delta > 0.0 or rng.random() < np.exp(delta / temperature):
            # accept the move
            best_centres = proposal
            best_radii  = new_radii
            best_total  = new_total

        # cool down
        temperature *= cooling_rate
        step_scale   *= cooling_rate
        if step_scale < min_step:
            step_scale = min_step

        # occasional re‑anneal: reset temperature a little to keep exploration
        if (it + 1) % 1000 == 0:
            temperature = 0.6

    # final LP solve (just to be safe that we have the truly optimal radii
    # for the final centre set)
    final_radii, final_total = optimal_radii(best_centres)

    return best_centres, final_radii, final_total
```
