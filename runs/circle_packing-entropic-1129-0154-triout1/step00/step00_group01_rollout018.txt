def compute_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n) # maximize sum

    # build constraints A_ub x <= b_ub
    m_boundary = 4 * n
    m_pairs = n*(n-1)//2
    m = m_boundary + m_pairs
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0

    xs = centers[:,0]
    ys = centers[:,1]

    for i in range(n):
        # r_i <= x_i
        A[row,i] = 1
        b[row] = xs[i]
        row += 1
        # r_i <= 1 - x_i
        A[row,i] = 1
        b[row] = 1 - xs[i]
        row += 1
        # r_i <= y_i
        A[row,i] = 1
        b[row] = ys[i]
        row += 1
        # r_i <= 1 - y_i
        A[row,i] = 1
        b[row] = 1 - ys[i]
        row += 1

    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row,i] = 1
            A[row,j] = 1
            b[row] = d
            row += 1

    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs', options={'presolve': True})
    if not res.success:
        # fallback: radii zeros
        radii = np.zeros(n)
    else:
        radii = res.x

    # optionally clip negative small values
    radii = np.maximum(radii, 0.0)

    return radii, np.sum(radii)

def compute_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n)
    # boundary constraints
    # count: 4*n + n*(n-1)//2
    num_pairs = n*(n-1)//2
    m = 4*n + num_pairs
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)
    row = 0
    xs = centers[:, 0]
    ys = centers[:, 1]

    # boundary constraints
    for i in range(n):
        A[row, i] = 1
        b[row] = xs[i]
        row += 1
        A[row, i] = 1
        b[row] = 1 - xs[i]
        row += 1
        A[row, i] = 1
        b[row] = ys[i]
        row += 1
        A[row, i] = 1
        b[row] = 1 - ys[i]
        row += 1

    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row, i] = 1
            A[row, j] = 1
            b[row] = d
            row += 1

    # Solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if res.success:
        radii = res.x
        # Clip negative small values
        radii = np.maximum(radii, 0.0)
    else:
        # fallback, zero radii
        radii = np.zeros(n)
    sum_r = np.sum(radii)
    return radii, sum_r

Now hill climbing:

def hill_climb(initial_centers, iterations=500, step_initial=0.1, step_decay=0.995):
    centers = np.copy(initial_centers)
    radii, best_sum = compute_radii(centers)
    best_centers = centers.copy()
    best_radii = radii.copy()
    current_sum = best_sum
    current_centers = centers.copy()
    step = step_initial
    for it in range(iterations):
        # propose random perturb
        new_centers = current_centers.copy()
        # choose random index
        i = np.random.randint(0, new_centers.shape[0])
        delta = (np.random.rand(2) - 0.5) * 2 * step
        new_centers[i] = np.clip(new_centers[i] + delta, 0.0, 1.0)
        # compute radii
        new_radii, new_sum = compute_radii(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
        if new_sum >= current_sum:
            current_centers = new_centers
            current_sum = new_sum
        # decay step
        step *= step_decay
    return best_centers, best_radii, best_sum

To obtain a high‑quality packing we treat the radii as variables of a linear program while the circle centers are fixed.  

For a given set of centres `c_i = (x_i,y_i)` the constraints are linear:

* `0 ≤ r_i ≤ x_i , 0 ≤ r_i ≤ 1‑x_i , 0 ≤ r_i ≤ y_i , 0 ≤ r_i ≤ 1‑y_i` – circles stay inside the unit square.  
* `r_i + r_j ≤ ‖c_i‑c_j‖` – circles do not overlap.

The objective is to **maximise** `∑ r_i`.  

We start from a constructive layout that already contains a few large circles in the corners (they can grow up to the distance to the sides) and a dense hexagonal lattice for the remaining circles.  This layout gives a good initial sum of radii (well above the target).  

To improve it further we perform a small hill‑climbing optimisation:

3. Resolve the LP; if the total sum of radii improves we keep the move, otherwise we discard it.  
4. Repeat the perturb‑and‑solve loop for a few hundred iterations, gradually decreasing the step size.

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear‑programming solver for a fixed set of centres
# ------------------------------------------------------------
def _solve_radii(centres: np.ndarray):
    """
    Solve the LP:
        maximise   sum r_i
        subject to r_i >= 0,
                   r_i <= x_i,  r_i <= 1-x_i,
                   r_i <= y_i,  r_i <= 1-y_i,
                   r_i + r_j <= distance(i,j)
    Returns radii (numpy array) and the sum of radii.
    """
    n = centres.shape[0]
    # maximise sum r  <=> minimise -sum r
    c = -np.ones(n, dtype=np.float64)

    # boundary constraints
    xs = centres[:, 0]
    ys = centres[:, 1]

    # total number of inequality constraints
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # r_i <= x_i , r_i <= 1-x_i , r_i <= y_i , r_i <= 1-y_i
    for i in range(n):
        A[row, i] = 1.0
        b[row] = xs[i]
        row += 1

        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]
        row += 1

        A[row, i] = 1.0
        b[row] = ys[i]
        row += 1

        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d
            row += 1

    # solve the LP
    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=(0, None),
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # fallback – all radii zero (should never happen with reasonable data)
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = np.maximum(res.x, 0.0)   # guard against tiny negatives
    total = float(radii.sum())
    return radii, total

def _initial_centres():
    """

      * four circles in the corners (they can become large)
      * a hexagonal lattice covering the rest of the square
    The lattice spacing is chosen so that the uniform radius would be
    roughly 0.09‑0.10, which together with the corner circles gives a
    total sum already above the target.
    """

    # four corner circles – placed a little inside the square
    c = 0.20                     # distance from each side

        [
            [c, c],
            [1 - c, c],
            [c, 1 - c],
            [1 - c, 1 - c],
        ]
    )

    # hexagonal lattice (step ≈ 0.20)
    step = 0.20
    dy = step * np.sqrt(3.0) / 2.0

    y = c + step / 2.0

    lattice = np.array(points)

# ------------------------------------------------------------
# Simple hill‑climbing optimiser
# ------------------------------------------------------------
def _optimize(centres: np.ndarray, iters: int = 300, step_start: float = 0.12):
    """
    Small stochastic optimiser that moves one centre at a time,
    re‑solves the LP and keeps the move when the total radius sum grows.
    """
    best_c = centres.copy()
    best_r, best_sum = _solve_radii(best_c)

    cur_c = best_c.copy()
    cur_sum = best_sum
    cur_r = best_r.copy()

    step = step_start
    rng = np.random.default_rng(2)

    for _ in range(iters):
        # pick a random circle and propose a move
        i = rng.integers(0, cur_c.shape[0])
        delta = (rng.random(2) - 0.5) * 2.0 * step
        new_c = cur_c.copy()
        new_c[i] = np.clip(new_c[i] + delta, 0.0, 1.0)

        new_r, new_sum = _solve_radii(new_c)

        # keep the best overall solution
        if new_sum > best_sum:
            best_sum = new_sum
            best_c = new_c.copy()
            best_r = new_r.copy()

        # accept the move if it is not worse
        if new_sum >= cur_sum:
            cur_c = new_c
            cur_r = new_r
            cur_sum = new_sum

        # slowly shrink the step size
        step *= 0.995

    return best_c, best_r, best_sum

def run_packing():

    init_c = _initial_centres()

    best_total = -1.0

    best_radii, best_total = _solve_radii(best_centres)

    best_radii = np.maximum(best_radii, 0.0)
