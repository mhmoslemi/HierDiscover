def compute_radii(centers):
    n = centers.shape[0]
    # compute distances
    dists = np.linalg.norm(centers[:, np.newaxis, :] - centers[np.newaxis, :, :], axis=2)
    # boundary distances
    b = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    # Build constraints
    num_constraints = n + n*(n-1)//2
    A = np.zeros((num_constraints, n))
    ub = np.zeros(num_constraints)
    # boundary constraints
    for i in range(n):
        A[i, i] = 1.0
        ub[i] = b[i]
    # pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            ub[idx] = dists[i, j]
            idx += 1
    # linear program
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback use zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r

def init_positions(num_points, min_dist):
    points = []

    while len(points) < num_points and attempts < max_attempts:

        attempts += 1
    # If we didn't get enough points, fallback to random
    while len(points) < num_points:

assistantfinal**Strategy**

The problem is a non‑convex optimisation: we must choose centres \((x_i,y_i)\) and radii \(r_i\) to maximise \(\sum r_i\) under  

* \(0\le r_i\)  
* \(r_i \le \min(x_i,1-x_i,y_i,1-y_i)\)  (stay inside the square)  
* \(\|c_i-c_j\|\ge r_i+r_j\) for all pairs (no overlap).

For a *fixed* set of centres the radii problem is linear:
\[
\max\;\sum_i r_i\qquad
\text{s.t. } r_i\!+\!r_j\le d_{ij},\;
r_i\le b_i,\; r_i\ge0,
\]
where \(d_{ij}\) is the Euclidean distance of the two centres and
\(b_i\) is the distance of centre \(i\) to the closest side of the unit square.
Thus, for any placement of the points we can obtain the optimal radii by a
linear programme (LP).

The quality of a packing is therefore driven by the *positions* of the
points.  We use a simple stochastic optimiser that repeatedly perturbs the
positions, solves the LP for the radii and keeps the best solution found.
The optimiser works as follows:

1. **Initial layout** – points are generated with a Poisson‑disc style
   rejection sampler to guarantee a moderate minimum separation.
2. **LP solution** – for the current set of centres the LP (via
   `scipy.optimize.linprog`) yields the maximal radii.
3. **Hill‑climbing / simulated‑annealing** – one centre is moved
   by a small random offset; the LP is solved again.  
   The move is accepted if the total radius sum improves,
   otherwise it may be accepted with a temperature‑controlled probability.
4. The process is repeated for a fixed number of iterations
   (a few thousand – fast enough for the test environment).  
   The best configuration encountered is returned.

Because the LP always yields a feasible solution (all radii can be set to 0),
the algorithm never produces an invalid packing.  The final
\((\text{centres},\text{radii})\) pair is guaranteed to satisfy the
validation routine, and the sum of radii is the true maximal sum for the
chosen centres, so the optimiser strives toward the global optimum.

Below is a self‑contained implementation of `run_packing` that follows this
plan.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: solve the linear programme for a *fixed* set of centres
# ------------------------------------------------------------
def _solve_radii(centres: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Given centre coordinates (n,2), compute the maximal radii that satisfy
    the non‑overlap and boundary constraints.
    Returns (radii, sum_of_radii).
    """
    n = centres.shape[0]
    # distances between centres
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # distance to the closest square side -> upper bound for each radius
    bound = np.minimum.reduce([centres[:, 0], 1 - centres[:, 0],
                               centres[:, 1], 1 - centres[:, 1]])

    # Build inequality matrix A_ub * r <= b_ub
    #  - one row per boundary bound  (r_i <= bound_i)
    #  - one row per pair i<j      (r_i + r_j <= d_ij)
    n_pairs = n * (n - 1) // 2
    n_constraints = n + n_pairs
    A = np.zeros((n_constraints, n), dtype=np.float64)
    b = np.zeros(n_constraints, dtype=np.float64)

    # boundary constraints
    for i in range(n):
        A[i, i] = 1.0
        b[i] = bound[i]

    # pairwise non‑overlap constraints
    row = n
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # Linear programme: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'integrality': False})

    if not res.success:
        # a feasible (zero‑radius) solution always exists; fall back to it
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = np.maximum(res.x, 0.0)   # guard against tiny negatives

    total = float(radii.sum())
    return radii, total


# ------------------------------------------------------------
# Helper: generate an initial set of points with a minimum distance
# ------------------------------------------------------------
def _initial_points(n: int, min_dist: float = 0.15, seed: int = 0) -> np.ndarray:
    rng = np.random.default_rng(seed)
    pts = []
    attempts = 0
    max_attempts = n * 5000
    while len(pts) < n and attempts < max_attempts:
        cand = rng.random(2)
        if all(np.linalg.norm(cand - np.array(p)) >= min_dist for p in pts):
            pts.append(cand)
        attempts += 1
    # If we failed to place enough points (very unlikely), fall back to pure random
    while len(pts) < n:
        pts.append(rng.random(2))
    return np.array(pts, dtype=np.float64)


# ------------------------------------------------------------
# Main routine required by the problem statement
# ------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a packing of 26 circles in the unit square.
    The tuple consists of:
        centres : (26,2) array of (x,y) coordinates,
        radii   : (26,)   array of radii,
        sum_r   : float    sum of all radii.
    The packing satisfies all constraints checked by `validate_packing`.
    """
    rng = np.random.default_rng(42)          # deterministic randomness
    n_circles = 26

    # ---- 1. start from a reasonably spread configuration ----
    centres = _initial_points(n_circles, min_dist=0.18, seed=1)
    radii, best_sum = _solve_radii(centres)
    best_centres = centres.copy()
    best_radii = radii.copy()

    # ---- 2. stochastic local optimisation (simulated annealing) ----
    max_iters = 3000
    sigma_initial = 0.15   # initial step size for centre moves
    temp_initial = 0.02    # initial acceptance temperature

    sigma = sigma_initial
    temp = temp_initial

    for it in range(max_iters):
        # linear cooling schedule
        frac = it / max_iters
        sigma = sigma_initial * (1.0 - frac)
        temp = temp_initial * (1.0 - frac)

        # propose a move for one randomly chosen centre
        i = rng.integers(0, n_circles)
        proposal = best_centres[i] + (rng.random(2) - 0.5) * 2.0 * sigma
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centres = best_centres.copy()
        new_centres[i] = proposal

        new_radii, new_sum = _solve_radii(new_centres)

        # accept if better, otherwise maybe accept probabilistically
        accept = False
        if new_sum > best_sum:
            accept = True
        else:
            # Metropolis criterion
            if temp > 0 and rng.random() < np.exp((new_sum - best_sum) / temp):
                accept = True

        if accept:
            best_centres = new_centres
            best_radii = new_radii
            best_sum = new_sum

        # optional early stop if we already exceed a reasonable target
        if best_sum > 2.65:
            break

    # Ensure the output shapes are exactly as required
    best_centres = best_centres.astype(np.float64)
    best_radii = best_radii.astype(np.float64)
    best_sum = float(best_sum)

    return best_centres, best_radii, best_sum
```
