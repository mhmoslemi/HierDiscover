def compute_optimal_radii(centers):
    n = len(centers)
    c = -np.ones(n)  # maximize sum
    A_ub = []
    b_ub = []
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # r_i <= x_i
        a = np.zeros(n); a[i] = 1; A_ub.append(a); b_ub.append(x)
        # r_i <= 1 - x_i
        a = np.zeros(n); a[i] = 1; A_ub.append(a); b_ub.append(1 - x)
        # r_i <= y_i
        a = np.zeros(n); a[i] = 1; A_ub.append(a); b_ub.append(y)
        # r_i <= 1 - y_i
        a = np.zeros(n); a[i] = 1; A_ub.append(a); b_ub.append(1 - y)
    # overlap constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            a = np.zeros(n); a[i] = 1; a[j] = 1
            A_ub.append(a); b_ub.append(d)
    bounds = [(0, None) for _ in range(n)]
    result = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        raise RuntimeError('LP failed')
    radii = result.x
    return radii

assistantfinal**Explanation of the approach**

1. **Linear programming for the radii**  
   For a *fixed* set of circle centres the optimal radii are obtained by a very small linear program:  

   ```
   maximise   Σ r_i
   subject to r_i ≤ x_i , r_i ≤ 1‑x_i , r_i ≤ y_i , r_i ≤ 1‑y_i          (inside the square)
               r_i + r_j ≤ distance(centre_i , centre_j)                (no overlap)
               r_i ≥ 0
   ```

   `scipy.optimize.linprog` (HiGHS) solves this LP instantly for 26 circles.

2. **Optimising the centres**  
   The positions are the only non‑linear part of the problem.  
   * A deterministic Latin‑Hypercube design gives a good, almost uniform start.
   * A simple simulated‑annealing loop repeatedly perturbs a randomly chosen centre,
     recomputes the optimal radii by the LP and keeps the change if the total
     radius sum improves (or with a small probability when the temperature is
     still high – this helps escaping local minima).

   The loop is deterministic because the NumPy RNG is seeded (`seed=0`).  
   It stops as soon as a packing with a sum of radii ≥ 2.636 is found
   (the target value) or after a fixed number of iterations.

3. **Result**  
   The best centre set found during the search is returned together with its
   radii and the exact sum of the radii.  By construction the pair satisfies all
   constraints of `validate_packing`.

```python
import numpy as np
from scipy.optimize import linprog
from scipy.stats import qmc

# ------------------------------------------------------------
# Helper: solve the LP that gives the maximal radii for given centres
# ------------------------------------------------------------
def _optimal_radii(centres: np.ndarray) -> np.ndarray:
    """
    LinProg: maximise sum(r_i) subject to
       * r_i ≤ x_i, 1‑x_i, y_i, 1‑y_i
       * r_i + r_j ≤ dist(i,j)
       * r_i ≥ 0
    Returns the optimal radii vector.
    """
    n = centres.shape[0]
    # objective: minimise -sum(r)  (linprog minimises)
    c = -np.ones(n)

    # inequality matrix A_ub * r ≤ b_ub
    rows = []
    rhs = []

    # boundary constraints
    for i in range(n):
        x, y = centres[i]
        e = np.zeros(n)
        e[i] = 1.0
        # r_i ≤ x_i
        rows.append(e); rhs.append(x)
        # r_i ≤ 1 - x_i
        rows.append(e); rhs.append(1.0 - x)
        # r_i ≤ y_i
        rows.append(e); rhs.append(y)
        # r_i ≤ 1 - y_i
        rows.append(e); rhs.append(1.0 - y)

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # bounds for each variable: r_i ≥ 0, no explicit upper bound (handled by A_ub)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # In the unlikely event the LP fails, fall back to zero radii
        return np.zeros(n, dtype=np.float64)
    # tiny negative values may appear due to numerical noise – clip them
    return np.clip(res.x, 0.0, None)


# ------------------------------------------------------------
# Main routine: simulated‑annealing on the centres
# ------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, sum_of_radii) for 26 circles inside the unit square.
    The packing is obtained by a deterministic simulated‑annealing search that
    maximises the sum of the radii.
    """
    rng = np.random.default_rng(0)          # deterministic randomness
    n_circles = 26
    target_sum = 2.636                       # stop as soon as we reach this

    # ---- initial centres: Latin‑Hypercube (uniformly spread) ----
    sampler = qmc.LatinHypercube(d=2, seed=0)
    centres = sampler.random(n_circles)     # points in [0,1]²

    # compute the first radii and its total length
    radii = _optimal_radii(centres)
    best_sum = float(np.sum(radii))
    best_centres = centres.copy()
    best_radii = radii.copy()

    # ---- simulated annealing parameters ----
    temperature = 0.5            # initial step magnitude
    cooling = 0.9995            # exponential cooling schedule
    max_iters = 12000           # enough iterations for a deterministic run

    current_centres = centres.copy()
    current_sum = best_sum

    for it in range(max_iters):
        # gradually shrink the proposal step with the temperature
        step = temperature * 0.05

        # pick a random circle and move its centre
        i = rng.integers(n_circles)
        trial = current_centres.copy()
        trial[i] += rng.normal(scale=step, size=2)
        trial[i] = np.clip(trial[i], 0.0, 1.0)   # stay inside the square

        # evaluate the new configuration
        trial_radii = _optimal_radii(trial)
        trial_sum = float(np.sum(trial_radii))

        # acceptance test (Metropolis)
        accept = False
        if trial_sum > current_sum:
            accept = True
        else:
            # occasional uphill moves while temperature is still high
            prob = np.exp((trial_sum - current_sum) / temperature)
            if rng.random() < prob:
                accept = True

        if accept:
            current_centres = trial
            current_sum = trial_sum
            if trial_sum > best_sum:
                best_sum = trial_sum
                best_centres = trial.copy()
                best_radii = trial_radii.copy()
                # early exit if we already beat the required target
                if best_sum >= target_sum:
                    break

        # cool down
        temperature *= cooling

    # Final radii for the best geometry (re‑evaluate to avoid numerical drift)
    best_radii = _optimal_radii(best_centres)
    best_sum = float(np.sum(best_radii))

    return best_centres, best_radii, best_sum
```
