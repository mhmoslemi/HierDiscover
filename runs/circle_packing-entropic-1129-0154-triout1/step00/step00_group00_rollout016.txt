def optimal_radii(centers):
    n = centers.shape[0]
    # compute distances between centers
    d = scipy.spatial.distance.cdist(centers, centers) # shape (n,n)
    # pair constraints: for i < j, r_i + r_j <= d[i,j]
    num_pairs = n*(n-1)//2
    A = np.zeros((num_pairs, n))
    b = np.zeros(num_pairs)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1
            A[idx, j] = 1
            b[idx] = d[i, j]
            idx += 1
    # lower bound 0, upper bound compute from boundary distances
    # compute max radius per center due to edges
    x = centers[:,0]
    y = centers[:,1]
    ub = np.minimum.reduce([x, 1-x, y, 1-y]) # shape (n,)
    bounds = [(0, ub_i) for ub_i in ub]
    # maximize sum r_i -> minimize -sum r_i
    c = -np.ones(n)
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # Could fallback to small radii or raise
        raise RuntimeError("LP failed")
    r = res.x
    return r

import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    n = centers.shape[0]
    # Distances between centers
    dists = cdist(centers, centers)
    # Build constraints for pairs
    m = n*(n-1)//2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
    # Upper bounds from boundaries
    x = centers[:, 0]
    y = centers[:, 1]
    ub = np.minimum.reduce([x, 1.0 - x, y, 1.0 - y])
    bounds = [(0.0, ub_i) for ub_i in ub]
    # Objective: maximize sum r -> minimize -sum r
    c = -np.ones(n, dtype=np.float64)
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # If failed, fallback to zeros (should not happen)
        return np.zeros(n)
    return res.x

def sum_of_radii_for_centers(centers):
    radii = compute_optimal_radii(centers)
    return radii.sum(), radii

def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    rng = np.random.default_rng(42)  # fixed seed
    n = 26
    # global best
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # Precompute a grid (5x5) as one possible start
    grid = np.linspace(0.1, 0.9, 5)
    grid_centers = np.array([[x, y] for x in grid for y in grid])
    # Ensure we have exactly 25 points
    assert grid_centers.shape[0] == 25

    # We'll run several independent restarts
    num_restarts = 30
    iterations_per_restart = 400
    # annealing schedule
    initial_step = 0.2
    final_step = 0.001

    for restart in range(num_restarts):
        # start from random perturbation of grid or pure random
        if restart % 3 == 0:
            # start from grid + small jitter
            centers = grid_centers.copy()
            # add jitter to all points
            jitter = rng.uniform(-0.03, 0.03, size=centers.shape)
            centers += jitter
            # ensure inside [0,1]
            centers = np.clip(centers, 0.0, 1.0)
            # add 26th circle at random location
            extra = rng.uniform(0.0, 1.0, size=(1,2))
            centers = np.vstack([centers, extra])
        else:
            # fully random
            centers = rng.uniform(0.0, 1.0, size=(n,2))

        # compute initial radii and sum
        radii = compute_optimal_radii(centers)
        cur_sum = radii.sum()
        cur_centers = centers.copy()
        cur_radii = radii.copy()

        # maintain best for this restart
        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = cur_centers.copy()
            best_radii = cur_radii.copy()

        for it in range(iterations_per_restart):
            # linear annealing for step size
            t = it / iterations_per_restart
            step = initial_step * (1 - t) + final_step * t
            # propose new centers by moving a random subset
            new_centers = cur_centers.copy()
            # number of circles to move, maybe 3
            k = rng.integers(1, 4)  # move 1 to 3 circles
            idxs = rng.choice(n, size=k, replace=False)
            # Gaussian perturbation
            moves = rng.normal(scale=step, size=(k,2))
            new_centers[idxs] += moves
            # clamp
            new_centers = np.clip(new_centers, 0.0, 1.0)

            new_radii = compute_optimal_radii(new_centers)
            new_sum = new_radii.sum()

            # Accept if better, else with probability exp((new_sum - cur_sum)/temp)
            # Use temperature decaying from 0.1 to 0
            temp = 0.05 * (1 - t) + 1e-6
            if new_sum > cur_sum or rng.random() < np.exp((new_sum - cur_sum) / temp):
                cur_centers = new_centers
                cur_radii = new_radii
                cur_sum = new_sum
                # update global best
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centers = cur_centers.copy()
                    best_radii = cur_radii.copy()
    # End restarts

    # Ensure final solution meets target; if not, we can attempt a final improvement phase.
    # Run a few extra iterations with best as start
    for extra_it in range(2000):
        step = 0.005 * (1 - extra_it / 2000) + 0.0005
        new_centers = best_centers.copy()
        # move a few circles
        k = 3
        idxs = rng.choice(n, size=k, replace=False)
        moves = rng.normal(scale=step, size=(k,2))
        new_centers[idxs] += moves
        new_centers = np.clip(new_centers, 0.0, 1.0)
        new_radii = compute_optimal_radii(new_centers)
        new_sum = new_radii.sum()
        # simple hill climbing: accept only if improvement
        if new_sum > best_sum:
            best_sum = new_sum
            best_centers = new_centers
            best_radii = new_radii

    # Return the best solution found
    return best_centers, best_radii, float(best_sum)

def compute_optimal_radii(centers):
    n=centers.shape[0]
    d = cdist(centers, centers)
    # pair indices
    iu = np.triu_indices(n, k=1)
    row_count = len(iu[0])
    A = np.zeros((row_count, n))
    A[np.arange(row_count), iu[0]] = 1
    A[np.arange(row_count), iu[1]] = 1
    b = d[iu]
    # bounds from boundary
    x = centers[:,0]
    y = centers[:,1]
    ub = np.minimum.reduce([x, 1-x, y, 1-y])
    bounds = [(0.0, ub_i) for ub_i in ub]
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: maybe try different method
        raise RuntimeError("LP failed")
    return res.x

    cur_cent = initial_centers.copy()
    cur_radii = compute_optimal_radii(cur_cent)
    cur_sum = cur_radii.sum()

    best_radii = cur_radii.copy()
    best_sum = cur_sum
    for it in range(iterations):
        t = it / iterations
        step = step_start * (1 - t) + step_end * t
        # propose new centers
        new_cent = cur_cent.copy()
        # move a random subset of circles
        k = rng.integers(1, 5)  # 1 to 4 circles (inclusive)
        idx = rng.choice(len(cur_cent), size=k, replace=False)
        new_cent[idx] += rng.normal(scale=step, size=(k,2))
        # keep within bounds
        np.clip(new_cent, 0.0, 1.0, out=new_cent)
        new_radii = compute_optimal_radii(new_cent)
        new_sum = new_radii.sum()
        # accept if improves
        if new_sum > cur_sum:
            cur_cent = new_cent
            cur_radii = new_radii
            cur_sum = new_sum

import numpy as np
from scipy.optimize import linprog
from scipy.spatial.distance import cdist

# ----------------------------------------------------------------------
# Linear‑program to obtain the maximal radii for a fixed set of centres
# ----------------------------------------------------------------------
def optimal_radii(centres: np.ndarray) -> np.ndarray:
    """Return the radii that maximise the sum of radii for the given centres."""
    n = centres.shape[0]
    # pairwise Euclidean distances
    d = cdist(centres, centres)

    # Build pairwise constraints r_i + r_j <= d_ij  (i < j)
    iu = np.triu_indices(n, k=1)
    pair_cnt = iu[0].size
    A = np.zeros((pair_cnt, n), dtype=np.float64)
    A[np.arange(pair_cnt), iu[0]] = 1.0
    A[np.arange(pair_cnt), iu[1]] = 1.0
    b = d[iu]

    # Upper bounds from the four sides of the unit square
    x = centres[:, 0]
    y = centres[:, 1]
    ub = np.minimum.reduce([x, 1.0 - x, y, 1.0 - y])

    bounds = [(0.0, float(ub_i)) for ub_i in ub]

    # Maximise sum(r_i)  <=> minimise -sum(r_i)
    c = -np.ones(n, dtype=np.float64)

    result = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not result.success:
        # Fall back to a zero solution – it is always feasible.
        return np.zeros(n, dtype=np.float64)
    return result.x

# Helper: generate a simple hexagonal lattice (used as one of the starts)

def hex_lattice_guess(r_guess: float, need: int, rng: np.random.Generator) -> np.ndarray:
    """Create points on a hexagonal lattice with spacing based on r_guess."""
    dx = 2.0 * r_guess
    dy = np.sqrt(3.0) * r_guess
    pts = []
    row = 0
    y = r_guess
    while y <= 1.0 - r_guess + 1e-12:
        offset = (row % 2) * r_guess
        x = r_guess + offset
        while x <= 1.0 - r_guess + 1e-12:
            pts.append([x, y])
            x += dx
        y += dy
        row += 1
    pts = np.asarray(pts, dtype=np.float64)
    if pts.shape[0] >= need:
        idx = rng.choice(pts.shape[0], size=need, replace=False)
        return pts[idx]
    else:
        extra = need - pts.shape[0]
        extra_pts = rng.uniform(0.0, 1.0, size=(extra, 2))
        return np.vstack([pts, extra_pts])


# ----------------------------------------------------------------------
# Core hill‑climbing / simulated‑annealing optimiser on the centres
# ----------------------------------------------------------------------
def hill_climb(
    start_centres: np.ndarray,
    rng: np.random.Generator,
    iters: int,
    step_start: float,
    step_end: float,
) -> tuple[np.ndarray, np.ndarray, float]:
    """Perform a simple stochastic hill climb starting from start_centres."""
    centres = start_centres.copy()
    radii = optimal_radii(centres)
    cur_sum = radii.sum()
    best_cent = centres.copy()
    best_rad = radii.copy()
    best_sum = cur_sum
    n = centres.shape[0]

    for it in range(iters):
        # linear annealing of the step size
        t = it / iters
        step = step_start * (1.0 - t) + step_end * t

        # propose a move: perturb a few random circles
        proposal = centres.copy()
        k = rng.integers(1, 5)  # move between 1 and 4 circles
        idxs = rng.choice(n, size=k, replace=False)
        proposal[idxs] += rng.normal(scale=step, size=(k, 2))
        np.clip(proposal, 0.0, 1.0, out=proposal)

        prop_radii = optimal_radii(proposal)
        prop_sum = prop_radii.sum()

        # Accept only improvements (with a tiny probability for worsening moves)
        if prop_sum > cur_sum:
            centres, radii, cur_sum = proposal, prop_radii, prop_sum
            if prop_sum > best_sum:
                best_sum = prop_sum
                best_cent = proposal.copy()
                best_rad = prop_radii.copy()
        else:
            # Metropolis criterion with a very low temperature to allow occasional escapes
            temp = 0.01 * (1.0 - t) + 1e-6
            if rng.random() < np.exp((prop_sum - cur_sum) / temp):
                centres, radii, cur_sum = proposal, prop_radii, prop_sum

    return best_cent, best_rad, best_sum
