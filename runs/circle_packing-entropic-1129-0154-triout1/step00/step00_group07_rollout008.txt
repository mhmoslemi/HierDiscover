import numpy as np
from scipy.optimize import linprog

def max_radii(centers):
    N=centers.shape[0]
    # compute distances matrix d
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    d = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    # Number of constraints = 4N + N(N-1)/2
    M=4*N + N*(N-1)//2
    A = np.zeros((M, N))
    b = np.zeros(M)
    row=0
    # wall constraints
    for i in range(N):
        A[row,i] = 1
        b[row] = centers[i,0]  # x_i
        row+=1
        A[row,i] = 1
        b[row] = 1 - centers[i,0]  # 1-x_i
        row+=1
        A[row,i] = 1
        b[row] = centers[i,1]  # y_i
        row+=1
        A[row,i] = 1
        b[row] = 1 - centers[i,1]  # 1-y_i
        row+=1
    # pairwise constraints
    for i in range(N):
        for j in range(i+1,N):
            A[row,i] = 1
            A[row,j] = 1
            b[row] = d[i,j]
            row+=1
    # objective
    c = -np.ones(N)
    # bounds
    bounds = [(0, None)]*N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback to zero radii
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

def sum_radii_given(centers):
    radii = max_radii(centers)
    return radii.sum(), radii

def improve_centers(initial_centers, iterations=5000, step=0.05, seed=0):
    np.random.seed(seed)
    centers = initial_centers.copy()
    best_sum, best_radii = sum_radii_given(centers)
    best_centers = centers.copy()
    cur_sum = best_sum
    cur_centers = centers.copy()
    cur_radii = best_radii.copy()
    T_initial = 0.1
    T_final = 0.001
    for it in range(iterations):
        # temperature schedule
        t = T_initial * ((T_final/T_initial) ** (it/iterations))
        # pick random circle
        i = np.random.randint(0, len(centers))
        # propose random move
        new_center = cur_centers[i] + np.random.uniform(-step, step, size=2)
        # clamp to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # try to ensure enough margin for radius: we can keep as is.
        # create new centers
        new_centers = cur_centers.copy()
        new_centers[i] = new_center
        new_sum, new_radii = sum_radii_given(new_centers)
        delta = new_sum - cur_sum
        if delta > 0 or np.exp(delta / t) > np.random.rand():
            # accept move
            cur_centers = new_centers
            cur_sum = new_sum
            cur_radii = new_radii
            # update best
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_radii = cur_radii.copy()
                best_centers = cur_centers.copy()
    return best_centers, best_radii, best_sum

    cur = initial.copy()
    cur_sum, cur_radii = sum_radii_given(cur)

    best_centers=cur.copy()
    best_radii=cur_radii.copy()

        T = T0 * (Tf/T0)**(k/iters)

        cand_sum, cand_radii = sum_radii_given(candidate)
        delta = cand_sum - cur_sum
        if delta > 0 or np.exp(delta / T) > np.random.rand():
            cur = candidate
            cur_sum = cand_sum
            cur_radii = cand_radii

def compute_radii(centers):
    N = centers.shape[0]
    # distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    M = 4*N + N*(N-1)//2
    A = np.zeros((M, N))
    b = np.zeros(M)
    row=0
    # left, right, bottom, top constraints
    for i in range(N):
        A[row, i] = 1
        b[row] = centers[i,0]  # x_i
        row+=1
        A[row, i] = 1
        b[row] = 1 - centers[i,0]  # 1 - x_i
        row+=1
        A[row, i] = 1
        b[row] = centers[i,1]  # y_i
        row+=1
        A[row, i] = 1
        b[row] = 1 - centers[i,1]  # 1 - y_i
        row+=1
    # pairwise
    for i in range(N):
        for j in range(i+1,N):
            A[row,i] = 1
            A[row,j] = 1
            b[row] = dists[i,j]
            row+=1
    # objective
    c = -np.ones(N)
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(N)
    else:
        radii = res.x
    return radii

def anneal(initial_centers, iterations=5000, seed=0):
    rng = np.random.default_rng(seed)
    N = initial_centers.shape[0]
    cur_centers = initial_centers.copy()
    cur_sum, cur_radii = evaluate(cur_centers)
    best_sum = cur_sum
    best_centers = cur_centers.copy()
    best_radii = cur_radii.copy()
    # annealing schedule
    T_start = 0.1
    T_end = 1e-4
    for k in range(iterations):
        T = T_start * (T_end/T_start)**(k/iterations)
        # decide move type
        if rng.random() < 0.1:
            # global jitter
            proposal = cur_centers + rng.normal(scale=0.03, size=(N,2))
        else:
            # single move
            i = rng.integers(N)
            proposal = cur_centers.copy()
            proposal[i] += rng.uniform(-0.05, 0.05, size=2)
        # clip to [0,1]
        proposal = np.clip(proposal, 0.0, 1.0)
        cand_sum, cand_radii = evaluate(proposal)
        delta = cand_sum - cur_sum
        if delta > 0 or rng.random() < np.exp(delta / T) if T>0 else False:
            cur_centers = proposal
            cur_sum = cand_sum
            cur_radii = cand_radii

                best_sum = cur_sum
                best_centers = cur_centers.copy()
                best_radii = cur_radii.copy()
    return best_centers, best_radii, best_sum

import numpy as np
from scipy.optimize import linprog

def _optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear program that, for a given set of centre points,
    yields the maximum possible radii (sum of radii maximised).
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of inequality constraints: 4 per circle (walls) + one per pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0

    # wall constraints
    for i in range(n):
        # left side   : r_i <= x_i
        A[row, i] = 1.0
        b[row] = centers[i, 0]
        row += 1
        # right side  : r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 0]
        row += 1
        # bottom side : r_i <= y_i
        A[row, i] = 1.0
        b[row] = centers[i, 1]
        row += 1
        # top side    : r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 1]
        row += 1

    # non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  <=> minimise -Σ r_i
    c = -np.ones(n)
    bounds = [(0.0, None) for _ in range(n)]

    # use the HiGHS interior‑point/simplex solver (fast and reliable)
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # In the extremely unlikely case of failure, fall back to zeros.
        return np.zeros(n, dtype=float)
    return result.x


def _evaluate(centers: np.ndarray) -> Tuple[float, np.ndarray]:
    """Return total radius and the optimal radii for the given centres."""
    radii = _optimal_radii(centers)
    return float(radii.sum()), radii

    """Simple simulated‑annealing optimiser on the centre positions."""
    rng = np.random.default_rng(seed)
    n = initial.shape[0]

    cur_centers = initial.copy()
    cur_sum, cur_radii = _evaluate(cur_centers)

    best_sum = cur_sum
    best_centers = cur_centers.copy()
    best_radii = cur_radii.copy()

    # exponential temperature schedule
    T_start = 0.10
    T_end = 1e-4

    for it in range(iterations):
        # temperature decreasing from T_start to T_end
        T = T_start * (T_end / T_start) ** (it / iterations)

        # 10 % of the time propose a global jitter,
        # otherwise move a single circle.
        if rng.random() < 0.10:
            proposal = cur_centers + rng.normal(scale=0.03, size=(n, 2))
        else:
            i = rng.integers(n)
            proposal = cur_centers.copy()
            proposal[i] += rng.uniform(-0.05, 0.05, size=2)

        # keep points inside the unit square
        proposal = np.clip(proposal, 0.0, 1.0)

        cand_sum, cand_radii = _evaluate(proposal)
        delta = cand_sum - cur_sum

        # Metropolis acceptance
        if delta > 0.0 or rng.random() < np.exp(delta / T):
            cur_centers = proposal
            cur_sum = cand_sum
            cur_radii = cand_radii
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_centers = cur_centers.copy()
                best_radii = cur_radii.copy()

    return best_centers, best_radii, best_sum


def _polish(centers: np.ndarray,
            radii: np.ndarray,
            eps: float = 0.004,
            max_passes: int = 5) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Small deterministic hill‑climbing: try to nudge each circle a little
    in the four cardinal directions and keep any improvement.
    """
    n = centers.shape[0]
    cur_c = centers.copy()
    cur_r = radii.copy()
    cur_sum = cur_r.sum()

    for _ in range(max_passes):
        improved = False
        for i in range(n):
            for dx, dy in ((eps, 0), (-eps, 0), (0, eps), (0, -eps)):
                new_c = cur_c[i].copy()
                new_c[0] = np.clip(new_c[0] + dx, 0.0, 1.0)
                new_c[1] = np.clip(new_c[1] + dy, 0.0, 1.0)
                cand = cur_c.copy()
                cand[i] = new_c
                cand_sum, cand_r = _evaluate(cand)
                if cand_sum > cur_sum + 1e-9:
                    cur_c = cand
                    cur_r = cand_r
                    cur_sum = cand_sum
                    improved = True
                    break
            if improved:
                break
        if not improved:
            break
    return cur_c, cur_r, cur_sum


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Produce 26 circles inside the unit square with a large total radius.
    Returns (centres, radii, total_radius).
    """
    N = 26

    # ---- 1. deterministic starting layout (5×5 grid + one extra point) ----
    grid = np.array([0.1, 0.3, 0.5, 0.7, 0.9])
    xv, yv = np.meshgrid(grid, grid)
    base = np.column_stack([xv.ravel(), yv.ravel()])            # 25 points
    extra = np.array([[0.5, 0.62]])                             # not on the grid
    start = np.vstack([base, extra])                            # now 26 points

    # a tiny random jitter so the optimiser does not start from a perfectly
    # symmetric configuration (helps escaping degenerate plateaus)
    rng = np.random.default_rng(2023)
    start += (rng.random((N, 2)) - 0.5) * 0.02
    start = np.clip(start, 0.02, 0.98)

    # ---- 2. several annealing runs, keep the best result ----
    best_total = -1.0
    best_centers = None
    best_radii = None

    for restart in range(3):
        # each restart gets a fresh random perturbation of the initial layout
        pert = start + (rng.random((N, 2)) - 0.5) * 0.05
        pert = np.clip(pert, 0.02, 0.98)

        centres, radii, total = _anneal(pert, iterations=8000, seed=restart)
        if total > best_total:
            best_total = total
            best_centers = centres
            best_radii = radii

    # ---- 3. final deterministic polishing ----
    best_centers, best_radii, best_total = _polish(best_centers, best_radii)

    # ---- 4. tiny safety shrink so the validator’s 1e‑12 margin is respected ----
    shrink = 1.0 - 1e-7
    best_radii *= shrink
    best_total = float(best_radii.sum())

    return best_centers, best_radii, best_total
