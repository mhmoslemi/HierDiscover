```
    n = centers.shape[0]
    # Border limits
    limits = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Pairwise distances
    # compute all pairs i<j
    idx_i, idx_j = np.triu_indices(n, k=1)  # but we need lower triangular? Actually upper.
    # distances:
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs**2, axis=1))
    # Construct A_ub and b_ub
    m = n + len(dists)
    A_ub = np.zeros((m, n))
    b_ub = np.zeros(m)
    # border constraints rows 0..n-1
    for i in range(n):
        A_ub[i, i] = 1.0
        b_ub[i] = limits[i]
    # pairwise constraints rows n to ...
    for k in range(len(dists)):
        i = idx_i[k]; j = idx_j[k]
        A_ub[n + k, i] = 1.0
        A_ub[n + k, j] = 1.0
        b_ub[n + k] = dists[k]
    c = -np.ones(n)
    bounds = [(0, None)] * n
    result = opt.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not result.success:
        # If fails, fallback to zero radii
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = result.x
        sum_r = -result.fun
    return radii, sum_r
```

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: given fixed centre positions, compute the optimal radii
# that maximise the sum of radii while respecting the non‑overlap
# and border constraints.
# ------------------------------------------------------------
def compute_optimal_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Solve the linear programme for fixed circle centres.

    Parameters
    ----------
    centers : np.ndarray, shape (n,2)
        The (x,y) coordinates of the circle centres.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        Radii that maximise the sum while satisfying all constraints.
    total : float
        Sum of the radii.
    """
    n = centers.shape[0]

    # ---------- border limits ----------
    # r_i must be ≤ distance to each side; the smallest of those is enough.
    border_limit = np.minimum.reduce(
        [
            centers[:, 0],            # distance to left side
            1.0 - centers[:, 0],      # distance to right side
            centers[:, 1],            # distance to bottom side
            1.0 - centers[:, 1],      # distance to top side
        ]
    )

    # ---------- pairwise distance constraints ----------
    # indices of the upper‑triangular part (i<j)
    idx_i, idx_j = np.triu_indices(n, k=1)
    # Euclidean distances between all distinct pairs
    diffs = centers[idx_i] - centers[idx_j]
    dists = np.sqrt(np.sum(diffs * diffs, axis=1))

    # ---------- build linear programme ----------
    # Variables: r_0 … r_{n‑1}
    # Objective: maximise Σ r_i   → minimise −Σ r_i
    c = -np.ones(n)

    # Inequality matrix A_ub * r ≤ b_ub
    #   * border constraints   :   r_i ≤ border_limit_i
    #   * pairwise constraints:   r_i + r_j ≤ d_ij
    m_border = n
    m_pair = len(dists)
    m_total = m_border + m_pair

    A_ub = np.zeros((m_total, n))
    b_ub = np.zeros(m_total)

    # border rows
    A_ub[np.arange(m_border), np.arange(m_border)] = 1.0
    b_ub[:m_border] = border_limit

    # pairwise rows
    A_ub[m_border:, idx_i] = 1.0
    A_ub[m_border:, idx_j] = 1.0
    b_ub[m_border:] = dists

    # variable bounds: radii ≥ 0, no explicit upper bound
    bounds = [(0.0, None) for _ in range(n)]

    # solve
    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True, "dual_feasibility_tolerance": 1e-12},
    )

    if not res.success:
        # In the highly unlikely event of failure, fall back to zeros.
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = -float(res.fun)  # because we minimised the negative sum

    return radii, total

    jitter = (rng.random(grid.shape) - 0.5) * 0.04
    grid = np.clip(grid + jitter, 0.0, 1.0)

# ------------------------------------------------------------
# Hill‑climbing optimiser for the centre positions.
# Only moves a single centre at a time; accepts the move if it
# increases the total sum of radii.  The step size shrinks
# gradually, allowing a fine‑grained final search.
# ------------------------------------------------------------
def hill_climb(
    start_centres: np.ndarray,
    max_iter: int = 3000,
    initial_step: float = 0.12,
    step_decay: float = 0.9995,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a simple hill‑climbing optimisation of the centre positions.

    Parameters
    ----------
    start_centres : np.ndarray, shape (26,2)
        Initial centre coordinates.
    max_iter : int
        Number of attempted moves.
    initial_step : float
        Maximum absolute displacement for a proposed move at iteration 0.
    step_decay : float
        Multiplicative factor applied to the step size after each iteration.

    Returns
    -------
    best_centres : np.ndarray, shape (26,2)
        The best centre set found.
    best_radii : np.ndarray, shape (26,)
        Radii corresponding to the best centre set.
    best_sum : float
        Sum of those radii.
    """
    rng = np.random.default_rng()
    centres = start_centres.copy()
    radii, best_sum = compute_optimal_radii(centres)
    best_centres = centres.copy()
    best_radii = radii.copy()

    step = initial_step

    for _ in range(max_iter):
        # pick a random circle to move
        i = rng.integers(0, centres.shape[0])

        # propose a new position
        delta = (rng.random(2) - 0.5) * 2.0 * step
        new_pos = np.clip(centres[i] + delta, 0.0, 1.0)

        # evaluate the new configuration
        trial = centres.copy()
        trial[i] = new_pos
        trial_radii, trial_sum = compute_optimal_radii(trial)

        # accept only improving moves
        if trial_sum > best_sum + 1e-12:
            centres = trial
            best_sum = trial_sum
            best_centres = trial.copy()
            best_radii = trial_radii.copy()

        # shrink step size
        step *= step_decay

    # final radii for the best centre set
    final_radii, final_sum = compute_optimal_radii(best_centres)
    return best_centres, final_radii, final_sum

    """

    the sum of their radii.

    Returns
    -------
    centres : np.ndarray, shape (26, 2)
        (x, y) coordinates of the circle centres.
    radii : np.ndarray, shape (26,)

        Sum of the radii (objective value).
    """
    np.random.seed(0)  # reproducibility

    best_total = -1.0
    best_centres = None
    best_radii = None

    # Perform a few independent restarts to increase the chance of a good optimum.
    for _ in range(4):
        init = generate_initial_centres()
        centres, radii, total = hill_climb(init, max_iter=2500)

        if total > best_total:
            best_total = total
            best_centres = centres
            best_radii = radii

        # early exit if we already beat the target comfortably
        if best_total > 2.70:
            break

    # Ensure the returned radii exactly match the best centre configuration.
    # (the hill‑climb already solved the LP, but recompute for safety)
    final_radii, final_total = compute_optimal_radii(best_centres)

    return best_centres, final_radii, float(final_total)
```
