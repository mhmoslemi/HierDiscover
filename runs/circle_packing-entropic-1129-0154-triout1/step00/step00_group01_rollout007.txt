    ub = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Build constraints: pairwise
    m = n*(n-1)//2 + n
    A = np.zeros((m, n))
    b = np.empty(m)
    idx=0
    # pairwise constraints
    for i in range(n):
        for j in range(i+1,n):
            A[idx,i]=1
            A[idx,j]=1
            b[idx] = np.linalg.norm(centers[i]-centers[j])
            idx+=1
    # wall constraints
    for i in range(n):
        A[idx,i]=1
        b[idx]=ub[i]
        idx+=1
    # objective:
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # maybe degrade to default with small tolerance
        raise RuntimeError('LP failed')
    radii = res.x
    return radii

def solve_lp(centers):
    n = centers.shape[0]
    # compute wall distances
    wall_limits = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])  # shape (n,)

    # pairwise distances
    # use broadcast to compute upper triangle distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Only need i<j entries.

    # Number constraints: npair = n*(n-1)//2
    pair_count = n*(n-1)//2
    total_constraints = pair_count + n

    A = np.zeros((total_constraints, n))
    b = np.empty(total_constraints)

    row = 0
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1

    # wall constraints: each radius <= wall limit
    for i in range(n):
        A[row, i] = 1
        b[row] = wall_limits[i]
        row += 1

    c = -np.ones(n)
    # Use bounds (0, None)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError('LP failed')
    radii = res.x
    # adjust epsilon
    radii = np.maximum(radii - 1e-9, 0.0)
    sum_r = radii.sum()
    return radii, sum_r

We need a feasible placement of 26 circles inside the unit square that maximises the sum of their radii.  
For a *fixed* set of centre positions the radii that satisfy the non‑overlap and boundary constraints can be found by a **linear program**:

```
max   Σ r_i
s.t.  r_i + r_j ≤ distance(centre_i , centre_j)      ∀ i<j
      r_i ≤ min(x_i, 1‑x_i, y_i, 1‑y_i)            ∀ i
      r_i ≥ 0
```

The constraints are linear in the radii, therefore the LP gives the optimal radii for any given placement of the centres.

import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _solve_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the linear programme that maximises the sum of radii for a
    fixed set of centres.

    Returns the optimal radii (slightly reduced by a tiny epsilon) and
    their sum.
    """
    n = centers.shape[0]

    # distance from each centre to the four walls
    wall_limits = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.einsum("ijk,ijk->ij", diff, diff))

    # Build the inequality matrix A_ub * r <= b_ub
    pair_cnt = n * (n - 1) // 2
    total_cnt = pair_cnt + n
    A = np.zeros((total_cnt, n), dtype=float)
    b = np.empty(total_cnt, dtype=float)

    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    for i in range(n):
        A[row, i] = 1.0
        b[row] = wall_limits[i]
        row += 1

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=float)

    # bounds: radii are non‑negative, no explicit upper bound (walls already limit them)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the extremely unlikely case of failure fall back to zero radii
        radii = np.zeros(n, dtype=float)
    else:
        radii = res.x

    # Tiny safety margin to avoid the 1e‑12 tolerance in the validator
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)

    return radii, float(radii.sum())

                        best_local_sum = trial_sum
                        best_local_pos = new_pos
                        trial_radii = trial_r

                if best_local_sum > cur_sum + 1e-9:
                    # accept the move
                    cur_c[i] = best_local_pos
                    radii = trial_radii
                    cur_sum = best_local_sum
                    improved = True
    return cur_c, radii, cur_sum


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Find a packing of 26 circles in the unit square that exceeds the
    required sum of radii (2.636).  The routine builds several
    candidate centre sets, refines them with a tiny hill‑climbing
    procedure and keeps the best feasible solution.
    """
    TARGET = 2.636
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # deterministic RNG for reproducibility
    base_rng = np.random.RandomState(123456)

    # ------------------------------------------------------------------
    # 1. try a few hand‑crafted candidates
    candidates = []

    # regular grid + extra random point
    candidates.append(_grid_candidate())

    # farthest‑point (blue‑noise) set
    candidates.append(_farthest_point_candidate(base_rng))

    # a few pure random sets
    for seed in range(5):
        rng = np.random.RandomState(seed + 1000)
        candidates.append(_random_candidate(rng))

    # ------------------------------------------------------------------
    # 2. evaluate and locally improve each candidate
    for cand in candidates:
        # local hill‑climbing refinement
        improved_c, improved_r, improved_sum = _hill_climb(
            cand, base_rng, max_iters=30
        )

        if improved_sum > best_sum:
            best_sum = improved_sum
            best_centers = improved_c
            best_radii = improved_r

        if best_sum >= TARGET:
            break

    # ------------------------------------------------------------------
    # 3. if still not enough, fall back to a longer random search
    if best_sum < TARGET:
        for trial in range(2000):
            rng = np.random.RandomState(trial + 5000)
            centres = _random_candidate(rng)
            radii, s = _solve_radii(centres)
            if s > best_sum:
                best_sum = s
                best_centers = centres
                best_radii = radii
            if best_sum >= TARGET:
                break

    # Ensure we always return something valid
    if best_centers is None:
        # (should never happen) fall back to the simple grid
        best_centers = _grid_candidate()
        best_radii, best_sum = _solve_radii(best_centers)

    return best_centers, best_radii, float(best_sum)
