```python
def solve_radii(centers):
    n = len(centers)
    # compute boundary distances
    bdist = np.minimum.reduce([centers[:,0], centers[:,1], 1-centers[:,0], 1-centers[:,1]])
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build A_ub
    # pair constraints
    pairs = [(i,j) for i in range(n) for j in range(i+1,n)]
    M = len(pairs) + n
    A_ub = np.zeros((M, n))
    b_ub = np.zeros(M)
    # pair constraints
    for idx, (i,j) in enumerate(pairs):
        A_ub[idx, i] = 1
        A_ub[idx, j] = 1
        b_ub[idx] = dists[i,j]
    # boundary constraints
    for i in range(n):
        A_ub[len(pairs) + i, i] = 1
        b_ub[len(pairs) + i] = bdist[i]
    # objective
    c = -np.ones(n)  # we minimize -sum
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        raise RuntimeError(res.message)
    radii = res.x
    return radii
```

```python
def solve_radii(centers):
    n = centers.shape[0]
    # boundary distances
    bdist = np.minimum.reduce([centers[:,0], centers[:,1], 1.0 - centers[:,0], 1.0 - centers[:,1]])
    # pairwise distances
    diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diffs**2, axis=2))
    # Build constraints
    m_pair = n*(n-1)//2
    total_constraints = m_pair + n
    A = np.zeros((total_constraints, n), dtype=np.float64)
    b = np.zeros(total_constraints, dtype=np.float64)
    # pair constraints
    idx = 0
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
    # boundary constraints
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = bdist[i]
        idx += 1
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii
```

```
def approx_sum(centers):
    n = centers.shape[0]
    # boundary distances
    bdist = np.minimum.reduce([centers[:,0], centers[:,1], 1 - centers[:,0], 1 - centers[:,1]])
    # compute nearest neighbor distances
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    # set diagonal large
    np.fill_diagonal(dists, np.inf)
    nearest = np.min(dists, axis=1)
    radii = np.minimum(bdist, nearest/2.0)
    return radii.sum(), radii
```

assistantfinal**Strategy**

To maximise the total sum of radii we treat the radii as a linear‑programming (LP) problem **once the circle centres are fixed**.  
For a given set of centre points  

* `r_i ≥ 0`  
* `r_i ≤ distance to the four sides of the square`  
* `r_i + r_j ≤ distance(centre_i , centre_j)`  

are all linear constraints, therefore the optimal radii for those centres can be obtained by a single call to `scipy.optimize.linprog`.

The remaining difficulty is to choose the 26 centre positions.  
We start from a simple regular grid (5 × 5 points) plus one extra point.  
That configuration already respects the non‑overlap constraints when all radii are 0.1, giving a total sum of 2.5.

From this start we run a **black‑box hill‑climbing / simulated‑annealing** loop:

1. Pick a random circle and move its centre by a small Gaussian step (step size is slowly reduced).  
2. Evaluate a *cheap* lower‑bound on the possible radii:  
   `r_i = min( distance to the square border, ½·nearest neighbour distance )`.  
   The sum of these lower bounds is quick to compute and is a reliable indicator of improvement.  
3. If the lower‑bound sum improves we accept the move (and keep the best configuration found).  
4. Whenever a new best lower‑bound is discovered we resolve the full LP for that configuration to obtain the *exact* radii and the true sum.  
5. The loop runs for a few × 10⁴ cheap evaluations; only a few hundred LP solves are needed, so the whole procedure finishes quickly.

At the end we return the centres, the radii from the last LP solve (the best found), and the corresponding total sum of radii.  
Running the optimisation with a fixed random seed makes the result deterministic, and the final LP guarantees that all circles satisfy the non‑overlap and boundary constraints required by the validator.

```python
import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Helper: solve the linear programme for radii given centres
# ------------------------------------------------------------
def solve_radii(centres: np.ndarray) -> np.ndarray:
    """
    Returns the optimal radii (maximising sum of radii) for a fixed set
    of centre positions inside the unit square.
    """
    n = centres.shape[0]

    # distance to the four sides of the square
    border = np.minimum.reduce(
        [centres[:, 0], centres[:, 1], 1.0 - centres[:, 0], 1.0 - centres[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # Build the inequality matrix A_ub * r <= b_ub
    #   – one entry per pair (i,j):   r_i + r_j <= d_ij
    #   – one entry per centre i:     r_i          <= border_i
    n_pairs = n * (n - 1) // 2
    m = n_pairs + n                     # total number of constraints
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    # pair constraints
    idx = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1

    # boundary constraints
    for i in range(n):
        A[idx, i] = 1.0
        b[idx] = border[i]
        idx += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    # solve (highs gives a fast, reliable interior‑point method)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0.0, None)] * n, method="highs")
    if not res.success:
        # In the unlikely event the LP fails we fall back to all zeros
        return np.zeros(n, dtype=np.float64)
    return res.x


# ------------------------------------------------------------
# Cheap lower‑bound on radii (used for fast hill‑climbing)
# ------------------------------------------------------------
def lower_bound_sum(centres: np.ndarray) -> float:
    """
    Returns a fast lower bound for the total radius sum:
        r_i = min( distance to border, ½·nearest neighbour distance )
    """
    n = centres.shape[0]

    # border distances
    border = np.minimum.reduce(
        [centres[:, 0], centres[:, 1], 1.0 - centres[:, 0], 1.0 - centres[:, 1]]
    )

    # pairwise distances
    diff = centres[:, np.newaxis, :] - centres[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))
    np.fill_diagonal(dists, np.inf)               # ignore self‑distance
    nearest = np.min(dists, axis=1)                # distance to nearest neighbour

    r = np.minimum(border, nearest * 0.5)
    return r.sum()

# Main routine required by the problem statement
# ------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centres, radii, total_sum) for 26 circles packed in the unit square.
    The routine uses a simple stochastic optimisation with an LP solver to
    obtain a high‑quality arrangement.
    """
    np.random.seed(0)                     # deterministic output

    n = 26

    # ----- initialise with a regular 5×5 grid + one extra point -----
    grid_vals = np.linspace(0.1, 0.9, 5)
    xv, yv = np.meshgrid(grid_vals, grid_vals)
    centres = np.column_stack((xv.ravel(), yv.ravel()))   # 25 points
    # an extra point that is not on the grid (to avoid a duplicate)
    extra = np.array([[0.2, 0.6]])
    centres = np.vstack([centres, extra])                # shape (26,2)

    # ----- keep track of the best configuration found -----
    best_centres = centres.copy()
    best_radii = solve_radii(best_centres)
    best_sum = float(best_radii.sum())
    best_lb_sum = lower_bound_sum(best_centres)          # lower‑bound version

    # ----- hill‑climbing parameters -----
    max_iters = 20000
    step_initial = 0.30
    step_final = 0.005

    current_centres = centres.copy()
    current_lb = lower_bound_sum(current_centres)

    for it in range(max_iters):
        # linearly anneal the step size
        step = step_initial - (step_initial - step_final) * (it / max_iters)

        # pick a random circle and propose a move
        i = np.random.randint(n)
        proposal = current_centres[i] + np.random.normal(scale=step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centres[i] = proposal

        new_lb = lower_bound_sum(new_centres)

            sum_candidate = float(radii_candidate.sum())

                best_centres = new_centres.copy()
                best_radii = radii_candidate.copy()
                best_lb_sum = new_lb

    # final sanity check: recompute the exact LP for the best centres
    best_radii = solve_radii(best_centres)

    return best_centres, best_radii, best_sum
```
