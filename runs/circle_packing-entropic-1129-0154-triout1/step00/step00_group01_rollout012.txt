def compute_optimal_radii(centers):
    n = centers.shape[0]
    # Build A_ub and b_ub
    rows = []
    b = []
    # pairwise constraints
    for i in range(n):
        xi, yi = centers[i]
        # boundary constraints: r_i <= xi, <= 1-xi, <= yi, <= 1-yi
        rows.append(np.eye(n)[i])  # r_i <= xi
        b.append(xi)
        rows.append(np.eye(n)[i]) # r_i <= 1-xi
        b.append(1 - xi)
        rows.append(np.eye(n)[i]) # r_i <= yi
        b.append(yi)
        rows.append(np.eye(n)[i]) # r_i <= 1-yi
        b.append(1 - yi)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(d)
    A_ub = np.array(rows)
    b_ub = np.array(b)
    c = -np.ones(n)  # maximize sum of r -> minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        radii = res.x
        # Clip small negatives
        radii = np.maximum(radii, 0.0)
        return radii
    else:
        raise RuntimeError('LP failed: ' + res.message)

def con_nonoverlap(x):
    n=26
    xs = x[0:n]; ys = x[n:2*n]; rs = x[2*n:3*n]
    # compute distances matrix
    # We compute for each pair i<j: squared distances - (ri+rj)^2
    cons = []
    for i in range(n):
        for j in range(i+1,n):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            cons.append(dx*dx + dy*dy - (rs[i] + rs[j])**2)
    return np.array(cons)

def con_nonoverlap(x):
    n = 26
    xs = x[:n]
    ys = x[n:2*n]
    rs = x[2*n:]
    # compute pairwise constraints
    cons = []
    for i in range(n):
        xi = xs[i]; yi = ys[i]; ri = rs[i]
        for j in range(i+1, n):
            dx = xi - xs[j]
            dy = yi - ys[j]
            rsum = ri + rs[j]
            cons.append(dx*dx + dy*dy - rsum*rsum)
    return np.array(cons)

def compute_optimal_radii(centers):
    import numpy as np
    from scipy.optimize import linprog
    n = centers.shape[0]
    # Build constraints
    # Pairwise
    d_ij = []
    # Use list of arrays
    rows = []
    # We'll fill A_ub and b_ub
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            d_ij.append(d)
    # boundary constraints for each i
    for i in range(n):
        x, y = centers[i]
        # left/right
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        d_ij.append(x) # r_i <= x
        rows.append(row)
        d_ij.append(1 - x) # r_i <= 1 - x
        rows.append(row)
        d_ij.append(y) # r_i <= y
        rows.append(row)
        d_ij.append(1 - y) # r_i <= 1 - y
    A_ub = np.array(rows)
    b_ub = np.array(d_ij)
    c = -np.ones(n)  # maximize sum r_i
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set all radii zero maybe
        return np.zeros(n)
    radii = res.x
    # Clip any small negative due to numeric tolerance
    radii = np.maximum(radii, 0.0)
    return radii

assistantfinal**Solution Overview**

The task is to place 26 circles inside the unit square `[0,1]×[0,1]`.  
For a *fixed* set of centre positions the problem of maximizing the sum of the radii is a **linear program**:

```
max   Σ r_i
s.t.  r_i + r_j ≤ dist( centre_i , centre_j )        for all i < j
      r_i ≤ x_i ,   r_i ≤ 1‑x_i                       (left / right wall)
      r_i ≤ y_i ,   r_i ≤ 1‑y_i                       (bottom / top wall)
      r_i ≥ 0
```

The linear program can be solved very quickly with `scipy.optimize.linprog`.

The whole optimisation problem (positions + radii) is therefore tackled in two steps:

1. **Choose a set of centre positions.**  
   We start from a regular 5 × 5 grid (25 points) and add a 26‑th point.
   The grid points are slightly jittered and several random initialisations are tried.

2. **Improve the positions.**  
   Using the radii obtained from the LP as a feasible starting point, we run a
   non‑linear optimiser (`SLSQP`).  
   The optimiser can move the centres while keeping all constraints satisfied.
   After the optimisation finishes we solve the LP *once more* for the final
   centre positions – this yields the *optimal* radii for those positions.

The algorithm is repeated for a few random starts; the best solution found
(its centre coordinates, radii and the achieved total radius) is returned.

All helper functions are defined at module level – no closures or lambda
expressions are used – and the code only relies on `numpy` and `scipy`.

```python
import numpy as np
from scipy.optimize import linprog, minimize

# ----------------------------------------------------------------------
# Global constant – number of circles
NN = 26
# ----------------------------------------------------------------------


def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear program that, for a given set of centre positions,
    maximises the sum of the radii.
    """
    n = centers.shape[0]
    # ----- pairwise distance constraints -----
    A = []
    b = []
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A.append(row)
            b.append(d)

    # ----- wall constraints -----
    for i in range(n):
        x, y = centers[i]
        row = np.zeros(n)

        # left wall  :  r_i ≤ x_i
        row[i] = 1.0
        A.append(row.copy())
        b.append(x)

        # right wall :  r_i ≤ 1‑x_i
        A.append(row.copy())
        b.append(1.0 - x)

        # bottom wall:  r_i ≤ y_i
        A.append(row.copy())
        b.append(y)

        # top wall   :  r_i ≤ 1‑y_i
        A.append(row.copy())
        b.append(1.0 - y)

    A = np.asarray(A, dtype=np.float64)
    b = np.asarray(b, dtype=np.float64)

    # ----- linear objective (maximise Σ r_i) -----
    c = -np.ones(n)                     # linprog minimises, so minimise -Σr_i
    bounds = [(0.0, None)] * n          # radii are non‑negative

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – return zeros (still a feasible point)
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(res.x, 0.0)
    return radii


# ----------------------------------------------------------------------
# Constraint functions for the non‑linear optimiser (SLSQP)
# ----------------------------------------------------------------------
def con_boundary(x: np.ndarray) -> np.ndarray:
    """
    Wall constraints:
        x_i - r_i          ≥ 0
        1 - x_i - r_i      ≥ 0
        y_i - r_i          ≥ 0
        1 - y_i - r_i      ≥ 0
    """
    xs = x[:NN]
    ys = x[NN:2 * NN]
    rs = x[2 * NN:3 * NN]
    return np.concatenate([xs - rs,
                           1.0 - xs - rs,
                           ys - rs,
                           1.0 - ys - rs])


def con_nonoverlap(x: np.ndarray) -> np.ndarray:
    """
    Non‑overlap constraints for every pair (i,j):
        (x_i - x_j)^2 + (y_i - y_j)^2  -  (r_i + r_j)^2   ≥ 0
    """
    xs = x[:NN]
    ys = x[NN:2 * NN]
    rs = x[2 * NN:3 * NN]
    cons = []
    for i in range(NN):
        xi, yi, ri = xs[i], ys[i], rs[i]
        for j in range(i + 1, NN):
            dx = xi - xs[j]
            dy = yi - ys[j]
            rsum = ri + rs[j]
            cons.append(dx * dx + dy * dy - rsum * rsum)
    return np.array(cons, dtype=np.float64)


def objective(x: np.ndarray) -> float:
    """We maximise Σ r_i → minimise -Σ r_i."""
    rs = x[2 * NN:3 * NN]
    return -np.sum(rs)


# ----------------------------------------------------------------------
# Main routine
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_of_radii) for a packing of 26 circles
    inside the unit square that tries to maximise the total radius.
    """
    rng = np.random.default_rng(123456)      # reproducible randomness
    best_sum = -1.0
    best_centers = None
    best_radii = None

    # ------------------------------------------------------------------
    # A deterministic 5×5 grid (25 points) – serves as a good start.
    # ------------------------------------------------------------------
    grid_vals = np.linspace(0.1, 0.9, 5)      # 0.1,0.3,0.5,0.7,0.9
    grid_points = np.array([(x, y) for y in grid_vals for x in grid_vals])
    # add a 26‑th point (random, but kept inside the square)
    extra_pt = rng.uniform(0.0, 1.0, size=(1, 2))
    base_centers = np.vstack([grid_points, extra_pt])

    # ------------------------------------------------------------------
    # Perform several independent attempts (different initial positions)
    # ------------------------------------------------------------------
    MAX_ATTEMPTS = 20

    for attempt in range(MAX_ATTEMPTS):
        # ----- initialise centre positions --------------------------------
        if attempt == 0:
            # start from the grid (good deterministic baseline)
            centres = base_centers.copy()
        else:
            # completely random start
            centres = rng.uniform(0.0, 1.0, size=(NN, 2))

        # Small jitter – helps the optimiser leave the perfectly symmetric grid
        jitter = rng.uniform(-0.03, 0.03, size=centres.shape)
        centres = np.clip(centres + jitter, 0.0, 1.0)

        # ----- obtain a feasible set of radii for those centres -----------
        radii_start = compute_optimal_radii(centres)
        # give a tiny slack so the optimiser has room to move
        radii_start = np.maximum(radii_start - 1e-6, 0.0)

        # ----- pack everything into one vector for SLSQP -----------------
        x0 = np.concatenate([centres[:, 0], centres[:, 1], radii_start])

        # bounds for (x, y, r)
        bounds = [(0.0, 1.0)] * (2 * NN) + [(0.0, 1.0)] * NN

        # constraints (inequalities returned as arrays)
        cons = [
            {'type': 'ineq', 'fun': con_nonoverlap},
            {'type': 'ineq', 'fun': con_boundary}
        ]

        # ----- local optimisation -----------------------------------------
        res = minimize(fun=objective,
                       x0=x0,
                       method='SLSQP',
                       bounds=bounds,
                       constraints=cons,
                       options={'maxiter': 400,
                                'ftol': 1e-9,
                                'disp': False})

        # use the result if optimisation succeeded; otherwise fall back
        if res.success:
            final_centres = np.stack([res.x[:NN], res.x[NN:2 * NN]], axis=1)
        else:
            final_centres = centres

        # ----- recompute the *optimal* radii for the final centre set ------
        final_radii = compute_optimal_radii(final_centres)
        # tiny uniform shrink to stay safely inside the tolerances of the validator
        final_radii = np.maximum(final_radii - 1e-9, 0.0)

        current_sum = float(final_radii.sum())
        if current_sum > best_sum:
            best_sum = current_sum
            best_centers = final_centres.copy()
            best_radii = final_radii.copy()
            if best_sum >= 2.636:          # target already reached – stop early
                break

    # ------------------------------------------------------------------
    # Safety net – if nothing better than the plain grid was found,
    # return the grid solution (it is always a valid packing).
    # ------------------------------------------------------------------
    if best_centers is None:
        best_centers = base_centers
        best_radii = compute_optimal_radii(best_centers)
        best_sum = float(best_radii.sum())

    return best_centers, best_radii, best_sum
```
