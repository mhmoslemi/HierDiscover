import numpy as np
from scipy.optimize import linprog
import random

def optimal_radii(centers):
    n = centers.shape[0]
    # compute distances
    # Make A_ub as list of rows
    # We'll build dense matrix
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # right
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # bottom
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # top
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d
            row += 1
    # objective
    c = -np.ones(n) # minimize -sum => maximize sum
    # bounds for each variable: (0, None)
    bounds = [(0, None)] * n
    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii = 0
        return np.zeros(n)
    radii = res.x
    # subtract a tiny epsilon for safety
    epsilon = 1e-12
    radii = np.maximum(radii - epsilon, 0.0)
    return radii

def anneal(centers, max_iter=5000, step=0.05, seed=0):
    np.random.seed(seed)
    best_centers = centers.copy()
    best_radii = optimal_radii(best_centers)
    best_sum = best_radii.sum()
    current_centers = best_centers.copy()
    current_radii = best_radii
    current_sum = best_sum
    T0 = 0.1
    for it in range(max_iter):
        T = T0 * (0.99 ** it)
        # choose random circle
        i = np.random.randint(0, centers.shape[0])
        dx = (np.random.rand() - 0.5) * 2 * step
        dy = (np.random.rand() - 0.5) * 2 * step
        new_center = current_centers[i] + np.array([dx, dy])
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        new_centers = current_centers.copy()
        new_centers[i] = new_center
        new_radii = optimal_radii(new_centers)
        new_sum = new_radii.sum()
        # accept if improvement or with probability
        if new_sum > current_sum or np.random.rand() < np.exp((new_sum - current_sum) / max(T, 1e-12)):
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
        # optionally reduce step after certain iterations
        if (it+1) % 1000 == 0:
            step *= 0.8
    return best_centers, best_radii, best_sum

import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear program that maximises the sum of radii for a fixed
    set of centre positions.
    """
    n = centers.shape[0]

    # Number of constraints: 4 per circle (left, right, bottom, top) + one per pair
    n_bc = 4 * n
    n_pair = n * (n - 1) // 2
    m = n_bc + n_pair

    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # left  : r_i <= x_i
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # right : r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # bottom: r_i <= y_i
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # top   : r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # non‑overlap constraints
    for i in range(n):
        ci = centers[i]
        for j in range(i + 1, n):
            d = np.linalg.norm(ci - centers[j])
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d
            row += 1

    # maximise sum(r)  → minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the unlikely event of a failure, fall back to zeros
        return np.zeros(n, dtype=np.float64)

    radii = res.x
    # shrink a tiny bit to give a safety margin for the validator
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)
    return radii


def _initial_grid() -> np.ndarray:
    """
    Produce 25 points on a 5×5 grid (spacing 0.2) plus one random extra point.
    """
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([0.1 + i * 0.2, 0.1 + j * 0.2])
    # extra point – use a reproducible random location
    rng = np.random.default_rng(12345)
    extra = rng.random(2)
    pts.append(extra.tolist())
    return np.array(pts, dtype=np.float64)


def _anneal(
    init_centers: np.ndarray,
    max_iter: int = 3000,
    step: float = 0.07,
    seed: int = 0,
) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Simple simulated‑annealing / hill‑climbing that moves the centres
    and always re‑optimises the radii by the LP.
    """
    rng = np.random.default_rng(seed)

    best_centers = init_centers.copy()
    best_radii = _optimal_radii(best_centers)
    best_sum = best_radii.sum()

    cur_centers = best_centers.copy()
    cur_radii = best_radii.copy()
    cur_sum = best_sum

    # temperature schedule
    T0 = 0.12

    for it in range(max_iter):
        T = T0 * (0.995 ** it)  # slowly cooling

        # propose a move
        i = rng.integers(0, cur_centers.shape[0])
        dx = (rng.random() - 0.5) * 2.0 * step
        dy = (rng.random() - 0.5) * 2.0 * step
        new_pos = cur_centers[i] + np.array([dx, dy])
        new_pos = np.clip(new_pos, 0.0, 1.0)

        new_centers = cur_centers.copy()
        new_centers[i] = new_pos

        new_radii = _optimal_radii(new_centers)
        new_sum = new_radii.sum()

        # acceptance test
        accept = False
        if new_sum > cur_sum:
            accept = True
        else:
            # Metropolis criterion
            prob = np.exp((new_sum - cur_sum) / max(T, 1e-12))
            if rng.random() < prob:
                accept = True

        if accept:
            cur_centers = new_centers
            cur_radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

        # slowly reduce the move size to fine‑tune the later stages
        if (it + 1) % 800 == 0:
            step *= 0.8

    return best_centers, best_radii, float(best_sum)


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Return a feasible packing of 26 circles inside the unit square.
    The function uses a small stochastic optimisation to obtain a high
    total sum of radii.
    """
    # deterministic start
    np.random.seed(0)
    init = _initial_grid()
    centres, radii, total = _anneal(init, max_iter=3000, step=0.07, seed=42)

    # final safety check – ensure everything is inside the square
    # (the optimiser already respects the constraints, we only
    #  subtract a tiny epsilon, but we double‑check for robustness)
    for i in range(centres.shape[0]):
        x, y = centres[i]
        r = radii[i]
        assert 0.0 <= x - r + 1e-12
        assert x + r <= 1.0 + 1e-12
        assert 0.0 <= y - r + 1e-12
        assert y + r <= 1.0 + 1e-12

    return centres, radii, total
