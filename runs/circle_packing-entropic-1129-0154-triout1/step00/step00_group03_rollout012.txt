def compute_optimal_radii(centers):
    n = centers.shape[0]
    xs = centers[:,0]
    ys = centers[:,1]
    # distances
    dists = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)
    # Setup constraints
    num_pair = n*(n-1)//2
    num_constraints = 5*n + num_pair
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    idx = 0
    # r_i - x_i <= 0
    for i in range(n):
        A[idx,i]=1
        b[idx]=xs[i]
        idx+=1
    # r_i + x_i <= 1 -> r_i <= 1 - x_i
    for i in range(n):
        A[idx,i]=1
        b[idx]=1 - xs[i]
        idx+=1
    # r_i - y_i <= 0
    for i in range(n):
        A[idx,i]=1
        b[idx]=ys[i]
        idx+=1
    # r_i + y_i <= 1
    for i in range(n):
        A[idx,i]=1
        b[idx]=1 - ys[i]
        idx+=1
    # Pair constraints
    for i in range(n):
        for j in range(i+1,n):
            A[idx,i]=1
            A[idx,j]=1
            b[idx]=dists[i,j]
            idx+=1
    # Solve using linprog
    res = linprog(c=-np.ones(n), A_ub=A, b_ub=b, bounds=[(0,None)]*n, method='highs')
    if not res.success:
        # maybe fallback
        raise RuntimeError('LP failed')
    radii = res.x
    return radii, radii.sum()

```
def compute_optimal_radii(centers):
    n = centers.shape[0]
    xs = centers[:,0]
    ys = centers[:,1]
    # distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Constraint matrix
    num_pair = n * (n - 1) // 2
    num_constraints = 5 * n + num_pair
    A = np.zeros((num_constraints, n))
    b = np.zeros(num_constraints)
    idx = 0
    # r_i <= x_i
    A[np.arange(n), np.arange(n)] = 1
    b[:n] = xs
    idx = n
    # r_i <= 1 - x_i
    A[np.arange(n, 2*n), np.arange(n)] = 1
    b[n:2*n] = 1 - xs
    idx = 2*n
```

assistantfinal**Strategy**

To maximize the sum of the radii of 26 non‑overlapping circles inside the unit square we treat the radii as linear variables and the centre positions as free parameters.

For a *fixed* set of centre positions the radii that maximise the total sum are the solution of a **linear program**

```
max   Σ r_i
s.t.  r_i ≥ 0
      r_i ≤ x_i ,  r_i ≤ 1‑x_i ,  r_i ≤ y_i ,  r_i ≤ 1‑y_i      (stay inside the square)
      r_i + r_j ≤ dist(i , j)                                    (no overlap)
```

where `dist(i , j)` is the Euclidean distance between centres `i` and `j`.  
Thus, given any configuration of points we can obtain the optimal radii instantly with `scipy.optimize.linprog`.

The overall problem is non‑convex because the distances depend on the centre coordinates.  
We use a simple **simulated‑annealing / hill‑climbing** meta‑heuristic:

1. **Initial layout** – a regular 5×5 grid (shifted by 0.1 from the walls) gives a good starting point (≈2.5 for the sum).  
   One extra point is added with a small random offset to obtain 26 circles.

2. **Iterative improvement** – at each iteration a new candidate layout is generated by perturbing the current centres:
   * most of the time a small Gaussian‑like jitter,
   * occasionally a larger random jump for a single circle.
   The new layout is fed to the LP, the resulting sum of radii is computed and the move is accepted  
   – always if it improves the sum, otherwise with a probability that slowly decreases (temperature schedule).

3. **Multiple restarts** – a few independent runs (different random seeds) are performed; the best packing found overall is kept.

4. **Finalisation** – after the optimisation loop the radii are recomputed once more, a tiny safety margin (`‑1e‑9`) is subtracted to avoid numerical tolerance issues, and the final `(centers, radii, sum)` tuple is returned.

The algorithm runs only a few thousand linear‑program solves (each with 26 variables and ≤ 455 constraints), which is fast with the HiGHS backend.  
All code is written with top‑level functions, uses only `numpy` and `scipy`, and obeys the required interface.

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _optimal_radii(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a fixed set of centres solve the linear programme that yields the
    maximal radii and their total sum.
    """
    n = centers.shape[0]
    xs = centers[:, 0]
    ys = centers[:, 1]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of constraints: 5 per circle + one for each unordered pair
    n_pairs = n * (n - 1) // 2
    n_cons = 5 * n + n_pairs
    A = np.zeros((n_cons, n))
    b = np.zeros(n_cons)

    row = 0
    # r_i <= x_i
    A[np.arange(row, row + n), np.arange(n)] = 1.0
    b[row:row + n] = xs
    row += n

    # r_i <= 1 - x_i
    A[np.arange(row, row + n), np.arange(n)] = 1.0
    b[row:row + n] = 1.0 - xs
    row += n

    # r_i <= y_i
    A[np.arange(row, row + n), np.arange(n)] = 1.0
    b[row:row + n] = ys
    row += n

    # r_i <= 1 - y_i
    A[np.arange(row, row + n), np.arange(n)] = 1.0
    b[row:row + n] = 1.0 - ys
    row += n

    # pairwise: r_i + r_j <= distance_ij
    i_idx, j_idx = np.triu_indices(n, k=1)
    A[row:, i_idx] = 1.0
    A[row:, j_idx] = 1.0
    b[row:] = dists[i_idx, j_idx]

    # linear objective: maximise sum(r)  => minimise -sum(r)
    c = -np.ones(n)

    # bounds: radii are non‑negative
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the extremely unlikely case the LP fails, fall back to zeros.
        radii = np.zeros(n)
    else:
        radii = np.maximum(res.x, 0.0)

    return radii, float(radii.sum())


def _initial_grid() -> np.ndarray:
    """
    Produce 25 points on a 5×5 regular grid shifted by 0.1 from the
    square boundaries and add one extra point with a tiny random offset.
    """
    rng = np.random.default_rng(0)
    grid_vals = np.linspace(0.1, 0.9, 5)            # 0.1,0.3,0.5,0.7,0.9
    xv, yv = np.meshgrid(grid_vals, grid_vals)
    points = np.column_stack([xv.ravel(), yv.ravel()])   # 25 points

    # extra point – place it near the centre but not exactly on an existing grid point
    extra = np.array([0.55, 0.55]) + (rng.random(2) - 0.5) * 0.02
    extra = np.clip(extra, 0.0, 1.0)
    points = np.vstack([points, extra])
    # a tiny jitter for all points makes the layout non‑degenerate
    jitter = (rng.random(points.shape) - 0.5) * 0.02
    points += jitter
    return np.clip(points, 0.0, 1.0)


def _random_initial(rng: np.random.Generator) -> np.ndarray:
    """
    Completely random start inside the square, kept away from the borders
    by a small margin (0.08) to avoid pathological zero‑radius points.
    """
    return rng.random((26, 2)) * 0.84 + 0.08


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_of_radii) for a packing of 26 circles
    inside the unit square.  The implementation uses a simple simulated
    annealing loop that repeatedly solves a linear programme to obtain the
    optimal radii for a given centre configuration.
    """
    rng = np.random.default_rng(42)

    best_centers = None
    best_radii = None
    best_total = -1.0

    # ---- three independent restarts (different initial layouts) ----
    for restart in range(3):
        if restart == 0:
            cur_centers = _initial_grid()
        else:
            cur_centers = _random_initial(rng)

        cur_radii, cur_total = _optimal_radii(cur_centers)

        # best of this restart
        best_restart_centers = cur_centers.copy()
        best_restart_radii = cur_radii.copy()
        best_restart_total = cur_total

        # simulated‑annealing parameters
        max_iter = 2500
        init_temp = 0.02
        final_temp = 5e-4
        max_step_initial = 0.25
        min_step = 0.003

        for it in range(max_iter):
            # linear temperature schedule
            t = init_temp * ((final_temp / init_temp) ** (it / max_iter))

            # step size shrinks from max_step_initial to min_step
            step = max_step_initial * (1.0 - it / max_iter) + min_step

            # propose a new layout
            proposal = cur_centers.copy()
            # with small probability make a large jump for a single circle
            if rng.random() < 0.05:
                idx = rng.integers(26)
                jump = (rng.random(2) - 0.5) * 2.0 * step
                proposal[idx] += jump
            else:
                # jitter all circles a little
                jitter = (rng.random(proposal.shape) - 0.5) * 2.0 * step
                proposal += jitter

            # enforce the square limits
            proposal = np.clip(proposal, 0.0, 1.0)

            # evaluate the new configuration
            try:
                prop_radii, prop_total = _optimal_radii(proposal)
            except Exception:
                continue  # numerical trouble – ignore this move

            delta = prop_total - cur_total
            if delta > 0.0 or np.exp(delta / (t + 1e-12)) > rng.random():
                # accept the move
                cur_centers = proposal
                cur_radii = prop_radii
                cur_total = prop_total

                # record the best ever seen in this restart
                if cur_total > best_restart_total:
                    best_restart_total = cur_total
                    best_restart_centers = cur_centers.copy()
                    best_restart_radii = cur_radii.copy()

        # keep the globally best restart
        if best_restart_total > best_total:
            best_total = best_restart_total
            best_centers = best_restart_centers
            best_radii = best_restart_radii

    # one final LP evaluation to clean up any tiny numerical inconsistencies
    final_radii, final_sum = _optimal_radii(best_centers)
    # subtract a negligible epsilon to stay safely inside the tolerances
    final_radii = np.maximum(final_radii - 1e-9, 0.0)
    final_sum = float(final_radii.sum())

    return best_centers, final_radii, final_sum
```
