```
def compute_optimal_radii(centers):
    n = len(centers)
    # distances
    diff = centers[:, None, :] - centers[None, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints for LP
    # variable bounds
    ub = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    bounds = [(0, ub_i) for ub_i in ub]
    # Pair constraints
    pairs = [(i,j) for i in range(n) for j in range(i+1,n)]
    m = len(pairs)
    A = np.zeros((m, n))
    b = np.zeros(m)
    for idx, (i,j) in enumerate(pairs):
        A[idx,i] = 1
        A[idx,j] = 1
        b[idx] = D[i,j]
    c = -np.ones(n)  # maximize sum
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii
```

We have to place 26 circles inside the unit square so that they do not overlap and the total sum of their radii is as large as possible.  
For a **fixed set of centres** the optimal radii are obtained by solving a small linear program:

* variables = radii \(r_i\)  
* maximise \(\sum_i r_i\) (equivalently minimise \(-\mathbf 1^T r\))  
* bounds \(0 \le r_i \le \min(x_i, 1-x_i, y_i, 1-y_i)\) – from the square boundary  
* pair‑constraints \(r_i+r_j \le d_{ij}\) where \(d_{ij}\) is the Euclidean distance between the two centres.

Thus the whole problem reduces to finding a good set of centre points.  
A good initial layout is obtained by **farthest‑point sampling** on a fine interior grid – this maximises the minimal distance to already placed points *and* to the square’s sides, giving a well‑spread configuration.

Starting from that layout we run a cheap **hill‑climbing** meta‑heuristic:

1. pick a circle at random,
2. perturb its centre by a small jitter,
3. resolve the linear program for the new set of centres,
4. keep the change only if the total sum of radii increases.

Several short restarts are performed, and the best solution found is returned.  
All computations are deterministic (fixed RNG seed) and use only NumPy and SciPy, which are allowed.  
The linear program guarantees that the returned radii satisfy every non‑overlap and boundary constraint, so the `validate_packing` routine will accept the result.

```python
import numpy as np
from scipy.optimize import linprog

def _max_radii_for_centers(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear program that, for a given set of centres,
    maximises the sum of radii subject to non‑overlap and boundary constraints.
    Returns an array of optimal radii.
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # upper bound from the square’s sides
    ub = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )
    bounds = [(0.0, ub_i) for ub_i in ub]

    # pair constraints r_i + r_j <= d_ij
    pairs_i = []
    pairs_j = []
    for i in range(n):
        for j in range(i + 1, n):
            pairs_i.append(i)
            pairs_j.append(j)
    m = len(pairs_i)
    A = np.zeros((m, n))
    b = np.empty(m)
    for idx, (i, j) in enumerate(zip(pairs_i, pairs_j)):
        A[idx, i] = 1.0
        A[idx, j] = 1.0
        b[idx] = dists[i, j]

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    # solve the LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs", options={"presolve": True})
    if not res.success:
        # fallback – all radii zero
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(res.x, 0.0)  # guard against tiny negative values
    return radii


def _farthest_point_sampling(num_points: int, candidates: np.ndarray) -> np.ndarray:
    """
    Select `num_points` points from `candidates` using farthest‑point sampling.
    The effective distance of a candidate = min( distance to already selected points,
    distance to the nearest square side ). This yields a well‑spread set.
    """
    selected = []

    # work on a mutable copy
    cand = candidates.copy()

    while len(selected) < num_points and cand.shape[0] > 0:
        if selected:
            # distance to the already selected points
            d = np.linalg.norm(
                cand[:, None, :] - np.array(selected)[None, :, :], axis=2
            )
            min_to_selected = np.min(d, axis=1)
        else:
            # no points selected yet -> treat distance as infinite
            min_to_selected = np.full(cand.shape[0], np.inf)

        # distance to the nearest side of the unit square
        side_dist = np.minimum.reduce(
            [cand[:, 0], 1.0 - cand[:, 0], cand[:, 1], 1.0 - cand[:, 1]]
        )

        # effective distance we want to maximise
        effective = np.minimum(min_to_selected, side_dist)
        idx = np.argmax(effective)
        selected.append(cand[idx])
        cand = np.delete(cand, idx, axis=0)

    return np.array(selected)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns:
        centres  – (26,2) array of (x,y) coordinates inside [0,1]^2
        radii   – (26,) array of non‑negative radii satisfying all constraints
        sum_r   – float, the sum of the radii (objective value)
    The routine uses farthest‑point sampling for an initial layout and a
    short hill‑climbing optimisation, each evaluation solved by a linear program.
    """
    rng = np.random.default_rng(0)  # deterministic randomness

    # a fine interior grid that will serve as candidate points
    grid_n = 30
    eps = 1.0 / (grid_n + 1)  # keep points away from the very border
    xs = np.linspace(eps, 1.0 - eps, grid_n)
    ys = np.linspace(eps, 1.0 - eps, grid_n)
    grid = np.stack(np.meshgrid(xs, ys), -1).reshape(-1, 2)

    best_sum = -1.0
    best_centers = None
    best_radii = None

    # perform a few independent restarts
    for restart in range(4):
        # ---- initial layout -------------------------------------------------
        centres = _farthest_point_sampling(26, grid.copy())
        radii = _max_radii_for_centers(centres)
        cur_sum = radii.sum()

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centres.copy()
            best_radii = radii.copy()

        # ---- simple hill‑climbing -------------------------------------------
        for it in range(2500):
            # pick a circle and propose a small random move
            i = rng.integers(0, 26)
            new_centres = centres.copy()
            # jitter within a box of side ≈0.1
            jitter = (rng.random(2) - 0.5) * 0.1
            cand = new_centres[i] + jitter
            # stay safely inside the unit square
            cand = np.clip(cand, 1e-6, 1.0 - 1e-6)
            new_centres[i] = cand

            new_radii = _max_radii_for_centers(new_centres)
            new_sum = new_radii.sum()

            if new_sum > cur_sum + 1e-9:  # accept only genuine improvements
                centres = new_centres
                radii = new_radii
                cur_sum = new_sum
                if new_sum > best_sum:
                    best_sum = new_sum
                    best_centers = new_centres.copy()
                    best_radii = new_radii.copy()

        # a tiny random perturbation of the whole set before the next restart
        # (helps escaping from a possibly poor region)
        if restart < 3:
            noise = (rng.random(best_centers.shape) - 0.5) * 0.05
            perturbed = best_centers + noise
            perturbed = np.clip(perturbed, 1e-6, 1.0 - 1e-6)
            best_centers = perturbed

    # Ensure the returned radii correspond exactly to the best centres
    # (re‑solve once more for numerical safety)
    best_radii = _max_radii_for_centers(best_centers)
    best_sum = float(best_radii.sum())

    return best_centers, best_radii, best_sum
```
