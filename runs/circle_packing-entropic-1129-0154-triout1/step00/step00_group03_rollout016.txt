    # 5×5 grid coordinates: 0.1, 0.3, 0.5, 0.7, 0.9
    grid_vals = np.linspace(0.1, 0.9, 5)
    xs, ys = np.meshgrid(grid_vals, grid_vals)
    grid_centers = np.column_stack([xs.ravel(), ys.ravel()])   # shape (25, 2)

    # Extra circle at the origin
    extra_center = np.array([[0.0, 0.0]])                     # shape (1, 2)

    # Combine to get 26 centres
    centers = np.vstack([grid_centers, extra_center])        # shape (26, 2)

    # Radii: 0.1 for the grid circles, 0.04 for the extra one
    radii = np.full(26, 0.1, dtype=np.float64)
    radii[-1] = 0.04

    # Compute total sum of radii
    sum_radii = float(np.sum(radii))

    return centers, radii, sum_radii
