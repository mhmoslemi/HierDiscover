import numpy as np
from scipy.optimize import linprog
import random

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances
    dists = []
    rows = []
    b = []
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            dist = np.linalg.norm(centers[i] - centers[j])
            rows.append(row)
            b.append(dist)
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        # <= x
        row = np.zeros(n); row[i]=1; rows.append(row); b.append(x)
        # <= y
        row = np.zeros(n); row[i]=1; rows.append(row); b.append(y)
        # <= 1-x
        row = np.zeros(n); row[i]=1; rows.append(row); b.append(1-x)
        # <= 1-y
        row = np.zeros(n); row[i]=1; rows.append(row); b.append(1-y)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = np.sum(radii)
    return radii, total
```

def spread_points(centers, n_iter=200, step=0.01):
    for _ in range(n_iter):
        # compute pairwise vectors
        for i in range(centers.shape[0]):
            for j in range(i+1, centers.shape[0]):

                dist = np.linalg.norm(diff)
                if dist < 0.2 and dist>1e-12:

```python
import numpy as np
from scipy.optimize import linprog
import random
import time

def _solve_radii(centers: np.ndarray):
    """
    Given centre coordinates (n,2) compute the optimal radii that maximise
    the sum of radii while satisfying non‑overlap and boundary constraints.
    Returns radii (n,) and the total sum of radii.
    """
    n = centers.shape[0]

    # ----- build linear constraints -----
    # pairwise distance constraints r_i + r_j <= d_ij
    rows = []
    rhs = []

    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            d = np.linalg.norm(centers[i] - centers[j])
            rows.append(row)
            rhs.append(d)

    # boundary constraints r_i <= x_i , y_i , 1-x_i , 1-y_i
    for i in range(n):
        x, y = centers[i]
        for b in (x, y, 1.0 - x, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            rhs.append(b)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    bounds = [(0.0, None) for _ in range(n)]

    # solve LP (small problem → method='highs' is very fast)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # fallback – all radii zero (should never happen for a feasible layout)
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = res.x
        total = float(radii.sum())

    return radii, total


def _initial_grid():
    """
    Create a 5x5 uniform grid (margin 0.1, spacing 0.2) and add one extra
    point close to a corner.  Returns an (26,2) array of centres.
    """
    margin = 0.1
    step = 0.2
    pts = []
    for i in range(5):
        for j in range(5):
            pts.append([margin + i * step, margin + j * step])
    # extra point in a far corner – very small radius, hardly interferes
    pts.append([0.01, 0.01])
    return np.array(pts[:26])


def _random_initial():
    """Uniform random points inside the unit square."""
    return np.random.rand(26, 2)


def _local_search(centers: np.ndarray,
                  max_iters: int = 5000,
                  init_step: float = 0.15,
                  init_temp: float = 0.05,
                  cooling: float = 0.9995):
    """
    Greedy / simulated‑annealing style local optimisation.
    Moves a single centre at a time, resolves the LP and keeps the move
    if the total radius increases (or with a small probability if it
    decreases, controlled by the temperature).
    """
    best_cent = centers.copy()
    best_rad, best_sum = _solve_radii(best_cent)

    step = init_step
    temp = init_temp
    n = best_cent.shape[0]

    for it in range(max_iters):
        # pick a centre to perturb
        idx = random.randrange(n)
        new_center = best_cent[idx] + np.random.uniform(-step, step, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)

        cand_cent = best_cent.copy()
        cand_cent[idx] = new_center

        cand_rad, cand_sum = _solve_radii(cand_cent)
        delta = cand_sum - best_sum

        # accept if better, otherwise with Metropolis probability
        if delta > 0.0 or random.random() < np.exp(delta / max(temp, 1e-12)):
            best_cent = cand_cent
            best_rad = cand_rad
            best_sum = cand_sum

        # annealing schedule
        temp *= cooling
        if (it + 1) % 500 == 0:
            step *= 0.9  # gradually reduce step size

    return best_cent, best_rad, best_sum


def run_packing():
    """
    Returns (centers, radii, sum_of_radii) for 26 circles packed inside
    the unit square.  The routine builds a reasonable initial layout,
    improves it with a local search, and keeps the best result over a few
    random restarts.
    """
    random.seed(0)
    np.random.seed(0)

    best_total = -1.0
    best_centers = None
    best_radii = None

    # ---- first try the structured grid ----
    grid_cent = _initial_grid()
    grid_rad, grid_sum = _solve_radii(grid_cent)
    if grid_sum > best_total:
        best_total = grid_sum
        best_centers = grid_cent.copy()
        best_radii = grid_rad.copy()

    # local optimisation starting from the grid
    cent_opt, rad_opt, sum_opt = _local_search(best_centers,
                                               max_iters=8000,
                                               init_step=0.2,
                                               init_temp=0.04,
                                               cooling=0.9994)
    if sum_opt > best_total:
        best_total = sum_opt
        best_centers = cent_opt
        best_radii = rad_opt

    # ---- a few completely random restarts ----
    for _ in range(8):
        rnd_cent = _random_initial()
        rad, s = _solve_radii(rnd_cent)
        if s > best_total:
            best_total, best_centers, best_radii = s, rnd_cent, rad

        # short local search from this random start
        cent_opt, rad_opt, sum_opt = _local_search(rnd_cent,
                                                   max_iters=4000,
                                                   init_step=0.25,
                                                   init_temp=0.06,
                                                   cooling=0.9993)
        if sum_opt > best_total:
            best_total = sum_opt
            best_centers = cent_opt
            best_radii = rad_opt

    # Ensure the result respects the required shape
    best_centers = np.asarray(best_centers).reshape(26, 2)
    best_radii = np.asarray(best_radii).reshape(26)

    # Final sanity check (optional, can be removed in production)
    # The provided validation routine will be used by the scorer.
    return best_centers, best_radii, float(best_total)
