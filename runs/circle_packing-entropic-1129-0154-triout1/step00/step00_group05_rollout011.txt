def solve_lp(centers):
    import numpy as np
    from scipy.optimize import linprog

    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Bounds for each radius
    boundary = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]), np.minimum(centers[:,1], 1-centers[:,1]))
    bounds = [(0, b) for b in boundary]

    # Prepare A_ub and b_ub
    # Pairwise constraints
    pairs = []
    for i in range(n):
        for j in range(i+1, n):
            pairs.append((i,j))
    m = len(pairs)
    A = np.zeros((m, n))
    b = np.empty(m)
    for idx, (i,j) in enumerate(pairs):
        A[idx,i] = 1
        A[idx,j] = 1
        b[idx] = dists[i,j]

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        radii = np.zeros(n)
    else:
        radii = np.maximum(res.x, 0)
    return radii, np.sum(radii)

def generate_maximin(num_points, num_candidates=5000):
    import numpy as np
    points = []
    # Start with a random point at center
    points.append(np.array([0.5,0.5]))
    while len(points) < num_points:
        # sample random candidates
        cand = np.random.rand(num_candidates, 2)
        # compute min distance to existing points and boundary
        min_vals = np.full(num_candidates, np.inf)
        for p in points:
            d = np.linalg.norm(cand - p, axis=1) - 0 # we don't have radii yet
            min_vals = np.minimum(min_vals, d)
        # boundary distance
        bound = np.minimum(np.minimum(cand[:,0], 1-cand[:,0]), np.minimum(cand[:,1], 1-cand[:,1]))
        min_vals = np.minimum(min_vals, bound)
        # Choose candidate with max min_vals
        idx = np.argmax(min_vals)
        points.append(cand[idx])
    return np.array(points)

def generate_hex_lattice(spacing):
    import numpy as np
    points = []
    row = 0
    y = spacing/2
    while y <= 1 - spacing/2 + 1e-12:
        offset = (row % 2) * spacing/2
        x = spacing/2 + offset
        while x <= 1 - spacing/2 + 1e-12:
            points.append([x, y])
            x += spacing
        row += 1
        y += spacing * np.sqrt(3)/2
    return np.array(points)

import numpy as np
from scipy.optimize import linprog

def solve_lp(centers):
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))
    # radius upper bounds from sides
    max_r = np.minimum(np.minimum(centers[:,0], 1-centers[:,0]),
                       np.minimum(centers[:,1], 1-centers[:,1]))
    bounds = [(0.0, float(b)) for b in max_r]
    # pairwise constraints ri + rj <= dist
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # If the solver fails for any reason we return zero radii.
        radii = np.zeros(n)
    else:
        radii = np.maximum(res.x, 0.0)
    return radii, float(np.sum(radii))

    points = []
    # start with centre
    points.append(np.array([0.5, 0.5]))
    while len(points) < num_points:
        cand = rng.random((n_candidates, 2))
        # compute distance to existing points
        min_dist = np.full(n_candidates, np.inf)
        for p in points:
            d = np.linalg.norm(cand - p, axis=1)
            min_dist = np.minimum(min_dist, d)

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear program that, for a fixed set of centres, gives the
# largest possible sum of radii while respecting the non‑overlap
# and boundary constraints.
# ------------------------------------------------------------
def solve_lp(centers: np.ndarray):
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # radius cannot exceed distance to the closest side
    max_r = np.minimum(np.minimum(centers[:, 0], 1.0 - centers[:, 0]),
                       np.minimum(centers[:, 1], 1.0 - centers[:, 1]))
    bounds = [(0.0, float(b)) for b in max_r]

    # constraints: r_i + r_j <= d_ij   for every i < j
    m = n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)
    k = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[k, i] = 1.0
            A[k, j] = 1.0
            b[k] = dists[i, j]
            k += 1

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        radii = np.zeros(n)
    else:
        radii = np.maximum(res.x, 0.0)

    return radii, float(np.sum(radii))


# ------------------------------------------------------------
# Produce a set of points that are as far apart from each other
# (and from the sides) as possible – a simple greedy “maximin”.
# ------------------------------------------------------------
def generate_maximin(num_points: int, rng: np.random.Generator,
                    n_candidates: int = 5000) -> np.ndarray:
    pts = []
    pts.append(np.array([0.5, 0.5]))                     # seed with the centre
    while len(pts) < num_points:
        cand = rng.random((n_candidates, 2))
        # distance to already placed points
        dmin = np.full(n_candidates, np.inf)
        for p in pts:
            d = np.linalg.norm(cand - p, axis=1)
            dmin = np.minimum(dmin, d)
        # also consider distance to the nearest side
        side = np.minimum(np.minimum(cand[:, 0], 1.0 - cand[:, 0]),
                          np.minimum(cand[:, 1], 1.0 - cand[:, 1]))
        dmin = np.minimum(dmin, side)
        idx = np.argmax(dmin)
        pts.append(cand[idx])
    return np.vstack(pts)

# Simple hexagonal lattice generator (spacing = centre‑to‑centre).
# ------------------------------------------------------------
def generate_hex_lattice(spacing: float) -> np.ndarray:
    pts = []
    row = 0
    y = spacing / 2.0
    sqrt3_2 = np.sqrt(3.0) / 2.0

        while x <= 1.0 - spacing / 2.0 + 1e-12:
            pts.append([x, y])
            x += spacing
        row += 1
        y += spacing * sqrt3_2
    return np.array(pts)


# ------------------------------------------------------------
# Very light hill‑climbing: move a single centre a little,
# accept the move if the total sum of radii grows.
# ------------------------------------------------------------
def hill_climb(start_centers: np.ndarray,
               rng: np.random.Generator,
               max_iters: int = 800,
               step: float = 0.05):
    centres = start_centers.copy()
    radii, best_val = solve_lp(centres)
    n = centres.shape[0]

    for _ in range(max_iters):
        i = rng.integers(0, n)
        delta = rng.uniform(-step, step, size=2)
        new_c = centres[i] + delta
        new_c = np.clip(new_c, 0.0, 1.0)
        trial = centres.copy()
        trial[i] = new_c

        if trial_val > best_val + 1e-12:
            centres = trial
            radii = trial_r
            best_val = trial_val

    return centres, radii, best_val
