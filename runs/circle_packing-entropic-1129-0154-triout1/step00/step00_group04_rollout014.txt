import numpy as np
from scipy.optimize import minimize

N = 26

def unpack(v):
    xs = v[:N]
    ys = v[N:2*N]
    rs = v[2*N:]
    return xs, ys, rs

def constraint_all(v):
    xs, ys, rs = unpack(v)
    cons = []
    # boundary constraints
    cons.append(xs - rs)               # x_i >= r_i
    cons.append(1 - xs - rs)           # x_i + r_i <= 1
    cons.append(ys - rs)               # y_i >= r_i
    cons.append(1 - ys - rs)           # y_i + r_i <= 1
    # radii non-negative (bounds already enforce but keep)
    cons.append(rs)                    # rs >=0
    # pairwise constraints
    # Compute pairwise distances
    # Use loops for each pair
    for i in range(N):
        for j in range(i+1, N):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            cons.append(dx*dx + dy*dy - (rs[i] + rs[j])**2)
    return np.concatenate(cons)

def objective(v):
    # maximize sum of radii => minimize negative sum
    return -np.sum(v[2*N:])

def run_optimizer(seed, max_iter=1000):
    rng = np.random.default_rng(seed)
    # generate initial positions
    # start from grid plus jitter
    # 5x5 grid (25 points)
    grid_coords = np.linspace(0.1, 0.9, 5)
    xs_grid, ys_grid = np.meshgrid(grid_coords, grid_coords)
    xs0 = xs_grid.ravel()
    ys0 = ys_grid.ravel()
    # extra point near centre
    xs0 = np.append(xs0, 0.5)
    ys0 = np.append(ys0, 0.5)
    # Jitter each coordinate a bit
    jitter = rng.uniform(-0.02, 0.02, size=xs0.shape)
    xs0 = np.clip(xs0 + jitter, 0.0, 1.0)
    jitter = rng.uniform(-0.02, 0.02, size=ys0.shape)
    ys0 = np.clip(ys0 + jitter, 0.0, 1.0)
    # Radii initial small
    rs0 = np.full(N, 0.02)
    v0 = np.concatenate([xs0, ys0, rs0])
    # bounds
    bounds = [(0.0, 1.0)] * (2*N) + [(0.0, 1.0)] * N
    # define constraint dict
    cons = {'type': 'ineq', 'fun': constraint_all}
    res = minimize(objective, v0, method='SLSQP', bounds=bounds,
                   constraints=cons, options={'maxiter': max_iter, 'ftol': 1e-12, 'disp': False})
    return res

def run_packing():
    best_sum = -np.inf
    best_res = None
    # try several seeds
    for seed in range(5):
        res = run_optimizer(seed, max_iter=1500)
        if not res.success:
            continue
        # extract radii sum
        xs, ys, rs = unpack(res.x)
        total = np.sum(rs)
        if total > best_sum:
            best_sum = total
            best_res = (xs, ys, rs)
    if best_res is None:
        # fallback: simple grid solution
        grid_coords = np.linspace(0.1, 0.9, 5)
        xs_grid, ys_grid = np.meshgrid(grid_coords, grid_coords)
        xs = xs_grid.ravel()
        ys = ys_grid.ravel()
        # add extra at centre
        xs = np.append(xs, 0.5)
        ys = np.append(ys, 0.5)
        # radii: 0.1 for first 25, small for last
        rs = np.full(N, 0.1)
        # reduce last radius to fit
        rs[-1] = 0.04
        best_sum = np.sum(rs)
        best_res = (xs, ys, rs)
    xs, ys, rs = best_res
    # Clip small negatives
    rs = np.maximum(rs, 0.0)
    centers = np.column_stack((xs, ys))
    sum_radii = float(np.sum(rs))
    return centers, rs, sum_radii

def compute_optimal_radii(xs, ys):
    n = len(xs)
    # compute distance matrix
    D = np.sqrt(((xs[:,None] - xs[None,:])**2) + ((ys[:,None] - ys[None,:])**2))
    # Build A_ub: each row has shape n
    # We'll generate list rows and b's.
    rows = []
    b = []
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        rows.append(e.copy())
        b.append(xs[i])
        rows.append(e.copy())
        b.append(1 - xs[i])
        rows.append(e.copy())
        b.append(ys[i])
        rows.append(e.copy())
        b.append(1 - ys[i])
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            b.append(D[i,j])
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback set radii zero
        rs = np.zeros(n)
    else:
        rs = res.x
    return rs, float(np.sum(rs))

def hill_climb(seed, max_iter=2000, step0=0.1):
    rng = np.random.default_rng(seed)
    # initial positions: grid with jitter
    base = np.linspace(0.1, 0.9, 5)
    xs_grid, ys_grid = np.meshgrid(base, base)
    xs = xs_grid.ravel()
    ys = ys_grid.ravel()
    xs = np.append(xs, 0.5)
    ys = np.append(ys, 0.5)
    # jitter
    xs += rng.uniform(-0.02, 0.02, size=xs.shape)
    ys += rng.uniform(-0.02, 0.02, size=ys.shape)
    xs = np.clip(xs, 0, 1)
    ys = np.clip(ys, 0, 1)
    # compute initial LP
    rs, total = compute_optimal_radii(xs, ys)
    best_total = total
    best_xs = xs.copy()
    best_ys = ys.copy()
    best_rs = rs.copy()
    cur_xs = xs.copy()
    cur_ys = ys.copy()
    cur_total = total
    step = step0
    for it in range(max_iter):
        # propose move
        i = rng.integers(N)
        new_x = cur_xs[i] + rng.uniform(-step, step)
        new_y = cur_ys[i] + rng.uniform(-step, step)
        new_x = np.clip(new_x, 0, 1)
        new_y = np.clip(new_y, 0, 1)
        xs_new = cur_xs.copy()
        ys_new = cur_ys.copy()
        xs_new[i] = new_x
        ys_new[i] = new_y
        rs_new, total_new = compute_optimal_radii(xs_new, ys_new)
        if total_new > cur_total + 1e-9:
            # accept move
            cur_xs = xs_new
            cur_ys = ys_new
            cur_total = total_new
            # update best if improve
            if total_new > best_total:
                best_total = total_new
                best_xs = xs_new.copy()
                best_ys = ys_new.copy()

import numpy as np
from scipy.optimize import linprog

# number of circles
N = 26
EPS = 1e-9  # tiny shrink to guarantee feasibility after LP


def _unpack(v):
    """Split the big variable vector into x, y, r."""
    xs = v[:N]
    ys = v[N:2 * N]
    rs = v[2 * N:]
    return xs, ys, rs


def _constraint_all(v):
    """
    Gather *all* inequality constraints for SLSQP.
    Each entry must be ≥ 0.
    """
    xs, ys, rs = _unpack(v)
    cons = []

    # boundary constraints
    cons.append(xs - rs)           # x_i - r_i ≥ 0
    cons.append(1.0 - xs - rs)     # 1 - x_i - r_i ≥ 0
    cons.append(ys - rs)           # y_i - r_i ≥ 0
    cons.append(1.0 - ys - rs)     # 1 - y_i - r_i ≥ 0

    # radii non‑negative (bounds are already set, but add robustness)
    cons.append(rs)

    # pairwise non‑overlap:   (dx)^2 + (dy)^2 - (r_i+r_j)^2 ≥ 0
    for i in range(N):
        for j in range(i + 1, N):
            dx = xs[i] - xs[j]
            dy = ys[i] - ys[j]
            cons.append(dx * dx + dy * dy - (rs[i] + rs[j]) ** 2)

    return np.concatenate(cons)


def _objective(v):
    """We maximise sum(r_i) ⇒ minimise -sum(r_i)."""
    return -np.sum(v[2 * N:])


def _optimal_radii(xs, ys):
    """
    For a given set of centres solve the linear programme that yields the
    maximal radii.
    Returns (radii, sum_of_radii).
    """
    # pairwise distances
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    D = np.sqrt(dx * dx + dy * dy)

    # constraint matrix construction
    rows = []
    b = []

    # boundary constraints r_i ≤ x_i , 1-x_i , y_i , 1-y_i
    for i in range(N):
        e = np.zeros(N)
        e[i] = 1.0
        rows.append(e)            # r_i ≤ x_i
        b.append(xs[i])
        rows.append(e)            # r_i ≤ 1-x_i
        b.append(1.0 - xs[i])
        rows.append(e)            # r_i ≤ y_i
        b.append(ys[i])
        rows.append(e)            # r_i ≤ 1-y_i
        b.append(1.0 - ys[i])

    # pairwise constraints r_i + r_j ≤ d_ij
    for i in range(N):
        for j in range(i + 1, N):
            e = np.zeros(N)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            b.append(D[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(b)

    # maximise sum(r_i)  ⇒ minimise -sum(r_i)
    c = -np.ones(N)
    bounds = [(0.0, None)] * N

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback – shouldn't happen, but keep a feasible (zero) solution
        radii = np.zeros(N)
        total = 0.0
    else:
        radii = res.x
        total = float(np.sum(radii))

    # tiny shrink to stay inside validator tolerances
    radii = np.maximum(radii - EPS, 0.0)
    total = float(np.sum(radii))
    return radii, total


def _initial_grid(rng):
    """
    Produce a jittered 5×5 grid (25 points) plus one centre point.
    This is a good starting layout.
    """
    base = np.linspace(0.1, 0.9, 5)
    xs_grid, ys_grid = np.meshgrid(base, base)
    xs = xs_grid.ravel()
    ys = ys_grid.ravel()
    xs = np.append(xs, 0.5)
    ys = np.append(ys, 0.5)

    # small jitter – kept inside the square
    xs += rng.uniform(-0.02, 0.02, size=xs.shape)
    ys += rng.uniform(-0.02, 0.02, size=ys.shape)
    xs = np.clip(xs, 0.0, 1.0)
    ys = np.clip(ys, 0.0, 1.0)
    return xs, ys


def _hill_climb(seed, max_iter=2000, step0=0.12):
    """
    Simple stochastic hill‑climbing on the centre positions.
    At each step a random circle is moved a little; the LP is solved;
    the move is kept only if the total sum of radii increases.
    """
    rng = np.random.default_rng(seed)

    # start from the jittered grid
    xs, ys = _initial_grid(rng)

    # radii for the initial layout
    rs, cur_total = _optimal_radii(xs, ys)

    best_xs, best_ys, best_rs, best_total = xs.copy(), ys.copy(), rs.copy(), cur_total

    step = step0
    for it in range(max_iter):
        # propose a move for a random circle
        i = rng.integers(N)
        new_x = xs[i] + rng.uniform(-step, step)
        new_y = ys[i] + rng.uniform(-step, step)
        new_x = np.clip(new_x, 0.0, 1.0)
        new_y = np.clip(new_y, 0.0, 1.0)

        xs_new = xs.copy()
        ys_new = ys.copy()
        xs_new[i] = new_x
        ys_new[i] = new_y

        rs_new, total_new = _optimal_radii(xs_new, ys_new)

        # accept only improving moves
        if total_new > cur_total + 1e-9:
            xs, ys = xs_new, ys_new
            cur_total = total_new
            if total_new > best_total:
                best_xs, best_ys, best_rs, best_total = xs_new.copy(), ys_new.copy(), rs_new.copy(), total_new

        # slowly shrink the random step
        step *= 0.9995

    return best_xs, best_ys, best_rs, best_total

    best_radii = np.maximum(best_radii, 0.0)
