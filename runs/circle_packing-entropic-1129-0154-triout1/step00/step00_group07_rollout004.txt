def solve_lp(centers):
    n = centers.shape[0]
    # compute distances
    x = centers[:,0]
    y = centers[:,1]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]  # shape (n,n,2)
    distances = np.sqrt(np.sum(diff**2, axis=2))
    # build constraints
    # A_ub shape (m, n)
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)

    # pointer
    k = 0
    # wall constraints:
    for i in range(n):
        A[k, i] = 1.0
        b[k] = x[i]
        k+=1
        A[k,i]=1.0
        b[k] = 1-x[i]
        k+=1
        A[k,i]=1.0
        b[k] = y[i]
        k+=1
        A[k,i]=1.0
        b[k] = 1-y[i]
        k+=1

    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            A[k,i] = 1.0
            A[k,j] = 1.0
            b[k] = distances[i,j]
            k+=1

    # linear program
    c = -np.ones(n)   # minimize -sum r_i = maximize sum r_i
    # bounds for r_i: (0, None)
    bounds = [(0.0, None)] * n
    # use high-level method
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In case of failure, maybe set radii to zero.
        return 0.0, np.zeros(n)
    radii = res.x
    sum_r = np.sum(radii)
    return sum_r, radii

import numpy as np

from scipy.optimize import linprog, differential_evolution

def solve_lp(centers):
    # compute distances
    n = centers.shape[0]
    x = centers[:,0]
    y = centers[:,1]
    # distances matrix
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    # wall constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = x[i]
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x[i]
        row += 1
        A[row, i] = 1.0
        b[row] = y[i]
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y[i]
        row += 1
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i,j]
            row += 1
    # Solve LP
    c = -np.ones(n)    # maximize sum
    bounds = [(0.0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # In case of failure, use zero radii
        return 0.0, np.zeros(n)
    radii = result.x
    sum_r = np.sum(radii)
    return sum_r, radii

def run_packing():
    import numpy as np
    from scipy.optimize import differential_evolution, linprog

    n = 26
    # random seed for reproducibility
    np.random.seed(0)

    # Define objective for DE
    def obj_flat(flat):
        centers = flat.reshape((n, 2))
        sum_r, _ = solve_lp(centers)
        return -sum_r

    bounds = [(0.0, 1.0)] * (2*n)

    # Use DE with limited generations
    result = differential_evolution(obj_flat, bounds, maxiter=30, popsize=15, polish=False, seed=1, recombination=0.7, init='latinhypercube')

    best_sum_current = best_sum

    # Simulated annealing style local search
    steps = 3000
    for iter_idx in range(steps):
        # Choose random index to move
        i = np.random.randint(n)
        # step size decreasing

            best_centers_current = new_centers
            best_radii_current = radii

    # Return final best
    total_sum = float(best_sum_current)
    return best_centers_current, best_radii_current, total_sum

def solve_lp(centers):
    n = centers.shape[0]
    # Distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    D = np.sqrt(np.sum(diff**2, axis=2))
    # Build A and b
    num_constraints = 4 * n + n * (n - 1) // 2
    A = np.zeros((num_constraints, n), dtype=np.float64)
    b = np.zeros(num_constraints, dtype=np.float64)
    row = 0
    for i in range(n):
        A[row, i] = 1.0
        b[row] = centers[i, 0]  # left boundary
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 0]  # right boundary
        row += 1
        A[row, i] = 1.0
        b[row] = centers[i, 1]  # bottom
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 1]  # top
        row += 1
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = D[i, j]
            row += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # At worst return zeros
        return 0.0, np.zeros(n)
    radii = res.x
    sum_r = np.sum(radii)
    return sum_r, radii

The problem is a non‑convex optimization:  
choose the centers \((x_i,y_i)\) of 26 circles inside the unit square and their radii
\(r_i\) to maximise \(\sum_i r_i\) subject to  

* \(0\le x_i\pm r_i\le1,\;0\le y_i\pm r_i\le1\)  (circles stay inside)  
* \(\|(x_i,y_i)-(x_j,y_j)\|\ge r_i+r_j\)        (no overlap).

For a *fixed* set of centers the radii can be chosen optimally by a **linear program**:

```
maximise   Σ r_i
subject to  r_i ≤ x_i,
            r_i ≤ 1‑x_i,
            r_i ≤ y_i,
            r_i ≤ 1‑y_i,
            r_i + r_j ≤ d_ij   (d_ij = distance between centres i and j)
            r_i ≥ 0
```

Thus the whole task becomes: find a placement of the 26 centres that gives the
largest LP‑value.  
The search over centre positions is performed with a two‑stage, derivative‑free
optimizer:

1. **Global search** – `scipy.optimize.differential_evolution` explores the
   52‑dimensional space (26 × 2) with modest population size and iteration count.
   For each candidate it builds the LP described above and returns the negative
   total radius (DE minimises).

2. **Local refinement** – starting from the best point found by DE we perform a
   simple hill‑climbing / simulated‑annealing style walk: repeatedly pick one
   centre, move it by a small random offset (shrinking the step size over time),
   resolve the LP and keep the move if the total radius improves.

Both stages are cheap because the LP has only 26 variables and at most  
\(4·26+{26\choose2}=429\) linear constraints.  
The algorithm stops early as soon as a sum of radii ≥ 2.636 is obtained or after a
time‑limited number of attempts.  The final radii are taken directly from the LP,
so the returned configuration is guaranteed to satisfy all geometric constraints
to the strict tolerance required by the validator.

Below is a self‑contained implementation that follows the rules (no file I/O,
no lambdas, all helpers are top‑level).  It returns the centre coordinates,
the radii, and the achieved sum of radii.

```python
import numpy as np
import time
from scipy.optimize import linprog, differential_evolution

# ----------------------------------------------------------------------
# Problem size (fixed, but kept as a global for the helper functions)
N_CIRCLES = 26
TARGET_SUM = 2.636  # the value we would like to reach


def _max_sum_radii(centers: np.ndarray) -> tuple[float, np.ndarray]:
    """
    Solve the linear programme for a *fixed* set of centres.
    Returns the optimal total radius and the vector of radii.
    """
    n = centers.shape[0]
    # distances between centres
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # number of linear constraints:
    # 4 per circle (left, right, bottom, top) + one per unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.zeros(m, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(n):
        # left side   x_i - r_i >= 0   -> r_i <= x_i
        A[row, i] = 1.0
        b[row] = centers[i, 0]
        row += 1
        # right side  x_i + r_i <= 1 -> r_i <= 1 - x_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 0]
        row += 1
        # bottom      y_i - r_i >= 0   -> r_i <= y_i
        A[row, i] = 1.0
        b[row] = centers[i, 1]
        row += 1
        # top        y_i + r_i <= 1 -> r_i <= 1 - y_i
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i, 1]
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  <=> minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    # use the efficient HiGHS solver that ships with recent SciPy
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # In the unlikely event of failure fall back to zeros
        return 0.0, np.zeros(n, dtype=np.float64)

    radii = res.x
    total = float(np.sum(radii))
    return total, radii


def _objective(flat: np.ndarray) -> float:
    """
    DE calls this; it receives a flat array of length 2*N_CIRCLES.
    Returns the *negative* total radius because DE performs minimisation.
    """
    centers = flat.reshape((N_CIRCLES, 2))
    total, _ = _max_sum_radii(centers)
    return -total


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Finds a placement of 26 circles in the unit square that maximises the sum of
    radii.  Returns (centres, radii, sum_of_radii).
    """
    np.random.seed(0)               # reproducibility
    start_time = time.time()
    best_total = -np.inf
    best_centers = None
    best_radii = None

    # search parameters – they are modest enough to finish well within typical
    # time limits while still giving a good chance of hitting the target.
    max_attempts = 5            # number of independent DE runs
    de_maxiter = 15            # generations per DE run
    de_popsize = 10            # population factor (popsize*dim individuals)
    local_steps = 2500         # hill‑climbing steps after each DE run

    bounds = [(0.0, 1.0)] * (2 * N_CIRCLES)

    attempt = 0
    while attempt < max_attempts and best_total < TARGET_SUM:
        # ------------------------------------------------------------------
        # 1) Global search with Differential Evolution
        # ------------------------------------------------------------------
        result = differential_evolution(
            _objective,
            bounds,
            maxiter=de_maxiter,
            popsize=de_popsize,
            polish=False,
            seed=attempt + 1,
            init="latinhypercube",
            disp=False,
        )
        centres = result.x.reshape((N_CIRCLES, 2))
        total, radii = _max_sum_radii(centres)
        if total > best_total:
            best_total = total
            best_centers = centres
            best_radii = radii

        # ------------------------------------------------------------------
        # 2) Local refinement – simple stochastic hill climbing
        # ------------------------------------------------------------------
        for step in range(local_steps):
            # pick a random circle to move
            i = np.random.randint(N_CIRCLES)

            # step size slowly shrinks
            scale = 0.2 * (1.0 - step / local_steps)
            proposal = best_centers[i] + (np.random.rand(2) - 0.5) * 2.0 * scale
            proposal = np.clip(proposal, 0.0, 1.0)

            new_centers = best_centers.copy()
            new_centers[i] = proposal

            new_total, new_radii = _max_sum_radii(new_centers)

            if new_total > best_total:
                best_total = new_total
                best_centers = new_centers
                best_radii = new_radii

        attempt += 1
        # stop early if we already reached the required quality
        if best_total >= TARGET_SUM:
            break

    # Safety net – should never happen, but ensures a valid return value.
    if best_centers is None:
        best_centers = np.zeros((N_CIRCLES, 2), dtype=np.float64)
        best_radii = np.zeros(N_CIRCLES, dtype=np.float64)
        best_total = 0.0

    return best_centers, best_radii, float(best_total)
```
