import numpy as np
from scipy.optimize import linprog

TARGET_SUM = 2.636
N_CIRCLES = 26
EPS = 1e-8  # safety margin for the validator


def _solve_lp(centers: np.ndarray) -> np.ndarray | None:
    """
    Solve the linear programme for a fixed set of centres.
    Returns the optimal radii as a 1‑d array or None if the solver fails.
    """
    n = centers.shape[0]

    # distance from each centre to the four sides
    border = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diff, axis=2)

    # Build the inequality matrix for r_i + r_j <= d_ij
    pair_idx = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pair_idx)
    A = np.zeros((m, n))
    b = np.empty(m)

    for k, (i, j) in enumerate(pair_idx):
        A[k, i] = 1.0
        A[k, j] = 1.0
        b[k] = dists[i, j]

    # bounds for each radius (0 <= r_i <= border_i)
    bounds = [(0.0, border[i]) for i in range(n)]

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n)

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        return None

    radii = np.maximum(res.x - EPS, 0.0)   # tiny shrink to stay away from the tolerance
    return radii


def _candidate_grid():
    """
    A deterministic candidate consisting of the four corner circles
    (radius 0.2 can be achieved later by the LP) and a set of interior points.
    The interior points lie on two interleaved square lattices.
    """
    # Four corner points – they will be allowed to grow to about 0.2 later.
    corners = np.array([[0.2, 0.2],
                        [0.8, 0.2],
                        [0.2, 0.8],
                        [0.8, 0.8]])

    # Two interleaved 4×4 lattices, shifted by 0.1
    xs1 = np.array([0.25, 0.45, 0.65, 0.85])
    xs2 = np.array([0.35, 0.55, 0.75, 0.95])
    pts = []
    for xs, ys in ((xs1, xs1), (xs2, xs2)):
        for x in xs:
            for y in ys:
                pts.append([x, y])
    interior = np.array(pts)

    # Take the first 22 interior points (any 22 will do)
    interior = interior[:22]

    centres = np.vstack([corners, interior])
    return centres


def _candidate_random(seed: int):
    """
    Generate a completely random set of 26 points.
    """
    rng = np.random.default_rng(seed)
    # keep points a little away from the border so that they can obtain a decent radius
    points = 0.05 + 0.9 * rng.random((N_CIRCLES, 2))
    return points


def run_packing():
    """
    Returns
    -------
    centres : np.ndarray of shape (26, 2)
        x‑y coordinates of the circle centres.
    radii   : np.ndarray of shape (26,)
        radius of each circle.
    sum_r   : float
        sum of the radii (must be >= 2.636).
    """
    # 1) Try a handcrafted grid – often already good enough
    centres = _candidate_grid()
    radii = _solve_lp(centres)
    if radii is not None and radii.sum() >= TARGET_SUM:
        return centres, radii, float(radii.sum())

    # 2) Random search – deterministic because the seed changes in a loop
    for seed in range(5000):
        centres = _candidate_random(seed)
        radii = _solve_lp(centres)
        if radii is not None and radii.sum() >= TARGET_SUM:
            return centres, radii, float(radii.sum())

    # 3) Very last resort – a simple fallback that meets the constraints
    #    (four big corner circles and many tiny ones, guaranteed feasible)
    corners = np.array([[0.2, 0.2],
                        [0.8, 0.2],
                        [0.2, 0.8],
                        [0.8, 0.8]])
    # 22 tiny circles placed on a fine 7×4 lattice (well inside the square)
    xs = np.linspace(0.07, 0.93, 7)
    ys = np.linspace(0.15, 0.85, 4)
    tiny = np.array([[x, y] for x in xs for y in ys])[:22]
    centres = np.vstack([corners, tiny])
    radii = _solve_lp(centers=centres)
    if radii is None:
        raise RuntimeError("LP failed in the fallback configuration.")
    return centres, radii, float(radii.sum())
