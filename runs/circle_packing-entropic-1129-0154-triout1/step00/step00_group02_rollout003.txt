```
def solve_lp(centers):
    n = len(centers)
    # Compute distances matrix
    d = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)  # n x n
    # Build constraints
    rows = []
    rhs = []
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        rows.append(np.eye(1, n, i).reshape(-1))
        rhs.append(x)
        rows.append(np.eye(1, n, i).reshape(-1))
        rhs.append(1 - x)
        rows.append(np.eye(1, n, i).reshape(-1))
        rhs.append(y)
        rows.append(np.eye(1, n, i).reshape(-1))
        rhs.append(1 - y)
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            # Add constraint r_i + r_j <= d[i][j]
            a = np.zeros(n)
            a[i] = 1
            a[j] = 1
            rows.append(a)
            rhs.append(d[i,j])
    A = np.vstack(rows)
    b = np.array(rhs)
    c = -np.ones(n)  # maximize sum => minimize -sum
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback maybe small epsilon
        return np.zeros(n), 0.0
    radii = res.x
    return radii, radii.sum()
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    n = 26
    rng = np.random.default_rng(12345)

    # Initialize positions: use a perturbed grid to avoid duplicate points
    # Start with 5x5 grid (25 points) and additional random point.
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    init = np.array([[x, y] for y in ys for x in xs])
    # add a random point not too close to others
    extra = rng.uniform(0.2, 0.8, size=(1,2))
    centers = np.vstack([init, extra])

    # Ensure shape (26,2)
    # Solve LP for radii
    def solve_radii(centers):
        n = centers.shape[0]
        dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
        # Build A_ub and b_ub
        # Number of constraints:
        # 4 per circle + pair constraints
        m = 4*n + n*(n-1)//2
        A = np.zeros((m, n))
        b = np.zeros(m)
        row = 0
        # boundary constraints
        for i in range(n):
            x, y = centers[i]
            # r_i <= x
            A[row, i] = 1
            b[row] = x
            row += 1
            # r_i <= 1 - x
            A[row, i] = 1
            b[row] = 1 - x
            row += 1
            # r_i <= y
            A[row, i] = 1
            b[row] = y
            row += 1
            # r_i <= 1 - y
            A[row, i] = 1
            b[row] = 1 - y
            row += 1
        # pair constraints
        for i in range(n):
            for j in range(i+1, n):
                A[row, i] = 1
                A[row, j] = 1
                b[row] = dists[i, j]
                row += 1
        c = -np.ones(n)  # maximize sum
        bounds = [(0, None)] * n
        res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
        if not res.success:
            # fallback to zeros
            return np.zeros(n), 0.0
        radii = res.x
        return radii, radii.sum()

    best_radii, best_sum = solve_radii(centers)
    best_centers = centers.copy()

    # Simulated annealing parameters
    temperature = 0.05
    cooling = 0.9995
    step_scale = 0.1
    n_iter = 5000

    for it in range(n_iter):
        # propose new centers by perturbing a random subset
        new_centers = best_centers.copy()
        # Choose a random number of points to move (1-3)
        k = rng.integers(1, 4)
        idxs = rng.choice(n, size=k, replace=False)
        # Apply Gaussian perturbations
        delta = rng.normal(scale=step_scale, size=(k,2))
        new_centers[idxs] += delta
        # Clip to [0,1]
        new_centers = np.clip(new_centers, 0.0, 1.0)
        # recompute radii
        new_radii, new_sum = solve_radii(new_centers)
        # acceptance
        if new_sum > best_sum or rng.random() < np.exp((new_sum - best_sum) / temperature):
            best_sum = new_sum
            best_centers = new_centers.copy()
            best_radii = new_radii.copy()
        # update temperature
        temperature *= cooling
        # gradually reduce step scale
        step_scale *= cooling

    # final radii for best_centers
    final_radii, final_sum = solve_radii(best_centers)

    return best_centers, final_radii, final_sum
```

```
def solve_radii(centers):
    n = centers.shape[0]
    xs = centers[:,0]
    ys = centers[:,1]
    # Pair distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
```

**Goal:** Produce a feasible packing of 26 circles inside the unit square whose total radius sum is as large as possible (target ≈ 2.636).

**Key observations**

* For a *fixed* set of centre positions the maximal admissible radii are obtained from a **linear program**  

  \[
  \max\; \sum_i r_i\qquad
  \text{s.t.}\;
  \begin{cases}
      r_i \ge 0,\\[2pt]
      r_i \le x_i,\; r_i \le 1-x_i,\; r_i \le y_i,\; r_i \le 1-y_i,\\[2pt]

  \end{cases}
  \]
  where \(d_{ij}\) is the Euclidean distance of centres \(i\) and \(j\).

* Thus the problem reduces to **optimising the centre positions**; for every trial set of centres we can instantly compute the optimal radii by solving the LP.

* A simple *simulated‑annealing / hill‑climbing* on the centre coordinates works well:

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple


def _solve_lp(centers: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    For a given set of centre coordinates solve the linear program that
    maximises the sum of radii under the non‑overlap and boundary constraints.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        Optimal radii (may contain tiny negatives due to numerical noise).
    total : float
        Sum of the radii.
    """
    n = centers.shape[0]
    xs = centers[:, 0]
    ys = centers[:, 1]

    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # Build inequality matrix A_ub * r <= b_ub
    #   * 4 bounds per circle
    #   * n*(n-1)//2 pair constraints
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = xs[i]          # r_i <= x_i
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]    # r_i <= 1 - x_i
        row += 1
        A[row, i] = 1.0
        b[row] = ys[i]          # r_i <= y_i
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]    # r_i <= 1 - y_i
        row += 1

    # non‑overlap constraints: r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # objective: maximise sum(r)  => minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero (still feasible)
        radii = np.zeros(n, dtype=np.float64)
        return radii, 0.0

    radii = res.x
    total = float(radii.sum())
    return radii, total


def _clip_radii(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
    """
    Clip radii to the exact geometric limits to eliminate possible
    tiny numerical overshoots.
    """
    xs = centers[:, 0]
    ys = centers[:, 1]
    max_allowed = np.minimum.reduce([xs, 1.0 - xs, ys, 1.0 - ys])
    radii = np.minimum(radii, max_allowed)
    radii = np.maximum(radii, 0.0)
    return radii


def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Produces 26 circles inside the unit square that maximise the sum of
    radii (heuristically).  Returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(42)  # deterministic seed

    n = 26

    # ---- 1. Initialise a reasonable pattern (5×5 grid + one extra point) ----
    # Grid spacing that yields radius 0.1 when circles are tangent.
    grid_vals = np.linspace(0.1, 0.9, 5)
    grid = np.array([[x, y] for y in grid_vals for x in grid_vals])
    # grid currently has 25 points
    extra = rng.uniform(0.2, 0.8, size=(1, 2))   # a well‑inside point
    centres = np.vstack([grid, extra])           # shape (26,2)

    # ---- 2. Evaluate initial LP ----
    radii, best_sum = _solve_lp(centres)
    radii = _clip_radii(centres, radii)
    best_radii = radii.copy()
    best_centres = centres.copy()

    # ---- 3. Simulated‑annealing / hill climbing on centre positions ----
    temperature = 0.05
    cooling = 0.9995
    step_scale = 0.08   # initial perturbation size
    iterations = 5000

    for it in range(iterations):
        # propose a move: perturb 1‑3 randomly chosen circles
        new_centres = best_centres.copy()
        k = rng.integers(1, 4)                     # how many circles to move
        idxs = rng.choice(n, size=k, replace=False)

        new_centres[idxs] += delta

        new_centres = np.clip(new_centres, 0.0, 1.0)

        new_radii = _clip_radii(new_centres, new_radii)

            best_sum = new_sum
            best_centres = new_centres.copy()
            best_radii = new_radii.copy()

        # cool down
        temperature *= cooling
        step_scale *= cooling

    # ---- 4. Final refinement (one more exact LP solve) ----
    final_radii, final_sum = _solve_lp(best_centres)
    final_radii = _clip_radii(best_centres, final_radii)
    final_sum = float(final_radii.sum())
```
