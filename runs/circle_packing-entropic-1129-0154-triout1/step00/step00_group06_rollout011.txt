def compute_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    # Use numpy broadcasting
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]  # shape (n,n,2)
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # For i==j, we can set to np.inf or ignore.
    # Build constraints
    # A_ub will have shape (4*n + n*(n-1)/2, n)
    # Build row list
    rows = []
    b = []
    # boundary constraints
    # for each i: r_i <= x_i
    # Could use loop to add each row.
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        rows.append(e)
        b.append(centers[i,0])  # x
        rows.append(e)
        b.append(1.0 - centers[i,0]) # 1-x
        rows.append(e)
        b.append(centers[i,1]) # y
        rows.append(e)
        b.append(1.0 - centers[i,1]) # 1-y

    # pairwise constraints r_i + r_j <= d_ij
    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            b.append(dists[i,j])

    A_ub = np.array(rows)
    b_ub = np.array(b)

    c = -np.ones(n)  # maximize sum radii

    # bounds: (0, None)
    bounds = [(0, None)] * n

    # Solve via linprog
    # Use method='highs' for speed & reliability
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: maybe reduce tolerance, but for now raise
        raise RuntimeError("LP failed")
    radii = res.x
    # enforce small negativity due to numerical errors:
    radii[radii <0] =0.0
    return radii

def hill_climb(initial_centers, steps, step_size, rng):
    current_centers = initial_centers.copy()
    current_radii = compute_radii(current_centers)
    current_sum = current_radii.sum()
    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum
    for it in range(steps):
        # random index
        i = rng.integers(len(current_centers))
        # propose new center
        new_center = current_centers[i].copy()
        delta = rng.uniform(-step_size, step_size, size=2)
        new_center += delta
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not too close to others? not needed.
        # create trial centers copy
        trial_centers = current_centers.copy()
        trial_centers[i] = new_center
        # compute radii
        trial_radii = compute_radii(trial_centers)
        trial_sum = trial_radii.sum()
        # accept if better
        if trial_sum > current_sum + 1e-12:
            current_centers = trial_centers
            current_radii = trial_radii
            current_sum = trial_sum
            # update best
            if trial_sum > best_sum:
                best_sum = trial_sum
                best_centers = trial_centers.copy()
                best_radii = trial_radii.copy()
        # optionally reduce step or random acceptance...
    return best_centers, best_radii, best_sum

def farthest_point_sampling(N, pool_size, margin, rng):
    # generate pool of random points within [margin,1-margin]
    candidates = rng.uniform(margin, 1 - margin, size=(pool_size, 2))
    # select first point randomly

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers):
    n = centers.shape[0]
    # distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    rows = []
    b = []
    # boundary constraints
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        rows.append(e)
        b.append(centers[i,0])          # x_i
        rows.append(e)
        b.append(1.0 - centers[i,0])    # 1-x_i
        rows.append(e)
        b.append(centers[i,1])          # y_i
        rows.append(e)
        b.append(1.0 - centers[i,1])    # 1-y_i
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            b.append(dists[i,j])
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)  # maximize sum radii
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        raise RuntimeError("Linear program failed")
    radii = res.x
    # Safety: clip tiny negatives & small epsilon
    radii[radii < 0] = 0.0
    return radii

def farthest_point_sampling(N, pool_size=2000, margin=0.08, rng=None):
    if rng is None:
        rng = np.random.default_rng()
    # generate random candidates within [margin, 1-margin]
    candidates = rng.uniform(margin, 1.0 - margin, size=(pool_size, 2))
    # Choose first point randomly
    idx = rng.integers(candidates.shape[0])
    selected = [candidates[idx].copy()]
    # remove selected
    candidates = np.delete(candidates, idx, axis=0)
    while len(selected) < N:
        sel_arr = np.array(selected)
        # Compute distances to current selected set
        # shape (n_candidates, n_selected)
        diffs = candidates[:, np.newaxis, :] - sel_arr[np.newaxis, :, :]
        d2 = np.sum(diffs**2, axis=2)
        min_d2 = d2.min(axis=1)
        # choose candidate maximizing min distance
        i = np.argmax(min_d2)
        selected.append(candidates[i].copy())
        candidates = np.delete(candidates, i, axis=0)
    return np.array(selected)

import numpy as np
from scipy.optimize import linprog

def compute_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear program that gives the maximal radii for a fixed set of centres.
    """
    n = centers.shape[0]

    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    rows = []
    rhs = []

    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0

        rows.append(e)                # r_i <= x_i
        rhs.append(centers[i, 0])

        rows.append(e)                # r_i <= 1 - x_i
        rhs.append(1.0 - centers[i, 0])

        rows.append(e)                # r_i <= y_i
        rhs.append(centers[i, 1])

        rows.append(e)                # r_i <= 1 - y_i
        rhs.append(1.0 - centers[i, 1])

    # non‑overlap constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            rhs.append(dists[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # maximise Σ r_i  -> minimise -Σ r_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # in the extremely unlikely case of failure, fall back to zeros
        return np.zeros(n, dtype=float)

    radii = res.x
    radii[radii < 0] = 0.0          # clean tiny negatives
    return radii


def farthest_point_sampling(
    N: int,
    pool_size: int = 3000,
    margin: float = 0.08,
    rng: np.random.Generator = None
) -> np.ndarray:
    """
    Choose N points inside [margin,1‑margin]² by farthest‑point (max‑min) sampling.
    """
    if rng is None:
        rng = np.random.default_rng()
    # random candidate pool
    candidates = rng.uniform(margin, 1.0 - margin, size=(pool_size, 2))

    # pick the first point randomly
    idx = rng.integers(candidates.shape[0])
    selected = [candidates[idx].copy()]
    candidates = np.delete(candidates, idx, axis=0)

    while len(selected) < N:
        sel_arr = np.asarray(selected)                     # (k,2)
        # distance from each candidate to the already selected points
        diff = candidates[:, np.newaxis, :] - sel_arr[np.newaxis, :, :]  # (m,k,2)
        d2 = np.sum(diff ** 2, axis=2)                     # (m,k)
        min_d2 = d2.min(axis=1)                            # (m,)
        i = np.argmax(min_d2)                              # farthest candidate
        selected.append(candidates[i].copy())
        candidates = np.delete(candidates, i, axis=0)

    return np.asarray(selected)


def hill_climb(
    initial_centers: np.ndarray,
    steps: int = 2500,
    init_step: float = 0.07,
    rng: np.random.Generator = None
):
    """
    Optimise the centre positions by a simple simulated‑annealing hill climb.
    Each move re‑solves the LP and accepts the move if the sum of radii improves
    (or, with a temperature‑controlled probability, if it gets slightly worse).
    """
    if rng is None:
        rng = np.random.default_rng()
    N = initial_centers.shape[0]

    current_centers = initial_centers.copy()
    current_radii = compute_radii(current_centers)
    current_sum = float(current_radii.sum())

    best_centers = current_centers.copy()
    best_radii = current_radii.copy()
    best_sum = current_sum

    T0 = 0.001                         # initial temperature for occasional downhill moves

    for it in range(steps):
        # linear cooling
        T = T0 * (1.0 - it / steps)
        step = init_step * (1.0 - it / steps)

        # pick a circle to move
        i = rng.integers(N)
        new_center = current_centers[i] + rng.uniform(-step, step, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)

        trial_centers = current_centers.copy()
        trial_centers[i] = new_center

        trial_radii = compute_radii(trial_centers)
        trial_sum = float(trial_radii.sum())
        delta = trial_sum - current_sum

        # accept if better, otherwise with Metropolis probability
        if delta > 0.0 or rng.random() < np.exp(delta / max(T, 1e-12)):
            current_centers = trial_centers
            current_radii = trial_radii
            current_sum = trial_sum

            if trial_sum > best_sum:
                best_sum = trial_sum
                best_centers = trial_centers.copy()
                best_radii = trial_radii.copy()

    return best_centers, best_radii, best_sum


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a packing of 26 circles in the unit square.
    Returns (centres, radii, sum_radii).
    """
    N = 26
    best_sum = -1.0
    best_centres = None
    best_radii = None

    base_seed = 20231129
    # several independent restarts increase the chance of hitting the target sum
    for restart in range(5):
        rng = np.random.default_rng(base_seed + restart)

        # 1) initialise centres with a far‑point set
        init_centres = farthest_point_sampling(N, pool_size=3000, margin=0.08, rng=rng)

        centres, radii, s = hill_climb(init_centres, steps=2500, init_step=0.07, rng=rng)

        if s > best_sum:
            best_sum = s
            best_centres = centres
            best_radii = radii

        if best_sum >= 2.636:           # good enough – stop early
            break

    # Safety margin: shrink every radius by a tiny epsilon so that we stay
    # strictly inside the validation tolerances.
    if best_radii is not None:
        eps = 1e-9
        best_radii = np.maximum(0.0, best_radii - eps)
        best_sum = float(best_radii.sum())

    return best_centres, best_radii, best_sum
