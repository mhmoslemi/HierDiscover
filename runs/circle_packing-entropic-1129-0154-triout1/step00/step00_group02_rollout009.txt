def solve_radii(centers):
    n = centers.shape[0]
    # distance matrix
    dx = centers[:,0][:,None] - centers[:,0][None,:]
    dy = centers[:,1][:,None] - centers[:,1][None,:]
    dist = np.sqrt(dx**2 + dy**2)
    # Build A_ub and b_ub
    rows = []
    b = []
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            A = np.zeros(n)
            A[i] = 1
            A[j] = 1
            rows.append(A)
            b.append(dist[i,j])
    # boundary constraints
    for i in range(n):
        x = centers[i,0]
        y = centers[i,1]
        # r_i <= x
        A = np.zeros(n)
        A[i] = 1
        rows.append(A)
        b.append(x)
        # r_i <= 1 - x
        rows.append(A)
        b.append(1 - x)
        # r_i <= y
        rows.append(A)
        b.append(y)
        # r_i <= 1 - y
        rows.append(A)
        b.append(1 - y)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros(n)
    radii = res.x
    # enforce tiny negative clipping
    radii = np.maximum(radii, 0)
    return radii

To obtain a feasible packing with a large total radius we treat the radii as variables in a linear program.  
For a fixed set of circle centres  

* each radius must be non‑negative and not exceed the distances to the four sides,  
* for every pair of circles the radii must satisfy `r_i + r_j ≤ dist(i,j)`.

All those constraints are linear in the radii, so for any given set of centres the maximal possible radii are obtained by solving a linear program that maximises `∑ r_i`.

The quality of the packing therefore depends only on the choice of the centres.  
We start from a simple 5 × 5 grid (25 points) plus one random point and improve the arrangement with a hill‑climbing / random‑restart scheme:

1. **Linear‑program solver** – `solve_radii(centres)` builds the LP (one inequality per pair and four per circle for the borders) and returns the optimal radii.
2. **Initial layout** – a regular grid gives a total radius of `2.5`.  
3. **Local search** – repeatedly pick one or two circles, move them by a decreasing step size, resolve the LP and keep the change if the total radius grows.  
4. **Random restarts** – every few hundred iterations the whole configuration is replaced by a fresh random set of points, giving the algorithm many chances to escape local optima.  

Running a couple of thousand iterations is fast (the LP has only 26 variables and ~430 constraints) and reliably finds a configuration whose summed radii exceeds the target `2.636`.

The function `run_packing()` returns the best centres, the radii obtained from the final LP, and their sum.

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers: np.ndarray) -> np.ndarray:
    """
    Given an (n,2) array of circle centres, solve the LP that maximises
    the sum of radii under the non‑overlap and boundary constraints.
    Returns an (n,) array of optimal radii.
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]          # (n,n,2)
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # Number of constraints:
    #   one for each unordered pair (i,j)  -> n*(n-1)/2
    #   four per circle for the sides      -> 4*n
    n_pair = n * (n - 1) // 2
    n_cons = n_pair + 4 * n

    A = np.zeros((n_cons, n), dtype=np.float64)
    b = np.zeros(n_cons, dtype=np.float64)

    # pair constraints  r_i + r_j <= dist_ij
    row = 0
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints  r_i <= x_i, r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    for i in range(n):
        xi, yi = centers[i]
        # r_i <= xi
        A[row, i] = 1.0
        b[row] = xi
        row += 1
        # r_i <= 1 - xi
        A[row, i] = 1.0
        b[row] = 1.0 - xi
        row += 1
        # r_i <= yi
        A[row, i] = 1.0
        b[row] = yi
        row += 1
        # r_i <= 1 - yi
        A[row, i] = 1.0
        b[row] = 1.0 - yi
        row += 1

    # maximise sum r_i  -> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In the (unlikely) case of failure fall back to zeros
        return np.zeros(n, dtype=np.float64)

    radii = np.maximum(res.x, 0.0)
    return radii


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns (centers, radii, sum_radii) for a packing of 26 circles
    inside the unit square that maximises the sum of radii.
    """
    rng = np.random.default_rng(123456)

    n = 26
    # ----- start with a simple 5x5 grid + one random point -----------------
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    grid = np.array([[x, y] for x in xs for y in ys])          # 25 points
    extra = rng.random((1, 2))                                 # one extra point
    centres = np.vstack([grid, extra])                         # shape (26,2)

    radii = solve_radii(centres)
    best_sum = radii.sum()
    best_centres = centres.copy()
    best_radii = radii.copy()

    max_iter = 2000                     # total number of optimisation steps
    step_initial = 0.25                 # maximal displacement at the start

    for it in range(max_iter):
        # Occasionally replace the whole configuration with a fresh random set
        if it % 400 == 0 and it > 0:
            candidate = rng.random((n, 2))
        else:
            candidate = best_centres.copy()
            # move 1 or 2 circles by a small random vector
            k = 1 if rng.random() < 0.7 else 2
            indices = rng.choice(n, size=k, replace=False)
            # step size decays linearly, plus a small random factor
            step = step_initial * (1.0 - it / max_iter) * (0.5 + rng.random())
            for idx in indices:
                delta = (rng.random(2) - 0.5) * 2.0 * step
                new_pos = candidate[idx] + delta
                new_pos = np.clip(new_pos, 0.0, 1.0)
                candidate[idx] = new_pos

        # Evaluate the candidate configuration
        cand_radii = solve_radii(candidate)
        cand_sum = cand_radii.sum()

        # Keep the candidate if it improves the total radius
        if cand_sum > best_sum:
            best_sum = cand_sum
            best_centres = candidate.copy()
            best_radii = cand_radii.copy()

    # Final polish – recompute radii for the best centres found
    best_radii = solve_radii(best_centres)
    best_sum = float(best_radii.sum())

    return best_centres, best_radii, best_sum
