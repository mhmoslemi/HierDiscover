def solve_radii(centers):
    n = centers.shape[0]
    # else...
    # build constraints
    A_ub = []
    b_ub = []
    # boundaries
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(x)
        # r_i <= 1-x
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1 - x)
        # r_i <= y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(y)
        # r_i <= 1-y
        row = np.zeros(n)
        row[i] = 1.0
        A_ub.append(row)
        b_ub.append(1 - y)
    # neighbor constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            A_ub.append(row)
            b_ub.append(d)
    A_ub = np.array(A_ub)
    b_ub = np.array(b_ub)
    c = -np.ones(n) # maximize sum -> minimize -sum.

    # solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # In case of failure, return zeros
        return np.zeros(n), 0.0
    radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r

def farthest_points(num, candidates=1000):
    points = []
    # start with random
    first = np.random.rand(2)
    points.append(first)
    for _ in range(1,num):
        # generate many candidate points
        cand = np.random.rand(candidates, 2)
        # compute min distance to existing points
        dists = np.min(np.linalg.norm(cand[:, None, :] - np.array(points)[None, :, :], axis=2), axis=1)
        best_idx = np.argmax(dists)
        points.append(cand[best_idx])
    return np.array(points)

def improve_centers(centers, steps=1000, step_size=0.05):
    best_centers = centers.copy()
    best_radii, best_sum = solve_radii(best_centers)
    cur_centers = best_centers.copy()
    cur_sum = best_sum
    cur_radii = best_radii
    for iter in range(steps):
        i = np.random.randint(len(centers))
        delta = (np.random.rand(2)-0.5)*2*step_size
        new_center = cur_centers[i] + delta
        new_center = np.clip(new_center, 0, 1)
        new_centers = cur_centers.copy()
        new_centers[i] = new_center
        # solve LP
        new_radii, new_sum = solve_radii(new_centers)
        if new_sum > cur_sum + 1e-9:
            cur_centers = new_centers
            cur_sum = new_sum
            cur_radii = new_radii
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
        # optionally decrease step size gradually
        step_size *= 0.9999 # slowly reduce

    return best_centers, best_radii, best_sum

assistantfinal**Strategy**

We need a set of 26 circles inside the unit square that do not overlap and maximise the sum of their radii.  

For a *fixed* placement of the circle centres the optimal radii are obtained by a linear program:

```
max   Σ r_i
s.t.  r_i ≤ x_i,  r_i ≤ 1‑x_i,
      r_i ≤ y_i,  r_i ≤ 1‑y_i,
      r_i + r_j ≤ dist( centre_i , centre_j )   (i < j)
      r_i ≥ 0
```

Thus the problem reduces to finding a good arrangement of the 26 centres.  
A simple heuristic works well:

1. **Initial placement** – use a *farthest‑point* (max‑min) sampling that spreads the points uniformly over the square.  
2. **Local optimisation** – repeatedly pick a random circle, move its centre a little (the step size is slowly reduced) and recompute the optimal radii with the LP above.  
   *If the total radius sum improves, keep the move; otherwise keep the old configuration.*  
   A tiny simulated‑annealing acceptance (`exp(Δ/temperature)`) is added so the search can escape local plateaus.  
3. **Multiple restarts** – the whole procedure is performed several times with different random seeds; the best result is returned.

All calculations are deterministic (the global NumPy random seed is fixed) and use only NumPy and SciPy, which satisfy the rules.

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programme: for a given set of centres compute the optimal radii
# ----------------------------------------------------------------------
def solve_radii(centres: np.ndarray):
    """
    Solve the LP that gives the maximal radii for fixed centre positions.

    Parameters
    ----------
    centres : np.ndarray, shape (n, 2)
        The (x, y) coordinates of the circles.

    Returns
    -------
    radii : np.ndarray, shape (n,)
        Radii that maximise the total sum while respecting all constraints.
    total : float
        Sum of the radii.
    """
    n = centres.shape[0]
    # ----------- build inequality matrix A_ub * r <= b_ub ---------------
    rows = []

    # boundary constraints
    for i in range(n):
        x, y = centres[i]
        e = np.zeros(n)
        e[i] = 1.0
        rows.append((e, x))           # r_i <= x_i
        rows.append((e, 1.0 - x))     # r_i <= 1 - x_i
        rows.append((e, y))           # r_i <= y_i
        rows.append((e, 1.0 - y))     # r_i <= 1 - y_i

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append((e, d))

    A_ub = np.stack([r[0] for r in rows])
    b_ub = np.array([r[1] for r in rows])

    # minimise -sum(r_i)  <=> maximise sum(r_i)
    c = -np.ones(n)

    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A_ub,
        b_ub=b_ub,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )
    if not res.success:
        # should never happen – return zeros if it does
        radii = np.zeros(n)
    else:
        radii = res.x
    total = float(radii.sum())
    return radii, total


# ----------------------------------------------------------------------
# Farthest‑point (max‑min) sampling – creates a well‑spaced set of points
# ----------------------------------------------------------------------
def farthest_points(n: int, n_candidates: int = 1500) -> np.ndarray:
    """
    Return n points inside [0,1]² that are spread out by the farthest‑point
    heuristic.
    """
    points = []
    # first point is completely random
    points.append(np.random.rand(2))
    for _ in range(1, n):
        cand = np.random.rand(n_candidates, 2)
        existing = np.asarray(points)          # shape (k,2)
        # distance from each candidate to the nearest existing point
        diffs = cand[:, None, :] - existing[None, :, :]   # (cand, k, 2)
        dists = np.linalg.norm(diffs, axis=2)            # (cand, k)
        mindist = dists.min(axis=1)                      # (cand,)
        best = np.argmax(mindist)
        points.append(cand[best])
    return np.asarray(points)


# ----------------------------------------------------------------------
# Local improvement of a set of centres
# ----------------------------------------------------------------------
def improve_centres(
    centres: np.ndarray,
    steps: int = 3000,
    init_step: float = 0.15,
    init_temp: float = 0.02,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a simple simulated‑annealing / hill‑climbing search on the centre
    positions.  Radii are recomputed by solve_radii after each move.
    """
    n = centres.shape[0]
    cur_c = centres.copy()
    cur_r, cur_sum = solve_radii(cur_c)

    best_c = cur_c.copy()
    best_r = cur_r.copy()
    best_sum = cur_sum

    step = init_step
    temp = init_temp

    for it in range(steps):
        i = np.random.randint(n)                     # which circle to move
        # propose a move
        delta = (np.random.rand(2) - 0.5) * 2.0 * step
        new_pos = cur_c[i] + delta
        new_pos = np.clip(new_pos, 0.0, 1.0)

        new_c = cur_c.copy()
        new_c[i] = new_pos

        new_r, new_sum = solve_radii(new_c)

        delta_sum = new_sum - cur_sum
        accept = False
        if delta_sum > 0:
            accept = True
        else:
            # occasional downhill moves (simulated annealing)
            prob = np.exp(delta_sum / max(temp, 1e-12))
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_c = new_c
            cur_r = new_r
            cur_sum = new_sum
            if cur_sum > best_sum:
                best_sum = cur_sum
                best_c = cur_c.copy()
                best_r = cur_r.copy()

        # gradually shrink the step size and the temperature
        step *= 0.9995
        temp *= 0.9995

    return best_c, best_r, best_sum


# ----------------------------------------------------------------------
# Main entry point required by the evaluator
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produce a packing of 26 circles in the unit square.
    Returns (centres, radii, sum_of_radii).
    """
    np.random.seed(123456)          # deterministic behaviour

    N = 26
    best_sum_overall = -1.0
    best_centres = None
    best_radii = None

    # several independent attempts – keep the best result
    for attempt in range(5):
        # 1) initialise centres with a max‑min spread
        centres0 = farthest_points(N, n_candidates=2000)

        # small random jitter to avoid perfect symmetry
        centres0 += (np.random.rand(N, 2) - 0.5) * 0.02
        centres0 = np.clip(centres0, 0.0, 1.0)

        # 2) optimise those centres locally
        centres_opt, radii_opt, sum_opt = improve_centres(
            centres0, steps=3000, init_step=0.15, init_temp=0.02
        )

        if sum_opt > best_sum_overall:
            best_sum_overall = sum_opt
            best_centres = centres_opt
            best_radii = radii_opt

    # final safety check – recompute radii for the reported centres
    final_radii, final_sum = solve_radii(best_centres)
    # (the LP may increase the sum a little further)
    return best_centres, final_radii, float(final_sum)
