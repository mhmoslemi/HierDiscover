Implementation:

```python
def solve_lp(centers):
    n = len(centers)
    # objective: maximize sum radii -> minimize negative
    c = -np.ones(n)
    # bounds for variables
    bounds = [(0, None)] * n
    # constraints
    A = []
    b = []
    xs = centers[:,0]
    ys = centers[:,1]
    for i in range(n):
        # x_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(xs[i])
        # 1-x_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - xs[i])
        # y_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(ys[i])
        # 1-y_i
        row = np.zeros(n)
        row[i] = 1
        A.append(row)
        b.append(1 - ys[i])
    # pairwise non-overlap
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(d)
    A_ub = np.array(A)
    b_ub = np.array(b)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # maybe infeasible due to numeric? fallback to zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r
```

    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=float)

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback zero radii (should never happen)
        radii = np.zeros(n)
    else:
        radii = res.x
        # protect against tiny negative values due to numerics
        radii = np.maximum(radii, 0.0)
    return radii, float(np.sum(radii))

import numpy as np
from scipy.optimize import linprog


def solve_lp(centers: np.ndarray):
    """
    Solve the linear programme that, for a fixed set of centres,
    yields the maximal radii respecting the square and non‑overlap constraints.
    Returns radii and the total sum of radii.
    """
    n = centers.shape[0]
    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n)
    bounds = [(0.0, None) for _ in range(n)]

    xs = centers[:, 0]
    ys = centers[:, 1]

    rows = []
    rhs = []

    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        # r_i <= x_i
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(xs[i])

        # r_i <= 1 - x_i
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - xs[i])

        # r_i <= y_i
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(ys[i])

        # r_i <= 1 - y_i
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(1.0 - ys[i])

    # non‑overlap constraints: r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=float)

    # use the HiGHS solver (fast and reliable for this size)
    result = linprog(
        c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs", options={"presolve": True}
    )

    if not result.success:
        # In the extremely unlikely case of failure return zeros
        radii = np.zeros(n)
        total = 0.0
    else:
        radii = np.maximum(result.x, 0.0)   # guard against tiny negatives
        total = float(np.sum(radii))

    return radii, total


def compute_solution(centers: np.ndarray):
    """Convenient wrapper returning radii and total sum."""
    radii, total = solve_lp(centers)
    return radii, total


def hill_climb(initial: np.ndarray,
               max_iters: int = 1500,
               step_max: float = 0.04,
               seed: int = 0):
    """
    Simple stochastic hill‑climbing (with a tiny SA component) that
    adjusts centre positions to increase the total radii.
    """
    rng = np.random.default_rng(seed)
    n = initial.shape[0]

    best_centers = initial.copy()
    best_radii, best_sum = compute_solution(best_centers)

    cur_centers = best_centers.copy()
    cur_sum = best_sum

    for it in range(max_iters):
        # step size linearly shrinks
        step = step_max * (1.0 - it / max_iters)

        # pick a circle and propose a tiny move
        i = rng.integers(n)
        proposal = cur_centers.copy()
        delta = rng.uniform(-step, step, size=2)
        proposal[i] = np.clip(proposal[i] + delta, 0.0, 1.0)

        radii_prop, sum_prop = compute_solution(proposal)

            best_centers = proposal.copy()
