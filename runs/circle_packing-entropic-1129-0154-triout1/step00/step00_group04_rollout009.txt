Implementation details for LP:

```python
def solve_lp(centers):
    n = centers.shape[0]
    # compute distance between centers
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # create constraints matrix A_ub and b_ub
    # total constraints M = n (bounds) + n*(n-1)//2 (pair constraints)
    M = n + n*(n-1)//2
    A = np.zeros((M, n))
    b = np.zeros(M)
    # boundaries constraints
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # each row i: A[i,i]=1, b[i]=bounds[i]
    for i in range(n):
        A[i,i] = 1.0
        b[i] = bounds[i]
    # pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx,i] = 1.0
            A[idx,j] = 1.0
            b[idx] = dists[i,j]
            idx += 1
    # objective maximize sum r -> minimize negative sum
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=(0, None), method='highs')
    if not res.success:
        # maybe infeasible; set radii zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    sum_r = np.sum(radii)
    return radii, sum_r
```

```
def solve_lp(centers):
    n = centers.shape[0]
    # compute bounds to edges
    min_edge = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # distances matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # build A_ub and b_ub
    M = n + n*(n-1)//2
    A = np.zeros((M, n))
    b = np.zeros(M)
    # edge constraints
    for i in range(n):
        A[i, i] = 1.0
        b[i] = min_edge[i]
    # pair constraints
    idx = n
    for i in range(n):
        for j in range(i+1, n):
            A[idx, i] = 1.0
            A[idx, j] = 1.0
            b[idx] = dists[i, j]
            idx += 1
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None) for _ in range(n)], method='highs')
    if not res.success:
        # Infeasible (shouldn't happen); set radii to zeros
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = radii.sum()
    return radii, sum_r
```

```python
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programming: for a fixed set of centres compute the optimal radii
# ----------------------------------------------------------------------
def solve_lp(centers: np.ndarray):
    """
    Solve the LP that maximises sum of radii for a given set of centres.

    Returns (radii, total_sum)
    """
    n = centers.shape[0]

    # distance from each centre to the four sides of the square
    edge_bound = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )  # shape (n,)

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # Build inequality matrix A_ub * r <= b_ub
    #  – one row per centre (boundary)
    #  – one row per unordered pair (non‑overlap)
    m_boundary = n
    m_pairs = n * (n - 1) // 2
    m_total = m_boundary + m_pairs

    A = np.zeros((m_total, n), dtype=np.float64)
    b = np.zeros(m_total, dtype=np.float64)

    # boundary constraints  r_i <= edge_bound_i
    for i in range(n):
        A[i, i] = 1.0
        b[i] = edge_bound[i]

    # non‑overlap constraints  r_i + r_j <= dist_ij
    row = m_boundary
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise sum r_i  <=> minimise -sum r_i
    c = -np.ones(n, dtype=np.float64)

    # variable bounds: r_i >= 0, no explicit upper bound (handled by A_ub)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=bounds,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        # Infeasible should not happen – fall back to zeros
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = res.x
        total = np.sum(radii)

    return radii, total

# Helper functions to generate initial point sets

def generate_rotated_hex(dx: float, rows: int, angle: float, max_points: int = None):
    """
    Hexagonal lattice with horizontal spacing dx, vertical spacing dy = sqrt(3)/2*dx,
    optionally rotated by `angle` (radians) around the centre (0.5,0.5).
    Returns an (M,2) array (M may be < max_points).
    """
    dy = np.sqrt(3.0) / 2.0 * dx
    pts = []

    for r in range(rows):
        # offset for the alternating rows
        offset = (r % 2) * dx / 2.0
        y = dx / 2.0 + r * dy
        if y > 1.0 - dx / 2.0 + 1e-12:
            continue
        x_start = dx / 2.0 + offset
        xs = np.arange(x_start, 1.0 - dx / 2.0 + 1e-12, dx)
        for x in xs:
            pts.append([x, y])

    pts = np.asarray(pts, dtype=np.float64)

    # rotate about centre if needed
    if angle != 0.0:
        shift = pts - 0.5
        rot = np.array(
            [[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]],
            dtype=np.float64,
        )
        shift = shift @ rot.T
        pts = shift + 0.5

    # keep only points that stayed inside the unit square
    mask = (
        (pts[:, 0] >= 0.0)
        & (pts[:, 0] <= 1.0)
        & (pts[:, 1] >= 0.0)
        & (pts[:, 1] <= 1.0)
    )
    pts = pts[mask]

    if max_points is not None and pts.shape[0] > max_points:
        pts = pts[:max_points]

    return pts


def generate_grid_plus_extra():
    """
    Regular 5×5 grid (spacing 0.2) plus a centre point.
    """
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    grid = np.array([[x, y] for y in ys for x in xs], dtype=np.float64)
    extra = np.array([[0.5, 0.5]], dtype=np.float64)
    return np.vstack([grid, extra])

def run_packing():
    """
    Returns (centers, radii, sum_radii) for 26 circles.
    The routine performs a stochastic search that maximises the sum of radii.
    """
    N = 26
    rng = np.random.default_rng(0)               # deterministic seed

    # ------------------------------------------------------------------
    # 1) Build a list of diverse seed configurations
    # ------------------------------------------------------------------
    seed_list = []

    # a) 5‑row hex lattices, several spacings and rotations
    for angle in [0.0, np.pi / 12.0, np.pi / 8.0, np.pi / 6.0, np.pi / 4.0]:
        for dx in np.linspace(0.18, 0.22, 5):
            pts = generate_rotated_hex(dx=dx, rows=5, angle=angle, max_points=None)
            if pts.shape[0] >= N:
                seed = pts[:N]
            else:
                # pad with random points
                extra_needed = N - pts.shape[0]
                extra = rng.random((extra_needed, 2))
                seed = np.vstack([pts, extra])
            seed_list.append(seed)

    # b) 5×5 grid + centre
    seed_list.append(generate_grid_plus_extra())

    # c) a few completely random point sets
    for _ in range(5):
        seed_list.append(rng.random((N, 2)))

    # ------------------------------------------------------------------
    # 2) Evaluate all seeds once, keep the best
    # ------------------------------------------------------------------
    best_sum = -1.0
    best_centers = None
    best_radii = None

    for seed in seed_list:
        # ensure we really have exactly N points
        if seed.shape[0] != N:
            raise RuntimeError("Seed generation error")
        radii, total = solve_lp(seed)
        if total > best_sum:
            best_sum = total
            best_centers = seed.copy()
            best_radii = radii.copy()

    # ------------------------------------------------------------------
    # 3) Local stochastic optimisation (hill‑climbing + simulated annealing)
    # ------------------------------------------------------------------
    centers = best_centers.copy()
    radii = best_radii.copy()
    current_sum = best_sum

    max_iter = 5000
    for it in range(max_iter):
        # Temperature schedule (for occasional uphill moves)
        temperature = max(0.001, 0.02 * (max_iter - it) / max_iter)

        # Propose a new set of centres
        new_centers = centers.copy()

        # move a single randomly chosen circle
        i = rng.integers(N)
        step = 0.04
        # occasional larger jump
        if rng.random() < 0.1:
            step = 0.08
        new_centers[i] += rng.normal(scale=step, size=2)
        new_centers[i] = np.clip(new_centers[i], 0.0, 1.0)

        # with small probability jiggle all circles a little (helps escaping plateaux)
        if rng.random() < 0.05:
            new_centers += rng.normal(scale=0.01, size=new_centers.shape)
            new_centers = np.clip(new_centers, 0.0, 1.0)

        # Re‑solve the LP for the proposed geometry
        new_radii, new_sum = solve_lp(new_centers)

        # Acceptance rule
        delta = new_sum - current_sum
        accept = False
        if delta > 0:
            accept = True
        else:
            # Metropolis criterion
            if rng.random() < np.exp(delta / temperature):
                accept = True

        if accept:
            centers = new_centers
            radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

        # Every few iterations restart from a fresh random layout
        if (it + 1) % 500 == 0:
            rnd = rng.random((N, 2))
            rnd_radii, rnd_sum = solve_lp(rnd)
            if rnd_sum > best_sum:
                best_sum = rnd_sum
                best_centers = rnd.copy()
                best_radii = rnd_radii.copy()
                centers = rnd.copy()
                radii = rnd_radii.copy()
                current_sum = rnd_sum

    # Final polishing – one more LP solve on the best geometry
    final_radii, final_sum = solve_lp(best_centers)

    # Return exactly the required tuple
    return best_centers, final_radii, float(final_sum)
```
