```
def compute_radii_sum(centers):
    n = len(centers)
    # Build constraints matrices for linprog
    # A_ub x <= b_ub
    # Variables: r_i
    # boundary constraints:
    A = []
    b = []
    for i in range(n):
        x, y = centers[i]
        # r_i <= x
        coeff = np.zeros(n)
        coeff[i] = 1
        A.append(coeff)
        b.append(x)
        # r_i <= 1 - x
        coeff = np.zeros(n); coeff[i] = 1
        A.append(coeff)
        b.append(1 - x)
        # r_i <= y
        coeff = np.zeros(n); coeff[i] = 1
        A.append(coeff)
        b.append(y)
        # r_i <= 1 - y
        coeff = np.zeros(n); coeff[i] = 1
        A.append(coeff)
        b.append(1 - y)
    # neighbor constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            coeff = np.zeros(n)
            coeff[i] = 1
            coeff[j] = 1
            A.append(coeff)
            b.append(d)
    A = np.array(A)
    b = np.array(b)
    c = -np.ones(n)  # minimize -sum(r)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: set radii zero
        return np.zeros(n), 0.0
    radii = res.x
    return radii, radii.sum()
```

```
def run_packing():
    import numpy as np
    from scipy.optimize import linprog
    
    np.random.seed(0)
    n = 26
    # initial grid
    base_coords = np.arange(0.1, 1.0, 0.2)  # [0.1,0.3,0.5,0.7,0.9]
    centers = []
    for x in base_coords:
        for y in base_coords:
            centers.append([x, y])
    centers = np.array(centers)  # 25 points
    # extra point
    extra = np.array([0.5, 0.2])  # not in grid
    centers = np.vstack([centers, extra])
    
    best_centers = centers.copy()
    best_radii, best_sum = compute_radii_sum(best_centers)
    
    # function to evaluate
    def evaluate(centers):
        r, s = compute_radii_sum(centers)
        return s, r
    # local search
    steps = [0.15, 0.07, 0.03, 0.015, 0.007]
    for step in steps:
        for _ in range(800):  # adjust iteration count
            i = np.random.randint(n)
            new_center = best_centers[i] + np.random.uniform(-step, step, size=2)
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = best_centers.copy()
            new_centers[i] = new_center
            s, r = evaluate(new_centers)
            if s > best_sum + 1e-12:
                best_sum = s
                best_radii = r
                best_centers = new_centers
    # final compute radii again to ensure consistency
    best_radii, best_sum = compute_radii_sum(best_centers)
    return best_centers, best_radii, float(best_sum)
```

```
def hill_climb(centers, rng, steps, max_iter_per_step):
    best_centers = centers.copy()
    best_radii, best_sum = compute_optimal_radii(best_centers)
    n = centers.shape[0]
    for step in steps:
        for _ in range(max_iter_per_step):
            i = rng.integers(n)
            proposal = best_centers[i] + rng.uniform(-step, step, size=2)
            proposal = np.clip(proposal, 0.0, 1.0)
            new_centers = best_centers.copy()
            new_centers[i] = proposal
            new_radii, new_sum = compute_optimal_radii(new_centers)
            if new_sum > best_sum:
                best_sum = new_sum
                best_radii = new_radii
                best_centers = new_centers
    return best_centers, best_radii, best_sum
```

```
def compute_optimal_radii(centers):
    n = centers.shape[0]
    # number of constraints
    # Build A_ub and b_ub arrays
    # total constraints = 4*n + n*(n-1)//2
    num_constraints = 4*n + n*(n-1)//2
    A = np.zeros((num_constraints, n), dtype=np.float64)
    b = np.zeros(num_constraints, dtype=np.float64)
    row = 0
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1
        b[row] = x
        row += 1
        A[row, i] = 1
        b[row] = 1 - x
        row += 1
        A[row, i] = 1
        b[row] = y
        row += 1
        A[row, i] = 1
        b[row] = 1 - y
        row += 1
```

1. **Linear programming for the radii**  
   For a *fixed* set of centre points the maximal radii are obtained by the linear program  

   \[
   \max\;\sum_i r_i
   \]

   with the constraints  

   * \(0\le r_i\)  
   * \(r_i \le x_i,\; r_i \le 1-x_i,\; r_i \le y_i,\; r_i \le 1-y_i\)  (stay inside the square)  
   * \(r_i+r_j \le d_{ij}\)  where \(d_{ij}=‖c_i-c_j‖\) (no overlap)

   All constraints are linear in the variables \(r_i\); we solve the LP with
   `scipy.optimize.linprog` (method *highs*).  
   The returned radii are reduced by a tiny \(\varepsilon=10^{-9}\) to stay safely
   inside the tolerances used by the validator.

2. **Improving the centre positions**  
   The sum of the optimal radii depends on the geometry of the centres.
   Starting from a good deterministic layout (a 5 × 5 grid plus one extra
   point) we perform a simple *hill‑climbing* / simulated‑annealing style
   optimisation:

   * at each iteration a random centre is displaced by a small amount,
     the new optimal radii are recomputed,
   * the move is kept only if the total radius sum increases.
   * The step size is gradually reduced (large moves first, fine adjustments later).

   A few random restarts are performed as well – they increase the chance of
   escaping poor local optima.

3. **Result**  
   The routine returns the best centre array, the corresponding radii and the
   achieved sum.  
   In practice the heuristic easily exceeds the requested value

```python
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear programme for a given set of centres
# ----------------------------------------------------------------------
def compute_optimal_radii(centres: np.ndarray):
    """
    Given an (n,2) array of centre positions, solve the LP that maximises
    the sum of radii under the non‑overlap and containment constraints.
    Returns the radii vector and the sum of radii.
    """
    n = centres.shape[0]

    # number of linear constraints:
    # 4 per centre (distance to the 4 sides) + one for each unordered pair
    n_pair = n * (n - 1) // 2
    n_cons = 4 * n + n_pair

    A = np.zeros((n_cons, n), dtype=np.float64)
    b = np.zeros(n_cons, dtype=np.float64)

    # ---- boundary constraints ----
    row = 0
    for i in range(n):
        x, y = centres[i]
        # r_i <= x
        A[row, i] = 1.0
        b[row] = x
        row += 1
        # r_i <= 1 - x
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        # r_i <= y
        A[row, i] = 1.0
        b[row] = y
        row += 1
        # r_i <= 1 - y
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # ---- pairwise non‑overlap constraints ----
    for i in range(n):
        for j in range(i + 1, n):
            dx = centres[i, 0] - centres[j, 0]
            dy = centres[i, 1] - centres[j, 1]
            d = np.hypot(dx, dy)
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d
            row += 1

    # objective: maximise sum(r)  --> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    # bounds: radii are non‑negative
    bounds = [(0.0, None) for _ in range(n)]

    # solve the LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fall back to the trivial feasible solution
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x

    # a tiny safety margin – the validator uses a 1e‑12 tolerance
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)

    return radii, float(radii.sum())


# ----------------------------------------------------------------------
# Helper: hill‑climbing / simulated‑annealing improvement of centres
# ----------------------------------------------------------------------
def improve_centres(initial_centres: np.ndarray,
                    rng: np.random.Generator,
                    total_iters: int = 2500,
                    step_start: float = 0.12,
                    step_end: float = 0.001):
    """
    Starting from `initial_centres` perform a simple stochastic hill
    climbing.  At each step a single centre is perturbed; the move is kept
    iff it raises the total sum of radii.
    """
    centres = initial_centres.copy()
    best_radii, best_sum = compute_optimal_radii(centres)
    best_centres = centres.copy()
    n = centres.shape[0]

    for it in range(total_iters):
        # linear schedule for the step size
        t = it / total_iters
        step = (1.0 - t) * step_start + t * step_end

        i = rng.integers(n)                     # which centre to move
        proposal = best_centres[i] + rng.uniform(-step, step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)   # keep inside the square

        new_centres = best_centres.copy()
        new_centres[i] = proposal

        new_radii, new_sum = compute_optimal_radii(new_centres)

        if new_sum > best_sum:
            best_sum = new_sum
            best_radii = new_radii
            best_centres = new_centres

    return best_centres, best_radii, best_sum


# ----------------------------------------------------------------------
# Main entry point required by the task
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Returns a tuple (centres, radii, sum_of_radii) for 26 circles placed
    inside the unit square.  The configuration is obtained by the
    optimisation routine described above.
    """
    rng = np.random.default_rng(0)          # deterministic randomness
    n = 26

    # ---- deterministic start: a 5×5 grid (25 points) + one extra point ----
    base = np.arange(0.1, 1.0, 0.2)          # 0.1,0.3,…,0.9
    grid = np.array([[x, y] for x in base for y in base])   # 25 points
    extra = np.array([[0.5, 0.2]])          # a point not coincident with the grid
    start = np.vstack([grid, extra])       # shape (26,2)

    # first improvement from the deterministic layout
    best_c, best_r, best_sum = improve_centres(start, rng,
                                               total_iters=3000,
                                               step_start=0.12,
                                               step_end=0.001)

    # a few random restarts – they often discover better global layouts
    for _ in range(4):
        rnd_start = rng.random((n, 2))      # uniform random points
        c, r, s = improve_centres(rnd_start, rng,
                                  total_iters=2500,
                                  step_start=0.12,
                                  step_end=0.001)
        if s > best_sum:
            best_c, best_r, best_sum = c, r, s

    # final sanity recomputation (guards against tiny numerical drift)
    best_r, best_sum = compute_optimal_radii(best_c)

    return best_c, best_r, best_sum
```
