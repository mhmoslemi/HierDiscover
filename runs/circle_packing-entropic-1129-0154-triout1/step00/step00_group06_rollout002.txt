def solve_radii(centers):
   n = centers.shape[0]
   # compute distances matrix
   diffs = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]  # shape (n,n,2)
   distances = np.linalg.norm(diffs, axis=2)   # shape (n,n)
   # compute constraints:
   # We'll create list of rows for A_ub and b_ub
   # For each i: r_i <= x_i => row: e_i (1 at i), b = x_i
   # Similarly r_i <= 1-x_i etc.
   # For each i<j: r_i + r_j <= distances[i,j] => row: e_i + e_j =1 for i,j, b=distances[i,j]
   # Actually we also need to ensure distances>0 else ignore constraints for same index.
   # Use SciPy's linear programming: minimize -sum(r_i)
   c = -np.ones(n)
   num_bounds = n
   # Build A_ub:
   rows = []
   b = []
   # boundary constraints
   for i in range(n):
       x, y = centers[i]
       # r_i <= x
       row = np.zeros(n)
       row[i] = 1
       rows.append(row)
       b.append(x)
       # r_i <= 1 - x
       row = np.zeros(n)
       row[i] = 1
       rows.append(row)
       b.append(1 - x)
       # r_i <= y
       row = np.zeros(n)
       row[i] = 1
       rows.append(row)
       b.append(y)
       # r_i <= 1 - y
       row = np.zeros(n)
       row[i] = 1
       rows.append(row)
       b.append(1 - y)
   # pair constraints
   for i in range(n):
       for j in range(i+1, n):
           d = distances[i, j]
           # no need to ignore tiny distances (they are >0 if positions distinct)
           row = np.zeros(n)
           row[i] = 1
           row[j] = 1
           rows.append(row)
           b.append(d)
   A_ub = np.array(rows)
   b_ub = np.array(b)
   # bounds per variable: (0, None)
   bounds = [(0, None)] * n
   # Solve LP
   res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
   if not res.success:
       # Some failure? Return zeros maybe.
       radii = np.zeros(n)
   else:
       radii = res.x
   sum_radii = np.sum(radii)
   return radii, sum_radii

def hex_grid(radius_spacing, margin):
   # generate rows & columns for positions within [margin, 1-margin].
   positions=[]
   y=margin
   row=0
   vertical_spacing = np.sqrt(3)/2 * radius_spacing
   while y <= 1 - margin + 1e-9:
       # number of columns = floor((1-2*margin)/radius_spacing) + 1
       # Actually we compute x positions: start at margin (+ offset if odd row)
       offset = (row % 2) * radius_spacing/2
       x = margin + offset

           positions.append([x, y])
           x += radius_spacing
       row+=1
       y += vertical_spacing
   return np.array(positions)

def local_search(centers, rng, steps=2000, initial_temp=0.02):
    best_centers = centers.copy()
    best_radii, best_sum = solve_radii(centers)
    curr_centers = centers.copy()
    curr_radii = best_radii
    curr_sum = best_sum
    T = initial_temp
    n = centers.shape[0]
    for k in range(steps):
        i = rng.integers(n)
        # propose a move: random normal shift
        step_size = 0.02  # adjust later
        new_center = curr_centers[i] + rng.normal(scale=step_size, size=2)
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure not too close to boundaries? Not needed.
        old_center = curr_centers[i].copy()
        curr_centers[i] = new_center
        new_radii, new_sum = solve_radii(curr_centers)
        delta = new_sum - curr_sum
        # acceptance
        if delta > 0 or rng.random() < np.exp(delta / T):
            curr_radii = new_radii
            curr_sum = new_sum
            # update best if improved
            if curr_sum > best_sum:
                best_sum = curr_sum
                best_centers = curr_centers.copy()
                best_radii = curr_radii.copy()
        else:
            # revert
            curr_centers[i] = old_center
        # cooling schedule
        T *= 0.9995
    return best_centers, best_radii, best_sum

def solve_radii(centers):
    n = centers.shape[0]
    # distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))
    # build constraints
    rows = []
    bh = []
    # boundary
    xs = centers[:, 0]
    ys = centers[:, 1]
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        # x_i
        rows.append(e.copy())
        bh.append(xs[i])
        # 1-x_i
        rows.append(e.copy())
        bh.append(1.0 - xs[i])
        # y_i
        rows.append(e.copy())
        bh.append(ys[i])
        # 1-y_i
        rows.append(e.copy())
        bh.append(1.0 - ys[i])
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            bh.append(dists[i,j])
    A_ub = np.vstack(rows)
    b_ub = np.array(bh)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii, float(np.sum(radii))

The problem is to choose 26 circle centres inside the unit square and radii that are as large as possible while keeping the circles disjoint and fully inside the square.  
For a fixed set of centre positions the radii that maximise the sum are given by the linear program  

```
max   Σ r_i
s.t.  r_i ≤ x_i, r_i ≤ 1−x_i, r_i ≤ y_i, r_i ≤ 1−y_i          (border constraints)
      r_i + r_j ≤ d_ij                                         (pairwise non‑overlap)
      r_i ≥ 0
```

where `d_ij` is the Euclidean distance between centres *i* and *j*.  
Thus the optimisation problem is “choose the centre positions so that the optimal LP
value is as large as possible”.  The LP is fast (≈ 400 constraints for 26 circles)
and its solution is unique for a given placement.

To search the space of centre positions a lightweight simulated‑annealing /
hill‑climbing scheme is used:

1. **Initial layout** – a regular hexagonal grid with spacing 0.2 (23 points) is
   generated inside a 0.1‑margin square, and the missing 3 points are added at

2. **Local moves** – repeatedly pick a random circle and propose a new centre:
   *with 90 % probability* a small Gaussian perturbation (σ≈0.03) and
   *with 10 % probability* a completely new random position.

   The search is deterministic because the random generator is seeded.

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Given centre coordinates, solve the linear program that maximises the sum
    of radii while respecting border and non‑overlap constraints.
    Returns (radii, sum_of_radii).
    """
    n = centers.shape[0]
    # pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    rows = []
    rhs = []

    # border constraints
    xs = centers[:, 0]
    ys = centers[:, 1]
    for i in range(n):
        e = np.zeros(n)
        e[i] = 1.0
        # x_i
        rows.append(e.copy())
        rhs.append(xs[i])
        # 1 - x_i
        rows.append(e.copy())
        rhs.append(1.0 - xs[i])
        # y_i
        rows.append(e.copy())
        rhs.append(ys[i])
        # 1 - y_i
        rows.append(e.copy())
        rhs.append(1.0 - ys[i])

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            e = np.zeros(n)
            e[i] = 1.0
            e[j] = 1.0
            rows.append(e)
            rhs.append(dists[i, j])

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)

    # maximise Σ r_i  → minimise −Σ r_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method="highs")
    if not res.success:
        # a feasible solution always exists (all radii zero)
        radii = np.zeros(n)
    else:
        radii = res.x

    # tiny safety margin to avoid numerical equality issues
    radii = np.maximum(radii - 1e-12, 0.0)
    return radii, float(np.sum(radii))


def hex_grid(d: float, margin: float) -> np.ndarray:
    """
    Generate a hexagonal lattice inside [margin, 1‑margin]².
    d      – horizontal spacing (distance between neighbour centres)
    margin – minimal distance from the outer square walls
    """
    positions = []
    vert = np.sqrt(3.0) / 2.0 * d
    y = margin
    row = 0
    while y <= 1.0 - margin + 1e-12:
        offset = (row % 2) * d / 2.0
        x = margin + offset
        while x <= 1.0 - margin + 1e-12:
            positions.append([x, y])
            x += d
        row += 1
        y += vert
    return np.array(positions)


def local_search(
    centers: np.ndarray,
    rng: np.random.Generator,
    steps: int = 2000,
    initial_temp: float = 0.05,
) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simulated‑annealing style optimisation of centre positions.
    At each iteration a single centre is perturbed, the LP is solved,
    and the move is accepted according to a Metropolis rule.
    """
    n = centers.shape[0]
    best_centers = centers.copy()
    best_radii, best_sum = solve_radii(best_centers)

    cur_centers = centers.copy()
    cur_radii = best_radii
    cur_sum = best_sum
    T = initial_temp

    for _ in range(steps):
        i = rng.integers(n)

        # 10 % of the time propose a completely new random location
        if rng.random() < 0.1:
            new_center = rng.uniform(0.0, 1.0, size=2)
        else:
            new_center = cur_centers[i] + rng.normal(scale=0.03, size=2)

        new_center = np.clip(new_center, 0.0, 1.0)
        old_center = cur_centers[i].copy()
        cur_centers[i] = new_center

        new_radii, new_sum = solve_radii(cur_centers)
        delta = new_sum - cur_sum

        if delta > 0.0 or rng.random() < np.exp(delta / T):
            # accept move
            cur_radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = cur_centers.copy()
                best_radii = new_radii.copy()
        else:
            # reject – revert
            cur_centers[i] = old_center

        T *= 0.9995  # geometric cooling

    return best_centers, best_radii, best_sum


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Entry point required by the grader.
    Returns (centres, radii, total_sum_of_radii) for 26 circles.
    """
    rng = np.random.default_rng(42)       # deterministic randomness
    n_circles = 26
    target = 2.636

    # a reasonably spread deterministic start – hexagonal grid with spacing 0.2
    base = hex_grid(d=0.20, margin=0.10)   # 23 points

    best_centers = None

    for _ in range(8):
