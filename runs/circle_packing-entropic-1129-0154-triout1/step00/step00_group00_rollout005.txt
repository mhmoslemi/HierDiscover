import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances
    dists = np.linalg.norm(centers[:, None, :] - centers[None, :, :], axis=2)
    # Build A_ub and b_ub
    rows = []
    b = []
    # pair constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            b.append(dists[i, j])
    # boundary constraints
    for i in range(n):
        x, y = centers[i]
        maxr = min(x, 1-x, y, 1-y)
        row = np.zeros(n); row[i] = 1
        rows.append(row); b.append(maxr)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    # objective: maximize sum r -> minimize -c^T r
    c = -np.ones(n)
    bounds = [(0, None) for _ in range(n)]
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # return zero radii
        return np.zeros(n), 0.0
    radii = res.x
    return radii, radii.sum()

def run_packing():
    import numpy as np
    from scipy.optimize import linprog
    n = 26
    # initial arrangement: 5x5 grid + one center
    n_rows = 5
    n_cols = 5
    r0 = 0.1
    xs = np.linspace(r0, 1 - r0, n_cols)
    ys = np.linspace(r0, 1 - r0, n_rows)
    xv, yv = np.meshgrid(xs, ys)
    centers = np.column_stack((xv.ravel(), yv.ravel()))
    # take first 25 circles and add extra at center
    if centers.shape[0] > 25:
        centers = centers[:25]
    centers = np.vstack([centers, np.array([0.5, 0.5])])
    # solve initial
    best_radii, best_sum = solve_radii(centers)
    best_centers = centers.copy()
    np.random.seed(0)
    step = 0.2
    for it in range(3000):
        # propose a move
        new_centers = best_centers.copy()
        i = np.random.randint(n)
        new_centers[i,0] += np.random.normal(scale=step)
        new_centers[i,1] += np.random.normal(scale=step)
        new_centers[i] = np.clip(new_centers[i], 0.0, 1.0)
        # solve
        new_radii, new_sum = solve_radii(new_centers)
        if new_sum > best_sum:
            best_sum = new_sum
            best_radii = new_radii
            best_centers = new_centers.copy()
        # gradually reduce step
        step *= 0.9995
    return best_centers, best_radii, float(best_sum)

    n = centers.shape[0]

    dists = np.sqrt(np.sum(diff**2, axis=2))
    # number constraints
    num_pair = n*(n-1)//2

import numpy as np
from scipy.optimize import linprog

def _solve_radii(centers: np.ndarray):
    """
    For a given set of centre points compute the maximum radii
    (linear programme) and the corresponding sum of radii.
    """
    n = centers.shape[0]
    # pairwise distances
    dif = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.einsum('...i,...i', dif, dif))

    # number of constraints:
    #   - one for each pair (i<j)
    #   - one for the boundary (tightest of the four side constraints)
    n_pair = n * (n - 1) // 2
    n_cons = n_pair + n
    A = np.zeros((n_cons, n), dtype=np.float64)
    b = np.zeros(n_cons, dtype=np.float64)

    row = 0
    # pairwise non‑overlap: r_i + r_j ≤ distance_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # boundary constraints: r_i ≤ min(x_i,1‑x_i,y_i,1‑y_i)
    for i in range(n):
        x, y = centers[i]
        max_r = min(x, 1.0 - x, y, 1.0 - y)
        A[row, i] = 1.0
        b[row] = max_r
        row += 1

    # maximise Σ r_i  → minimise -Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # if the LP fails (should not happen) fall back to zeros
        radii = np.zeros(n, dtype=np.float64)
        total = 0.0
    else:
        radii = np.maximum(res.x, 0.0)
        total = float(radii.sum())
    return radii, total


def _initial_grid():
    """
    Produce 25 points on a 5×5 regular grid with spacing 0.2
    (radius 0.1 would be tight).  The 26‑th point is placed in the centre.
    """
    r = 0.1
    xs = np.linspace(r, 1.0 - r, 5)
    ys = np.linspace(r, 1.0 - r, 5)
    xv, yv = np.meshgrid(xs, ys)
    grid = np.column_stack((xv.ravel(), yv.ravel()))  # 25 points
    centre = np.array([[0.5, 0.5]])
    return np.vstack([grid, centre])  # shape (26,2)


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a short simulated‑annealing search over centre positions.
    Returns (centres, radii, sum_of_radii) for 26 circles.
    """
    np.random.seed(42)        # deterministic result
    n = 26

    # start from the regular grid + centre
    cur_centres = _initial_grid()
    cur_radii, cur_sum = _solve_radii(cur_centres)

    best_centres = cur_centres.copy()
    best_radii = cur_radii.copy()
    best_sum = cur_sum

    # annealing parameters
    temperature = 1.0
    step_size = 0.15
    max_iter = 1200

    for it in range(max_iter):
        # propose a move of a single centre
        new_centres = cur_centres.copy()
        idx = np.random.randint(n)
        new_centres[idx] += np.random.normal(scale=step_size, size=2)
        new_centres[idx] = np.clip(new_centres[idx], 0.0, 1.0)

        new_radii, new_sum = _solve_radii(new_centres)
        delta = new_sum - cur_sum

        # Metropolis acceptance
        accept = False
        if delta > 0:
            accept = True
        else:
            prob = np.exp(delta / temperature) if temperature > 1e-12 else 0.0
            if np.random.rand() < prob:
                accept = True

        if accept:
            cur_centres = new_centres
            cur_radii = new_radii
            cur_sum = new_sum
            # record the best ever seen
            if new_sum > best_sum:
                best_sum = new_sum
                best_centres = new_centres.copy()
                best_radii = new_radii.copy()

        # cooling schedules
        temperature *= 0.9985
        step_size *= 0.9990

    return best_centres, best_radii, float(best_sum)
