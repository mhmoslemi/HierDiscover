def solve_radii(centers):
    n = centers.shape[0]
    c = -np.ones(n) # minimize -sum(r) = maximize sum(r)
    # constraints: A_ub @ r <= b_ub
    # pairwise
    rows = []
    rhs = []
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            d = np.linalg.norm(centers[i] - centers[j])
            rhs.append(d)
    # boundary constraints
    for i in range(n):
        x = centers[i,0]
        y = centers[i,1]
        # r_i <= x
        row = np.zeros(n); row[i] = 1; rows.append(row); rhs.append(x)
        # r_i <= 1 - x
        row = np.zeros(n); row[i] = 1; rows.append(row); rhs.append(1 - x)
        # r_i <= y
        row = np.zeros(n); row[i] = 1; rows.append(row); rhs.append(y)
        # r_i <= 1 - y
        row = np.zeros(n); row[i] = 1; rows.append(row); rhs.append(1 - y)
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    bounds = [(0, None) for _ in range(n)]
    # Solve
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if res.success:
        r = res.x
        # Clip small negative due to numeric
        r = np.maximum(r, 0.0)
        return r, np.sum(r)
    else:
        # Return zeros if fails
        return np.zeros(n), 0.0
```

def solve_lp(centers):
    n = centers.shape[0]
    # compute distances matrix
    dist_matrix = np.sqrt(((centers[:, None, :] - centers[None, :, :])**2).sum(axis=2))
    # constraints
    rows = []
    rhs = []
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.
            row[j] = 1.
            rows.append(row)
            rhs.append(dist_matrix[i, j])
    for i in range(n):
        x = centers[i,0]
        y = centers[i,1]
        row = np.zeros(n); row[i] = 1.; rows.append(row); rhs.append(x)
        row = np.zeros(n); row[i] = 1.; rows.append(row); rhs.append(1-x)
        row = np.zeros(n); row[i] = 1.; rows.append(row); rhs.append(y)
        row = np.zeros(n); row[i] = 1.; rows.append(row); rhs.append(1-y)
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: zero
        r = np.zeros(n)
    else:
        r = np.maximum(res.x, 0.0)
    return r, np.sum(r)
```

    n = centers.shape[0]

    diff = centers[:, None, :] - centers[None, :, :]

    rows = []
    b = []
    # pairwise

            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            b.append(dists[i, j])
    # boundaries
    for i in range(n):
        x, y = centers[i]
        for bound in (x, 1-x, y, 1-y):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            b.append(bound)
    A_ub = np.vstack(rows)
    b_ub = np.array(b)
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,

        total = float(np.sum(radii))

        return np.zeros(n), 0.0

import numpy as np
import scipy.optimize
import time

def _solve_lp(centers: np.ndarray):
    """
    For a fixed set of centre points solve the linear program that
    maximises the sum of radii under non‑overlap and boundary constraints.
    Returns (radii, total_sum).
    """
    n = centers.shape[0]
    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))

    # build A_ub * r <= b_ub
    rows = []
    rhs = []

    # r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        for bound in (x, 1.0 - x, y, 1.0 - y):
            row = np.zeros(n)
            row[i] = 1.0
            rows.append(row)
            rhs.append(bound)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=np.float64)

    c = -np.ones(n)                         # maximise sum(r)  <=> minimise -sum(r)
    bounds = [(0.0, None) for _ in range(n)]

    # use the HiGHS solver – fast and reliable for this size
    res = scipy.optimize.linprog(c, A_ub=A_ub, b_ub=b_ub,
                                 bounds=bounds, method='highs')
    if not res.success:
        # fallback – all radii zero (should never happen)
        return np.zeros(n), 0.0
    radii = np.maximum(res.x, 0.0)
    return radii, float(radii.sum())

def _hill_climb(initial_centers: np.ndarray,
                max_iter: int = 300,
                step_initial: float = 0.15) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Simple hill‑climbing on the centre positions.
    Each proposal moves a single centre, recomputes the optimal radii (LP)
    and keeps the move only when the total radius sum grows.
    """
    centers = initial_centers.copy()
    radii, cur_sum = _solve_lp(centers)
    best_sum = cur_sum
    best_centers = centers.copy()
    best_radii = radii.copy()
    n = centers.shape[0]

    for it in range(max_iter):
        # linearly decreasing step size
        step = step_initial * (1.0 - it / max_iter)
        i = np.random.randint(n)

        proposal = centers[i] + (np.random.rand(2) - 0.5) * 2.0 * step
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centers = centers.copy()
        new_centers[i] = proposal

        new_radii, new_sum = _solve_lp(new_centers)

        if new_sum > cur_sum + 1e-9:          # accept only improvements
            centers = new_centers
            radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

    return best_centers, best_radii, best_sum


def _prune_dense(target_n: int = 26,
                 candidates: int = 70) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Create a dense random point set, then iteratively delete the point with
    the smallest LP‑radius until exactly `target_n` points remain.
    """
    pts = np.random.rand(candidates, 2)
    while pts.shape[0] > target_n:
        radii, _ = _solve_lp(pts)
        # delete the worst (smallest radius)
        del_idx = np.argmin(radii)
        pts = np.delete(pts, del_idx, axis=0)
    final_radii, total = _solve_lp(pts)
    return pts, final_radii, total


def _jittered_grid():
    """
    Produce a 5x5 grid (25 points) with a tiny random perturbation and one
    extra random point to reach 26 points.
    """
    xs = np.linspace(0.1, 0.9, 5)
    ys = np.linspace(0.1, 0.9, 5)
    X, Y = np.meshgrid(xs, ys)
    grid = np.vstack([X.ravel(), Y.ravel()]).T                     # (25,2)

    # add a 26th point at a random location reasonably far from the grid
    extra = np.random.rand(1, 2)
    points = np.vstack([grid, extra])

    # jitter all points a little (±0.04) to break perfect symmetry
    jitter = 0.04
    noise = (np.random.rand(*points.shape) - 0.5) * 2.0 * jitter
    points = np.clip(points + noise, 0.0, 1.0)
    return points


def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Search for a packing of 26 circles inside the unit square
    that maximises the sum of radii.
    Returns (centres, radii, sum_of_radii).
    """
    np.random.seed(0)                     # reproducibility
    best_sum = -1.0
    best_centers = None
    best_radii = None

    time_limit = 8.0                      # seconds
    start = time.time()

    while time.time() - start < time_limit:
        # randomly pick a strategy
        if np.random.rand() < 0.55:
            # hill‑climb from a jittered grid start
            init = _jittered_grid()
            centers, radii, total = _hill_climb(init,
                                                max_iter=400,
                                                step_initial=0.18)
        else:
            # dense‑prune approach
            centers, radii, total = _prune_dense(target_n=26,
                                                candidates=80)

        if total > best_sum:
            best_sum = total
            best_centers = centers.copy()
            best_radii = radii.copy()
            # stop early if we have clearly passed the target
            if best_sum >= 2.636 - 1e-6:
                break

    # final sanity – ensure we really have 26 circles
    assert best_centers.shape == (26, 2)
    assert best_radii.shape == (26,)

    return best_centers, best_radii, float(best_sum)
