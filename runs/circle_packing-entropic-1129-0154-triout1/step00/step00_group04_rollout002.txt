import numpy as np
from scipy.optimize import minimize

def make_constraints(N: int):
    """Return a list of SLSQP constraint dictionaries for the packing problem."""
    cons = []

    # boundary and non‑negativity constraints for each circle
    for i in range(N):
        # x_i - r_i >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2 * i] - x[2 * N + i]
        })
        # 1 - x_i - r_i >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2 * i] - x[2 * N + i]
        })
        # y_i - r_i >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2 * i + 1] - x[2 * N + i]
        })
        # 1 - y_i - r_i >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: 1.0 - x[2 * i + 1] - x[2 * N + i]
        })
        # r_i >= 0
        cons.append({
            'type': 'ineq',
            'fun': lambda x, i=i: x[2 * N + i]
        })

    # pairwise non‑overlap constraints
    for i in range(N):
        for j in range(i + 1, N):
            cons.append({
                'type': 'ineq',
                'fun': lambda x, i=i, j=j: (
                    (x[2 * i] - x[2 * j]) ** 2 +
                    (x[2 * i + 1] - x[2 * j + 1]) ** 2 -
                    (x[2 * N + i] + x[2 * N + j]) ** 2
                )
            })
    return cons


def initial_guess(N: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a feasible starting vector (centres + radii).
    Four circles are placed near the corners, the rest on a regular grid.
    """
    # small initial radius for every circle
    r0 = 0.02

    # corner circles – a little away from the exact corner to keep a margin
    corner_margin = 0.15
    fixed = [
        (corner_margin, corner_margin),
        (1.0 - corner_margin, corner_margin),
        (corner_margin, 1.0 - corner_margin),
        (1.0 - corner_margin, 1.0 - corner_margin)
    ]

    # number of circles that still have to be placed
    remaining = N - len(fixed)

    # a simple rectangular grid for the remaining points
    cols = 6
    rows = 5
    xs = np.linspace(r0, 1.0 - r0, cols)
    ys = np.linspace(r0, 1.0 - r0, rows)

    grid_pts = [(float(x), float(y)) for y in ys for x in xs]

    # discard grid points that are too close to the pre‑placed corner circles
    filtered = []
    for pt in grid_pts:
        too_close = False
        for fx, fy in fixed:
            if np.hypot(pt[0] - fx, pt[1] - fy) < 0.12:
                too_close = True
                break
        if not too_close:
            filtered.append(pt)
        if len(filtered) >= remaining:
            break

    points = fixed + filtered[:remaining]
    points = np.array(points, dtype=np.float64)

    # add a tiny random perturbation so that repeated restarts explore different
    # regions of the search space
    perturb = 0.02 * (rng.random(points.shape) - 0.5)
    points += perturb
    points = np.clip(points, r0, 1.0 - r0)

    radii = np.full(N, r0, dtype=np.float64)

    return np.concatenate([points.ravel(), radii])


def expand_radii(centers: np.ndarray, radii: np.ndarray) -> np.ndarray:
    """
    Greedy enlargement of radii for a *fixed* set of centres.
    For each circle we set its radius to the largest value that still respects
    the wall constraints and the current radii of the other circles.
    The procedure is iterated a few times until no radius changes appreciably.
    """
    N = len(radii)
    radii = radii.copy()

    for _ in range(5):          # a small fixed number of passes is sufficient
        changed = False
        for i in range(N):
            # distance to the four walls
            max_r = min(
                centers[i, 0],          # left wall
                1.0 - centers[i, 0],    # right wall
                centers[i, 1],          # bottom wall
                1.0 - centers[i, 1]     # top wall
            )
            # distance to all other circles
            for j in range(N):
                if i == j:
                    continue
                d = np.linalg.norm(centers[i] - centers[j])
                max_r = min(max_r, d - radii[j])
            max_r = max(max_r, 0.0)   # safety
            if max_r > radii[i] + 1e-9:
                radii[i] = max_r
                changed = True
        if not changed:
            break
    return radii

def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns
    -------
    centers : np.ndarray shape (26,2)
        x‑ and y‑coordinates of the circle centres.
    radii   : np.ndarray shape (26,)
        radii of the circles.
    sum_radii : float
        total sum of all radii (the objective value).
    """
    N = 26
    rng = np.random.default_rng(12345)

    # SLSQP constraints and variable bounds
    constraints = make_constraints(N)
    bounds = [(0.0, 1.0)] * (2 * N) + [(0.0, 0.5)] * N  # radii can never exceed 0.5

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # Several independent optimisation runs – keep the best result
    for attempt in range(15):
        x0 = initial_guess(N, rng)

        # Objective: maximise sum(radii)  <=> minimise -sum(radii)
        def objective(v):
            return -np.sum(v[2 * N:])

        res = minimize(
            fun=objective,
            x0=x0,
            method='SLSQP',
            bounds=bounds,
            constraints=constraints,
            options={'maxiter': 2000, 'ftol': 1e-9, 'disp': False}
        )

        if not res.success:
            continue

        # Retrieve centres and radii from the solution
        sol_vars = res.x
        centres = sol_vars[:2 * N].reshape(N, 2)
        radii = sol_vars[2 * N:]

        # Greedy enlargement of radii (with fixed centres)
        radii = expand_radii(centres, radii)

        # Safety shrink to guarantee a strict feasible point for the validator
        radii *= 0.9995

        total = float(np.sum(radii))
        if total > best_sum:
            best_sum = total
            best_centers = centres.copy()
            best_radii = radii.copy()

    # -----------------------------------------------------------------
    # Fallback – a simple regular grid (used only if optimisation never succeeded)
    # -----------------------------------------------------------------
    if best_centers is None:
        # 5 × 5 grid (25 circles) with radius 0.1, plus one tiny extra circle
        grid = np.linspace(0.1, 0.9, 5)

        best_centers = np.vstack([centers_grid, extra_center])
        best_radii = np.concatenate([radii_grid, extra_radius])
        best_sum = float(best_radii.sum())

    return best_centers, best_radii, best_sum
