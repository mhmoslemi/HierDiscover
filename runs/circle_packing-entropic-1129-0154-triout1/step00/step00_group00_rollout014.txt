def compute_sum_radii(centers):
    # returns radii, sum
    N = centers.shape[0]

    # compute distances
    d_mat = np.linalg.norm(centers[:,None,:] - centers[None,:,:], axis=2)
    # get pairwise distances i<j

    # build A_ub and b_ub
    m = 4*N + num_pairs
    A = np.zeros((m, N))
    b = np.zeros(m)
    row = 0
    # left boundary: r_i <= x_i
    for i in range(N):
        A[row, i] = 1.0
        b[row] = centers[i,0]
        row += 1
    # right
    for i in range(N):
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i,0]
        row += 1
    # bottom
    for i in range(N):
        A[row, i] = 1.0
        b[row] = centers[i,1]
        row += 1
    # top
    for i in range(N):
        A[row, i] = 1.0
        b[row] = 1.0 - centers[i,1]
        row += 1
    # non-overlap
    for (i,j) in pair_indices:
        A[row, i] = 1.0
        A[row, j] = 1.0
        b[row] = d_mat[i,j]
        row += 1

    c = -np.ones(N) # maximize sum r => minimize -sum

    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*N, method='highs')
    if not res.success:
        # return zeros maybe
        radii = np.zeros(N)
        sum_r = 0.0
    else:
        radii = res.x
        sum_r = np.sum(radii)
    return radii, sum_r

```
def compute_radii(centers):
    import numpy as np
    from scipy.optimize import linprog

    N = centers.shape[0]
    # pairwise dist matrix
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff ** 2, axis=2))
    # Build constraints
    # number of constraints
    num_pairs = N * (N - 1) // 2
    m = 4 * N + num_pairs
    A = np.zeros((m, N))
    b = np.zeros(m)
    row = 0
    # left bound
    A[row:row+N, np.arange(N)] = 1
    b[row:row+N] = centers[:, 0]
    row += N
    # right bound
    A[row:row+N, np.arange(N)] = 1
    b[row:row+N] = 1 - centers[:, 0]
    row += N
    # bottom bound
    A[row:row+N, np.arange(N)] = 1
    b[row:row+N] = centers[:, 1]
    row += N
    # top bound
    A[row:row+N, np.arange(N)] = 1
    b[row:row+N] = 1 - centers[:, 1]
    row += N
    # pair constraints
    for i in range(N):
        for j in range(i+1, N):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1
    c = -np.ones(N)
    bounds = [(0, None)] * N
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs', options={'presolve': True})
    if not res.success:
        # fallback: zero radii
        radii = np.zeros(N)
    else:
        radii = res.x
    # tiny safety margin
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)
    sum_r = np.sum(radii)
    return radii, sum_r
```

def run_packing():
    import numpy as np
    from scipy.optimize import linprog

    N = 26
    # base positions: 5x5 grid + extra point at (0.2, 0.2)
    xs = np.array([0.1,0.3,0.5,0.7,0.9])
    grid = np.array([(x,y) for y in xs for x in xs])
    # Use 25 grid; maybe include extra point random later.

    best_sum = -np.inf
    best_centers = None
    best_radii = None

    # multiple restarts
    for restart in range(20):
        # start from grid with jitter small
        jitter_scale = 0.05
        centers = grid + np.random.uniform(-jitter_scale, jitter_scale, size=grid.shape)
        centers = np.clip(centers, 0.0, 1.0)
        # add extra point random
        extra = np.random.rand(1,2)
        centers = np.vstack([centers, extra])

        # compute radii
        radii, sum_r = compute_radii(centers)

        # improve by local adjustments
        curr_centers = centers.copy()
        curr_radii = radii.copy()
        curr_sum = sum_r

        for iter in range(500):
            # pick random index
            i = np.random.randint(N)
            # propose new location
            step = np.random.normal(scale=0.05, size=2) # can also adapt scale
            new_center = curr_centers[i] + step
            new_center = np.clip(new_center, 0.0, 1.0)
            new_centers = curr_centers.copy()
            new_centers[i] = new_center
            # compute new radii
            new_radii, new_sum = compute_radii(new_centers)
            # accept if better
            if new_sum > curr_sum:
                curr_centers = new_centers
                curr_radii = new_radii
                curr_sum = new_sum
        # after local search, compare with global best
        if curr_sum > best_sum:
            best_sum = curr_sum
            best_centers = curr_centers
            best_radii = curr_radii
        # maybe early exit if sum > target
        if best_sum >= 2.636:
            break
    return best_centers, best_radii, float(best_sum)

import numpy as np
from scipy.optimize import linprog

def _optimal_radii(centers: np.ndarray):
    """
    Given centre coordinates (N,2) return the radii that maximise the sum of radii
    under the non‑overlap and boundary constraints.
    """
    N = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of pairwise constraints
    pair_i, pair_j = np.triu_indices(N, k=1)
    P = pair_i.size

    # total number of inequality rows
    M = 4 * N + P
    A = np.zeros((M, N), dtype=np.float64)
    b = np.zeros(M, dtype=np.float64)

    # boundary constraints  r_i <= x_i,  r_i <= 1-x_i, r_i <= y_i, r_i <= 1-y_i
    rows = np.arange(N)
    A[rows, rows] = 1.0               # r_i <= x_i
    b[rows] = centers[:, 0]

    start = N
    A[start:start + N, rows] = 1.0    # r_i <= 1 - x_i
    b[start:start + N] = 1.0 - centers[:, 0]

    start += N
    A[start:start + N, rows] = 1.0    # r_i <= y_i
    b[start:start + N] = centers[:, 1]

    start += N
    A[start:start + N, rows] = 1.0    # r_i <= 1 - y_i
    b[start:start + N] = 1.0 - centers[:, 1]

    # pairwise non‑overlap  r_i + r_j <= d_ij
    start += N
    pair_rows = np.arange(P) + start
    A[pair_rows, pair_i] = 1.0
    A[pair_rows, pair_j] = 1.0
    b[pair_rows] = dists[pair_i, pair_j]

    # maximise sum(r_i)  <=> minimise -sum(r_i)
    c = -np.ones(N, dtype=np.float64)
    bounds = [(0.0, None)] * N

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                  method='highs', options={'presolve': True})
    if not res.success:
        # fallback – all radii zero (should never happen)
        radii = np.zeros(N, dtype=np.float64)
    else:
        # tiny safety margin to stay away from the 1e-12 tolerance
        radii = np.maximum(res.x - 1e-9, 0.0)

    return radii, float(radii.sum())

        radii, total = _optimal_radii(centers)
        return total, radii

    best_total = -1.0
    best_centers = None
    best_radii = None

            if cur_total > best_total:
                best_total = cur_total
                best_centers = cur_centers.copy()
                best_radii = cur_radii.copy()

        # early stop if we already pass the required threshold
        if best_total >= 2.636:
            break

    # final safety: ensure radii are non‑negative and respect bounds
    eps = 1e-9
    best_radii = np.maximum(best_radii - eps, 0.0)

    return best_centers, best_radii, float(best_total)
