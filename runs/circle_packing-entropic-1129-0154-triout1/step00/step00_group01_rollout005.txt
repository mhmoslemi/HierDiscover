def boundary_cons(vars):
    xs = vars[0::3]; ys = vars[1::3]; rs = vars[2::3];
    c = np.concatenate([xs - rs,
                        1 - xs - rs,
                        ys - rs,
                        1 - ys - rs])
    return c

def pairwise_cons(vars):
    xs = vars[0::3]; ys = vars[1::3]; rs = vars[2::3];
    n = len(rs)
    # Build all pairs
    # We can compute pairwise difference matrices.
    # Use broadcasting.
    dx = xs[:,None] - xs[None,:]  # shape (n,n)
    dy = ys[:,None] - ys[None,:]
    d2 = dx**2 + dy**2
    rsum = rs[:,None] + rs[None,:]
    # For i<j, compute d2 - (rsum)^2
    # We'll take upper triangular entries, flatten.
    tril_indices = np.triu_indices(n, k=1)
    val = d2[tril_indices] - (rsum[tril_indices])**2
    return val

def solve_radii(centers):
    n = centers.shape[0]
    # Build constraints
    # boundary constraints:
    xs = centers[:,0]
    ys = centers[:,1]
    # bound_i = np.minimum.reduce([xs, 1 - xs, ys, 1 - ys])
    bounds_rad = np.minimum(np.minimum(xs, 1 - xs), np.minimum(ys, 1 - ys))

    # Pairwise distances matrix
    dists = np.sqrt(((xs[:,None] - xs[None,:])**2 + (ys[:,None] - ys[None,:])**2))
    # Extract upper triangular distances
    triu_idx = np.triu_indices(n, k=1)
    d_ij = dists[triu_idx]

    # Build A_ub and b_ub
    # For each i (0..n-1): r_i <= bounds_rad[i]
    # For each pair (i,j): r_i + r_j <= d_ij
    m_constraints = 4*n + len(d_ij)
    A = np.zeros((m_constraints, n))
    b = np.zeros(m_constraints)
    # boundary constraints rows 0..4n-1
    for i in range(n):
        # x - r >= 0 => r <= x, we add row for r_i <= xs[i]
        A[4*i, i] = 1
        b[4*i] = xs[i]
        # 1 - x - r >=0 => r <= 1 - x
        A[4*i+1, i] = 1
        b[4*i+1] = 1 - xs[i]
        # y - r >=0 => r <= y
        A[4*i+2, i] = 1
        b[4*i+2] = ys[i]
        # 1 - y - r >=0 => r <= 1 - y
        A[4*i+3, i] = 1
        b[4*i+3] = 1 - ys[i]

    # pair constraints rows start at 4*n
    start = 4*n
    for idx, (i, j) in enumerate(zip(triu_idx[0], triu_idx[1])):
        A[start + idx, i] = 1
        A[start + idx, j] = 1
        b[start + idx] = d_ij[idx]

    # Linear program: maximize sum r_i => minimize -c^T r, where c = ones
    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: set radii zeros
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii

def solve_radii(centers):
    n = centers.shape[0]
    xs = centers[:,0]
    ys = centers[:,1]
    # bound for each radius due to boundaries
    boundX = np.minimum(xs, 1 - xs)
    boundY = np.minimum(ys, 1 - ys)
    bounds_rad = np.minimum(boundX, boundY)  # each radius <= min distance to any side

    # number of constraints
    m_boundary = 4 * n
    # pairwise distances
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    d = np.sqrt(dx * dx + dy * dy)
    triu = np.triu_indices(n, k=1)
    d_ij = d[triu]

    m_pair = d_ij.size
    m_total = m_boundary + m_pair

    A = np.zeros((m_total, n))
    b = np.zeros(m_total)

    # boundary constraints
    for i in range(n):
        # r_i <= x_i
        A[4*i, i] = 1.0
        b[4*i] = xs[i]
        # r_i <= 1 - x_i
        A[4*i + 1, i] = 1.0
        b[4*i + 1] = 1.0 - xs[i]
        # r_i <= y_i
        A[4*i + 2, i] = 1.0
        b[4*i + 2] = ys[i]
        # r_i <= 1 - y_i
        A[4*i + 3, i] = 1.0
        b[4*i + 3] = 1.0 - ys[i]

    # pair constraints r_i + r_j <= d_ij
    start = m_boundary
    for idx, (i, j) in enumerate(zip(triu[0], triu[1])):
        A[start + idx, i] = 1.0
        A[start + idx, j] = 1.0
        b[start + idx] = d_ij[idx]

    # Linear program to maximize sum of r_i
    c = -np.ones(n)   # minimize -sum r_i
    # bounds for each r_i
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if res.success:
        radii = res.x
    else:
        # fall back to zeros to keep feasibility
        radii = np.zeros(n)

    # slight shrink to avoid numerical equality issues
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)
    return radii

import numpy as np
from scipy.optimize import linprog
from typing import Tuple

# ------------------------------------------------------------
# Linear program: for a given set of centres compute the
# maximal radii that satisfy all constraints.
# ------------------------------------------------------------
def _optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises the sum of radii for fixed centre positions.
    Parameters
    ----------
    centers : (n,2) array
        x‑y coordinates of the circle centres.

    Returns
    -------
    radii : (n,) array
        Radii that maximise Σr_i under the non‑overlap and boundary constraints.
    """
    n = centers.shape[0]
    xs = centers[:, 0]
    ys = centers[:, 1]

    # ----- boundary limits for each radius -----
    # Each radius must be ≤ distance to every side of the square.
    # The bound for circle i is   b_i = min(x_i, 1‑x_i, y_i, 1‑y_i)
    bound_x_low = xs
    bound_x_up = 1.0 - xs
    bound_y_low = ys
    bound_y_up = 1.0 - ys

    # ----- pairwise distance constraints (r_i + r_j ≤ d_ij) -----
    # Compute the full distance matrix once.
    dx = xs[:, None] - xs[None, :]
    dy = ys[:, None] - ys[None, :]
    d_mat = np.sqrt(dx * dx + dy * dy)

    # Indices of the upper‑triangular part (i<j)
    iu, ju = np.triu_indices(n, k=1)
    d_ij = d_mat[iu, ju]          # shape (n_pairs,)

    # ----- Build the linear program matrices -----
    # Number of inequality rows:
    #   4 per circle for the four side constraints + one per unordered pair.
    m_boundary = 4 * n
    m_pair = d_ij.size
    m_total = m_boundary + m_pair

    A = np.zeros((m_total, n))
    b = np.zeros(m_total)

    # 4 boundary constraints for each circle i
    for i in range(n):
        row = 4 * i
        A[row, i] = 1.0          # r_i ≤ x_i
        b[row] = bound_x_low[i]

        A[row + 1, i] = 1.0      # r_i ≤ 1‑x_i
        b[row + 1] = bound_x_up[i]

        A[row + 2, i] = 1.0      # r_i ≤ y_i
        b[row + 2] = bound_y_low[i]

        A[row + 3, i] = 1.0      # r_i ≤ 1‑y_i
        b[row + 3] = bound_y_up[i]

    # Pairwise non‑overlap constraints
    start = m_boundary
    for idx, (i, j) in enumerate(zip(iu, ju)):
        A[start + idx, i] = 1.0
        A[start + idx, j] = 1.0
        b[start + idx] = d_ij[idx]

    # Objective: maximise Σ r_i  → minimise –Σ r_i
    c = -np.ones(n)

    # Variable bounds: r_i ≥ 0, no explicit upper bound (the A‑matrix already limits it)
    var_bounds = [(0.0, None) for _ in range(n)]

    # Solve the LP with the HiGHS interior‑point solver (fast for this size)
    res = linprog(c, A_ub=A, b_ub=b, bounds=var_bounds, method="highs")
    if not res.success:
        # If the solver fails (should be very rare) fall back to zero radii
        radii = np.zeros(n)
    else:
        radii = res.x

    # A tiny safety margin – guarantees strict feasibility for the validator
    radii = np.maximum(radii - 1e-9, 0.0)
    return radii


# ------------------------------------------------------------
# Stochastic hill‑climbing that moves the centres and always
# re‑optimises the radii via the LP above.
# ------------------------------------------------------------
def _hill_climb(
    init_centers: np.ndarray,
    steps: int = 3000,
    init_step: float = 0.2,
    seed: int = 0,
) -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Perform a simple random‑walk hill‑climb on the centre positions.
    Returns the best centres found, the corresponding optimal radii,
    and the achieved total radius sum.
    """
    rng = np.random.default_rng(seed)
    n = init_centers.shape[0]
    centres = init_centers.copy()

    # Radii for the starting layout
    radii = _optimal_radii(centres)
    best_sum = radii.sum()
    best_centres = centres.copy()
    best_radii = radii.copy()

    step = init_step
    for it in range(steps):
        # Reduce step size periodically for fine‑tuning
        if it % 500 == 0 and it > 0:
            step *= 0.8

        i = rng.integers(n)                     # which circle to move
        delta = (rng.random(2) - 0.5) * 2 * step
        new_pos = centres[i] + delta
        new_pos = np.clip(new_pos, 0.0, 1.0)     # enforce staying inside the square

        if np.allclose(new_pos, centres[i]):
            continue  # no actual move

        old_pos = centres[i].copy()
        centres[i] = new_pos

        new_radii = _optimal_radii(centres)
        new_sum = new_radii.sum()

        if new_sum > best_sum + 1e-8:            # accept only genuine improvements
            best_sum = new_sum
            best_centres = centres.copy()
            best_radii = new_radii.copy()
            # keep the move (centres already updated)
        else:
            # reject – revert the move
            centres[i] = old_pos

    return best_centres, best_radii, float(best_sum)

# Helper to create a reasonably spread initial layout.
# ------------------------------------------------------------
def _initial_grid_with_jitter(N: int, rng: np.random.Generator) -> np.ndarray:

    jitter = (rng.random(pts.shape) - 0.5) * 0.05
    pts = np.clip(pts + jitter, 0.0, 1.0)
    return pts

# ------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Returns
    -------
    centres : (26,2) ndarray
        x‑y coordinates of the 26 circles.
    radii   : (26,)   ndarray
        Corresponding radii (non‑negative, non‑overlapping, inside the unit square).
    total   : float
        Sum of all radii (the optimisation objective).
    """
    N = 26
    rng = np.random.default_rng(42)   # deterministic seed for reproducibility

    best_total = -1.0
    best_centres = None
    best_radii = None

    # Perform several independent restarts – both grid‑based and fully random.
    for restart in range(5):
        # 1) Grid + jitter start
        init_grid = _initial_grid_with_jitter(N, rng)
        centres, radii, total = _hill_climb(init_grid, steps=3000,

        if total2 > best_total:
            best_total = total2
            best_centres = centres2
            best_radii = radii2

    # One final LP pass on the best centre set to ensure the radii are truly optimal.
    final_radii = _optimal_radii(best_centres)
    final_total = float(final_radii.sum())

    return best_centres, final_radii, final_total
