```
def solve_radii(centers):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r >= 0]
    # boundary constraints
    constraints += [r <= centers[:,0]]
    constraints += [r <= 1 - centers[:,0]]
    constraints += [r <= centers[:,1]]
    constraints += [r <= 1 - centers[:,1]]
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d)
    objective = cp.Maximize(cp.sum(r))
    prob = cp.Problem(objective, constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    # The solver may leave r_i as np.ndarray
    radii = np.maximum(r.value, 0.0)   # ensure nonnegative
    return radii, float(np.sum(radii))
```

```
def solve_radii(centers):
    n = centers.shape[0]
    # bounds to boundaries
    bound_vals = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Build constraints
    rows = []
    rhs = []
    # boundary constraints
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(bound_vals[i])
    # pairwise
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            d = np.linalg.norm(centers[i] - centers[j])
            rhs.append(d)
    A = np.vstack(rows)
    b = np.array(rhs)
    c = -np.ones(n)
    # Solve
    res = scipy.optimize.linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback maybe zero radii
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = res.x
        # clip small negative due to tolerance
        radii = np.maximum(radii, 0.0)
    sum_r = float(radii.sum())
    return radii, sum_r
```

```
def solve_radii(centers):
    n=centers.shape[0]
    bound_vals = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # Pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dist_mat = np.sqrt(np.sum(diff**2, axis=2))
    # Extract upper triangular distances
    iu = np.triu_indices(n, k=1)
    distances = dist_mat[iu]

    # Build A matrix
    # first n rows: boundary constraints
    A = np.zeros((n + distances.shape[0], n))
    b = np.empty(n + distances.shape[0])

    # boundary rows
    for i in range(n):
        A[i,i] = 1.0
        b[i] = bound_vals[i]

    # pairwise rows
    start = n
    for idx, (i,j) in enumerate(zip(iu[0], iu[1])):
        A[start+idx, i] = 1.0
        A[start+idx, j] = 1.0
        b[start+idx] = distances[idx]

    c = -np.ones(n)
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        radii = np.zeros(n)
    else:
        radii = res.x
    # Clip to constraints
    radii = np.maximum(radii, 0.0)
    radii = np.minimum(radii, bound_vals)
    # Subtract epsilon
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)
    sum_r = float(radii.sum())
    return radii, sum_r
```

```python
import numpy as np
from scipy.optimize import linprog
import time

# ----------------------------------------------------------------------
# Helper: solve the linear programme for a fixed set of centres
# ----------------------------------------------------------------------
def solve_radii(centres: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Given centre coordinates (n,2), compute the radii that maximise the sum
    of radii while respecting all constraints.
    Returns (radii, sum_of_radii).  Radii are post‑processed to be safely
    inside the feasible region.
    """
    n = centres.shape[0]

    # distance to the four sides of the square
    bound_vals = np.minimum.reduce(
        [centres[:, 0], 1.0 - centres[:, 0], centres[:, 1], 1.0 - centres[:, 1]]
    )

    # pairwise distances
    iu = np.triu_indices(n, k=1)
    diff = centres[iu[0]] - centres[iu[1]]
    pair_dists = np.sqrt(np.einsum("ij,ij->i", diff, diff))

    # Build A_ub and b_ub for the LP:
    #   r_i                     <= bound_vals[i]
    #   r_i + r_j               <= pair_dists[k]
    n_boundary = n
    n_pairs = pair_dists.shape[0]
    total_constraints = n_boundary + n_pairs

    A = np.zeros((total_constraints, n), dtype=np.float64)
    b = np.empty(total_constraints, dtype=np.float64)

    # boundary rows
    for i in range(n):
        A[i, i] = 1.0
        b[i] = bound_vals[i]

    # pairwise rows
    start = n_boundary
    A[start:start + n_pairs, iu[0]] = 1.0
    A[start:start + n_pairs, iu[1]] = 1.0
    b[start:start + n_pairs] = pair_dists

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    res = linprog(
        c,
        A_ub=A,
        b_ub=b,
        bounds=[(0.0, None)] * n,
        method="highs",
        options={"presolve": True},
    )

    if not res.success:
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = np.maximum(res.x, 0.0)

    # Clip to the exact boundary limits (protect against tiny numerical overshoot)
    radii = np.minimum(radii, bound_vals)

    # Subtract a tiny epsilon so that the validator’s 1e-12 tolerance is safely met
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)

    return radii, float(radii.sum())

# Helper: generate a jittered triangular lattice with at least n points
# ----------------------------------------------------------------------
def init_lattice(n: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce an initial set of n centre points.
    A triangular lattice (spacing ≈0.2) is created, jittered a little,
    then n points are sampled from it.
    """
    d = 0.20                      # horizontal lattice spacing
    dy = d * np.sqrt(3.0) / 2.0   # vertical spacing for a triangular lattice

    points = []
    y = 0.0
    row = 0
    while y <= 1.0:
        offset = 0.0 if (row % 2 == 0) else d / 2.0
        x = offset
        while x <= 1.0:
            points.append([x, y])
            x += d
        row += 1
        y += dy

    points = np.array(points, dtype=np.float64)

    # add a small random jitter so the search space is not perfectly regular
    jitter = rng.uniform(-0.02, 0.02, size=points.shape)
    points += jitter
    points = np.clip(points, 0.0, 1.0)

    if points.shape[0] >= n:
        chosen = rng.choice(points.shape[0], n, replace=False)
        centres = points[chosen]
    else:
        # not enough points – fill the remainder with uniform random points
        extra = rng.random((n - points.shape[0], 2))
        centres = np.vstack([points, extra])
        chosen = rng.choice(centres.shape[0], n, replace=False)
        centres = centres[chosen]

    return centres

    """
    rng = np.random.default_rng()

    TARGET = 2.636        # we stop as soon as we reach (or beat) this value

    best_sum = -1.0

    # optimisation parameters
    n_restarts = 8               # independent random starts

        cur_radii = radii.copy()

        # update global best

            best_radii = cur_radii.copy()
            if best_sum >= TARGET:

    return best_centres, best_radii, best_sum
```
