def hill_climb(initial_centers, iterations=2000, step=0.05):
    centers = initial_centers.copy()
    radii, sum_rad = get_sum_radii(centers)
    best_sum = sum_rad
    best_centers = centers.copy()
    best_radii = radii.copy()
    for it in range(iterations):
        # pick random index to perturb
        i = np.random.randint(len(centers))
        # propose new position by adding random delta uniform -step to step
        new_center = centers[i] + np.random.uniform(-step, step, size=2)
        # clip new center into [0,1] (ensuring not out)
        new_center = np.clip(new_center, 0.0, 1.0)
        # ensure some margin to allow radius >0 maybe ensure not at boundary exactly? But we can accept.
        old_center = centers[i].copy()
        centers[i] = new_center
        # compute new radii & sum
        rad_new, sum_new = get_sum_radii(centers)
        if sum_new > best_sum:
            best_sum = sum_new
            best_centers = centers.copy()
            best_radii = rad_new.copy()
            # keep change
        else:
            # revert
            centers[i] = old_center
    return best_centers, best_radii, best_sum

def precompute_constraints(n):
    pairs = [(i,j) for i in range(n) for j in range(i+1, n)]
    num_constraints = n + len(pairs)
    A_ub = np.zeros((num_constraints, n), dtype=np.float64)
    # variable bounds rows
    for i in range(n):
        A_ub[i, i] = 1.0
    # pairwise rows
    for idx, (i,j) in enumerate(pairs):
        A_ub[n+idx, i] = 1.0
        A_ub[n+idx, j] = 1.0
    return A_ub, pairs

def optimal_radii(centers, A_ub, pairs):
    n = centers.shape[0]
    # compute bound per circle
    bound = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    # Compute distances for pairs
    # Use np.linalg.norm for each pair
    b = np.empty(A_ub.shape[0], dtype=np.float64)
    b[:n] = bound
    # fill pairwise part
    # Vectorized: we could compute all pair distances at once using loops
    for idx, (i, j) in enumerate(pairs):
        b[n+idx] = np.linalg.norm(centers[i] - centers[j])
    # Solve LP: maximize sum r => minimize -sum r
    c = -np.ones(n, dtype=np.float64)
    res = linprog(c, A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # Infeasible? fallback to zeros
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = res.x
    total = float(radii.sum())
    return radii, total

def optimal_radii(centers, A_ub, pairs):

    _SCIPY_OK = False

if not _SCIPY_OK:                     # pragma: no cover
    import cvxpy as cp

# --------------------------------------------------------------------------- #
# Pre‑compute the constant part of the linear program (matrix A_ub)
# --------------------------------------------------------------------------- #
def _precompute_constraints(n_circles: int):
    """Return (A_ub, pair_list) for a given number of circles."""
    pairs = [(i, j) for i in range(n_circles) for j in range(i + 1, n_circles)]
    n_pairs = len(pairs)
    n_constraints = n_circles + n_pairs                # one row per circle + one per pair
    A = np.zeros((n_constraints, n_circles), dtype=np.float64)

    # circle–boundary rows: r_i <= bound_i  -> coefficient 1 on r_i
    for i in range(n_circles):
        A[i, i] = 1.0

    # pair rows: r_i + r_j <= d_ij  -> coefficients 1 on both variables
    for idx, (i, j) in enumerate(pairs):
        A[n_circles + idx, i] = 1.0
        A[n_circles + idx, j] = 1.0

    return A, pairs


# --------------------------------------------------------------------------- #
# Solve the linear programme for a fixed set of centres
# --------------------------------------------------------------------------- #
def _solve_lp(c, A_ub, b_ub, bounds):
    """Solve the LP – maximise c·x (implemented as minimise -c·x)."""
    if _SCIPY_OK:                               # pragma: no branch
        res = linprog(-c, A_ub=A_ub, b_ub=b_ub,
                      bounds=bounds, method='highs')
        if res.success:
            return res.x
        # fall back to CVXPY if SciPy failed
    # ------------------- CVXPY fallback ------------------- #
    n = len(c)
    x = cp.Variable(n)
    constraints = [x >= 0]
    for i in range(A_ub.shape[0]):
        constraints.append(A_ub[i] @ x <= b_ub[i])
    prob = cp.Problem(cp.Maximize(c @ x), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True)
    if x.value is not None:
        return np.array(x.value).reshape(-1)
    # if everything fails, return zeros
    return np.zeros(n, dtype=np.float64)


def optimal_radii(centers: np.ndarray, A_ub: np.ndarray, pairs):
    """
    For the given centre positions compute the radii that maximise the sum.
    Returns (radii, total_sum).
    """
    n = centers.shape[0]

    # distance to the four sides → wall bound for each circle
    wall_bound = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0],
         centers[:, 1], 1.0 - centers[:, 1]]
    )

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]   # (n,n,2)
    dmat = np.sqrt(np.sum(diff * diff, axis=2))

    # right‑hand side vector for the LP
    b = np.empty(A_ub.shape[0], dtype=np.float64)
    b[:n] = wall_bound
    for idx, (i, j) in enumerate(pairs):
        b[n + idx] = dmat[i, j]

    # objective: maximise sum(r)  (linprog does minimise, so we pass -c)
    c = np.ones(n, dtype=np.float64)          # we will maximise c·r
    bounds = [(0.0, None)] * n

    radii = _solve_lp(c, A_ub, b, bounds)

    # tiny safety clipping – guarantees validation tolerance
    eps = 1e-9
    radii = np.maximum(radii, 0.0)
    radii = np.minimum(radii, wall_bound - eps)

    # (the LP already respects pairwise constraints up to solver tolerance)
    total = float(radii.sum())
    return radii, total


# --------------------------------------------------------------------------- #
# Simple hill‑climbing optimiser
# --------------------------------------------------------------------------- #
def hill_climb(initial_centers: np.ndarray, A_ub: np.ndarray, pairs,
               iterations: int = 1500,
               step: float = 0.18,
               target: float = None):
    """
    Perform a stochastic hill‑climbing search on the centre positions.
    Returns the best (centres, radii, sum_radii) found.
    """
    n = initial_centers.shape[0]
    centres = initial_centers.copy()

    radii, cur_sum = optimal_radii(centres, A_ub, pairs)
    best_sum = cur_sum
    best_centres = centres.copy()
    best_radii = radii.copy()

    cur_step = step
    for it in range(iterations):
        i = np.random.randint(n)

        # propose a small random move, keep inside the unit square
        delta = (np.random.rand(2) - 0.5) * 2.0 * cur_step
        new_pos = centres[i] + delta
        new_pos = np.clip(new_pos, 0.0, 1.0)

        old_pos = centres[i].copy()
        centres[i] = new_pos

        new_radii, new_sum = optimal_radii(centres, A_ub, pairs)

        if new_sum > best_sum:                      # accept improving move
            best_sum = new_sum
            best_centres = centres.copy()
            best_radii = new_radii.copy()
            cur_sum = new_sum
        else:                                       # reject
            centres[i] = old_pos

        # decay the step size a little
        cur_step *= 0.9995

        if target is not None and best_sum >= target:
            break

    return best_centres, best_radii, best_sum


# --------------------------------------------------------------------------- #
# Main entry point required by the platform
# --------------------------------------------------------------------------- #
def run_packing():
    """
    Compute a packing of 26 circles inside the unit square that aims to
    maximise the sum of radii.  Returns (centres, radii, sum_of_radii).
    """
    np.random.seed(0)          # deterministic behaviour
    N = 26
    TARGET = 2.636

    # ----- constant LP data -------------------------------------------------
    A_ub, pair_list = _precompute_constraints(N)

    # ----- several random restarts with hill climbing -----------------------
    best_sum = -1.0
    best_centres = None
    best_radii = None

    # we use a mixture of a regular grid (good start) and pure random starts
    for restart in range(6):
        # ------- initial centre set ---------------------------------------
        # start from the 5×5 grid (25 points) + 1 random point
        xs = np.linspace(0.1, 0.9, 5)
        ys = np.linspace(0.1, 0.9, 5)
        grid = np.column_stack(np.meshgrid(xs, ys)).reshape(-1, 2)   # 25×2
        centres0 = np.empty((N, 2), dtype=np.float64)
        centres0[:25] = grid
        centres0[25] = np.random.rand(2)                            # extra point
        np.random.shuffle(centres0)

        # ------- local optimisation ---------------------------------------
        centres_opt, radii_opt, sum_opt = hill_climb(
            centres0, A_ub, pair_list,
            iterations=2000,
            step=0.18,
            target=TARGET
        )

        if sum_opt > best_sum:
            best_sum = sum_opt
            best_centres = centres_opt
            best_radii = radii_opt

        if best_sum >= TARGET:
            break

    # final safety recomputation (ensures radii truly correspond to centres)
    final_radii, final_sum = optimal_radii(best_centres, A_ub, pair_list)

    # Return exactly the radii we have checked with the validator
    return best_centres, final_radii, float(final_sum)
