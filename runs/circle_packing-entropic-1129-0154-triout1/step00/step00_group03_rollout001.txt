def compute_radii(centers):
    n = centers.shape[0]
    # distances
    dists = cdist(centers, centers)
    # Build A_ub, b_ub
    rows = []
    b = []
    # boundary constraints
    for i in range(n):
        x,y = centers[i]
        # xi <= x
        e = np.zeros(n)
        e[i] = 1
        rows.append(e)
        b.append(x)
        # xi <= 1-x
        rows.append(e)
        b.append(1-x)
        # yi <= y
        rows.append(e)
        b.append(y)
        # yi <= 1-y
        rows.append(e)
        b.append(1-y)
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            e = np.zeros(n)
            e[i] = 1
            e[j] = 1
            rows.append(e)
            b.append(dists[i,j])
    A_ub = np.vstack(rows)
    b_ub = np.array(b)

    c = -np.ones(n)  # maximize sum -> minimize -sum
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        return np.zeros(n)
    return res.x
```

def compute_optimal_radii(centers):
    import numpy as np
    from scipy.optimize import linprog
    n = centers.shape[0]
    # compute pairwise distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    # number of constraints: 4*n boundaries + n*(n-1)/2 pairwise
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)
    row = 0
    for i in range(n):
        x, y = centers[i]
        # ri <= x
        A[row, i] = 1
        b[row] = x
        row += 1
        # ri <= 1-x
        A[row, i] = 1
        b[row] = 1 - x
        row += 1
        # ri <= y
        A[row, i] = 1
        b[row] = y
        row += 1
        # ri <= 1-y
        A[row, i] = 1
        b[row] = 1 - y
        row += 1
    for i in range(n):
        for j in range(i+1, n):
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dists[i, j]
            row += 1
    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # If LP fails, return zeros
        return np.zeros(n)
    return np.maximum(res.x, 0.0)
```

For a *fixed* set of centre positions the best radii are obtained from a **linear programme**  

```
max   Σ r_k
s.t.  r_k ≤ x_k , r_k ≤ 1‑x_k , r_k ≤ y_k , r_k ≤ 1‑y_k          (boundary)
      r_i + r_j ≤ distance(c_i,c_j)   for all i<j                (non‑overlap)
      r_k ≥ 0
```

import numpy as np
from scipy.optimize import linprog

# ------------------------------------------------------------
# Linear programme: for a given set of centre points compute the
# maximum possible radii (sum of radii is maximised).
# ------------------------------------------------------------
def compute_optimal_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the LP that maximises sum of radii for the fixed centre positions.
    Returns an array of radii (length = number of circles).
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of inequality constraints:
    # 4 per circle for the four sides + one per unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n), dtype=np.float64)
    b = np.empty(m, dtype=np.float64)

    row = 0
    # boundary constraints: r_i <= x_i, 1-x_i, y_i, 1-y_i
    for i in range(n):
        x, y = centers[i]
        A[row, i] = 1.0
        b[row] = x
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - x
        row += 1
        A[row, i] = 1.0
        b[row] = y
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - y
        row += 1

    # non‑overlap constraints: r_i + r_j <= dist_ij
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    # solve
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method="highs")
    if not res.success:
        # fallback: return zeros (should not happen for reasonable inputs)
        return np.zeros(n, dtype=np.float64)

    # numerical safety – clip to the obvious geometric limits
    radii = np.maximum(res.x, 0.0)
    max_by_boundary = np.minimum.reduce(
        [centers[:, 0], 1.0 - centers[:, 0], centers[:, 1], 1.0 - centers[:, 1]]
    )
    radii = np.minimum(radii, max_by_boundary)
    return radii


# ------------------------------------------------------------
# Simple simulated‑annealing optimiser that moves centres and
# re‑optimises the radii via the LP.
# ------------------------------------------------------------
def simulated_annealing(
    init_centers: np.ndarray, iterations: int = 1500, initial_temp: float = 0.05
):
    """
    Perform a short annealing run starting from init_centers.
    Returns (best_centers, best_radii, best_sum).
    """
    n = init_centers.shape[0]
    best_centers = init_centers.copy()
    best_radii = compute_optimal_radii(best_centers)
    best_sum = best_radii.sum()

    cur_centers = best_centers.copy()
    cur_radii = best_radii.copy()
    cur_sum = best_sum
    temp = initial_temp

    for it in range(iterations):
        # pick a circle and propose a small random move
        i = np.random.randint(n)
        proposal = cur_centers.copy()

        # step size shrinks with temperature
        step_scale = 0.05 * (temp / initial_temp + 0.1)
        nx = proposal[i, 0] + np.random.normal(scale=step_scale)
        ny = proposal[i, 1] + np.random.normal(scale=step_scale)

        # keep the centre inside the open unit square
        nx = np.clip(nx, 1e-8, 1.0 - 1e-8)
        ny = np.clip(ny, 1e-8, 1.0 - 1e-8)
        proposal[i] = [nx, ny]

        # resolve the LP for the new layout
        new_radii = compute_optimal_radii(proposal)
        new_sum = new_radii.sum()

        delta = new_sum - cur_sum
        accept = False
        if delta > 0:
            accept = True
        else:
            # Metropolis criterion
            if temp > 1e-12:
                prob = np.exp(delta / temp)
                if np.random.rand() < prob:
                    accept = True

        if accept:
            cur_centers = proposal
            cur_radii = new_radii
            cur_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = proposal.copy()
                best_radii = new_radii.copy()

        # exponential cooling
        temp *= 0.9995
        if temp < 1e-12:
            temp = 1e-12

    return best_centers, best_radii, best_sum


# ------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------
def run_packing():
    """
    Returns (centres, radii, sum_of_radii) for 26 circles inside the unit square.
    The configuration is obtained by a short simulated‑annealing optimisation
    that repeatedly solves the LP for the radii.
    """
    np.random.seed(0)                     # reproducibility
    n = 26
    best_overall_sum = -1.0
    best_overall_centres = None
    best_overall_radii = None

    # We try a few independent restarts.
    for _ in range(6):
        # start from a regular 5×5 grid (25 points) plus one extra point
        grid_vals = np.linspace(0.1, 0.9, 5)
        X, Y = np.meshgrid(grid_vals, grid_vals)
        base = np.column_stack((X.ravel(), Y.ravel()))[:n]   # first 25 cells
        # add a small random jitter to break perfect symmetry
        jitter = np.random.uniform(-0.015, 0.015, size=base.shape)
        centres = np.clip(base + jitter, 1e-8, 1.0 - 1e-8)

        # if we have only 25 points (which is the case for n=26), add a random extra point
        if centres.shape[0] < n:
            extra = np.random.uniform(0.2, 0.8, size=(n - centres.shape[0], 2))
            centres = np.vstack([centres, extra])

        # run a short annealing optimisation
        centres_opt, radii_opt, sum_opt = simulated_annealing(
            centres, iterations=1800, initial_temp=0.07
        )

        if sum_opt > best_overall_sum:
            best_overall_sum = sum_opt
            best_overall_centres = centres_opt
            best_overall_radii = radii_opt

    # Ensure the returned arrays have the exact required shapes
    best_overall_centres = np.asarray(best_overall_centres, dtype=np.float64).reshape(n, 2)
    best_overall_radii = np.asarray(best_overall_radii, dtype=np.float64).reshape(n)

    # final safety clipping (tiny numerical overshoots)
    max_by_boundary = np.minimum.reduce(
        [
            best_overall_centres[:, 0],
            1.0 - best_overall_centres[:, 0],
            best_overall_centres[:, 1],
            1.0 - best_overall_centres[:, 1],
        ]
    )
    best_overall_radii = np.minimum(best_overall_radii, max_by_boundary)
    best_overall_radii = np.maximum(best_overall_radii, 0.0)

    total_sum = float(best_overall_radii.sum())
    return best_overall_centres, best_overall_radii, total_sum
