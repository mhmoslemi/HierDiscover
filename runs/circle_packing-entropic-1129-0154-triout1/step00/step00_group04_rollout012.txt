import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Global data for the problem (26 circles)
N = 26
# All unordered pairs (i,j) with i < j
PAIR_IDX = [(i, j) for i in range(N) for j in range(i + 1, N)]

# ----------------------------------------------------------------------
def _lp_max_radii(centers: np.ndarray) -> np.ndarray:
    """
    Solve the linear programme that maximises the sum of radii for a *fixed*
    set of centre positions.

    Parameters
    ----------
    centers : np.ndarray, shape (N,2)
        The (x,y) coordinates of the circles.

    Returns
    -------
    radii : np.ndarray, shape (N,)
        The optimal radii (may contain zeros).
    """
    # distances to the four sides for each centre
    side_dist = np.minimum.reduce(
        [centers[:, 0],                       # distance to left side
         1.0 - centers[:, 0],                 # distance to right side
         centers[:, 1],                       # distance to bottom side
         1.0 - centers[:, 1]]                 # distance to top side
    )

    # Upper bounds for each radius: limited by the nearest side
    ub = side_dist.copy()

    # Pairwise distance matrix (symmetric, zero diagonal)
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]  # (N,N,2)
    pair_dist = np.sqrt(np.sum(diff * diff, axis=2))

    # Build the inequality matrix A_ub * r <= b_ub
    #   * one row per radius (r_i <= ub_i)
    #   * one row per pair (r_i + r_j <= d_ij)
    num_constraints = N + len(PAIR_IDX)
    A = np.zeros((num_constraints, N), dtype=np.float64)
    b = np.zeros(num_constraints, dtype=np.float64)

    # radius <= side distance
    for i in range(N):
        A[i, i] = 1.0
        b[i] = ub[i]

    # pairwise constraints
    base = N
    for k, (i, j) in enumerate(PAIR_IDX):
        A[base + k, i] = 1.0
        A[base + k, j] = 1.0
        b[base + k] = pair_dist[i, j]

    # Objective: maximise sum(r)  → minimise -sum(r)
    c = -np.ones(N, dtype=np.float64)

    # Bounds for each variable (r_i >= 0)
    bounds = [(0.0, None) for _ in range(N)]

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In the unlikely event the LP fails, fall back to zero radii
        return np.zeros(N, dtype=np.float64)
    return res.x

def _initial_grid(jitter: float = 0.0, rng: np.random.Generator = None):
    """
    Produce a base 5×5 grid (25 points) and a 26th auxiliary point.
    Optional random jitter can be added to all points.
    """
    # regular 5×5 grid with spacing 0.2, first centre at 0.1
    step = 0.2
    offset = 0.1

    ys = offset + step * np.arange(5)

    extra = np.array([[0.25, 0.25]])  # corner of the central free square

        points = np.clip(points + noise, 0.0, 1.0)

    return points

        # Solve the LP for this fixed geometry
        radii = _lp_max_radii(centres)

        # Compute the objective value
        cur_sum = float(np.sum(radii))

        if cur_sum > best_sum:
            best_sum = cur_sum
            best_centers = centres.copy()
            best_radii = radii.copy()

    # ------------------------------------------------------------------
    # Fallback: pure 5×5 grid with 25 circles of radius 0.1 and a dummy circle
    # (radius zero).  This is always feasible and guarantees a non‑negative
    # result even if the optimisation somehow failed.
    if best_centers is None or best_radii is None:
        step = 0.2
        offset = 0.1
        xs = offset + step * np.arange(5)
        ys = offset + step * np.arange(5)
        xv, yv = np.meshgrid(xs, ys)
        grid = np.column_stack([xv.ravel(), yv.ravel()])   # 25 points
        dummy = np.array([[0.0, 0.0]])                     # dummy centre
        best_centers = np.vstack([grid, dummy])
        best_radii = np.concatenate([np.full(25, 0.1), np.array([0.0])])
        best_sum = float(np.sum(best_radii))

    # Reshape the centre array to the required (26,2) form
    centers_out = best_centers.reshape(N, 2)
    radii_out = best_radii.reshape(N)

    return centers_out, radii_out, best_sum
