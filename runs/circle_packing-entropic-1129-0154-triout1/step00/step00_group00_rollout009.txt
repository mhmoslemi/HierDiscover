import numpy as np
import cvxpy as cp

def solve_radii(centers):
    n = centers.shape[0]
    r = cp.Variable(n)
    constraints = [r >= 0]
    # boundary constraints
    constraints += [r <= centers[:,0]]
    constraints += [r <= 1 - centers[:,0]]
    constraints += [r <= centers[:,1]]
    constraints += [r <= 1 - centers[:,1]]
    # non overlap
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            constraints.append(r[i] + r[j] <= d)
    prob = cp.Problem(cp.Maximize(cp.sum(r)), constraints)
    prob.solve(solver=cp.ECOS, warm_start=True, verbose=False)
    if r.value is None:
        raise RuntimeError("LP failed to solve")
    radii = np.maximum(r.value, 0.0)
    sum_radii = np.sum(radii)
    return radii, sum_radii
```

    for i in range(n):
        # r_i <= left[i]
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(left[i])
        # r_i <= right[i]

        rows.append(row); rhs.append(right[i])
        # r_i <= bottom[i]
        row = np.zeros(n); row[i] = 1.0
        rows.append(row); rhs.append(bottom[i])
        # r_i <= top[i]
        row = np.zeros(n); row[i] = 1.0
        rows.append(row); rhs.append(top[i])
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(d)
    A = np.vstack(rows)
    b = np.array(rhs)
    # objective maximize sum r => minimize -sum r
    c = -np.ones(n)
    # variable bounds (0, None)
    var_bounds = [(0, None)] * n
    res = opt.linprog(c, A_ub=A, b_ub=b, bounds=var_bounds, method='highs')
    if not res.success:
        raise RuntimeError("LP failed")
    radii = res.x
    sum_radii = radii.sum()
    return radii, sum_radii

import numpy as np
from scipy.optimize import linprog

def solve_radii(centers):
    n = centers.shape[0]
    # compute distances and boundaries
    left = centers[:,0]
    right = 1 - centers[:,0]
    bottom = centers[:,1]
    top = 1 - centers[:,1]

    # total constraints count: 4*n + n*(n-1)//2
    m = 4*n + n*(n-1)//2
    A = np.zeros((m, n))
    b = np.zeros(m)

    row = 0
    for i in range(n):
        A[row,i] = 1; b[row] = left[i]; row += 1
        A[row,i] = 1; b[row] = right[i]; row += 1
        A[row,i] = 1; b[row] = bottom[i]; row += 1
        A[row,i] = 1; b[row] = top[i]; row += 1

    for i in range(n):
        for j in range(i+1, n):
            d = np.linalg.norm(centers[i] - centers[j])
            A[row,i] = 1
            A[row,j] = 1
            b[row] = d
            row += 1

    c = -np.ones(n)
    bounds = [(0, None)] * n
    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # fallback: return zeros
        radii = np.zeros(n)
        sum_r = 0.0
    else:
        radii = np.maximum(res.x, 0.0)
        sum_r = radii.sum()
    return radii, sum_r
```

def generate_hex_grid(dx):
    dy = np.sqrt(3) / 2 * dx
    points = []
    i = 0
    y = dy/2.0
    while y < 1.0:
        offset = (i % 2) * (dx/2.0)
        x = dx/2.0 + offset
        while x < 1.0:
            points.append([x, y])
            x += dx
        i += 1
        y += dy
    return np.array(points)

1. **Radii optimisation for fixed centres.**  
   For a given set of centre points the radii are the solution of a linear program  

```
max   Σ r_i

```

   The constraints are linear in the radii, therefore we can obtain the optimal radii for any

   `scipy.optimize.linprog` with the HiGHS solver solves these LPs in a few milliseconds.

   The sum of the optimal radii is a fairly smooth function of the centres.

import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Linear programming: optimal radii for a fixed set of centres
# ----------------------------------------------------------------------
def solve_radii(centres: np.ndarray):
    """
    Solve the LP that maximises the sum of radii for the given centres.
    Returns the radii vector and the sum of radii.
    """
    n = centres.shape[0]

    # distances to the four sides
    left   = centres[:, 0]
    right  = 1.0 - centres[:, 0]
    bottom = centres[:, 1]
    top    = 1.0 - centres[:, 1]

    # Number of constraints: 4 per circle + one per unordered pair
    m = 4 * n + n * (n - 1) // 2
    A = np.zeros((m, n))
    b = np.zeros(m)

    row = 0
    # side constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = left[i]
        row += 1

        A[row, i] = 1.0
        b[row] = right[i]
        row += 1

        A[row, i] = 1.0
        b[row] = bottom[i]
        row += 1

        A[row, i] = 1.0
        b[row] = top[i]
        row += 1

    # pairwise non‑overlap constraints
    for i in range(n):
        for j in range(i + 1, n):
            d = np.linalg.norm(centres[i] - centres[j])
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = d
            row += 1

    # Maximise Σ r_i  <=> minimise -Σ r_i
    c = -np.ones(n)
    bounds = [(0.0, None)] * n

    res = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not res.success:
        # In the very unlikely case of a failure return all zeros
        return np.zeros(n), 0.0

    radii = np.maximum(res.x, 0.0)   # guard against tiny negative values
    return radii, radii.sum()


# ----------------------------------------------------------------------
# Helper to evaluate a centre configuration and shrink radii a tiny bit
# ----------------------------------------------------------------------
def evaluate(centres: np.ndarray):
    radii, total = solve_radii(centres)
    eps = 1e-9
    radii = np.maximum(radii - eps, 0.0)
    return radii, radii.sum()

def hex_lattice(dx: float):
    dy = np.sqrt(3.0) / 2.0 * dx
    points = []
    i = 0
    y = dy / 2.0
    while y < 1.0:
        offset = (i % 2) * (dx / 2.0)
        x = dx / 2.0 + offset
        while x < 1.0:
            points.append([x, y])
            x += dx
        i += 1
        y += dy
    return np.array(points)

    # 1.  Deterministic seeds – hex lattice, many random subsets
    # --------------------------------------------------------------
    hex_pts = hex_lattice(dx=0.2)          # gives 27 points
    m_pts = hex_pts.shape[0]

    for _ in range(200):
        if m_pts > n_circles:
            idx = np.random.choice(m_pts, size=n_circles, replace=False)
            centres = hex_pts[idx].copy()
        else:
            centres = hex_pts.copy()

        # tiny jitter avoids exact degeneracy
        jitter = (np.random.rand(n_circles, 2) - 0.5) * 0.02
        centres = np.clip(centres + jitter, 0.0, 1.0)

        radii, total = evaluate(centres)
        if total > best_sum:
            best_sum = total
            best_centres = centres.copy()
            best_radii = radii.copy()
        if best_sum >= 2.636:
            break

    # --------------------------------------------------------------
    # 2.  Purely random initial configurations
    # --------------------------------------------------------------
    for _ in range(500):
        centres = np.random.rand(n_circles, 2)
        radii, total = evaluate(centres)
        if total > best_sum:
            best_sum = total
            best_centres = centres.copy()
            best_radii = radii.copy()
        if best_sum >= 2.636:
            break

    # --------------------------------------------------------------
    # 3.  Stochastic hill‑climbing refinement
    # --------------------------------------------------------------
    max_iters = 3000
    step = 0.2                     # initial move size
    for it in range(max_iters):
        # pick a random circle to move
        i = np.random.randint(n_circles)

        # decide between a global jump or a local jitter
        if np.random.rand() < 0.25:
            new_pos = np.random.rand(2)          # anywhere in the square
        else:
            delta = (np.random.rand(2) - 0.5) * 2.0 * step
            new_pos = best_centres[i] + delta

        new_pos = np.clip(new_pos, 0.0, 1.0)

        trial_centres = best_centres.copy()
        trial_centres[i] = new_pos

        radii, total = evaluate(trial_centres)

        if total > best_sum:
            best_sum = total
            best_centres = trial_centres
            best_radii = radii

        # slowly shrink the step size – focus on fine adjustments later
        step *= 0.9995

        if best_sum >= 2.636:
            break

    # --------------------------------------------------------------
    # 4.  Final safety shrink (already performed inside evaluate)
    # --------------------------------------------------------------
    final_radii = best_radii
    final_sum = final_radii.sum()

    return best_centres, final_radii, float(final_sum)
