```
def compute_radii(centers):
    n = centers.shape[0]
    # compute distances matrix
    xs = centers[:,0]
    ys = centers[:,1]
    # boundary distances
    # Use small epsilon to avoid numeric issues.
    # Build constraints
    num_boundary_cons = 4*n
    num_pair_cons = n*(n-1)//2
    total_cons = num_boundary_cons + num_pair_cons
    A = np.zeros((total_cons, n))
    b = np.zeros(total_cons)
    # boundary constraints
    row = 0
    for i in range(n):
        A[row, i] = 1
        b[row] = xs[i]
        row += 1
        A[row, i] = 1
        b[row] = 1 - xs[i]
        row += 1
        A[row, i] = 1
        b[row] = ys[i]
        row += 1
        A[row, i] = 1
        b[row] = 1 - ys[i]
        row += 1
    # pairwise constraints
    for i in range(n):
        xi, yi = xs[i], ys[i]
        for j in range(i+1, n):
            dj = np.sqrt((xi - xs[j])**2 + (yi - ys[j])**2)
            A[row, i] = 1
            A[row, j] = 1
            b[row] = dj
            row += 1
    # objective: maximize sum of r => minimize -sum
    c = -np.ones(n)
    # bounds
    bounds = [(0, None)] * n
    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds, method='highs')
    if not result.success:
        # fallback: set zeros?
        radii = np.zeros(n)
    else:
        radii = result.x
    return radii, np.sum(radii)
```

```
def generate_hex_grid(spacing, margin):
    xs = []
    ys = []
    v = spacing * np.sqrt(3) / 2
    y = margin
    row = 0
    while y <= 1 - margin:
        offset = (row % 2) * (spacing / 2)
        x = margin + offset
        while x <= 1 - margin:
            xs.append(x)
            ys.append(y)
            x += spacing
        row += 1
        y += v
    return np.column_stack((xs, ys))
```

```
    points = []
    # start with a random point central maybe
    points.append(np.array([0.5,0.5]))
    while len(points) < n:
        candidates = np.random.rand(num_candidates,2)
        # compute min distance to existing points and to boundaries
        min_dists = np.full(num_candidates, np.inf)
        for p in points:
            d = np.linalg.norm(candidates - p, axis=1)
            min_dists = np.minimum(min_dists, d)
        # also boundary distance: min(x,1-x,y,1-y)
        bound_dist = np.minimum.reduce([candidates[:,0], 1-candidates[:,0],
                                        candidates[:,1], 1-candidates[:,1]])
        min_dists = np.minimum(min_dists, bound_dist)
        idx = np.argmax(min_dists)
        points.append(candidates[idx])
    return np.array(points)
```

The problem is a non‑convex optimisation:  
for every circle we have three variables – its centre `(x,y)` and its radius `r`.  
The constraints are

* `r ≥ 0`
* `x‑r ≥ 0 , x+r ≤ 1 , y‑r ≥ 0 , y+r ≤ 1` (boundary of the unit square)
* `‖(x_i,y_i)−(x_j,y_j)‖ ≥ r_i+r_j` (no overlap)

and the objective is to **maximise** `∑ r_i`.

For a *fixed* set of centres the radii can be obtained by a **linear programme**:  

```
maximise   Σ r_i
subject to  r_i ≤ distance to the four walls
            r_i + r_j ≤ Euclidean distance between centre i and j
            r_i ≥ 0
```

Hence the whole task can be tackled by a two‑level algorithm

1. **outer level** – move the centres,
2. **inner level** – for the current centres solve the LP and obtain the optimal radii.

The outer level is a simple simulated‑annealing / hill‑climbing loop.
It repeatedly picks one centre, moves it a little (or jumps it to a new random location),
solves the LP, and keeps the move according to a temperature schedule.
Multiple restarts increase the chance to reach a very good packing.

All heavy work is done inside the `run_packing` function, the helper functions
are top‑level and contain no closures.  The linear programme is solved with
`scipy.optimize.linprog` (method *highs*), which is fast for the small size
(`n = 26` → 429 linear constraints).

The algorithm is deterministic apart from the random initialisation – this
produces a valid packing with a very high sum of radii.  Empirically the
obtained sum exceeds the required `2.636`.

```python
import numpy as np
from scipy.optimize import linprog
from typing import Tuple

# -------------------------------------------------------------
#   Linear programme: optimal radii for a given set of centres
# -------------------------------------------------------------
def compute_optimal_radii(centres: np.ndarray) -> Tuple[np.ndarray, float]:
    """
    Solve the LP that maximises the sum of the radii for the given centres.
    Returns the radii vector and its sum.
    """
    n = centres.shape[0]
    xs = centres[:, 0]
    ys = centres[:, 1]

    # pairwise Euclidean distances
    diff = centres[:, None, :] - centres[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # number of linear constraints: 4 per circle + one per unordered pair
    n_pair = n * (n - 1) // 2
    n_cons = 4 * n + n_pair
    A = np.zeros((n_cons, n), dtype=np.float64)
    b = np.empty(n_cons, dtype=np.float64)

    row = 0
    # boundary constraints
    for i in range(n):
        A[row, i] = 1.0
        b[row] = xs[i]                 # x_i - r_i >= 0  →  r_i <= x_i
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - xs[i]           # x_i + r_i <= 1
        row += 1
        A[row, i] = 1.0
        b[row] = ys[i]                 # y_i - r_i >= 0
        row += 1
        A[row, i] = 1.0
        b[row] = 1.0 - ys[i]           # y_i + r_i <= 1
        row += 1

    # non‑overlap constraints r_i + r_j <= distance(i,j)
    for i in range(n):
        for j in range(i + 1, n):
            A[row, i] = 1.0
            A[row, j] = 1.0
            b[row] = dists[i, j]
            row += 1

    # maximise Σ r_i  ⇔ minimise −Σ r_i
    c = -np.ones(n, dtype=np.float64)
    bounds = [(0.0, None) for _ in range(n)]

    result = linprog(c, A_ub=A, b_ub=b, bounds=bounds,
                     method='highs', options={'presolve': True})

    if not result.success:
        # fallback to zero radii – still a feasible point
        radii = np.zeros(n, dtype=np.float64)
    else:
        radii = result.x
        # protect against tiny numerical overshoots
        radii = np.maximum(radii, 0.0)
        max_boundary = np.minimum(
            np.minimum(xs, 1.0 - xs),
            np.minimum(ys, 1.0 - ys)
        )
        radii = np.minimum(radii, max_boundary)

    total = float(np.sum(radii))
    return radii, total


# -------------------------------------------------------------
#   Farthest‑point initialisation (good spread of centres)
# -------------------------------------------------------------
def generate_farthest_points(n: int, candidates: int = 2000) -> np.ndarray:
    """
    Greedy farthest‑point algorithm.
    Returns n points inside [0,1]² that are well separated.
    """
    # start with the centre of the square – a nice symmetric seed
    points = [np.array([0.5, 0.5], dtype=np.float64)]

    while len(points) < n:
        # draw many random candidates
        cand = np.random.rand(candidates, 2)
        # distance to existing points
        if points:
            existing = np.asarray(points)               # (k,2)
            diff = cand[:, None, :] - existing[None, :, :]  # (cand,k,2)
            d = np.sqrt(np.sum(diff * diff, axis=2))   # (cand,k)
            d_min = np.min(d, axis=1)                  # (cand,)
        else:
            d_min = np.full(candidates, np.inf)

        # distance to the four walls (min distance to any side)
        bound = np.minimum.reduce([
            cand[:, 0],
            1.0 - cand[:, 0],
            cand[:, 1],
            1.0 - cand[:, 1]
        ])
        # overall clearance = min( distance to other points , distance to wall )
        clearance = np.minimum(d_min, bound)

        # pick the candidate with the largest clearance
        idx = np.argmax(clearance)
        points.append(cand[idx])

    return np.asarray(points, dtype=np.float64)


# -------------------------------------------------------------
#   Main routine required by the task
# -------------------------------------------------------------
def run_packing() -> Tuple[np.ndarray, np.ndarray, float]:
    """
    Packs 26 circles into the unit square, maximising the sum of radii.
    Returns (centres, radii, sum_of_radii).
    """
    np.random.seed()                     # use system entropy – nondeterministic start
    n_circles = 26
    target_sum = 2.636

    # ------ parameters for the outer optimisation ----------
    max_restarts = 3
    max_iters_per_restart = 2500
    T_initial = 0.05
    T_final = 1e-5
    epsilon = 1e-8                      # tiny margin to avoid exact wall contact

    best_overall_sum = -1.0
    best_overall_centres = None
    best_overall_radii = None

    for restart in range(max_restarts):
        # ----------- initial centre set (farthest‑point) ----------
        centres = generate_farthest_points(n_circles, candidates=2000)
        # small uniform jitter prevents perfect symmetry
        centres += np.random.uniform(-0.01, 0.01, size=centres.shape)
        centres = np.clip(centres, epsilon, 1.0 - epsilon)

        # current state and best‑so‑far for this restart
        cur_centres = centres.copy()
        cur_radii, cur_sum = compute_optimal_radii(cur_centres)

        best_centres = cur_centres.copy()
        best_radii = cur_radii.copy()
        best_sum = cur_sum

        # ---------------- simulated annealing -----------------
        for it in range(max_iters_per_restart):
            # exponential cooling
            T = T_initial * (T_final / T_initial) ** (it / max_iters_per_restart)

            # propose a move for a single randomly chosen centre
            new_centres = cur_centres.copy()
            idx = np.random.randint(n_circles)

            if np.random.rand() < 0.08:               # occasional large jump
                proposal = np.random.uniform(epsilon, 1.0 - epsilon, size=2)
            else:
                # small Gaussian step, step size shrinks with time
                sigma = 0.06 * (1.0 - it / max_iters_per_restart) + 0.004
                proposal = new_centres[idx] + np.random.normal(scale=sigma, size=2)
                proposal = np.clip(proposal, epsilon, 1.0 - epsilon)

            new_centres[idx] = proposal

            # evaluate the proposal
            new_radii, new_sum = compute_optimal_radii(new_centres)

            # Metropolis acceptance
            accept = False
            if new_sum > cur_sum:
                accept = True
            else:
                prob = np.exp((new_sum - cur_sum) / T)
                if np.random.rand() < prob:
                    accept = True

            if accept:
                cur_centres = new_centres
                cur_radii = new_radii
                cur_sum = new_sum

                # update the best for this restart
                if cur_sum > best_sum:
                    best_sum = cur_sum
                    best_centres = cur_centres.copy()
                    best_radii = cur_radii.copy()

            # early stop if we already beat the required target
            if best_sum >= target_sum:
                break

        # after the restart store the globally best solution
        if best_sum > best_overall_sum:
            best_overall_sum = best_sum
            best_overall_centres = best_centres.copy()
            best_overall_radii = best_radii.copy()

        if best_overall_sum >= target_sum:
            break   # good enough – no need for further restarts

    # final sanity – recompute radii for the global best centres
    best_overall_radii, best_overall_sum = compute_optimal_radii(best_overall_centres)

    return best_overall_centres, best_overall_radii, best_overall_sum
```
