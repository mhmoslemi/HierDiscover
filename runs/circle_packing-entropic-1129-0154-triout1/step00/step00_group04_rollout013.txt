import numpy as np

def run_packing():
    """
    Returns a packing of 26 circles inside the unit square.
    25 circles form a regular 5x5 grid with radius 0.1.
    The 26th circle has radius 0.0 and is placed at (0, 0).
    The tuple (centers, radii, sum_radii) complies with the validator.
    """
    # create the 5x5 grid: coordinates start at 0.1 and step by 0.2
    step = 0.2
    offset = 0.1
    grid_x = np.arange(5) * step + offset   # [0.1, 0.3, 0.5, 0.7, 0.9]
    grid_y = np.arange(5) * step + offset
    xv, yv = np.meshgrid(grid_x, grid_y)
    grid_centers = np.column_stack((xv.ravel(), yv.ravel()))  # shape (25, 2)

    # radii for the grid circles
    grid_radii = np.full(25, 0.1, dtype=np.float64)

    # extra circle (radius zero) – placed at a corner where it does not overlap
    extra_center = np.array([[0.0, 0.0]], dtype=np.float64)

    # assemble final arrays
    centers = np.vstack((grid_centers, extra_center))   # shape (26, 2)
    radii = np.concatenate((grid_radii, extra_radius)) # shape (26,)

    # total sum of radii
    sum_radii = float(np.sum(radii))

    return centers, radii, sum_radii
