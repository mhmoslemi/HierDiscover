def hill_climbing(initial_centers, time_limit=some seconds):
    best_centers = initial_centers.copy()
    best_radii, best_sum = solve_lp(best_centers)
    current_centers = best_centers.copy()
    current_radii = best_radii
    current_sum = best_sum
    start = time.time()
    step = 0.1
    T = 0.01 # temperature?
    while time.time() - start < time_limit:
        # randomly select i
        i = rng.integers(N)
        # propose new position
        new_center = current_centers[i] + rng.normal(scale=step, size=2)
        # clip to [0,1]:
        new_center = np.clip(new_center, 0.0, 1.0)
        # assign
        new_centers = current_centers.copy()
        new_centers[i] = new_center
        # solve LP
        new_radii, new_sum = solve_lp(new_centers)
        # accept if better or with probability
        if new_sum > current_sum:
            current_centers = new_centers
            current_sum = new_sum
            current_radii = new_radii
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
        else:
            # accept with probability exp((new_sum - current_sum)/T)
            if rng.random() < np.exp((new_sum - current_sum) / max(T, 1e-6)):
                current_centers = new_centers
                current_sum = new_sum
                current_radii = new_radii
        # decay step and T
        step *= 0.99995
        T *= 0.99995

    return best_centers, best_radii, best_sum

def hill_climb(initial_centers, rng, max_time=5.0):
    best_centers = initial_centers.copy()
    best_radii, best_sum = solve_lp(best_centers)
    current_centers = best_centers.copy()
    current_radii = best_radii
    current_sum = best_sum
    start_time = time.time()
    step = 0.2  # max move
    T = 0.1
    while time.time() - start_time < max_time:
        i = rng.integers(len(current_centers))
        # propose move
        move = rng.normal(scale=step, size=2)
        new_center = current_centers[i] + move
        # clip to [0,1]
        new_center = np.clip(new_center, 0.0, 1.0)
        # if too close to other circles? Not needed; LP will adjust radii appropriately.
        new_centers = current_centers.copy()
        new_centers[i] = new_center
        new_radii, new_sum = solve_lp(new_centers)
        if new_sum > current_sum or rng.random() < np.exp((new_sum - current_sum) / max(1e-8, T)):
            current_centers = new_centers
            current_sum = new_sum
            current_radii = new_radii
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()
        # decay step and T
        step *= 0.9995
        T *= 0.9995
    return best_centers, best_radii, best_sum

import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Helper: solve the linear program for the maximal radii of given centers
# ----------------------------------------------------------------------
def solve_lp(centers: np.ndarray) -> tuple[np.ndarray, float]:
    """
    Given circle centers (n,2), compute the maximal radii that satisfy
    - non‑overlap:   r_i + r_j <= distance(i,j)
    - containment:  r_i <= min(x_i, 1-x_i, y_i, 1-y_i)
    Returns (radii, sum_of_radii).
    """
    n = centers.shape[0]

    # pairwise Euclidean distances
    diff = centers[:, np.newaxis, :] - centers[np.newaxis, :, :]
    dists = np.linalg.norm(diff, axis=2)

    # Build inequality matrix A_ub * r <= b_ub
    # 1) pair constraints r_i + r_j <= d_ij
    rows = []
    rhs = []

    for i in range(n):
        for j in range(i + 1, n):
            row = np.zeros(n)
            row[i] = 1.0
            row[j] = 1.0
            rows.append(row)
            rhs.append(dists[i, j])

    # 2) wall constraints r_i <= min distance to the four sides
    for i in range(n):
        x, y = centers[i]
        w = min(x, 1.0 - x, y, 1.0 - y)
        row = np.zeros(n)
        row[i] = 1.0
        rows.append(row)
        rhs.append(w)

    A_ub = np.vstack(rows)
    b_ub = np.array(rhs, dtype=np.float64)

    # Objective: maximise sum(r)  -> minimise -sum(r)
    c = -np.ones(n, dtype=np.float64)

    # Bounds for radii: r_i >= 0
    bounds = [(0.0, None) for _ in range(n)]

    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=bounds,
                  method='highs', options={'presolve': True})

    if not res.success:
        # In the very unlikely case of failure, fall back to zero radii
        radii = np.zeros(n, dtype=np.float64)
        return radii, 0.0

    radii = res.x
    return radii, float(radii.sum())


# ----------------------------------------------------------------------
# Helper: farthest point sampling (deterministic with a fixed RNG)
# ----------------------------------------------------------------------
def farthest_sampling(num_points: int, rng: np.random.Generator) -> np.ndarray:
    """
    Produce a set of `num_points` points inside the unit square using a
    greedy farthest‑point heuristic on a pool of random candidates.
    """
    # a generous pool of random candidates
    pool_size = max(5000, num_points * 200)
    candidates = rng.uniform(0.0, 1.0, size=(pool_size, 2))

    # start with a random point from the pool
    idx = rng.integers(pool_size)
    centers = [candidates[idx]]
    candidates = np.delete(candidates, idx, axis=0)

    # iteratively pick the candidate farthest from the existing set
    for _ in range(num_points - 1):
        # distance from each candidate to the nearest already‑chosen centre
        dists = np.min(
            np.linalg.norm(candidates[:, np.newaxis, :] -
                           np.array(centers)[np.newaxis, :, :], axis=2),
            axis=1)
        idx = np.argmax(dists)
        centers.append(candidates[idx])
        candidates = np.delete(candidates, idx, axis=0)

    return np.asarray(centers, dtype=np.float64)


# ----------------------------------------------------------------------
# Helper: simple hill‑climbing / simulated‑annealing optimisation
# ----------------------------------------------------------------------
def hill_climb(initial_centers: np.ndarray,
               rng: np.random.Generator,
               max_time: float = 5.0) -> tuple[np.ndarray, np.ndarray, float]:
    """
    Starting from `initial_centers`, perturb one centre at a time and keep
    changes that improve (or occasionally worsen) the total sum of radii.
    Returns the best (centres, radii, sum) encountered within `max_time`
    seconds.
    """
    n = initial_centers.shape[0]

    best_centers = initial_centers.copy()
    best_radii, best_sum = solve_lp(best_centers)

    current_centers = best_centers.copy()
    current_radii = best_radii
    current_sum = best_sum

    start = time.time()
    step = 0.2      # initial maximal displacement
    temperature = 0.1

    while time.time() - start < max_time:
        # pick a circle to move
        i = rng.integers(n)

        # propose a new centre (Gaussian perturbation)
        proposal = current_centers[i] + rng.normal(scale=step, size=2)
        proposal = np.clip(proposal, 0.0, 1.0)

        new_centers = current_centers.copy()
        new_centers[i] = proposal

        new_radii, new_sum = solve_lp(new_centers)

        # Metropolis acceptance
        improve = new_sum - current_sum
        if improve > 0 or rng.random() < np.exp(improve / max(temperature, 1e-12)):
            current_centers = new_centers
            current_radii = new_radii
            current_sum = new_sum
            if new_sum > best_sum:
                best_sum = new_sum
                best_centers = new_centers.copy()
                best_radii = new_radii.copy()

        # slowly anneal step size and temperature
        step *= 0.9995
        temperature *= 0.9995

    return best_centers, best_radii, best_sum


# ----------------------------------------------------------------------
# Main entry point required by the task description
# ----------------------------------------------------------------------
def run_packing() -> tuple[np.ndarray, np.ndarray, float]:
    """
    Produces a packing of 26 circles inside the unit square.
    Returns (centers, radii, sum_of_radii).
    """
    rng = np.random.default_rng(12345)   # deterministic seed for reproducibility
    N = 26

    # 1) initialise positions with a far‑point heuristic
    init_centres = farthest_sampling(N, rng)

    # 2) locally optimise the positions
    centres, radii, total = hill_climb(init_centres, rng, max_time=5.0)

    # Ensure the shapes are exactly as required
    centres = centres.reshape(N, 2)
    radii = radii.reshape(N)

    return centres, radii, float(total)
