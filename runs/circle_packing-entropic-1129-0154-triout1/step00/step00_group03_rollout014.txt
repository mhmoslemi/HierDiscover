```python
def compute_radii(centers):
    n = centers.shape[0]
    # compute bounds for each center
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff**2, axis=2))
    # Build constraints
    A = []
    b = []
    # pairwise constraints
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            A.append(row)
            b.append(dists[i,j])
    # boundary constraints
    for i in range(n):
        row = np.zeros(n, dtype=float)
        row[i] = 1
        A.append(row)
        b.append(bounds[i])
    A = np.array(A)
    b = np.array(b)
    c = -np.ones(n)
    # Solve LP
    res = linprog(c, A_ub=A, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # In case of infeasibility -> all radii zero perhaps
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii, np.sum(radii)
```

```
def solve_radii(centers):
    n = centers.shape[0]
    bounds = np.minimum.reduce([centers[:,0], 1 - centers[:,0], centers[:,1], 1 - centers[:,1]])
    # compute pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.linalg.norm(diff, axis=2)
    # Build A_ub and b_ub
    # Pairwise constraints: for i<j, r_i + r_j <= d_ij
    rows = []
    rhs = []
    for i in range(n):
        for j in range(i+1, n):
            row = np.zeros(n)
            row[i] = 1
            row[j] = 1
            rows.append(row)
            rhs.append(dists[i,j])
    # boundary constraints: r_i <= bound_i
    for i in range(n):
        row = np.zeros(n)
        row[i] = 1
        rows.append(row)
        rhs.append(bounds[i])
    A_ub = np.vstack(rows)
    b_ub = np.array(rhs)
    c = -np.ones(n)
    res = linprog(c, A_ub=A_ub, b_ub=b_ub, bounds=[(0, None)]*n, method='highs')
    if not res.success:
        # fallback: set radii to zeros to avoid errors.
        radii = np.zeros(n)
    else:
        radii = res.x
    return radii, np.sum(radii)
```

```
def solve_radii_linear(centers, A_ub, pair_idx):
    n = centers.shape[0]
    # compute bound_i
    bounds = np.minimum.reduce([centers[:,0], 1-centers[:,0], centers[:,1], 1-centers[:,1]])
    # compute distances for pairs
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff*diff, axis=2))
    # Build b vector
    b = np.empty(A_ub.shape[0])
    for k, (i,j) in enumerate(pair_idx):
        b[k] = dists[i,j]
    b[len(pair_idx):] = bounds
    # solve LP
    res = linprog(-np.ones(n), A_ub=A_ub, b_ub=b, bounds=[(0, None)]*n, method='highs')
    if res.success:
        r = res.x
    else:
        r = np.zeros(n)
    return r, r.sum()
```

Precompute A_ub and pair_idx:

```
def precompute_A_ub(n):
    pair_idx = [(i, j) for i in range(n) for j in range(i+1, n)]
    m = len(pair_idx) + n
    A = np.zeros((m, n))
    for row, (i,j) in enumerate(pair_idx):
        A[row,i] = 1
        A[row,j] = 1
    for i in range(n):
        A[len(pair_idx)+i,i] = 1
    return A, pair_idx
```

The problem can be written as a linear program once the circle centres are fixed:  

*Variables*: radii `r_i ≥ 0` for `i = 0 … n‑1`.  
*Constraints*  

```
r_i + r_j ≤ d_ij          for every pair i < j
r_i ≤ min(x_i,1-x_i,y_i,1-y_i)    (circle stays inside the square)
```

where `d_ij` is the Euclidean distance of the centres.  
For a given set of centres the LP gives the **maximum possible sum of radii**.

Therefore the optimisation is reduced to *choose centre positions* that make this LP value as large as possible.  
A simple stochastic hill‑climbing / simulated‑annealing scheme is used:

1. **Initial layout** – a “pseudo‑grid” of 9 relatively large circles (3×3) placed at  
   `(r, r) , (r, 0.5) , (r, 1‑r) , (0.5, r) , … , (1‑r, 1‑r)` with `r≈0.14`.  
   The remaining 17 circles are placed uniformly at random inside the square.  
   This gives a feasible starting point with many circles already large.

2. **LP evaluation** – for any centre configuration the LP (via `scipy.optimize.linprog`
   with the high‑performance “highs” solver) returns the optimal radii and the sum.
   The LP matrices (`A_ub` and the list of index pairs) are built once and reused.

3. **Search step** – at each iteration a random subset of circles is displaced by a

   After clipping the new centres to `[0,1]²` the LP is solved again.  
   If the new sum of radii is larger, the move is accepted.
   A simple temperature schedule allows occasional uphill moves (simulated annealing).

4. **Time bound** – the loop runs for a few seconds (default ≈ 8 s).  This is fast

```python
import time
import numpy as np
from scipy.optimize import linprog

# ----------------------------------------------------------------------
# Global problem size
N_CIRCLES = 26

# ----------------------------------------------------------------------
def _precompute_lp_matrices(n):
    """
    Build the constant matrix A_ub for the LP:
        r_i + r_j <= d_ij    (pairwise)
        r_i <= bound_i       (boundary)
    Returns A_ub (numpy array) and a list of index pairs (i,j) for the distances.
    """
    pair_idx = [(i, j) for i in range(n) for j in range(i + 1, n)]
    m = len(pair_idx) + n                 # pair constraints + boundary constraints
    A = np.zeros((m, n), dtype=float)

    # pairwise rows
    for row, (i, j) in enumerate(pair_idx):
        A[row, i] = 1.0
        A[row, j] = 1.0

    # boundary rows (one per circle)
    for i in range(n):
        A[len(pair_idx) + i, i] = 1.0

    return A, pair_idx

# Pre‑compute once
_A_UB, _PAIR_IDX = _precompute_lp_matrices(N_CIRCLES)

# ----------------------------------------------------------------------
def _solve_radii(centers):
    """
    Solve the LP for a fixed set of centres.
    Returns radii (ndarray shape (n,)) and the sum of radii.
    """
    n = centers.shape[0]
    # distance to the closest side -> boundary bound for each radius
    bounds = np.minimum.reduce([centers[:, 0],
                                1.0 - centers[:, 0],
                                centers[:, 1],
                                1.0 - centers[:, 1]])

    # pairwise distances
    diff = centers[:, None, :] - centers[None, :, :]
    dists = np.sqrt(np.sum(diff * diff, axis=2))

    # RHS vector for the LP
    b = np.empty(_A_UB.shape[0], dtype=float)

    # fill pairwise distances
    for k, (i, j) in enumerate(_PAIR_IDX):
        b[k] = dists[i, j]

    # boundary limits
    b[len(_PAIR_IDX):] = bounds

    # linear objective: maximise sum(r)  <=> minimise -sum(r)
    c = -np.ones(n, dtype=float)

    # solve
    res = linprog(c, A_ub=_A_UB, b_ub=b,
                  bounds=[(0.0, None) for _ in range(n)],
                  method='highs')
    if not res.success:
        # fall back to zero radii – should never happen for a feasible start
        radii = np.zeros(n, dtype=float)
        total = 0.0
    else:
        radii = res.x
        total = radii.sum()
    return radii, total

# ----------------------------------------------------------------------
def _initial_configuration(r_big=0.14, r_small=0.08, rng=None):
    """
    Build an initial set of centres:
    - nine relatively large circles placed on a 3×3 grid with spacing based on r_big,
    - seventeen small circles placed uniformly at random.
    """
    if rng is None:
        rng = np.random.default_rng(0)

    # 3×3 grid for the “big” circles
    big_coords = np.array([r_big, 0.5, 1.0 - r_big])
    big_centres = np.stack(np.meshgrid(big_coords, big_coords), -1).reshape(-1, 2)

    # random positions for the remaining circles
    n_small = N_CIRCLES - big_centres.shape[0]
    small_centres = rng.uniform(low=r_small, high=1.0 - r_small, size=(n_small, 2))

    centres = np.vstack([big_centres, small_centres])
    return centres

# ----------------------------------------------------------------------
def _local_search(time_limit=8.0, rng_seed=42):
    """
    Simple simulated‑annealing / hill‑climbing optimiser.
    Returns the best centre set, the corresponding radii and the total radius sum.
    """
    rng = np.random.default_rng(rng_seed)

    # start with a promising layout (big + random small circles)
    centres = _initial_configuration(rng=rng)
    radii, best_sum = _solve_radii(centres)
    best_centres = centres.copy()
    best_radii = radii.copy()

    start = time.time()
    it = 0
    while time.time() - start < time_limit:
        it += 1
        # linear schedule for step size and temperature
        elapsed = time.time() - start
        frac = elapsed / time_limit
        step = 0.05 * (1.0 - frac) + 0.001 * frac
        temperature = 0.02 * (1.0 - frac) + 1e-6 * frac

        # propose a new set of centres – move a random subset
        new_centres = best_centres.copy()
        mask = rng.random(N_CIRCLES) < 0.25          # move ≈ 1/4 of the circles each round
        perturb = rng.normal(scale=step, size=(N_CIRCLES, 2))
        new_centres[mask] += perturb[mask]
        new_centres = np.clip(new_centres, 0.0, 1.0)

        # evaluate
        new_radii, new_sum = _solve_radii(new_centres)

        # accept if better, otherwise accept with Metropolis probability
        delta = new_sum - best_sum
        if delta > 0.0 or rng.random() < np.exp(delta / temperature):
            best_sum = new_sum
            best_centres = new_centres
            best_radii = new_radii

    return best_centres, best_radii, best_sum

# ----------------------------------------------------------------------
def run_packing():
    """
    Entry point required by the evaluation harness.
    Returns (centres, radii, sum_of_radii).
    """
    # Run the optimiser (≈ 8 seconds).  The time budget can be changed if needed.
    centres, radii, total = _local_search(time_limit=8.0, rng_seed=123)

    # Final safety – clip any tiny negative values caused by numerical noise
    radii = np.maximum(radii, 0.0)

    # Ensure the returned quantities have the exact expected shapes
    centres = np.asarray(centres, dtype=float).reshape(N_CIRCLES, 2)
    radii   = np.asarray(radii,   dtype=float).reshape(N_CIRCLES)

    return centres, radii, float(total)
```
