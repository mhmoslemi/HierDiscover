"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input pair representation.
2. Linear projections (left_proj / right_proj) and 3 gating projections
   (left_gate, right_gate, out_gate) are performed in FP32.
3. Apply the pair mask (if present) and the left/right gates:
       left  = left  * mask * left_gate
       right = right * mask * right_gate
4. Compute the core O(N²·C) contraction
       out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   using a custom Triton kernel that tiles the i, j, k and channel
   dimensions (blocks of 16) and accumulates in FP32.
5. Layer‑norm the result over the hidden dimension, apply the out‑gate,
   and a final linear projection back to the original dimension.

The kernel operates on FP32 tensors, uses explicit bounds checks
and works for arbitrary batch size, sequence length and hidden
dimension (as long as they fit on the device).  Only the
contraction is performed in Triton – the surrounding linear layers
and normalisations use PyTorch (which already calls highly‑optimised
cuBLAS/cuDNN kernels).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the N×N matrix multiplication across the hidden dim
# ----------------------------------------------------------------------
@triton.jit
def trmul_outgoing_kernel(
    left_ptr,          # const float*   [B, N, N, C]
    right_ptr,         # const float*   [B, N, N, C]
    out_ptr,           #       float*   [B, N, N, C]
    B, N, C,           # dimension sizes (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,   # int64
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,  # int64
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,        # int64
    # ---------------------------------------------------------------
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_C: tl.constexpr,
):
    """Multiply   left[b,i,k,c] * right[b,j,k,c]   and sum over k.

    Tiles:
        i : BLOCK_I
        j : BLOCK_J
        k : BLOCK_K
        c : BLOCK_C
    The program ID encodes (b, i_tile, j_tile, c_tile).
    """

    # ---------------------------------------------------------------
    # 1) Decode the 1‑D program ID into (b, i_tile, j_tile, c_tile)
    # ---------------------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    num_c = tl.cdiv(C, BLOCK_C)

    pid = tl.program_id(0)

    b = pid // (num_i * num_j * num_c)
    pid_rest = pid % (num_i * num_j * num_c)

    i_tile = pid_rest // (num_j * num_c)
    pid_rest = pid_rest % (num_j * num_c)

    j_tile = pid_rest // num_c
    c_tile = pid_rest % num_c

    # ---------------------------------------------------------------
    # 2) Starting offsets for this tile
    # ---------------------------------------------------------------
    i0 = i_tile * BLOCK_I
    j0 = j_tile * BLOCK_J
    c0 = c_tile * BLOCK_C

    i_idx = i0 + tl.arange(0, BLOCK_I)
    j_idx = j0 + tl.arange(0, BLOCK_J)
    c_idx = c0 + tl.arange(0, BLOCK_C)

    # Out‑of‑bounds masks for the leading dimensions
    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_c = c_idx < C

    # ---------------------------------------------------------------
    # 3) Accumulator (FP32)
    # ---------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_C), dtype=tl.float32)

    # ---------------------------------------------------------------
    # 4) Loop over k dimension (blocked)
    # ---------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k_idx = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # -----------------------------------------------------------
        #   Load left[b,i,k,c]
        # -----------------------------------------------------------
        left_ptrs = left_ptr + (
            b * stride_left_b
            + i_idx[:, None, None] * stride_left_i
            + k_idx[None, :, None] * stride_left_k
            + c_idx[None, None, :] * stride_left_c
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_c[None, None, :],
            other=0.0,
        )   # (BLOCK_I, BLOCK_K, BLOCK_C) FP32

        # -----------------------------------------------------------
        #   Load right[b,j,k,c]   (note the j dimension)
        # -----------------------------------------------------------
        right_ptrs = right_ptr + (
            b * stride_right_b
            + j_idx[:, None, None] * stride_right_j
            + k_idx[None, :, None] * stride_right_k
            + c_idx[None, None, :] * stride_right_c
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_c[None, None, :],
            other=0.0,
        )   # (BLOCK_J, BLOCK_K, BLOCK_C) FP32

        # -----------------------------------------------------------
        #   Accumulate over the K‑block.
        #   For every k within the block we compute the outer product
        #   across i and j for each channel c.
        # -----------------------------------------------------------
        for kk_local in range(0, BLOCK_K):
            # left slice: (BLOCK_I, BLOCK_C)
            l = left[:, kk_local, :]          # (i, c)
            # right slice: (BLOCK_J, BLOCK_C)
            r = right[:, kk_local, :]         # (j, c)

            # Broadcast to (i, j, c) and accumulate
            acc += l[:, None, :] * r[None, :, :]   # (i, j, c)

    # ---------------------------------------------------------------
    # 5) Write back the result (respecting out‑of‑bounds)
    # ---------------------------------------------------------------
    out_ptrs = out_ptr + (
        b * stride_out_b
        + i_idx[:, None, None] * stride_out_i
        + j_idx[None, :, None] * stride_out_j
        + c_idx[None, None, :] * stride_out_c
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_c[None, None, :],
    )


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments:
        data = (input, mask, weights, config)
            input   : [B, N, N, dim]   (torch.float32)
            mask    : [B, N, N]        (torch.float32 or bool)
            weights : dict of tensors (all on the same device)
            config  : dict containing at least
                       - "dim"
                       - "hidden_dim"
    Returns:
        out : [B, N, N, dim] torch.float32
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    torch.set_grad_enabled(False)
    input_tensor, mask_tensor, weights, config = data

    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ---------------------------------------------------------------
    # 1) Initial LayerNorm
    # ---------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim] FP32

    # ---------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ---------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ---------------------------------------------------------------
    # 3) Gating values (sigmoid)
    # ---------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ---------------------------------------------------------------
    # 4) Apply mask (if any) and the per‑pair gates
    # ---------------------------------------------------------------
    if mask_tensor is not None:
        # mask shape: [B, N, N] -> [B, N, N, 1] for broadcasting
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ---------------------------------------------------------------
    # 5) Core contraction via Triton
    # ---------------------------------------------------------------
    # Allocate output buffer (same dtype as the inputs to the kernel)
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Strides (in element units, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_c = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_c = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_c = out.stride()

    # Block sizes (tuned for H100 – 16^4 fits comfortably in registers)
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 16
    BLOCK_C = 16

    # Grid dimensions: one program per (b, i_tile, j_tile, c_tile)
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    num_c_tiles = (hidden_dim + BLOCK_C - 1) // BLOCK_C
    total_progs = B * num_i_tiles * num_j_tiles * num_c_tiles
    grid = (total_progs,)

    trmul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_c,
        stride_right_b,
        stride_right_j,
        stride_right_k,
        stride_right_c,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_c,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_C=BLOCK_C,
        num_warps=4,          # a reasonable default for H100
    )

    # ---------------------------------------------------------------
    # 6) Post‑processing: layer‑norm, out‑gate, final linear
    # ---------------------------------------------------------------
    # Normalisation over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    # Apply the out‑gate
    out = out * out_gate
    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
