"""
TriMul (outgoing) forward implementation.

The operator works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} and performs:
1. LayerNorm over the last channel dimension.
2. Two linear projections (left/right) → hidden_dim (no bias).
3. Optional binary mask applied to the projected tensors.
4. Gating (sigmoid(linear(x))) applied to left/right and a final out‑gate.
5. Contraction over the middle sequence dimension:
       out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batched matrix multiplication for each hidden slice d.
   This step is executed with a custom Triton kernel.
6. LayerNorm over hidden_dim, multiplication by out‑gate,
   and a final linear projection back to the original channel size.

All linear layers and layer‑norms use the provided weight tensors.
The Triton kernel computes the N×N matrix product for every hidden
slice (B·hidden_dim independent GEMMs) using a tiled outer‑product
algorithm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_gemm_kernel(
    left_ptr,                     # ptr to left tensor   [B,N,N,HD]
    right_ptr,                    # ptr to right tensor  [B,N,N,HD]
    out_ptr,                      # ptr to output tensor [B,N,N,HD]

    B, N, HD,                     # shapes
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,

    BLOCK_M: tl.constexpr,        # tile size over i
    BLOCK_N: tl.constexpr,        # tile size over j
    BLOCK_K: tl.constexpr,        # tile size over k
):
    """Batched GEMM for TriMul:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       The hidden dimension (d) is fused into the batch axis (B*HD).
    """
    pid_m = tl.program_id(0)          # tile index over i
    pid_n = tl.program_id(1)          # tile index over j
    pid_h = tl.program_id(2)          # combined batch * hidden index

    # decode batch and hidden slice from pid_h
    b = pid_h // HD
    d = pid_h % HD

    # Compute base offsets for this (b,d) slice
    offset_left  = b * stride_left_b  + d * stride_left_d
    offset_right = b * stride_right_b + d * stride_right_d
    offset_out   = b * stride_out_b   + d * stride_out_d

    # Global i/j coordinates of the tile
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i < N
    j_mask = j < N

    # accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k < N

        # Load A tile: left[b,i,k,d]
        a_ptrs = left_ptr + offset_left + i[:, None] * stride_left_i + k[None, :] * stride_left_k
        a = tl.load(a_ptrs,
                    mask=(i_mask[:, None] & k_mask[None, :]),
                    other=0.0).to(tl.float32)   # (BLOCK_M, BLOCK_K)

        # Load B tile: right[b,j,k,d]
        b_ptrs = right_ptr + offset_right + j[:, None] * stride_right_i + k[None, :] * stride_right_k
        b = tl.load(b_ptrs,
                    mask=(j_mask[:, None] & k_mask[None, :]),
                    other=0.0).to(tl.float32)   # (BLOCK_N, BLOCK_K)

        # Accumulate outer‑product: sum_k a[i,k] * b[j,k]
        # tl.dot expects (M,K) @ (K,N) – set trans_b=True because b is (N,K)
        acc += tl.dot(a, b, trans_b=True)

    # Write the tile back to memory
    out_ptrs = out_ptr + offset_out + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=(i_mask[:, None] & j_mask[None, :]))


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] (float32, on the same device as input).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & shapes
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    B, N, _, _ = input_tensor.shape
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm over channel dim
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B,N,N,HD]
    right = F.linear(x, right_proj_weight)  # [B,N,N,HD]

    # ------------------------------------------------------------------
    # 3) Optional binary mask
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left  = left * mask
        right = right * mask
    # (If nomask==True we skip the multiplication – mask is effectively all‑ones)

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear projections of the normalized x)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Contraction over the middle sequence dimension using Triton
    # ------------------------------------------------------------------
    # Allocate output buffer for the hidden‑dim contraction
    out_hidden = torch.empty_like(left)  # shape [B,N,N,HD]

    # Strides for the three tensors (they are contiguous in C order)
    sL_b, sL_i, sL_k, sL_d = left.stride()
    sR_b, sR_i, sR_k, sR_d = right.stride()
    sO_b, sO_i, sO_j, sO_d = out_hidden.stride()

    # Tile sizes – these work well for N up to 1024 on an H100.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim                 # combined batch × hidden slice
    )

    _trimul_gemm_kernel[grid](
        left,
        right,
        out_hidden,
        B, N, hidden_dim,
        sL_b, sL_i, sL_k, sL_d,
        sR_b, sR_i, sR_k, sR_d,
        sO_b, sO_i, sO_j, sO_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    # LayerNorm over hidden_dim
    out = F.layer_norm(out_hidden,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # Multiply with out‑gate (broadcasted over the hidden dimension)
    out = out * out_gate

    # Final linear projection back to dim (no bias)
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
