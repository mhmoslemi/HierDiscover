    pid_j = tl.program_id(1)

@triton.jit
def apply_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i = i_base + tl.arange(0, BLOCK_I)
    j = j_base + tl.arange(0, BLOCK_J)
    mask_i = i < N
    mask_j = j < N

    # broadcast h
    h = tl.arange(0, BLOCK_H)
    # For each combination of i/j, we need to compute vector of length BLOCK_H.
    # We'll loop over i_idx and j_idx inside kernel to avoid large broadcast.
    for ii in range(0, BLOCK_I):
        for jj in range(0, BLOCK_J):
            i_idx = i[ii]
            j_idx = j[jj]
            valid = mask_i[ii] & mask_j[jj]
            if not valid:
                continue
            # load mask scalar
            mask = tl.load(mask_ptr + pid_b*stride_mask_b + i_idx*stride_mask_i + j_idx*stride_mask_j)
            # load proj vector
            proj = tl.load(proj_ptr + pid_b*stride_proj_b + i_idx*stride_proj_i + j_idx*stride_proj_j + h*stride_proj_h, mask=valid & (h < H))
            # load gate vector
            gate = tl.load(gate_ptr + pid_b*stride_gate_b + i_idx*stride_gate_i + j_idx*stride_gate_j + h*stride_gate_h, mask=valid & (h < H))
            out = proj * gate * mask
            tl.store(out_ptr + pid_b*stride_out_b + i_idx*stride_out_i + j_idx*stride_out_j + h*stride_out_h, out, mask=valid & (h < H))
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Normalization
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply mask and gating using Triton kernel
    if mask is None or mask.shape == ():
        mask_use = torch.ones_like(input_tensor[...,0], dtype=dtype, device=device)
    else:
        mask_use = mask

    # left_fused = left_proj * left_gate * mask
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_H = 32  # maybe split hidden dimension into blocks to handle large H.

    grid = (triton.cdiv(input_tensor.shape[1], BLOCK_I),
            triton.cdiv(input_tensor.shape[2], BLOCK_J),
            input_tensor.shape[0])

    # Need to launch kernel for left and right separately
    apply_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_use,
        left_fused,
        input_tensor.shape[0],
        input_tensor.shape[1],
        hidden_dim,
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask_use.stride(0), mask_use.stride(1), mask_use.stride(2),
        left_fused.stride(0), left_fused.stride(1), left_fused.stride(2), left_fused.stride(3),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H)

    apply_mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_use,
        right_fused,
        input_tensor.shape[0],
        input_tensor.shape[1],
        hidden_dim,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask_use.stride(0), mask_use.stride(1), mask_use.stride(2),
        right_fused.stride(0), right_fused.stride(1), right_fused.stride(2), right_fused.stride(3),
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_H=BLOCK_H)

    # Triangular multiplication: left_fused @ right_fused^T per hidden dim
    B, N, _, H = left_fused.shape
    # reshape to (B*H, N, N)
    left_reshape = left_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshape = right_fused.permute(0, 3, 1, 2).reshape(B * H, N, N)
    # compute batched matmul
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back to (B, N, N, H)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Output layernorm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Multiply by out_gate
    out = out * out_gate

    # Final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def apply_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Load mask scalar (broadcast to vector later)
    mask = tl.load(mask_ptr + pid_b * stride_mask_b
                            + pid_i * stride_mask_i
                            + pid_j * stride_mask_j)

    # Compute base pointer offsets for proj, gate, out
    proj_base = proj_ptr + pid_b * stride_proj_b \
                           + pid_i * stride_proj_i \
                           + pid_j * stride_proj_j
    gate_base = gate_ptr + pid_b * stride_gate_b \
                           + pid_i * stride_gate_i \
                           + pid_j * stride_gate_j
    out_base = out_ptr + pid_b * stride_out_b \
                           + pid_i * stride_out_i \
                           + pid_j * stride_out_j

    # Loop over hidden dimension in BLOCK_H chunks
    off = 0
    while off < H:
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < H
        # Load vector for this block
        proj = tl.load(proj_base + h * stride_proj_h, mask=mask_h, other=0.0)
        gate = tl.load(gate_base + h * stride_gate_h, mask=mask_h, other=0.0)
        # Compute fused result
        out = proj * gate * mask
        tl.store(out_base + h * stride_out_h, out, mask=mask_h)
        off += BLOCK_H

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing Triangle Multiplicative Update (TriMul).
    The kernel fuses masking and gating (projected * gate * mask) for the left and right
    pair representations. The heavy triple‑product is performed as a batched GEMM via
    torch.bmm, which is highly optimized on H100. The implementation respects the
    original PyTorch semantics (LayerNorm, sigmoid gates, final linear projection).
    """
    import torch
    import torch.nn.functional as F
    import triton
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --- LayerNorm on input ---
    norm_out = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # --- Linear projections (no bias) ---
    left_proj = F.linear(norm_out, weights["left_proj.weight"])
    right_proj = F.linear(norm_out, weights["right_proj.weight"])

    # --- Gating logits & sigmoid ---
    left_gate = F.linear(norm_out, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(norm_out, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(norm_out, weights["out_gate.weight"]).sigmoid()

    # --- Fuse mask and gating using Triton ---
    # Ensure mask is float32 (or same dtype) for multiplication.
    if mask is None:
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    B, N, _, _ = input_tensor.shape

    # Prepare output buffers
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # Strides (converted to python int)
    lp_strides = [int(s) for s in left_proj.stride()]
    rg_strides = [int(s) for s in right_proj.stride()]
    lg_strides = [int(s) for s in left_gate.stride()]
    rg_gate_strides = [int(s) for s in right_gate.stride()]
    mask_strides = [int(s) for s in mask_f.stride()]
    lf_strides = [int(s) for s in left_fused.stride()]
    rf_strides = [int(s) for s in right_fused.stride()]

    # Kernel launch configuration
    grid = (N, N, B)   # i, j, batch
    BLOCK_H = 32       # hidden dimension block size (must divide hidden_dim or be covered by loop)

    # Fuse left side
    apply_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left_fused,
        B,
        N,
        hidden_dim,
        lp_strides[0], lp_strides[1], lp_strides[2], lp_strides[3],
        lg_strides[0], lg_strides[1], lg_strides[2], lg_strides[3],
        mask_strides[0], mask_strides[1], mask_strides[2],
        lf_strides[0], lf_strides[1], lf_strides[2], lf_strides[3],
        BLOCK_H=BLOCK_H,
    )

    # Fuse right side (reuse same kernel)
    apply_mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right_fused,
        B,
        N,
        hidden_dim,
        rg_strides[0], rg_strides[1], rg_strides[2], rg_strides[3],
        rg_gate_strides[0], rg_gate_strides[1], rg_gate_strides[2], rg_gate_strides[3],
        mask_strides[0], mask_strides[1], mask_strides[2],
        rf_strides[0], rf_strides[1], rf_strides[2], rf_strides[3],
        BLOCK_H=BLOCK_H,
    )

    # --- Triple product: left_fused @ right_fused^T over the sequence dimension ---
    # reshape to (B*hidden_dim, N, N)
    left_reshape = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    # batched matrix multiplication (N x N) * (N x N)^T -> (N x N)
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # reshape back to (B, N, N, hidden_dim)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --- Output LayerNorm ---
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # --- Apply out gate ---
    out = out * out_gate

    # --- Final linear projection back to original dim ---
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass
===================================================

The implementation follows the reference PyTorch module closely:

1️⃣ Layer‑norm the input (last dimension).
2️⃣ Linear projections to a hidden dimension (left/right).
3️⃣ Compute three sigmoid gates (left, right, out).
4️⃣ **Fused Triton kernel**:   left = left_proj * left_gate * mask
                              right = right_proj * right_gate * mask
   (mask is broadcast over the hidden dimension.)
5️⃣ Batched matrix‑multiplication:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is performed as a GEMM on each hidden slice using
   ``torch.bmm`` (highly optimized on H100).
6️⃣ Layer‑norm + out‑gate multiplication.
7️⃣ Final linear projection back to the original channel size.

Only the mask‑and‑gate fusion is written in Triton; the heavy
triple‑product is executed with cuBLAS via ``torch.bmm`` for speed.
The code works for any batch ∈ {1,2}, sequence length ∈ {128…1024},
and hidden dimensions up to 384.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _apply_mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_H: tl.constexpr,
):
    """
    Fuse projection, sigmoid‑gate and mask:
        out[b,i,j,:] = proj[b,i,j,:] * gate[b,i,j,:] * mask[b,i,j]
    One thread processes a single (b,i,j) location and loops over the
    hidden dimension in blocks of size BLOCK_H.
    """
    pid_b = tl.program_id(2)   # batch index
    pid_i = tl.program_id(0)   # sequence index i
    pid_j = tl.program_id(1)   # sequence index j

    # Load mask scalar (broadcast later)
    mask = tl.load(mask_ptr + pid_b * stride_mask_b
                            + pid_i * stride_mask_i
                            + pid_j * stride_mask_j)

    # Base pointers for this (b,i,j) element
    proj_base = proj_ptr + pid_b * stride_proj_b \
                           + pid_i * stride_proj_i \
                           + pid_j * stride_proj_j
    gate_base = gate_ptr + pid_b * stride_gate_b \
                           + pid_i * stride_gate_i \
                           + pid_j * stride_gate_j
    out_base = out_ptr + pid_b * stride_out_b \
                         + pid_i * stride_out_i \
                         + pid_j * stride_out_j

    # Loop over the hidden dimension in BLOCK_H‑sized chunks
    off = 0
    while off < H:
        h = off + tl.arange(0, BLOCK_H)
        mask_h = h < H               # guard for the last partial block

        # Load a vector of size BLOCK_H from projection and gate tensors
        proj = tl.load(proj_base + h * stride_proj_h, mask=mask_h, other=0.0)
        gate = tl.load(gate_base + h * stride_gate_h, mask=mask_h, other=0.0)

        # Fuse: projection * gate * mask
        out = proj * gate * mask
        tl.store(out_base + h * stride_out_h, out, mask=mask_h)

        off += BLOCK_H


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton kernel for
    mask‑gate fusion and a batched GEMM for the O(N³) triangular update.

    Args:
        data (Tuple):
            - input_tensor (torch.Tensor): shape [B, N, N, dim]
            - mask (torch.Tensor): shape [B, N, N] (may be all‑ones)
            - weights (dict of torch.Tensor): model parameters
            - config (dict): contains "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Sigmoid gates
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4️⃣ Fuse mask & per‑position gating with a Triton kernel
    # ------------------------------------------------------------------
    # Convert mask to the same dtype as the tensors (float32/float16)
    if mask is None:
        mask_f = torch.ones_like(input_tensor[..., 0], dtype=dtype, device=device)
    else:
        mask_f = mask.to(dtype)

    B, N, _, _ = input_tensor.shape

    # Output buffers for the fused left/right tensors
    left_fused = torch.empty_like(left_proj)
    right_fused = torch.empty_like(right_proj)

    # strides for all tensors (as Python ints)
    lp_s = [int(s) for s in left_proj.stride()]       # proj (left)
    rp_s = [int(s) for s in right_proj.stride()]      # proj (right)
    lg_s = [int(s) for s in left_gate.stride()]       # gate (left)
    rg_s = [int(s) for s in right_gate.stride()]      # gate (right)
    mk_s = [int(s) for s in mask_f.stride()]          # mask
    lf_s = [int(s) for s in left_fused.stride()]      # out (left)
    rf_s = [int(s) for s in right_fused.stride()]     # out (right)

    # launch grid – one thread per (i, j, b)
    grid = (N, N, B)          # (i, j, batch)
    BLOCK_H = 32              # hidden‑dim tile size (must be ≥1)

    # left side fusion
    _apply_mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_f,
        left_fused,
        B,
        N,
        hidden_dim,
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        mk_s[0], mk_s[1], mk_s[2],
        lf_s[0], lf_s[1], lf_s[2], lf_s[3],
        BLOCK_H=BLOCK_H,
    )

    # right side fusion (same kernel)
    _apply_mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_f,
        right_fused,
        B,
        N,
        hidden_dim,
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        mk_s[0], mk_s[1], mk_s[2],
        rf_s[0], rf_s[1], rf_s[2], rf_s[3],
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5️⃣ Triangular multiplicative update (batched GEMM)
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) -> batch over hidden dimension
    left_reshape = left_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_reshape = right_fused.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    out = torch.bmm(left_reshape, right_reshape.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Output layer‑norm and out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear back to the original dimension
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
```

This implementation respects the exact algebra of the reference TriMul module,
uses a Triton kernel to fuse the mask and gating computation, and relies on
high‑performance batched GEMM for the O(N³) triangular multiplication, making
it suitable for the required test configurations on an H100 GPU.
