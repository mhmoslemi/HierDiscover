```
@triton.jit
def layernorm_kernel(x_ptr, y_ptr, weight_ptr, bias_ptr,
                     B, N, C,
                     stride_b0, stride_b1, stride_b2, stride_c,
                     eps: tl.constexpr,
                     BLOCK_C: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return
    # compute b,i,j
    i_idx = pid
    b = i_idx // (N * N)
    rem = i_idx - b * N * N
    i = rem // N
    j = rem - i * N
    # compute base pointer
    x_ptrs = x_ptr + b*stride_b0 + i*stride_b1 + j*stride_b2
    # compute sum and sumsq
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)
    for off in range(0, C, BLOCK_C):
        cur_off = off + tl.arange(0, BLOCK_C)
        mask = cur_off < C
        vals = tl.load(x_ptrs + cur_off * stride_c, mask=mask, other=0.0)
        vals_f = vals.to(tl.float32)
        sum_val += tl.sum(vals_f, mask=mask)
        sum_sq += tl.sum(vals_f * vals_f, mask=mask)
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    # compute rsqrt var+eps
    rstd = tl.rsqrt(var + eps)
    # second loop for normalized output
    for off in range(0, C, BLOCK_C):
        cur_off = off + tl.arange(0, BLOCK_C)
        mask = cur_off < C
        vals = tl.load(x_ptrs + cur_off * stride_c, mask=mask, other=0.0)
        vals_f = vals.to(tl.float32)
        norm = (vals_f - mean) * rstd
        # apply weight and bias
        w = tl.load(weight_ptr + cur_off, mask=mask, other=0.0).to(tl.float32)
        b_val = tl.load(bias_ptr + cur_off, mask=mask, other=0.0).to(tl.float32)
        out = norm * w + b_val
        # store as float32
        tl.store(y_ptr + b*stride_b0 + i*stride_b1 + j*stride_b2 + cur_off * stride_c, out.to(tl.float32), mask=mask)
```

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    k = 0
    while k < K:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # pointers
        a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # combined masks
        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]

        a = tl.load(a_ptrs, mask=mask_a, other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0).to(tl.float32)

    mask_c = mask_m[:, None] & mask_n[None, :]

"""
TriMul forward (outgoing) – a fused Triton implementation of the core
pairwise‑multiplicative update.

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Project the normalized tensor to a hidden dimension H with two linear layers
   (left_proj / right_proj) and optionally apply a binary mask.
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   normalized tensor and apply the first two to the projections.
4. The computational bottleneck is
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication of shape
       (B·H, N, N) × (B·H, N, N)ᵀ → (B·H, N, N).
   A custom Triton kernel performs this GEMM using tensor‑core friendly
   block‑wise dot‑products.
5. Reshape the result back to (B,N,N,H), apply a second layer‑norm,
   multiply by the out_gate, and finally map back to the original channel
   dimension C with a linear layer.

Only the matrix multiplication (step 4) is executed in Triton; the rest
uses PyTorch primitives, ensuring correctness while still gaining a large
speed‑up on H100 GPUs.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    # strides for tensor A (batch, M, K)
    stride_a_batch, stride_a_m, stride_a_k,
    # strides for tensor B (batch, K, N)
    stride_b_batch, stride_b_k, stride_b_n,
    # strides for tensor C (batch, M, N)
    stride_c_batch, stride_c_m, stride_c_n,
    # matrix dimensions
    M, N, K,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   (all tensors are float32)."""
    pid_batch = tl.program_id(0)  # batch dimension (B·H)
    pid_m = tl.program_id(1)      # tile along M (i‑axis)
    pid_n = tl.program_id(2)      # tile along N (j‑axis)

    # Offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    k = 0
    while k < K:
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers for the current block
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # Load with bounds checks
        mask_a = mask_m[:, None] & mask_k[None, :]
        mask_b = mask_k[:, None] & mask_n[None, :]
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Tile‑wise dot product (M×K)·(K×N) → (M×N)
        acc += tl.dot(a, b)

        k += BLOCK_K

    # Store the result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operator.
    Args:
        data: (input, mask, weights, config)
            input  – Tensor [B, N, N, C] (float32)
            mask   – Tensor [B, N, N] (bool/float) or None
            weights – dict with all linear / layer‑norm parameters
            config – dict containing:
                     "dim" (C), "hidden_dim" (H), and "nomask" flag.
    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------ #
    # unpack arguments
    # ------------------------------------------------------------------ #
    x, mask, weights, config = data
    device = x.device
    B, N, _, C = x.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", True)

    # ------------------------------------------------------------------ #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------ #
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B,N,N,dim] – still float32

    # ------------------------------------------------------------------ #
    # 2) Linear projections (left/right) and optional mask
    # ------------------------------------------------------------------ #
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    if use_mask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] for broadcast
        mask_fp = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_fp
        right = right * mask_fp

    # ------------------------------------------------------------------ #
    # 3) Gating
    # ------------------------------------------------------------------ #
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------ #
    # 4) Core batched matrix multiplication via Triton
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------ #
    # reshape to (B·H, N, N) for A and (B·H, N, N)ᵀ for B
    left_mat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # note the transpose on the last two dims of right
    right_mat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    out_mat = torch.empty((B * hidden_dim, N, N), device=device, dtype=torch.float32)

    # block sizes – tuned for H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # batch tiles
        (N + BLOCK_M - 1) // BLOCK_M,       # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,       # tiles along N (j)
    )

    batched_matmul_kernel[grid](
        a_ptr=left_mat,
        b_ptr=right_mat,
        c_ptr=out_mat,
        stride_a_batch=left_mat.stride(0),
        stride_a_m=left_mat.stride(1),
        stride_a_k=left_mat.stride(2),
        stride_b_batch=right_mat.stride(0),
        stride_b_k=right_mat.stride(1),
        stride_b_n=right_mat.stride(2),
        stride_c_batch=out_mat.stride(0),
        stride_c_m=out_mat.stride(1),
        stride_c_n=out_mat.stride(2),
        M=N,
        N=N,
        K=N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------ #
    # 5) Second LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------ #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])  # back to [B,N,N,dim]

    return out
