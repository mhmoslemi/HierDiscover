def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, N2, dim = input_tensor.shape
    assert N == N2

    # Config
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Broadcast to match shape (1,1,1,dim)
    x = input_tensor
    # Compute mean and var over last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    # Apply weight and bias
    x_norm = x_norm * norm_weight + norm_bias

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Compute gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Convert to half
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h = right_gate.to(torch.float16)
    out_gate_h = out_gate.to(torch.float16)

    # mask
    if mask is None or (config.get("nomask", False) and mask is None):
        # use all ones
        mask_tensor = torch.ones((bs, N, N), dtype=torch.float16, device=device)
    else:
        mask_tensor = mask.to(torch.float16)
    # Expand dims later broadcast via kernel.

    # Allocate outputs
    left_out = torch.empty_like(left_h)
    right_out = torch.empty_like(right_h)

    # Launch kernel
    # Compute grid size
    total_elements = left_h.numel()
    BLOCK_SIZE = 1024  # maybe tune. Must be multiple of 256 maybe.
    grid = lambda meta: ( (total_elements + meta['BLOCK_SIZE'] - 1) // meta['BLOCK_SIZE'], )
    mask_gate_kernel[grid](left_h, right_h, left_gate_h, right_gate_h, mask_tensor,
                           left_out, right_out,
                           B=bs, N=N, HD=hidden_dim,
                           BLOCK_SIZE=BLOCK_SIZE)

    # Now compute out = bmm
    # reshape
    left_reshaped = left_out.view(bs * hidden_dim, N, N)
    right_reshaped = right_out.view(bs * hidden_dim, N, N)

    out = torch.bmm(left_reshaped, right_reshaped.transpose(1, 2))
    out = out.view(bs, N, N, hidden_dim).to(torch.float32)

    # Apply to_out_norm layernorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Compute mean var across hidden_dim
    mean2 = out.mean(dim=-1, keepdim=True)
    var2 = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean2) / torch.sqrt(var2 + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # Multiply with out_gate (converted to float32)
    out_gate_f32 = out_gate_h.to(torch.float32)
    out = out_norm * out_gate_f32

    # Final linear to output dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    # shape [B,N,N,dim]
    return out

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The algorithm follows the PyTorch reference:
1. Layer‑norm the input tensor (shape [B,N,N,D]).
2. Linear projections (left/right) and three gating linear layers (no bias).
3. Apply sigmoid to gating outputs.
4. Fuse mask, left/right projections and their gates in a Triton kernel:
   left_out = left * left_gate * mask
   right_out = right * right_gate * mask
5. Compute the pairwise multiplicative update:
   out[b,i,j,d] = Σ_k left_out[b,i,k,d] * right_out[b,j,k,d]
   This is performed by reshaping to (B*HD,N,N) and using torch.bmm.
6. Layer‑norm the hidden tensor, multiply by the out‑gate, and a final
   linear projection back to the original dimension.
All heavy element‑wise work (step 4) is done in Triton; the large
batch‑matrix multiplication (step 5) uses the highly‑optimized cuBLAS
backend via torch.bmm.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, HD,
    BLOCK_SIZE: tl.constexpr
):
    """Fuse mask, left/right projections and gating.

    Arguments:
        *_ptr: pointers to contiguous tensors of shape [B,N,N,HD] (except mask).
        B, N, HD: runtime dimensions.
        mask_ptr points to a contiguous tensor of shape [B,N,N] (broadcasted
        across the HD dimension).
        BLOCK_SIZE: compile‑time block size.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)

    total = B * N * N * HD            # total elements in the 4‑D tensors
    mask_total = B * N * N            # total elements in the mask

    in_bounds = offs < total

    # Load inputs (masked loads return 0.0 outside valid range)
    left      = tl.load(left_ptr      + offs, mask=in_bounds, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=in_bounds, other=0.0)
    left_gate = tl.load(left_gate_ptr + offs, mask=in_bounds, other=0.0)
    right_gate= tl.load(right_gate_ptr+ offs, mask=in_bounds, other=0.0)

    # Broadcast mask: each mask entry corresponds to HD consecutive elements
    mask_idx = offs // HD
    mask_valid = mask_idx < mask_total
    mask_val = tl.load(mask_ptr + mask_idx, mask=mask_valid, other=1.0)

    # Apply gates and mask
    out_l = left * left_gate * mask_val
    out_r = right * right_gate * mask_val

    # Store results
    tl.store(out_left_ptr  + offs, out_l, mask=in_bounds)
    tl.store(out_right_ptr + offs, out_r, mask=in_bounds)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, D] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool/float, may be all‑ones)
            - weights: dict of model parameters (float32 tensors)
            - config: dict containing at least:
                * "dim": input/output channel dimension D
                * "hidden_dim": hidden dimension H
                * optional "nomask": bool indicating no masking

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 1️⃣ Input LayerNorm (weight & bias are from `weights["norm.*"]`)
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # Compute mean / variance across the last dimension (D)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_w + norm_b                         # broadcasting over D

    # --------------------------------------------------------------------- #
    # 2️⃣ Linear projections (no bias) and gating logits
    # --------------------------------------------------------------------- #
    left      = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right     = F.linear(x_norm, weights["right_proj.weight"]) # [B,N,N,H]

    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate= torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate  = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # --------------------------------------------------------------------- #
    # 3️⃣ Cast to half‑precision for the Triton kernel
    # --------------------------------------------------------------------- #
    left_h      = left.to(torch.float16)
    right_h     = right.to(torch.float16)
    left_gate_h = left_gate.to(torch.float16)
    right_gate_h= right_gate.to(torch.float16)
    out_gate_h  = out_gate.to(torch.float16)

    # --------------------------------------------------------------------- #
    # 4️⃣ Prepare mask (broadcasted across H)
    # --------------------------------------------------------------------- #
    if nomask:
        mask_f16 = torch.ones((B, N, N), dtype=torch.float16, device=device)
    else:
        mask_f16 = mask.to(torch.float16)

    # Allocate destination tensors for the kernel
    left_out  = torch.empty_like(left_h)
    right_out = torch.empty_like(right_h)

    # --------------------------------------------------------------------- #
    # 5️⃣ Launch the fused mask‑gate Triton kernel
    # --------------------------------------------------------------------- #
    total_elems = left_h.numel()
    BLOCK_SIZE = 1024  # Tunable; works for all tested sizes
    grid = lambda meta: ((total_elems + meta['BLOCK_SIZE'] - 1) // meta['BLOCK_SIZE'],)

    _mask_gate_kernel[grid](
        left_h, right_h, left_gate_h, right_gate_h, mask_f16,
        left_out, right_out,
        B, N, H,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------------- #
    # 6️⃣ Pairwise multiplicative update (einsum equivalent)
    #    out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
    #    Implemented via batched GEMM on (B*H) groups.
    # --------------------------------------------------------------------- #
    # Reshape to (B*H, N, N) for both tensors
    left_flat  = left_out.view(B * H, N, N)
    right_flat = right_out.view(B * H, N, N)
    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, N, N, H).to(torch.float32)  # back to FP32 for norm

    # --------------------------------------------------------------------- #
    # 7️⃣ Hidden‑dim LayerNorm, out‑gate, and final projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    # LayerNorm over hidden dimension H
    mean_h = out.mean(dim=-1, keepdim=True)
    var_h  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_h) / torch.sqrt(var_h + eps)
    out_norm = out_norm * to_out_norm_w + to_out_norm_b

    # Multiply by out‑gate (broadcasted to float32)
    out = out_norm * out_gate_h.to(torch.float32)

    # Final linear projection back to dimension D
    out = F.linear(out, weights["to_out.weight"])

    return out
