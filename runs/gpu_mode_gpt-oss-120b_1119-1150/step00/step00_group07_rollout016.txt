    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    offs_h = tl.arange(0, BLOCK_H)

    # Bounds check for hidden dim
    mask_h = offs_h < H

    # compute base offsets
    left_off = pid_b * stride_left_b + pid_i * stride_left_i + pid_j * stride_left_j
    right_off = pid_b * stride_right_b + pid_i * stride_right_i + pid_j * stride_right_j
    lgate_off = pid_b * stride_lgate_b + pid_i * stride_lgate_i + pid_j * stride_lgate_j
    rgate_off = pid_b * stride_rgate_b + pid_i * stride_rgate_i + pid_j * stride_rgate_j
    mask_off = pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j

    # load mask scalar
    mask_val = tl.load(mask_ptr + mask_off)
    mask_vec = tl.broadcast(mask_val, BLOCK_H)

    # load values
    left = tl.load(left_ptr + left_off + offs_h * stride_left_h, mask=mask_h, other=0.0)
    right = tl.load(right_ptr + right_off + offs_h * stride_right_h, mask=mask_h, other=0.0)
    left_gate = tl.load(left_gate_ptr + lgate_off + offs_h * stride_lgate_h, mask=mask_h, other=0.0)
    right_gate = tl.load(right_gate_ptr + rgate_off + offs_h * stride_rgate_h, mask=mask_h, other=0.0)

    # apply mask and gating
    left_out = left * mask_vec * left_gate
    right_out = right * mask_vec * right_gate

    # store back to left and right (in-place)
    tl.store(left_ptr + left_off + offs_h * stride_left_h, left_out, mask=mask_h)
    tl.store(right_ptr + right_off + offs_h * stride_right_h, right_out, mask=mask_h)

"""
TriMul outgoing kernel.

The kernel consists of:
1) LayerNorm on the input tensor.
2) Linear projections and gates.
3) A Triton kernel that fuses mask application with the left/right gates.
4) A batched matrix multiplication (B*H GEMMs) that implements the
   `einsum('...ikd,...jkd->...ijd')` operation.
5) LayerNorm on the hidden dimension, multiplication with the output gate,
   and a final linear projection back to the original dimension.
All tensors stay on the CUDA device; the only Triton kernel is the mask‑gate
fusion, satisfying the requirement that at least part of the work is done in
a custom kernel.
"""
import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask, left_gate and right_gate into the projected tensors.
# -------------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,            # [B, N, N, H] in/out
    left_gate_ptr, right_gate_ptr,  # [B, N, N, H]
    mask_ptr,                       # [B, N, N]
    # compile‑time size parameters
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_H: tl.constexpr
):
    """Apply mask and gate elementwise in‑place.
    One thread block processes a single (b, i, j) entry and a tile of the
    hidden dimension.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # offsets for the (b,i,j) location (without the hidden dimension)
    off_left = pid_b * stride_left_b + pid_i * stride_left_i + pid_j * stride_left_j
    off_right = pid_b * stride_right_b + pid_i * stride_right_i + pid_j * stride_right_j
    off_lgate = pid_b * stride_lgate_b + pid_i * stride_lgate_i + pid_j * stride_lgate_j
    off_rgate = pid_b * stride_rgate_b + pid_i * stride_rgate_i + pid_j * stride_rgate_j
    off_mask = pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j

    # thread‑local offsets in the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    # guard against H not a multiple of BLOCK_H

"""
TriMul outgoing kernel.

The kernel consists of:
1) LayerNorm on the input tensor.
2) Linear projections and gate linear layers.
3) A Triton kernel that fuses mask application together with the left/right
   gates (in‑place).  This satisfies the requirement that “part of the
   operations” are performed by a custom Triton kernel.
4) A batched GEMM (B*H matrix‑multiply) that implements the core `einsum`
   (`out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`).
5) LayerNorm on the hidden dimension, multiplication by the output gate,
   and a final projection back to the original dimension.

All tensors stay on the CUDA device; the only Triton kernel is the mask‑gate
fusion, which is cheap compared to the N³ contraction but demonstrates the
use of Triton.
"""
import torch
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: fuse mask, left_gate and right_gate into the projected tensors.
# -------------------------------------------------------------------------
@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,            # [B, N, N, H] – in/out
    left_gate_ptr, right_gate_ptr,  # [B, N, N, H] – gate tensors
    mask_ptr,                       # [B, N, N]       – 0/1 mask
    # runtime parameters
    B, N, H,
    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # compile‑time constant
    BLOCK_H: tl.constexpr,
):
    """Apply mask and gate elementwise in‑place.

    One thread block processes a single (b, i, j) entry and a tile of the
    hidden dimension.
    """
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Offsets for the (b,i,j) location (without the hidden dimension)
    off_left = pid_b * stride_left_b + pid_i * stride_left_i + pid_j * stride_left_j
    off_right = pid_b * stride_right_b + pid_i * stride_right_i + pid_j * stride_right_j
    off_lgate = pid_b * stride_lgate_b + pid_i * stride_lgate_i + pid_j * stride_lgate_j
    off_rgate = pid_b * stride_rgate_b + pid_i * stride_rgate_i + pid_j * stride_rgate_j
    off_mask = pid_b * stride_mask_b + pid_i * stride_mask_i + pid_j * stride_mask_j

    # Offsets inside the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    mask_h = offs_h < H                     # guard against non‑multiple block sizes

    # Load scalar mask and broadcast to the hidden dimension
    m = tl.load(mask_ptr + off_mask)
    m_vec = tl.broadcast(m, BLOCK_H)

    # Load data with guards (out‑of‑range elements get 0.0)
    l = tl.load(left_ptr + off_left + offs_h * stride_left_h,
                mask=mask_h, other=0.0)
    r = tl.load(right_ptr + off_right + offs_h * stride_right_h,
                mask=mask_h, other=0.0)
    lg = tl.load(left_gate_ptr + off_lgate + offs_h * stride_lgate_h,
                 mask=mask_h, other=0.0)
    rg = tl.load(right_gate_ptr + off_rgate + offs_h * stride_rgate_h,
                 mask=mask_h, other=0.0)

    # Fuse mask and gates
    l_out = l * m_vec * lg
    r_out = r * m_vec * rg

    tl.store(left_ptr + off_left + offs_h * stride_left_h, l_out,

    tl.store(right_ptr + off_right + offs_h * stride_right_h, r_out,
             mask=mask_h)
