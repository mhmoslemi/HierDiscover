import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def fused_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE_I: tl.constexpr, BLOCK_SIZE_J: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    # Map pid to (i0, j0)
    num_i = tl.cdiv(N, BLOCK_SIZE_I)
    i0 = (pid // num_i) * BLOCK_SIZE_I
    j0 = (pid % num_i) * BLOCK_SIZE_J

    # create offsets
    offs_i = i0 + tl.arange(0, BLOCK_SIZE_I)
    offs_j = j0 + tl.arange(0, BLOCK_SIZE_J)
    offs_h = tl.arange(0, H)

    mask = tl.load(mask_ptr + (offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j), mask= (offs_i[:, None] < N) & (offs_j[None, :] < N), other=0.0)

    # Compute left projection values
    left_proj = tl.load(left_proj_ptr + (offs_i[:, None, None] * stride_lp_i + offs_j[None, :, None] * stride_lp_j + offs_h[None, None, :] * stride_lp_h), mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H), other=0.0)

    left_gate = tl.load(left_gate_ptr + (offs_i[:, None, None] * stride_lg_i + offs_j[None, :, None] * stride_lg_j + offs_h[None, None, :] * stride_lg_h), mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H), other=0.0)

    right_proj = tl.load(right_proj_ptr + (offs_i[:, None, None] * stride_rp_i + offs_j[None, :, None] * stride_rp_j + offs_h[None, None, :] * stride_rp_h), mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H), other=0.0)

    right_gate = tl.load(right_gate_ptr + (offs_i[:, None, None] * stride_rg_i + offs_j[None, :, None] * stride_rg_j + offs_h[None, None, :] * stride_rg_h), mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H), other=0.0)

    # apply mask: mask broadcast over hidden dimension
    mask_expand = mask[:, :, None]

```
@triton.jit
def fused_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    num_i = tl.cdiv(N, BLOCK_I)
    i0 = (pid // num_i) * BLOCK_I
    j0 = (pid % num_i) * BLOCK_J
    
    # Offsets
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, H)
    
    mask = tl.load(mask_ptr + (offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j), mask= (offs_i[:, None] < N) & (offs_j[None, :] < N), other=0.0)
    mask = mask[:, :, None]  # broadcast over hidden
    # Load tensors
    left_proj = tl.load(left_proj_ptr + (offs_i[:, None, None] * stride_lp_i + offs_j[None, :, None] * stride_lp_j + offs_h[None, None, :] * stride_lp_h), mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H), other=0.0)
    left_gate = tl.load(left_gate_ptr + (offs_i[:, None, None] * stride_lg_i + offs_j[None, :, None] * stride_lg_j + offs_h[None, None, :] * stride_lg_h), mask=..., other=0.0)
    right_proj = tl.load(right_proj_ptr + (offs_i[:, None, None] * stride_rp_i + offs_j[None, :, None] * stride_rp_j + offs_h[None, None, :] * stride_rp_h), mask=..., other=0.0)
    right_gate = tl.load(right_gate_ptr + (offs_i[:, None, None] * stride_rg_i + offs_j[None, :, None] * stride_rg_j + offs_h[None, None, :] * stride_rg_h), mask=..., other=0.0)

    # Apply sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    out_left = left_proj * mask * left_gate_sig
    out_right = right_proj * mask * right_gate_sig

    # Store
    tl.store(out_left_ptr + (offs_i[:, None, None] * stride_lp_i + offs_j[None, :, None] * stride_lp_j + offs_h[None, None, :] * stride_lp_h), out_left, mask= (offs_i[:, None, None] < N) & (offs_j[None, :, None] < N) & (offs_h[None, None, :] < H))

```

    # mask: shape [B,N,N] or None

    assert N==N_

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])
    left_gate = F.linear(x, weights["left_gate.weight"])
    right_gate = F.linear(x, weights["right_gate.weight"])
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Mask handling
    if mask is None or config.get("nomask", False):
        mask = torch.ones(input_tensor.shape[:3], dtype=x.dtype, device=x.device)
    # Ensure mask dtype matches
    mask = mask.to(x.dtype)

    # Fuse mask and gates using Triton
    left_gated, right_gated = fused_mask_gate(left_proj, right_proj, left_gate, right_gate, mask)

    # Compute TriMul via batched GEMM
    B, N, _, H = left_gated.shape
    # reshape to (B*H, N, N)
    left_flat = left_gated.permute(0,3,1,2).contiguous().view(B*H, N, N).to(torch.float16)
    right_flat = right_gated.permute(0,3,1,2).contiguous().view(B*H, N, N).to(torch.float16)
    # right needs transpose over last two dimensions
    out_flat = torch.bmm(left_flat, right_flat.transpose(1,2))  # shape (B*H,N,N)
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).to(torch.float32)  # back to (B,N,N,H)

    # Apply out_gate
    out = out * out_gate

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Final linear projection
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape (B,N,N,dim)

    return out

@triton.jit
def fused_gate_mask_kernel(
    left_proj_ptr, right_proj_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    B, N, H,
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
):
    # Determine tile indices
    pid = tl.program_id(axis=0)
    batch_id = tl.program_id(axis=1)

    # Compute tile offsets for i and j
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    i0 = (pid // num_i_tiles) * BLOCK_I
    j0 = (pid % num_i_tiles) * BLOCK_J

    # Create ranges within tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, H)

    # Compute valid masks
    i_valid = offs_i < N
    j_valid = offs_j < N
    h_valid = offs_h < H

    # Load mask (2D) and broadcast to hidden dim
    mask = tl.load(
        mask_ptr + batch_id * stride_mask_b + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0
    )
    mask = mask[:, :, None]  # shape (BLOCK_I, BLOCK_J, 1)

    # Compute base pointers for each tensor for this batch
    lp_base = left_proj_ptr + batch_id * stride_lp_b
    rp_base = right_proj_ptr + batch_id * stride_rp_b
    lg_base = left_gate_ptr + batch_id * stride_lg_b
    rg_base = right_gate_ptr + batch_id * stride_rg_b
    outl_base = out_left_ptr + batch_id * stride_lp_b  # same layout as left_proj
    outr_base = out_right_ptr + batch_id * stride_rp_b

    # Load data
    # left_proj
    left = tl.load(
        lp_base + offs_i[:, None, None] * stride_lp_i
                     + offs_j[None, :, None] * stride_lp_j
                     + offs_h[None, None, :] * stride_lp_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )
    # left_gate
    left_gate = tl.load(
        lg_base + offs_i[:, None, None] * stride_lg_i
                     + offs_j[None, :, None] * stride_lg_j
                     + offs_h[None, None, :] * stride_lg_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )
    # right_proj
    right = tl.load(
        rp_base + offs_i[:, None, None] * stride_rp_i
                     + offs_j[None, :, None] * stride_rp_j
                     + offs_h[None, None, :] * stride_rp_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )
    # right_gate
    right_gate = tl.load(
        rg_base + offs_i[:, None, None] * stride_rg_i
                     + offs_j[None, :, None] * stride_rg_j
                     + offs_h[None, None, :] * stride_rg_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )

    # Sigmoid
    left_gate_sig = 1.0 / (1.0 + tl.exp(-left_gate))
    right_gate_sig = 1.0 / (1.0 + tl.exp(-right_gate))

    # Apply mask and gates
    left_out = left * mask * left_gate_sig
    right_out = right * mask * right_gate_sig

    # Store results
    tl.store(
        outl_base + offs_i[:, None, None] * stride_lp_i
                     + offs_j[None, :, None] * stride_lp_j
                     + offs_h[None, None, :] * stride_lp_h,
        left_out,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
    )
    tl.store(
        outr_base + offs_i[:, None, None] * stride_rp_i
                     + offs_j[None, :, None] * stride_rp_j
                     + offs_h[None, None, :] * stride_rp_h,
        right_out,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
    )
```

@triton.jit
def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
):
    pid = tl.program_id(axis=0)
    batch_id = tl.program_id(axis=1)

    num_i_tiles = tl.cdiv(N, BLOCK_I)
    i0 = (pid // num_i_tiles) * BLOCK_I
    j0 = (pid % num_i_tiles) * BLOCK_J

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, H)

    i_valid = offs_i < N
    j_valid = offs_j < N
    h_valid = offs_h < H

    # Base pointers for current batch
    proj_base = proj_ptr + batch_id * stride_proj_b
    gate_base = gate_ptr + batch_id * stride_gate_b
    out_base = out_ptr + batch_id * stride_proj_b
    mask_base = mask_ptr + batch_id * stride_mask_b

    # Load mask (2D) and broadcast
    mask = tl.load(
        mask_base + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0
    )
    mask = mask[:, :, None]

    # Load proj and gate
    proj = tl.load(
        proj_base + offs_i[:, None, None] * stride_proj_i
                     + offs_j[None, :, None] * stride_proj_j
                     + offs_h[None, None, :] * stride_proj_h,
        mask= i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )
    gate = tl.load(
        gate_base + offs_i[:, None, None] * stride_gate_i
                     + offs_j[None, :, None] * stride_gate_j
                     + offs_h[None, None, :] * stride_gate_h,
        mask= i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0
    )
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    out = proj * mask * gate_sig

    tl.store(
        out_base + offs_i[:, None, None] * stride_proj_i
                     + offs_j[None, :, None] * stride_proj_j
                     + offs_h[None, None, :] * stride_proj_h,
        out,
        mask= i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
    )
```

"""
TriMul (outgoing) forward pass – fast Triton‑accelerated implementation.

The heavy O(N³·C_hidden) “triangle multiplicative update” is performed with a
batched GEMM (leveraging cuBLAS via torch.bmm) on FP16 tensors.
A small Triton kernel fuses the element‑wise mask, left/right gating and sigmoid,
which reduces memory traffic compared with a pure PyTorch implementation.
All other linear / LayerNorm steps are kept in PyTorch for simplicity and
to benefit from highly‑optimized kernels.

Input:
    data = (x, mask, weights, config)
    x        : [B, N, N, C]                (float32)
    mask     : [B, N, N] or None           (float32 or bool)
    weights  : dict of all model parameters
    config   : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

Output:
    out : [B, N, N, C] (float32)

The implementation follows the reference PyTorch module exactly, except that the
mask+gate fusion is done with Triton, and the N³ reduction uses an efficient
batched half‑precision matrix multiplication.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fuse mask, gate (sigmoid) and projection.
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,          # pointers
    B, N, H,                                        # sizes
    stride_proj_b, stride_proj_i, stride_proj_j, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
):
    """
    Compute  out = proj * mask * sigmoid(gate)
    for a tensor of shape (B, N, N, H).
    """
    pid = tl.program_id(0)        # tile over the (i, j) plane
    batch_id = tl.program_id(1)   # batch dimension

    # ----- tile offsets -------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    i0 = (pid // num_i_tiles) * BLOCK_I
    j0 = (pid %  num_i_tiles) * BLOCK_J

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = tl.arange(0, H)

    i_valid = offs_i < N
    j_valid = offs_j < N
    h_valid = offs_h < H

    # ----- base pointers for the current batch -------------------------
    proj_base = proj_ptr + batch_id * stride_proj_b
    gate_base = gate_ptr + batch_id * stride_gate_b
    out_base  = out_ptr  + batch_id * stride_proj_b   # same layout as proj
    mask_base = mask_ptr + batch_id * stride_mask_b

    # ----- load mask (2‑D) and broadcast -------------------------------
    mask = tl.load(
        mask_base + offs_i[:, None] * stride_mask_i + offs_j[None, :] * stride_mask_j,
        mask=i_valid[:, None] & j_valid[None, :],
        other=0.0,
    )
    mask = mask[:, :, None]                     # shape (BLOCK_I, BLOCK_J, 1)

    # ----- load projection and gate ------------------------------------
    proj = tl.load(
        proj_base + offs_i[:, None, None] * stride_proj_i
                     + offs_j[None, :, None] * stride_proj_j
                     + offs_h[None, None, :] * stride_proj_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    gate = tl.load(
        gate_base + offs_i[:, None, None] * stride_gate_i
                     + offs_j[None, :, None] * stride_gate_j
                     + offs_h[None, None, :] * stride_gate_h,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
        other=0.0,
    )
    # sigmoid
    gate_sig = 1.0 / (1.0 + tl.exp(-gate))

    # ----- compute fused output ----------------------------------------
    out = proj * mask * gate_sig

    # ----- store --------------------------------------------------------
    tl.store(
        out_base + offs_i[:, None, None] * stride_proj_i
                     + offs_j[None, :, None] * stride_proj_j
                     + offs_h[None, None, :] * stride_proj_h,
        out,
        mask=i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the above kernel
# ----------------------------------------------------------------------
def apply_gate_mask(proj, gate, mask):
    """
    proj, gate: (B, N, N, H) float32
    mask      : (B, N, N)   float32 (0.0/1.0)
    Returns: (B, N, N, H) float32 = proj * mask * sigmoid(gate)
    """
    B, N, _, H = proj.shape
    out = torch.empty_like(proj)

    BLOCK_I = 64
    BLOCK_J = 64
    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid = (grid_i * grid_j, B)   # (tiles, batch)

    apply_gate_mask_kernel[grid](
        proj.data_ptr(),
        gate.data_ptr(),
        mask.data_ptr(),
        out.data_ptr(),
        B, N, H,
        proj.stride(0), proj.stride(1), proj.stride(2), proj.stride(3),
        gate.stride(0), gate.stride(1), gate.stride(2), gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
    )
    return out


# ----------------------------------------------------------------------
# Main entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing Triangle Multiplicative Update.
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]               # C
    hidden_dim = config["hidden_dim"] # H
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate_raw  = F.linear(x, weights["left_gate.weight"])
    right_gate_raw = F.linear(x, weights["right_gate.weight"])
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 3. Mask (if any)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones(x.shape[:3], dtype=dtype, device=device)
    else:
        # ensure float mask (0/1) on same device/dtype
        mask = mask.to(dtype=dtype, device=device)

    # ------------------------------------------------------------------
    # 4. Fuse mask + sigmoid(gate) for left and right tensors (Triton)
    # ------------------------------------------------------------------
    left = apply_gate_mask(left_proj, left_gate_raw, mask)   # [B,N,N,H]
    right = apply_gate_mask(right_proj, right_gate_raw, mask)

    # ------------------------------------------------------------------
    # 5. Triangle multiplicative update: batched GEMM over the "k" dimension
    #    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape

    # reshape to (B*H, N, N) and cast to fp16 for speed
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * H, N, N).to(torch.float16)

    # GEMM: left * right^T  (right needs its k‑dim transposed)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))          # (B*H, N, N)

    # reshape back to (B, N, N, H) in fp32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7. Output LayerNorm over hidden dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 8. Final linear projection back to dim
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]       # shape (dim, hidden_dim)
    out = F.linear(out, to_out_weight)             # (B, N, N, dim)

    return out
