def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # Apply mask if present
    if mask_tensor is not None:
        mask_exp = mask_tensor.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # Strides
    stride_bl, stride_il, stride_kl, stride_dl = left_fp16.stride()
    stride_br, stride_ir, stride_kr, stride_dr = right_fp16.stride()
    stride_bo, stride_io, stride_jo, stride_do = out.stride()

    BLOCK_N = 64
    BLOCK_K = 64

"""
TriMul (outgoing) – Triton implementation
-----------------------------------------

The TriMul operation processes a 4‑D tensor X ∈ ℝ^{B×N×N×C}
(the pairwise representation of a protein sequence) and returns a tensor
with the same shape.  The forward pass consists of:

1. Layer‑norm over the last channel dimension (C).
2. Linear projections to a hidden dimension H (no bias):
   left = X·W_left ,   right = X·W_right
3. Gating networks (sigmoid) also on the normalized X:
   g_left  = σ(X·W_gleft)   (∈ ℝ^{B×N×N×H})
   g_right = σ(X·W_gright)
   g_out   = σ(X·W_gout)   (used after the multiplicative update)
4. Optional binary mask M ∈ {0,1}^{B×N×N}:
   left  ← left  * M   * g_left
   right ← right * M   * g_right
5. **Core kernel** – for every batch b and hidden channel h compute

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   which is an outer‑product over the sequence dimension.
   This is exactly a batched matrix multiplication
   (left_h @ right_hᵀ) for each hidden channel.
   The kernel tiles the i‑ and j‑dimensions (BLOCK_N×BLOCK_N) and
   reduces over k in blocks of size BLOCK_K.  Input tensors are stored
   in FP16 for bandwidth, accumulation is performed in FP32 and the
   final result is written as FP32.
6. Layer‑norm over the hidden dimension, element‑wise multiplication with
   g_out, and a final linear projection back to the original channel size.

Only the heavy O(N³·H) reduction (step 5) is executed in Triton; all
other operations use native PyTorch kernels, which are already highly
optimized.  The implementation works for any
B∈{1,2}, N∈{128,256,512,1024}, C∈{128,384}, H=128 (as used in the
test configuration) and supports both masked and un‑masked inputs.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Strides (in elements)
    stride_bl, stride_il, stride_kl, stride_dl,
    stride_br, stride_ir, stride_kr, stride_dr,
    stride_bo, stride_io, stride_jo, stride_do,
    # Compile‑time constants
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a tile (BLOCK_N × BLOCK_N) of the (i,j) matrix.
    The kernel is instantiated for every (b,h) pair.
    """
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile column
    pid_bd = tl.program_id(2)         # combined batch & hidden index

    # decode batch and hidden channel from pid_bd
    b = pid_bd // H
    h = pid_bd - b * H

    # start indices of the tile
    i_start = pid_i * BLOCK_N
    j_start = pid_j * BLOCK_N

    # block coordinates
    i_offset = i_start + tl.arange(0, BLOCK_N)
    j_offset = j_start + tl.arange(0, BLOCK_N)

    # predicates for out‑of‑bounds rows / cols
    mask_i = i_offset < N
    mask_j = j_offset < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_N, BLOCK_N), dtype=tl.float32)

    # number of k‑blocks
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_offset = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- load left[i, k, h] -------------------------------------------------
        # left layout: (b, i, k, h)
        left_ptrs = (
            left_ptr
            + b * stride_bl
            + i_offset[:, None] * stride_il
            + k_offset[None, :] * stride_kl
            + h * stride_dl
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)  # cast from FP16 → FP32

        # ----- load right[j, k, h] transposed to (k, j) ---------------------------
        # right original layout: (b, j, k, h)
        # we need right_T[k, j, h] = right[b, j, k, h]
        right_ptrs = (
            right_ptr
            + b * stride_br
            + k_offset[:, None] * stride_kr   # k dimension first
            + j_offset[None, :] * stride_ir   # j dimension second
            + h * stride_dr
        )
        right_T = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- dot product over the k‑dimension ------------------------------------
        # left: (BLOCK_N, BLOCK_K) , right_T: (BLOCK_K, BLOCK_N)
        acc += tl.dot(left, right_T)

    # write back the tile
    out_ptrs = (
        out_ptr
        + b * stride_bo
        + i_offset[:, None] * stride_io
        + j_offset[None, :] * stride_jo
        + h * stride_do
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------------
    # unpack input
    # -------------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]            # original channel dimension C
    hidden_dim = config["hidden_dim"]  # projection dimension H

    device = input_tensor.device
    dtype = input_tensor.dtype

    # -------------------------------------------------------------------------
    # 1) Input layer‑norm
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [B,N,N,dim]

    # -------------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B,N,N,hidden_dim]
    # -------------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # -------------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # -------------------------------------------------------------------------
    # 4) Optional mask
    # -------------------------------------------------------------------------
    # config["nomask"] == True  → ignore mask even if supplied
    if not config.get("nomask", False) and mask_tensor is not None:
        # mask shape [B,N,N] → broadcast over hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -------------------------------------------------------------------------
    # 5) Apply gates
    # -------------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 6) Cast to FP16 for the heavy kernel (bandwidth‑save)
    # -------------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, H = left_fp16.shape  # B, N, N, hidden_dim

    # output buffer (FP32 as required by the rest of the model)
    out = torch.empty_like(left_fp16, dtype=torch.float32)

    # -------------------------------------------------------------------------
    # 7) Strides (in elements) – needed by the Triton kernel
    # -------------------------------------------------------------------------
    s_bl, s_il, s_kl, s_dl = left_fp16.stride()
    s_br, s_ir, s_kr, s_dr = right_fp16.stride()
    s_bo, s_io, s_jo, s_do = out.stride()

    # -------------------------------------------------------------------------
    # 8) Triton launch configuration
    # -------------------------------------------------------------------------
    BLOCK_N = 64   # tile size for i‑ and j‑axes
    BLOCK_K = 64   # reduction tile size (k‑axis)

    grid_i = (N + BLOCK_N - 1) // BLOCK_N
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_bd = B * H                     # one program per (batch, hidden channel)

    trimul_outgoing_kernel[ (grid_i, grid_j, grid_bd) ](
        left_fp16,
        right_fp16,
        out,
        s_bl, s_il, s_kl, s_dl,
        s_br, s_ir, s_kr, s_dr,
        s_bo, s_io, s_jo, s_do,
        B, N, H,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 9) Output layer‑norm, gate, and final linear projection
    # -------------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )
    out_norm = out_norm * out_gate      # element‑wise multiplication with out_gate

    final_out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    # final_out shape: [B, N, N, dim] – same dtype as model (float32)
    return final_out
