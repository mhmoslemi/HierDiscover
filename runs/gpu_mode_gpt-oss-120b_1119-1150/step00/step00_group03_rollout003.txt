    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # program_id(1) => block index i (tile rows)
    # program_id(2) => block index j (tile cols)

import torch
import triton
import triton.language as tl

def layer_norm(x, weight, bias, eps=1e-5):
    return torch.nn.functional.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)

@triton.jit
def triton_trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program id 0: batch_head (b * H + h)
    batch_head = tl.program_id(0)
    b = batch_head // H
    h = batch_head % H

    pid_m = tl.program_id(1)  # tile row
    pid_n = tl.program_id(2)  # tile col

    # offsets for each tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds rows / cols
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Iterate over reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offsets (global)
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load tiles of left and right
        # left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k

        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_j

        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)

        # accumulate dot product
        acc += tl.dot(left_tile, right_tile)

    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j

    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    Implements the forward path of the AlphaFold3 TriMul module:
    - LayerNorm on the input.
    - Linear projections (left/right) and three gates.
    - Optional mask application.
    - Batched matrix multiplication across the sequence dimension using a
      Triton kernel (the O(N³) core).
    - Output LayerNorm, gating and final linear projection.
    All tensors reside on the CUDA device. The kernel computes the
    reduction `out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]`.
    """
    import torch
    import triton
    import triton.language as tl

    # -------------------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ensure tensors are contiguous for Triton
    x = input_tensor.contiguous()
    mask = mask_tensor.contiguous() if mask_tensor is not None else None

    # -------------------------------------------------------------------------
    # Helper: LayerNorm (bias/weight provided)
    # -------------------------------------------------------------------------
    def layer_norm(x, weight, bias, eps=1e-5):
        # torch.nn.functional.layer_norm normalizes over the last dimension
        return torch.nn.functional.layer_norm(
            x, normalized_shape=weight.shape, weight=weight, bias=bias, eps=eps
        )

    # -------------------------------------------------------------------------
    # Pull weight tensors (they are already on the same device)
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # -------------------------------------------------------------------------
    # 1. Input layer norm
    # -------------------------------------------------------------------------
    x_norm = layer_norm(x, norm_weight, norm_bias)  # shape [B,N,N,dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections and gates (no bias)
    # -------------------------------------------------------------------------
    # Linear operation: y = x @ W.T
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    # -------------------------------------------------------------------------
    # 3. Apply optional mask
    # -------------------------------------------------------------------------
    if nomask or mask is None:
        mask_ = torch.ones_like(x_norm[..., :1])  # shape [B,N,N,1]
    else:
        # mask dtype may be bool/integer; convert to float
        mask_ = mask.to(x.dtype).unsqueeze(-1)   # shape [B,N,N,1]
    # broadcast mask over hidden_dim
    left = left * mask_ * left_gate
    right = right * mask_ * right_gate

    # -------------------------------------------------------------------------
    # 4. Core TriMul: batched GEMM left @ right.T per hidden channel
    # -------------------------------------------------------------------------
    B, N, _, _ = x.shape
    # Cast to half for the kernel (more efficient). The kernel accumulates in FP32.
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Allocate output tensor in FP32
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # Compute strides (contiguous layout)
    # left/right shape: (B,N,N,hidden_dim) => (b, i, k, h)
    stride_left_b = left_half.stride(0)
    stride_left_i = left_half.stride(1)
    stride_left_k = left_half.stride(2)
    stride_left_h = left_half.stride(3)

    stride_right_b = right_half.stride(0)
    stride_right_j = right_half.stride(1)  # j dimension maps to the same as i
    stride_right_k = right_half.stride(2)
    stride_right_h = right_half.stride(3)

    stride_out_b = out.stride(0)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)
    stride_out_h = out.stride(3)

    # Block sizes – tuned for H100 and hidden_dim ≤ 384
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                      # program id 0 – batch * head
        (N + BLOCK_M - 1) // BLOCK_M,        # program id 1 – row tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # program id 2 – col tiles
    )

    # Launch kernel
    triton_trimul_matmul_kernel[grid](
        left_ptr=left_half,
        right_ptr=right_half,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # small number of warps works well for these sizes
    )
    # -------------------------------------------------------------------------
    # 5. Output layer norm + gating
    # -------------------------------------------------------------------------
    out = layer_norm(out, to_out_norm_weight, to_out_norm_bias)

    # Apply output gate (shape compatible: [B,N,N,hidden_dim])
    out = out * out_gate

    # 6. Final linear projection back to original dim
    output = torch.nn.functional.linear(out, to_out_weight)  # bias=None

    return output

"""
TriMul (outgoing) – Triton‑accelerated forward pass

The implementation follows the AlphaFold3 TriMul module:
1. Layer‑norm over the channel dimension.
2. Linear projections (`left_proj`, `right_proj`) and three gates
   (`left_gate`, `right_gate`, `out_gate`).  All linear layers have no bias.
3. Optional pair‑wise mask (broadcasted over the hidden dimension) applied
   before gating.
4. Core O(N³·C) reduction:
       out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]
   realized as a batched GEMM `L @ Rᵀ` per hidden‑channel using a fused
   Triton kernel.  The kernel works on half‑precision inputs, accumulates in
   FP32 and writes FP32 output.
5. Output layer‑norm (`to_out_norm`), element‑wise gating (`out_gate`) and a
   final linear projection back to the original dimension.

All tensors are assumed to reside on the CUDA device.  The kernel only
covers the heavy matrix‑multiplication; the surrounding linear/normalization
steps are kept in PyTorch for clarity.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul reduction
# -------------------------------------------------------------------------
@triton.jit
def triton_trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute, for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    left  : (B, N, N, H) interpreted as (b, i, k, h)
    right : (B, N, N, H) interpreted as (b, j, k, h)
    out   : (B, N, N, H) interpreted as (b, i, j, h)
    """
    # -----------------------------------------------------------------
    # identifiers
    # -----------------------------------------------------------------
    batch_head = tl.program_id(0)          # 0‑th grid dimension: b*H + h
    b = batch_head // H
    h = batch_head % H

    pid_m = tl.program_id(1)               # row‑tile index
    pid_n = tl.program_id(2)               # col‑tile index

    # -----------------------------------------------------------------
    # tile offsets
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulation in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # reduction over k
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile (M×K)
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            h * stride_left_h + \
            offs_m[:, None] * stride_left_i + \
            offs_k[None, :] * stride_left_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        # right tile (K×N)
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            h * stride_right_h + \
            offs_k[:, None] * stride_right_k + \
            offs_n[None, :] * stride_right_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)

        # accumulate dot product
        acc += tl.dot(left_tile, right_tile)

    # -----------------------------------------------------------------
    # store result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        h * stride_out_h + \
        offs_m[:, None] * stride_out_i + \
        offs_n[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
