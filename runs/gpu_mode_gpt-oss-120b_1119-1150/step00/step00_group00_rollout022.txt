    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_aa, stride_am, stride_ak,
    stride_bb, stride_bk, stride_bn,
    stride_cc, stride_cm, stride_cn,

):

"""
TriMul "outgoing" implementation.

The kernel performs the heavy pairwise multiplication:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batched GEMM for each hidden channel:
    out_batch = left_batch @ right_batch.T

Steps:
1. Layer‑norm the input.
2. Linear projections for left/right and their gates (all in FP32).
3. Apply the optional mask, then multiply by the learned gates.
4. Cast to FP16 and reorder to (batch, hidden, N, N).
5. Transpose the right‑hand operand, flatten batch*hidden and launch a
   custom Triton kernel that computes a batched matrix‑multiply
   (BLOCK_M×BLOCK_N tiles, accumulating over K).  The kernel works with
   arbitrary batch sizes, sequence lengths up to 1024 and hidden_dim=128.
6. Reshape back, convert to FP32 and apply the output LayerNorm,
   the output gate and the final linear projection.
The result is a tensor of shape [bs, seq_len, seq_len, dim] (FP32).
"""

import torch
import triton
import triton.language as tl
from torch.nn.functional import linear, layer_norm, sigmoid

# --------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B.T) where A,B are (B, M, K)
# --------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    a_ptr,                     # left  (batch, M, K)
    b_ptr,                     # right_T (batch, K, N)
    c_ptr,                     # out   (batch, M, N)
    batch, M, N, K,
    stride_aa, stride_am, stride_ak,      # strides for A
    stride_bb, stride_bk, stride_bn,      # strides for B
    stride_cc, stride_cm, stride_cn,      # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------
    # program IDs
    # ------------------------------
    b_idx = tl.program_id(0)          # batch*hidden index
    pid_m = tl.program_id(1)          # block row
    pid_n = tl.program_id(2)          # block col

    # ------------------------------
    # block offsets
    # ------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    # ------------------------------
    # pointer base for this batch
    # ------------------------------
    a_base = a_ptr + b_idx * stride_aa
    b_base = b_ptr + b_idx * stride_bb
    c_base = c_ptr + b_idx * stride_cc

    # ------------------------------
    # initialise accumulator
    # ------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------
    # masks for out‑of‑bounds rows/cols
    # ------------------------------
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------
    # loop over K dimension
    # ------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        # compute current k offset
        k_offset = kk * BLOCK_K
        # masks for the current K‑tile
        cur_k = k_offset + k_range
        mask_k = cur_k < K

        # pointer for this tile
        a_ptrs = a_base + offs_m[:, None] * stride_am + cur_k[None, :] * stride_ak  # (BM, BK)
        b_ptrs = b_base + cur_k[:, None] * stride_bk + offs_n[None, :] * stride_bn  # (BK, BN)

        # load with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # accumulate
        acc += tl.dot(a, b)

    # ------------------------------
    # write result
    # ------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask_store = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_store)

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # shape: (bs, hidden, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    # transpose the right operand (now (bs, hidden, N, N) -> (bs, hidden, N, N) transposed)
    right_t = right.transpose(-2, -1).contiguous()

    # ----------------------------------------------------------
    # 6) Flatten batch & hidden dims for the batched GEMM
    # ----------------------------------------------------------
    B_total = bs * hidden_dim                     # effective batch size for the kernel
    left_flat = left.view(B_total, N, N)          # (B_total, M, K)
    right_flat = right_t.view(B_total, N, N)      # (B_total, K, N)
    out_flat = torch.empty_like(left_flat)        # (B_total, M, N), FP16

    # ----------------------------------------------------------
    # 7) Extract strides (in element counts)
    # ----------------------------------------------------------
    stride_aa = left_flat.stride(0)
    stride_am = left_flat.stride(1)
    stride_ak = left_flat.stride(2)

    stride_bb = right_flat.stride(0)
    stride_bk = right_flat.stride(1)
    stride_bn = right_flat.stride(2)

    stride_cc = out_flat.stride(0)
    stride_cm = out_flat.stride(1)
    stride_cn = out_flat.stride(2)

    # ----------------------------------------------------------
    # 8) Launch the Triton kernel
    # ----------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B_total,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B_total,
        N,
        N,
        N,
        stride_aa,
        stride_am,
        stride_ak,
        stride_bb,
        stride_bk,
        stride_bn,
        stride_cc,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------
    # 9) Restore original layout and dtype (FP32)
    # ----------------------------------------------------------
    out = out_flat.to(torch.float32)
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)   # (bs, N, N, hidden)

    # ----------------------------------------------------------
    # 10) Output LayerNorm, gate, and final linear projection
    # ----------------------------------------------------------
    out = layer_norm(out, normalized_shape=(hidden_dim,),
                     weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = linear(out, to_out_weight)   # bias=False as in the reference module

    return out
```
