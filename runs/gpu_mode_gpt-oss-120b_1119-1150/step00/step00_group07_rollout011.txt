def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # LayerNorm: compute mean/var across dim using torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # input_norm = (input_tensor - mean) / sqrt(var+eps) * norm_weight + norm_bias
    # using torch.nn.functional.layer_norm
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Proj left and right
    left_proj_weight = weights["left_proj.weight"]  # shape [hidden_dim, dim]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.einsum('b i j d, h d -> b i j h', x_norm, left_proj_weight)  # shape [B,N,N,hidden]
    right = torch.einsum('b i j d, h d -> b i j h', x_norm, right_proj_weight)
    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp
    else:
        # treat mask as ones - already fine
        pass
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.einsum('b i j d, h d -> b i j h', x_norm, left_gate_weight).sigmoid()
    right_gate = torch.einsum('b i j d, h d -> b i j h', x_norm, right_gate_weight).sigmoid()
    out_gate = torch.einsum('b i j d, h d -> b i j h', x_norm, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Now compute out = sum over k left[..., i,k,d] * right[..., j,k,d]
    # Use Triton kernel.
    out = tri_mul_outgoing_kernel(left, right, hidden_dim)  # returns shape [B,N,N,hidden]
    # to_out_norm: layernorm across hidden_dim; weight & bias config.
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_normed = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out_normed = out_normed * out_gate
    # final linear: to_out.weight shape [dim, hidden_dim]; no bias
    to_out_weight = weights["to_out.weight"]
    final = torch.einsum('b i j h, d h -> b i j d', out_normed, to_out_weight)  # shape [B,N,N,dim]
    return final
```

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)  # row tile
    pid_n = tl.program_id(1)  # col tile
    pid_b = tl.program_id(2)  # batch*hidden tile

    # decode batch and hidden dimension
    # Hidden dimension not known at compile time; use argument H
    # Compute b = pid_b // H, h = pid_b % H
    b = pid_b // H
    h = pid_b % H

    # compute offsets for start of this batch and hidden
    # base pointers
    a_base = left_ptr + b * stride_left_b + h * stride_left_h
    b_base = right_ptr + b * stride_right_b + h * stride_right_h
    c_base = out_ptr   + b * stride_out_b  + h * stride_out_h

    # compute block offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers for A and B tiles
    a_ptrs = a_base + (offs_m[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
    b_ptrs = b_base + (offs_k[:, None] * stride_right_k) + (offs_n[None, :] * stride_right_j)

    # accumulation
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # number of K iterations
    num_k = (N + BLOCK_K - 1) // BLOCK_K
    for k in range(num_k):
        # compute masks for k dimension
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        mask_k = cur_k < N

        # load A and B with masking
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # dot product
        acc += tl.dot(a, b)

        # increment pointers for next K block
        a_ptrs += BLOCK_K * stride_left_k
        b_ptrs += BLOCK_K * stride_right_k

    # store result

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operation.
    It fuses the expensive N^3 contraction (sum over k) using a Triton
    kernel while the surrounding linear projections, layer‑norms and gates
    are performed with native PyTorch for simplicity.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------- LayerNorm ----------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---------- Linear projections ----------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)

    # ---------- Optional mask ----------
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ---------- Gates ----------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # ---------- Triton contraction ----------
    out = tri_mul_outgoing_triton(left, right)  # shape [B,N,N,hidden]

    # ---------- Output norm ----------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate

    # ---------- Final linear ----------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out
```

def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):

    pid_n = tl.program_id(1)
    pid_b = tl.program_id(2)

    # decode batch and hidden indices
    b = pid_b // H
    h = pid_b % H

    # base pointers for this (b,h)
    a_base = left_ptr + b * stride_left_b + h * stride_left_h
    b_base = right_ptr + b * stride_right_b + h * stride_right_h
    c_base = out_ptr + b * stride_out_b + h * stride_out_h

    # offsets for this block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(num_k):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        mask_k = cur_k < N

        # load A tile (M x K)
        a = tl.load(a_base + (offs_m[:, None] * stride_left_i) + (cur_k[None, :] * stride_left_k),

"""
TriMul (outgoing) – Triton‑accelerated core contraction.

The TriMul operator works on a tensor x of shape [B, N, N, D] and
produces an output of the same shape.  The most expensive part is the
N³ contraction

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

which is equivalent to a batched matrix multiplication
left[..., :, h] @ right[..., :, h]ᵀ for each hidden dimension `h`.
This kernel implements that contraction with a custom Triton GEMM.
All surrounding LayerNorms, Linear projections and gates are handled
with native PyTorch for simplicity.

The function `custom_kernel` receives a tuple
(input_tensor, mask, weights, config) and returns the forward pass
result (no gradients are required).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------
# Triton kernel: batched GEMM for the N³ contraction
# ---------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid_m = tl.program_id(0)        # row tile index
    pid_n = tl.program_id(1)        # col tile index
    pid_b = tl.program_id(2)        # combined batch‑hidden index

    # Decode batch (b) and hidden (h) from pid_b
    b = pid_b // H
    h = pid_b % H

    # Base pointers for this (b, h) pair
    a_base = left_ptr + b * stride_left_b + h * stride_left_h
    b_base = right_ptr + b * stride_right_b + h * stride_right_h
    c_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_m = offs_m < N
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for k in range(num_k):
        k_offset = k * BLOCK_K
        cur_k = offs_k + k_offset
        mask_k = cur_k < N

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + (offs_m[:, None] * stride_left_i)
            + (cur_k[None, :] * stride_left_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # Load Bᵀ tile: shape (BLOCK_K, BLOCK_N)
        # Note: after transpose the row‑stride is stride_right_k and the
        # column‑stride is stride_right_j.
        b = tl.load(
            b_base
            + (cur_k[:, None] * stride_right_k)
            + (offs_n[None, :] * stride_right_j),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        acc += tl.dot(a, b)

    # Store the result into out[b,i,j,h]
    c_ptrs = c_base + (offs_m[:, None] * stride_out_i) + (offs_n[None, :] * stride_out_j)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[:, None])


def tri_mul_outgoing_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    left, right: tensors of shape [B, N, N, H] (float32, on CUDA).
    Returns out of the same shape.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Block sizes – tuned for H100 (adjustable if needed)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Strides (ints) for address calculations
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()
    # `right` uses the same layout as `left`; the j‑dimension is dim‑1,
    # the k‑dimension is dim‑2.
    stride_right_j = stride_right_i   # original second axis (j)
    stride_right_k = stride_right_j   # original third axis (k)

    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


# ---------------------------------------------------------------
# High‑level custom kernel entry point
# ---------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, D] on CUDA
            - mask        : torch.Tensor [B, N, N] (or None)
            - weights     : dict of model weights (see body)
            - config      : dict containing "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32, on CUDA)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # LayerNorm over the channel dimension `dim`
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (broadcast on hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating (sigmoid) – also bias‑free linear layers
    # -----------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Heavy N³ contraction via Triton
    # -----------------------------------------------------------------
    out = tri_mul_outgoing_triton(left, right)   # [B, N, N, hidden]

    # -----------------------------------------------------------------
    # Output LayerNorm + gating
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # Final linear projection back to `dim`
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])  # [B, N, N, dim]

    return out
