    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    EPS: tl.constexpr = 1e-5

    p_j = tl.program_id(1)   # j tile index
    pid_h = tl.program_id(2) # batch*head index

    batch_h = pid_h
    b = batch_h // H
    d = batch_h % H

    # Compute offsets for the start of this block
    i_offset = pid * BLOCK_M
    j_offset = p_j * BLOCK_N

    # Loop over k dimension
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k range
        k_offset = k * BLOCK_K

        # Load left block (M,K)
        offs_i = i_offset + tl.arange(0, BLOCK_M)
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        mask_i = offs_i < N
        mask_k = offs_k < N

        left_ptrs = left_ptr + b*stride_left_b + d*stride_left_d + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k

def layer_norm(x, weight, bias, eps=1e-5):
    # x shape: [*, dim]
    # compute mean and variance along last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) * torch.rsqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm

"""
custom_kernel – Outgoing TriMul (AlphaFold‑3) implemented with Triton

The kernel fuses the expensive N³ contraction:
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

where left/right are the linearly projected and gated pairwise
representations of shape [B, N, N, H] (H = hidden_dim).  The kernel
computes a batched GEMM for each head `d` using Tensor‑Cores (FP16
inputs, FP32 accumulator) and writes the result back as FP16.
All other linear / LayerNorm / sigmoid steps are performed with PyTorch
(easily fused by the compiler).  This yields a fast forward‑only
implementation suitable for the benchmark configurations listed in the
prompt.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the einsum contraction
# ----------------------------------------------------------------------
@triton.jit
def trmul_einsum_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # shapes
    B, N, H,
    # strides (in elements) for left = [B,N,N,H]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right = [B,N,N,H]
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for out = [B,N,N,H]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for each head (hidden dimension).

    Computes for a single (batch, head) pair:
        out[i, j] = Σₖ left[i, k] * right[k, j]
    where left has shape (N, N) and right is accessed as (k, j) to match the
    `einsum('... i k d, ... j k d -> ... i j d')` semantics.
    """

    # ------------------------------------------------------------------
    #  Program IDs
    # ------------------------------------------------------------------
    pid_m = tl.program_id(0)                # block row index
    pid_n = tl.program_id(1)                # block column index
    pid_hd = tl.program_id(2)               # combined batch*head index

    # ------------------------------------------------------------------
    #  Decode batch and head
    # ------------------------------------------------------------------
    batch_id = pid_hd // H
    head_id  = pid_hd % H

    # ------------------------------------------------------------------
    #  Offsets for this tile
    # ------------------------------------------------------------------
    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for out‑of‑bounds rows/cols (needed for the last partial tile)
    mask_i = i0 < N
    mask_j = j0 < N

    # ------------------------------------------------------------------
    #  Accumulator (FP32 for numeric stability)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #  Loop over K dimension in blocks
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k0 = kb * BLOCK_K + tl.arange(0, BLOCK_K)

        # Masks for the K dimension
        mask_k = k0 < N

        # ------------------------------------------------------------------
        #  Load left tile   shape (BLOCK_M, BLOCK_K)
        #    left[batch_id, i, k, head_id]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch_id * stride_left_b
            + head_id  * stride_left_h
            + i0[:, None] * stride_left_i
            + k0[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptrs,
            mask=(mask_i[:, None] & mask_k[None, :]),
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        #  Load right tile   shape (BLOCK_K, BLOCK_N)
        #    right[batch_id, j, k, head_id]  --> we need (k, j) order
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch_id * stride_right_b
            + head_id  * stride_right_h
            + k0[:, None] * stride_right_k   # k dimension first
            + j0[None, :] * stride_right_j   # j dimension second
        )
        right = tl.load(
            right_ptrs,
            mask=(mask_k[:, None] & mask_j[None, :]),
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        #  Block‑level matrix multiplication (Tensor‑Core)
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)

    # ------------------------------------------------------------------
    #  Write the result back (store as FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch_id * stride_out_b
        + head_id  * stride_out_h
        + i0[:, None] * stride_out_i
        + j0[None, :] * stride_out_j
    )
    tl.store(
        out_ptrs,
        acc.to(tl.float16),
        mask=(mask_i[:, None] & mask_j[None, :]),
    )


# ----------------------------------------------------------------------
# Helper:  Layer‑Norm over the last dimension using torch.nn.functional
# ----------------------------------------------------------------------
def _layer_norm(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor, eps: float = 1e-5) -> torch.Tensor:
    """
    x:    [..., C]
    weight, bias: (C,)
    """
    return F.layer_norm(x, (x.shape[-1],), weight, bias, eps)


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        input    : torch.Tensor, shape [B, N, N, dim]
        mask     : torch.Tensor or None, shape [B, N, N] (optional)
        weights  : dict of torch.Tensor containing all model parameters
        config   : dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nominal_no_mask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑Norm on the input (float32)
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (float32)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (broadcast over hidden dim)
    # ------------------------------------------------------------------
    if mask is not None and not nominal_no_mask:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – computed from the *unnormalized* x
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates
    # ------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # 7️⃣ Allocate output buffer (FP16)
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float16, device=device)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 8️⃣ Launch the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Strides (in **elements**, not bytes)
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_j, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Block sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid definition: (M‑blocks, N‑blocks, batch*head)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trmul_einsum_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_b=s_left_b,
        stride_left_i=s_left_i,
        stride_left_k=s_left_k,
        stride_left_h=s_left_h,
        stride_right_b=s_right_b,
        stride_right_j=s_right_j,
        stride_right_k=s_right_k,
        stride_right_h=s_right_h,
        stride_out_b=s_out_b,
        stride_out_i=s_out_i,
        stride_out_j=s_out_j,
        stride_out_h=s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # ------------------------------------------------------------------
    # 9️⃣ Cast contraction result back to FP32
    # ------------------------------------------------------------------
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 🔟 Post‑contraction Layer‑Norm over hidden_dim
    # ------------------------------------------------------------------
    out = _layer_norm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
    )  # still [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 1️⃣1️⃣ Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 1️⃣2️⃣ Final linear back to original dim
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
```

**Explanation of the implementation**

1. **Layer‑Norm** – performed with `torch.nn.functional.layer_norm` on the last (`dim`) axis.
2. **Linear projections** – `left_proj`, `right_proj`, and the three gates are computed with `torch.nn.functional.linear`. No bias terms are used, matching the reference model.
3. **Mask handling** – if a mask tensor is supplied and `config["nomask"]` is False, it is broadcast and applied after the projections (exactly as in the reference code).
4. **Gating** – sigmoid gates are multiplied element‑wise with the projected tensors.
5. **Half‑precision contraction** – the heavy `einsum` is replaced by a custom Triton kernel that executes a batched GEMM for each head (`hidden_dim`).  The kernel:
   * treats each `(batch, head)` pair as an independent program,
   * tiles the output with `BLOCK_M × BLOCK_N` tiles,
   * iterates over `K` in `BLOCK_K`‑sized chunks,
   * uses `tl.dot` (Tensor‑Core) with FP16 inputs and FP32 accumulation,
   * writes the result back as FP16.
6. **Post‑processing** – the result is cast back to FP32, layer‑normalized over the hidden dimension, multiplied by the `out_gate`, and finally projected back to the original channel dimension with a linear layer.

The kernel covers the asymptotically dominant `O(N³·H)` operation, while the remaining light‑weight operations stay in PyTorch.  This split satisfies the requirement of “implementing at least part of the operations in a kernel” and yields a performant forward‑only TriMul implementation suitable for the listed benchmark configurations.
