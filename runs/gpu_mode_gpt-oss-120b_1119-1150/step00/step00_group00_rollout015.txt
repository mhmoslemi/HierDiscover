1. LayerNorm over the last dimension (dim). Input shape: [B, N, N, dim]. So compute normalized x = (x - mean)/sqrt(var+eps) * weight + bias.

2. Linear projections (left_proj, right_proj) from dim to hidden_dim, both weight matrices of shape [hidden_dim, dim] (since bias=False). So left = x @ left_proj_weight.T, right = x @ right_proj_weight.T.

3. Multiply with mask (unsqueeze -1) (mask shape [B,N,N] -> [B,N,N,1]; broadcast). left = left * mask, right = right * mask.

4. Compute gating: left_gate = sigmoid(x @ left_gate_weight.T), right_gate = sigmoid(x @ right_gate_weight.T), out_gate = sigmoid(x @ out_gate_weight.T). (All with bias = False).

5. left = left * left_gate, right = right * right_gate.

6. Compute out = einsum('... i k d, ... j k d -> ... i j d', left, right). That is for each b and i,j, sum over k: left[..., i, k, d] * right[..., j, k, d], shape B,N,N,hidden_dim (i.e., dim of left/right). Note that left and right share the same hidden_dim sized last dimension.

7. then apply to_out_norm (LayerNorm) over hidden_dim, then multiply out * out_gate, then linear to_out: out = out @ to_out_weight.T (dim output). Finally returns shape B,N,N,dim.

def batched_matmul_triton(A, B):
    # A and B shape: [batch, M, K] and [batch, N, K]
    # assume A and B are fp16.
    batch, M, K = A.shape
    _, N, K2 = B.shape
    assert K2 == K
    # Output shape [batch, M, N] -> float32
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A.device)

    # Compute strides
    stride_a_batch, stride_a_m, stride_a_k = A.stride()
    stride_b_batch, stride_b_n, stride_b_k = B.stride()  # note B shape [batch,N,K]
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Choose block sizes
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (batch, (M + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

"""
TriMul custom kernel implementation (outgoing version).

This kernel implements the heavy pairwise multiplicative update:
    out_{b,i,j,d} = sum_k left_{b,i,k,d} * right_{b,j,k,d}
where left/right are linear projections of the normalized input and are gated.
The outer sum over k is performed as a batched matrix multiplication using a
custom Triton kernel (fp16 inputs, fp32 accumulation). The surrounding
layer‑norms, linear projections and gating are performed with PyTorch
operations for simplicity. The result matches the reference PyTorch
implementation of AlphaFold3's TriMul (outgoing variant).
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of A (M×K) with B (N×K) transposed -> C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B^T for a batch dimension."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load a block of A (shape BLOCK_M × BLOCK_K)
        a_ptr = (
            A_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptr,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # Load a block of B^T (shape BLOCK_K × BLOCK_N)
        # Remember B is stored as (batch, N, K); we need B[j, k] -> B^T[k, j]
        b_ptr = (
            B_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptr,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # Multiply‑accumulate
        acc += tl.dot(a, b)  # a: (Mblk, Kblk), b: (Kblk, Nblk)

    # Write back the result
    c_ptr = (
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptr,
        acc,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

# Helper that wraps the Triton kernel for a batched matmul
# ----------------------------------------------------------------------
def batched_matmul_fp16_to_fp32(A_fp16: torch.Tensor, B_fp16: torch.Tensor) -> torch.Tensor:
    """
    A_fp16: [batch, M, K]  (fp16)
    B_fp16: [batch, N, K]  (fp16)
    Returns C: [batch, M, N] (fp32)
    """
    assert A_fp16.is_contiguous()
    assert B_fp16.is_contiguous()
    batch, M, K = A_fp16.shape
    _, N, K2 = B_fp16.shape
    assert K2 == K

    # Allocate output
    C = torch.empty((batch, M, N), dtype=torch.float32, device=A_fp16.device)

    # Compute strides (in elements, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A_fp16.stride()
    stride_b_batch, stride_b_n, stride_b_k = B_fp16.stride()
    stride_c_batch, stride_c_m, stride_c_n = C.stride()

    # Choose block sizes; tuned for H100 and typical N up to 1024.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A_fp16,
        B_fp16,
        C,
        M,
        N,
        K,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,   # note: k stride comes before n stride for the kernel
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    return C

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for the
    core N³ pairwise multiplication.

    Arguments
    ---------
    data: tuple (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor of shape [B, N, N, dim]
        - mask: torch.Tensor of shape [B, N, N] (may be None if nomask=True)
        - weights: dict mapping string names to torch.Tensors (already on device)
        - config: dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------------
    # LayerNorm over the input dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Gating linear layers (no bias)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output layernorm & projection
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim], dtype float32

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (float32)
    # ------------------------------------------------------------------
    if not nomask:
        # mask may be bool or float; ensure float32 and broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid() # [B,N,N,hidden_dim]
    out_gate = F.linear(x, out_gate_weight).sigmoid()     # [B,N,N,hidden_dim]

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ multiplication using Triton (fp16 inputs, fp32 output)
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Move hidden dimension to the batch axis to obtain independent matmuls
    # left_perm: [B, hidden_dim, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left_perm.shape  # _ == N
    flat_batch = B * H

    # Reshape to [flat_batch, N, N] for the kernel
    left_mat = left_perm.view(flat_batch, N, N)
    right_mat = right_perm.view(flat_batch, N, N)

    # Perform batched matmul: out = left_mat @ right_mat^T   (shape [flat_batch, N, N])
    out_flat = batched_matmul_fp16_to_fp32(left_mat, right_mat)  # fp32

    # Reshape back to [B, N, N, hidden_dim]
    out_perm = out_flat.view(B, H, N, N)                # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # still [B,N,N,hidden_dim]

    # Broadcasted multiplication with out_gate (already float32)
    out = out * out_gate

    # Final projection back to original dim
    out = F.linear(out, to_out_weight)  # [B,N,N,dim]

    return out

"""
TriMul custom kernel (outgoing variant).

This implementation follows the PyTorch reference in AlphaFold3 but replaces the
heavy N³ pairwise multiplication

    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}

with a fused Triton kernel.  The surrounding layer‑norms, linear projections and
gating are performed with PyTorch for simplicity.  The kernel runs on fp16
inputs with fp32 accumulation, giving the same numerical result as the reference
(up to typical fp16 rounding) while achieving a large speed‑up for the large
sequence lengths used in protein models.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul of A (M×K) with B (N×K)ᵀ → C (M×N)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """C = A @ Bᵀ for a batch dimension (fp16 → fp32)."""
    batch_id = tl.program_id(0)          # which head (batch × hidden_dim)
    pid_m = tl.program_id(1)             # block row
    pid_n = tl.program_id(2)             # block column

    # Offsets within the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dim in blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K
        offs_k = k_off + tl.arange(0, BLOCK_K)

        # A block: (BLOCK_M x BLOCK_K)
        a_ptr = (
            A_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptr,
            mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
            other=0.0,
        )

        # Bᵀ block: (BLOCK_K x BLOCK_N)  (B is stored as (batch, N, K))
        b_ptr = (
            B_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(
            b_ptr,
            mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
            other=0.0,
        )

        # dot: (Mblk, Kblk) @ (Kblk, Nblk) → (Mblk, Nblk)
        acc += tl.dot(a, b)

    # Write result back
    c_ptr = (
        C_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptr,
        acc,
        mask=(offs_m[:, None] < M) & (offs_n[None, :] < N),
    )

def batched_matmul_fp16_to_fp32(A_fp16: torch.Tensor, B_fp16: torch.Tensor) -> torch.Tensor:
    """
    Batched matmul using the kernel above.

    Arguments
    ----------
    A_fp16 : torch.Tensor[batch, M, K] (half)

    assert B_fp16.is_contiguous()

    _, N, K2 = B_fp16.shape
    assert K2 == K

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (bool or float) – may be None
        weights      : dict of torch.Tensors containing all linear / LN params
        config       : dict with keys:
                       - "dim"          (int)
                       - "hidden_dim"   (int)
                       - "nomask"       (bool, optional)

    Returns
    -------
    torch.Tensor [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # fetch weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (float32)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight) # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # Ensure mask has same dtype as x and broadcast over hidden dim
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (float32)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()   # [B, N, N, hidden_dim]
    right_gate = F.linear(x, right_gate_weight).sigmoid() # [B, N, N, hidden_dim]
    out_gate = F.linear(x, out_gate_weight).sigmoid()     # [B, N, N, hidden_dim]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core N³ multiplicative update via Triton
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Move hidden dimension to batch axis so each head performs an NxN matmul
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_fp16.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    flat_batch = B * H

    # Reshape to [flat_batch, N, N] for the kernel
    left_mat = left_perm.view(flat_batch, N, N)
    right_mat = right_perm.view(flat_batch, N, N)

    # Batched matmul: out = left_mat @ right_matᵀ  →  [flat_batch, N, N] (fp32)
    out_flat = batched_matmul_fp16_to_fp32(left_mat, right_mat)

    # Reshape back to [B, N, N, hidden_dim]
    out_perm = out_flat.view(B, H, N, N)               # [B, H, N, N]
    out = out_perm.permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim] (float32)

    out = out * out_gate                         # element‑wise gate
    out = F.linear(out, to_out_weight)           # [B, N, N, dim] (float32)

    return out
