@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,   # B is transposed shape (batch, K, N) maybe earlier we passed as B_T shape?
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0) # batch id
    pid_m = tl.program_id(1) # block row
    pid_n = tl.program_id(2) # block col

    # ... compute offsets

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        cur_K = min(BLOCK_K, N - k)
        # Pointers for A block: shape (BLOCK_M, cur_K)
        a_ptrs = A_ptr + (pid_batch * stride_Ab
                          + (pid_m*BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_Am
                          + (k + tl.arange(0, cur_K))[None, :] * stride_Ak)
        a = tl.load(a_ptrs, mask=(pid_m*BLOCK_M + tl.arange(0, BLOCK_M) < N)[:, None] & (k + tl.arange(0, cur_K) < N)[None, :], other=0.0)

        # B block: shape (cur_K, BLOCK_N)
        b_ptrs = B_ptr + (pid_batch * stride_Bb
                          + (k + tl.arange(0, cur_K))[:, None] * stride_Bk
                          + (pid_n*BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_Bn)
        b = tl.load(b_ptrs, mask=(k + tl.arange(0, cur_K) < N)[:, None] & (pid_n*BLOCK_N + tl.arange(0, BLOCK_N) < N)[None, :], other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

        # increment pointer offsets for next block - but we directly compute offset based on k each iteration

    # Write output
    c_ptrs = C_ptr + (pid_batch * stride_Cb
                      + (pid_m*BLOCK_M + tl.arange(0, BLOCK_M))[:, None] * stride_Ci
                      + (pid_n*BLOCK_N + tl.arange(0, BLOCK_N))[None, :] * stride_Cj)
    tl.store(c_ptrs, acc.to(tl.float16) if accumulate in float32 else acc, mask=... )

def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, N, BLOCK_K):
        cur_K = tl.where(k + tl.arange(0, BLOCK_K) < N, 1, 0)
        # load A block
        a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + (k + tl.arange(0, BLOCK_K))[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k + tl.arange(0, BLOCK_K) < N)[None, :], other=0.0)

        # load B block
        b_ptrs = B_ptr + pid_batch * stride_Bb + (k + tl.arange(0, BLOCK_K))[:, None] * stride_Bk + offs_n[None, :] * stride_Bn

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Ci, stride_Cj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute coordinates
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Load A block (M x K)
        a_ptrs = A_ptr + pid_batch * stride_Ab + offs_m[:, None] * stride_Am + cur_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B block (K x N)
        b_ptrs = B_ptr + pid_batch * stride_Bb + cur_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Compute block multiplication
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + pid_batch * stride_Cb + offs_m[:, None] * stride_Ci + offs_n[None, :] * stride_Cj

def custom_kernel(data):
    """
    Triton‑accelerated implementation of the outgoing triangular multiplicative
    update (TriMul) used in AlphaFold3. The heavy O(N³) pairwise reduction
    ``out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`` is performed by a
    custom batched matrix‑multiply kernel; all other operations (LayerNorm,
    linear projections, sigmoid gates) are carried out with PyTorch for
    simplicity.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # LayerNorm over the last dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # Linear projections (bias=False)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply optional mask (broadcast on hidden dimension)
    if not nomask:
        mask_expanded = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_expanded
        right = right * mask_expanded

    # Gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------- Triangular multiplicative update (heavy part) ------------
    # Prepare right transposed on the two sequence dimensions: right_T[b, k, j, d]
    right_T = right.permute(0, 2, 1, 3).contiguous()

    # Fuse batch and channel dimensions to expose a pure batched matmul:
    #   left_reshape  : [B*hidden_dim, N, N] (i, k)
    #   right_reshape : [B*hidden_dim, N, N] (k, j)
    B, N, _, D = left.shape  # D == hidden_dim
    left_reshape = left.permute(0, 3, 1, 2).reshape(B * D, N, N)
    right_reshape = right_T.permute(0, 3, 1, 2).reshape(B * D, N, N)

    # Compute using Triton batched GEMM
    out_reshape = triton_batched_matmul(left_reshape, right_reshape)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, D, N, N).permute(0, 2, 3, 1)

    # ---------------------------------------------------------------------

    # Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply final gating
    out = out * out_gate

    # Final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) kernel in Triton.

The TriMul module performs a quadratic‑cubic update on a 4‑D pair representation.
The most expensive part is the reduction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which can be expressed as a batched matrix multiplication
for each (b,d) pair:

    left_reshape  : (b,d,i,k) → (B*D, N, N)   (rows = i, cols = k)
    right_reshape : (b,d,k,j) → (B*D, N, N)   (rows = k, cols = j)

    out_reshape   = left_reshape @ right_reshape   (shape B*D, N, N)

All other operations (LayerNorm, linear projections, sigmoid gates) are
implemented with PyTorch for simplicity.  The heavy matrix multiplication
is carried out by a custom Triton kernel that processes each (b,d) batch
independently and tiles the computation with explicit K‑loop accumulation.
The kernel uses FP32 accumulation (the inputs are float32) and respects
arbitrary sequence lengths N via masking of out‑of‑range indices.

The function `custom_kernel` follows the signature required by the test
harness and returns a tensor of shape [bs, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr,                # left  matrix  [B*D, N, N]
    B_ptr,                # right matrix  [B*D, N, N] (already transposed on the two seq dims)
    C_ptr,                # output matrix [B*D, N, N]
    N: tl.constexpr,      # sequence length
    stride_Ab, stride_Am, stride_Ak,   # strides of A (batch, row, col)
    stride_Bb, stride_Bk, stride_Bn,   # strides of B (batch, row, col)
    stride_Cb, stride_Ci, stride_Cj,   # strides of C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ B   where A,B,C are (batch, N, N)."""
    pid_batch = tl.program_id(0)   # batch index = b*D
    pid_m = tl.program_id(1)       # tile on M (i dimension)
    pid_n = tl.program_id(2)       # tile on N (j dimension)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the K dimension (the summed index)
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- Load A tile: (M, K) ----
        a_ptrs = (
            A_ptr
            + pid_batch * stride_Ab
            + offs_m[:, None] * stride_Am
            + cur_k[None, :] * stride_Ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---- Load B tile: (K, N) ----
        b_ptrs = (
            B_ptr
            + pid_batch * stride_Bb
            + cur_k[:, None] * stride_Bk
            + offs_n[None, :] * stride_Bn
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Accumulate the matrix product for this K‑chunk
        acc += tl.dot(a, b)

    # ---- Store the result ----
    c_ptrs = (
        C_ptr
        + pid_batch * stride_Cb
        + offs_m[:, None] * stride_Ci
        + offs_n[None, :] * stride_Cj
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def triton_batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ B where A,B are (batch, N, N).
    The kernel tiles the matrices with configurable block sizes.
    """
    assert A.shape == B.shape, "A and B must have identical shapes"
    batch, N, N2 = A.shape
    assert N == N2, "Matrices must be square"

    C = torch.empty_like(A)

    # Tunable tile sizes – work well on an H100 (adjust if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        A,
        B,
        C,
        N,
        A.stride(0),
        A.stride(1),
        A.stride(2),
        B.stride(0),
        B.stride(1),
        B.stride(2),
        C.stride(0),
        C.stride(1),
        C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul.
    data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns: Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # Unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (bias=False) -> shape [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    if not nomask:
        mask_exp = mask.unsqueeze(-1)                 # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4. Compute sigmoid gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply the gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Heavy triangular multiplicative update (batched GEMM)
    #    right_T = right transposed on the two sequence dimensions.
    right_T = right.permute(0, 2, 1, 3).contiguous()   # [B, N, N, hidden_dim] with (k, j) ordering

    B, N, _, D = left.shape   # D == hidden_dim
    # Merge batch and channel dimensions to obtain pure 3‑D tensors
    left_reshape = (
        left.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    )  # (B*D, i, k)
    right_reshape = (
        right_T.permute(0, 3, 1, 2).contiguous().view(B * D, N, N)
    )  # (B*D, k, j)

    # Perform the batched matrix multiplication with Triton
    out_reshape = triton_batched_matmul(left_reshape, right_reshape)  # (B*D, i, j)

    # Reshape back to [B, N, N, hidden_dim]
    out = out_reshape.view(B, D, N, N).permute(0, 2, 3, 1)  # [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 6. Output LayerNorm over the hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Apply the final output gate
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to the original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
