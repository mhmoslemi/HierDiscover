import triton
import triton.language as tl

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    b = pid // H
    h = pid % H

    # offsets for block start
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-range
    mask_i = i < N
    mask_j = j < N

    # Compute base pointers for left and right and out for this batch and head
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        # Offsets for K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        # mask for k range
        mask_k = k_idx < N

        # Load left tile
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # Load right tile (transposed): we need right[j,k] as rows j? Actually we need right_T shape (BLOCK_K, BLOCK_N)
        # Right base for original orientation: index (j,k) corresponds to (row=j, col=k)
        # To treat as transposed for multiplication, we will load with rows = k, cols = j:
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Perform dot product across K dimension: accumulate
        acc += tl.dot(left, right)  # left: (BLOCK_M, BLOCK_K); right: (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)

    # Write out
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) – Triton‑accelerated forward implementation.

The forward pass consists of:
1. Layer‑norm on the input pair representation (last dimension C = dim).
2. Two linear projections (left/right) followed by an optional mask.
3. Three element‑wise sigmoid gates (left, right, output) derived from the
   normalized input.
4. Core “outgoing” update:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a batched matrix multiplication of left with the transpose of right
   for every head * batch* pair.  This O(N³·H) kernel is written in Triton.
5. A second Layer‑norm on the hidden dimension, multiplication by the output
   gate and a final linear projection back to the original channel size.

All tensors are kept on the device; only the O(N³·H) contraction is
implemented in Triton, the rest uses PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # *const float* left matrix (B,N,N,H)
    right_ptr,              # *const float* right matrix (B,N,N,H)
    out_ptr,                # *float* output matrix (B,N,N,H)
    B, N, H,                # problem dimensions
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes for a single (batch, head) pair:
        out[i,j] = Σ_k left[i,k] * right[j,k]
    using a tiled GEMM (left @ rightᵀ).  The kernel is launched with a
    grid of (B*H, ceil(N/BLOCK_M), ceil(N/BLOCK_N)).
    """
    pid = tl.program_id(0)          # combined batch * head index
    pid_m = tl.program_id(1)        # block row (i)
    pid_n = tl.program_id(2)        # block column (j)

    # decode batch and head from combined pid
    b = pid // H
    h = pid % H

    # row / column coordinates for this block
    i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # range masks (to avoid out‑of‑bounds stores/loads)
    mask_i = i < N
    mask_j = j < N

    # base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_h

    # accumulator in FP32 (even if inputs are FP16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        k_idx = k_start + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ----- load left[i, k] ------------------------------------------------
        # shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i[:, None] * stride_left_i + k_idx[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float32)

        # ----- load rightᵀ[k, j]  (i.e. right[j, k]) -------------------------
        # shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + k_idx[:, None] * stride_right_k + j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float32)

        # ----- accumulate -------------------------------------------------------
        # tl.dot does a matrix‑multiply (M×K)·(K×N) → (M×N)
        acc += tl.dot(left, right)

    # ----- write result ---------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask        : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights     : dict of torch.Tensors
        - config      : dict containing `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (mask may be None or all‑ones)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure same dtype and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul – batched (batch×head) matrix multiplication
    # ------------------------------------------------------------------
    B, N, _, _ = x.shape
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=dtype, device=device)

    # strides in elements (not bytes) – PyTorch gives them directly
    s_lb, s_li, s_lk, s_lh = left.stride()
    s_rb, s_ri, s_rk, s_rh = right.stride()
    s_ob, s_oi, s_oj, s_oh = out_hidden.stride()

    # Tile sizes – chosen to fit in registers on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * hidden_dim,                         # batch * head
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
    )

    # Launch the Triton kernel
    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out_hidden,
        B, N, hidden_dim,
        s_lb, s_li, s_lk, s_lh,
        s_rb, s_ri, s_rk, s_rh,
        s_ob, s_oi, s_oj, s_oh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6) Second layer‑norm on the hidden dimension
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 7) Output gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out_scaled = out_norm * out_gate
    out = F.linear(out_scaled, weights["to_out.weight"])   # [B,N,N,dim]

    return out
