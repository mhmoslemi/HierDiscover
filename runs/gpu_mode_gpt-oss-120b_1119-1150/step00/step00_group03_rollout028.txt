"""
Trident Computational Kernel for TriMul (outgoing version).

The kernel performs a batched matrix multiplication across the
pairwise tensor dimensions:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]

This corresponds to the einsum in the PyTorch reference
'...i k d, ...j k d -> ...i j d'. The implementation fuses the
reduction over the sequence dimension k in a Triton kernel, while
the surrounding layer‑norms, linear projections and gating are
handled with PyTorch for clarity.

The kernel operates on contiguous tensors of shape [B, N, N, H]
using fp16 arithmetic (accumulating in fp32) and writes the result
in fp16, which is later cast back to fp32 for the remaining
operations. This yields a fast GPU implementation suitable for
the configuration ranges used in the tests.
"""
import torch
import triton
import triton.language as tl

# -----------------------------------------------------------------------------
# Triton kernel: batched GEMM for TriMul
# -----------------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for each (b, h):
        out[b, i, j, h] = sum_{k=0}^{N-1} left[b, i, k, h] * right[b, j, k, h]

    The kernel uses a standard tiled GEMM with fp16 inputs and fp32
    accumulation.
    """
    pid_bh = tl.program_id(0)      # combined batch and hidden dimension
    pid_i  = tl.program_id(1)      # tile row (i)
    pid_j  = tl.program_id(2)      # tile column (j)

    # Resolve batch index and hidden index from the combined dimension
    b = pid_bh // H
    h = pid_bh %  H

    # Compute the start offsets of the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Allocate accumulator in float32 (more stable than fp16)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # Loop over the reduction dimension k
    # -------------------------------------------------------------------------
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointers to the current K‑block of A (left) and B (right^T)
        # A layout: [b, i, k, h]   -> row stride = stride_l_i, col stride = stride_l_k
        a_ptrs = left_ptr + \
                 b * stride_l_b + h * stride_l_h + \
                 offs_i[:, None] * stride_l_i + \
                 offs_k[None, :] * stride_l_k

        # B layout (transposed): we want [k, j] from the original [j, k] tensor.
        b_ptrs = right_ptr + \
                 b * stride_r_b + h * stride_r_h + \
                 offs_k[:, None] * stride_r_k + \
                 offs_j[None, :] * stride_r_i

        # Load A and B (fp16) with proper masking
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Perform the matrix multiply for this K‑block and accumulate.
        # a (M, K), b (K, N) => (M, N)
        acc += tl.dot(a, b).to(tl.float32)

        # The next block simply advances the K‑pointer by BLOCK_K elements.
        # (Handled implicitly by recomputing a_ptrs / b_ptrs each iteration.)
    # -------------------------------------------------------------------------

    # Write the result (cast back to fp16 for storage)
    out_ptrs = out_ptr + \
               b * stride_o_b + h * stride_o_h + \
               offs_i[:, None] * stride_o_i + \
               offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

# -----------------------------------------------------------------------------
# Forward entry point
# -----------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing version) using a fused
    Triton GEMM for the costly inner‑product reduction.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)

    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # Extract dimensions
    # -----------------------------------------------------------------
    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # Load weights (they already live on the correct device)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # LayerNorm on the input (over the last dimension)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
    )   # shape [B, N, N, D]

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # Gating (sigmoid after a linear)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # Optional mask (broadcast over the hidden dimension)
    # -----------------------------------------------------------------
    if mask is not None:
        # Ensure mask is float and broadcastable
        mask = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask
    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Batched GEMM over the reduction dimension `k`.
    # We compute: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    # -----------------------------------------------------------------
    # Cast to fp16 for the kernel (fast tensor‑core path)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (fp16)
    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (int64) required by the Triton kernel
    stride_l = left_fp16.stride()    # (b,i,k,h)
    stride_r = right_fp16.stride()
    stride_o = out_fp16.stride()

    # Kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                               # combined batch × hidden dimension
        (N + BLOCK_M - 1) // BLOCK_M,                 # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,                 # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        stride_l[0], stride_l[1], stride_l[2], stride_l[3],
        stride_r[0], stride_r[1], stride_r[2], stride_r[3],
        stride_o[0], stride_o[1], stride_o[2], stride_o[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # a reasonable default for H100
    )

    # Cast back to fp32 for the remaining operations
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # Output layer‑norm (over hidden dimension)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # Apply the output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,D]

    return out

"""
TriMul (outgoing) forward with a fused Triton GEMM.

The heavy part of TriMul is the reduction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
which is a batched matrix multiplication (one matmul per hidden channel).
This kernel computes that reduction in FP16 using Tensor‑cores (accumulating in FP32)
and the rest of the module (LayerNorms, Linear projections and gates) is
implemented with PyTorch for clarity.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum reduction
# ----------------------------------------------------------------------
@triton.jit
def trimul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Computes, for each (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to contiguous tensors of shape
        [B, N, N, H] (FP16). The last dimension H is treated as a separate batch.
    B, N, H : sizes of the tensors.
    stride_* : per‑dimension strides of the tensors (in elements).
    BLOCK_M, BLOCK_N, BLOCK_K : tile sizes.
    """
    pid_bh = tl.program_id(0)      # combined batch * hidden index
    pid_i  = tl.program_id(1)      # tile row (i)
    pid_j  = tl.program_id(2)      # tile column (j)

    # Resolve original batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Tile start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_iter * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Pointer to left[b,i,k,h]   -> shape (M, K)
        a_ptrs = left_ptr + \
                 b * stride_l_b + h * stride_l_h + \
                 offs_i[:, None] * stride_l_i + \
                 offs_k[None, :] * stride_l_k

        # Pointer to right[b,j,k,h] but accessed as (K, N) = right^T
        b_ptrs = right_ptr + \
                 b * stride_r_b + h * stride_r_h + \
                 offs_k[:, None] * stride_r_k + \
                 offs_j[None, :] * stride_r_i

        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # MatMul for this K‑block → accumulate in FP32
        acc += tl.dot(a, b).to(tl.float32)

    # Write the result (cast back to FP16)
    out_ptrs = out_ptr + \
               b * stride_o_b + h * stride_o_h + \
               offs_i[:, None] * stride_o_i + \
               offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Forward entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask         : torch.Tensor [B, N, N] (float32 or bool) or None
        - weights      : dict of torch.Tensor containing all linear/ln weights
        - config       : dict with at least ``dim`` and ``hidden_dim`` entries

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, D = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Load model weights (already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the feature dimension D
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(D,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, D]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Gating networks (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast over the hidden dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # Apply gating
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core reduction: batched GEMM over the reduction dim k
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    # ------------------------------------------------------------------
    # Cast to FP16 to exploit Tensor‑cores (accumulate in FP32)
    left_fp16  = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    out_fp16 = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides for the Triton kernel (in elements, not bytes)
    s_l = left_fp16.stride()   # (b,i,k,h)
    s_r = right_fp16.stride()
    s_o = out_fp16.stride()

    # Tile sizes – chosen to fit well on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                     # combined batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,       # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,       # j‑tiles
    )

    trimul_gemm_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B, N, hidden_dim,
        s_l[0], s_l[1], s_l[2], s_l[3],
        s_r[0], s_r[1], s_r[2], s_r[3],
        s_o[0], s_o[1], s_o[2], s_o[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # H100 performs well with a few warps per block
    )

    # Convert back to FP32 for the remaining FP32‑only ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # 7) Apply the final output gate and linear projection back to D
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # -> [B, N, N, D]

    return out
