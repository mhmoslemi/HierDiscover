@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hd = tl.program_id(2)  # combined batch*hidden

    # Compute batch and hidden index
    num_h = H
    b = pid_hd // num_h
    d = pid_hd % num_h

    # Compute start offsets in i and j dimensions
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Compute pointers base for this batch and hidden channel
    # left base
    left_base = left_ptr + b * stride_left_b + d * stride_left_d
    # right base
    right_base = right_ptr + b * stride_right_b + d * stride_right_d
    # out base
    out_base = out_ptr + b * stride_out_b + d * stride_out_d

    # allocate accumulator

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    This is a batched matrix multiplication with the second operand transposed.
    The kernel iterates over the K dimension in blocks and accumulates the result.
    """
    pid_i = tl.program_id(0)  # tile over i
    pid_j = tl.program_id(1)  # tile over j
    pid_hd = tl.program_id(2) # combined batch * hidden

    # Resolve batch and hidden indices
    hidden = H
    b = pid_hd // hidden
    h = pid_hd % hidden

    # Compute start indices for this tile
    i_base = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_base = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute base pointer offsets for the selected batch and hidden channel
    off_b = b * stride_left_b
    off_h = h * stride_left_d
    left_base = left_ptr + off_b + off_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_d
    out_base = out_ptr + b * stride_out_b + h * stride_out_d

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + i_base[:, None] * stride_left_i + k_base[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=(i_base[:, None] < N) & (k_base[None, :] < N),
                       other=0.0).to(tl.float32)

        # Load right tile: shape (BLOCK_N, BLOCK_K) (note: we need the k dimension as the second axis)
        right_ptrs = right_base + j_base[:, None] * stride_right_i + k_base[None, :] * stride_right_k
        right = tl.load(right_ptrs,
                        mask=(j_base[:, None] < N) & (k_base[None, :] < N),
                        other=0.0).to(tl.float32)

        # Accumulate product. tl.dot with trans_b=True computes Σ_k left[i,k] * right[j,k].
        acc += tl.dot(left, right, trans_b=True)

    # Write result
    out_ptrs = out_base + i_base[:, None] * stride_out_i + j_base[None, :] * stride_out_j
    tl.store(out_ptrs,
             acc,
             mask=(i_base[:, None] < N) & (j_base[None, :] < N))

def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    Steps:
      1. Layer‑norm the input.
      2. Linear projections (left/right) and gating.
      3. Optional mask multiplication.
      4. Core triplet multiplication performed by a Triton kernel:
            out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
      5. Layer‑norm + out‑gate + final linear projection back to `dim`.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch.nn.functional as F

    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ---- 1. LayerNorm over the channel dimension ----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ---- 2. Linear projections (no bias) ----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ---- 3. Apply mask if present ----
    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ---- 4. Gating ----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---- 5. Core TriMul via Triton kernel ----
    B, N, _, _ = left.shape

    # Allocate output buffer
    out = torch.empty_like(left)   # shape (B,N,N,hidden_dim)

    # Compute strides (in elements) for pointer arithmetic
    stride_left = left.stride()      # (b,i,k,d)
    stride_right = right.stride()
    stride_out = out.stride()

    stride_left_b, stride_left_i, stride_left_k, stride_left_d = stride_left
    stride_right_b, stride_right_i, stride_right_k, stride_right_d = stride_right
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = stride_out

    # Block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Grid dimensions
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )

    trimul_matmul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_i, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # ---- 6. Post‑processing (layer‑norm, gate, final linear) ----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” operator – Triton‑accelerated core multiplication.

The heavy part of the layer is the tensor contraction

    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which can be seen as a batch of hidden‑dim independent matrix
multiplications:   left_h @ right_h.T   (size N×N).

The kernel below tiles the i‑ and j‑dimensions (BLOCK_M × BLOCK_N)
and iterates over the reduction axis k in blocks (BLOCK_K).  For every
(batch, hidden) pair we accumulate a float‑32 matrix and store the
result.  All other components (LayerNorms, linear projections,
gating, optional masking) are handled in PyTorch – only the O(N³)
contraction is offloaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_i, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """
    Compute:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Parameters
    ----------
    left_ptr, right_ptr, out_ptr : pointers to the input / output tensors
    B, N, H : batch size, sequence length, hidden dimension
    stride_* : strides (in elements) of the respective tensors
    BLOCK_M, BLOCK_N, BLOCK_K : compile‑time tile sizes
    """
    pid_i = tl.program_id(0)               # tile index over i
    pid_j = tl.program_id(1)               # tile index over j
    pid_hd = tl.program_id(2)              # combined batch * hidden

    # Resolve batch (b) and hidden channel (h) from the flattened axis
    hidden = H
    b = pid_hd // hidden
    h = pid_hd % hidden

    # Starting coordinates of the current tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Base pointers for this (b, h) slice
    left_base  = left_ptr  + b * stride_left_b  + h * stride_left_d
    right_base = right_ptr + b * stride_right_b + h * stride_right_d
    out_base   = out_ptr   + b * stride_out_b   + h * stride_out_d

    # Accumulator for the block
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks of size BLOCK_K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        kk = k * BLOCK_K + tl.arange(0, BLOCK_K)               # k‑indices for this block

        # ---- load left[i, k] -------------------------------------------------
        left_ptrs = left_base + i[:, None] * stride_left_i + kk[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=(i[:, None] < N) & (kk[None, :] < N),
            other=0.0
        ).to(tl.float32)                                     # (BLOCK_M, BLOCK_K)

        # ---- load right[j, k] ------------------------------------------------
        # Right operand is NOT transposed in memory; we ask tl.dot to
        # treat it as transposed via trans_b=True.
        right_ptrs = right_base + j[:, None] * stride_right_i + kk[None, :] * stride_right_k
        right = tl.load(
            right_ptrs,
            mask=(j[:, None] < N) & (kk[None, :] < N),
            other=0.0
        ).to(tl.float32)                                     # (BLOCK_N, BLOCK_K)

        # ---- accumulate ------------------------------------------------------
        # tl.dot(A, B, trans_b=True) computes Σ_k A[i,k] * B[j,k]
        acc += tl.dot(left, right, trans_b=True)               # (BLOCK_M, BLOCK_N)

    # ---- write back -----------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(
        out_ptrs,
        acc,
        mask=(i[:, None] < N) & (j[None, :] < N)
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module with a Triton core.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask          : torch.Tensor, shape [B, N, N]  (may be ignored)
        - weights       : dict of torch.Tensors containing all model parameters
        - config        : dict with at least keys ``dim`` and ``hidden_dim`` and
                         optional ``nomask`` (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # -------------------------------------------------------------------------
    # Unpack inputs
    # -------------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension
    # -------------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )   # [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (bias‑free)
    # -------------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the last dim)
    # -------------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(dtype)        # [B,N,N,1]
        left  = left  * mask_exp
        right = right * mask_exp

    # -------------------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. Core TriMul contraction via Triton
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Output buffer for the intermediate hidden representation
    out_hidden = torch.empty_like(left)   # shape [B, N, N, hidden_dim]

    # Strides (in elements) for pointer arithmetic
    stride_l = left.stride()   # (b,i,k,d)
    stride_r = right.stride()
    stride_o = out_hidden.stride()

    (s_l_b, s_l_i, s_l_k, s_l_d) = stride_l
    (s_r_b, s_r_i, s_r_k, s_r_d) = stride_r
    (s_o_b, s_o_i, s_o_j, s_o_d) = stride_o

    # Tile sizes – empirically good for H100 (shared‑memory friendly)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_matmul_kernel[grid](
        left, right, out_hidden,
        B, N, hidden_dim,
        s_l_b, s_l_i, s_l_k, s_l_d,
        s_r_b, s_r_i, s_r_k, s_r_d,
        s_o_b, s_o_i, s_o_j, s_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 6. Post‑processing: layer‑norm, gate, final linear back to `dim`
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # shape [B, N, N, dim]

    return out
