Pseudo:

```
def trimul_outgoing_kernel(...):
    B, N, D = ... const ints
    K = N
    BLOCK_K = 64

    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # ensure within bounds
    if i >= N or j >= N:
        return

    # allocate accumulate vector for each d
    acc = tl.zeros((D,), dtype=tl.float16)

    # loop over k in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)  # shape (BLOCK_K)
        # mask out-of-range
        mask_k = k_off < N

        # left offset: b*stride_left_b + i*stride_left_i + k_off*stride_left_k + d*stride_left_d
        # we need to load vector [BLOCK_K, D].
        # Use tl.load with pointer shape (BLOCK_K, D)
        left_ptrs = left_ptr + b*stride_left_b + i*stride_left_i + k_off[:, None] * stride_left_k + tl.arange(0, D)[None, :] * stride_left_d
        left_val = tl.load(left_ptrs, mask=mask_k[:, None] & (tl.arange(0, D)[None,:] < D), other=0.0).to(tl.float16)

        # right offset: b*stride_right_b + j*stride_right_j + k_off*stride_right_k + d*stride_right_d
        right_ptrs = right_ptr + b*stride_right_b + j*stride_right_j + k_off[:, None] * stride_right_k + tl.arange(0, D)[None, :] * stride_right_d
        right_val = tl.load(right_ptrs, mask=mask_k[:, None] & (tl.arange(0, D)[None,:] < D), other=0.0).to(tl.float16)

        # multiply elementwise and sum across k dimension:
        # For each d, we need sum over k of left_val[k,d] * right_val[k,d].
        # We can compute elementwise multiplication and sum across k using tl.sum.
        prod = left_val * right_val  # shape (BLOCK_K, D)
        # sum across k
        acc += tl.sum(prod, axis=0)  # shape (D,)
    # Now write acc to out.
    out_ptrs = out_ptr + b*stride_out_b + i*stride_out_i + j*stride_out_j + tl.arange(0, D) * stride_out_d
    tl.store(out_ptrs, acc.to(tl.float32))  # store as float32?
```

    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: pid_m, pid_n, pid_b
    pid_m = tl.program_id(0) # row block
    pid_n = tl.program_id(1) # col block
    pid_b = tl.program_id(2) # batch block
    
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)
    
    # compute masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N
    
    # compute base pointers for a, b, c
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0).to(tl.float32)
        # dot
        acc += tl.dot(a, b)
        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    
    # store result
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) kernel implementation with Triton.

Algorithm
---------
1. Layer‑normalize the input tensor `x`  (shape [B,N,N,C]).
2. Project to a hidden dimension `H` with two independent linear layers:
       left  = x @ left_proj_weight
       right = x @ right_proj_weight
3. (Optional) Apply the pairwise mask.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) and
   multiply them into `left`, `right` and the final output respectively.
5. The core operation is
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix‑multiply for every hidden channel `h`.
   By reshaping the tensors we obtain a batch dimension `B*H` and run a
   custom Triton GEMM kernel (`batched_matmul_kernel`) that computes
   `left_flat @ right_flatᵀ` for each channel in parallel, using FP16 for
   the loads and FP32 accumulation.
6. Reshape the GEMM result back to [B,N,N,H], apply a second LayerNorm,
   multiply by `out_gate` and a final linear projection to the original
   dimension `C`.
7. Return the float‑32 output tensor.

The Triton kernel is deliberately written as a generic batched GEMM and
re‑used for the TriMul contraction.  All heavy arithmetic (the N³
contraction) is performed on the GPU via tensor‑core‑compatible
operations, while the surrounding linear‑layer and LayerNorm logic is
handled with standard PyTorch calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Batched GEMM kernel (FP16 inputs, FP32 accumulation)
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,                     # problem size
    stride_ab, stride_am, stride_ak,    # a: (batch, M, K)
    stride_bb, stride_bk, stride_bn,    # b: (batch, K, N)
    stride_cb, stride_cm, stride_cn,    # c: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute C = A @ Bᵀ for each batch entry (A: M×K, B: N×K)."""
    pid_m = tl.program_id(0)   # row block
    pid_n = tl.program_id(1)   # col block
    pid_b = tl.program_id(2)   # batch block (0 … batch‑1)

    # ---- compute offsets within the block --------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out‑of‑bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_ptrs = a_ptr + pid_b * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_b * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- loop over K ------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_blocks):
        # mask for the current K tile
        cur_k = k_iter * BLOCK_K + offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)               # FP16 load
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)               # FP16 load

        # dot product (FP16 inputs, FP32 accumulation)
        acc += tl.dot(a, b, out_dtype=tl.float32)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ---- write back --------------------------------------------------------
    c_ptrs = c_ptr + pid_b * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Custom TriMul forward entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            input   – torch.Tensor of shape [B, N, N, C]  (float32)
            mask    – torch.Tensor of shape [B, N, N] (bool/float) or None
            weights – dict with all linear‑layer weights and LayerNorm params
            config  – dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    (x, mask, weights, config) = data
    device = x.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B,N,N,dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Apply mask if present (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask is [B,N,N]; expand to [B,N,N,1] and multiply
        mask_exp = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates to the projected tensors
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 7. Reshape for batched GEMM:
    #    left_flat  : (B*H, N, N)   where each batch entry is one hidden channel
    #    right_flat : (B*H, N, N)   (already transposed on the last two dims)
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # left: (B,N,N,H) -> (B,H,N,N) -> (B*H,N,N)
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    # right: (B,N,N,H) -> (B,H,N,N) then transpose the N,N dims -> (K,J)
    right_flat = right.permute(0, 3, 2, 1).contiguous().view(B * hidden_dim, N, N)

    # ------------------------------------------------------------------
    # 8. Allocate output buffer (FP32 accumulation)
    # ------------------------------------------------------------------
    out_flat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 9. Launch the custom batched GEMM kernel
    # ------------------------------------------------------------------
    # block sizes – tuned for H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B * hidden_dim,          # batch
        N, N, N,                # M, N, K
        left_flat.stride(0),    # stride_ab (batch stride)
        left_flat.stride(1),    # stride_am (row stride)
        left_flat.stride(2),    # stride_ak (col stride)
        right_flat.stride(0),   # stride_bb
        right_flat.stride(1),   # stride_bk
        right_flat.stride(2),   # stride_bn
        out_flat.stride(0),     # stride_cb
        out_flat.stride(1),     # stride_cm
        out_flat.stride(2),     # stride_cn
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 10. Reshape back to [B,N,N,hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 11. Final LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
