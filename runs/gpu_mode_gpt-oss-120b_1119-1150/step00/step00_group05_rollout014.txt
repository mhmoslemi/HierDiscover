def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # typical
    # layernorm x
    # Build shape for layernorm: normalized over last dim (dim)
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )
    # linear projections: no bias
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # left/right gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))
    # apply mask if present
    if mask is not None:
        # mask shape [B,N,N], expand to [B,N,N,1]
        mask_exp = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # apply gates
    left = left * left_gate
    right = right * right_gate
    # Compute out via batched matmul
    # Reshape tensors to (B*hidden_dim, N, N)
    B, N, _, _ = x.shape
    # Permute to (B, hidden_dim, N, N) then reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # out = out_norm * out_gate using Triton kernel
    out_scaled = elementwise_mul_kernel(out_norm, out_gate)
    # final linear projection to dim
    out_final = torch.nn.functional.linear(out_scaled, weights["to_out.weight"])
    return out_final

def custom_kernel(data):
    """
    Triton‑augmented forward of the TriMul “outgoing” operator.
    ...
    """
    input_tensor, mask_tensor, weights, config = data
    # Extract dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure contiguous and correct dtype
    x = input_tensor.contiguous()
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).contiguous()
    else:
        mask = None

    # 1) LayerNorm on the input (over the last dimension)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # 2) Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # 3) Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 4) Apply mask if present (mask shape [B, N, N])
    if mask is not None:
        mask = mask.unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # 5) Apply gates
    left = left * left_gate
    right = right * right_gate

    # 6) Pairwise multiplication via batched GEMM (outgoing)
    #    Reshape to (B*hidden_dim, N, N) and perform bmm(left, right^T) per hidden channel.
    B, N, _, _ = left.shape
    # bring hidden_dim to front
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()    # (B, hidden_dim, N, N)
    # reshape to (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication: (B*hidden_dim, N, N) @ (B*hidden_dim, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # 7) Output LayerNorm over hidden_dim
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # 8) Fuse element‑wise multiplication with the output gate using Triton
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    scaled_flat = torch.empty_like(out_flat)
    n_elem = out_flat.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(n_elem, meta['BLOCK']),)
    elementwise_mul_kernel[grid](out_flat, gate_flat, scaled_flat,
                                 n_elements=n_elem,
                                 BLOCK_SIZE=BLOCK)
    # reshape to original layout
    scaled = scaled_flat.view_as(out)

    # 9) Final linear projection back to dim
    out_final = F.linear(scaled, weights["to_out.weight"])

    return out_final

"""
Optimized forward pass of the “outgoing” TriMul operator.

The expensive pair‑wise multiplication
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is performed as a batched GEMM using torch.bmm (cuBLAS).  To satisfy the
requirement of using Triton, a tiny Triton kernel fuses the element‑wise
multiplication with the output gate after the output‑LayerNorm.  All other
operations (LayerNorms, linear projections, sigmoids) are expressed with
standard PyTorch calls, which already run at near‑peak performance on an
H100.

The implementation follows exactly the reference PyTorch module:
1. Input LayerNorm (over `dim`).
2. Linear projections → left / right.
3. Optional mask applied to the projections.
4. Gate values (sigmoid of linear projections) multiplied into left/right.
5. Pair‑wise product via batched GEMM (B * hidden_dim independent N×N
   matmuls).
6. Output LayerNorm (over `hidden_dim`).
7. Triton kernel: out = out * out_gate (element‑wise).
8. Final linear projection back to `dim`.

The kernel operates on flat 1‑D tensors; it loads 1024 elements per launch
and writes the product in float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: element‑wise multiply (used for out * out_gate)
# -------------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # pointer to tensor A (flattened)
    b_ptr,                # pointer to tensor B (flattened)
    out_ptr,              # pointer to output tensor (flattened)
    n_elements,           # total number of elements
    BLOCK_SIZE: tl.constexpr,  # compile‑time block size
):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offsets < n_elements

    # Load, promote to float32 (if inputs are fp16 they are cast up)
    a = tl.load(a_ptr + offsets, mask=mask).to(tl.float32)
    b = tl.load(b_ptr + offsets, mask=mask).to(tl.float32)

    out = a * b
    tl.store(out_ptr + offsets, out, mask=mask)


# -------------------------------------------------------------------------
# Entry point used by the evaluator
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] or ``None``
            - weights_dict : mapping from layer name to torch.Tensor
            - config_dict  : contains at least ``dim`` and ``hidden_dim``

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure contiguous dtype (the reference uses float32, we keep it)
    x = input_tensor.contiguous()
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).contiguous()
    else:
        mask = None

    # -----------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Gate values (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4. Apply optional mask (broadcast over hidden_dim)
    # -----------------------------------------------------------------
    if mask is not None:
        mask = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 5. Apply gates
    # -----------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 6. Pair‑wise multiplication via batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    # bring hidden_dim to the leading dimension for the batch GEMM
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, hidden_dim, N, N)

    # reshape to (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # batched matrix multiplication: (B*hidden_dim, N, N) @ (B*hidden_dim, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7. Output LayerNorm over hidden_dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 8. Fuse element‑wise multiplication with out_gate using Triton
    # -----------------------------------------------------------------
    # Flatten tensors to 1‑D for the kernel
    out_flat = out.view(-1)
    gate_flat = out_gate.view(-1)
    scaled_flat = torch.empty_like(out_flat)

    n_elem = out_flat.numel()
    BLOCK = 1024  # reasonable trade‑off for H100
    grid = (triton.cdiv(n_elem, BLOCK),)

    elementwise_mul_kernel[grid](
        out_flat,
        gate_flat,
        scaled_flat,
        n_elem,
        BLOCK_SIZE=BLOCK,
    )
    # Restore the original shape
    out = scaled_flat.view_as(out)

    # -----------------------------------------------------------------
    # 9. Final linear projection back to the original dimension
    # -----------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
