def tri_mul_triton(left, right, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    B, N, N2, H = left.shape
    assert N == N2
    out = torch.empty_like(left)  # shape [B,N,N,H]
    # compute strides in elements (int64)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()
    # grid: total number of program ids
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = B * H * num_tiles_i * num_tiles_j
    grid = (total_blocks,)
    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)
    # compute mapping
    total_h_tiles = H
    # each program processes (b,d,i_tile,j_tile)
    pid2 = pid
    batch_id = pid2 // (total_h_tiles * num_tiles_i * num_tiles_j)
    pid2 = pid2 % (total_h_tiles * num_tiles_i * num_tiles_j)
    d = pid2 // (num_tiles_i * num_tiles_j)
    pid2 = pid2 % (num_tiles_i * num_tiles_j)
    i_tile = pid2 // num_tiles_j
    j_tile = pid2 % num_tiles_j

    # compute base offsets
    batch_offset_l = batch_id * stride_l_b
    batch_offset_r = batch_id * stride_r_b
    batch_offset_o = batch_id * stride_o_b

    channel_offset_l = d * stride_l_h
    channel_offset_r = d * stride_r_h
    channel_offset_o = d * stride_o_h

    # compute start indices
    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # accumulate in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K

        # compute offsets for left (M,K)
        offs_i = tl.arange(0, BLOCK_M)
        offs_k = tl.arange(0, BLOCK_K)

        i_idx = i_base + offs_i
        k_idx = k_base + offs_k

        # mask for bounds
        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_ik = mask_i[:, None] & mask_k[None, :]

        # pointers
        left_ptrs = left_ptr + batch_offset_l + channel_offset_l + i_idx[:, None] * stride_l_i + k_idx[None, :] * stride_l_k

        left = tl.load(left_ptrs, mask=mask_ik, other=0.0, dtype=tl.float16)

        # compute offsets for right (K,N) (actually shape (K,N))
        offs_j = tl.arange(0, BLOCK_N)
        j_idx = j_base + offs_j
        # mask for j
        mask_j = j_idx < N
        mask_kj = mask_k[:, None] & mask_j[None, :]

        right_ptrs = right_ptr + batch_offset_r + channel_offset_r + k_idx[:, None] * stride_r_k + j_idx[None, :] * stride_r_j

        right = tl.load(right_ptrs, mask=mask_kj, other=0.0, dtype=tl.float16)

        # dot: left (M,K) dot right (K,N)
        acc += tl.dot(left, right)

    # store result
    # Mask for output i,j
    mask_i_out = (i_base + tl.arange(0, BLOCK_M)) < N
    mask_j_out = (j_base + tl.arange(0, BLOCK_N)) < N
    mask_ij = mask_i_out[:, None] & mask_j_out[None, :]

    out_ptrs = out_ptr + batch_offset_o + channel_offset_o + (i_base + tl.arange(0, BLOCK_M))[:, None] * stride_o_i + (j_base + tl.arange(0, BLOCK_N))[None, :] * stride_o_j

    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # bias=False
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Apply mask if present
    mask_unsq = mask[..., None]  # shape [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Multiply using Triton
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)
    # ensure contiguous for performance
    left_fp16 = left_fp16.contiguous()
    right_fp16 = right_fp16.contiguous()
    out_fp16 = tri_mul_triton(left_fp16, right_fp16, BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)
    out = out_fp16.to(torch.float32)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    # final linear
    to_out_weight = weights["to_out.weight"]

"""
TriMul (outgoing) – Triton‑accelerated implementation
====================================================

The module operates on a pairwise tensor ``x`` of shape
``[B, N, N, C]`` (B – batch, N – sequence length, C – channel).

The forward pass consists of:

1. **LayerNorm** over the last channel dimension (C = dim).
2. Two linear projections producing ``left`` and ``right`` tensors of shape
   ``[B, N, N, Hid]`` (``Hid = hidden_dim``).
3. Element‑wise masking and gating (sigmoid) applied to ``left`` and ``right``.
4. **Core TriMul operation** – a batched matrix multiplication for every
   hidden channel:

   ``out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]``

   This is exactly ``out = left @ rightᵀ`` for each hidden slice.
   The expensive ∑ₖ is implemented in a custom Triton kernel that
   tiles the ``i`` and ``j`` dimensions (BLOCK_M × BLOCK_N) and iterates
   over the ``k`` dimension in blocks (BLOCK_K).  All arithmetic is
   performed in FP16 (for bandwidth) with FP32 accumulation.

5. LayerNorm on the hidden dimension, an output‑gate (sigmoid) and a final
   linear projection back to the original dimension.

Only the massive “einsum” (step 4) is executed on the GPU via Triton –
the surrounding linear layers, normalisations, and gating are done with
standard PyTorch calls (which are already highly optimised).

The kernel works for any batch size (1 or 2), any sequence length (≤ 1024)
and any hidden dimension (e.g. 128 or 384).  It automatically masks out‑of‑bounds
indices, so the same kernel works for all test configurations.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the core TriMul operation
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                # const float16 *
    right_ptr,               # const float16 *
    out_ptr,                 # float16 *
    B, N, H,                 # int32 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # int64 strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # int64 strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # int64 strides for output
    BLOCK_M: tl.constexpr,   # tile size in i direction
    BLOCK_N: tl.constexpr,   # tile size in j direction
    BLOCK_K: tl.constexpr,   # tile size in k direction
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    for a single hidden channel ``h`` per kernel launch.
    """

    pid = tl.program_id(axis=0)

    # Number of tiles in the i / j dimensions
    num_tiles_i = tl.cdiv(N, BLOCK_M)
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    # ------------------------------------------------------------------
    # De‑compose the 1‑D program id into (batch, hidden, i‑tile, j‑tile)
    # ------------------------------------------------------------------
    batch = pid // (H * num_tiles_i * num_tiles_j)
    pid_batch = pid % (H * num_tiles_i * num_tiles_j)

    hidden = pid_batch // (num_tiles_i * num_tiles_j)
    pid_hidden = pid_batch % (num_tiles_i * num_tiles_j)

    i_tile = pid_hidden // num_tiles_j
    j_tile = pid_hidden % num_tiles_j

    # ------------------------------------------------------------------
    # Base indices for this tile
    # ------------------------------------------------------------------
    i_base = i_tile * BLOCK_M
    j_base = j_tile * BLOCK_N

    # Strides for the current batch and hidden channel
    batch_off_l = batch * stride_l_b
    batch_off_r = batch * stride_r_b
    batch_off_o = batch * stride_o_b

    chan_off_l = hidden * stride_l_h
    chan_off_r = hidden * stride_r_h
    chan_off_o = hidden * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction axis K in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_tile * BLOCK_K

        # ----- left: shape (BLOCK_M, BLOCK_K) -----
        offs_i = tl.arange(0, BLOCK_M)
        offs_k = tl.arange(0, BLOCK_K)

        i_idx = i_base + offs_i
        k_idx = k_base + offs_k

        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_ik = mask_i[:, None] & mask_k[None, :]

        left_ptrs = (
            left_ptr
            + batch_off_l
            + chan_off_l
            + i_idx[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_k
        )
        left = tl.load(left_ptrs, mask=mask_ik, other=0.0, dtype=tl.float16)

        # ----- right: shape (BLOCK_K, BLOCK_N) -----
        offs_j = tl.arange(0, BLOCK_N)
        j_idx = j_base + offs_j

        mask_j = j_idx < N
        mask_kj = mask_k[:, None] & mask_j[None, :]

        right_ptrs = (
            right_ptr
            + batch_off_r
            + chan_off_r
            + k_idx[:, None] * stride_r_k
            + j_idx[None, :] * stride_r_j
        )
        right = tl.load(right_ptrs, mask=mask_kj, other=0.0, dtype=tl.float16)

        # Accumulate the dot product over the K dimension
        acc += tl.dot(left, right)          # (M,K) • (K,N) → (M,N)

    # ------------------------------------------------------------------
    # Write the result back (store as FP16)
    # ------------------------------------------------------------------
    offs_i = tl.arange(0, BLOCK_M)
    offs_j = tl.arange(0, BLOCK_N)

    i_idx = i_base + offs_i
    j_idx = j_base + offs_j

    mask_i_out = i_idx < N
    mask_j_out = j_idx < N
    mask_ij = mask_i_out[:, None] & mask_j_out[None, :]

    out_ptrs = (
        out_ptr
        + batch_off_o
        + chan_off_o
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j
    )
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_ij)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor,
                   BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Execute the TriMul kernel on FP16 tensors ``left`` and ``right``.
    Both inputs must be contiguous and have shape [B, N, N, H].
    Returns a FP16 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2, "pairwise tensors must be square"

    # Output tensor (FP16)
    out = torch.empty_like(left)

    # Strides are expressed in *elements* (not bytes) – perfect for Triton.
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Number of tiles for the i/j dimensions
    num_tiles_i = (N + BLOCK_M - 1) // BLOCK_M
    num_tiles_j = (N + BLOCK_N - 1) // BLOCK_N

    # One program (block) per (batch, hidden, i_tile, j_tile)
    total_blocks = B * H * num_tiles_i * num_tiles_j
    grid = (total_blocks,)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    return out

# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments:
        data = (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    input_tensor, mask, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & hyper‑parameters
    # ------------------------------------------------------------------
    dim = config["dim"]                # C  (input channel dimension)
    hidden_dim = config["hidden_dim"]  # H  (projected hidden dimension)
    eps = 1e-5                         # LayerNorm epsilon (default)

    # ------------------------------------------------------------------
    # Load all weight tensors (they are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # (H, dim) @ (B,N,N,dim)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Apply the pairwise mask (if any)
    #    mask shape: [B, N, N]; we broadcast over the hidden dimension.
    # ------------------------------------------------------------------
    mask_unsq = mask[..., None].to(left.dtype)   # [B,N,N,1]
    left = left * mask_unsq
    right = right * mask_unsq

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) → element‑wise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul (batched matmul over the k‑dimension) – Triton
    # ------------------------------------------------------------------
    # Convert to FP16 for the kernel (memory‑bandwidth friendly)
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    # Triton kernel output (still FP16)
    out_fp16 = tri_mul_triton(left_fp16, right_fp16,
                              BLOCK_M=64, BLOCK_N=64, BLOCK_K=32)

    # Back to FP32 for the remaining FP32 ops
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Output gating
    # ------------------------------------------------------------------
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8) Final linear projection back to ``dim``
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, to_out_weight)   # (dim, hidden_dim) @ ...

    return out
