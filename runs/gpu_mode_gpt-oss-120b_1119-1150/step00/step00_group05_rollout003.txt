                        stride_bb, stride_bi, stride_bk,   # B batch, j, k? We'll use B to index as (batch, k, j)
                        stride_cb, stride_ci, stride_cj,

@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    # compute batch, tile indices
    total_m = tl.cdiv(M, BLOCK_M)
    total_n = tl.cdiv(N, BLOCK_N)
    batch = (pid // (total_m * total_n))
    pid_tile = pid % (total_m * total_n)
    pid_m = pid_tile // total_n
    pid_n = pid_tile % total_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = offs_k < K

        # load A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = A_ptr + batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B as transposed: B_T shape (BLOCK_K, BLOCK_N)

        b_ptrs = B_ptr + batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

        # compute dot
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # write out
    c_ptrs = C_ptr + batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) forward pass implemented with Triton‑accelerated batched GEMM.

Algorithm
---------
1. Layer‑norm the input tensor (last dimension = dim) using the provided
   `norm.weight` / `norm.bias`.
2. Project the normalized tensor to the hidden dimension (`hidden_dim`) with
   three linear layers (`left_proj`, `right_proj`, gates).
3. Apply the optional mask (broadcasted on the hidden dim).
4. Multiply the projections by their respective sigmoid gates.
5. For each hidden channel `h` compute the pairwise update:
       out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h]
   This is a batched matrix multiplication of shape
   (batch * hidden_dim, N, N) @ (batch * hidden_dim, N, N).T and is
   performed by a tiled Triton kernel (`batched_gemm_kernel`).
6. Reshape the result back to (B, N, N, hidden_dim) and apply a second
   Layer‑norm (`to_out_norm`) followed by the output gate.
7. Final linear projection back to `dim` (`to_out`).

The heavy O(N³·hidden_dim) work is done in the Triton kernel, while
the remaining tensor algebra is expressed with PyTorch for brevity.
All computations are performed in float32 for numerical stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A : (batch, M, K)    – here M = N, K = N
#   B : (batch, N, K)    – we read it as Bᵀ to obtain (batch, K, N)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr,                    # *Pointer* to A
    B_ptr,                    # *Pointer* to B (will be accessed transposed)
    C_ptr,                    # *Pointer* to output C
    M, N, K,                  # Matrix dimensions (all int32)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,    # Tile size in M dimension
    BLOCK_N: tl.constexpr,    # Tile size in N dimension
    BLOCK_K: tl.constexpr,    # Tile size in K dimension
):
    pid = tl.program_id(0)

    # --------------------------------------------------------------
    # 1) Decode program id into batch, tile‑row and tile‑col indices
    # --------------------------------------------------------------
    num_tiles_m = tl.cdiv(M, BLOCK_M)
    num_tiles_n = tl.cdiv(N, BLOCK_N)

    batch = pid // (num_tiles_m * num_tiles_n)                     # which batch
    tile_id = pid % (num_tiles_m * num_tiles_n)                    # tile within batch
    tile_m = tile_id // num_tiles_n                                # tile row
    tile_n = tile_id % num_tiles_n                                 # tile col

    # --------------------------------------------------------------
    # 2) Compute the absolute offsets for the current tile
    # --------------------------------------------------------------
    offs_m = tile_m * BLOCK_M + tl.arange(0, BLOCK_M)              # [BLOCK_M]
    offs_n = tile_n * BLOCK_N + tl.arange(0, BLOCK_N)              # [BLOCK_N]

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # 3) Loop over the K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A block: (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B block, read as transposed => (BLOCK_K, BLOCK_N)
        #   B_ptr layout is (batch, K, N) with stride_b_k for K and stride_b_n for N
        b_ptrs = (
            B_ptr
            + batch * stride_b_batch
            + offs_k[:, None] * stride_b_k   # K dimension (original i)
            + offs_n[None, :] * stride_b_n   # N dimension (original j)
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # FP32 dot product accumulation
        acc += tl.dot(a, b)   # a,b are FP32, result FP32

    # --------------------------------------------------------------
    # 4) Write back the tile
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module, using a Triton kernel for the
    cubic O(N³·hidden) pairwise update.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be all‑ones)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing "dim", "hidden_dim", "nomask"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Extract model parameters
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # ensure mask is float and has trailing dim for broadcasting
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cubic pairwise update using Triton batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # B = batch size
    # Rearrange to (batch, hidden, N, N) for contiguous slices per hidden channel
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    batch_h = B * hidden_dim                         # total number of independent GEMMs

    # Flatten the first two dimensions so that each hidden channel of each batch
    # becomes a separate “batch” for the kernel
    A = left_perm.view(batch_h, N, N)   # shape (batch_h, M=N, K=N)
    B_mat = right_perm.view(batch_h, N, N)  # shape (batch_h, K=N, N) – we will read transposed

    # Allocate output
    C = torch.empty((batch_h, N, N), dtype=torch.float32, device=device)

    # Strides (in element counts, not bytes)
    stride_a_batch, stride_a_m, stride_a_k = A.stride()          # (N*N, N, 1)
    stride_b_batch, stride_b_k, stride_b_n = B_mat.stride()     # (N*N, N, 1) – k uses original i stride
    stride_c_batch, stride_c_m, stride_c_n = C.stride()          # (N*N, N, 1)

    # Triton block sizes – tuned for H100 (64 works well, fall back to 32 for very small N)
    BLOCK_M = 64 if N >= 256 else 32
    BLOCK_N = 64 if N >= 256 else 32
    BLOCK_K = 64 if N >= 256 else 32

    # Compute launch grid
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * grid_m * grid_n

    # Launch kernel
    batched_gemm_kernel[(total_blocks,)](
        A_ptr=A.data_ptr(),
        B_ptr=B_mat.data_ptr(),
        C_ptr=C.data_ptr(),
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,      # this is the stride for the K dimension (original i)
        stride_b_n=stride_b_n,      # stride for the N dimension (original j)
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, hidden)
    out_hidden = C.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6) To‑out LayerNorm + output gate
    # ------------------------------------------------------------------
    out_norm = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_gated = out_norm * out_gate  # out_gate already float32

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original dim
    # ------------------------------------------------------------------
    output = F.linear(out_gated, to_out_weight)  # [B, N, N, dim]

    return output
```
