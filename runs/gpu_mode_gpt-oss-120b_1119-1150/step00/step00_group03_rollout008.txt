def custom_kernel(data):
    """
    TriMul (outgoing) forward pass.
    Implements the same computation as the reference PyTorch TriMul module,
    fusing mask, gate and projection element‑wise multiplication with a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # Mask handling
    if mask is None:
        # If no mask supplied, treat it as all‑ones.
        mask_tensor = torch.ones(x.shape[:3], dtype=dtype, device=device)
    else:
        # Ensure mask is float (0/1) and broadcastable
        mask_tensor = mask.to(dtype)
    # mask_tensor shape [B,N,N]; broadcast in kernel

    # ------------------------------------------------------------------
    # Fuse mask and gates with the projections using Triton
    left = fused_mul3(left_proj, mask_tensor, left_gate)
    right = fused_mul3(right_proj, mask_tensor, right_gate)

    # ------------------------------------------------------------------
    # Core multiplicative update: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    # Perform as a batched matrix multiplication over the k dimension.
    # Reshape to (B*hidden_dim, N, N) and use torch.bmm.
    B, N, _, _ = left.shape
    # Permute to B, hidden_dim, N, N
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Collapse batch and channel dimensions
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)
    # Batched matmul: left * right^T   (sum over k)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    # Restore shape [B, hidden_dim, N, N] -> [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output layer norm (hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Gate after norm
    out = out * out_gate

    # Final linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

"""
custom_kernel: Triton‑accelerated outgoing TriMul (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,dim]).
2. Linear projections (no bias) → `left_proj` / `right_proj`   (dim → hidden_dim).
3. Compute three sigmoid‑gates from the same normalized input:
   `left_gate`, `right_gate`, `out_gate` (dim → hidden_dim).
4. Fuse element‑wise multiplication of projection, mask and gate
   with a small Triton kernel:
        left  = left_proj  * mask * left_gate
        right = right_proj * mask * right_gate
   (`mask` has shape [B,N,N]; it is broadcast over the hidden dimension).
5. Core multiplicative update:
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   Per‑channel this is a matrix‑multiply: for each d,
   `out[...,d] = left[...,d] @ right[...,d].T`.
   We perform it with a batched GEMM (`torch.bmm`) after reshaping:
        (B*hidden_dim, N, N) × (B*hidden_dim, N, N)ᵀ → (B*hidden_dim,N,N)
6. Layer‑norm the result over the hidden dimension,
   apply the `out_gate` (broadcasted), and finally linearly project back
   to `dim` with `to_out`.
7. Return the tensor of shape [B,N,N,dim] (float32).

Only the element‑wise mask‑gate‑projection fusion is written in Triton;
all linear layers and the large batched matrix multiply use PyTorch’s
high‑performance kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: fused multiply three tensors (A * B * C)
# A, C : [B,N,N,D]   (projected tensor & gate)
# B    : [B,N,N]     (binary mask, broadcast over D)
# ----------------------------------------------------------------------
@triton.jit
def fused_mul3_kernel(
    a_ptr, b_ptr, c_ptr, out_ptr,
    B, N, D,
    stride_a_batch, stride_a_i, stride_a_j, stride_a_d,
    stride_b_batch, stride_b_i, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,
    BLOCK: tl.constexpr
):
    pid = tl.program_id(0)
    offs = tl.arange(0, BLOCK)
    idx = pid * BLOCK + offs

    # Out‑of‑bounds mask (bool vector)
    valid = idx < (B * N * N * D)

    # De‑compose linear index into (b,i,j,d)
    d = idx % D
    tmp = idx // D
    j = tmp % N
    tmp = tmp // N
    i = tmp % N
    b = tmp // N

    # Compute element offsets respecting the supplied strides
    a_off = b * stride_a_batch + i * stride_a_i + j * stride_a_j + d * stride_a_d
    b_off = b * stride_b_batch + i * stride_b_i + j * stride_b_j
    c_off = b * stride_c_batch + i * stride_c_i + j * stride_c_j + d * stride_c_d

    a = tl.load(a_ptr + a_off, mask=valid, other=0.0)
    bm = tl.load(b_ptr + b_off, mask=valid, other=0.0)   # mask is 0/1
    c = tl.load(c_ptr + c_off, mask=valid, other=0.0)

    tl.store(out_ptr + a_off, a * bm * c, mask=valid)


def fused_mul3(a: torch.Tensor, b: torch.Tensor, c: torch.Tensor) -> torch.Tensor:
    """Element‑wise a * b * c where a,c are [B,N,N,D] and b is [B,N,N].

    The computation is performed by a single Triton kernel to reduce
    memory traffic.  All tensors must reside on the same CUDA device.
    """
    assert a.shape == c.shape
    B, N, N2, D = a.shape
    assert N == N2, "Tensor must be square in the two sequence dimensions."
    total = B * N * N * D

    BLOCK = 1024  # threads per program (tune if needed)
    grid = lambda meta: (triton.cdiv(total, BLOCK),)

    # strides in elements (not bytes)
    stride_a = a.stride()
    stride_b = b.stride()
    stride_c = c.stride()

    out = torch.empty_like(a)
    fused_mul3_kernel[grid](
        a, b, c, out,
        B, N, D,
        stride_a[0], stride_a[1], stride_a[2], stride_a[3],
        stride_b[0], stride_b[1], stride_b[2],
        stride_c[0], stride_c[1], stride_c[2], stride_c[3],
        BLOCK=BLOCK,
    )
    return out

    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating (sigmoid of linear projection)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Mask handling (binary mask 0/1, broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is None:
        mask_tensor = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=dtype,
            device=device,
        )
    else:
        # Ensure dtype matches the rest of the tensors
        mask_tensor = mask.to(dtype)

    # ------------------------------------------------------------------
    # 5) Fuse mask, projection and gate with Triton
    # ------------------------------------------------------------------
    left = fused_mul3(left_proj, mask_tensor, left_gate)   # [B,N,N,hidden_dim]
    right = fused_mul3(right_proj, mask_tensor, right_gate)

    # ------------------------------------------------------------------
    # 6) Core multiplicative update:
    #    out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    Implemented as batched GEMM over the k dimension.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # Rearrange to (B, hidden_dim, N, N) for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & channel dimensions → (B*hidden_dim, N, N)
    left_mat = left_perm.view(B * hidden_dim, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)

    # Batched matrix multiplication: each slice does Σ_k left[i,k] * right[j,k]
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))

    # Restore original layout: [B, N, N, hidden_dim]
    out = (
        out_mat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )

    # ------------------------------------------------------------------
    # 7) Output LayerNorm (over hidden_dim) + out_gate
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    out = out * out_gate  # element‑wise gating

    # ------------------------------------------------------------------
    # 8) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
```
