    N, H,

@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_li, stride_lk, stride_lh,
    stride_rj, stride_rk, stride_rh,
    stride_oi, stride_oj, stride_oh,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # Compute offsets
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # Masks for range
    mask_i = i < N
    mask_j = j < N

    # Pointers with hidden channel offset
    left_ptr = left_ptr + pid_h * stride_lh
    right_ptr = right_ptr + pid_h * stride_rh
    out_ptr = out_ptr + pid_h * stride_oh

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k
    k = 0
    while k < N:
        cur_k = tl.minimum(BLOCK_K, N - k)
        k_range = k + tl.arange(0, cur_k)

        # Load left tile (i,k)
        left = tl.load(left_ptr + i[:, None] * stride_li + k_range[None, :] * stride_lk,
                       mask=mask_i[:, None] & (k_range[None, :] < N), other=0.0)

        # Load right tile (k,j) - note shape (cur_k, BLOCK_J)
        right = tl.load(right_ptr + k_range[:, None] * stride_rk + j[None, :] * stride_rj,
                        mask=(k_range[:, None] < N) & mask_j[None, :], other=0.0)

        # Compute dot product (i,j) = left @ right (where left shape (BLOCK_I, cur_k), right shape (cur_k, BLOCK_J))

        k += BLOCK_K

    # Store output

"""
TriMul (outgoing) implementation with a fused Triton kernel.

The heavy work of the module is the 4‑D einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden dimension *d* this is a matrix multiplication
    (N×N) @ (N×N)^T.
We launch a Triton kernel that computes a block of the (i,j) matrix
for one hidden channel at a time:
    - program_id(0) → i‑tile
    - program_id(1) → j‑tile
    - program_id(2) → hidden‑channel index
Inside the kernel we loop over the k‑dimension in tiles, load a
( BLOCK_I × BLOCK_K ) block from *left* and a ( BLOCK_K × BLOCK_J )
block from *right* (notice the transposed layout for *right*) and
accumulate a dot‑product with `tl.dot`.  All arithmetic is done in
float32; the kernel is launched for every batch separately.

The remaining operations (LayerNorm, Linear projections, sigmoids,
masking and the final projection) are performed with PyTorch
functions, which are already highly optimized.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # *float32   (B,N,N,H) – base pointer for a single batch
    right_ptr,         # *float32   (B,N,N,H)
    out_ptr,           # *float32   (B,N,N,H)
    N,                 # int32      sequence length
    stride_li, stride_lk, stride_lh,   # int32  left strides: i, k, h
    stride_rj, stride_rk, stride_rh,   # int32  right strides: j, k, h
    stride_oi, stride_oj, stride_oh,   # int32  out strides: i, j, h
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)          # i‑tile
    pid_j = tl.program_id(1)          # j‑tile
    pid_h = tl.program_id(2)          # hidden channel

    # ------------------------------------------------------------------
    # 1) Compute global indices of the tiles we are responsible for
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i < N
    mask_j = j < N

    # Advance pointers to the current hidden channel (stride_lh == stride_rh == stride_oh == 1)
    left_ptr  = left_ptr  + pid_h * stride_lh
    right_ptr = right_ptr + pid_h * stride_rh
    out_ptr   = out_ptr   + pid_h * stride_oh

    # Accumulator for this tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over the k dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    k = 0
    while k < N:
        # k‑indices for this tile
        k_offsets = k + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # ------------------------------------------------------------------
        # Load left block: shape (BLOCK_I, BLOCK_K)
        # left[i, k, h]   → i‑axis stride = stride_li, k‑axis stride = stride_lk
        # ------------------------------------------------------------------
        left = tl.load(
            left_ptr + i[:, None] * stride_li + k_offsets[None, :] * stride_lk,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ------------------------------------------------------------------
        # Load right block transposed: shape (BLOCK_K, BLOCK_J)
        # right[j, k, h]   → k‑axis stride = stride_rk, j‑axis stride = stride_rj
        # We want (k, j) so we place k in the leading dimension.
        # ------------------------------------------------------------------
        right = tl.load(
            right_ptr + k_offsets[:, None] * stride_rk + j[None, :] * stride_rj,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # Block matrix multiplication: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right)

        k += BLOCK_K

    # ------------------------------------------------------------------
    # 3) Write the result back to memory
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Public entry point: custom_kernel(data)
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor of shape [B, N, N, C] (C = dim)
            - mask         : torch.Tensor of shape [B, N, N] (bool/float) or None
            - weights      : dict of torch.Tensors with keys:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config       : dict containing at least:
                * "dim" (input feature dimension)
                * "hidden_dim" (the hidden dimension used inside TriMul)
                * optional "nomask" (bool) – if True mask is ignored
    Returns:
        torch.Tensor of shape [B, N, N, dim] – the forward pass result.
    """

    device = input_tensor.device
    dtype = input_tensor.dtype

    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # 0) Optional mask handling
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        # a mask of all ones – broadcasting will have no effect
        mask_expanded = None
    else:
        mask_expanded = mask.unsqueeze(-1)                     # [B,N,N,1]

    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Compute mean/var over the last dimension (C)
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x = (input_tensor - mean) * torch.rsqrt(var + eps)
    x = x * norm_weight + norm_bias                     # shape [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])    # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Apply mask if present (before gating, exactly as in the reference)
    # ------------------------------------------------------------------
    if mask_expanded is not None:
        left = left * mask_expanded
        right = right * mask_expanded

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core TriMul contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left)                      # [B,N,N,hidden_dim]

    # Kernel launch configuration
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    for b in range(B):
        # Slice the batch dimension – each slice is a contiguous [N,N,hidden_dim] tensor
        left_b = left[b]
        right_b = right[b]
        out_b = out[b]

        # Ensure contiguous layout (important for stride correctness)
        left_b = left_b.contiguous()
        right_b = right_b.contiguous()
        out_b = out_b.contiguous()

        # Strides in *elements* (not bytes)
        stride_li, stride_lk, stride_lh = left_b.stride()
        stride_rj, stride_rk, stride_rh = right_b.stride()
        stride_oi, stride_oj, stride_oh = out_b.stride()

        grid = (
            triton.cdiv(N, BLOCK_I),          # i‑tiles
            triton.cdiv(N, BLOCK_J),          # j‑tiles
            hidden_dim,                       # one program per hidden channel
        )

        tri_mul_outgoing_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            stride_li, stride_lk, stride_lh,
            stride_rj, stride_rk, stride_rh,
            stride_oi, stride_oj, stride_oh,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
        )
        # implicit synchronization when we next use `out`

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gate + final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply the output gate (broadcast over the hidden dimension)
    out = out * out_gate

    # Final linear projection back to the original dimension (no bias)
    out = F.linear(out, weights["to_out.weight"])

    return out
```
