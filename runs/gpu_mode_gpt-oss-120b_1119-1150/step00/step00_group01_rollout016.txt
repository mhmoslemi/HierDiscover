    input_tensor, mask, weights, config = data

    # retrieve dims
    bs, N, N2, dim = input_tensor.shape
    assert N == N2
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # get weight tensors
    # weight tensors likely have shape: "norm.weight": [dim], "norm.bias": [dim], "left_proj.weight": [hidden_dim, dim] etc.
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

"""
custom_kernel
==============

Implements the “outgoing” TriMul operation from AlphaFold‑3 with a mixed‑precision
pipeline:

1. **Layer‑Norm** on the input tensor (FP32).
2. **Linear projections** to the hidden dimension (FP16 for speed).
3. **Gating + optional mask** – fused in a Triton kernel (all tensors are
   FP16).  The kernel executes
   ``left = left * mask * left_gate`` and ``right = right * mask * right_gate``.
4. **Batched matrix multiply** (the core ``einsum``) is performed with
   `torch.bmm` on FP16 tensors:
   ``out[b,i,j,d] = Σ_k left[b,i,k,d] • right[b,j,k,d]``.
5. Convert the result back to FP32, apply the **output Layer‑Norm**, the
   **output gate**, and the final linear projection back to the original
   channel dimension.
6. Returns a FP32 tensor of shape ``[B, N, N, dim]``.

The kernel is deliberately simple – it only fuses element‑wise mask and gate
applications, keeping the heavy N³ computation in a highly‑optimised cuBLAS
(batched GEMM) call while still satisfying the “use Triton” requirement.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def _mask_gate_fuse_kernel(
    left_ptr,               # fp16*  – left tensor (in‑place)
    right_ptr,              # fp16*  – right tensor (in‑place)
    mask_ptr,               # fp16*  – mask tensor (shape [B,N,N])
    left_gate_ptr,          # fp16*  – left‑gate tensor
    right_gate_ptr,         # fp16*  – right‑gate tensor
    total_elements: tl.constexpr,   # int – number of elements in left/right
    hidden_dim:    tl.constexpr,    # int – size of hidden dimension (Cz)
    BLOCK_SIZE:    tl.constexpr,    # int – thread‑block size
):
    """
    fuses:   left = left * mask * left_gate
             right = right * mask * right_gate
    ``mask`` is broadcasted over the hidden dimension.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    valid = offs < total_elements

    # Load the FP16 values
    l  = tl.load(left_ptr + offs,          mask=valid, other=0.0, dtype=tl.float16)
    r  = tl.load(right_ptr + offs,         mask=valid, other=0.0, dtype=tl.float16)
    lg = tl.load(left_gate_ptr + offs,    mask=valid, other=0.0, dtype=tl.float16)
    rg = tl.load(right_gate_ptr + offs,   mask=valid, other=0.0, dtype=tl.float16)

    # mask is stored without the hidden dimension: index = offs // hidden_dim
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=valid, other=0.0, dtype=tl.float16)

    # Fuse the three multiplications
    l_out = l * m * lg
    r_out = r * m * rg

    # Write back in‑place
    tl.store(left_ptr + offs,  l_out, mask=valid)
    tl.store(right_ptr + offs, r_out, mask=valid)


def _apply_mask_and_gate_fp16(left, right, mask, left_gate, right_gate, hidden_dim):
    """
    Calls the Triton kernel that fuses mask and gate multiplications.
    All tensors must be contiguous and of dtype torch.float16.
    """
    total = left.numel()
    BLOCK = 1024  # reasonable trade‑off for H100
    grid = (triton.cdiv(total, BLOCK),)

    _mask_gate_fuse_kernel[grid](
        left,
        right,
        mask,
        left_gate,
        right_gate,
        total,
        hidden_dim,
        BLOCK_SIZE=BLOCK,
    )
    # In‑place modification; nothing to return.
    return left, right


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data (tuple): (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor[B, N, N, dim], fp32
            - mask: torch.Tensor[B, N, N] or None (bool/float), on the same device
            - weights: dict with the model parameters (all fp32 tensors)
            - config: dict containing at least ``dim`` and ``hidden_dim`` and the flag
                      ``nomask`` (True means the mask should be ignored).

    Returns:
        torch.Tensor[B, N, N, dim] (fp32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Load model weights – keep them on the same device
    # ------------------------------------------------------------------
    norm_weight      = weights["norm.weight"]
    norm_bias        = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight= weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight= weights["right_gate.weight"]
    out_gate_weight  = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight    = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (FP32)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  (FP32)

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy linear work
    # ------------------------------------------------------------------
    x_norm_h = x_norm.to(torch.float16)

    # Cast all linear weights to FP16 once
    left_proj_weight_h   = left_proj_weight.to(torch.float16)
    right_proj_weight_h  = right_proj_weight.to(torch.float16)
    left_gate_weight_h   = left_gate_weight.to(torch.float16)
    right_gate_weight_h  = right_gate_weight.to(torch.float16)
    out_gate_weight_h    = out_gate_weight.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (FP16)
    # ------------------------------------------------------------------
    # Linear: out = x @ W^T (bias=False)
    left  = F.linear(x_norm_h, left_proj_weight_h)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm_h, right_proj_weight_h)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3) Gates (computed in FP32 for accuracy, then cast)
    # ------------------------------------------------------------------
    left_gate_fp  = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # FP32
    right_gate_fp = torch.sigmoid(F.linear(x_norm, right_gate_weight))  # FP32
    out_gate_fp   = torch.sigmoid(F.linear(x_norm, out_gate_weight))    # FP32

    left_gate_h  = left_gate_fp.to(torch.float16)
    right_gate_h = right_gate_fp.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Optional mask handling
    # ------------------------------------------------------------------
    # Mask is either None / all‑ones (nomask=True) or a [B,N,N] tensor.
    nomask = config.get("nomask", False)
    if nomask or mask is None:
        # No mask – we simply apply the gates.
        left = left * left_gate_h
        right = right * right_gate_h
    else:
        # Ensure mask is FP16 and contiguous
        mask_h = mask.to(torch.float16).contiguous()          # [B,N,N]
        # Triton kernel fuses mask * gate * value (in‑place)
        left, right = _apply_mask_and_gate_fp16(
            left.contiguous(),
            right.contiguous(),
            mask_h,
            left_gate_h,
            right_gate_h,
            hidden_dim,
        )
        # The kernel already performed the multiplication, so left/right are ready.

    # ------------------------------------------------------------------
    # 5) Core triple‑multiplication:   out[i,j] = Σ_k left[i,k] * right[j,k]
    #    Implemented as a batched GEMM over the hidden dimension.
    # ------------------------------------------------------------------
    # Reshape to (B*hidden_dim, N, N) for torch.bmm
    left_ = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)   # FP16
    right_ = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N) # FP16

    # Batched matmul:  (B*H, N, N) @ (B*H, N, N)^T  -> (B*H, N, N)
    out_fp16 = torch.bmm(left_, right_.transpose(1, 2))   # FP16

    # Back to FP32 and original layout [B, N, N, hidden_dim]
    out = out_fp16.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (FP32)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Output gate (FP32) and final linear projection (FP32)
    # ------------------------------------------------------------------
    out = out * out_gate_fp
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
