@triton.jit
def trimul_outgoing_kernel( left_ptr, right_ptr, out_ptr,
                            B, N, H,
                            stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
                            stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
                            stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
                            BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
                            BLOCK_H: tl.constexpr):
    pid_b = tl.program_id(2) # batch index b
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute start indices
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = tl.arange(0, BLOCK_H)  # hidden dims within block

    # Mask for bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # Base pointers for each batch
    left_batch_ptr = left_ptr + pid_b * stride_left_batch
    right_batch_ptr = right_ptr + pid_b * stride_right_batch
    out_batch_ptr = out_ptr + pid_b * stride_out_batch

    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset range
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers with broadcasting: left[i,k,:]
        left_ptrs = left_batch_ptr + \
            (offs_i[:, None, None] * stride_left_i) + \
            (offs_k[None, :, None] * stride_left_k) + \
            (offs_h[None, None, :] * stride_left_h)

        # pointers for right[j,k,:] but we need layout (k, j) for dot
        right_ptrs = right_batch_ptr + \
            (offs_k[:, None, None] * stride_right_k) + \
            (offs_j[None, :, None] * stride_right_i) + \
            (offs_h[None, None, :] * stride_right_h)

        # load data
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :], other=0.0)

        # dot product across k dimension
        # shape: (BLOCK_M, BLOCK_K, BLOCK_H) dot (BLOCK_K, BLOCK_N, BLOCK_H) -> (BLOCK_M, BLOCK_N, BLOCK_H)
        acc += tl.dot(left, right)

    # Write back output
    out_ptrs = out_batch_ptr + \
        (offs_i[:, None, None] * stride_out_i) + \
        (offs_j[None, :, None] * stride_out_j) + \
        (offs_h[None, None, :] * stride_out_h)

    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel computing out[i,j] = sum_k left[i,k] * right[j,k] per batch and head.
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_j, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched per-head matrix multiplication (i,k)·(k,j) → (i,j).

    The kernel is launched with a 3‑D grid:
        axis 0 → i‑tile,
        axis 1 → j‑tile,
        axis 2 → combined (batch, head) index.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hd = tl.program_id(2)

    # decode batch and head from the combined index
    b = pid_hd // H          # batch id
    h = pid_hd % H           # head id (hidden‑dim index)

    # range of rows/cols for this program instance
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # pointers that already include the batch and head offsets
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left:  (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right: (k, j) – note the swap of i/j indices compared to the storage order
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # accumulate the product
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # write back the result
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    TriMul “outgoing” forward pass.
    The heavy N³ reduction is performed by a Triton kernel that evaluates,
    for each batch and hidden channel, the matrix product
        out[b, :, :, h] = left[b, :, :, h] @ right[b, :, :, h].T
    The surrounding LayerNorm/Linear/Gates are carried out with PyTorch
    (still on‑device) for simplicity.
    """
    # unpack arguments
    input_tensor, mask, weights, config = data

    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # config may also contain dim, but we rely on input shape

    # ----- LayerNorm on the input ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- Linear projections -----------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ----- Optional masking -------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.to(dtype).unsqueeze(-1)   # (bs, N, N, 1)
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gating ----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Triton kernel for the N³ reduction ------------------------------------
    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape (bs, N, N, hidden_dim)

    # kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,
    )

    # strides (batch, i, j, h)
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_j, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        stride_left_batch,
        stride_left_i,
        stride_left_j,
        stride_left_h,
        stride_right_batch,
        stride_right_i,
        stride_right_j,
        stride_right_h,
        stride_out_batch,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ----- Post‑processing --------------------------------------------------------
    # LayerNorm on the hidden dimension
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply output gate
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # (bs, N, N, dim)

    return out
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
# Each program instance computes a tile of the (i,j) matrix for a single
# (batch, hidden) pair.  The reduction over k is performed with a
# standard blocked GEMM loop.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_j, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched per‑head matrix multiplication (i,k)·(k,j) → (i,j).

    Grid layout:
        axis 0 → i‑tile,
        axis 1 → j‑tile,
        axis 2 → combined (batch, hidden‑dim) index.
    """
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile column
    pid_hd = tl.program_id(2)         # batch * hidden index

    # Decode batch and hidden‑dim index
    b = pid_hd // H
    h = pid_hd % H

    # Offsets for the current tile
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointers that already incorporate the batch and hidden offsets
    left_base  = left_ptr  + b * stride_left_batch  + h * stride_left_h
    right_base = right_ptr + b * stride_right_batch + h * stride_right_h
    out_base   = out_ptr   + b * stride_out_batch   + h * stride_out_h

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Reduction over k (the sequence dimension)
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left block:   (i, k)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_j
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

        # right block:  (k, j)  – note the swap of i/j indices w.r.t. storage order
        b_ptrs = right_base + offs_k[:, None] * stride_right_i + offs_j[None, :] * stride_right_j
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)

        # Update accumulator
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_N)

    # Write the tile back to global memory
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + offs_j[None, :] * stride_out_j
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry‑point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” Triangle‑Multiplicative‑Update (TriMul)
    module.  All heavy N³ work (the einsum) is executed by the Triton
    kernel defined above; surrounding LayerNorm/Linear/Gating steps are
    performed with native PyTorch ops on‑device.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left  = F.linear(x, left_proj_weight)   # (bs, N, N, hidden_dim)
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_exp = mask.to(dtype).unsqueeze(-1)          # (bs, N, N, 1)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating values (sigmoid of linear(x))
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) N³ reduction via Triton: out = left @ rightᵀ (per hidden dim)
    # ------------------------------------------------------------------
    left  = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)          # (bs, N, N, hidden_dim)

    # Triton launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,                 # combine batch and hidden‑dim index
    )

    # Strides for the three tensors (batch, i, j, h)
    sL_b, sL_i, sL_j, sL_h = left.stride()
    sR_b, sR_i, sR_j, sR_h = right.stride()
    sO_b, sO_i, sO_j, sO_h = out.stride()

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        bs,
        seq_len,
        hidden_dim,
        sL_b, sL_i, sL_j, sL_h,
        sR_b, sR_i, sR_j, sR_h,
        sO_b, sO_i, sO_j, sO_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                               # apply output gate

    to_out_weight = weights["to_out.weight"]           # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                 # (bs, N, N, dim)

    return out
