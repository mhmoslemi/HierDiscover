"""
TriMul (outgoing) implementation with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm over the last dimension (dim) of the input tensor.
2. Linear projections (no bias) to a hidden dimension `hidden_dim`:
   * left = x @ W_left
   * right = x @ W_right
3. Optional binary mask (broadcast over the hidden dim) is applied after projection.
4. Gated multiplicative updates:
   * left  = left  * sigmoid(x @ W_left_gate)
   * right = right * sigmoid(x @ W_right_gate)
5. Core triangular multiplication:
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of `hidden_dim` matrix‑multiplications
   (left[b,d] @ right[b,d].T) and is evaluated with a Triton kernel.
6. Layer‑norm over the hidden dimension, gated by `out_gate`,
   and a final linear projection back to `dim`.

The kernel works on `float16` tensors (to halve memory traffic) while
accumulating in `float32` for numerical stability.  The final result is
returned in `float32`.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------
# Triton kernel for the triangular multiplication:
# ---------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # *float16  [B, N, N, D]
    right_ptr,         # *float16  [B, N, N, D]
    out_ptr,           # *float16  [B, N, N, D]
    N,                 # seq length (int)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    B, D,              # batch size, hidden dim (int)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute, for every batch `b` and hidden channel `d`,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Block sizes are controllable via constexprs.
    """
    # -------------------------------------------------
    # 1) Identify which (b, d) pair this program works on
    # -------------------------------------------------
    pid = tl.program_id(axis=0)      # combined (b, d) dimension
    pid_i = tl.program_id(axis=1)    # block row   (i)
    pid_j = tl.program_id(axis=2)    # block col   (j)

    b = pid // D                     # batch index
    d = pid % D                      # hidden‑channel index

    # -------------------------------------------------
    # 2) Offsets for the tile we are responsible for
    # -------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base pointer offsets for this (b, d)
    off_b_d_left  = b * stride_left_b  + d * stride_left_d
    off_b_d_right = b * stride_right_b + d * stride_right_d
    off_b_d_out   = b * stride_out_b   + d * stride_out_d

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------
    # 3) Loop over the reduction dimension k
    # -------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile   : [BLOCK_M, BLOCK_K]
        left_ptrs = left_ptr + off_b_d_left + \
                    (offs_i[:, None] * stride_left_i) + \
                    (offs_k[None, :] * stride_left_k)
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        a = a.to(tl.float32)   # promote to fp32 for accumulation

        # right tile  : [BLOCK_N, BLOCK_K] (note the j‑index first)
        right_ptrs = right_ptr + off_b_d_right + \
                     (offs_j[:, None] * stride_right_j) + \
                     (offs_k[None, :] * stride_right_k)
        b_tile = tl.load(right_ptrs,
                         mask=mask_j[:, None] & mask_k[None, :],
                         other=0.0)
        b_tile = b_tile.to(tl.float32)

        # matrix‑multiply the K‑tiles (reduction over K)
        acc += tl.dot(a, b_tile)   # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------
    # 4) Write the result back (cast to fp16)
    # -------------------------------------------------
    out_ptrs = out_ptr + off_b_d_out + \
               (offs_i[:, None] * stride_out_i) + \
               (offs_j[None, :] * stride_out_j)

    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


# ---------------------------------------------------------
# Python entry‑point that follows the required `custom_kernel`
# signature.
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module, using a fused Triton
    kernel for the O(N³) triangular multiplication.

    Args:
        data: Tuple
            - input_tensor : torch.Tensor of shape [B, N, N, dim]
            - mask         : torch.Tensor of shape [B, N, N] (or None)
            - weights      : dict of torch.Tensors (see skeleton)
            - config       : dict with keys 'dim' and 'hidden_dim'

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # -----------------------------------------------------------------
    # 0) Cast everything to the desired dtypes.
    #    We keep the heavy tensors (projected values) in fp16
    #    to save memory, while the final output is fp32.
    # -----------------------------------------------------------------
    device = input_tensor.device
    x = input_tensor  # [B, N, N, dim] (float32 by default)

    # ----- LayerNorm over the last dim -----
    x = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # still FP32

    # ----- Linear projections (FP32) -----
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Cast projected tensors to fp16 for the heavy kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ----- Optional mask (broadcast over hidden dim) -----
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(left.dtype)   # [B, N, N, 1]
        left = left * mask_
        right = right * mask_

    # ----- Gating values (compute in FP32, then cast) -----
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid().to(left.dtype)
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid().to(right.dtype)
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()  # keep FP32 for later

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 1) Triangular multiplication with Triton
    # -----------------------------------------------------------------
    B, N, _, D = left.shape  # D == hidden_dim
    assert D == hidden_dim, "Hidden dimension mismatch."

    # Output placeholder (fp16)
    out = torch.empty_like(left, dtype=torch.float16, device=device)

    # Strides (in elements, not bytes)
    stride_left = left.stride()    # (b, i, k, d)
    stride_right = right.stride()  # (b, j, k, d)
    stride_out = out.stride()      # (b, i, j, d)

    # Choose block sizes – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_i_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_j_blocks = (N + BLOCK_N - 1) // BLOCK_N
    total_instances = B * hidden_dim   # each (batch, channel) pair gets its own program

    # Launch Triton kernel
    tri_mul_kernel[(total_instances, num_i_blocks, num_j_blocks)](
        left,
        right,
        out,
        N,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        B, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 2) Remaining post‑processing in FP32
    # -----------------------------------------------------------------
    out = out.to(torch.float32)   # [B, N, N, hidden_dim]

    # Layer‑norm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # Gating with out_gate (still FP32)
    out = out * out_gate

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
