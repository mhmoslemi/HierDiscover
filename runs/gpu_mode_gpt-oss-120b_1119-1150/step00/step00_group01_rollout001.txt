def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # mask
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul: transpose right's second dimension and permute
    # left_perm: [B, hidden_dim, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous() # swap k and j

    B, H, N, _ = left_perm.shape
    # Flatten batch: batch_total = B*H
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)
    # allocate output
    out_flat = torch.empty_like(left_flat)

    # launch triton kernel
    grid = (B*H, (N + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N -1)//BLOCK_N)
    batched_matmul_kernel[grid](

        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_aba, stride_abm, stride_abk,
    stride_bba, stride_bbk, stride_bbn,
    stride_cca, stride_ccm, stride_ccn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # check mask
    mask_m = offs_m < M
    mask_n = offs_n < N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointers for A and B
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        a_ptrs = a_ptr + pid_batch * stride_aba + offs_m[:, None] * stride_abm + offs_k[None, :] * stride_abk
        b_ptrs = b_ptr + pid_batch * stride_bba + offs_k[:, None] * stride_bbk + offs_n[None, :] * stride_bbn

        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(a, b)   # dot expects shape (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N), returns (BLOCK_M,BLOCK_N)

"""
TriMul forward (outgoing) with Triton‑accelerated pairwise multiplication.

Algorithm
---------
1. Layer‑norm the 4‑D input tensor (B, N, N, C) using the provided
   `norm.weight` and `norm.bias`.
2. Linear projections → `left` and `right` (C → H) and three gating
   linear layers (C → H) are performed with torch.nn.functional.linear.
3. Optional mask (B,N,N) is broadcast and applied to `left`/`right`.
4. Gating: `left *= sigmoid(left_gate)`, `right *= sigmoid(right_gate)`.
5. Core operation – the “triangle multiplicative update”:
   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h].
   This is a batched matrix multiplication over the hidden dimension:
   For each hidden channel h we compute
       out_h = left_h @ right_hᵀ .
   The computation is carried out by a custom Triton kernel that
   processes all (B·H) batches in parallel with a tiled GEMM.
6. Layer‑norm over the hidden dimension, gate with `out_gate`,
   and final linear projection back to C.
All other operations stay in PyTorch; only the O(N³·H) reduction
   (step 5) is performed on the GPU with Triton.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# --------------------------------------------------------------------------- #
# Triton kernel: batched matrix multiplication C = A @ Bᵀ
#   A: (batch, M, K)   – here M = N, K = N
#   B: (batch, K, N)   – B already contains the transposed dimension
#   C: (batch, M, N)
# --------------------------------------------------------------------------- #
@triton.jit
def batched_matmul_kernel(
        a_ptr, b_ptr, c_ptr,
        BATCH, M, N, K,
        stride_a_batch, stride_a_m, stride_a_k,
        stride_b_batch, stride_b_k, stride_b_n,
        stride_c_batch, stride_c_m, stride_c_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """ Compute C = A @ Bᵀ for many independent batches.

    Arguments
    ---------
    a_ptr, b_ptr, c_ptr : pointer to the first element of each tensor.
    BATCH, M, N, K       : integer sizes (batch, rows, cols, reduction axis).
    stride_*            : stride (in elements) for each dimension.
    BLOCK_*             : compile‑time tile sizes.
    """
    pid_batch = tl.program_id(0)   # batch index   (0 … BATCH‑1)
    pid_m = tl.program_id(1)       # block row index
    pid_n = tl.program_id(2)       # block col index

    # ------------------------------- tile coordinates -------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------- reduction over K -------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A has shape (batch, M, K)
        a_ptrs = a_ptr + pid_batch * stride_a_batch \
                 + offs_m[:, None] * stride_a_m \
                 + offs_k[None, :] * stride_a_k

        # B has shape (batch, K, N) – we already have Bᵀ in memory
        b_ptrs = b_ptr + pid_batch * stride_b_batch \
                 + offs_k[:, None] * stride_b_k \
                 + offs_n[None, :] * stride_b_n

        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & mask_k[None, :]),
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_n[None, :]),
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)          # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # ------------------------------- write back result -------------------------------
    c_ptrs = c_ptr + pid_batch * stride_c_batch \
             + offs_m[:, None] * stride_c_m \
             + offs_n[None, :] * stride_c_n

    tl.store(c_ptrs, acc,
             mask=(mask_m[:, None] & mask_n[None, :]))


# --------------------------------------------------------------------------- #
# Public entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)
        - input_tensor : torch.Tensor, shape (B, N, N, C)
        - mask_tensor  : torch.Tensor, shape (B, N, N) or None
        - weight_dict  : dict with the LayerNorm / Linear weights
        - config_dict  : dict with at least keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C) – the TriMul output.
    """
    # --------------------------------------------------------------- unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # --------------------------------------------------------------- LayerNorm (nn.LayerNorm style)
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # --------------------------------------------------------------- Linear projections
    left = F.linear(x, weights["left_proj.weight"])        # (B,N,N,H)
    right = F.linear(x, weights["right_proj.weight"])      # (B,N,N,H)

    # --------------------------------------------------------------- optional mask
    if mask is not None:
        # mask may be bool; cast to same dtype as x
        mask = mask.to(dtype=x.dtype, device=x.device)
        mask = mask.unsqueeze(-1)          # (B,N,N,1)
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------- Gating (sigmoid)
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()   # (B,N,N,H)
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()     # (B,N,N,H)

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------- Core TriMul (batched GEMM)
    # Transpose to shape (B, H, N, N) so that each hidden channel becomes a batch.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    # right needs to be transposed on the second N dimension (k ↔ j)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    total_batches = B * H

    # Collapse the (B, H) dimensions into a single batch dimension for the kernel.
    left_flat = left_perm.view(total_batches, N, N)
    right_flat = right_perm.view(total_batches, N, N)
    out_flat = torch.empty_like(left_flat)

    # --------------------------------------------------------------- launch Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    grid = (
        total_batches,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batches,
        N,
        N,
        N,
        left_flat.stride(0),   # batch stride
        left_flat.stride(1),   # M stride (rows)
        left_flat.stride(2),   # K stride (cols)
        right_flat.stride(0),  # batch stride
        right_flat.stride(1),  # K stride
        right_flat.stride(2),  # N stride
        out_flat.stride(0),    # batch stride
        out_flat.stride(1),    # M stride
        out_flat.stride(2),    # N stride
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------- post‑processing
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # (B, N, N, C)

    return out
