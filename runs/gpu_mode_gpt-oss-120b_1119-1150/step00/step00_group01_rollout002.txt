    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    right = torch.nn.functional.linear(x, right_proj_weight)

        left = left * mask_exp
        right = right * mask_exp
    else:
        # skip
        pass

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    right = right * right_gate

    # Convert to fp16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Prepare output tensor

    BLOCK_N = 64
    BLOCK_K = 64

    right_ptr = right_fp16
    out_ptr = out_fp16

    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_k, stride_out_h = out_fp16.stride()

         stride_out_b, stride_out_i, stride_out_k, stride_out_h,
         BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Cast back to float32
    out = out_fp16.to(torch.float32)

    # to_out_norm and out_gate

    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # result shape [B,N,N,dim]

    return out_final

"""
TriMul custom forward (outgoing) implementation.

- Applies LayerNorm on the last channel of the input tensor.
- Projects the normalized tensor with two linear layers (left/right) and
  computes gated versions using three additional linear layers + sigmoid.
- The expensive contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
  is executed with a fused Triton kernel that performs a batched GEMM
  for each (batch, hidden) pair, treating the hidden dimension as an
  extra batch axis.  The kernel tiles the M, N and K dimensions with
  configurable block sizes and writes the result back in FP16 (accumulated
  in FP32 for numerical stability).  The final output is cast back to
  FP32, normalized, gated and linearly projected to the original channel
  dimension.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel that computes the Einstein‑summation:
#   out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
# ----------------------------------------------------------------------
@triton.jit
def _trimul_einsum_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Shapes / strides
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # Compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute one tile of the TriMul contraction for a specific (batch, hidden)
    pair.  The kernel is launched with a three‑dimensional grid:
        (num_tiles_M, num_tiles_N, B*H)
    where each program instance processes a BLOCK_M × BLOCK_N output tile.
    """
    pid_m = tl.program_id(0)          # tile row   (i dimension)
    pid_n = tl.program_id(1)          # tile col   (j dimension)
    pid_bh = tl.program_id(2)         # flattened batch * hidden index

    # ------------------------------------------------------------------
    # decode batch and hidden indices
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Compute the start offsets for the current tile
    # ------------------------------------------------------------------
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / columns
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension (the summed axis)
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- load left tile (M × K) -----
        # left[b, i, k, h]
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + h * stride_left_h \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left_tile = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        ).to(tl.float32)

        # ----- load right tile (K × N) -----
        # right[b, j, k, h]  (we need it as K × N for the dot product)
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + h * stride_right_h \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        ).to(tl.float32)

        # ----- matrix multiplication & accumulate -----
        #   (M×K) · (K×N) -> (M×N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + h * stride_out_h \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (or None)
        - weights      : dict of torch.Tensor containing all linear / LN params
        - config       : dict with keys "dim", "hidden_dim", optionally "layernorm_eps"

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (FP32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ------------------------------------------------------------------
    # Extract weight tensors
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w   = weights["left_proj.weight"]
    right_proj_w  = weights["right_proj.weight"]
    left_gate_w   = weights["left_gate.weight"]
    right_gate_w  = weights["right_gate.weight"]
    out_gate_w    = weights["out_gate.weight"]

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional masking (mask is broadcast over the channel dim)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)       # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate  = F.linear(x, left_gate_w).sigmoid()
    right_gate = F.linear(x, right_gate_w).sigmoid()
    out_gate   = F.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Heavy contraction – execute in FP16 on Triton, accumulate in FP32
    # ------------------------------------------------------------------
    # Cast to FP16 to reduce memory traffic (the kernel converts back to FP32 internally)
    left_fp16  = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, H = left_fp16.shape  # H == hidden_dim

    # Output buffer (still FP16, will be cast back later)
    out_fp16 = torch.empty_like(left_fp16)

    # Tiling parameters – these work well on H100 for the tested N ranges
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (convert to python ints for Triton)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Grid: (tiles_M, tiles_N, B*H)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * H,
    )

    _trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # modest occupancy; can be tuned
        num_stages=2,              # double‑buffering
    )

    # Cast back to FP32 for the remaining layers
    out = out_fp16.to(torch.float32)

    # ------------------------------------------------------------------
    # 6️⃣ Output normalisation, gating and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    out = out * out_gate                         # element‑wise gating
    out = F.linear(out, to_out_w)                # [B, N, N, dim]

    return out
