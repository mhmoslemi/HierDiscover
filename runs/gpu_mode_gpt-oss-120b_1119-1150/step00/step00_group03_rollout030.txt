2. left = Linear(dim->hidden_dim, bias=False) (weights left_proj_weight) apply to last dim: result [B,N,N,hidden_dim].

4. mask shape [B,N,N]; if mask exists (not nomask), we need to multiply left and right in place: mask.unsqueeze(-1). Probably mask is boolean or float (0 or 1). In kernel we can do that after projection.

6. right_gate = sigmoid(Linear(dim->hidden_dim, bias=False)(x_norm)).
7. out_gate = sigmoid(Linear(dim->hidden_dim, bias=False)(x_norm)). Wait out_gate is Linear(dim->hidden_dim, bias=False). 9. Compute out = einsum('... i k d, ... j k d -> ... i j d', left, right). Equivalent per hidden d: out[:,:,,:,d] = left_d @ right_d.T.

10. out_norm = LayerNorm(hidden_dim)(out). Weighted by to_out_norm weight and bias.

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_bl, stride_il, stride_kl, stride_hl,
                   stride_br, stride_jr, stride_kr, stride_hr,
                   stride_bo, stride_io, stride_jo, stride_ho,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
                   offset_h: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute block offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    # compute indices
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    d_offs = tl.arange(0, BLOCK_D) + offset_h

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = d_offs < H

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M, K, D)
        # compute pointer offsets
        left_ptrs = left_ptr + \
            pid_b * stride_bl + \
            offs_i[:, None, None] * stride_il + \
            offs_k[None, :, None] * stride_kl + \
            d_offs[None, None, :] * stride_hl
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # Load right tile (K, N, D) with transposed layout
        right_ptrs = right_ptr + \
            pid_b * stride_br + \
            offs_k[:, None, None] * stride_kr + \
            offs_j[None, :, None] * stride_jr + \
            d_offs[None, None, :] * stride_hr
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # Compute partial dot products for each hidden slice
        for d in range(BLOCK_D):
            # extract 2D slice
            left_slice = left[:, :, d]
            right_slice = right[:, :, d]
            # accumulate
            acc[:, :, d] += tl.dot(left_slice, right_slice)

    # Store output
    out_ptrs = out_ptr + \
        pid_b * stride_bo + \
        offs_i[:, None, None] * stride_io + \
        offs_j[None, :, None] * stride_jo + \
        d_offs[None, None, :] * stride_ho
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

@triton.jit
def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, N, H,
                   stride_bl, stride_il, stride_kl, stride_hl,
                   stride_br, stride_jr, stride_kr, stride_hr,
                   stride_bo, stride_io, stride_jo, stride_ho,
                   BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
                   offset_h: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # range idx within block
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    d = offset_h + tl.arange(0, BLOCK_D)

    # masks for bounds
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of k tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left tile load (M, K, D)
        left_ptrs = left_ptr + \
            pid_b * stride_bl + \
            i[:, None, None] * stride_il + \
            k[None, :, None] * stride_kl + \
            d[None, None, :] * stride_hl
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # right tile load (K, N, D) with transposition (K, J)
        right_ptrs = right_ptr + \
            pid_b * stride_br + \
            k[:, None, None] * stride_kr + \
            j[None, :, None] * stride_jr + \
            d[None, None, :] * stride_hr
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # compute dot for each hidden slice
        for dh in range(BLOCK_D):
            acc[:, :, dh] += tl.dot(left[:, :, dh], right[:, :, dh])

    # write to output
    out_ptrs = out_ptr + \
        pid_b * stride_bo + \
        i[:, None, None] * stride_io + \
        j[None, :, None] * stride_jo + \
        d[None, None, :] * stride_ho
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)
```

def tri_mul_forward(input_tensor, mask_tensor, weights, config):
    # parse dims
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", True):
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Compute einsum via Triton
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # get strides in elements
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # grid dimensions
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8  # hidden channels per kernel

    grid = (B,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # iterate over hidden channel slices
    for offset_h in range(0, hidden_dim, BLOCK_D):
        # launch kernel
        tri_mul_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            hidden_dim,

"""
Fast Triton implementation of the “outgoing” Triangle Multiplicative Update (TriMul)
used in AlphaFold‑3 and related protein‑structure models.

The heavy O(N³·H) reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
is off‑loaded to a custom Triton kernel while the surrounding LayerNorms,
linear projections, sigmoids and gating are performed with PyTorch.
The kernel processes a (BLOCK_M×BLOCK_N) tile of the (i,j) output and a small
slice of the hidden dimension (BLOCK_D) per launch, using block‑wise loads and
`tl.dot` for the inner K‑reduction.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def _tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_jr, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
    offset_h: tl.constexpr,
):
    """
    Compute a tiled slice of the TriMul contraction:
          out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]

    Arguments
    ---------
    left_ptr, right_ptr, out_ptr : pointers to the tensors
    B, N, H                     : batch, sequence length and hidden dim (constexpr)
    stride_*                    : element‑wise strides for the tensors
    BLOCK_*                     : tile sizes (constexpr)
    offset_h                    : start index of the processed hidden‑dim slice (constexpr)
    """
    pid_b = tl.program_id(0)               # batch index
    pid_i = tl.program_id(1)               # tile row (i)
    pid_j = tl.program_id(2)               # tile col (j)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # indices inside the tile
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)
    d = offset_h + tl.arange(0, BLOCK_D)

    # bounds masks
    mask_i = i < N
    mask_j = j < N
    mask_d = d < H

    # accumulator for the (M,N,D) output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # number of K‑tiles
    num_k = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k):
        k0 = kk * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------------------------------------------------------
        # Load left [M, K, D]  (shape respects the original layout)
        # left[b, i, k, d]
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + pid_b * stride_bl
            + i[:, None, None] * stride_il
            + k[None, :, None] * stride_kl
            + d[None, None, :] * stride_hl
        )
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)   # (M,K,D)

        # ------------------------------------------------------------------
        # Load right transposed [K, N, D] so that we effectively have
        # right[b, j, k, d] → accessed as (k, j, d)   (shape K,N,D)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + pid_b * stride_br
            + k[:, None, None] * stride_kr
            + j[None, :, None] * stride_jr
            + d[None, None, :] * stride_hr
        )
        right_mask = mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)   # (K,N,D)

        # ------------------------------------------------------------------
        # Accumulate the outer‑product for each hidden slice
        # ------------------------------------------------------------------
        for dh in range(BLOCK_D):
            # left[:, :, dh] -> (M, K), right[:, :, dh] -> (K, N)
            acc[:, :, dh] += tl.dot(left[:, :, dh], right[:, :, dh])

    # ----------------------------------------------------------------------
    # Store the result tile
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_b * stride_bo
        + i[:, None, None] * stride_io
        + j[None, :, None] * stride_jo
        + d[None, None, :] * stride_ho
    )
    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel on the whole [B,N,N,H] tensors.
    """
    B, N, _, H = left.shape
    device = left.device

    out = torch.empty((B, N, N, H), dtype=left.dtype, device=device)

    # strides in element units (Triton expects element strides)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_jr, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # Tile sizes – chosen to fit comfortably in registers on H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden‐dim slice processed per kernel launch

    grid = (
        B,                                    # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,         # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,         # j‑tiles
    )

    # Launch one kernel per hidden‑dim slice
    for off_h in range(0, H, BLOCK_D):
        _tri_mul_kernel[grid](
            left,
            right,
            out,
            B,
            N,
            H,
            stride_bl, stride_il, stride_kl, stride_hl,
            stride_br, stride_jr, stride_kr, stride_hr,
            stride_bo, stride_io, stride_jo, stride_ho,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
            offset_h=off_h,
        )
    return out


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be omitted when config["nomask"] is True)
        - weights      : dict of torch.Tensor containing all linear / layer‑norm parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the result of the TriMul forward pass.
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. LayerNorm on the input tensor (dim = last dimension)
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias) -> shape [B,N,N,hidden_dim]
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional masking (mask is broadcast over the hidden dimension)
    # ----------------------------------------------------------------------
    if not nomask:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)      # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----------------------------------------------------------------------
    # 4. Gating values (sigmoid of linear projections of normalized input)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Core TriMul contraction (heavy O(N³·H) part) via Triton
    # ----------------------------------------------------------------------
    # Ensure tensors are contiguous for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_triton(left, right)               # shape [B,N,N,hidden_dim]

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm + output gating
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ----------------------------------------------------------------------
    # 7. Final linear back to original dimension
    # ----------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
