import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# Triton kernel for TriMul batched matrix multiplication across hidden channels.
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program IDs
    pid_m = tl.program_id(0)   # block index over output rows (i)
    pid_n = tl.program_id(1)   # block index over output cols (j)
    pid_bh = tl.program_id(2)  # combined batch & hidden channel index

    # decompose combined batch channel
    b = pid_bh // H
    h = pid_bh % H
    # Base offsets for this batch/channel
    off_batch = b * stride_l_b + h * stride_l_h

    # Compute row / column offsets within the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        # k offsets for this tile
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + off_batch \
                    + (offs_m[:, None] * stride_l_i) \
                    + (offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_m[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        # We load it as (k, j) so that tl.dot(left, right) yields (i, j)
        right_ptrs = right_ptr + off_batch \
                     + (offs_k[:, None] * stride_r_k) \
                     + (offs_n[None, :] * stride_r_i)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_n[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate product
        acc += tl.dot(left, right)

    # Write out the result tile
    out_ptrs = out_ptr + b * stride_o_b + h * stride_o_h \
               + (offs_m[:, None] * stride_o_i) \
               + (offs_n[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    TriMul outgoing kernel (forward only).

    This implementation follows the PyTorch reference:
    1. Layer‑norm on the input.
    2. Linear projections for left/right branches and their gates.
    3. Apply mask (if present) and gate sigmoids.
    4. Compute the core ``einsum('... i k d, ... j k d -> ... i j d')`` using
       a custom Triton kernel that treats each hidden channel as an
       independent batched GEMM: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
    5. Post‑processing LayerNorm, gate, and final linear projection.

    All heavy arithmetic (the O(N³·H) reduction) is performed inside the Triton
    kernel. The rest of the operations use fast PyTorch primitives.
    """
    # unpack inputs
    input_tensor, mask_tensor, weights, config = data
    # Get shapes & config
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # Load weight tensors (they are expected to be on the same device)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # 1. LayerNorm over dim
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # 2. Linear projections (no bias)
    left = F.linear(x, left_proj_weight)          # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)        # same shape

    # 3. Mask handling (if mask is provided and not nomask)
    if not config.get("nomask", False):
        # mask shape [bs, N, N] -> broadcast to hidden dim
        mask = mask_tensor.to(dtype).unsqueeze(-1)   # [bs, N, N, 1]
        left = left * mask
        right = right * mask

    # 4. Gate computations
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()
    # Allocate output buffer
    out = torch.empty_like(left)  # shape [bs, N, N, hidden_dim]

    # Triton kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    num_blocks_m = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (seq_len + BLOCK_N - 1) // BLOCK_N
    batch_hidden = bs * hidden_dim

    grid = (num_blocks_m, num_blocks_n, batch_hidden)

    # Launch kernel
    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),
        stride_r_i=right.stride(1),
        stride_r_k=right.stride(2),
        stride_r_h=right.stride(3),
        stride_o_b=out.stride(0),
        stride_o_i=out.stride(1),
        stride_o_j=out.stride(2),
        stride_o_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # 5. Post‑norm and final linear projection
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate

    # Final linear back to original dimension
    out = F.linear(out, to_out_weight)  # [bs, N, N, dim]

    return out

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The module follows the reference PyTorch implementation:
1. Layer‑norm over the input channel dimension.
2. Linear projections to a hidden dimension (left/right) and three sigmoid gates.
3. Optional mask multiplication.
4. Core compute:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a batch of GEMMs (one per hidden channel).
   This O(N³·H) reduction is performed by a custom Triton kernel.
5. Post‑normalisation, multiplication by the output gate and a final linear
   projection back to the original channel dimension.

Only the heavy reduction is fused in Triton; the surrounding operations use
high‑performance PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    Each (b,h) pair is treated as an independent batch.  The kernel tiles the
    output in (BLOCK_M, BLOCK_N) and iterates over k in blocks of size
    BLOCK_K.  All arithmetic is performed in float32.
    """
    # -------------------------------------------------------------------------
    # Identify the current block
    pid_m = tl.program_id(0)      # output‑row block index (i)
    pid_n = tl.program_id(1)      # output‑col block index (j)
    pid_bh = tl.program_id(2)     # combined batch‑ & hidden‑channel index

    # Recover batch and hidden‑channel indices
    b = pid_bh // H
    h = pid_bh % H

    # Base offsets for the selected (b,h) slice
    off_l = b * stride_l_b + h * stride_l_h
    off_r = b * stride_r_b + h * stride_r_h
    off_o = b * stride_o_b + h * stride_o_h

    # Row / column offsets inside the output tile
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # Loop over the reduction dimension k
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k_iter * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------- left tile: shape (BLOCK_M, BLOCK_K)
        # left[b, i, k, h]
        left_ptrs = left_ptr + off_l \
                    + (offs_i[:, None] * stride_l_i) \
                    + (offs_k[None, :] * stride_l_k)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # -------- right tile transposed: shape (BLOCK_K, BLOCK_N)
        # we load as right[b, k, j, h] so that tl.dot(left, right) → (i, j)
        right_ptrs = right_ptr + off_r \
                     + (offs_k[:, None] * stride_r_k) \
                     + (offs_j[None, :] * stride_r_i)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # Accumulate the product over k
        acc += tl.dot(left_tile, right_tile)

    # -------------------------------------------------------------------------
    # Write the computed tile to the output tensor
    out_ptrs = out_ptr + off_o \
               + (offs_i[:, None] * stride_o_i) \
               + (offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a Triton kernel for
    the O(N³·H) reduction.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor of shape [bs, seq_len, seq_len, dim]
            - mask_tensor: torch.Tensor of shape [bs, seq_len, seq_len] (may be ignored)
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------ unpack ------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------ weights ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------ 1. Input LayerNorm ------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------ 2. Linear projections (no bias) ----------------------
    left = F.linear(x, left_proj_weight)   # [bs, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # same shape

    # ------------------------ 3. Optional mask --------------------------------------
    if not config.get("nomask", False):
        # mask: [bs, N, N] -> [bs, N, N, 1] for broadcasting
        mask = mask_tensor.to(dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------ 4. Gates ------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [bs, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------ 5. Core TriMul (Triton) -------------------------------
    # Ensure contiguous layout for Triton
    left = left.contiguous()
    right = right.contiguous()

    # Output buffer for the reduction result
    out = torch.empty_like(left)  # shape [bs, N, N, hidden_dim]

    # Tiling parameters – chosen for good occupancy on an H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid dimensions
    num_blocks_m = (seq_len + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (seq_len + BLOCK_N - 1) // BLOCK_N
    batch_hidden = bs * hidden_dim
    grid = (num_blocks_m, num_blocks_n, batch_hidden)

    # Launch the kernel
    tri_mul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=bs,
        N=seq_len,
        H=hidden_dim,
        stride_l_b=left.stride(0),
        stride_l_i=left.stride(1),
        stride_l_k=left.stride(2),
        stride_l_h=left.stride(3),
        stride_r_b=right.stride(0),
        stride_r_i=right.stride(1),
        stride_r_k=right.stride(2),
        stride_r_h=right.stride(3),
        stride_o_b=out.stride(0),
        stride_o_i=out.stride(1),
        stride_o_j=out.stride(2),
        stride_o_h=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------ 6. Post‑norm, output gate, final linear --------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)  # final projection, shape [bs, N, N, dim]

    return out
