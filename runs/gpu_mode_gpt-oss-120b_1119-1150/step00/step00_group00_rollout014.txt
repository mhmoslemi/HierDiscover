```
@triton.jit
def triton_matmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    bh = tl.program_id(0)  # combined batch and hidden dimension
    i_block = tl.program_id(1)
    j_block = tl.program_id(2)
    # compute batch and hidden index
    batch_idx = bh // H
    hidden_idx = bh % H
    # compute base pointers for this batch & hidden
    # Starting address for this (batch, hidden)
    base_l = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    base_r = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    base_o = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # compute start offsets for i, j
    i_offset = i_block * BLOCK_M
    j_offset = j_block * BLOCK_N

    # create range vectors
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # mask to handle out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        # compute pointer offsets for a (left) and b (right)
        # A_tile shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        # B_tile shape (BLOCK_K, BLOCK_N) => rows k, cols j
        b_ptrs = base_r + offs_j[:, None] * stride_r_i + offs_k[None, :] * stride_r_k
        # Load with appropriate masks
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)
        # Compute dot product: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)
        # tl.dot works on 2-D matrices
        acc = acc + tl.dot(a, b)
        # increment pointers automatically in next loop for k
    # write back output
    o_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(o_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

    j_offset = j_block * BLOCK_N

    # range vectors
    offs_i = i_offset + tl.arange(0, BLOCK_M)
    offs_j = j_offset + tl.arange(0, BLOCK_N)

    # masks for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs, mask=(mask_i[:, None] & mask_k[None, :]), other=0.0)

        # B tile: shape (BLOCK_K, BLOCK_N) -> rows = k, cols = j
        b_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs, mask=(mask_k[:, None] & mask_j[None, :]), other=0.0)

        # Accumulate
        acc = acc + tl.dot(a, b)  # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # write back
    o_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
Out‑going TriMul (AlphaFold3) implemented with Triton.

Algorithm
---------
1️⃣ Layer‑norm the input tensor over the last dimension.
2️⃣ Two linear projections (left/right) produce tensors of shape [B,N,N,H].
3️⃣ Optional mask (broadcasted) zeros out invalid entries.
4️⃣ Gating: three sigmoid‑gated linear layers (left, right, out).
5️⃣ Multiply left/right by their respective gates.
6️⃣ Heavy O(N³·H) contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication C = L @ Rᵀ, realised by a custom
   Triton kernel that tiles the (i,j) plane and reduces over k.
7️⃣ Second Layer‑norm over the hidden dimension, multiply by out‑gate,
   then a final linear projection back to the original dimension.

Only the O(N³) part is fused in Triton; the surrounding linear / norm / sigmoid
operations use fused PyTorch kernels. The kernel works for any batch size (≤2),
sequence length up to 1024 and hidden dimensions up to 384 (the tests use H=128).
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _triton_outgoing_matmul_kernel(
    left_ptr,               # *tl.float16
    right_ptr,              # *tl.float16
    out_ptr,                # *tl.float32
    B, N, H,                # sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute C[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]  (L @ Rᵀ)
    Grid: (bh, i_tile, j_tile) where bh = B * H.
    """
    bh = tl.program_id(0)            # batch * hidden
    i_tile = tl.program_id(1)        # i‑tile index
    j_tile = tl.program_id(2)        # j‑tile index

    # ---- split combined dimension into batch and hidden index ----
    batch_idx = bh // H
    hidden_idx = bh % H

    # ---- base pointers for this (batch, hidden) pair ----
    base_l = left_ptr + batch_idx * stride_l_b + hidden_idx * stride_l_h
    base_r = right_ptr + batch_idx * stride_r_b + hidden_idx * stride_r_h
    base_o = out_ptr + batch_idx * stride_o_b + hidden_idx * stride_o_h

    # ---- tile offsets in i / j ----
    i_offset = i_tile * BLOCK_M
    j_offset = j_tile * BLOCK_N

    # ---- range vectors for the tile ----
    offs_i = i_offset + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_j = j_offset + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # ---- accumulator (float32) ----
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- reduction over k dimension ----
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)   # [BLOCK_K]
        mask_k = offs_k < N

        # Left tile  : (BLOCK_M, BLOCK_K)
        a_ptrs = base_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & mask_k[None, :]),
                    other=0.0)

        # Right tile : (BLOCK_K, BLOCK_N)   rows = k, cols = j
        b_ptrs = base_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=(mask_k[:, None] & mask_j[None, :]),
                    other=0.0)

        # Accumulate dot product: (M,K) @ (K,N) → (M,N)
        acc = acc + tl.dot(a, b)

    # ---- write back the result tile ----
    o_ptrs = base_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(o_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
              - input_tensor: torch.Tensor [B, N, N, dim]               (float32)
              - mask        : torch.Tensor [B, N, N]  (bool/int or float)
              - weights     : dict of torch.Tensor containing trainable params
              - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # same default as torch.nn.LayerNorm

    # ------------------------------------------------------------------
    # 1) First LayerNorm over the feature dimension
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the last dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Heavy contraction via Triton (L @ Rᵀ)
    # ------------------------------------------------------------------
    # Cast to half‑precision for the matmul (still accumulate in fp32)
    left_h = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    # Allocate output buffer for the contraction (float32)
    out_mat = torch.empty(
        (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2], hidden_dim),
        dtype=torch.float32,
        device=input_tensor.device,
    )

    # Strides for the three tensors (in elements, not bytes)
    l_str = left_h.stride()   # (B, i, k, h)
    r_str = right_h.stride()
    o_str = out_mat.stride()

    # Tile sizes – chosen for good occupancy on H100 (fp16 dot product)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        input_tensor.shape[0] * hidden_dim,                      # bh = B * H
        triton.cdiv(input_tensor.shape[1], BLOCK_M),           # i‑tiles
        triton.cdiv(input_tensor.shape[2], BLOCK_N),           # j‑tiles
    )

    _triton_outgoing_matmul_kernel[grid](
        left_h,
        right_h,
        out_mat,
        input_tensor.shape[0],   # B
        input_tensor.shape[1],   # N
        hidden_dim,               # H
        l_str[0], l_str[1], l_str[2], l_str[3],
        r_str[0], r_str[1], r_str[2], r_str[3],
        o_str[0], o_str[1], o_str[2], o_str[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_mat,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out_norm * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
