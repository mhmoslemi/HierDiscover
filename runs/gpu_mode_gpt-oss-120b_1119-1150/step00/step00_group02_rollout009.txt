```
@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr, weight_ptr, bias_ptr,
    B, N, dim,
    stride_ipb, stride_ipi, stride_ipj, stride_ipd,
    stride_opb, stride_opi, stride_opj, stride_opd,
    stride_w, stride_b,
    eps: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid = tl.program_id(0)
    # Compute b,i,j from pid
    total = B * N * N
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # Offsets for vectors
    offs_d = tl.arange(0, BLOCK_D)
    # linear index base:
    inp_base = inp_ptr + b * stride_ipb + i * stride_ipi + j * stride_ipj
    # offsets for each dim:
    inp_offsets = inp_base + offs_d * stride_ipd

    mask = offs_d < dim
    x = tl.load(inp_offsets, mask=mask, other=0.0).to(tl.float32)

    # Compute mean
    sum_x = tl.sum(x, axis=0)
    mean = sum_x / dim

    # Compute var: sum((x-mean)^2) / dim
    diff = x - mean
    var = tl.sum(diff * diff, axis=0) / dim

    eps_val = eps
    inv_std = 1.0 / tl.sqrt(var + eps_val)

    # Normalize
    y = (x - mean) * inv_std
    # Apply weight and bias
    # Load weight, bias
    w = tl.load(weight_ptr + offs_d, mask=mask, other=0.0)
    b_ = tl.load(bias_ptr + offs_d, mask=mask, other=0.0)
    y = y * w + b_
    # Store
    out_offsets = out_ptr + b * stride_opb + i * stride_opi + j * stride_opj + offs_d * stride_opd
    tl.store(out_offsets, y, mask=mask)
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,D]).
2. Linear projections to a hidden dimension H (left_proj, right_proj) and
   the corresponding gating vectors (left_gate, right_gate).
3. A single Triton kernel fuses
      * mask broadcasting (optional),
      * sigmoid gating,
      * element‑wise multiplication (proj * mask * sigmoid(gate)),
   producing the gated tensors `left` and `right` of shape [B,N,N,H].
4. Convert `left/right` to fp16 and perform a batched matrix‑multiply:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is realised as many independent (N×N) GEMMs (B·H of them) via
   `torch.bmm` (uses Tensor‑cores on H100).
5. Layer‑norm the result over the hidden dimension, apply the output gate
   (sigmoid of a linear projection), and finally a linear projection back
   to the original dimension D.

Only step 3 is written in Triton; all other heavy operations use
high‑performance PyTorch kernels (cuBLAS / Tensor Cores). The implementation
supports both masked and un‑masked inputs and works for the test configurations
specified in the prompt.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: fuse mask, sigmoid gating and element‑wise multiplication
# -------------------------------------------------------------------------
@triton.jit
def gate_fuse_kernel(
    # pointers
    left_proj_ptr,   # [B,N,N,H]   proj   (float32)
    right_proj_ptr,  # [B,N,N,H]   proj   (float32)
    left_gate_ptr,   # [B,N,N,H]   raw gating (float32)
    right_gate_ptr,  # [B,N,N,H]   raw gating (float32)
    mask_ptr,        # [B,N,N,1]   broadcast mask (float32)
    left_out_ptr,    # [B,N,N,H]   result after gating (float32)
    right_out_ptr,   # [B,N,N,H]   result after gating (float32)

    # dimensions
    B, N,                # batch size, sequence length
    stride_lp_b, stride_lp_i, stride_lp_j, stride_lp_h,
    stride_rp_b, stride_rp_i, stride_rp_j, stride_rp_h,
    stride_lg_b, stride_lg_i, stride_lg_j, stride_lg_h,
    stride_rg_b, stride_rg_i, stride_rg_j, stride_rg_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    stride_lo_b, stride_lo_i, stride_lo_j, stride_lo_h,
    stride_ro_b, stride_ro_i, stride_ro_j, stride_ro_h,

    HIDDEN_DIM: tl.constexpr,   # actual hidden dimension (e.g. 128)
    BLOCK_H: tl.constexpr,      # tile size in hidden dimension (must ≥ HIDDEN_DIM)
):
    pid = tl.program_id(0)                 # one program per (b,i,j)
    total_ij = N * N
    b = pid // total_ij
    ij = pid % total_ij
    i = ij // N
    j = ij % N

    # offsets for the start of the row (b,i,j,0)
    lp_base = left_proj_ptr + b * stride_lp_b + i * stride_lp_i + j * stride_lp_j
    rp_base = right_proj_ptr + b * stride_rp_b + i * stride_rp_i + j * stride_rp_j
    lg_base = left_gate_ptr + b * stride_lg_b + i * stride_lg_i + j * stride_lg_j
    rg_base = right_gate_ptr + b * stride_rg_b + i * stride_rg_i + j * stride_rg_j
    mask_base = mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # broadcast mask scalar (mask shape [...,1])
    mask_val = tl.load(mask_base).to(tl.float32)

    # vector indices inside the hidden dimension
    offs_h = tl.arange(0, BLOCK_H)
    valid = offs_h < HIDDEN_DIM

    # load vectors (float32)
    lp = tl.load(lp_base + offs_h * stride_lp_h, mask=valid, other=0.0).to(tl.float32)
    rp = tl.load(rp_base + offs_h * stride_rp_h, mask=valid, other=0.0).to(tl.float32)
    lg_raw = tl.load(lg_base + offs_h * stride_lg_h, mask=valid, other=0.0).to(tl.float32)
    rg_raw = tl.load(rg_base + offs_h * stride_rg_h, mask=valid, other=0.0).to(tl.float32)

    # sigmoid activation
    lg = 1.0 / (1.0 + tl.exp(-lg_raw))
    rg = 1.0 / (1.0 + tl.exp(-rg_raw))

    # apply mask and gating
    left = lp * mask_val * lg
    right = rp * mask_val * rg

    # store results
    lo_ptr = left_out_ptr + b * stride_lo_b + i * stride_lo_i + j * stride_lo_j
    ro_ptr = right_out_ptr + b * stride_ro_b + i * stride_ro_i + j * stride_ro_j
    tl.store(lo_ptr + offs_h * stride_lo_h, left, mask=valid)
    tl.store(ro_ptr + offs_h * stride_ro_h, right, mask=valid)

# -------------------------------------------------------------------------
# Entry point used by the evaluation harness
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # D  (input/output channel)
    hidden_dim = config["hidden_dim"]  # H  (projection dimension)
    eps = 1e-5                         # LayerNorm epsilon

    device = input_tensor.device
    B, N, _, _ = input_tensor.shape  # input shape [B, N, N, D]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # -----------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # -----------------------------------------------------------------
    # 2. Linear projections (pre‑gate) – keep float32 for maximum accuracy
    # -----------------------------------------------------------------
    left_proj = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right_proj = F.linear(x_norm, weights["right_proj.weight"]) # [B,N,N,H]
    left_gate_raw = F.linear(x_norm, weights["left_gate.weight"])   # [B,N,N,H]
    right_gate_raw = F.linear(x_norm, weights["right_gate.weight"]) # [B,N,N,H]

    # -----------------------------------------------------------------
    # 3. Optional mask handling
    # -----------------------------------------------------------------
    nomask = config.get("nomask", True)
    if nomask:
        mask_4d = torch.ones((B, N, N, 1), dtype=input_tensor.dtype, device=device)
    else:
        # mask_tensor has shape [B,N,N]; broadcast to last dim
        mask_4d = mask_tensor.unsqueeze(-1).to(input_tensor.dtype)

    # ensure contiguous layout for Triton
    left_proj = left_proj.contiguous()
    right_proj = right_proj.contiguous()
    left_gate_raw = left_gate_raw.contiguous()
    right_gate_raw = right_gate_raw.contiguous()
    mask_4d = mask_4d.contiguous()

    # allocate output tensors for the gated values
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # -----------------------------------------------------------------
    # 4. Triton kernel: fuse mask, sigmoid gating and elementwise multiplication
    # -----------------------------------------------------------------
    lp_s = left_proj.stride()
    rp_s = right_proj.stride()
    lg_s = left_gate_raw.stride()
    rg_s = right_gate_raw.stride()
    m_s = mask_4d.stride()
    lo_s = left.stride()
    ro_s = right.stride()

    grid = (B * N * N,)
    gate_fuse_kernel[grid](
        left_proj,
        right_proj,
        left_gate_raw,
        right_gate_raw,
        mask_4d,
        left,
        right,
        # dimensions
        B,
        N,
        # strides (b,i,j,h)
        lp_s[0], lp_s[1], lp_s[2], lp_s[3],
        rp_s[0], rp_s[1], rp_s[2], rp_s[3],
        lg_s[0], lg_s[1], lg_s[2], lg_s[3],
        rg_s[0], rg_s[1], rg_s[2], rg_s[3],
        m_s[0],  m_s[1],  m_s[2],
        lo_s[0], lo_s[1], lo_s[2], lo_s[3],
        ro_s[0], ro_s[1], ro_s[2], ro_s[3],
        # compile‑time constants
        HIDDEN_DIM=hidden_dim,
        BLOCK_H=hidden_dim,
        num_warps=4,
    )

    # -----------------------------------------------------------------
    # 5. Batched matrix multiplication (the N³ core)
    #    Compute out_{i,j,h} = Σ_k left_{i,k,h} * right_{j,k,h}
    #    We use half‑precision for the GEMM to exploit Tensor Cores.
    # -----------------------------------------------------------------
    # Convert to fp16 (memory + speed)
    left_h = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # [B,H,N,N]
    right_h = right.permute(0, 3, 1, 2).contiguous().to(torch.float16) # [B,H,N,N]

    # reshape to (B*H, N, N) for torch.bmm
    left_mat = left_h.view(B * hidden_dim, N, N)          # (B*H,N,N)
    right_mat_T = right_h.view(B * hidden_dim, N, N).transpose(1, 2)  # (B*H,N,N)

    # batched GEMM (fp16, Tensor‑cores)
    out_h = torch.bmm(left_mat, right_mat_T)             # (B*H,N,N)

    # back to FP32 and original layout [B,N,N,H]
    out = out_h.to(torch.float32).view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, output gating and final linear projection
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # [B,N,N,H]

    out_gate_raw = F.linear(x_norm, weights["out_gate.weight"])   # [B,N,N,H]
    out_gate = torch.sigmoid(out_gate_raw)

    out_gated = out_norm * out_gate

    # final linear back to dimensionality D
    output = F.linear(out_gated, weights["to_out.weight"])      # [B,N,N,D]

    return output
