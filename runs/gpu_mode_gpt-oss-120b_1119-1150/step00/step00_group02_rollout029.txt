    left_ptr, right_ptr, out_ptr,

    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,

        h = h_base + tl.arange(0, BLOCK_H)

"""
TriMul (outgoing) forward pass – a fused Triton kernel for the heavy
pairwise contraction.

The tensor layout is (batch, N, N, C) where C = hidden_dim after the
projections.  The core operation is
    out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:]                     (1)

which is equivalent to a batched matrix‑multiply left @ rightᵀ for each
hidden channel.  The kernel computes (1) directly, loading the hidden
vector for every (i,k) and (j,k) and accumulating the outer‑product over
k.  All other operations (LayerNorm, projections, gating, final
Linear) are performed with native PyTorch kernels; the kernel therefore
covers the O(N³·C) work that dominates runtime.

The implementation uses mixed‑precision: the projected tensors are stored
as fp16, but the accumulation is performed in fp32 to preserve numerical
stability.  The kernel processes the hidden dimension in blocks
(`BLOCK_H`) to fit in registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the triple contraction (outgoing version)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,          # pointers
    B, N, H,                               # B = batch, N = seq_len, H = hidden_dim
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""

    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # 1) map program id to (batch, i‑block, j‑block)
    # ------------------------------------------------------------------
    num_i_blocks = tl.cdiv(N, BLOCK_M)
    num_j_blocks = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_i_blocks * num_j_blocks

    batch_idx = pid // blocks_per_batch
    block_idx = pid % blocks_per_batch
    i_block   = block_idx // num_j_blocks
    j_block   = block_idx %  num_j_blocks

    # --------------------------------------------------------------
    # 2) offsets for the current i/j tile
    # --------------------------------------------------------------
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)   # (M,)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)   # (N,)
    i_valid = i < N
    j_valid = j < N

    # ------------------------------------------------------------------
    # 3) loop over hidden dimension in sub‑blocks
    # ------------------------------------------------------------------
    for hb in range(0, tl.cdiv(H, BLOCK_H)):
        # hidden offsets within this sub‑block
        h = hb * BLOCK_H + tl.arange(0, BLOCK_H)       # (H_sub,)
        h_valid = h < H

        # accumulator in fp32 (M,N,H_sub)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # --------------------------------------------------------------
        # 4) loop over the reduction dimension k
        # --------------------------------------------------------------
        for kb in range(0, tl.cdiv(N, BLOCK_K)):
            k = kb * BLOCK_K + tl.arange(0, BLOCK_K)   # (K,)
            k_valid = k < N

            # ------------------------------------------------------------------
            # 5) compute pointers for a (i,k,h) tile and a (j,k,h) tile
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + batch_idx * stride_left_batch
                + i[:, None, None] * stride_left_i
                + k[None, :, None] * stride_left_k
                + h[None, None, :] * stride_left_h
            )
            right_ptrs = (
                right_ptr
                + batch_idx * stride_right_batch
                + j[:, None, None] * stride_right_i
                + k[None, :, None] * stride_right_k
                + h[None, None, :] * stride_right_h
            )

            # mask all out‑of‑bounds loads
            load_mask = (
                i_valid[:, None, None]
                & j_valid[None, :, None]  # j is not used for loading right, but keep symmetry
                & k_valid[None, :, None]
                & h_valid[None, None, :]
            )
            # left/right are stored as fp16 → cast to fp32 for accumulation
            left_tile  = tl.load(left_ptrs,  mask=load_mask, other=0.0).to(tl.float32)   # (M,K,H_sub)
            right_tile = tl.load(right_ptrs, mask=load_mask, other=0.0).to(tl.float32)   # (N,K,H_sub)

            # --------------------------------------------------------------
            # 6) accumulate Σ_k left[i,k,:] * right[j,k,:]  (outer‑product over i/j)
            # --------------------------------------------------------------
            for kk in range(0, BLOCK_K):
                # left_vec : (M, H_sub)   ; right_vec : (N, H_sub)
                l = left_tile[:, kk, :]   # (M, H_sub)
                r = right_tile[:, kk, :]  # (N, H_sub)
                # broadcast to (M,N,H_sub) and add
                acc += l[:, None, :] * r[None, :, :]

        # ------------------------------------------------------------------
        # 7) write the result tile back to global memory
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + batch_idx * stride_out_batch
            + i[:, None, None] * stride_out_i
            + j[None, :, None] * stride_out_j
            + h[None, None, :] * stride_out_h
        )
        store_mask = i_valid[:, None, None] & j_valid[None, :, None] & h_valid[None, None, :]
        tl.store(out_ptrs, acc, mask=store_mask)


# ----------------------------------------------------------------------
# Python entry point – builds the full TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.
    Uses PyTorch for LayerNorm / Linear / gating and a custom Triton
    kernel for the cubic contraction (the dominant cost).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask – broadcast to the channel dimension
    # ------------------------------------------------------------------
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # Cast to half‑precision for the heavy kernel (saves bandwidth)
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # 5) Triton kernel for the cubic contraction
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # strides for pointer arithmetic
    stride_left_batch, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_batch, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tunable block sizes – chosen to fit H100 register limits
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 64
    BLOCK_H = 32

    grid = (B * ((N + BLOCK_M - 1) // BLOCK_M) *
            ((N + BLOCK_N - 1) // BLOCK_N),)

    tri_mul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        stride_left_batch=stride_left_batch,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_batch=stride_right_batch,
        stride_right_i=stride_right_i,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_batch=stride_out_batch,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,               # typical for H100
    )

    # ------------------------------------------------------------------
    # 6) post‑processing: LayerNorm over hidden dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )  # still [B,N,N,hidden]

    # multiply by the out‑gate (broadcasted)
    out = out * out_gate

    # final projection back to the original channel dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
