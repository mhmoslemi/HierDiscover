@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # Compute batch and hidden idx
    bh = pid_bc
    b = bh // H
    h = bh % H

    # Compute i and j offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Range within block
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)

    # Masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Compute reduction over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # Compute left tile (i,k)
        left_ptrs = left_ptr + b*stride_l_b + h*stride_l_h + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Compute right tile (k,j) (need transpose)
        right_ptrs = right_ptr + b*stride_r_b + h*stride_r_h + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate
        acc += tl.dot(left, right)   # shape (BLOCK_I, BLOCK_J)

    # Write out
    out_ptrs = out_ptr + b*stride_o_b + h*stride_o_h + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul forward (outgoing) implementation with Triton acceleration.

Algorithm
---------
1. Layer‑norm the input over the channel dimension (dim).
2. Linear projections to a hidden dimension `hidden_dim` for the left and right
   tensors, plus three gating projections (left_gate, right_gate, out_gate).
3. Apply an optional 2‑D mask (broadcasted on the hidden dimension) and the
   sigmoid gates.
4. Compute the core contraction:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix‑multiply of `left[...,h]` (shape B×N×N) with the
   transpose of `right[...,h]`.  The heavy O(N³) work is performed by a
   custom Triton kernel that tiles the i, j and reduction‑k dimensions.
5. Layer‑norm over the hidden dimension, multiply by the `out_gate`,
   and a final linear projection back to `dim`.

The Triton kernel processes a block of (BLOCK_I × BLOCK_J) output elements
and reduces over k in tiles of size BLOCK_K.  Batch index and hidden channel
are fused into a single program‑id dimension, allowing the kernel to be
launched with a 3‑D grid.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# -------------------------------------------------------------------------
# Triton kernel: computes the pairwise inner‑product over the third sequence
# dimension for every hidden channel.
# -------------------------------------------------------------------------
@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,               # tensor pointers
    B, N, H,                                    # batch, seq_len, hidden_dim
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # strides for left
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # strides for right
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # strides for out
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Grid:
        program_id(0) -> block i
        program_id(1) -> block j
        program_id(2) -> batch * hidden (linearized)
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # -----------------------------------------------------------------
    # Decode batch index and hidden channel from the fused program id.
    # -----------------------------------------------------------------
    b = pid_bc // H            # batch index
    h = pid_bc % H             # hidden channel index

    # -----------------------------------------------------------------
    # Compute absolute offsets for the i‑ and j‑tiles.
    # -----------------------------------------------------------------
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in float32 (regardless of input dtype)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Reduction over k dimension (the third sequence axis).
    # -----------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -------------------------------------------------------------
        # Load left tile: shape (BLOCK_I, BLOCK_K)
        # -------------------------------------------------------------
        left_ptrs = (left_ptr
                     + b * stride_l_b
                     + h * stride_l_h
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # -------------------------------------------------------------
        # Load right tile transposed: shape (BLOCK_K, BLOCK_J)
        #   right original layout is (j, k); we want (k, j) for dot.
        # -------------------------------------------------------------
        right_ptrs = (right_ptr
                      + b * stride_r_b
                      + h * stride_r_h
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # -------------------------------------------------------------
        # Accumulate: (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        # -------------------------------------------------------------
        acc += tl.dot(left, right)

    # -----------------------------------------------------------------
    # Store the result tile.
    # -----------------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + h * stride_o_h
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# -------------------------------------------------------------------------
# Entry point called by the evaluator.
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be all‑ones)
        - weights      : dict of torch.Tensors containing all learnable parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # Load weight tensors
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) Input layer‑norm over `dim`
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    # Linear works on the last dimension; we reuse `F.linear`.
    left = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # 4) Gating projections + sigmoid
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction via Triton (out shape = [B, N, N, hidden_dim])
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty_like(left, dtype=torch.float32, device=device)

    # Triton kernel launch parameters
    BLOCK_I = 64   # rows per block
    BLOCK_J = 64   # cols per block
    BLOCK_K = 64   # reduction tile size

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,                     # fused batch & hidden dimension
    )

    # Strides (in elements) for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Launch the kernel
    tr_mul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6) Output layer‑norm (over hidden_dim), gating and final linear projection
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # back to `dim`

    return out
```
