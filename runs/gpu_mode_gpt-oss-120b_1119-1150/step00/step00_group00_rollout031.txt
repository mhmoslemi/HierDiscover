    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_j,
    stride_right_batch, stride_right_i, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Triton kernel performing the pairwise multiplication reduction:
    out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h].
    """

    pid_j = tl.program_id(1)

    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # 1) LayerNorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    
    # 2) Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)    # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not nomask:

        mask = mask_tensor.unsqueeze(-1)
        left = left * mask
        right = right * mask

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    
    # Output placeholder
    out_fp32 = torch.empty_like(left_fp16, dtype=torch.float32)
    
    # Get strides
    stride_left_batch, stride_left_i, stride_left_j, stride_left_h = left_fp16.stride()
    stride_right_batch, stride_right_i, stride_right_j, stride_right_h = right_fp16.stride()
    stride_out_batch, stride_out_i, stride_out_j, stride_out_h = out_fp32.stride()
    
    # Block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    
    # Grid dimensions
    B, N, _, _ = left_fp16.shape
    grid = ( (N + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             B * hidden_dim )

    triton_trimul_kernel[
        grid
    ](
        left_fp16.data_ptr(),
        right_fp16.data_ptr(),
        out_fp32.data_ptr(),

        stride_right_batch, stride_right_i, stride_right_j,

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

The heavy cubic reduction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
is performed by a tiled FP16 mat‑mul kernel (batched over batch×head)
with FP32 accumulation. The rest of the TriMul pipeline (LayerNorm,
linear projections, gating, final LayerNorm + linear) is carried out in
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def triton_trimul_kernel(
    left_ptr,          # const half *
    right_ptr,         # const half *
    out_ptr,           # float *
    B, N, H,           # int32 scalars: batch, seq_len, hidden_dim
    stride_left_batch, stride_left_i, stride_left_j,
    stride_right_batch, stride_right_i, stride_right_j,
    stride_out_batch, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel treats (batch, head) as the third launch dimension.
    Each program processes a tile (BLOCK_M × BLOCK_N) of the output
    matrix for a single head h of a single batch b.
    """
    # -----------------------------------------------------------------
    # Program IDs
    pid_i = tl.program_id(0)               # tile row (i)
    pid_j = tl.program_id(1)               # tile col (j)
    pid_hb = tl.program_id(2)              # flattened batch*head index

    # De‑compose batch and head index
    b = pid_hb // H
    h = pid_hb % H

    # Base pointer for this (b, h)
    left_base  = left_ptr  + b * stride_left_batch  + h
    right_base = right_ptr + b * stride_right_batch + h
    out_base   = out_ptr   + b * stride_out_batch   + h

    # Tile offsets within the (i, j) dimensions
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the reduction dimension k
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_start = kb * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A tile: left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + offs_i[:, None] * stride_left_i + \
                 offs_k[None, :] * stride_left_j
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B_T tile (right transposed): right[b, j, k, h] as (k, j)
        b_ptrs = right_base + offs_k[:, None] * stride_right_j + \
                 offs_j[None, :] * stride_right_i
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Matrix‑multiply the tiles and accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) -> (BLOCK_M, BLOCK_N)

    # -----------------------------------------------------------------
    # Store the result tile
    c_ptrs = out_base + offs_i[:, None] * stride_out_i + \
             offs_j[None, :] * stride_out_j
    tl.store(c_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul outgoing update.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor [B, N, N, dim], float32
            - mask_tensor:  torch.Tensor [B, N, N] or None
            - weights: dict of model weights (see code)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (float32)
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)          # [B,N,N,dim]

    # Cast to fp16 for the heavy computation
    x_half = x.half()

    # 2) Linear projections (fp16)
    left = F.linear(x_half, left_proj_w.half())   # [B,N,N,hidden]
    right = F.linear(x_half, right_proj_w.half())

    # 3) Optional mask (broadcast over hidden dim)
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(left.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # 4) Gating (fp16 for left/right gates)
    left_gate = torch.sigmoid(F.linear(x_half, left_gate_w.half()))
    right_gate = torch.sigmoid(F.linear(x_half, right_gate_w.half()))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))   # keep out_gate in FP32

    # Apply gates
    left = (left * left_gate).contiguous()
    right = (right * right_gate).contiguous()

    # -----------------------------------------------------------------
    # 5) Cubic reduction via Triton kernel
    B, N, _, _ = left.shape

    # Output tensor (float32)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Strides (in elements) – left/right/out are contiguous [B,N,N,hidden]
    stride_left_batch, stride_left_i, stride_left_j, _ = left.stride()
    stride_right_batch, stride_right_i, stride_right_j, _ = right.stride()
    stride_out_batch, stride_out_i, stride_out_j, _ = out.stride()

    # Tile sizes – chosen to map well to Tensor Cores on H100
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    triton_trimul_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_left_batch, stride_left_i, stride_left_j,
        stride_right_batch, stride_right_i, stride_right_j,
        stride_out_batch, stride_out_i, stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from more warps per block
    )

    # -----------------------------------------------------------------
    # 6) Output LayerNorm (float32)
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=1e-5)

    # Apply final gating
    out = out * out_gate

    # 7) Final linear projection back to dim
    out = F.linear(out, to_out_w)   # [B,N,N,dim]

    return out
