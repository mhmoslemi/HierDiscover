def tri_mul_kernel(left_ptr, right_ptr, out_ptr,
                   B, D, N,
                   stride_l_b, stride_l_d, stride_l_i, stride_l_k,
                   stride_r_b, stride_r_d, stride_r_j, stride_r_k,
                   stride_o_b, stride_o_d, stride_o_i, stride_o_j,
                   BLOCK_M: tl.constexpr,
                   BLOCK_N: tl.constexpr,
                   BLOCK_K: tl.constexpr):

    batch_id = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute batch and channel indices
    b = batch_id // D
    d = batch_id % D

    # Compute root pointers for this batch+channel
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

    # Compute start offsets for i and j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & (offs_k < N), other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs, mask=(offs_k < N)[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Dot product: left (M,K) dot right (K,N)
        acc += tl.dot(left, right)

        # Increment pointers are not needed because we compute from base each iteration

    # Store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_i[:, None] & mask_j[None, :])

@end

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_l_b, stride_l_d, stride_l_i, stride_l_k,
    stride_r_b, stride_r_d, stride_r_j, stride_r_k,
    stride_o_b, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_and_channel = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_channels = D
    b = batch_and_channel // total_channels
    d = batch_and_channel % total_channels

    # base pointers for this batch+channel
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

    # Compute start offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile [BLOCK_I, BLOCK_K]
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed [BLOCK_K, BLOCK_J]
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)

    # store result
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

import torch
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_l_b, stride_l_d, stride_l_i, stride_l_k,
    stride_r_b, stride_r_d, stride_r_j, stride_r_k,
    stride_o_b, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    batch_channel = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    total_channels = D
    b = batch_channel // total_channels
    d = batch_channel % total_channels

    # base pointers for this batch+channel
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

    # compute offsets for this tile
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load left tile [BLOCK_I, BLOCK_K]
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile transposed [BLOCK_K, BLOCK_J]
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        acc += tl.dot(left, right)

    # write out
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    Triton-accelerated outgoing triangle multiplicative update (TriMul).
    Implements the forward pass of the AlphaFold3 TriMul module:
    1. LayerNorm over the feature dimension.
    2. Linear projections to a hidden dimension.
    3. Optional masking and gating (sigmoid) applied per element.
    4. Core multiplication: for each batch and hidden channel,
       out_{i,j} = sum_k left_{i,k} * right_{j,k}.
       Implemented as a batched matrix multiplication using a Triton kernel.
    5. LayerNorm over hidden dimension, gating and final linear projection back
       to the original feature dimension.
    """
    import torch
    import triton
    import triton.language as tl

    # unpack inputs
    inp, mask, weights, config = data
    # inp: [B, N, N, dim]
    # mask: [B, N, N] (may be None)
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim]

    # Linear projections (float32)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask (broadcast on hidden dim)
    if mask is not None:
        # ensure mask as float32 on same device
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the kernel to reduce bandwidth
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to [B, hidden_dim, N, N] and make contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, D, N_i, N_k = left_perm.shape
    assert N_i == N_k == inp.shape[1], "Sequence dimension mismatch"

    # Prepare output tensor (float32)
    out = torch.empty((B, D, N_i, N_k), dtype=torch.float32, device=inp.device)

    # Get strides (in elements)
    stride_l_b, stride_l_d, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_d, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_d, stride_o_i, stride_o_j = out.stride()

    # Prepare grid dimensions
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    grid = (B * D,
            (N_i + BLOCK_I - 1) // BLOCK_I,
            (N_k + BLOCK_J - 1) // BLOCK_J)

    # launch kernel
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out,
        B, D, N_i,
        stride_l_b, stride_l_d, stride_l_i, stride_l_k,
        stride_r_b, stride_r_d, stride_r_i, stride_r_k,  # note: we pass i stride as j stride
        stride_o_b, stride_o_d, stride_o_i, stride_o_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # The kernel expects right stride for j and k as (stride_r_j, stride_r_k). Here we have
    # stride_r_i as the stride for dimension that corresponds to `j` (since right and left share layout).
    # So we map stride_r_j = stride_r_i, stride_r_k = stride_r_k.
    # The argument ordering matches the kernel signature.

    # Permute back to original layout [B, N, N, hidden_dim]
    out_hidden = out.permute(0, 2, 3, 1).contiguous()

    # Second LayerNorm over hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gating
    out_gated = out_norm * out_gate

    # Final linear projection back to dim
    output = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) – Triton accelerated forward pass.

Algorithm
---------
1. Layer‑norm the input over the last feature dimension.
2. Linear projections to a hidden dimension (`left_proj`, `right_proj`).
3. Optional binary mask applied (broadcast on the hidden dim).
4. Gating: sigmoid(linear(x)) → `left_gate`, `right_gate`, `out_gate`.
   Multiply `left`/`right` by their gates.
5. Core operation:
   For each batch `b` and hidden channel `d`

        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

   This is a batched matrix‑multiply `left @ rightᵀ`.  Implemented with a
   tiled Triton kernel (`tri_mul_kernel`).  Operands are cast to fp16,
   accumulation is performed in fp32.
6. Layer‑norm over the hidden dimension, multiply by `out_gate`,
   and a final linear projection back to the original feature dimension.

The kernel fuses the O(N³) reduction without materialising any intermediate
tensors and is the only Triton‑coded part (the surrounding ops use
torch.nn.functional for clarity and speed).
"""

import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, D, N,
    stride_l_b, stride_l_d, stride_l_i, stride_l_k,
    stride_r_b, stride_r_d, stride_r_j, stride_r_k,
    stride_o_b, stride_o_d, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matrix multiplication for the TriMul core.

    out[b,d,i,j] = Σ_k left[b,d,i,k] * right[b,d,j,k]
    """
    # 3‑D grid: (batch*channel, i‑tile, j‑tile)
    batch_chan = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # decode batch and hidden channel
    b = batch_chan // D
    d = batch_chan % D

    # base pointers for this (b,d)
    left_base = left_ptr + b * stride_l_b + d * stride_l_d
    right_base = right_ptr + b * stride_r_b + d * stride_r_d
    out_base = out_ptr + b * stride_o_b + d * stride_o_d

    # tile offsets
    offs_i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k]   -> shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_base + (offs_i[:, None] * stride_l_i
                                 + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[j,k] transposed to (k,j) -> shape (BLOCK_K, BLOCK_J)
        right_ptrs = right_base + (offs_k[:, None] * stride_r_k
                                   + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # accumulate dot product over the K dimension
        acc += tl.dot(left, right)

    # write the tile back
    out_ptrs = out_base + (offs_i[:, None] * stride_o_i
                           + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Triton‑accelerated outgoing TriMul forward pass.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor  [B, N, N, dim]   (float32)
        - mask_tensor  : torch.Tensor  [B, N, N]       (bool/float, may be None)
        - weights_dict : mapping from the layer names used in the reference
                         implementation to torch.Tensors on the same device.
        - config_dict  : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor : [B, N, N, dim]  (float32)
    """
    # -------------------------------------------------------------
    # unpack
    # -------------------------------------------------------------
    inp, mask, weights, config = data
    device = inp.device
    eps = 1e-5

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # -------------------------------------------------------------
    # 1. Input LayerNorm over the feature dimension
    # -------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        inp,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -------------------------------------------------------------
    # 2. Linear projections
    # -------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # -------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(left.dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------
    # 4. Gating
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------
    # 5. Core TriMul – batched matmul via Triton
    # -------------------------------------------------------------
    # cast to fp16 for bandwidth reduction; accumulation stays fp32
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape to [B, hidden_dim, N, N] (contiguous)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    B, D, N_i, N_k = left_perm.shape
    assert N_i == N_k == inp.shape[1], "Seq‑len mismatch"

    # output in float32, shape [B, D, N, N]
    out = torch.empty((B, D, N_i, N_k), dtype=torch.float32, device=device)

    # strides (in elements)
    stride_l_b, stride_l_d, stride_l_i, stride_l_k = left_perm.stride()
    stride_r_b, stride_r_d, stride_r_i, stride_r_k = right_perm.stride()
    stride_o_b, stride_o_d, stride_o_i, stride_o_j = out.stride()

    # Triton launch configuration
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    grid = (
        B * D,
        (N_i + BLOCK_I - 1) // BLOCK_I,
        (N_k + BLOCK_J - 1) // BLOCK_J,
    )

    # The right‑hand side uses (j,k) layout; in memory j ↔ i stride.
    tri_mul_kernel[grid](
        left_perm,
        right_perm,
        out,
        B,
        D,
        N_i,
        stride_l_b,
        stride_l_d,
        stride_l_i,
        stride_l_k,
        stride_r_b,
        stride_r_d,
        stride_r_i,          # ← stride for `j` (original i‑stride)
        stride_r_k,
        stride_o_b,
        stride_o_d,
        stride_o_i,
        stride_o_j,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to original 4‑D layout: [B, N, N, hidden_dim]
    out_hidden = out.permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension
    # -------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # apply the output gate
    out_gated = out_norm * out_gate

    # -------------------------------------------------------------
    # 7. Final projection back to `dim`
    # -------------------------------------------------------------
    output = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return output
