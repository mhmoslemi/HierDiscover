```
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_Ab, stride_Am, stride_Ak,
                          stride_Bb, stride_Bk, stride_Bn,
                          stride_Cb, stride_Cm, stride_Cn,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers for A and B
    a_ptrs = A_ptr + batch * stride_Ab + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
    b_ptrs = B_ptr + batch * stride_Bb + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        # accumulate
        acc += tl.dot(a, b)
        # advance pointers
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write back
    c_ptrs = C_ptr + batch * stride_Cb + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])
```

Now the custom_kernel function:

```
def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Apply LayerNorm to input
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections: x shape (bs,N,N,dim)
    bs, N, _, _ = x.shape
    x_flat = x.reshape(bs * N * N, dim)  # (B*N*N, dim)

    # left_proj
    left = torch.matmul(x_flat, left_proj_weight.T).reshape(bs, N, N, hidden_dim)
    right = torch.matmul(x_flat, right_proj_weight.T).reshape(bs, N, N, hidden_dim)

    # gates
    left_gate = torch.sigmoid(torch.matmul(x_flat, left_gate_weight.T).reshape(bs, N, N, hidden_dim))
    right_gate = torch.sigmoid(torch.matmul(x_flat, right_gate_weight.T).reshape(bs, N, N, hidden_dim))
    out_gate = torch.sigmoid(torch.matmul(x_flat, out_gate_weight.T).reshape(bs, N, N, hidden_dim))

    # Apply mask if present
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # shape (bs,N,N,1)
        left = left * mask
        right = right * mask
    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matmul
    # Permute to (bs, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch & hidden dimension
    batch_hidden = bs * hidden_dim
    A = left_perm.reshape(batch_hidden, N, N)
    B = right_perm.reshape(batch_hidden, N, N)
    # Transpose B for matmul using kernel
    B_T = B.transpose(1, 2).contiguous()

    # Cast to fp16 for kernel bandwidth
    A_fp16 = A.to(torch.float16)
    B_T_fp16 = B_T.to(torch.float16)
    # Allocate output fp16
    C_fp16 = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # Kernel launch parameters
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), batch_hidden)

    batched_matmul_kernel[grid](
        A_fp16, B_T_fp16, C_fp16,
        M=N, N=N, K=N,
        stride_Ab=A_fp16.stride(0),
        stride_Am=A_fp16.stride(1),
        stride_Ak=A_fp16.stride(2),
        stride_Bb=B_T_fp16.stride(0),
        stride_Bk=B_T_fp16.stride(1),
        stride_Bn=B_T_fp16.stride(2),
        stride_Cb=C_fp16.stride(0),
        stride_Cm=C_fp16.stride(1),
        stride_Cn=C_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert result back to float32
    out = C_fp16.to(torch.float32)

    # Reshape back to (bs, N, N, hidden_dim)
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply final LayerNorm (to_out_norm) over hidden_dim
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection to dim
    out_flat = out.reshape(bs * N * N, hidden_dim)
    output_flat = torch.matmul(out_flat, to_out_weight.T)  # shape (bs*N*N, dim)
    output = output_flat.view(bs, N, N, dim)

    return output
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B where A, B, C are 3‑D tensors (batch, M/K, K/N).
    This is the core of the TriMul “einsum”:
        out[i,j] = Σ_k left[i,k] * right[j,k]
    The batch dimension encodes (batch * hidden_dim).
    """
    batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # coordinate offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers with batch offset
    a_ptrs = A_ptr + batch * stride_Ab + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
    b_ptrs = B_ptr + batch * stride_Bb + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # write output
    c_ptrs = C_ptr + batch * stride_Cb + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    Triton‑accelerated forward pass of the “outgoing” TriMul operation.

    Steps:
      1. Layer‑norm the input tensor.
      2. Linear projections to hidden_dim (left/right) and three sigmoidal gates.
      3. Apply mask (if any) and element‑wise gating.
      4. Compute the costly einsum via a custom batched‑matmul kernel.
      5. Apply a second LayerNorm, out‑gate and final linear projection.

    All heavy matrix‑multiply (step 4) is performed by the Triton kernel.
    """
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load model parameters (all tensors are already on the correct device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections & gates
    # ------------------------------------------------------------------
    # flatten for efficient matmul
    x_flat = x.reshape(bs * N * N, dim)                      # (B*N*N, dim)

    left = torch.matmul(x_flat, left_proj_weight.T).reshape(bs, N, N, hidden_dim)
    right = torch.matmul(x_flat, right_proj_weight.T).reshape(bs, N, N, hidden_dim)

    left_gate = torch.sigmoid(
        torch.matmul(x_flat, left_gate_weight.T).reshape(bs, N, N, hidden_dim)
    )
    right_gate = torch.sigmoid(
        torch.matmul(x_flat, right_gate_weight.T).reshape(bs, N, N, hidden_dim)
    )
    out_gate = torch.sigmoid(
        torch.matmul(x_flat, out_gate_weight.T).reshape(bs, N, N, hidden_dim)
    )

    # ------------------------------------------------------------------
    # 3) Mask and gating (fused element‑wise multiplication)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)       # (bs, N, N, 1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4) Core einsum via Triton batched matmul
    #    Compute: out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    We reshape (b,d) → batch dimension for the kernel.
    # ------------------------------------------------------------------
    # Move hidden dim to batch axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (bs, hidden_dim, N, N)

    batch_hidden = bs * hidden_dim
    A = left_perm.reshape(batch_hidden, N, N)          # (batch_h, M=N, K=N)
    B = right_perm.reshape(batch_hidden, N, N)         # (batch_h, N, K=N)

    # we need B^T for the GEMM
    B_T = B.transpose(1, 2).contiguous()               # (batch_h, K=N, N)

    # cast to half for bandwidth‑limited kernel
    A_fp16 = A.to(torch.float16)
    B_T_fp16 = B_T.to(torch.float16)

    # allocate output buffer (half precision)
    C_fp16 = torch.empty((batch_hidden, N, N), dtype=torch.float16, device=device)

    # Triton launch configuration
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    batched_matmul_kernel[grid](
        A_fp16, B_T_fp16, C_fp16,
        M=N, N=N, K=N,
        stride_Ab=A_fp16.stride(0),
        stride_Am=A_fp16.stride(1),
        stride_Ak=A_fp16.stride(2),
        stride_Bb=B_T_fp16.stride(0),
        stride_Bk=B_T_fp16.stride(1),
        stride_Bn=B_T_fp16.stride(2),
        stride_Cb=C_fp16.stride(0),
        stride_Cm=C_fp16.stride(1),
        stride_Cn=C_fp16.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert result back to FP32 and reshape to original layout
    out = C_fp16.to(torch.float32)
    out = out.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()   # (bs, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 5) Post‑processing: second LayerNorm, out‑gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate

    # final linear projection back to `dim`
    out_flat = out.reshape(bs * N * N, hidden_dim)
    output_flat = torch.matmul(out_flat, to_out_weight.T)                     # (bs*N*N, dim)
    output = output_flat.view(bs, N, N, dim)

    return output

"""
custom Triton‑accelerated forward pass for the “outgoing” Triangle‑Multiplicative
Update (TriMul) used in AlphaFold‑3 style models.

Algorithm
---------
1. Layer‑norm the input tensor.
2. Linear projections (left/right) and three sigmoidal gates (left, right, out).
3. Apply optional mask and the element‑wise gates (fusion of mask/gate).
4. Core O(N³·C) operation:
   out[b,i,j,c] = Σₖ left[b,i,k,c] * right[b,j,k,c]
   This is a batched matrix multiplication for every hidden channel.
   The implementation flattens (batch, hidden) → a single batch dimension and
   invokes a custom Triton kernel `batched_matmul_kernel` that computes
   C = A @ B where A,B,C are (batch, M, K)/(batch, K, N)/(batch, M, N).
5. Second Layer‑norm over the hidden dimension, multiplication by the out‑gate,
   and a final linear projection back to the original feature dimension.
Only step 4 is performed in Triton; the rest uses optimized PyTorch ops.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for 3‑D tensors (batch, M, K) × (batch, K, N).
    The kernel is the heavy part of TriMul:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    Here the *hidden* dimension is fused into the batch axis
    (batch_hidden = batch * hidden_dim).
    """
    batch = tl.program_id(axis=2)               # batch * hidden_idx
    pid_m = tl.program_id(axis=0)               # tile over M (i)
    pid_n = tl.program_id(axis=1)               # tile over N (j)

    # Offsets inside the tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Bounds checking
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers that include the batch offset
    a_ptrs = A_ptr + batch * stride_Ab + (offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak)
    b_ptrs = B_ptr + batch * stride_Bb + (offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn)

    # Accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)                      # (BLOCK_M, BLOCK_N)
        a_ptrs += BLOCK_K * stride_Ak
        b_ptrs += BLOCK_K * stride_Bk

    # Store output tile (in FP16 to save bandwidth)
    c_ptrs = C_ptr + batch * stride_Cb + (offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    out_flat = out.reshape(bs * N * N, hidden_dim)
    output_flat = torch.matmul(out_flat, to_out_weight.T)               # (B*N*N, dim)
    output = output_flat.view(bs, N, N, dim)

    return output
