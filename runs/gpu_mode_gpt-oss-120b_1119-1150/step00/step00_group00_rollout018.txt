@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, H, N,
    stride_left_b, stride_left_h, stride_left_i, stride_left_k,
    stride_right_b, stride_right_h, stride_right_j, stride_right_k,
    stride_out_b, stride_out_h, stride_out_i, stride_out_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute indices
    bh_id = tl.program_id(0)
    batch = bh_id // H
    hidden = bh_id % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # base pointers
    left_base = left_ptr + batch * stride_left_b + hidden * stride_left_h
    right_base = right_ptr + batch * stride_right_b + hidden * stride_right_h

        k_mask = k_offset < N

        acc += tl.dot(a, b)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # compute dims
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Norm
    x = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Masks
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()

    # Output tensor
    out = torch.empty_like(left)  # shape [B,N,N,hidden_dim]

    # Launch Triton kernel
    B, N, _, H = left.shape
    # Compute strides (int)
    stride_left_b = left.stride(0)
    stride_left_h = left.stride(3)
    stride_left_i = left.stride(1)
    stride_left_k = left.stride(2)
    stride_right_b = right.stride(0)
    stride_right_h = right.stride(3)
    stride_right_j = right.stride(1)  # dimension for j (i.e., second N)
    stride_right_k = right.stride(2)
    stride_out_b = out.stride(0)
    stride_out_h = out.stride(3)
    stride_out_i = out.stride(1)
    stride_out_j = out.stride(2)

    # Choose block sizes based on N
    # We'll set BLOCK_M = BLOCK_N = 64, BLOCK_K = 32. For small N maybe need to adjust.
    # But ensure block sizes not larger than N
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute grid dimensions
    grid = (B * H, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # kernel launch
    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_left_b=stride_left_b,
        stride_left_h=stride_left_h,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_right_b=stride_right_b,
        stride_right_h=stride_right_h,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_out_b=stride_out_b,
        stride_out_h=stride_out_h,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # After kernel, out shape [B,N,N,H]
    # Apply to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Apply out_gate
    out = out * out_gate

    # Final projection
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
TriMul (outgoing) forward implementation using Triton.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C}:
1. Layer‑norm over the last dimension C.
2. Two linear projections → left, right ∈ ℝ^{B×N×N×H}.
3. Optional binary mask (broadcasted on the hidden dimension).
4. Gated projections using sigmoid(Linear(X)).
5. Core “tri‑multiplicative” update:
        out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   which is a batch of H independent matrix multiplications:
        out_d = left_d @ right_dᵀ   with d∈[H].
   This heavy contraction (O(N³·H)) is performed with a custom
   Triton kernel that tiles the output (i, j) and the reduction
   dimension k.
6. Layer‑norm over H, gated by out_gate and a final linear
   projection back to dimension C.

Only the inner contraction (step 5) is executed on Triton; all
other ops use PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr,          # const * left  (B,N,N,H)
    right_ptr,         # const * right (B,N,N,H)
    out_ptr,           # * out (B,N,N,H)
    B: tl.int32,       # batch size
    H: tl.int32,       # hidden dimension
    N: tl.int32,       # sequence length
    # strides for left
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    # strides for right
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    # strides for out
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # each program handles a tile (BLOCK_M × BLOCK_N) of the output
    # for a specific (batch, hidden) pair.
    # ---------------------------------------------------------------
    bh_id = tl.program_id(0)                     # combined batch*hidden id
    batch = bh_id // H
    hidden = bh_id % H

    pid_m = tl.program_id(1)                     # tile row
    pid_n = tl.program_id(2)                     # tile col

    # Offsets for output rows/cols
    i_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # Base pointers for the selected batch and hidden slice
    left_base  = left_ptr  + batch * stride_l_b + hidden * stride_l_h
    right_base = right_ptr + batch * stride_r_b + hidden * stride_r_h

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # reduction over k (the third dimension of left / right)
    # ---------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_offset = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # left tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = left_base + i_offset[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None] & k_mask[None, :],
                    other=0.0)

        # right tile (transposed on the last two dimensions):
        # we need right[k, j] → shape (BLOCK_K, BLOCK_N)
        b_ptrs = right_base + k_offset[:, None] * stride_r_k + j_offset[None, :] * stride_r_j
        b = tl.load(b_ptrs,
                    mask=k_mask[:, None] & j_mask[None, :],
                    other=0.0)

        # Matrix multiply‑accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # ---------------------------------------------------------------
    # write the result back
    # ---------------------------------------------------------------
    c_ptrs = (out_ptr + batch * stride_o_b + hidden * stride_o_h +
              i_offset[:, None] * stride_o_i + j_offset[None, :] * stride_o_j)
    tl.store(c_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block (AlphaFold‑3).

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (float32/float16)
            - mask: torch.Tensor of shape [B, N, N] (bool or 0/1) – may be ignored.
            - weights: dict of torch.Tensor containing all model parameters.
            - config: dict with keys:
                * "dim" (C)
                * "hidden_dim" (H)
                * "nomask" (bool) – when True the mask can be skipped.

    Returns:
        torch.Tensor of shape [B, N, N, C] (same dtype/device as input).
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1️⃣ Layer‑norm over C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask could be bool or float; cast to same dtype as X
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating scalars (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # make contiguous for Triton pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, H = left.shape                     # left.shape == (B,N,N,H)

    # ------------------------------------------------------------------
    # 5️⃣ Core Tri‑multiplicative update via Triton
    # ------------------------------------------------------------------
    out = torch.empty_like(left)                # shape (B,N,N,H)

    # strides (int) – note that PyTorch gives strides in elements, not bytes
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_j = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    # Tile sizes – chosen to fit comfortably in registers on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                             # one program id per (batch, hidden) pair
        (N + BLOCK_M - 1) // BLOCK_M,      # rows
        (N + BLOCK_N - 1) // BLOCK_N,      # cols
    )

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        H=H,
        N=N,
        stride_l_b=stride_l_b,
        stride_l_h=stride_l_h,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_r_b=stride_r_b,
        stride_r_h=stride_r_h,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_o_b=stride_o_b,
        stride_o_h=stride_o_h,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # reasonable default for H100
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing (LayerNorm → gate → final linear)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
