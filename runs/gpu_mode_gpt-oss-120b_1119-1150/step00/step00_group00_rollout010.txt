    # backup features: extract dimensions.
    B, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    
    # Normalization
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = triton_layer_norm(input_tensor, norm_weight, norm_bias)  # maybe we do torch.nn.functional.layer_norm

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    right = torch.nn.functional.linear(x, right_proj_weight)

        right = right * mask_

    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()

    right = right * right_gate

Now design the triton kernel:

```
@triton.jit
def trimul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_p = tl.program_id(2)

    # recover batch and hidden indices
    # We assume H is hidden_dim.
    # Compute b and h from pid_p
    b = pid_p // H
    h = pid_p % H

    # compute offsets for i, j
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute masks for bounds

    left  → (B, H, I, K)   (I = i, K = k)
    right → (B, H, J, K)   (J = j, K = k)

For each (b, h) we compute `out[b,h] = left[b,h] @ right[b,h].T`,
i.e. a standard M×K times K×N multiplication with M = N = seq_len.
The kernel `trimul_matmul_kernel` tiles the M and N dimensions (i‑, j‑axes)
and reduces over K using `tl.dot`.  All other operations (LayerNorm,
Linear projections, gating, final LayerNorm and linear) are performed
with PyTorch (they are cheap compared to the N³ work).

The kernel works for any batch size B≤2, any sequence length N up to 1024,
and any hidden dimension H (here H = hidden_dim).  It runs on H100 GPUs
using Triton 3.3.1.
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul for the triple‑wise contraction
# --------------------------------------------------------------------------- #
@triton.jit
def trimul_matmul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    B, N, H,
    # left strides: (b, h, i, k)
    stride_l_b, stride_l_h, stride_l_i, stride_l_k,
    # right strides: (b, h, j, k)
    stride_r_b, stride_r_h, stride_r_j, stride_r_k,
    # out strides: (b, h, i, j)
    stride_o_b, stride_o_h, stride_o_i, stride_o_j,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b,h) pair:
        out[b,h,i,j] = Σ_k left[b,h,i,k] * right[b,h,j,k]
    using a standard tiled matrix multiplication.
    """
    pid_i = tl.program_id(0)   # block row   (i)
    pid_j = tl.program_id(1)   # block col   (j)
    pid_p = tl.program_id(2)   # flattened (b * H)

    # ------------------------------------------------------------------- #
    # Recover batch and hidden indices from pid_p
    # ------------------------------------------------------------------- #
    b = pid_p // H
    h = pid_p % H

    # ------------------------------------------------------------------- #
    # Compute start offsets for the current tile
    # ------------------------------------------------------------------- #
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointers for this (b,h) slice
    left_base  = left_ptr  + b * stride_l_b  + h * stride_l_h
    right_base = right_ptr + b * stride_r_b + h * stride_r_h
    out_base   = out_ptr   + b * stride_o_b + h * stride_o_h

    # Accumulator in FP32 (regardless of input dtype)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over the reduction dimension K
    # ------------------------------------------------------------------- #
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        offs_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left[i,k] tile   → shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_base + \
                    offs_i[:, None] * stride_l_i + \
                    offs_k[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0)

        # right[j,k] tile but we need its transpose (k,j) for dot:
        #   right[j,k] stored at stride_r_j (j) and stride_r_k (k)
        #   we load as (k,j) → shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_base + \
                     offs_k[:, None] * stride_r_k + \
                     offs_j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0)

        # Accumulate dot product: (M,K) @ (K,N) → (M,N)
        acc += tl.dot(left_tile, right_tile)

    # ------------------------------------------------------------------- #
    # Write results back to memory
    # ------------------------------------------------------------------- #
    out_ptrs = out_base + \
               offs_i[:, None] * stride_o_i + \
               offs_j[None, :] * stride_o_j
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------------------- #
# Public entry point required by the evaluation harness
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask         : torch.Tensor  [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensors containing all model parameters
        - config       : dict with keys 'dim', 'hidden_dim', 'nomask' (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as inputs).
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------- #
    # LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight) # [B,N,N,hidden]

    # ------------------------------------------------------------------- #
    # Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------- #
    if not nomask:
        # mask: [B, N, N] → [B, N, N, 1] → same dtype as tensors
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()   # [B,N,N,hidden]
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Prepare tensors for the batched matmul kernel:
    #   left  → (B, H, I, K)  where I=seq, K=seq (intermediate)
    #   right → (B, H, J, K)  where J=seq
    # ------------------------------------------------------------------- #
    # Permute to bring the hidden dimension in front.
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape  # H == hidden_dim

    # Allocate output buffer in the same layout as left_perm
    out_perm = torch.empty_like(left_perm)  # [B, H, N, N]

    # ------------------------------------------------------------------- #
    # Triton launch configuration
    # ------------------------------------------------------------------- #
    BLOCK_M = 64   # tile size for i‑axis
    BLOCK_N = 64   # tile size for j‑axis
    BLOCK_K = 64   # tile size for reduction axis (k)

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    num_blocks_p = B * H

    grid = (num_blocks_m, num_blocks_n, num_blocks_p)

    # Extract strides (in element counts, not bytes) for the kernel
    s_l_b, s_l_h, s_l_i, s_l_k = left_perm.stride()
    s_r_b, s_r_h, s_r_j, s_r_k = right_perm.stride()
    s_o_b, s_o_h, s_o_i, s_o_j = out_perm.stride()

    # Launch the kernel (synchronisation handled by Triton)
    trimul_matmul_kernel[grid](
        left_perm,
        right_perm,
        out_perm,
        B, N, H,
        s_l_b, s_l_h, s_l_i, s_l_k,
        s_r_b, s_r_h, s_r_j, s_r_k,
        s_o_b, s_o_h, s_o_i, s_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Bring the tensor back to the original layout: [B, N, N, H]
    # ------------------------------------------------------------------- #
    out = out_perm.permute(0, 2, 3, 1).contiguous()  # [B, N, N, hidden]

    # ------------------------------------------------------------------- #
    # Final LayerNorm over the hidden dimension
    # ------------------------------------------------------------------- #
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # [B, N, N, hidden]

    # ------------------------------------------------------------------- #
    # Gating after to_out_norm
    # ------------------------------------------------------------------- #
    out = out * out_gate  # element‑wise

    # ------------------------------------------------------------------- #
    # Final linear projection back to `dim`
    # ------------------------------------------------------------------- #
    to_out_weight = weights["to_out.weight"]  # (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # [B, N, N, dim]

    return out
```
