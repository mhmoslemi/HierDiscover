    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    B, N1, N2, C,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_y_b, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    epsilon: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # Compute base offset for x and y
    # Offsets across last dimension will be added later within loops
    # Compute pointers for each (b,i,j) element
    x_ptrs_base = x_ptr + pid_b * stride_x_b + pid_i * stride_x_i + pid_j * stride_x_j
    y_ptrs_base = y_ptr + pid_b * stride_y_b + pid_i * stride_y_i + pid_j * stride_y_j

    # compute mean and variance
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)
    # iterate over channel dimension
    for start_c in range(0, C, BLOCK_C):
        cur_block = tl.minimum(C - start_c, BLOCK_C)
        mask = tl.arange(0, BLOCK_C) < cur_block
        offs_c = start_c + tl.arange(0, BLOCK_C)
        x = tl.load(x_ptrs_base + offs_c * stride_x_c, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq += tl.sum(x * x, axis=0)
    mean = sum_val / C
    var = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # second loop store normalized values
    for start_c in range(0, C, BLOCK_C):
        cur_block = tl.minimum(C - start_c, BLOCK_C)
        mask = tl.arange(0, BLOCK_C) < cur_block

        w = tl.load(weight_ptr + offs_c * stride_w, mask=mask, other=0.0).to(tl.float32)
        b = tl.load(bias_ptr + offs_c * stride_b, mask=mask, other=0.0).to(tl.float32)
        y = (x - mean) * inv_std
        y = y * w + b
        tl.store(y_ptrs_base + offs_c * stride_y_c, y.to(tl.float32), mask=mask)

@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    B, N1, N2, C,
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_y_b, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    epsilon: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    x_ptrs = x_ptr + b*stride_x_b + i*stride_x_i + j*stride_x_j
    y_ptrs = y_ptr + b*stride_y_b + i*stride_y_i + j*stride_y_j

    # compute mean and var
    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    # loop over C
    for start_c in range(0, C, BLOCK_C):
        cur_block = tl.minimum(C - start_c, BLOCK_C)
        mask = tl.arange(0, BLOCK_C) < cur_block
        offs_c = start_c + tl.arange(0, BLOCK_C)
        # load
        x = tl.load(x_ptrs + offs_c * stride_x_c, mask=mask, other=0.0)
        x_f = x.to(tl.float32)
        sum_val += tl.sum(x_f, axis=0)
        sum_sq += tl.sum(x_f * x_f, axis=0)

    C_f = tl.float32(C)
    mean = sum_val / C_f
    var = sum_sq / C_f - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    for start_c in range(0, C, BLOCK_C):
        cur_block = tl.minimum(C - start_c, BLOCK_C)
        mask = tl.arange(0, BLOCK_C) < cur_block
        offs_c = start_c + tl.arange(0, BLOCK_C)
        x = tl.load(x_ptrs + offs_c * stride_x_c, mask=mask, other=0.0).to(tl.float32)
        w = tl.load(weight_ptr + offs_c * stride_w, mask=mask, other=0.0).to(tl.float32)

"""
TriMul (outgoing) forward pass with Triton acceleration.

Algorithm
---------

1. **Layer‑norm on the input**  
   A custom Triton kernel (`layernorm_kernel`) normalises the last dimension
   of the 4‑D tensor `[B, N, N, C]`.  The mean/variance are accumulated in
   FP32 while the data are stored in FP16.  The kernel writes the normalised
   tensor back to memory.

2. **Linear projections and gating**  
   The normalised tensor is projected to the hidden dimension (`H`) with three
   linear layers (left, right, and three gates).  All linear layers are
   bias‑free and implemented with `torch.nn.functional.linear` (highly
   optimised cuBLAS kernels).  Gate values are passed through a sigmoid.

3. **Masking** (if a mask is supplied) – the mask is broadcast to `[B,N,N,1]`
   and applied to the left/right tensors before gating.

4. **TriMul contraction**  

   The core operation is  

   ```
   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   ```

   which is equivalent to a batched matrix multiplication per hidden channel:
   `out_d = left_d @ right_dᵀ`.  By reshaping we obtain a single batched GEMM
   (`torch.bmm`) of shape `(B*H, N, N)`.  This leverages the highly tuned
   cuBLAS kernels on the H100.

5. **Post‑norm, output gate and final projection**  
   The result of the contraction is normalised again with the same Triton
   layer‑norm kernel (now `C = H`).  The output gate (produced in step 2) is
   applied element‑wise, and a final linear layer maps the hidden dimension
   back to the original `dim`.

All heavy arithmetic (the two layer‑norms and the mask‑gate multiplication)
runs on the GPU via Triton, while the large matrix multiplication uses the
state‑of‑the‑art cuBLAS implementation.  The final tensor is returned in
`torch.float32`.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def layernorm_kernel(
    x_ptr,               # *️⃣ input tensor   [B,N,N,C]
    y_ptr,               # *️⃣ output tensor  [B,N,N,C] (layer‑normed)
    weight_ptr,          # *️⃣ gamma          [C]
    bias_ptr,            # *️⃣ beta           [C]
    B, N1, N2, C,        # static sizes
    stride_x_b, stride_x_i, stride_x_j, stride_x_c,
    stride_y_b, stride_y_i, stride_y_j, stride_y_c,
    stride_w, stride_b,
    epsilon: tl.constexpr,
    BLOCK_C: tl.constexpr,
):
    """Layer‑norm over the last dimension (C).  Uses FP16 loads, FP32 reduction."""
    b = tl.program_id(0)   # batch axis
    i = tl.program_id(1)   # first sequence axis
    j = tl.program_id(2)   # second sequence axis

    # base pointers for this (b,i,j) element
    x_base = x_ptr + b * stride_x_b + i * stride_x_i + j * stride_x_j
    y_base = y_ptr + b * stride_y_b + i * stride_y_i + j * stride_y_j

    # ------------------------------------------------------------------
    # 1️⃣ Compute mean and variance (FP32 accumulator)
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    for cur_start in range(0, C, BLOCK_C):
        cur_end   = tl.minimum(C - cur_start, BLOCK_C)
        mask      = tl.arange(0, BLOCK_C) < cur_end
        offs_c    = cur_start + tl.arange(0, BLOCK_C)

        x = tl.load(x_base + offs_c * stride_x_c,
                    mask=mask,
                    other=0.0).to(tl.float32)

        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    C_f  = tl.float32(C)
    mean = sum_val / C_f
    var  = sum_sq / C_f - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ------------------------------------------------------------------
    # 2️⃣ Normalise, apply scale (weight) and shift (bias)
    # ------------------------------------------------------------------
    for cur_start in range(0, C, BLOCK_C):
        cur_end   = tl.minimum(C - cur_start, BLOCK_C)
        mask      = tl.arange(0, BLOCK_C) < cur_end
        offs_c    = cur_start + tl.arange(0, BLOCK_C)

        x = tl.load(x_base + offs_c * stride_x_c,
                    mask=mask,
                    other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + offs_c * stride_w,
                    mask=mask,
                    other=0.0).to(tl.float32)

        b_ = tl.load(bias_ptr + offs_c * stride_b,
                     mask=mask,
                     other=0.0).to(tl.float32)

        y = (x - mean) * inv_std
        y = y * w + b_

        tl.store(y_base + offs_c * stride_y_c,
                 y,
                 mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul (outgoing) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] in ``float32``.
    """
    # ------------------------------------------------------------------
    # 0️⃣ Unpack arguments
    # ------------------------------------------------------------------
    inp, mask, weights, config = data
    dim        = config["dim"]          # input feature dimension (C)
    hidden_dim = config["hidden_dim"]   # hidden dimension (H)

    # ------------------------------------------------------------------
    # 1️⃣ Cast to fp16 for speed (except where higher precision is needed)
    # ------------------------------------------------------------------
    x = inp.to(torch.float16)
    if mask is None:
        # no mask – use a tensor of ones
        mask = torch.ones(x.shape[0], x.shape[1], x.shape[2],
                          dtype=x.dtype, device=x.device)
    mask = mask.to(torch.float16).unsqueeze(-1)       # [B,N,N,1]

    # ------------------------------------------------------------------
    # 2️⃣ Layer‑norm on the input (custom Triton kernel)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(torch.float16)
    norm_bias   = weights["norm.bias"].to(torch.float16)

    B, N, _, _ = x.shape
    # allocate output tensor for the normalised input
    x_norm = torch.empty_like(x, dtype=torch.float16)

    # strides (PyTorch gives them as ints)
    sxb, sxi, sxj, sxc = x.stride()
    syb, syi, syj, syc = x_norm.stride()
    sw = norm_weight.stride(0)
    sb = norm_bias.stride(0)

    # launch Triton kernel: one program per (b,i,j)
    grid = (B, N, N)
    layernorm_kernel[grid](
        x, x_norm,
        norm_weight, norm_bias,
        B, N, N, dim,
        sxb, sxi, sxj, sxc,
        syb, syi, syj, syc,
        sw, sb,
        epsilon=1e-5,
        BLOCK_C=32,
    )

    # ------------------------------------------------------------------
    # 3️⃣ Linear projections + gating (using torch’s highly‑optimised kernels)
    # ------------------------------------------------------------------
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)   # (H, C)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)  # (H, C)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)   # (H, C)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)  # (H, C)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)    # (H, C)

    # projections (bias‑free)
    left  = F.linear(x_norm, left_proj_w)   # [B,N,N,H]
    right = F.linear(x_norm, right_proj_w)  # [B,N,N,H]

    # mask (broadcast over H)
    left  = left * mask
    right = right * mask

    # gates
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))  # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))    # [B,N,N,H]

    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4️⃣ TriMul contraction (batched GEMM)
    #   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # bring hidden dimension to the batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    left_mat  = left_perm.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_mat = right_perm.view(B * hidden_dim, N, N)   # (B*H, N, N)

    # batched mat‑mul over the K‑axis (N)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5️⃣ Post‑norm, output gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)  # (H,)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)    # (H,)

    # allocate tensor for the second layer‑norm
    out_norm = torch.empty_like(out, dtype=torch.float16)

    # kernel launch (reuse same grid)
    sxb, sxi, sxj, sxc = out.stride()
    syb, syi, syj, syc = out_norm.stride()
    sw = to_out_norm_w.stride(0)
    sb = to_out_norm_b.stride(0)

    layernorm_kernel[grid](
        out, out_norm,
        to_out_norm_w, to_out_norm_b,
        B, N, N, hidden_dim,
        sxb, sxi, sxj, sxc,
        syb, syi, syj, syc,
        sw, sb,
        epsilon=1e-5,
        BLOCK_C=32,
    )

    # apply output gate
    out_norm = out_norm * out_gate

    # final linear mapping back to `dim`
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, H)
    final = F.linear(out_norm, to_out_w)   # [B,N,N,dim]

    # return in FP32 as required by the reference implementation
    return final.to(torch.float32)
