import triton
import triton.language as tl

@triton.jit
def apply_mask_gate_kernel(
    left_ptr, right_ptr,
    mask_ptr,
    left_gate_ptr, right_gate_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    k0 = pid_k * BLOCK_K

    # indices
    i = i0 + tl.arange(0, BLOCK_I)[:, None, None]
    k = k0 + tl.arange(0, BLOCK_K)[None, :, None]
    d = tl.arange(0, BLOCK_D)[None, None, :]

    # bounds
    mask_i = i < N
    mask_k = k < N
    mask_d = d < H

    # load mask and broadcast to hidden dim
    mask = tl.load(mask_ptr + pid_b * stride_mask_b + i * stride_mask_i + k * stride_mask_k,
                   mask=mask_i & mask_k,
                   other=0.0)
    mask = mask[:, :, None]  # shape (BLOCK_I, BLOCK_K, 1)

    # load left and gates
    left = tl.load(left_ptr + pid_b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_h,
                   mask=mask_i & mask_k & mask_d,
                   other=0.0)

    left_gate = tl.load(left_gate_ptr + pid_b * stride_lgate_b + i * stride_lgate_i + k * stride_lgate_k + d * stride_lgate_h,
                        mask=mask_i & mask_k & mask_d,
                        other=0.0)

    # load right and gates
    right = tl.load(right_ptr + pid_b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_h,
                    mask=mask_i & mask_k & mask_d,
                    other=0.0)

    right_gate = tl.load(right_gate_ptr + pid_b * stride_rgate_b + i * stride_rgate_i + k * stride_rgate_k + d * stride_rgate_h,
                         mask=mask_i & mask_k & mask_d,
                         other=0.0)

    # apply mask and gates
    left_out = left * mask * left_gate
    right_out = right * mask * right_gate

    # store back in place
    tl.store(left_ptr + pid_b * stride_left_b + i * stride_left_i + k * stride_left_k + d * stride_left_h,
             left_out,
             mask=mask_i & mask_k & mask_d)

    tl.store(right_ptr + pid_b * stride_right_b + i * stride_right_i + k * stride_right_k + d * stride_right_h,
             right_out,
             mask=mask_i & mask_k & mask_d)

    inp, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = inp.device
    dtype = inp.dtype

    norm_bias = weights["norm.bias"]

    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # compute gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # apply mask and gates using triton
    if not nomask:
        # Ensure mask shape matches [B,N,N], unsqueeze later if needed.

    else:

    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()

    BLOCK_D = 32  # may need to adjust for hidden_dim

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input `x` over the last dimension.
2. Linear‑project the normalized tensor to the hidden dimension:
   * left  = x_norm @ left_proj_weight.T
   * right = x_norm @ right_proj_weight.T
3. Compute three sigmoid gates (left_gate, right_gate, out_gate) from the
   same normalized tensor using their own linear layers.
4. Fuse mask, left‑gate and right‑gate in a custom Triton kernel:
      left  = left  * mask * left_gate
      right = right * mask * right_gate
   (If `nomask=True` the mask is an all‑ones tensor – the kernel still works.)
5. Compute the core “triangle‑multiply”:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of `hidden_dim` matrix–matrix multiplications
   (left[b,d] @ right[b,d].T) and is performed with a batched `torch.bmm`,
   which on H100 uses Tensor‑Cores.
6. Apply a hidden‑dim layer‑norm, multiply by `out_gate`,
   and finally a linear projection back to `dim`.
7. Return the result of shape `[B, N, N, dim]`.

Only step 4 (mask + gate fusion) is written in Triton; the remaining
operations use highly‑optimised PyTorch kernels.  The kernel processes
tiles of size (BLOCK_I × BLOCK_K × BLOCK_D) and works for any
`B∈{1,2}`, `N∈{128,256,512,1024}` and `hidden_dim≤384`.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: in‑place mask + gate fusion for the left/right tensors.
# ----------------------------------------------------------------------
@triton.jit
def mask_gate_kernel(
    left_ptr,               # *Tensor*  [B,N,N,H]   (in‑place write)
    right_ptr,              # *Tensor*  [B,N,N,H]   (in‑place write)
    mask_ptr,               # *Tensor*  [B,N,N]  (float32/float16)
    left_gate_ptr,          # *Tensor*  [B,N,N,H]
    right_gate_ptr,         # *Tensor*  [B,N,N,H]
    # meta‑parameters
    B: tl.constexpr,
    N: tl.constexpr,
    H: tl.constexpr,
    # strides (in elements, NOT bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    # tiling
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Program IDs
    # ------------------------------------------------------------------
    pid_b = tl.program_id(0)          # batch dimension
    pid_i = tl.program_id(1)          # tile over first sequence axis
    pid_k = tl.program_id(2)          # tile over second sequence axis (the summed “k”)

    i0 = pid_i * BLOCK_I
    k0 = pid_k * BLOCK_K

    # ------------------------------------------------------------------
    # Index tensors for the current tile
    # ------------------------------------------------------------------
    i = i0 + tl.arange(0, BLOCK_I)[:, None, None]   # (BLOCK_I,1,1)
    k = k0 + tl.arange(0, BLOCK_K)[None, :, None]   # (1,BLOCK_K,1)
    d = tl.arange(0, BLOCK_D)[None, None, :]        # (1,1,BLOCK_D)

    # bounds
    mask_i = i < N
    mask_k = k < N
    mask_d = d < H

    # ------------------------------------------------------------------
    # Load mask and broadcast to the hidden dimension
    # ------------------------------------------------------------------
    mask = tl.load(mask_ptr
                   + pid_b * stride_mask_b
                   + i * stride_mask_i
                   + k * stride_mask_k,
                   mask=mask_i & mask_k,
                   other=0.0)                # (BLOCK_I, BLOCK_K)
    mask = mask[:, :, None]                         # (BLOCK_I, BLOCK_K, 1)

    # ------------------------------------------------------------------
    # Load tensors
    # ------------------------------------------------------------------
    left = tl.load(left_ptr
                   + pid_b * stride_left_b
                   + i * stride_left_i
                   + k * stride_left_k
                   + d * stride_left_h,
                   mask=mask_i & mask_k & mask_d,
                   other=0.0)

    left_gate = tl.load(left_gate_ptr
                        + pid_b * stride_lgate_b
                        + i * stride_lgate_i
                        + k * stride_lgate_k
                        + d * stride_lgate_h,
                        mask=mask_i & mask_k & mask_d,
                        other=0.0)

    right = tl.load(right_ptr
                    + pid_b * stride_right_b
                    + i * stride_right_i
                    + k * stride_right_k
                    + d * stride_right_h,
                    mask=mask_i & mask_k & mask_d,
                    other=0.0)

    right_gate = tl.load(right_gate_ptr
                         + pid_b * stride_rgate_b
                         + i * stride_rgate_i
                         + k * stride_rgate_k
                         + d * stride_rgate_h,
                         mask=mask_i & mask_k & mask_d,
                         other=0.0)

    # ------------------------------------------------------------------
    # Apply mask and element‑wise gates
    # ------------------------------------------------------------------
    left = left * mask * left_gate
    right = right * mask * right_gate

    # ------------------------------------------------------------------
    # Store the results back in‑place
    # ------------------------------------------------------------------
    tl.store(left_ptr
             + pid_b * stride_left_b
             + i * stride_left_i
             + k * stride_left_k
             + d * stride_left_h,
             left,
             mask=mask_i & mask_k & mask_d)

    tl.store(right_ptr
             + pid_b * stride_right_b
             + i * stride_right_i
             + k * stride_right_k
             + d * stride_right_h,
             right,
             mask=mask_i & mask_k & mask_d)


# ----------------------------------------------------------------------
# Entrypoint used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward implementation of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask + gates via Triton (in‑place on left/right)
    # ------------------------------------------------------------------
    if nomask:
        # no mask → treat as all‑ones; skip the Triton kernel and just apply gates
        left = left * left_gate
        right = right * right_gate
    else:
        # ensure mask is float (0/1) and on the same device/dtype
        mask = mask_tensor.to(dtype=dtype, device=device)
        # make tensors contiguous for pointer arithmetic
        left = left.contiguous()
        right = right.contiguous()
        left_gate = left_gate.contiguous()
        right_gate = right_gate.contiguous()
        mask = mask.contiguous()

        # strides (in elements)
        stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
        stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
        stride_mask_b, stride_mask_i, stride_mask_k = mask.stride()
        stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h = left_gate.stride()
        stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h = right_gate.stride()

        B, N, _, _ = left.shape

        # tile sizes – chosen to give good occupancy on H100
        BLOCK_I = 64   # tile along the first sequence axis
        BLOCK_K = 64   # tile along the summed axis
        BLOCK_D = 32   # tile along the hidden dimension (fits in registers)

        grid = (
            B,
            triton.cdiv(N, BLOCK_I),
            triton.cdiv(N, BLOCK_K),
        )

        mask_gate_kernel[grid](
            left_ptr=left,
            right_ptr=right,
            mask_ptr=mask,
            left_gate_ptr=left_gate,
            right_gate_ptr=right_gate,
            B=B,
            N=N,
            H=hidden_dim,
            stride_left_b=stride_left_b,
            stride_left_i=stride_left_i,
            stride_left_k=stride_left_k,
            stride_left_h=stride_left_h,
            stride_right_b=stride_right_b,
            stride_right_i=stride_right_i,
            stride_right_k=stride_right_k,
            stride_right_h=stride_right_h,
            stride_mask_b=stride_mask_b,
            stride_mask_i=stride_mask_i,
            stride_mask_k=stride_mask_k,
            stride_lgate_b=stride_lgate_b,
            stride_lgate_i=stride_lgate_i,
            stride_lgate_k=stride_lgate_k,
            stride_lgate_h=stride_lgate_h,
            stride_rgate_b=stride_rgate_b,
            stride_rgate_i=stride_rgate_i,
            stride_rgate_k=stride_rgate_k,
            stride_rgate_h=stride_rgate_h,
            BLOCK_I=BLOCK_I,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )
        # Synchronise to make sure the kernel has finished before the next
        # PyTorch operation accesses the tensors.
        torch.cuda.synchronize()

    # ------------------------------------------------------------------
    # 5) Triangle‑multiply (batched GEMM over the hidden dimension)
    # ------------------------------------------------------------------
    # left/right now have shape [B, N, N, H]
    # reshape to (B, H, N, N) -> (B*H, N, N) for a single batched bmm
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # out_flat[b*H + h, i, j] = Σ_k left_flat[b*H+h, i, k] * right_flat[b*H+h, j, k]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout -> [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm + out‑gate
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    # Ensure output dtype is float32 as required by the spec
    return output.to(dtype=torch.float32)
