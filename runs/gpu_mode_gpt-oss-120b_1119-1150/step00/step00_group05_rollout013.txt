```
def fused_mul_mask_gate(proj_ptr, gate_ptr, mask_ptr, out_ptr, hidden_dim, total_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE

    mask_offsets = offsets // hidden_dim  # integer division

    proj = tl.load(proj_ptr + offsets, mask=within_bounds, other=0.0)
```

```
def elem_mul_kernel(a_ptr, b_ptr, c_ptr, total_elements, BLOCK_SIZE: tl.constexpr):

@triton.jit
def triple_mul_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr, hidden_dim, total_elements, BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)

    mask_val = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)
```

```
def custom_kernel(data):
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm for x (norm)
    # Use torch.nn.functional.layer_norm
    weight = weights["norm.weight"]
    bias = weights["norm.bias"]
    # shape: (...)
    x_norm = F.layer_norm(x, normalized_shape=(dim,), weight=weight, bias=bias, eps=eps)

    # Linear projections -> shape [bs,N,N,hidden_dim]
    left_proj = F.linear(x_norm, weights["left_proj.weight"])
    right_proj = F.linear(x_norm, weights["right_proj.weight"])

    # Gating preactivations
    left_gate_pre = F.linear(x_norm, weights["left_gate.weight"])
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"])
    out_gate_pre = F.linear(x_norm, weights["out_gate.weight"])

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # Apply mask & gates using Triton kernel
    device = x.device
    mask = mask.to(device).float()
    # shape mask: [bs,N,N]; we need broadcast using triple mul
    # ensure mask is contiguous
    mask = mask.contiguous()
    # allocate left and right
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_el = left_proj.numel()
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total_el, BLOCK),)

    # Use triple_mul_kernel if mask exists (i.e., config["nomask"] is False)
    if config.get("nomask", False) or mask.numel() == 0:
        # No mask, elementwise multiply proj * gate
        elem_mul_kernel[grid](
            a_ptr=left_proj, b_ptr=left_gate, c_ptr=left,
            total_elements=total_el, BLOCK_SIZE=BLOCK
        )
        elem_mul_kernel[grid](
            a_ptr=right_proj, b_ptr=right_gate, c_ptr=right,
            total_elements=total_el, BLOCK_SIZE=BLOCK
        )
    else:
        # mask broadcast across hidden_dim
        hidden_dim_val = left_proj.shape[-1]
        triple_mul_kernel[grid](
            proj_ptr=left_proj, gate_ptr=left_gate, mask_ptr=mask,
            out_ptr=left,
            hidden_dim=hidden_dim_val,
            total_elements=total_el,
            BLOCK_SIZE=BLOCK
        )
        triple_mul_kernel[grid](
            proj_ptr=right_proj, gate_ptr=right_gate, mask_ptr=mask,
            out_ptr=right,
            hidden_dim=hidden_dim_val,
            total_elements=total_el,
            BLOCK_SIZE=BLOCK
        )

    # Compute out = left @ right^T for each hidden_dim channel via batched GEMM
    # Permute to (bs, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N_seq, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N_seq, N_seq)
    right_flat = right_perm.view(B * H, N_seq, N_seq)
    # Batched matmul: left_flat @ right_flat.transpose(1,2)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N_seq, N_seq).permute(0, 2, 3, 1)  # shape [B,N,N,H]

    # Apply to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate (elementwise) using Triton
    # out_gate shape [B,N,N,hidden_dim]
    gated_out = torch.empty_like(out_norm)
    total_out_el = out_norm.numel()
    grid_out = lambda meta: (triton.cdiv(total_out_el, BLOCK),)
    elem_mul_kernel[grid_out](
        a_ptr=out_norm, b_ptr=out_gate, c_ptr=gated_out,
        total_elements=total_out_el, BLOCK_SIZE=BLOCK
    )
    # Final linear projection to dim
    to_out_weight = weights["to_out.weight"]
    output = F.linear(gated_out, to_out_weight)

    return output
```

"""
TriMul (outgoing) forward pass – Triton‑accelerated implementation.

The module processes a 4‑D tensor `x` of shape [B, N, N, D] and produces an
output of shape [B, N, N, D].  The heavy work consists of three parts:

1. Layer‑norm over the last dimension (`dim`).
2. Linear projections + gating + optional masking.  The element‑wise product
   (projection * gate * mask) is performed with a small Triton kernel.
3. Triangular multiplicative update:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   which is a batched matrix‑multiply `left @ rightᵀ` for each channel `h`.
   This is executed with PyTorch's batched GEMM (highly tuned on H100).

After the update we apply a second layer‑norm, gating, and a final linear
projection back to the original `dim`.  Two Triton kernels are used:
    * `triple_mul_kernel` – fuses `proj * gate * mask` (mask is broadcast over
      the hidden dimension).
    * `elem_mul_kernel`   – generic element‑wise product (used for the final
      gating step).

The implementation works for the full range of shapes required by the
evaluation harness (B∈{1,2}, N up to 1024, dim∈{128,384},
hidden_dim=128) and respects the `nomask` flag.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernels
# -------------------------------------------------------------------------

@triton.jit
def elem_mul_kernel(a_ptr, b_ptr, c_ptr,
                    total_elements,               # runtime argument
                    BLOCK_SIZE: tl.constexpr):   # compile‑time constant
    """c = a * b   (element‑wise)"""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offsets < total_elements

    a = tl.load(a_ptr + offsets, mask=mask, other=0.0)
    b = tl.load(b_ptr + offsets, mask=mask, other=0.0)
    tl.store(c_ptr + offsets, a * b, mask=mask)


@triton.jit
def triple_mul_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                      hidden_dim,                 # compile‑time (constexpr)
                      total_elements,            # runtime
                      BLOCK_SIZE: tl.constexpr):
    """out = proj * gate * mask   (mask broadcast over hidden_dim)"""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE, dtype=tl.int64)
    mask = offsets < total_elements

    proj = tl.load(proj_ptr + offsets, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offsets, mask=mask, other=0.0)

    # mask has shape (B, N, N) → flattened size B*N*N.
    # Each hidden_dim consecutive entries share the same mask value.
    mask_index = offsets // hidden_dim
    mask_val = tl.load(mask_ptr + mask_index, mask=mask, other=0.0)

    tl.store(out_ptr + offsets, proj * gate * mask_val, mask=mask)


# -------------------------------------------------------------------------
# Custom kernel entry point
# -------------------------------------------------------------------------

def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Args:
        data: Tuple containing
            - x:       torch.Tensor of shape [B, N, N, dim] (float32)
            - mask:    torch.Tensor of shape [B, N, N] (float32 or bool). May be
                       ignored when config["nomask"] is True.
            - weights: dict with the following keys (all torch.Tensors on the
                       same device as `x`):
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config: dict with keys
                * "dim"        – input feature dimension D
                * "hidden_dim" – hidden dimension H (typically 128)
                * "nomask"     – bool, True → ignore the mask
                * optional "eps" for layer‑norm (default 1e-5)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    # -----------------------------------------------------------------
    # 1️⃣  Layer‑norm over the last dimension (dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"].to(dtype).to(device)
    norm_bias   = weights["norm.bias"].to(dtype).to(device)
    x_norm = F.layer_norm(x, normalized_shape=(dim,),
                          weight=norm_weight, bias=norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) and gating pre‑activations
    # -----------------------------------------------------------------
    # NOTE: all linear layers are bias‑free as per the reference implementation
    left_proj   = F.linear(x_norm, weights["left_proj.weight"].to(dtype).to(device))
    right_proj  = F.linear(x_norm, weights["right_proj.weight"].to(dtype).to(device))

    left_gate_pre  = F.linear(x_norm, weights["left_gate.weight"].to(dtype).to(device))
    right_gate_pre = F.linear(x_norm, weights["right_gate.weight"].to(dtype).to(device))
    out_gate_pre   = F.linear(x_norm, weights["out_gate.weight"].to(dtype).to(device))

    left_gate  = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate   = torch.sigmoid(out_gate_pre)               # shape [B, N, N, hidden_dim]

    # -----------------------------------------------------------------
    # 3️⃣  Apply mask (if any) and combine projection with gates.
    #     Uses Triton to fuse the three element‑wise multiplications.
    # -----------------------------------------------------------------
    use_mask = not config.get("nomask", False) and mask is not None
    mask_f = None
    if use_mask:
        mask_f = mask.to(dtype).to(device)                 # [B, N, N]
        mask_f = mask_f.contiguous()                       # ensure 1‑D layout after view()
    else:
        # dummy mask of ones – the triple kernel will multiply by 1,
        # equivalent to skipping the mask.
        mask_f = torch.ones((B, N, N), dtype=dtype, device=device)

    # Allocate containers for the gated & masked tensors
    left  = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    total_elem = left_proj.numel()          # B * N * N * hidden_dim
    BLOCK = 1024

    # Triton launch configuration
    grid = lambda meta: (triton.cdiv(total_elem, BLOCK),)

    # Fuse (proj * gate * mask) for both left and right branches
    triple_mul_kernel[grid](
        proj_ptr=left_proj,
        gate_ptr=left_gate,
        mask_ptr=mask_f,
        out_ptr=left,
        hidden_dim=hidden_dim,
        total_elements=total_elem,
        BLOCK_SIZE=BLOCK,
    )
    triple_mul_kernel[grid](
        proj_ptr=right_proj,
        gate_ptr=right_gate,
        mask_ptr=mask_f,
        out_ptr=right,
        hidden_dim=hidden_dim,
        total_elements=total_elem,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # 4️⃣  Triangular multiplicative update: out = left @ rightᵀ
    # -----------------------------------------------------------------
    # Rearrange to (B, hidden_dim, N, N) → batch‑gemm over N×N
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    BH, H, N_, _ = left_perm.shape  # BH = B (same as B) after reshape
    left_flat  = left_perm.view(B * hidden_dim, N, N)    # [B*H, N, N]
    right_flat = right_perm.view(B * hidden_dim, N, N)   # [B*H, N, N]

    # Batched matrix multiplication: (B*H, N, N) × (B*H, N, N)ᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # -----------------------------------------------------------------
    # 5️⃣  Second Layer‑norm over hidden_dim
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"].to(dtype).to(device)
    to_out_norm_bias   = weights["to_out_norm.bias"].to(dtype).to(device)

    out_norm = F.layer_norm(out, normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias, eps=eps)

    # -----------------------------------------------------------------
    # 6️⃣  Apply final gate (out_gate) – again via a small Triton kernel
    # -----------------------------------------------------------------
    gated_out = torch.empty_like(out_norm)
    total_out_elem = out_norm.numel()
    grid_out = lambda meta: (triton.cdiv(total_out_elem, BLOCK),)

    elem_mul_kernel[grid_out](
        a_ptr=out_norm,
        b_ptr=out_gate,
        c_ptr=gated_out,
        total_elements=total_out_elem,
        BLOCK_SIZE=BLOCK,
    )

    # -----------------------------------------------------------------
    # 7️⃣  Final linear projection back to `dim`
    # -----------------------------------------------------------------
    to_out_weight = weights["to_out.weight"].to(dtype).to(device)  # [dim, hidden_dim]
    output = F.linear(gated_out, to_out_weight)   # [B, N, N, dim]

    return output
