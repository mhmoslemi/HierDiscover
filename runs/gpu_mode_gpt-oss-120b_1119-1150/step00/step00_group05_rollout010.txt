@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute which batch + block indices
    total_blocks = batch * tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    # need to map pid to batch, block_i, block_j
    batch_id = pid // (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    rem = pid % (tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N))
    block_i = rem // tl.cdiv(N, BLOCK_N)
    block_j = rem % tl.cdiv(N, BLOCK_N)

    # block offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute pointer offsets for a (starting at row offs_m) and b (col offs_n)
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + tl.arange(0, BLOCK_K)[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # accum
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32) # using float32 accumulation
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for this k block
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (k * BLOCK_K + tl.arange(0, BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0, BLOCK_K) < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        acc += tl.dot(a, b)
        # move pointers for next k block:
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b
    # store result
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])

def batched_matmul(a,b):
    # a: [B, M, K]; b: [B, K, N]; both dtype torch.float16
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    c = torch.empty((batch, M, N), dtype=torch.float16, device=a.device)
    # compute grid size: number of programs = batch * ceil(M/BLOCK_M) * ceil(N/BLOCK_N)
    grid = (batch * ((M + BLOCK_M - 1)//BLOCK_M) * ((N + BLOCK_N - 1)//BLOCK_N),)
    batched_matmul_kernel[grid](

        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)
    # Number of blocks per dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    # Total number of program instances needed: batch * num_blocks_m * num_blocks_n
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # Offsets
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for valid rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + tl.arange(0, BLOCK_K)[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Using masks
        cur_k = k * BLOCK_K
        cur_K = tl.minimum(BLOCK_K, K - cur_k)
        # Load blocks with mask
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K) < cur_K), other=0.0)
        b = tl.load(b_ptrs, mask=(tl.arange(0, BLOCK_K)[:, None] < cur_K) & mask_n[None, :], other=0.0)
        # Multiply-accumulate
        acc += tl.dot(a, b)
        # Increment pointers
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b
    # Write back
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_triton(a: torch.Tensor, b: torch.Tensor, block_m=64, block_n=64, block_k=64):
    assert a.ndim == 3 and b.ndim == 3
    assert a.shape[0] == b.shape[0]
    batch, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2
    # ensure contiguous
    a = a.contiguous()
    b = b.contiguous()
    c = torch.empty((batch, M, N), dtype=a.dtype, device=a.device)
    # Compute grid:
    num_blocks_m = (M + block_m - 1)//block_m
    num_blocks_n = (N + block_n - 1)//block_n
    total_blocks = batch * num_blocks_m * num_blocks_n
    grid = (total_blocks,)
    # launch kernel
    batched_matmul_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        batch=batch,
        M=M,
        N=N,
        K=K,
        stride_batch_a=a.stride(0),
        stride_m_a=a.stride(1),
        stride_k_a=a.stride(2),
        stride_batch_b=b.stride(0),
        stride_k_b=b.stride(1),
        stride_n_b=b.stride(2),
        stride_batch_c=c.stride(0),
        stride_m_c=c.stride(1),
        stride_n_c=c.stride(2),
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

def custom_kernel(data):
    """
    Triton-accelerated TriMul (outgoing) forward pass.
    ...
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data

    # hyper params
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights['norm.weight'],
        bias=weights['norm.bias'],
        eps=eps,
    )

    # Linear projections: use torch.nn.functional.linear
    # weight shape: (hidden_dim, dim)
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])

    # Apply mask if not nomask
    if mask_tensor is not None:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [bs,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for matmul
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute & reshape for batched matmul
    bs, N, _, _ = left_fp16.shape
    # left_perm: [bs, hidden_dim, N, N]
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    # right_t: transpose second two dims (k,j)
    right_t = right_fp16.permute(0, 3, 2, 1).contiguous()

    # flatten batch*hidden_dim
    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_t.view(bs * hidden_dim, N, N)

    # Perform batched matmul with Triton
    out_flat = batched_matmul_triton(left_flat, right_flat, block_m=64, block_n=64, block_k=64)

    # Reshape back
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # [bs, N, N, hidden_dim]

    # Convert back to float32 for LayerNorm
    out = out.to(torch.float32)

    # to_out_norm (LayerNorm over hidden_dim)
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights['to_out_norm.weight'],
        bias=weights['to_out_norm.bias'],
        eps=eps,
    )

    # Apply out_gate
    out = out_norm * out_gate

    # Final linear
    out = torch.nn.functional.linear(out, weights['to_out.weight'])

    return out

"""
custom_kernel – Triton‑accelerated “outgoing” TriMul (AlphaFold‑3)

Algorithm
---------
1. Layer‑norm over the last dimension (dim) using the provided
   `norm.weight` / `norm.bias`.
2. Two linear projections (`left_proj` and `right_proj`) from dim → hidden_dim.
3. Optional element‑wise mask (shape [B,N,N,1]) applied to both projections.
4. Three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) obtained
   with linear layers (bias‑free) followed by `sigmoid`.
5. Apply the left/right gates element‑wise.
6. Core operation – the “outgoing” triangular multiplicative update:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiply for every hidden channel:
        out = left @ rightᵀ   (batch = B·hidden_dim, M=N, K=N, N=N)
   Implemented with a custom Triton kernel that works on FP16 inputs
   and accumulates in FP32.
7. Layer‑norm over the hidden dimension (`to_out_norm`) and element‑wise
   multiplication with the `out_gate`.
8. Final linear projection `to_out` (hidden_dim → dim) producing the
   result tensor of shape [B, N, N, dim] (float32).

The kernel fuses the heavy N³ work (step 6) and dramatically reduces
runtime compared with a naïve PyTorch implementation.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


# ----------------------------------------------------------------------
# Triton batched GEMM kernel (A @ Bᵀ) used for the TriMul einsum.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch, M, N, K,
    stride_batch_a, stride_m_a, stride_k_a,
    stride_batch_b, stride_k_b, stride_n_b,
    stride_batch_c, stride_m_c, stride_n_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)

    # number of blocks per matrix dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # map pid -> (batch_id, block_i, block_j)
    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    block_i = block_id // num_blocks_n
    block_j = block_id % num_blocks_n

    # block start indices
    offs_m = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for OOB rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the start of the current block
    a_ptrs = a_ptr + batch_id * stride_batch_a + offs_m[:, None] * stride_m_a + tl.arange(0, BLOCK_K)[None, :] * stride_k_a
    b_ptrs = b_ptr + batch_id * stride_batch_b + tl.arange(0, BLOCK_K)[:, None] * stride_k_b + offs_n[None, :] * stride_n_b

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        # remaining length in K for this block (handles tail)
        rem_k = K - cur_k
        mask_k = tl.arange(0, BLOCK_K) < rem_k

        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)          # (BLOCK_M, BLOCK_K)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)          # (BLOCK_K, BLOCK_N)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

        # advance pointers to the next K‑tile
        a_ptrs += BLOCK_K * stride_k_a
        b_ptrs += BLOCK_K * stride_k_b

    # write back result
    c_ptrs = c_ptr + batch_id * stride_batch_c + offs_m[:, None] * stride_m_c + offs_n[None, :] * stride_n_c
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor,
                           B: torch.Tensor,
                           BLOCK_M: int = 64,
                           BLOCK_N: int = 64,
                           BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matrix multiplication C = A @ Bᵀ where each batched matrix has shape (M, K)
    and (K, N) respectively. A, B, C are expected to be contiguous and in FP16.
    """
    assert A.ndim == 3 and B.ndim == 3
    assert A.shape[0] == B.shape[0]
    batch, M, K = A.shape
    _, Kb, N = B.shape
    assert K == Kb

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in element units (not bytes)
    stride_batch_a, stride_m_a, stride_k_a = A.stride()
    stride_batch_b, stride_k_b, stride_n_b = B.stride()
    stride_batch_c, stride_m_c, stride_n_c = C.stride()

    # launch configuration
    grid_m = (M + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch * grid_m * grid_n
    grid = (total_blocks,)

    _batched_matmul_kernel[grid](
        A,
        B,
        C,
        batch,
        M,
        N,
        K,
        stride_batch_a,
        stride_m_a,
        stride_k_a,
        stride_batch_b,
        stride_k_b,
        stride_n_b,
        stride_batch_c,
        stride_m_c,
        stride_n_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Triton‑accelerated TriMul forward (outgoing) entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul module.

    Args
    ----
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim], float32
        - mask_tensor  : torch.Tensor, shape [B, N, N] (bool/int/float)
        - weights_dict : mapping of weight names to torch.Tensor objects
        - config_dict  : contains at least ``dim`` and ``hidden_dim``;
                         may contain ``nomask`` (True/False).

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # 2. Linear projections to hidden_dim (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Apply mask if present
    # ------------------------------------------------------------------
    if not nomask:
        # mask is broadcastable to the last dimension
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Compute sigmoid gates
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 5. Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. Core TriMul – batched gemm left @ rightᵀ
    #    Cast to FP16 for the gemm, keep accumulators in FP32.
    # ------------------------------------------------------------------
    # Cast to half‑precision (the kernel expects FP16)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Permute to bring the hidden dimension into the batch axis:
    #   left_perm : [B, hidden_dim, N, N]  (i,k)
    #   right_t   : [B, hidden_dim, N, N]  (k,j)   <-- transpose of (j,k)
    left_perm = left_fp16.permute(0, 3, 1, 2).contiguous()
    right_t = right_fp16.permute(0, 3, 2, 1).contiguous()   # transpose k ↔ j

    B, H, N, _ = left_perm.shape
    # Merge batch and hidden dimensions → [B*H, N, N]
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    # Batched matmul: each slice computes Σₖ left[i,k] * right[j,k]
    out_flat_fp16 = _batched_matmul_triton(left_flat, right_flat,
                                           BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # Restore the original layout: [B, N, N, hidden_dim]
    out = out_flat_fp16.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)   # back to fp32 for the following layers

    # ------------------------------------------------------------------
    # 7. LayerNorm over the hidden dimension + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------
    # 8. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return output
