@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):

    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Axes ranges
    i_range = tl.arange(0, BLOCK_I)
    j_range = tl.arange(0, BLOCK_J)

    # Mask for i, j boundaries
    i_valid = i_start + i_range < N
    j_valid = j_start + j_range < N

    for d_start in range(0, H, BLOCK_D):
        # accumulators per (i,j,d)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)
        d_range = tl.arange(0, BLOCK_D)
        d_valid = d_start + d_range < H

        for k_start in range(0, N, BLOCK_K):
            k_range = tl.arange(0, BLOCK_K)
            k_valid = k_start + k_range < N

            # Left tile (i, k, d)
            left_ptrs = left_ptr + \
                pid_b * stride_l_b + \
                (i_start + i_range)[:, None, None] * stride_l_i + \
                (k_start + k_range)[None, :, None] * stride_l_k + \
                (d_start + d_range)[None, None, :] * stride_l_h
            mask_left = i_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :]
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)

            # Right tile transposed (k, j, d)
            right_ptrs = right_ptr + \
                pid_b * stride_r_b + \
                (k_start + k_range)[:, None, None] * stride_r_k + \
                (j_start + j_range)[None, :, None] * stride_r_j + \
                (d_start + d_range)[None, None, :] * stride_r_h
            mask_right = k_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :]
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)

            # accumulate per d slice
            for d_offset in range(BLOCK_D):
                # check if this d is valid
                if not d_valid[d_offset]:
                    continue
                left_slice = left_tile[:, :, d_offset]    # (BLOCK_I, BLOCK_K)
                right_slice = right_tile[:, :, d_offset]  # (BLOCK_K, BLOCK_J)
                acc[:, :, d_offset] += tl.dot(left_slice, right_slice)

        # Store the result for this d block
        out_ptrs = out_ptr + \
            pid_b * stride_o_b + \
            (i_start + i_range)[:, None, None] * stride_o_i + \
            (j_start + j_range)[None, :, None] * stride_o_j + \
            (d_start + d_range)[None, None, :] * stride_o_h
        mask_out = i_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)

"""
TriMul “outgoing” implementation (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the 4‑D input tensor over the last dimension (dim).
2. Linear projections `left` and `right` (dim → hidden_dim) without bias.
3. Optional pair‑wise mask (broadcasted on the hidden dim) is applied.
4. Compute three gating vectors (sigmoid) from the normalized input:
   * left_gate   – multiplies `left`
   * right_gate  – multiplies `right`
   * out_gate    – applied after the inner update.
5. Core operation:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication for every hidden channel *d*.
   It is realised with a custom Triton kernel that tiles the (i,j) plane
   (BLOCK_I × BLOCK_J) and the hidden dimension (BLOCK_D) while iterating
   over *k* in blocks (BLOCK_K).  The kernel accumulates a (BLOCK_I, BLOCK_J,
   BLOCK_D) tensor using `tl.dot` on each hidden‑slice.
6. Layer‑norm the intermediate result over the hidden dimension,
   multiply by `out_gate`, and apply the final linear projection back to
   the original `dim`.

Only the O(N³·hidden_dim) reduction (step 5) is performed in Triton;
all other operations use PyTorch functional primitives.
"""

import torch
import triton
import triton.language as tl


# -----------------------------------------------------------------------
# Triton kernel for the outgoing triangular multiplicative update
# -----------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # [B, N, N, H]  (i, k, d)
    right_ptr,              # [B, N, N, H]  (j, k, d) – we will read transposed as (k, j, d)
    out_ptr,                # [B, N, N, H]  (i, j, d)

    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,

    BLOCK_I: tl.constexpr,  # tile size for output rows (i)
    BLOCK_J: tl.constexpr,  # tile size for output cols (j)
    BLOCK_K: tl.constexpr,  # tile size for reduction dimension (k)
    BLOCK_D: tl.constexpr,  # tile size for hidden dimension (d)
):
    pid_b = tl.program_id(0)                # batch index
    pid_i = tl.program_id(1)                # i‑tile index
    pid_j = tl.program_id(2)                # j‑tile index

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # -------------------------------------------------------------------
    # 1‑D range vectors (compile‑time length = block size)
    # -------------------------------------------------------------------
    i_range = tl.arange(0, BLOCK_I)
    j_range = tl.arange(0, BLOCK_J)
    d_range = tl.arange(0, BLOCK_D)
    k_range = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    i_valid = i_start + i_range < N
    j_valid = j_start + j_range < N

    # -------------------------------------------------------------------
    # Loop over hidden‑dim blocks
    # -------------------------------------------------------------------
    for d_start in range(0, H, BLOCK_D):
        d_valid = d_start + d_range < H
        # accumulator: (BLOCK_I, BLOCK_J, BLOCK_D)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # ----------------------------------------------------------------
        # Reduction over the sequence dimension k
        # ----------------------------------------------------------------
        for k_start in range(0, N, BLOCK_K):
            k_valid = k_start + k_range < N

            # ------------------------------------------------------------
            # Load left tile  : shape (BLOCK_I, BLOCK_K, BLOCK_D)
            #   indexes = (b, i, k, d)
            # ------------------------------------------------------------
            left_ptrs = left_ptr + \
                pid_b * stride_l_b + \
                (i_start + i_range)[:, None, None] * stride_l_i + \
                (k_start + k_range)[None, :, None] * stride_l_k + \
                (d_start + d_range)[None, None, :] * stride_l_h
            mask_left = i_valid[:, None, None] & k_valid[None, :, None] & d_valid[None, None, :]
            left_tile = tl.load(left_ptrs, mask=mask_left, other=0.0)   # (I, K, D)

            # ------------------------------------------------------------
            # Load right tile transposed : shape (BLOCK_K, BLOCK_J, BLOCK_D)
            #   indexes = (b, k, j, d)
            # ------------------------------------------------------------
            right_ptrs = right_ptr + \
                pid_b * stride_r_b + \
                (k_start + k_range)[:, None, None] * stride_r_k + \
                (j_start + j_range)[None, :, None] * stride_r_j + \
                (d_start + d_range)[None, None, :] * stride_r_h
            mask_right = k_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :]
            right_tile = tl.load(right_ptrs, mask=mask_right, other=0.0)   # (K, J, D)

            # ------------------------------------------------------------
            # Accumulate dot products for every hidden slice
            # ------------------------------------------------------------
            for d_off in range(BLOCK_D):
                # left_slice : (BLOCK_I, BLOCK_K)
                # right_slice: (BLOCK_K, BLOCK_J)
                left_slice  = left_tile[:, :, d_off]
                right_slice = right_tile[:, :, d_off]
                acc[:, :, d_off] += tl.dot(left_slice, right_slice)

        # ----------------------------------------------------------------
        # Store the completed block
        # ----------------------------------------------------------------
        out_ptrs = out_ptr + \
            pid_b * stride_o_b + \
            (i_start + i_range)[:, None, None] * stride_o_i + \
            (j_start + j_range)[None, :, None] * stride_o_j + \
            (d_start + d_range)[None, None, :] * stride_o_h
        mask_out = i_valid[:, None, None] & j_valid[None, :, None] & d_valid[None, None, :]
        tl.store(out_ptrs, acc, mask=mask_out)


# -----------------------------------------------------------------------
# Entrypoint used by the evaluator
# -----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool or float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict with keys
              * "dim"         – input feature dimension
              * "hidden_dim" – hidden feature dimension
              * "nomask"      – True → ignore mask, False → use mask
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -------------------------------------------------------------------
    # unpack inputs
    # -------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # -------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )   # shape [B, N, N, dim]

    # -------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dimension)
    # -------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------
    # 4. Gating vectors (sigmoid)
    # -------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------
    # 5. Triangular multiplicative update – Triton kernel
    # -------------------------------------------------------------------
    # Ensure contiguous layout for efficient pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)          # [B, N, N, H]

    B, N, _, H = left.shape

    # Strides in number of elements (PyTorch gives stride in elements)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # Tunable block sizes – chosen to fit in registers on an H100
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_K = 16
    BLOCK_D = 8

    # Grid: (batch, i‑tiles, j‑tiles)
    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_h,
        stride_r_b,
        stride_r_j,
        stride_r_k,
        stride_r_h,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,
    )
    # Synchronise before using the result
    torch.cuda.synchronize()

    # -------------------------------------------------------------------
    # 6. Normalisation, gating and final projection
    # -------------------------------------------------------------------
    # LayerNorm over hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply output gate
    out = out * out_gate

    # Final linear back to original dimension (no bias)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
