def trilmul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_left, stride_i_left, stride_k_left, stride_h_left,
    stride_b_right, stride_i_right, stride_k_right, stride_h_right,
    stride_b_out, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, H_BLOCK: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute base offsets
    b = pid_b

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Initialize accumulators for each hidden block
    acc = tl.zeros([BLOCK_M, BLOCK_N, H_BLOCK], dtype=tl.float32)

    # Iterate over k dimension
    for k_outer in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_outer * BLOCK_K

        # Load left block (M, K, H_BLOCK)
        offs_i = i_offset + tl.arange(0, BLOCK_M)[:, None, None]
        offs_k = k_offset + tl.arange(0, BLOCK_K)[None, :, None]
        offs_h = tl.arange(0, H_BLOCK)[None, None, :]
        # mask OOB
        mask_i = offs_i < N
        mask_k = offs_k < N
        mask_h = offs_h < H

        left_ptrs = left_ptr + b*stride_b_left + offs_i*stride_i_left + offs_k*stride_k_left + offs_h*stride_h_left

        offs_j = j_offset + tl.arange(0, BLOCK_N)[:, None, None]  # shape (N,1,1)
        right_ptrs = right_ptr + b*stride_b_right + offs_j*stride_i_right + offs_k*stride_k_right + offs_h*stride_h_right
        # Wait stride_i_right corresponds to second dimension after batch? The shape is [B, N, N, H]; For right we want indexing (b, j, k, h).         mask_j = offs_j < N
        right = tl.load(right_ptrs, mask=mask_j & mask_k & mask_h, other=0.0)

        # Compute partial contributions

def trilmul_reduce_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, H_BLOCK: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # compute offsets of the start of the tile
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    b = pid_b

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_BLOCK), dtype=tl.float32)

    # loop over k dimension
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        # actual K size
        cur_k = tl.minimum(BLOCK_K, N - k_start * BLOCK_K)
        # offsets for left
        i_idx = i_start + tl.arange(0, BLOCK_M)[:, None, None]  # (M,1,1)
        k_idx = k_start * BLOCK_K + tl.arange(0, BLOCK_K)[None, :, None]  # (1,K,1)
        h_idx = tl.arange(0, H_BLOCK)[None, None, :]  # (1,1,H)
        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_h = h_idx < H

        left_ptrs = left_ptr + b*stride_b_l + i_idx*stride_i_l + k_idx*stride_k_l + h_idx*stride_h_l
        left = tl.load(left_ptrs, mask=mask_i & mask_k & mask_h, other=0.0)

        # offsets for right

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    # optionally get nomask flag: config.get("nomask", True)
    # but not needed.
    # step 1: layernorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # step 2: projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # step 3: mask
    if mask is not None:
        mask_ = mask.unsqueeze(-1).type_as(left)  # broadcast
        left = left * mask_
        right = right * mask_
    # step 4: gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # step5: trilmul reduction
    out = torch.empty_like(left)  # shape B,N,N,hidden_dim

    # call Triton kernel with block sizes.
    B, N, _, H = left.shape
    # compute strides
    stride_b_l, stride_i_l, stride_j_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_j_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # Choose block sizes based on N
    BLOCK_M = 16  # i tile
    BLOCK_N = 16  # j tile
    BLOCK_K = 32  # k tile
    H_BLOCK = 32  # hidden tile (must divide hidden_dim)
    # compute grid dims
    grid = (B, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    # launch kernel for each hidden block slice
    # Because H_BLOCK may be less than hidden_dim
    for h_start in range(0, hidden_dim, H_BLOCK):
        # slice tensors
        left_h = left[..., h_start:h_start+H_BLOCK]
        right_h = right[..., h_start:h_start+H_BLOCK]
        out_h = out[..., h_start:h_start+H_BLOCK]
        trilmul_reduce_kernel[grid](
            left_h, right_h, out_h,
            B, N, hidden_dim,
            stride_b_l, stride_i_l, stride_j_l, stride_h_l,
            stride_b_r, stride_i_r, stride_j_r, stride_h_r,
            stride_b_o, stride_i_o, stride_j_o, stride_h_o,
            BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, H_BLOCK=H_BLOCK
        )
    # Step6: out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])
    # step7: out_gate gating
    out = out * out_gate
    # step8: final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul "outgoing" kernel (AlphaFold3 style)

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections → left, right (B, N, N, H) and three gate vectors.
3. Apply optional mask to left/right, then apply the left/right gates.
4. Compute the bilinear reduction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   with a custom Triton kernel.  The hidden dimension H is processed
   in blocks of H_BLOCK (default 32) to keep register pressure low.
   For each (i‑tile, j‑tile, batch) we accumulate a tile
   (BLOCK_M × BLOCK_N × H_BLOCK) over the k‑dimension using a
   series of small GEMM‑like dot‑products (one per hidden sub‑block).
5. Layer‑norm the result over the hidden dimension, multiply by the
   output gate and apply the final linear projection back to C.

The kernel works for B∈{1,2}, N up to 1024 and hidden dimensions
divisible by H_BLOCK (the provided test configurations all satisfy this).
All operations except the bilinear reduction are performed with PyTorch
so that the implementation stays simple and numerically identical
to the reference code.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: sum_k left[...,i,k,:] * right[...,j,k,:]
# ------------------------------------------------------------
@triton.jit
def trilmul_reduce_kernel(
    left_ptr,          # *const float *
    right_ptr,         # *const float *
    out_ptr,           # *mut   float *
    B, N,              # int64
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    H_BLOCK: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The hidden dimension is processed in blocks of size H_BLOCK.
    """
    pid_b = tl.program_id(0)      # batch
    pid_i = tl.program_id(1)      # i‑tile
    pid_j = tl.program_id(2)      # j‑tile

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    b = pid_b

    # accumulator for the tile (M, N, H_BLOCK)
    acc = tl.zeros((BLOCK_M, BLOCK_N, H_BLOCK), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        k_base = kb * BLOCK_K

        # ------------------------------------------------------------------
        # Build offsets for left (shape B,N,N,H)
        # left[i, k, h]
        # ------------------------------------------------------------------
        i_idx = i_base + tl.arange(0, BLOCK_M)[:, None, None]                # (M,1,1)
        k_idx = k_base + tl.arange(0, BLOCK_K)[None, :, None]                # (1,K,1)
        h_idx = tl.arange(0, H_BLOCK)[None, None, :]                         # (1,1,H)

        mask_i = i_idx < N
        mask_k = k_idx < N
        mask_h = h_idx < N  # hidden dim always fits H_BLOCK (divisible)

        left_off = (
            b * stride_b_l
            + i_idx * stride_i_l
            + k_idx * stride_k_l
            + h_idx * stride_h_l
        )
        left = tl.load(left_ptr + left_off,
                       mask=mask_i & mask_k & mask_h,
                       other=0.0)                                            # (M,K,H)

        # ------------------------------------------------------------------
        # Offsets for right (shape B,N,N,H)
        # right[j, k, h]
        # ------------------------------------------------------------------
        j_idx = j_base + tl.arange(0, BLOCK_N)[:, None, None]                # (N,1,1)

        mask_j = j_idx < N

        right_off = (
            b * stride_b_r
            + j_idx * stride_i_r
            + k_idx * stride_k_r
            + h_idx * stride_h_r
        )
        right = tl.load(right_ptr + right_off,
                        mask=mask_j & mask_k & mask_h,
                        other=0.0)                                           # (N,K,H)

        # ------------------------------------------------------------------
        # Accumulate dot products for each hidden sub‑block.
        # For each h ∈ [0, H_BLOCK) we compute
        #   acc[...,h] += left[...,h] @ right[...,h]^T
        # ------------------------------------------------------------------
        for hi in range(H_BLOCK):
            # (M, K) and (N, K) slices
            left_h = left[:, :, hi]          # shape (M, K)
            right_h = right[:, :, hi]        # shape (N, K)
            # tl.dot computes Σ_k left_h[m,k] * right_h[n,k] → (M,N)
            acc[:, :, hi] += tl.dot(left_h, right_h)

    # ------------------------------------------------------------------
    # Store the accumulated tile to global memory
    # ------------------------------------------------------------------
    i_out = i_base + tl.arange(0, BLOCK_M)[:, None, None]
    j_out = j_base + tl.arange(0, BLOCK_N)[None, :, None]
    h_out = tl.arange(0, H_BLOCK)[None, None, :]

    mask_i_o = i_out < N
    mask_j_o = j_out < N
    mask_h_o = h_out < N  # always true for valid blocks

    out_off = (
        b * stride_b_o
        + i_out * stride_i_o
        + j_out * stride_j_o
        + h_out * stride_h_o
    )
    tl.store(out_ptr + out_off,
             acc,
             mask=mask_i_o & mask_j_o & mask_h_o)


# ------------------------------------------------------------
# Custom kernel entry point
# ------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, C] (float32)
        - mask        : torch.Tensor [B, N, N] (bool/float) or None
        - weights     : dict of model parameters (LayerNorm, Linear, etc.)
        - config      : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Load weights (names follow the PyTorch reference implementation)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # Step 1: LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Step 2: Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # Step 3: Apply optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        mask_bc = mask.unsqueeze(-1).to(dtype)          # [B, N, N, 1]
        left = left * mask_bc
        right = right * mask_bc

    # ------------------------------------------------------------------
    # Step 4: Compute gates and apply them
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Step 5: Bilinear reduction (TriMul core) using Triton
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    out = torch.empty_like(left)  # allocate output tensor [B, N, N, H]

    # Strides for the three tensors (batch, i, k/j, hidden)
    stride_b_l, stride_i_l, stride_k_l, stride_h_l = left.stride()
    stride_b_r, stride_i_r, stride_k_r, stride_h_r = right.stride()
    stride_b_o, stride_i_o, stride_j_o, stride_h_o = out.stride()

    # Tunable block sizes – chosen to fit typical H100 registers.
    BLOCK_M = 16      # tile size along i
    BLOCK_N = 16      # tile size along j
    BLOCK_K = 32      # tile size along k
    H_BLOCK = 32      # must divide hidden_dim in the provided configs

    # Grid: (batch, i‑tiles, j‑tiles)
    grid = (
        B,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Process hidden dimension in blocks of H_BLOCK
    for h_start in range(0, hidden_dim, H_BLOCK):
        # Views that start at the appropriate hidden offset
        left_h = left[..., h_start : h_start + H_BLOCK]
        right_h = right[..., h_start : h_start + H_BLOCK]
        out_h = out[..., h_start : h_start + H_BLOCK]

        # Launch the Triton kernel
        trilmul_reduce_kernel[grid](
            left_h,
            right_h,
            out_h,
            B,
            N,
            stride_b_l,
            stride_i_l,
            stride_k_l,
            stride_h_l,
            stride_b_r,
            stride_i_r,
            stride_k_r,
            stride_h_r,
            stride_b_o,
            stride_i_o,
            stride_j_o,
            stride_h_o,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            H_BLOCK=H_BLOCK,
        )
    # ------------------------------------------------------------------
    # Step 6: Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # ------------------------------------------------------------------
    # Step 7: Apply output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Step 8: Final linear projection back to original dimensionality
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
