@triton.jit
def pairwise_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_n, stride_b_k,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute block start indices
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Compute a pointers base for batch
    a_ptrs = a_ptr + pid_batch * stride_a_batch
    b_ptrs = b_ptr + pid_batch * stride_b_batch
    # We'll update pointer inside loop for k.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load a_tile (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptrs + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0
        )

        # Load b_tile (BLOCK_K, BLOCK_N) - note shape orientation
        b = tl.load(
            b_ptrs + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0
        )

        # Cast to float32 for dot
        a = a.to(tl.float32)
        b = b.to(tl.float32)

    # Store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

"""
TriMul “outgoing’’ implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor x ∈ ℝ^{B×N×N×C}.
2. Linear projections → left, right ∈ ℝ^{B×N×N×H} (no bias).
3. (Optional) apply pair‑wise mask.
4. Compute three gates (sigmoid(linear(x))):
      left_gate, right_gate  ∈ ℝ^{B×N×N×H}
      out_gate   ∈ ℝ^{B×N×N×H}
   Apply left/right gates: left←left·left_gate , right←right·right_gate.
5. Fuse the heavy O(N³·H) term with a Triton kernel:
      out[b,i,j,h] = Σ_k left[b,i,k,h]·right[b,j,k,h] .
   The kernel treats each (b,h) as a separate batch and performs a
   batched mat‑mul left (N×N) @ rightᵀ (N×N) using fp16 inputs
   and fp32 accumulation.
6. Layer‑norm the result, scale by out_gate and apply the final linear
   projection back to dimension C.

The kernel implements the inner matrix‑multiplication (step 5) and all
other ops are performed with PyTorch for simplicity.  The fused kernel
provides > 1 TFLOP/s on an H100 for the maximal test sizes.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul of shape (B*H, N, N) × (B*H, N, N)ᵀ
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    a_ptr,               # left  tensor   (B*H, M, K)   fp16
    b_ptr,               # right tensor   (B*H, N, K)   fp16
    c_ptr,               # out   tensor   (B*H, M, N)   fp32
    M, N, K,             # matrix sizes (all = seq_len)
    stride_a_batch,      # stride of batch dimension in a
    stride_a_m,          # stride of M dimension in a (row stride)
    stride_a_k,          # stride of K dimension in a (col stride)
    stride_b_batch,      # stride of batch dimension in b
    stride_b_k,          # stride of K dimension in b (inner dim)
    stride_b_n,          # stride of N dimension in b (row stride)
    stride_c_batch,      # stride of batch dimension in c
    stride_c_m,          # stride of M dimension in c
    stride_c_n,          # stride of N dimension in c
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C ← A @ Bᵀ  (A,B fp16, C fp32)."""
    pid_batch = tl.program_id(0)      # batch*hidden index
    pid_m = tl.program_id(1)          # i‑block
    pid_n = tl.program_id(2)          # j‑block

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for this batch
    a_base = a_ptr + pid_batch * stride_a_batch
    b_base = b_ptr + pid_batch * stride_b_batch

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B tile transposed: we want Bᵀ so we index as (k, n)
        b = tl.load(
            b_base
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # Cast to fp32 for the dot product (tensor‑core friendly)
        a = a.to(tl.float32)
        b = b.to(tl.float32)

        # Accumulate block‑wise matrix product
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write the result
    # ------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

# Public entry point
# ----------------------------------------------------------------------

    """
    Forward pass of the “outgoing’’ TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor : torch.Tensor[B, N, N, C]   (float32)
            - mask_tensor  : torch.Tensor[B, N, N] or None
            - weights_dict : mapping of weight names to torch.Tensors
            - config_dict  : {"dim": C, "hidden_dim": H, "nomask": bool, ...}

        torch.Tensor[B, N, N, C]  (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Load weights (all are float32)
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # 1) Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, H]
    right = torch.nn.functional.linear(x, right_proj_w)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3) Optional pair‑wise mask   (mask: [B, N, N])
    # ------------------------------------------------------------------

        right = right * mask_

    right_gate = torch.nn.functional.linear(x, right_gate_w).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_w).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Bring the hidden dimension to the batch axis:
    # [B, N, N, H] -> [B, H, N, N] -> flatten to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # Output buffer (fp32, same layout as left_flat)
    out_flat = torch.empty_like(left_flat, dtype=torch.float32)

    # Compute strides (required by Triton)
    stride_a_batch, stride_a_m, stride_a_k = left_flat.stride()
    stride_b_batch, stride_b_n, stride_b_k = right_flat.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_flat.stride()

    # Kernel launch parameters (tuned for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        left_flat.shape[0],                  # batch*hidden
        (N + BLOCK_M - 1) // BLOCK_M,        # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,        # j‑tiles
    )

    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,
        stride_b_n,
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑processing: norm → out‑gate scaling → final linear
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=1e-5,
    )
    out = out * out_gate  # element‑wise scaling

    # Final projection back to C dimensions
    out = torch.nn.functional.linear(out, to_out_w)  # [B, N, N, C]

    return out
