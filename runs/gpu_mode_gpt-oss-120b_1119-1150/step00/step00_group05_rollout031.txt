"""
TriMul (outgoing) fusion kernel for AlphaFold3‑style pairwise updates.

The forward pass consists of

1. Layer‑norm on the input tensor (shape [B, N, N, C]).
2. Linear projections to a hidden dimension H (no bias):
      left = x·W_left   , right = x·W_right
3. Gating: sigmoid(x·W_gate) and an optional binary mask.
   The mask, the projection and the gate are fused in a single
   Triton kernel (no extra memory traffic for the mask).
4. Core multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix‑multiply per hidden channel
   (equivalent to (left₍d₎ @ right₍d₎ᵀ) for each d).
   We implement it with a highly‑optimized Torch batched GEMM which
   on H100 uses Tensor‑Core kernels.
5. A second Layer‑norm on the hidden dimension, multiplication with a
   third gate (out_gate) and a final linear projection back to C.

Only the mask‑gate‑projection fusion is done in Triton; the heavy
N³ reduction is handled by Torch’s native batched GEMM which gives
optimal performance on modern GPUs.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: fuse mask, projection and gate (elementwise product)
# ----------------------------------------------------------------------
@triton.jit
def _apply_gate_kernel(
    proj_ptr,          # hidden projection (float32)
    gate_ptr,          # sigmoid gate (float32)
    mask_ptr,          # binary/float mask (float32)
    out_ptr,           # gated output (float32)

    # strides for projection
    proj_s_b, proj_s_i, proj_s_k, proj_s_d,
    # strides for gate
    gate_s_b, gate_s_i, gate_s_k, gate_s_d,
    # strides for mask (no hidden dim)
    mask_s_b, mask_s_i, mask_s_k,
    # strides for output
    out_s_b, out_s_i, out_s_k, out_s_d,

    B: tl.constexpr, N: tl.constexpr, H: tl.constexpr,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)                # batch
    pid_i = tl.program_id(1)                # i‑tile
    pid_k = tl.program_id(2)                # k‑tile (the third tensor axis)

    # ------------------------------------------------------------------
    # index ranges for this tile
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    k_offset = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    d_offset = tl.arange(0, H)

    # out‑of‑bounds masks
    i_valid = i_offset < N
    k_valid = k_offset < N
    ik_valid = i_valid[:, None] & k_valid[None, :]

    # ------------------------------------------------------------------
    # mask (broadcasted over hidden dimension)
    # ------------------------------------------------------------------
    mask = tl.load(
        mask_ptr
        + pid_b * mask_s_b
        + i_offset[:, None] * mask_s_i
        + k_offset[None, :] * mask_s_k,
        mask=ik_valid,
        other=1.0,
    )
    mask = mask[:, :, None]   # shape (BLOCK_I, BLOCK_K, 1)

    # ------------------------------------------------------------------
    # projection and gate
    # ------------------------------------------------------------------
    proj = tl.load(
        proj_ptr
        + pid_b * proj_s_b
        + i_offset[:, None, None] * proj_s_i
        + k_offset[None, :, None] * proj_s_k
        + d_offset[None, None, :] * proj_s_d,
        mask=ik_valid[:, :, None],
        other=0.0,
    )
    gate = tl.load(
        gate_ptr
        + pid_b * gate_s_b
        + i_offset[:, None, None] * gate_s_i
        + k_offset[None, :, None] * gate_s_k
        + d_offset[None, None, :] * gate_s_d,
        mask=ik_valid[:, :, None],
        other=0.0,
    )

    # ------------------------------------------------------------------
    # gated output = proj * gate * mask
    # ------------------------------------------------------------------
    out = proj * gate * mask

    tl.store(
        out_ptr
        + pid_b * out_s_b
        + i_offset[:, None, None] * out_s_i
        + k_offset[None, :, None] * out_s_k
        + d_offset[None, None, :] * out_s_d,
        out,
        mask=ik_valid[:, :, None],
    )

# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor_or_None, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (C == config["dim"])
    """
    # --------------------------------------------------------------
    # unpack arguments
    # --------------------------------------------------------------
    if len(data) == 4:
        input_tensor, mask_tensor, weights, config = data
    else:  # mask omitted (nomask case)
        input_tensor, weights, config = data
        mask_tensor = None

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # pick out the weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over the last dim = C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right_proj = F.linear(x, right_proj_weight) # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid of linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 4. Optional mask – if missing we use an all‑ones mask
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = torch.ones((bs, seq_len, seq_len), dtype=x.dtype, device=x.device)
    else:
        mask = mask_tensor.to(dtype=x.dtype)

    # ------------------------------------------------------------------
    # 5. Fuse mask, projection and gate with a Triton kernel
    # ------------------------------------------------------------------
    BLOCK_I = 8   # tile size in the first N dimension
    BLOCK_K = 8   # tile size in the second N dimension

    # allocate containers for the gated tensors
    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # Grid: (batch, ceil(N/BLOCK_I), ceil(N/BLOCK_K))
    grid = (
        bs,
        (seq_len + BLOCK_I - 1) // BLOCK_I,
        (seq_len + BLOCK_K - 1) // BLOCK_K,
    )

    # ----- left side -----
    _apply_gate_kernel[grid](
        left_proj,
        left_gate,
        mask,
        left_gated,
        # strides for projection
        left_proj.stride(0), left_proj.stride(1), left_proj.stride(2), left_proj.stride(3),
        # strides for gate
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        # strides for mask (no hidden dim)
        mask.stride(0), mask.stride(1), mask.stride(2),
        # strides for output
        left_gated.stride(0), left_gated.stride(1), left_gated.stride(2), left_gated.stride(3),
        bs, seq_len, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K,
    )

    # ----- right side -----
    _apply_gate_kernel[grid](
        right_proj,
        right_gate,
        mask,
        right_gated,
        right_proj.stride(0), right_proj.stride(1), right_proj.stride(2), right_proj.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        right_gated.stride(0), right_gated.stride(1), right_gated.stride(2), right_gated.stride(3),
        bs, seq_len, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Core multiplicative update:
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    This is a batched GEMM per hidden channel:
    #          (B, H, i, k) @ (B, H, k, j) -> (B, H, i, j)
    # ------------------------------------------------------------------
    # reshape for batched matmul
    left_perm = left_gated.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_perm = right_gated.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    # batched matmul on H100 → Tensor‑Core fast path
    out_perm = torch.matmul(left_perm, right_perm)          # (B, H, i, j)

    # back to original layout
    out = out_perm.permute(0, 2, 3, 1).contiguous()         # (B, i, j, H)

    # ------------------------------------------------------------------
    # 7. Second LayerNorm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8. Apply out_gate (element‑wise) and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # no bias, shape (B, N, N, C)

    return out
```
