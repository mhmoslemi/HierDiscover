    b_off_r = b * stride_r_b
    h_off_r = h * stride_r_h
    b_off_o = b * stride_o_b
    h_off_o = h * stride_o_h

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over reduction axis k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b_off_l + h_off_l + offs_i[:, None] * stride_l_i + offs_k[None, :] * stride_l_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        left = left.to(tl.float32)

        # Load right tile transposed: we want (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b_off_r + h_off_r + offs_k[:, None] * stride_r_k + offs_j[None, :] * stride_r_j
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        right = right.to(tl.float32)

        acc += tl.dot(left, right)

    # Write back
    out_ptrs = out_ptr + b_off_o + h_off_o + offs_i[:, None] * stride_o_i + offs_j[None, :] * stride_o_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul outgoing kernel (AlphaFold3 style)

The kernel implements the heavy N³ component of the TriMul operation:
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

where left/right are the gated linearly‑projected input tensors of shape
[batch, N, N, hidden_dim].  The reduction over the shared dimension *k* is
performed with a Triton matmul kernel (batched matmul for each hidden slice).
All other operations (LayerNorm, linear projections, sigmoids, masking and the
final linear projection) are expressed with PyTorch ops.  The Triton kernel
operates on float32 accumulation (inputs may be float16/BF16) and writes the
result in the hidden‑dim space, after which a second LayerNorm, gating and an
output linear layer produce the final tensor of shape [B,N,N,dim].

This implementation targets the “outgoing” formulation used in AlphaFold3,
AlphaFold‑3‑experimental, Chai, Protenix, etc.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_matmul_kernel(
    left_ptr,               # *float*, shape [B,N,N,H]
    right_ptr,              # *float*, shape [B,N,N,H]
    out_ptr,                # *float*, shape [B,N,N,H]
    B, N, H,                # runtime constants
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matmul for the TriMul einsum:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    The grid is 3‑D:
      x‑axis: i‑blocks of size BLOCK_M
      y‑axis: j‑blocks of size BLOCK_N
      z‑axis: flattened batch*hidden dimension (B*H)
    """
    pid_m = tl.program_id(0)            # i‑block index
    pid_n = tl.program_id(1)            # j‑block index
    pid_z = tl.program_id(2)            # flattened (batch, hidden) index

    # decompose pid_z into batch and hidden indices
    b = pid_z // H
    h = pid_z % H

    # start positions of the current tile
    i0 = pid_m * BLOCK_M
    j0 = pid_n * BLOCK_N

    # offsets inside the tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # base offsets for the current (b,h) slice
    b_off_l = b * stride_l_b
    h_off_l = h * stride_l_h
    b_off_r = b * stride_r_b
    h_off_r = h * stride_r_h
    b_off_o = b * stride_o_b
    h_off_o = h * stride_o_h

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # reduction over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        k0 = kk * BLOCK_K
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ---- load left tile (M x K) ----
        # left[b,i,k,h]
        left_ptrs = (left_ptr + b_off_l + h_off_l
                     + offs_i[:, None] * stride_l_i
                     + offs_k[None, :] * stride_l_k)
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)
        left = left.to(tl.float32)

        # ---- load right tile transposed (K x N) ----
        # we need right[b,j,k,h] but want (K,N) layout for dot
        right_ptrs = (right_ptr + b_off_r + h_off_r
                      + offs_k[:, None] * stride_r_k
                      + offs_j[None, :] * stride_r_j)
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)
        right = right.to(tl.float32)

        # matrix multiplication for this k‑tile
        acc += tl.dot(left, right)      # (M,K) @ (K,N) -> (M,N)

    # store the result
    out_ptrs = (out_ptr + b_off_o + h_off_o
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module using a Triton kernel for the
    N³ multiplication step.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] or None
            - weights     : dict of model parameters (LayerNorm, Linear, etc.)
            - config      : dict with keys "dim", "hidden_dim", "nomask" ...

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # unpack
    input_tensor, mask_tensor, weights, config = data

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (dim)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ----------------------------------------------------------------------
    # 2. Linear projections (dim -> hidden_dim)
    # ----------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    # flatten sequence dimensions for a single linear call (more efficient)
    x_flat = x.reshape(-1, dim)                     # [(B*N*N), dim]

    left = F.linear(x_flat, left_proj_weight).view(B, N, N, hidden_dim)
    right = F.linear(x_flat, right_proj_weight).view(B, N, N, hidden_dim)

    # ----------------------------------------------------------------------
    # 3. Optional mask
    # ----------------------------------------------------------------------
    if (mask_tensor is not None) and (not nomask):
        # mask shape: [B,N,N] -> [B,N,N,1] for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4. Gating (sigmoid of linear maps)
    # ----------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x_flat, left_gate_weight)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_weight)).view(B, N, N, hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x_flat, out_gate_weight)).view(B, N, N, hidden_dim)

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. TriMul core: batched matmul across the shared k‑dimension
    # ----------------------------------------------------------------------
    # Ensure contiguity for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()

    # allocate output tensor (hidden space)
    out_hidden = torch.empty_like(left)

    # strides needed by the kernel
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_hidden.stride()

    # Tunable block sizes (work well on H100 for hidden_dim=128)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑blocks
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑blocks
        B * hidden_dim                  # (batch, hidden) dimension
    )

    trimul_matmul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out_hidden,
        B=B,
        N=N,
        H=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_h=stride_l_h,
        stride_r_b=stride_r_b,
        stride_r_j=stride_r_j,
        stride_r_k=stride_r_k,
        stride_r_h=stride_r_h,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_h=stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # empirically good for the target hardware
        num_stages=2,
    )

    # ----------------------------------------------------------------------
    # 6. Output LayerNorm (hidden_dim) + gating
    # ----------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out_hidden,
                            (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    out = out_norm * out_gate

    # ----------------------------------------------------------------------
    # 7. Final linear projection (hidden_dim -> dim)
    # ----------------------------------------------------------------------
    to_out_weight = weights["to_out.weight"]
    out_flat = out.view(-1, hidden_dim)
    out_final = F.linear(out_flat, to_out_weight).view(B, N, N, dim)

    return out_final
