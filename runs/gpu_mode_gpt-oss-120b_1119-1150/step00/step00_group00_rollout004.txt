"""
custom_kernel
--------------

Implements the “outgoing” TriMul operation from AlphaFold3.

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor (shape [B, N, N, C]).
2. Linear projections → left, right (shape [B, N, N, H]).
3. Apply the pair‑mask (if present) and three sigmoidal gates.
4. Fuse the expensive contraction
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that performs a batched matrix‑multiply
   over the (i,k) × (j,k) dimensions for every hidden channel `h`.
   The kernel works on FP16 data (loads as fp16, accumulates in FP32) to
   reduce memory traffic while keeping the final result in FP32.
5. Layer‑normalize the result over the hidden dimension, multiply by the
   output gate and apply the final linear projection back to the original
   channel dimension.

The kernel processes the combined batch‑and‑hidden index (`bh = b*H + h`)
as its third program dimension, enabling massive parallelism even for
large sequence lengths (N up to 1024).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,          # ptr to left tensor  (fp16)
    right_ptr,         # ptr to right tensor (fp16)
    out_ptr,           # ptr to output tensor (fp16)
    B, N, H,           # batch, seq_len, hidden_dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched contraction over the `k` dimension:
         out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    (no transpose of the right operand; `trans_b=True` in tl.dot achieves it)."""
    pid_bh = tl.program_id(2)          # combined batch*hidden index
    pid_m  = tl.program_id(0)          # i‑tile
    pid_n  = tl.program_id(1)          # j‑tile

    # decode batch & hidden channel
    hid   = pid_bh % H
    batch = pid_bh // H

    # tile start indices
    i0 = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    j0 = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds checks
    mask_i = i0 < N
    mask_j = j0 < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension in BLOCK_K chunks
    for k in range(0, N, BLOCK_K):
        k0 = k + tl.arange(0, BLOCK_K)
        mask_k = k0 < N

        # pointers for left[b,i,k,h]
        left_off = (
            batch * stride_left_b
            + hid * stride_left_h
            + i0[:, None] * stride_left_i
            + k0[None, :] * stride_left_k
        )
        left = tl.load(
            left_ptr + left_off,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_M, BLOCK_K), fp16

        # pointers for right[b,j,k,h]
        right_off = (
            batch * stride_right_b
            + hid * stride_right_h
            + j0[:, None] * stride_right_i
            + k0[None, :] * stride_right_k
        )
        right = tl.load(
            right_ptr + right_off,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )  # (BLOCK_N, BLOCK_K), fp16

        # dot product over k (left * right^T)
        acc += tl.dot(left, right, trans_b=True, out_dtype=tl.float32)

    # write back result (cast to fp16)
    out_off = (
        batch * stride_out_b
        + hid * stride_out_h
        + i0[:, None] * stride_out_i
        + j0[None, :] * stride_out_j
    )
    tl.store(
        out_ptr + out_off,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """
    Entry point for the Triton‑accelerated TriMul forward pass.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]          # original channel dimension C
    hidden_dim = config["hidden_dim"]   # projection dimension H
    eps = 1e-5

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension (C)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast to hidden dim)
    # ------------------------------------------------------------------
    if mask_tensor is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            device=device,
            dtype=dtype,
        )
    else:
        mask = mask_tensor
    mask = mask.unsqueeze(-1)               # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and apply to projected tensors
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Triton kernel for the contraction over k
    # ------------------------------------------------------------------
    # Cast to fp16 for the kernel (reduces memory bandwidth)
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, N2, H = left_fp16.shape
    assert N == N2, "Input must be square in the two middle dimensions."

    out_fp16 = torch.empty_like(left_fp16)   # [B,N,N,H] fp16

    # strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left_fp16.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right_fp16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_fp16.stride()

    # Block sizes – chosen to fit well on H100 Tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_batch = B * H   # combined batch*hidden dimension

    # Launch kernel
    tri_mul_kernel[(grid_m, grid_n, grid_batch)](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        H,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to fp32 for the remaining ops
    out = out_fp16.to(torch.float32)        # [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden dimension H)
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Output gate and final linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,dim]

    return out
