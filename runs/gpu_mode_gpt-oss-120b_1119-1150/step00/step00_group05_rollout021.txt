@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    total_batch, N,
    stride_l_batch, stride_l_i, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_j,
    stride_o_batch, stride_o_i, stride_o_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    
    # Compute start indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # Compute masks
    mask_i = i < N
    mask_j = j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)  # accumulate in fp32

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + pid_batch * stride_l_batch + i[:, None] * stride_l_i + k_offset[None, :] * stride_l_k
        left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # Load right tile (K x N)
        # right shape: (batch, K, J) where stride_r_k for K dimension and stride_r_j for J dimension
        right_ptrs = right_ptr + pid_batch * stride_r_batch + k_offset[:, None] * stride_r_k + j[None, :] * stride_r_j
        right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Compute partial product
        # Use dot to compute matrix multiplication: left_tile (M,K) dot right_tile (K,N)
        acc += tl.dot(left_tile, right_tile)

    # Write output
    out_ptrs = out_ptr + pid_batch * stride_o_batch + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask
    if not nomask:
        mask_ = mask.unsqueeze(-1)   # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    total_batch = B * hidden_dim

    # Permute and reshape for kernel
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where second N is k, third is j

    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    out_flat = torch.empty_like(left_flat)

    # Compute strides
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_j = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    # Define block sizes, typical: 128? Choose 64 to fit within shared memory
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Launch kernel
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             total_batch )

    tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_j,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape and permute back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # shape [B,N,N,hidden]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Linear projection back to dim
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out
```

"""
TriMul (outgoing) – highly‑optimized Triton implementation.

The core of the TriMul operation is the einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each hidden channel `d` this is a matrix multiplication
    left_d (N×N)  @  right_d.T (N×N) → (N×N).

We fuse all hidden channels into the batch dimension (B·hidden_dim) and
run a custom Triton kernel that performs a batched matmul with
FP16 arithmetic (accumulating in FP32).  The surrounding LayerNorm,
linear projections, gating and the final projection back to `dim` are
performed with PyTorch for simplicity.

The implementation works for both masked / un‑masked inputs and all
shapes listed in the test suite.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matmul (B·H, N, N) @ (B·H, N, N).T
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    left_ptr,          # ptr to left matrix   (batch, i, k)
    right_ptr,         # ptr to right matrix  (batch, k, j)
    out_ptr,           # ptr to output        (batch, i, j)

    total_batch: tl.constexpr,  # B*hidden_dim  (only needed for sanity)
    N: tl.constexpr,            # sequence length

    # strides for the left tensor  (batch, i, k)
    stride_l_batch, stride_l_i, stride_l_k,
    # strides for the right tensor (batch, k, j)
    stride_r_batch, stride_r_k, stride_r_j,
    # strides for the output tensor (batch, i, j)
    stride_o_batch, stride_o_i, stride_o_j,

    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute one tile of the batched matrix multiplication."""
    pid_batch = tl.program_id(2)        # batch dimension = B * hidden_dim
    pid_i = tl.program_id(0)           # block row
    pid_j = tl.program_id(1)           # block column

    # ------------------------------------------------------------------
    # 1) Compute absolute offsets for the current tile
    # ------------------------------------------------------------------
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for out‑of‑bounds (edge tiles)
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator in FP32 (higher precision)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 2) Loop over the K dimension in blocks
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ----- Load a tile of the left matrix (M×K) -----
        left_ptrs = (
            left_ptr
            + pid_batch * stride_l_batch
            + i_offset[:, None] * stride_l_i
            + k_offset[None, :] * stride_l_k
        )
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)   # cast to FP32

        # ----- Load a tile of the right matrix (K×N) -----
        right_ptrs = (
            right_ptr
            + pid_batch * stride_r_batch
            + k_offset[:, None] * stride_r_k
            + j_offset[None, :] * stride_r_j
        )
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # ----- Compute partial product (M×K)·(K×N) -----
        acc += tl.dot(left_tile, right_tile)   # FP32 accumulation

    # ------------------------------------------------------------------
    # 3) Write the result (cast back to FP16)
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + pid_batch * stride_o_batch
        + i_offset[:, None] * stride_o_i
        + j_offset[None, :] * stride_o_j
    )
    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Public entry point – matches the expected signature
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # 0) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                     # input feature dim
    hidden_dim = config["hidden_dim"]       # projection dim
    nomask = config.get("nomask", True)    # True → ignore mask
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input (per‑channel)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape [B, N, N, dim], float32

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional masking (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if not nomask:
        # mask is assumed to be 0/1 (float or bool)
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy O(N³) kernel
    # ------------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape
    total_batch = B * hidden_dim                           # flattened batch

    # ------------------------------------------------------------------
    # 6) Rearrange tensors to (B*H, N, N) so that a normal matmul
    #    implements Σ_k left[i,k] * right[j,k].
    # ------------------------------------------------------------------
    # left:   [B, N, N, H] → [B, H, N, N] (i, k)
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    # right:  [B, N, N, H] → [B, H, N, N] (k, j)      (swap the two N axes)
    right_perm = right_h.permute(0, 3, 2, 1).contiguous()

    left_flat = left_perm.view(total_batch, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(total_batch, N, N) # (B*H, N, N)

    # Output buffer (FP16, will be up‑cast later)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # 7) Compute strides required by the Triton kernel
    # ------------------------------------------------------------------
    stride_l_batch, stride_l_i, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_j = right_flat.stride()
    stride_o_batch, stride_o_i, stride_o_j = out_flat.stride()

    # ------------------------------------------------------------------
    # 8) Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks along i
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks along j
        total_batch,                    # one program per (batch * hidden)
    )

    _tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        total_batch,
        N,
        stride_l_batch, stride_l_i, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_j,
        stride_o_batch, stride_o_i, stride_o_j,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 9) Reshape back to [B, N, N, hidden] (float32)
    # ------------------------------------------------------------------
    out = (
        out_flat
        .view(B, hidden_dim, N, N)               # (B, H, N, N)
        .permute(0, 2, 3, 1)                     # (B, N, N, H)
        .contiguous()
        .to(torch.float32)
    )

    # ------------------------------------------------------------------
    # 10) LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 11) Apply the output gating
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 12) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
