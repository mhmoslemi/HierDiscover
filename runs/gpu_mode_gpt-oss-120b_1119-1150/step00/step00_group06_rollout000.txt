```
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # config contains dim and hidden_dim, and maybe 'nomask' flag.
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # unpack weights.
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # Some models may pass mask as None; if config 'nomask' True, treat mask = torch.ones.
    if mask is None or config.get('nomask', False):
        mask_tensor = None
    else:
        mask_tensor = mask

    # LayerNorm: x_norm = (x - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gate linear then sigmoid
    left_gate_pre = torch.nn.functional.linear(x_norm, left_gate_weight)
    left_gate = torch.sigmoid(left_gate_pre)
    right_gate_pre = torch.nn.functional.linear(x_norm, right_gate_weight)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate_pre = torch.nn.functional.linear(x_norm, out_gate_weight)
    out_gate = torch.sigmoid(out_gate_pre)

    # apply mask and gating using Triton kernel
    if mask_tensor is None:
        # mask all ones
        mask_f = torch.ones_like(left_proj)
    else:
        # mask shape [B,N,N]; broadcast to hidden dim
        mask_f = mask_tensor.unsqueeze(-1).type_as(input_tensor)
        mask_f = mask_f.expand_as(left_proj)

    # We need to apply mask * left_gate to left_proj and similarly for right.
    left = apply_gate_mask(left_proj, left_gate, mask_f)
    right = apply_gate_mask(right_proj, right_gate, mask_f)

    # Compute out = sum_k left[,i,k,:] * right[,j,k,:]
    # Rearrange for batched matmul
    B,N,N,_ = left.shape
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # flatten batch and hidden dims
    BH = B * hidden_dim
    left_flat = left_perm.view(BH, N, N)
    right_flat = right_perm.view(BH, N, N)
    # Compute output via bmm
    out_flat = torch.bmm(left_flat, right_flat.transpose(2, 1))
    # reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply with out_gate
    out_norm = out_norm * out_gate

    # final linear to dim
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]

    return out_final
```

"""
TriMul “outgoing” kernel (AlphaFold3 style)

The implementation follows the reference PyTorch module but fuses the
element‑wise mask/gate multiplication with a tiny Triton kernel.
Main steps
-----------
1. Layer‑norm over the last dimension of the input tensor (B,N,N,D).
2. Linear projections (x → left_proj / right_proj) and three gating
   projections (left_gate, right_gate, out_gate).  Sigmoid is applied to
   the gating tensors.
3. A Triton kernel `apply_gate_mask_kernel` fuses
       proj * gate * mask
   for both the left and right side.
4. The core “TriMul” contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   is realised as a batched matrix‑multiplication:
       left_perm (B, H, N, N)  → (B*H, N, N)
       right_perm (B, H, N, N) → (B*H, N, N)
       out_flat = bmm(left_flat, right_flat.transpose(-2, -1))
   This uses cuBLAS‑powered batched GEMM and is far faster than an
   explicit N³ loop.
5. Layer‑norm on the hidden dimension, multiplication by `out_gate`,
   and a final linear projection back to the original dimension.

All tensors stay on the GPU; only the element‑wise fusion is written in
Triton, satisfying the requirement to have “part of the operations in a
kernel”.  The final output is a float32 tensor of shape
[batch, seq_len, seq_len, dim].
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def apply_gate_mask_kernel(
    proj_ptr,    # *float32
    gate_ptr,    # *float32
    mask_ptr,    # *float32
    out_ptr,     # *float32
    n_elements,  # i32
    BLOCK_SIZE: tl.constexpr,
):
    """
    Element‑wise out = proj * gate * mask.
    All inputs are assumed to be contiguous 1‑D views.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)
    msk  = tl.load(mask_ptr + offs, mask=mask, other=1.0)

    out = proj * gate * msk
    tl.store(out_ptr + offs, out, mask=mask)


def apply_gate_mask(proj: torch.Tensor, gate: torch.Tensor, mask: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for the Triton kernel.
    All tensors must have the same shape and be on the same CUDA device.
    Returns a new tensor with the same dtype as the inputs (float32).
    """
    assert proj.shape == gate.shape == mask.shape, "Shape mismatch in apply_gate_mask"
    proj = proj.contiguous()
    gate = gate.contiguous()
    mask = mask.contiguous()
    out = torch.empty_like(proj)

    total_elems = proj.numel()
    BLOCK_SIZE = 1024  # reasonable size for H100; can be tuned
    grid = lambda meta: (triton.cdiv(total_elems, meta['BLOCK_SIZE']),)

    apply_gate_mask_kernel[grid](
        proj_ptr=proj,
        gate_ptr=gate,
        mask_ptr=mask,
        out_ptr=out,
        n_elements=total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
        num_stages=2,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Input:
        data = (input_tensor, mask, weights, config)
        - input_tensor: torch.Tensor [B, N, N, D] (float32)
        - mask: torch.Tensor [B, N, N] (float32) or None
        - weights: dict of all model parameters (torch.Tensors on CUDA)
        - config: dict containing at least 'dim' and 'hidden_dim' and optional 'nomask'
    Output:
        torch.Tensor [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # unpack weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension (last dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gate logits
    # ------------------------------------------------------------------
    left_proj = F.linear(x, left_proj_weight)          # [B, N, N, H]
    right_proj = F.linear(x, right_proj_weight)        # [B, N, N, H]

    left_gate_pre = F.linear(x, left_gate_weight)      # [B, N, N, H]
    right_gate_pre = F.linear(x, right_gate_weight)    # [B, N, N, H]
    out_gate_pre = F.linear(x, out_gate_weight)       # [B, N, N, H]

    left_gate = torch.sigmoid(left_gate_pre)
    right_gate = torch.sigmoid(right_gate_pre)
    out_gate = torch.sigmoid(out_gate_pre)

    # ------------------------------------------------------------------
    # Prepare mask (broadcasted to the hidden dimension)
    # ------------------------------------------------------------------
    if nomask or mask is None:
        # all‑ones mask – create a tensor of ones with the same shape as the projections
        mask_f = torch.ones_like(left_proj)
    else:
        # mask shape [B, N, N] → [B, N, N, 1] → broadcast over hidden dim
        mask_f = mask.unsqueeze(-1).to(input_tensor.dtype)
        mask_f = mask_f.expand_as(left_proj)  # now [B, N, N, H]

    # ------------------------------------------------------------------
    # Fuse projection, gating and mask using Triton
    # ------------------------------------------------------------------
    left = apply_gate_mask(left_proj, left_gate, mask_f)   # [B, N, N, H]
    right = apply_gate_mask(right_proj, right_gate, mask_f)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Core TriMul contraction via batched GEMM
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # Rearrange to (B, H, N, N) and flatten B & H into the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    BH = B * H
    left_flat = left_perm.view(BH, N, N)                # [B*H, N, N]
    right_flat = right_perm.view(BH, N, N)              # [B*H, N, N]

    # Batched matrix multiply: for each hidden slice compute L @ Rᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(2, 1))  # [B*H, N, N]

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim layer norm, out_gate, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate  # element‑wise broadcast over hidden dim

    # Final linear back to original dimension D
    out = F.linear(out, to_out_weight)  # [B, N, N, D]

    return out
