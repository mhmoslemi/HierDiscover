"""
custom_kernel: Highly‑optimized TriMul “outgoing” forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` using the supplied LN weight/bias.
2. Linear projections (no bias) → left_raw / right_raw.
3. Gating projections → left_gate / right_gate / out_gate (sigmoid).
4. Apply optional mask and element‑wise gate: 
       left = left_raw * left_gate * mask
       right = right_raw * right_gate * mask
5. Compute the core tensor contraction
       out[b, i, j, h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched mat‑mul per head `h`.  Implemented in Triton:
   each program instance handles a (BLOCK_M × BLOCK_N) tile of the
   (i, j) matrix for a specific (batch, head) pair and reduces over k.
6. Layer‑norm the result (`out`) with its own weight/bias.
7. Apply the `out_gate` (element‑wise) and a final linear projection
   back to `dim`.

Only the O(N³) contraction is performed in Triton; the remaining
operations use highly‑optimized PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: compute out = left @ rightᵀ for each (batch, head) pair.
# ----------------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,
    # strides for out: (batch, i, j, head)
    stride_ob, stride_oi, stride_oj, stride_od,
    # strides for left: (batch, i, k, head)
    stride_lb, stride_li, stride_lk, stride_ld,
    # strides for right: (batch, j, k, head)
    stride_rb, stride_ri, stride_rk, stride_rd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program IDs
    pid_m = tl.program_id(0)               # tile row   (i)
    pid_n = tl.program_id(1)               # tile col   (j)
    pid_g = tl.program_id(2)               # batch * head

    # Decode batch and head indices
    b = pid_g // H
    h = pid_g % H

    # Starting coordinates of the tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Linear ranges for the tile
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base pointer for the current (batch, head)
    out_base = out_ptr + b * stride_ob + h * stride_od
    # Pointers to the output tile (M × N)
    out_tile_ptr = out_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in BLOCK_K chunks
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # -----------------
        # Load left tile: shape (M, K)
        # -----------------
        left_base = left_ptr + b * stride_lb + h * stride_ld
        left_ptrs = left_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )
        # -----------------
        # Load right tile: shape (K, N)  (k as rows, j as cols)
        # -----------------
        right_base = right_ptr + b * stride_rb + h * stride_rd
        right_ptrs = right_base + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )
        # Accumulate M×N product
        acc += tl.dot(left, right)   # left: (M,K), right: (K,N) → (M,N)

    # Write back the result
    tl.store(
        out_tile_ptr,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]
        - mask_tensor  : torch.Tensor[B, N, N] or None
        - weights      : dict mapping weight names to torch.Tensor
        - config       : dict containing at least `dim` and `hidden_dim`

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    torch.cuda.empty_cache()  # ensure deterministic launches
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_raw = F.linear(x, weights["left_proj.weight"])
    right_raw = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) projections
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # 4. Apply mask (if present) and gates
    # ------------------------------------------------------------------
    if mask_tensor is None or config.get("nomask", False):
        mask = 1.0
    else:
        # broadcast mask to [..., 1] and match dtype
        mask = mask_tensor.unsqueeze(-1).to(dtype)

    left = left_raw * left_gate * mask
    right = right_raw * right_gate * mask

    # Ensure tensors are contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5. Core Tensor Contraction via Triton
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (element‑wise, not byte)
    stride_ob, stride_oi, stride_oj, stride_od = out.stride()
    stride_lb, stride_li, stride_lk, stride_ld = left.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right.stride()

    # Tile sizes – chosen for H100 (adjustable if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),            # i‑tiles
        triton.cdiv(N, BLOCK_N),            # j‑tiles
        B * hidden_dim,                     # batch * head dimension
    )

    trimul_outgoing_kernel[grid](
        out_ptr=out,
        left_ptr=left,
        right_ptr=right,
        B=int(B),
        N=int(N),
        H=int(hidden_dim),
        stride_ob=int(stride_ob),
        stride_oi=int(stride_oi),
        stride_oj=int(stride_oj),
        stride_od=int(stride_od),
        stride_lb=int(stride_lb),
        stride_li=int(stride_li),
        stride_lk=int(stride_lk),
        stride_ld=int(stride_ld),
        stride_rb=int(stride_rb),
        stride_ri=int(stride_ri),
        stride_rk=int(stride_rk),
        stride_rd=int(stride_rd),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output LayerNorm, out‑gate and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    # Apply out‑gate (shape matches out)
    out = out * out_gate

    # Final linear back to original dimension
    out = F.linear(out, weights["to_out.weight"])

    return out
