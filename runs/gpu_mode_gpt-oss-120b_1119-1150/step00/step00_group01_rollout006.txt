def batched_matmul_triton(a, b):
    # a: [batch, M, K] (contiguous)
    # b: [batch, N, K] (contiguous) (no transposition). We'll compute a @ b.T
    assert a.shape[0] == b.shape[0]
    batch, M, K = a.shape
    _, N, Kb = b.shape
    assert K == Kb

    # We'll transpose b to (batch, K, N) for easier matmul
    b_t = b.transpose(1,2).contiguous()
    out = torch.empty((batch, M, N), dtype=torch.float32, device=a.device)

    # get strides
    a_ptr = a
    b_ptr = b_t
    c_ptr = out

    # compute grid
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = ( (M + BLOCK_M - 1)//BLOCK_M,
             (N + BLOCK_N - 1)//BLOCK_N,
             batch)

    batched_matmul_kernel[grid](
        a_ptr, b_ptr, c_ptr,
        a.stride(0), a.stride(1), a.stride(2),
        b_t.stride(0), b_t.stride(1), b_t.stride(2),
        out.stride(0), out.stride(1), out.stride(2),
        M, N, K,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out

@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_az, stride_am, stride_ak,
    stride_bz, stride_bk, stride_bn,
    stride_cz, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_z = tl.program_id(2) # batch
    pid_m = tl.program_id(0) # block row
    pid_n = tl.program_id(1) # block col

    # offset into dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointer offsets for a and b
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptrs, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)
        # Compute dot product
        acc += tl.dot(a, b)
        # stride to next K block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    # Store only valid positions
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask)

import torch
import triton
import triton.language as tl

def _batched_matmul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication (B, M, K) @ (B, N, K).T -> (B, M, N)
    Implemented in Triton for fast compute. Input a and b may be float16.
    Returns result in float32.
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, _ = b.shape

    # Transpose second argument so that we can compute a @ b^T as a @ b_t where b_t shape (batch, K, N)
    b_t = b.transpose(1, 2).contiguous()

    out = torch.empty((batch, M, N), dtype=torch.float32, device=a.device)

    # strides for a
    stride_az = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    # strides for b_t
    stride_bz = b_t.stride(0)
    stride_bk = b_t.stride(1)
    stride_bn = b_t.stride(2)
    # strides for c
    stride_cz = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        a,
        b_t,
        out,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


@triton.jit
def _batched_matmul_kernel(
    a_ptr,
    b_ptr,
    c_ptr,
    stride_az,
    stride_am,
    stride_ak,
    stride_bz,
    stride_bk,
    stride_bn,
    stride_cz,
    stride_cm,
    stride_cn,
    M,
    N,
    K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Triton kernel for batched matrix multiplication.
    Computes: C[b, i, j] = sum_{k} A[b, i, k] * B[b, j, k]
    where B is provided as (batch, K, N) i.e. transposed version of the original argument.
    """
    pid_z = tl.program_id(2)  # batch
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col

    # Coordinates in the output matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Pointers to the start of the block in A and B
    a_ptrs = a_ptr + pid_z * stride_az + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    b_ptrs = b_ptr + pid_z * stride_bz + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bn

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(
            a_ptrs,
            mask=(offs_m < M)[:, None] & (offs_k < K)[None, :],
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=(offs_k < K)[:, None] & (offs_n < N)[None, :],
            other=0.0,
        )
        # a,b are half (or bf16) -> dot yields FP32
        acc += tl.dot(a, b)
        # Move to the next K-block
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # Write back the result
    c_ptrs = c_ptr + pid_z * stride_cz + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc, mask=mask)


def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing TriMul operator.
    The heavy cubic einsum is implemented as a batched matrix multiplication
    with a custom Triton kernel.
    """
    input_tensor, mask, weights, config = data

    # -------------------------------------------------------------
    # 1. Retrieve config and weights
    # -------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm on the input (over the last dimension)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -------------------------------------------------------------
    # 3. Gating (sigmoid)
    # -------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # -------------------------------------------------------------
    # 4. Apply gates (and optional mask) and cast to fp16 for the kernel
    # -------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not config.get("nomask", False):
        # mask shape (B, N, N) -> (B, N, N, 1)
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Cast to half to reduce memory bandwidth for the cubic step
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -------------------------------------------------------------
    # 5. Compute the O(N^3) contraction with Triton
    #    out[i, j, d] = sum_k left[i, k, d] * right[j, k, d]
    # -------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim

    # Rearrange so that each (batch, hidden) pair becomes one independent matrix
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Batched matmul: (B*H, N, N) @ (B*H, N, N).T -> (B*H, N, N)
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # float32

    # Reshape back to original layout
    out_perm = out_flat.view(B, H, N, N)           # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1)             # (B, N, N, H)

    # -------------------------------------------------------------
    # 6. Final LayerNorm, gating, and linear projection back to dim
    # -------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

The heavy part of the module is the einsum
    out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
which can be rewritten as a batch of matrix multiplications:
    out[b,d] = left[b,d] @ right[b,d].T
where left[b,d] and right[b,d] are N×N matrices (one per batch and hidden
channel).  The kernel below performs this batched mat‑mul in fp16 (inputs)
with fp32 accumulation, achieving a large speed‑up over the naïve PyTorch
implementation.  All other operations (LayerNorm, Linear, Sigmoid, masking)
are done with PyTorch, but the cubic contraction is fused into Triton.
"""

import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr,                # const *fp16  [batch, M, K]
    b_ptr,                # const *fp16  [batch, K, N]   (right matrix already transposed)

        a: (B, M, K)   – fp16 or bf16
        b: (B, N, K)   – fp16 or bf16
    Returns:
        c: (B, M, N)   – fp32
    The operation performed is  c[b] = a[b] @ b[b].T
    """
    assert a.is_contiguous() and b.is_contiguous()
    batch, M, K = a.shape
    _, N, _ = b.shape

    # Transpose the second operand so that we can use the kernel that expects (B, K, N)
    b_t = b.transpose(1, 2).contiguous()

    out = torch.empty((batch, M, N), dtype=torch.float32, device=a.device)

    # Strides for a
    stride_az = a.stride(0)
    stride_am = a.stride(1)
    stride_ak = a.stride(2)
    # Strides for b_t
    stride_bz = b_t.stride(0)
    stride_bk = b_t.stride(1)
    stride_bn = b_t.stride(2)
    # Strides for output
    stride_cz = out.stride(0)
    stride_cm = out.stride(1)
    stride_cn = out.stride(2)

    # Tunable block sizes – good defaults for H100 fp16 GEMM
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        a,
        b_t,
        out,
        stride_az,
        stride_am,
        stride_ak,
        stride_bz,
        stride_bk,
        stride_bn,
        stride_cz,
        stride_cm,
        stride_cn,
        M,
        N,
        K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return out


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.
    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # -----------------------------------------------------------------
    # 1️⃣ unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 2️⃣ Input LayerNorm over the last dimension (dim)
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 3️⃣ Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 5️⃣ Apply gates (and optional mask) – keep fp32 until the final cast
    # -----------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    if not config.get("nomask", False):
        # mask: (B, N, N) → (B, N, N, 1) for broadcasting
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Cast to half to reduce memory traffic for the cubic step
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # -----------------------------------------------------------------
    # 6️⃣ Cubic contraction via Triton (batched mat‑mul)
    #     out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # -----------------------------------------------------------------
    B, N, _, H = left.shape                # H == hidden_dim
    # Move hidden dimension to the batch axis: (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Collapse (B, H) → batch dimension for the kernel
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_perm.view(B * H, N, N) # (B*H, N, N)

    # Perform batched mat‑mul in fp16 (result in fp32)
    out_flat = _batched_matmul_triton(left_flat, right_flat)   # (B*H, N, N) fp32

    # Restore original layout
    out_perm = out_flat.view(B, H, N, N)       # (B, H, N, N)
    out = out_perm.permute(0, 2, 3, 1)        # (B, N, N, H)

    # -----------------------------------------------------------------
    # 7️⃣ Final LayerNorm, output gating and projection back to dim
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
