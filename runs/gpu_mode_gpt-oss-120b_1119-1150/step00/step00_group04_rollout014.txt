@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_left_bh, stride_left_m, stride_left_n,
    stride_right_bh, stride_right_m, stride_right_n,
    stride_out_bh, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bh = tl.program_id(2)  # batch hidden index
    pid_m = tl.program_id(0)   # block row index
    pid_n = tl.program_id(1)   # block col index

    # compute offsets
    off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        off_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # load left block: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + pid_bh * stride_left_bh + off_m[:, None] * stride_left_m + off_k[None,:] * stride_left_n
        left_mask = (off_m < N)[:, None] & (off_k < N)[None,:]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # load right block transposed: shape (BLOCK_K, BLOCK_N)
        # we need right[k, j] for j in off_n; original right layout (rows=j, cols=k)
        right_ptrs = right_ptr + pid_bh * stride_right_bh + off_k[:, None] * stride_right_n + off_n[None, :] * stride_right_m
        # Explanation: we treat original right stride: row stride = stride_right_m (over 'rows' dimension j?) Actually original shape is (N rows aka j), (N cols aka k). So row stride = stride_right_m (for j), col stride = stride_right_n (for k).
        # For transposed version (k as row, j as col), row stride = stride_right_n, col stride = stride_right_m.
        right_mask = (off_k < N)[:, None] & (off_n < N)[None,:]
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # accumulate
        acc += tl.dot(left, right)  # left (M,K), right (K,N) -> (M,N)
    # store result
    out_ptrs = out_ptr + pid_bh * stride_out_bh + off_m[:, None] * stride_out_m + off_n[None,:] * stride_out_n
    out_mask = (off_m < N)[:, None] & (off_n < N)[None,:]

    tl.store(out_ptrs, acc, mask=out_mask)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for batched matmul (left @ right^T) over hidden dimensions
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_hidden, N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_m, stride_r_k,
    stride_o_batch, stride_o_m, stride_o_k,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out = left @ right^T for each (batch_hidden) slice.
    left/right/out shapes: (batch_hidden, N, N)
    left indexed as (i,k), right as (j,k). The kernel loads blocks
    of size BLOCK_M x BLOCK_K from left and BLOCK_K x BLOCK_N from the
    transposed view of right, accumulating into a BLOCK_M x BLOCK_N tile.
    """
    pid_batch = tl.program_id(2)
    pid_m = tl.program_id(0)   # block row index (i)
    pid_n = tl.program_id(1)   # block col index (j)

    # Offsets for rows and cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Load left block (M x K)
        left_ptrs = left_ptr + pid_batch * stride_l_batch + offs_m[:, None] * stride_l_m + offs_k[None, :] * stride_l_k
        mask_l = (offs_m < N)[:, None] & (offs_k < N)[None, :]
        left = tl.load(left_ptrs, mask=mask_l, other=0.0)

        # Load right block transposed (K x N)
        # Original right layout: (row=j, col=k). We want (k,j)
        right_ptrs = right_ptr + pid_batch * stride_r_batch + offs_k[:, None] * stride_r_k + offs_n[None, :] * stride_r_m
        mask_r = (offs_k < N)[:, None] & (offs_n < N)[None, :]
        right = tl.load(right_ptrs, mask=mask_r, other=0.0)

        # Accumulate dot product
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # Store result
    out_ptrs = out_ptr + pid_batch * stride_o_batch + offs_m[:, None] * stride_o_m + offs_n[None, :] * stride_o_k
    mask_o = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(out_ptrs, acc, mask=mask_o)

# Main entry point
def custom_kernel(data):
    """
    Triton implementation of the outgoing Triangle Multiplicative Update (TriMul).
    Steps:
    1. Layer‑norm the input tensor.
    2. Linear projections (left, right) and gating (left_gate, right_gate, out_gate).
    3. Apply optional mask and gates to left/right.
    4. Compute out[i,j] = Σ_k left[i,k] * right[j,k] for each hidden dimension
       using a custom Triton kernel (batched matmul with transposed RHS).
    5. Layer‑norm over hidden dimension, multiply by out_gate, and final linear to dim.
    """
    # Unpack
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -------------------------------------------------------------------------
    # 1. Input layer‑norm
    # -------------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -------------------------------------------------------------------------
    # 2. Linear projections & gates
    # -------------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Optional masking
    if not config.get("nomask", False):
        # mask shape [B, N, N]; broadcast to last dim
        mask_f = mask.unsqueeze(-1).to(dtype)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 3. Batched matmul left @ right.T using Triton
    # -------------------------------------------------------------------------
    # Rearrange to (batch_hidden, N, N) where batch_hidden = B * hidden_dim
    left_t = left.permute(0, 3, 1, 2).contiguous()
    right_t = right.permute(0, 3, 1, 2).contiguous()
    B, N, _, _ = input_tensor.shape  # note: dim may differ
    batch_hidden = left_t.shape[0]   # B * hidden_dim

    # Allocate output tensor
    out_t = torch.empty_like(left_t)

    # Strides (int)
    stride_l_batch, stride_l_m, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_m, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_m, stride_o_k = out_t.stride()

    # Tile sizes – chosen to fit H100 shared memory and provide good occupancy
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch_hidden,
    )

    tri_mul_kernel[grid](
        left_t,
        right_t,
        out_t,
        batch_hidden,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------
# Triton kernel: batched matmul where the RHS is logically transposed.
# Input/Output shape per batch‑hidden slice : (N, N)
#   out[bh, i, j] = Σ_k left[bh, i, k] * right[bh, j, k]
# --------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,          # const float*   left  (batch_hidden, N, N)
    right_ptr,         # const float*   right (batch_hidden, N, N)
    out_ptr,           # float*         out   (batch_hidden, N, N)
    batch_hidden,      # i32            total # of (batch × hidden) slices
    N,                 # i32            sequence length
    stride_l_batch,    # i32            stride for batch_hidden dim of left
    stride_l_m,        # i32            stride for first N dim (i) of left
    stride_l_k,        # i32            stride for second N dim (k) of left
    stride_r_batch,    # i32            stride for batch_hidden dim of right
    stride_r_m,        # i32            stride for first N dim (j) of right
    stride_r_k,        # i32            stride for second N dim (k) of right
    stride_o_batch,    # i32            stride for batch_hidden dim of out
    stride_o_m,        # i32            stride for first N dim (i) of out
    stride_o_k,        # i32            stride for second N dim (j) of out
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute out = left @ rightᵀ for every (batch, hidden) pair.

    The tensors are laid out as (batch_hidden, N, N).  For each tile we load
    a BLOCK_M×BLOCK_K block from `left` and a BLOCK_K×BLOCK_N block from the
    transposed view of `right` (original layout (j,k) → we access (k,j)).
    The partial products are accumulated in FP32 and written back.
    """
    pid_bh = tl.program_id(2)                     # which (batch, hidden) slice
    pid_m = tl.program_id(0)                      # tile row index (i)
    pid_n = tl.program_id(1)                      # tile col index (j)

    # Offsets of the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    num_k = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ------- left block (M × K) -------
        left_ptrs = (
            left_ptr
            + pid_bh * stride_l_batch
            + offs_m[:, None] * stride_l_m
            + offs_k[None, :] * stride_l_k
        )
        mask_l = (offs_m < N)[:, None] & (offs_k < N)[None, :]
        left = tl.load(left_ptrs, mask=mask_l, other=0.0)

        # ------- right block transposed (K × N) -------
        # Original layout of right: (j, k).  We need (k, j) → swap the two strides.
        right_ptrs = (
            right_ptr
            + pid_bh * stride_r_batch
            + offs_k[:, None] * stride_r_k   # k becomes the row index
            + offs_n[None, :] * stride_r_m   # j becomes the column index
        )
        mask_r = (offs_k < N)[:, None] & (offs_n < N)[None, :]
        right = tl.load(right_ptrs, mask=mask_r, other=0.0)

        # ----- accumulate product -----
        acc += tl.dot(left, right)   # (M,K) @ (K,N) -> (M,N)

    # ------- write result -------
    out_ptrs = (
        out_ptr
        + pid_bh * stride_o_batch
        + offs_m[:, None] * stride_o_m
        + offs_n[None, :] * stride_o_k
    )
    mask_o = (offs_m < N)[:, None] & (offs_n < N)[None, :]
    tl.store(out_ptrs, acc, mask=mask_o)

def custom_kernel(data):
    """
    Outgoing TriMul (AlphaFold‑3) forward pass.

    The implementation follows the reference PyTorch module but replaces the
    cubic O(N³) tensor contraction with a custom Triton kernel that performs
    a batched matrix multiplication of shape (B·H, N, N) @ (B·H, N, N)ᵀ.
    All other operations (LayerNorm, linear projections, gating) are kept in
    PyTorch for simplicity and numerical stability.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                # input‐/output feature dimension
    hidden_dim = config["hidden_dim"]  # internal hidden dimension
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    if not config.get("nomask", False):
        # mask shape: [B, N, N] → broadcast to last dim
        mask_f = mask.unsqueeze(-1).to(dtype)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 3. Gates (sigmoid activations)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Batched matmul left @ rightᵀ using Triton
    # ------------------------------------------------------------------
    # reshape to (batch_hidden, N, N) where batch_hidden = B * hidden_dim
    # layout after permute: (B, hidden_dim, N, N) → contiguous
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()
    B, N, _, _ = input_tensor.shape      # input shape: [B, N, N, dim]
    batch_hidden = B * hidden_dim

    # allocate output tensor
    out_t = torch.empty_like(left_t)

    # strides for the flattened (batch_hidden, N, N) tensors
    stride_l_batch, stride_l_m, stride_l_k = left_t.stride()
    stride_r_batch, stride_r_m, stride_r_k = right_t.stride()
    stride_o_batch, stride_o_m, stride_o_k = out_t.stride()

    # Tile sizes – tuned for H100 (shared‑memory friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks over the i‑dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks over the j‑dimension
        batch_hidden,                   # one program per (batch, hidden) slice
    )

    tri_mul_kernel[grid](
        left_t,
        right_t,
        out_t,
        batch_hidden,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_m,
        stride_r_k,
        stride_o_batch,
        stride_o_m,
        stride_o_k,
        BLOCK_M,
        BLOCK_N,
        BLOCK_K,
    )

    # reshape back to the original layout [B, N, N, hidden_dim]
    out = out_t.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # → [B, N, N, dim]

    return out

**Explanation of the algorithm**

1. **Layer‑norm the input** – identical to the reference implementation.
2. **Project to a hidden space** (`left_proj`, `right_proj`) and optionally apply the supplied mask.
3. **Compute three sigmoid gates** (`left_gate`, `right_gate`, `out_gate`) and multiply the projected tensors by the corresponding gates.
4. **Core TriMul contraction**  
   The contraction  
   `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`  
   is a batch of independent matrix multiplications `L_d @ R_dᵀ`.  
   By reshaping to `(B·H, N, N)` we launch a Triton kernel where each
   program instance works on a tile `(BLOCK_M, BLOCK_N)` of the output matrix.
   The kernel loads a `(BLOCK_M, BLOCK_K)` tile from `left` and a
   `(BLOCK_K, BLOCK_N)` tile from the *transposed* view of `right`,
   accumulates the dot‑product in FP32 and finally stores the result.
5. **Post‑processing** – Layer‑norm over the hidden dimension, element‑wise
   multiplication by `out_gate`, and a final linear projection back to the
   original feature dimension.

All heavy‑compute work (the `O(N³)` contraction) is performed on the GPU by the custom Triton kernel, while the surrounding linear layers, normalizations and gating use highly‑optimized PyTorch primitives. This fulfills the requirement of implementing “part of the operations” in Triton while keeping the overall implementation correct and efficient for the provided test configurations.
