"""
TriMul “outgoing” forward pass with Triton acceleration.

Algorithm
---------
1. Normalise the input 4‑D tensor (B,N,N,C) with a custom Triton LayerNorm.
2. Linear projections (left/right) and three gating projections are performed
   with torch.nn.functional.linear (no bias). The gating logits are passed
   through a sigmoid.
3. An optional mask (B,N,N) is broadcast and applied to the left/right
   projections, after which the element‑wise gates are multiplied in.
4. Core multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is mathematically equivalent to a batched matrix multiplication
   per hidden channel: for every hidden index `h`,
        out_h = left_h @ right_hᵀ .
   We realise it with a single torch.matmul on tensors of shape
   (B, hidden, N, N), which maps to a highly‑optimised cuBLAS kernel.
5. A second LayerNorm (on the hidden dimension) is applied via the same
   Triton kernel, followed by multiplication with the out‑gate.
6. Finally a linear projection maps the hidden dimension back to the original
   channel size.

Only the LayerNorm is implemented in Triton; the remaining dense operations
are delegated to PyTorch’s fast BLAS kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton LayerNorm over the last dimension of a 4‑D tensor
# ------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr, w_ptr, b_ptr,            # pointers
    B, N, D, eps,                          # runtime scalars
    stride_xb, stride_xi, stride_xj, stride_xd,
    stride_yb, stride_yi, stride_yj, stride_yd,
    BLOCK_D: tl.constexpr                  # compile‑time constant
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # decode flat pid -> (b,i,j)
    b = pid // (N * N)
    rem = pid % (N * N)
    i = rem // N
    j = rem % N

    # base offsets for the vector (b,i,j,:)
    x_base = b * stride_xb + i * stride_xi + j * stride_xj
    y_base = b * stride_yb + i * stride_yi + j * stride_yj

    # ---------- first pass: compute mean & variance ----------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)
    num_blocks = tl.cdiv(D, BLOCK_D)
    for block in range(num_blocks):
        d = block * BLOCK_D + tl.arange(0, BLOCK_D)          # [BLOCK_D]
        mask = d < D
        x = tl.load(x_ptr + x_base + d * stride_xd, mask=mask).to(tl.float32)
        sum_val += tl.sum(x)
        sum_sq_val += tl.sum(x * x)

    mean = sum_val / D
    var = sum_sq_val / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ---------- second pass: normalise and apply affine ----------
    for block in range(num_blocks):
        d = block * BLOCK_D + tl.arange(0, BLOCK_D)
        mask = d < D
        x = tl.load(x_ptr + x_base + d * stride_xd, mask=mask).to(tl.float32)
        w = tl.load(w_ptr + d, mask=mask).to(tl.float32)
        b_ = tl.load(b_ptr + d, mask=mask).to(tl.float32)
        y = (x - mean) * inv_std          # normalisation
        y = y * w + b_                    # scale & shift
        tl.store(y_ptr + y_base + d * stride_yd, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias:   torch.Tensor,
                     eps: float = 1e-5,
                     block_dim: int = 32) -> torch.Tensor:

    Apply LayerNorm over the last dimension of a 4‑D tensor using Triton.
    The function assumes a contiguous layout (B, N, N, C).
    """

    B, N, N2, C = x.shape
    assert N == N2, "input must be square in the two sequence axes"

        x, out,
        weight, bias,
        B, N, C, eps,
        sxb, sxi, sxj, sxd,
        syb, syi, syj, syd,
        BLOCK_D=block_dim,
    )

    right_perm = right.permute(0, 3, 1, 2)  # (B, hidden, N, N)

    # batch matmul: (B, hidden, N, N) @ (B, hidden, N, N)ᵀ  -> (B, hidden, N, N)
    out_perm = torch.matmul(left_perm, right_perm.transpose(-2, -1))
    out = out_perm.permute(0, 2, 3, 1)      # back to [B, N, N, hidden]

    # ------------------------------------------------------------------
    # 6) Hidden‑dim LayerNorm (Triton) + out‑gate
    # ------------------------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps)
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    output = F.linear(out_norm, to_out_weight)   # (dim, hidden) weight → [B,N,N,dim]

    return output
```
