```
    X_ptr, Y_ptr,
    weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xn, stride_xc,
    stride_yb, stride_yn, stride_yc,
    eps,
    BLOCK_SIZE: tl.constexpr,
    BLOCK_C: tl.constexpr):

    pid = tl.program_id(0)

    idx = pid * BLOCK_SIZE + bs
    mask = idx < total
    # compute coordinates
    b = idx // (N * N)
    rem = idx % (N * N)
    i = rem // N
    j = rem % N

    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for off_c in range(0, C, BLOCK_C):
        c_off = off_c + tl.arange(0, BLOCK_C)
        # compute pointer offsets
        x_ptrs = X_ptr + b[:,None]*stride_xb + i[:,None]*stride_xn + j[:,None]*stride_xn + c_off[None,:]*stride_xc
        x = tl.load(x_ptrs, mask=mask[:,None] & (c_off[None,:] < C), other=0.0)
        sum_val += tl.sum(x, axis=1) # sum across C block -> shape BLOCK_SIZE
        sum_sq += tl.sum(x * x, axis=1)
    # after loop, compute mean and var
    mean = sum_val / C
    var = sum_sq / C - mean * mean

    for off_c in range(0, C, BLOCK_C):

        x_ptrs = X_ptr + b[:,None]*stride_xb + i[:,None]*stride_xn + j[:,None]*stride_xn + c_off[None,:]*stride_xc
```

```
# Triton LayerNorm kernel
@triton.jit
def layernorm_fwd(
    X_ptr, Y_ptr, weight_ptr, bias_ptr,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    eps,
    BLOCK_SIZE: tl.constexpr, BLOCK_C: tl.constexpr
):
    pid = tl.program_id(0)
    total = B * N * N
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_sq = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        c_mask = c < C
        ptrs = X_ptr + (b[:,None] * stride_xb) + (i[:,None] * stride_xi) + (j[:,None] * stride_xj) + (c[None,:] * stride_xc)
```

```
def custom_kernel(data):
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ensure tensors contiguous
    input_tensor = input_tensor.contiguous()
    if mask is not None:
        mask = mask.contiguous()
    # extract weights
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()
    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()
    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()
    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # -------------------------------------------------
    # First LayerNorm (dim)
    # -------------------------------------------------
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps, block_size=128, block_c=32)

    B, N, _, _ = x_norm.shape

    # -------------------------------------------------
    # Linear projections and gates (float32)
    # -------------------------------------------------
    flat = x_norm.view(-1, dim)  # (B*N*N, dim)

    left_proj = torch.nn.functional.linear(flat, left_proj_weight)  # (B*N*N, hidden)
    right_proj = torch.nn.functional.linear(flat, right_proj_weight)
    left_gate_p = torch.nn.functional.linear(flat, left_gate_weight)
    right_gate_p = torch.nn.functional.linear(flat, right_gate_weight)
    out_gate = torch.nn.functional.linear(flat, out_gate_weight)

    left_gate = left_gate_p.sigmoid()
    right_gate = right_gate_p.sigmoid()
    out_gate = out_gate.sigmoid()

    # reshape
    left = left_proj.view(B, N, N, hidden_dim)
    right = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    if mask is not None and not config.get("nomask", True):
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # Compute pairwise product_sum (einsum) via batched GEMM
    # -------------------------------------------------
    # permute to (B, hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # optional half precision for speed
    left_hp = left_perm.to(torch.float16)
    right_hp = right_perm.to(torch.float16)

    # flatten batch & hidden
    left_flat = left_hp.view(-1, N, N)
    right_flat = right_hp.view(-1, N, N)

    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*hidden, N, N)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -------------------------------------------------
    # Output layernorm (hidden_dim)
    # -------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps, block_size=128, block_c=32)

    # -------------------------------------------------
    # Apply out gate
    # -------------------------------------------------
    out_norm = out_norm * out_gate

    # -------------------------------------------------
    # Final linear to original dim
    # -------------------------------------------------
    out_flat2 = out_norm.view(-1, hidden_dim)
    output = torch.nn.functional.linear(out_flat2, to_out_weight)  # shape (B*N*N, dim)
    output = output.view(B, N, N, dim)

    return output
```

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for LayerNorm (per token) w/ weight and bias
@triton.jit
def layernorm_fwd(
    X_ptr, Y_ptr, weight_ptr, bias_ptr,
    B, N,                     # batch size and sequence length
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    eps,
    C: tl.constexpr,          # channel dimension (dim or hidden_dim)
    BLOCK_SIZE: tl.constexpr, # number of positions per program
    BLOCK_C: tl.constexpr     # channel tile size
):
    # compute flattened position index
    pid = tl.program_id(0)
    total = B * N * N
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # recover (b,i,j) coordinates
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # accumulate sum and sum of squares across channel dimension C
    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        c_valid = c < C
        # pointer to X[b,i,j,c]
        x_ptrs = X_ptr + (b[:, None] * stride_xb) \
                        + (i[:, None] * stride_xi) \
                        + (j[:, None] * stride_xj) \
                        + (c[None, :] * stride_xc)
        x = tl.load(x_ptrs,
                    mask=mask[:, None] & c_valid[None, :],
                    other=0.0)
        sum_val += tl.sum(x, axis=1)
        sum_sq += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # second pass: write normalized values with affine weight/bias
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        c_valid = c < C
        x_ptrs = X_ptr + (b[:, None] * stride_xb) \
                        + (i[:, None] * stride_xi) \
                        + (j[:, None] * stride_xj) \
                        + (c[None, :] * stride_xc)
        x = tl.load(x_ptrs,
                    mask=mask[:, None] & c_valid[None, :],
                    other=0.0)
        norm = (x - mean[:, None]) * inv_std[:, None]

        w = tl.load(weight_ptr + c, mask=c_valid, other=0.0)
        b_val = tl.load(bias_ptr + c, mask=c_valid, other=0.0)

        y = norm * w[None, :] + b_val[None, :]

        y_ptrs = Y_ptr + (b[:, None] * stride_yb) \
                        + (i[:, None] * stride_yi) \
                        + (j[:, None] * stride_yj) \
                        + (c[None, :] * stride_yc)
        tl.store(y_ptrs,
                 y,
                 mask=mask[:, None] & c_valid[None, :])

    Triton‑accelerated implementation of the “outgoing” TriMul operator.

    The algorithm follows the reference PyTorch code:
    1. Layer‑norm over ``dim`` (Triton).
    2. Linear projections for left/right and three gates (torch.nn.functional.linear).
    3. Optional mask multiplication.
    4. Apply gates to the projected tensors.
    5. Compute the pairwise contraction
         out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
       using a batch of GEMMs (one per hidden channel).
    6. Layer‑norm over the hidden dimension (Triton).
    7. Multiply by the output gate and a final linear projection back to ``dim``.
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # make everything contiguous for simplicity
    input_tensor = input_tensor.contiguous()
    if mask is not None:
        mask = mask.contiguous()

    # ------------------------------------------------------------------
    # retrieve the model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias   = weights["norm.bias"].contiguous()

    left_proj_weight  = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()

    left_gate_weight  = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight   = weights["out_gate.weight"].contiguous()

    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias   = weights["to_out_norm.bias"].contiguous()
    to_out_weight      = weights["to_out.weight"].contiguous()

    # ------------------------------------------------------------------
    # 1. Layer‑norm on the input features (dim)
    # ------------------------------------------------------------------
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias,
                              eps=eps, block_size=128, block_c=32)

    B, N, _, _ = x_norm.shape

    # ------------------------------------------------------------------
    # 2. Linear projections & gating (float32)
    # ------------------------------------------------------------------
    # reshape to (B*N*N, dim) for fast linear layers
    x_flat = x_norm.view(-1, dim)                                 # (BNN, dim)

    left_proj   = F.linear(x_flat, left_proj_weight)              # (BNN, hidden)
    right_proj  = F.linear(x_flat, right_proj_weight)
    left_gate   = torch.sigmoid(F.linear(x_flat, left_gate_weight))
    right_gate  = torch.sigmoid(F.linear(x_flat, right_gate_weight))
    out_gate    = torch.sigmoid(F.linear(x_flat, out_gate_weight))

    # reshape back to (B, N, N, hidden)
    left    = left_proj.view(B, N, N, hidden_dim)
    right   = right_proj.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", True):
        mask_f = mask.to(x_norm.dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Apply gating
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Pairwise contraction via batched GEMM (one GEMM per hidden channel)
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N) and cast to half for speed
    left_perm  = left.permute(0, 3, 1, 2).contiguous().to(torch.float16)   # (B, hidden, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous().to(torch.float16)

    # fold batch & hidden into a single batch dimension for torch.bmm
    left_flat  = left_perm.view(-1, N, N)      # (B*hidden, N, N)
    right_flat = right_perm.view(-1, N, N)

    # GEMM: (B*hidden, N, N) @ (B*hidden, N, N)^T  -> (B*hidden, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden) and convert to fp32 for the next steps
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6. Layer‑norm over hidden dimension (Triton)
    # ------------------------------------------------------------------
    out_norm = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias,
                               eps=eps, block_size=128, block_c=32)

    # ------------------------------------------------------------------
    # 7. Output gating and final linear projection back to dim
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate

    out_flat2 = out_norm.view(-1, hidden_dim)          # (B*N*N, hidden)
    output = F.linear(out_flat2, to_out_weight)       # (B*N*N, dim)
    output = output.view(B, N, N, dim)

    return output

"""
Triton‑accelerated implementation of the “outgoing” TriMul operator.

Algorithm (mirrors the reference PyTorch version):
1️⃣  Layer‑norm over the input feature dimension (dim) – implemented in Triton.
2️⃣  Linear projections for left/right tensors and three gate tensors (no bias).
3️⃣  Optional mask multiplication (mask is broadcast over the hidden dimension).
4️⃣  Apply gates to the projected tensors.
5️⃣  Compute the pairwise contraction
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   using a batch of GEMMs (one GEMM per hidden channel).  The hidden dimension
   is folded into the batch dimension, the GEMMs run in float16 for speed,
   and the result is cast back to float32.
6️⃣  Layer‑norm over the hidden dimension (hidden_dim) – also implemented in Triton.
7️⃣  Multiply by the output gate and apply the final linear projection back to
    the original dimension (dim).

Only the two LayerNorm passes are written in Triton; the remaining linear
operations use highly‑optimized PyTorch kernels, which together deliver a fast
forward pass suitable for the large N² × N patterns used in AlphaFold‑3.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: per‑token LayerNorm with affine weight & bias
# ---------------------------------------------------------------------------
@triton.jit
def layernorm_fwd(
    X_ptr, Y_ptr, weight_ptr, bias_ptr,
    B, N,                                   # batch size, sequence length
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    eps,
    C: tl.constexpr,                        # channel dimension (dim or hidden_dim)
    BLOCK_SIZE: tl.constexpr,               # #positions processed by one program
    BLOCK_C: tl.constexpr                   # channel tile size
):
    """
    Normalises X over its last dimension C:
        Y = ((X - μ) / σ) * weight + bias
    where μ,σ are computed per (b,i,j) entry.
    """
    pid = tl.program_id(0)
    total = B * N * N
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total

    # Decode flattened index → (b,i,j)
    b = offs // (N * N)
    rem = offs % (N * N)
    i = rem // N
    j = rem % N

    # ----------------- 1️⃣  Compute mean & variance -----------------
    sum_val = tl.zeros([BLOCK_SIZE], dtype=tl.float32)
    sum_sq  = tl.zeros([BLOCK_SIZE], dtype=tl.float32)

    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        c_valid = c < C
        # pointer → X[b,i,j,c]
        x_ptrs = X_ptr + (b[:, None] * stride_xb) \
                        + (i[:, None] * stride_xi) \
                        + (j[:, None] * stride_xj) \
                        + (c[None, :] * stride_xc)
        x = tl.load(x_ptrs,
                    mask=mask[:, None] & c_valid[None, :],
                    other=0.0)
        sum_val += tl.sum(x, axis=1)
        sum_sq  += tl.sum(x * x, axis=1)

    mean = sum_val / C
    var  = sum_sq / C - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ----------------- 2️⃣  Write normalised + affine output -------------
    for c0 in range(0, C, BLOCK_C):
        c = c0 + tl.arange(0, BLOCK_C)
        c_valid = c < C
        x_ptrs = X_ptr + (b[:, None] * stride_xb) \
                        + (i[:, None] * stride_xi) \
                        + (j[:, None] * stride_xj) \
                        + (c[None, :] * stride_xc)
        x = tl.load(x_ptrs,
                    mask=mask[:, None] & c_valid[None, :],
                    other=0.0)
        norm = (x - mean[:, None]) * inv_std[:, None]

        w = tl.load(weight_ptr + c, mask=c_valid, other=0.0)
        b_val = tl.load(bias_ptr + c, mask=c_valid, other=0.0)

        y = norm * w[None, :] + b_val[None, :]

        y_ptrs = Y_ptr + (b[:, None] * stride_yb) \
                        + (i[:, None] * stride_yi) \
                        + (j[:, None] * stride_yj) \
                        + (c[None, :] * stride_yc)
        tl.store(y_ptrs,
                 y,
                 mask=mask[:, None] & c_valid[None, :])


def layernorm_triton(x, weight, bias, eps=1e-5,
                    block_size: int = 128, block_c: int = 32):
    """
    Helper that launches the Triton LayerNorm kernel.
    x       : [B, N, N, C]   (float32)
    weight  : [C]            (float32)
    bias    : [C]            (float32)
    returns : tensor of same shape as x, already normalised.
    """
    assert x.is_contiguous(), "input must be contiguous"
    B, N, _, C = x.shape
    y = torch.empty_like(x)

    total = B * N * N
    grid = (triton.cdiv(total, block_size),)

    layernorm_fwd[grid](
        x, y,
        weight, bias,
        B, N,
        x.stride(0), x.stride(1), x.stride(2), x.stride(3),
        y.stride(0), y.stride(1), y.stride(2), y.stride(3),
        eps,
        C=C,                     # compile‑time constant for loop unrolling
        BLOCK_SIZE=block_size,
        BLOCK_C=block_c,
    )
    return y

    out_norm = out_norm * out_gate

    output = F.linear(out_flat2, to_out_weight)     # (B·N·N, dim)
    output = output.view(B, N, N, dim)
