```python
@triton.jit
def tri_mul_matmul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                          N,
                          stride_l_batch, stride_l_m, stride_l_k,
                          stride_r_batch, stride_r_k, stride_r_n,
                          stride_o_batch, stride_o_m, stride_o_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # compute mask for out of bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # pointer to start of each batch
    lhs_batch_ptr = lhs_ptr + batch_id * stride_l_batch
    rhs_batch_ptr = rhs_ptr + batch_id * stride_r_batch
    out_batch_ptr = out_ptr + batch_id * stride_o_batch

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load lhs: shape [BLOCK_M, BLOCK_K]
        lhs = tl.load(lhs_batch_ptr + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k),
                      mask=mask_m[:, None] & mask_k[None, :],
                      other=0.0)

        # load rhs transposed: we need rhs[k, j] which corresponds to rhs_ptr + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n)
        rhs = tl.load(rhs_batch_ptr + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n),
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0)

        # accumulate
        acc += tl.dot(lhs, rhs)

    # store result
    out = out_batch_ptr + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(out, acc, mask=mask_m[:, None] & mask_n[None, :])
```

We need stride definitions:

- For lhs_flat (batch, N, N), stride_l_batch = N * N, stride_l_m = N (row stride), stride_l_k = 1 (col stride). We can derive from .stride.

- For rhs_flat (batch, N, N) but we want its transposed view while reading. To avoid physically transposing, we can interpret (k, n) as offset with stride_r_k (row stride) = N, stride_r_n = 1. That is original layout, but we want to read as (k, j), which corresponds to that.

Our parameters: stride_r_batch = N*N, stride_r_k = N (since it's row stride moving across k dimension (first N), i.e., original row dimension), stride_r_n = 1.

The out stride: out_flat shape same as lhs; out_stride batch = N*N, out_stride_m = N, out_stride_n = 1.

    batch, N, _ = left_flat.shape
    out_flat = torch.empty_like(left_flat)
    # compute strides
    stride_l_batch, stride_l_m, stride_l_k = left_flat.stride()
    stride_r_batch, stride_r_k, stride_r_n = right_flat.stride()
    stride_o_batch, stride_o_m, stride_o_n = out_flat.stride()

    # block sizes
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))
    tri_mul_matmul_kernel[grid](

        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    return out_flat

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # get device
    device = input_tensor.device
    dtype = input_tensor.dtype

    # config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply mask if needed
    if not nomask and mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask_exp
        right = right * mask_exp

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Permute to (B, hidden_dim, N, N) and contiguously
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # Flatten batch and hidden dims for batched matmul
    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # Compute batch matmul via Triton: out = left @ right^T
    # Cast to fp16 for faster compute (optional)
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    # Triton matmul kernel launch
    _triton_bmm(left_fp16, right_fp16, out_fp16)

    # Back to fp32
    out = out_fp16.to(torch.float32)

    # Reshape back to (B, N, N, hidden_dim)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate
    out = out * out_gate

    # Final linear to dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)

    return out

@triton.jit
def tri_mul_matmul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                          N,
                          stride_l_batch, stride_l_m, stride_l_k,
                          stride_r_batch, stride_r_k, stride_r_n,
                          stride_o_batch, stride_o_m, stride_o_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # offsets within N
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # pointers to start of batch
    lhs_base = lhs_ptr + batch_id * stride_l_batch
    rhs_base = rhs_ptr + batch_id * stride_r_batch
    out_base = out_ptr + batch_id * stride_o_batch

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # load lhs[i, k]
        lhs = tl.load(lhs_base + (offs_m[:, None] * stride_l_m) + (offs_k[None, :] * stride_l_k),
                      mask=mask_m[:, None] & mask_k[None, :],
                      other=0.0,
                      # load as float16 for compute (assuming input dtype half)
                      # For auto detection, we use dtype=tl.float16
                      # But we can rely on default dtype (based on pointer type)
                     )
        # load rhs[k, j] (transposed view)
        rhs = tl.load(rhs_base + (offs_k[:, None] * stride_r_k) + (offs_n[None, :] * stride_r_n),
                      mask=mask_k[:, None] & mask_n[None, :],
                      other=0.0)

        # dot product
        # lhs and rhs are either fp16 or fp32; we ensure accumulate in fp32
        acc += tl.dot(lhs, rhs).to(tl.float32)

    # store result
    out_ptrs = out_base + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul outgoing kernel (AlphaFold3) – forward pass only.

Algorithm
---------
1. Layer‑norm the input tensor X ∈ ℝ^{B×N×N×C}.
2. Linear projections (no bias) → left, right ∈ ℝ^{B×N×N×H}.
3. Compute three sigmoid gates (left, right, out) from X.
4. If a mask is provided (nomask=False) zero‑out left/right entries.
5. Apply the element‑wise gates to left and right.
6. Perform the “TriMul” contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which can be written as a batched matrix‑multiply:
       out_{b,h} = left_{b,h} @ right_{b,h}ᵀ
   The heavy O(N³·H) work is done with a custom Triton kernel that
   executes a tiled GEMM for each (batch,head) pair.
7. Layer‑norm the intermediate tensor over the hidden dimension H.
8. Apply the out‑gate and a final linear projection back to C.

Only the core contraction (step 6) is implemented in Triton; all
auxiliary operations (layer‑norm, linear layers, gating, mask handling)
use regular PyTorch for simplicity and correctness.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul left @ rightᵀ for tensors of shape
# (B·H, N, N).  Input and output are half‑precision; the final result
# is cast back to FP32 in Python.
# ----------------------------------------------------------------------
@triton.jit
def _tri_mul_matmul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,            # pointers
    N,                                    # size of the square matrices
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program IDs
    batch_id = tl.program_id(0)                 # 0‑th axis → (batch * hidden)
    pid_m = tl.program_id(1)                    # row block
    pid_n = tl.program_id(2)                    # col block

    # offsets inside the N×N matrix
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # out‑of‑bounds masks
    mask_m = offs_m < N
    mask_n = offs_n < N

    # base pointers for this batch entry
    lhs_base = lhs_ptr + batch_id * stride_l_batch
    rhs_base = rhs_ptr + batch_id * stride_r_batch
    out_base = out_ptr + batch_id * stride_o_batch

    # accumulator (FP16 – matches the data type of the inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float16)

    # tiled K‑loop
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # lhs[i,k]  (shape BLOCK_M × BLOCK_K)
        lhs = tl.load(
            lhs_base
            + (offs_m[:, None] * stride_l_m)
            + (offs_k[None, :] * stride_l_k),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )

        # rhs[k,j] – we read the transposed view of rhs (shape BLOCK_K × BLOCK_N)
        rhs = tl.load(
            rhs_base
            + (offs_k[:, None] * stride_r_k)
            + (offs_n[None, :] * stride_r_n),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )

        # accumulate the dot product
        acc += tl.dot(lhs, rhs)   # fp16 → fp16 accumulation

    # write the result
    out_ptrs = out_base + (offs_m[:, None] * stride_o_m) + (offs_n[None, :] * stride_o_n)
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _triton_batched_matmul(lhs: torch.Tensor, rhs: torch.Tensor, out: torch.Tensor,
                           BLOCK_M=64, BLOCK_N=64, BLOCK_K=32):
    """
    Batched GEMM for tensors (B·H, N, N) → out (B·H, N, N)
    lhs, rhs, out must be contiguous and have identical dtype (float16).
    """
    batch, N, _ = lhs.shape
    grid = (batch, triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N))

    # strides for the three tensors (as returned by .stride())
    stride_l_batch, stride_l_m, stride_l_k = lhs.stride()
    stride_r_batch, stride_r_k, stride_r_n = rhs.stride()
    stride_o_batch, stride_o_m, stride_o_n = out.stride()

    _tri_mul_matmul_kernel[grid](
        lhs,
        rhs,
        out,
        N,
        stride_l_batch,
        stride_l_m,
        stride_l_k,
        stride_r_batch,
        stride_r_k,
        stride_r_n,
        stride_o_batch,
        stride_o_m,
        stride_o_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )


# ----------------------------------------------------------------------
# Public entry point used by the grader
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, C] (float32)
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict of model parameters (see code)
            - config: dict containing at least "dim", "hidden_dim", and optional "nomask"

    Returns:
        torch.Tensor with shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]       # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]     # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Apply element‑wise gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) TriMul contraction – batched matmul with Triton
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) then flatten B·H as the batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    left_flat = left_perm.view(-1, N, N)   # (B·H, N, N)
    right_flat = right_perm.view(-1, N, N)

    # compute in fp16 for speed, then cast back
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    _triton_batched_matmul(left_fp16, right_fp16, out_fp16)

    # back to fp32 and restore original shape
    out = out_fp16.to(torch.float32)
    out = out.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ------------------------------------------------------------------
    # 8) Apply out‑gate and final projection back to C
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]               # (dim, hidden_dim)
    out = F.linear(out, to_out_weight)                    # [B, N, N, dim]

    return out
