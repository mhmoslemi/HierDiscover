    num_n = tl.cdiv(N, BLOCK_N)

"""
TriMul custom kernel (outgoing version).

This implementation follows the reference PyTorch TriMul module but accelerates the
core O(N³·C_hidden) operation with a fused Triton kernel.  The heavy part is the
einsum
    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
which can be expressed as a batched matrix multiplication
    out_{b,d} = left_{b,d} @ right_{b,d}.T
where each batch‑hidden slice has shape (N, N).  The Triton kernel computes these
batched GEMMs (FP16 inputs, FP32 accumulation, FP16 output) using tensor cores.
All LayerNorms, linear projections and gating are performed with PyTorch
functional calls; they are cheap compared with the N³ term and benefit from
cuBLAS/cuDNN optimizations.

The kernel is written for Triton 3.3.1 and targets an H100 GPU.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A, B, C,               # pointers to 3‑D tensors [batch, M, K], [batch, N, K], [batch, M, N]
    M, N, K,               # matrix dimensions
    stride_a_batch, stride_am, stride_ak,   # strides for A
    stride_b_batch, stride_bn, stride_bk,   # strides for B
    stride_c_batch, stride_cm, stride_cn,   # strides for C
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM C = A @ Bᵀ  (A: [batch,M,K], B: [batch,N,K])"""
    pid = tl.program_id(axis=0)          # tile id within a batch
    batch_id = tl.program_id(axis=1)     # which batch/hidden slice

    # -------------------------------------------------------------
    # 1) Compute tile coordinates (i_tile, j_tile)
    # -------------------------------------------------------------
    num_m = tl.cdiv(M, BLOCK_M)
    num_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_n                # tile row
    pid_n = pid % num_n                 # tile column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)   # [BLOCK_M]
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)   # [BLOCK_N]

    # -------------------------------------------------------------
    # 2) Pointers to the first element of the current batch
    # -------------------------------------------------------------
    a_batch = A + batch_id * stride_a_batch
    b_batch = B + batch_id * stride_b_batch
    c_batch = C + batch_id * stride_c_batch

    # -------------------------------------------------------------
    # 3) Accumulator in FP32
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # 4) Loop over k dimension in tiles of size BLOCK_K
    # -------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)     # [BLOCK_K]

        # Masks for out‑of‑bounds loads
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        b_mask = (offs_n < N)[:, None] & (offs_k < K)[None, :]

        # Load tiles.  A: (BLOCK_M, BLOCK_K), B: (BLOCK_K, BLOCK_N)
        a = tl.load(
            a_batch
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak,
            mask=a_mask,
            other=0.0,
        )
        b = tl.load(
            b_batch
            + offs_k[:, None] * stride_bk      # k → rows
            + offs_n[None, :] * stride_bn,     # n → cols
            mask=b_mask,
            other=0.0,
        )
        # Tensor‑core dot (FP16·FP16 → FP32)
        acc += tl.dot(a, b)

    # -------------------------------------------------------------
    # 5) Write back result (stored as FP16 to save bandwidth)
    # -------------------------------------------------------------
    c = acc.to(tl.float16)
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(
        c_batch + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn,
        c,
        mask=c_mask,
    )


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - input_tensor: torch.Tensor of shape [B, N, N, C] (float32)
              - mask_tensor : torch.Tensor of shape [B, N, N] (bool or float)
              - weights_dict: mapping from weight names to torch.Tensor
              - config_dict : must contain "dim" and "hidden_dim", also
                              optional "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, C] (float32)
    """
    # -----------------------------------------------------------------
    # 0) unpack
    # -----------------------------------------------------------------
    (x, mask, weights, config) = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1) LayerNorm on input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [B,N,N,dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]

    left = F.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on the hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(dtype).unsqueeze(-1)            # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4) Gating linear layers + sigmoid
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core O(N³·C_hidden) operation:
    #    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
    #    Implemented as batched GEMM: C = A @ Bᵀ  for each hidden slice.
    # -----------------------------------------------------------------
    # 5.1) Rearrange to [batch, hidden, N, N] and make them contiguous
    B, N, _, _ = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, hidden, N, N]

    # 5.2) Transpose the second operand so that we can treat it as (N, K)
    right_t = right_perm.transpose(-2, -1).contiguous()  # now shape [B, hidden, N, K] (K == N)

    # 5.3) Cast to FP16 for tensor‑core speed‑up
    left_fp16 = left_perm.to(torch.float16)
    right_fp16 = right_t.to(torch.float16)

    # 5.4) Flatten the batch‑hidden dimension
    batch_hidden = B * hidden_dim
    left_flat = left_fp16.view(batch_hidden, N, N)
    right_flat = right_fp16.view(batch_hidden, N, N)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16)

    # 5.5) Extract strides (in elements, not bytes)
    stride_a_batch, stride_am, stride_ak = left_flat.stride()
    stride_b_batch, stride_bn, stride_bk = right_flat.stride()
    stride_c_batch, stride_cm, stride_cn = out_flat.stride()

    # 5.6) Launch the Triton kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid_x = tl.cdiv(N, BLOCK_M) * tl.cdiv(N, BLOCK_N)   # number of tiles per batch
    grid = (grid_x, batch_hidden)                        # (tile_id, batch_hidden_id)

    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,
        N,
        N,
        stride_a_batch,
        stride_am,
        stride_ak,
        stride_b_batch,
        stride_bn,
        stride_bk,
        stride_c_batch,
        stride_cm,
        stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # typical for FP16 matmul on H100
    )
    # -----------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] (float32)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).to(dtype)   # [B,N,N,hidden]

    # -----------------------------------------------------------------
    # 7) Output LayerNorm, gating, final linear projection
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B,N,N,hidden]

    out_norm = out_norm * out_gate                       # element‑wise gating

    to_out_weight = weights["to_out.weight"]             # (dim, hidden)
    out_final = F.linear(out_norm, to_out_weight)       # [B,N,N,dim]

    return out_final
