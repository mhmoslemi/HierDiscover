import triton.language as tl

def trimul_kernel(left, right):
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2
    # allocate output
    out = torch.empty((B, N, N, H), dtype=torch.float32, device=left.device)

    # compute strides
    left_stride_b, left_stride_i, left_stride_k, left_stride_h = left.stride()
    right_stride_b, right_stride_j, right_stride_k, right_stride_h = right.stride()
    out_stride_b, out_stride_i, out_stride_j, out_stride_h = out.stride()

    # define block sizes
    BLOCK_M = 32  # i dimension
    BLOCK_N = 32  # j dimension
    BLOCK_K = 32  # k tile
    BLOCK_D = 8   # hidden dimension tile

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B)

    trimul_kernel_impl[grid](
        left_ptr = left,
        right_ptr = right,
        out_ptr = out,
        N = N,
        H = H,
        stride_left_b = left_stride_b,
        stride_left_i = left_stride_i,
        stride_left_k = left_stride_k,
        stride_left_h = left_stride_h,
        stride_right_b = right_stride_b,
        stride_right_j = right_stride_j,
        stride_right_k = right_stride_k,
        stride_right_h = right_stride_h,
        stride_out_b = out_stride_b,
        stride_out_i = out_stride_i,
        stride_out_j = out_stride_j,
        stride_out_h = out_stride_h,
        BLOCK_M = BLOCK_M,
        BLOCK_N = BLOCK_N,

Define kernel:

```
@triton.jit
def trimul_kernel_impl(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # offsets
    i_idx = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_idx = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_idx = tl.arange(0, BLOCK_D)

    mask_i = i_idx < N
    mask_j = j_idx < N
    mask_d = d_idx < H

    # Init accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # Compute pointers for left and right
        left_ptrs = left_ptr + pid_b * stride_left_b \
                     + i_idx[:, None, None] * stride_left_i \
                     + cur_k[None, :, None] * stride_left_k \
                     + d_idx[None, None, :] * stride_left_h
        right_ptrs = right_ptr + pid_b * stride_right_b \
                     + j_idx[:, None, None] * stride_right_j \
                     + cur_k[None, :, None] * stride_right_k \
                     + d_idx[None, None, :] * stride_right_h

        # Load with mask
        # Build mask for left: combine mask_i, mask_k, mask_d.
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
        right_mask = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]

        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # Broadcast and multiply
        # left shape (BM, BK, BD), right shape (BN, BK, BD)
        # expand to (BM, 1, BK, BD) and (1, BN, BK, BD)
        prod = left[:, None, :, :] * right[None, :, :, :]  # (BM, BN, BK, BD)
        # sum over k dimension
        acc += tl.sum(prod, axis=2)  # sum over K -> shape (BM, BN, BD)

    # Write out
    out_ptrs = out_ptr + pid_b * stride_out_b \
               + i_idx[:, None, None] * stride_out_i \
               + j_idx[None, :, None] * stride_out_j \
               + d_idx[None, None, :] * stride_out_h

    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

@triton.jit
def triu_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_li_b, stride_li_i, stride_li_k, stride_li_h,
    stride_ri_b, stride_ri_j, stride_ri_k, stride_ri_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid_i = tl.program_id(0) # i-block
    pid_j = tl.program_id(1) # j-block
    pid_b = tl.program_id(2) # batch

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    d_base = tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_j = j < N

    # initialize accumulator: (BLOCK_M, BLOCK_N, BLOCK_D)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    for cur_k in range(0, N, BLOCK_K):
        k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # pointer offsets for left (b,i,k,d)
        left_ptrs = left_ptr + pid_b * stride_li_b \
                    + i[:, None, None] * stride_li_i \
                    + k[None, :, None] * stride_li_k \
                    + d_base[None, None, :] * stride_li_h
        left_mask = mask_i[:, None, None] & mask_k[None, :, None] & (d_base[None, None, :] < H)
        left = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # pointer offsets for right (b,j,k,d)
        right_ptrs = right_ptr + pid_b * stride_ri_b \
                     + j[:, None, None] * stride_ri_j \
                     + k[None, :, None] * stride_ri_k \
                     + d_base[None, None, :] * stride_ri_h
        right_mask = mask_j[:, None, None] & mask_k[None, :, None] & (d_base[None, None, :] < H)
        right = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # compute product and sum over k dimension: need to multiply left[i,k,d] * right[j,k,d] and sum across k.
        # Broadcasting left and right across i and j axes
        # left: (BM, BK, BD), right: (BN, BK, BD)
        # produce partial = sum over k of left[i,k,d] * right[j,k,d]
        # Efficient way: we can compute for each d slice using dot product: left[:,:,d] (BM,BK) dot right[:,:,d].T (BK,BN) -> (BM,BN)
        # We'll loop over d within BLOCK_D
        for db in range(0, BLOCK_D):
            # check bounds for this d
            if d_base[db] >= H:
                continue
            left_slice = left[:, :, db]   # shape (BM, BK)
            right_slice = right[:, :, db] # shape (BN, BK)
            # compute dot
            # need to transpose right for dot
            prod = tl.dot(left_slice, right_slice, trans_b=True)   # shape (BM, BN)
            # accumulate
            acc[:, :, db] += prod

    # write out
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + i[:, None, None] * stride_o_i \
               + j[None, :, None] * stride_o_j \
               + d_base[None, None, :] * stride_o_h

    out_mask = mask_i[:, None, None] & mask_j[None, :, None] & (d_base[None, None, :] < H)
    tl.store(out_ptrs, acc, mask=out_mask)
```

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm x using torch
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    # apply weight and bias
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # cast to half for linear ops
    x_norm_h = x_norm.to(torch.float16)

    # linear projections
    left = F.linear(x_norm_h, weights["left_proj.weight"])
    right = F.linear(x_norm_h, weights["right_proj.weight"])

    # gates (compute in float32 then cast)
    left_gate_f = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate_f = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate_f = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # cast gates to half for multiplication
    left_gate = left_gate_f.to(torch.float16)
    right_gate = right_gate_f.to(torch.float16)

    # Apply mask if provided
    if mask is not None:
        mask_f = mask.to(torch.float16).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Apply gates (elementwise)
    left = left * left_gate
    right = right * right_gate

    # Compute TriMul using Triton kernel
    out = trimul_triton(left, right)  # output dtype float32

    # to_out_norm (LayerNorm on hidden_dim)
    eps_norm = 1e-5
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps_norm)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Multiply by out_gate (float32)
    out_norm = out_norm * out_gate_f.unsqueeze(-1)  # Broadcast output_gate shape [B,N,N,H] (already H)

"""
custom_kernel implements the outgoing Triangular Multiplicative Update (TriMul)
operator from AlphaFold‑3 (and related models). The heavy O(N³·C) pairwise
multiplication

    out[b,i,j,c] = Σ_k left[b,i,k,c] * right[b,j,k,c]

is performed with a custom Triton kernel. All other operations
(LayerNorms, linear projections, gating, final projection) are carried out
with PyTorch for simplicity and numerical stability.

The Triton kernel computes a matrix multiplication for a single hidden
channel (c) and a single batch element, i.e. it evaluates a standard GEMM
on slices left[b,:, :,c] and right[b,:, :,c] (with the second operand
transposed on the K dimension).  The kernel is launched with a 3‑D grid:
(i‑tile, j‑tile, batch·hidden) and accumulates the result in fp32 while
the inputs are stored in fp16 to exploit Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: one program computes out[b,i,j,c] for a fixed (b,c)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr
):
    pid_i = tl.program_id(0)               # i‑tile
    pid_j = tl.program_id(1)               # j‑tile
    pid_c = tl.program_id(2)               # combined batch·channel index

    # --------------------------------------------------------------
    # decode batch index and hidden‑channel index
    # --------------------------------------------------------------
    batch = pid_c // H
    d = pid_c % H

    # --------------------------------------------------------------
    # tile offsets
    # --------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # base pointers for the current (batch, channel)
    left_base  = left_ptr  + batch * stride_l_b + d * stride_l_h
    right_base = right_ptr + batch * stride_r_b + d * stride_r_h
    out_base   = out_ptr   + batch * stride_o_b + d * stride_o_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left[b,i,k,c]
        left_ptrs = left_base + i[:, None] * stride_l_i + cur_k[None, :] * stride_l_k
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right[b,j,k,c]
        right_ptrs = right_base + j[:, None] * stride_r_i + cur_k[None, :] * stride_r_k
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0).to(tl.float32)

        # Σ_k left[i,k] * right[j,k]   (note trans_b=True -> right is treated as (K,N))
        acc += tl.dot(left, right, trans_b=True)

    # --------------------------------------------------------------
    # write back the result
    # --------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_o_i + j[None, :] * stride_o_j
    out_mask = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Helper that wraps the Triton kernel
# ----------------------------------------------------------------------
def _trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left/right: [B, N, N, C]  (dtype=torch.float16)
    returns:    out   : [B, N, N, C] (dtype=torch.float32)
    """
    assert left.shape == right.shape
    B, N, _, C = left.shape

    out = torch.empty((B, N, N, C), dtype=torch.float32, device=left.device)

    # strides (in elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # tuning parameters – they work well for the sizes required in the tests
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * C,                         # one program per (batch, channel)
    )

    _trimul_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        N          = N,
        stride_l_b = stride_l_b,
        stride_l_i = stride_l_i,
        stride_l_k = stride_l_k,
        stride_l_h = stride_l_h,
        stride_r_b = stride_r_b,
        stride_r_i = stride_r_i,
        stride_r_k = stride_r_k,
        stride_r_h = stride_r_h,
        stride_o_b = stride_o_b,
        stride_o_i = stride_o_i,
        stride_o_j = stride_o_j,
        stride_o_h = stride_o_h,
        BLOCK_M    = BLOCK_M,
        BLOCK_N    = BLOCK_N,
        BLOCK_K    = BLOCK_K,
        H          = C,                # compile‑time constant for modulo/div
    )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    data = (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns the forward pass of the outgoing TriMul block.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm on the input tensor (over the feature dim)
    # ------------------------------------------------------------------
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) * torch.rsqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias) → cast to fp16 for the heavy part
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"]).to(torch.float16)
    right = F.linear(x_norm, weights["right_proj.weight"]).to(torch.float16)

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid of a linear projection). Keep them in fp32 for
    #    numerical stability, then cast to fp16 for the element‑wise mul.
    # ------------------------------------------------------------------
    left_gate_f  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate_f = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate_f   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left_gate  = left_gate_f.to(torch.float16)
    right_gate = right_gate_f.to(torch.float16)

    # ------------------------------------------------------------------
    # 4) Optional mask (shape [B,N,N]) – broadcast on the hidden dim.
    # ------------------------------------------------------------------
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)          # [B,N,N,1]
        left = left * mask_h
        right = right * mask_h

    # ------------------------------------------------------------------
    # 5) Apply the per‑channel gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core O(N³·C) multiplication using the Triton kernel
    # ------------------------------------------------------------------
    out = _trimul_triton(left, right)          # [B,N,N,hidden_dim], fp32

    # ------------------------------------------------------------------
    # 7) LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out_mean = out.mean(dim=-1, keepdim=True)
    out_var  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) * torch.rsqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # ------------------------------------------------------------------
    # 8) Multiply by the output gate (broadcasted automatically)
    # ------------------------------------------------------------------
    out_norm = out_norm * out_gate_f

    # ------------------------------------------------------------------
    # 9) Final linear projection back to the original dimension
    # ------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])
    return output
