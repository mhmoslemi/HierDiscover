def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_batch, stride_left_i, stride_left_k, stride_left_c,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_c,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bch = tl.program_id(2)  # flatten batch and channel
    b = pid_bch // H
    c = pid_bch % H
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K
        # compute bounds
        i_offset = i0 + tl.arange(0, BLOCK_M)
        j_offset = j0 + tl.arange(0, BLOCK_N)
        k_offset = k0 + tl.arange(0, BLOCK_K)

        # mask to avoid out-of-bounds
        mask_i = i_offset < N
        mask_j = j_offset < N
        mask_k = k_offset < N

        # offsets for left: [b, i, k, c]
        left_ptrs = left_ptr + b*stride_left_batch + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_k + c*stride_left_c
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    # LayerNorm input
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=1e-5)
    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    # Expand mask and apply
    if mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate
    # Compute out via triton kernel
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty(bs, N, N, hidden_dim, dtype=dtype, device=device)
    # Launch kernel
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    # compute strides
    stride_lb = left.stride(0)
    stride_li = left.stride(1)
    stride_lk = left.stride(2)
    stride_lc = left.stride(3)
    stride_rb = right.stride(0)
    stride_rj = right.stride(1)
    stride_rk = right.stride(2)
    stride_rc = right.stride(3)
    stride_ob = out.stride(0)
    stride_oi = out.stride(1)
    stride_oj = out.stride(2)
    stride_oc = out.stride(3)

        stride_ob, stride_oi, stride_oj, stride_oc,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K

    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=1e-5)

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation (AlphaFold‑3 Triangular Multiplicative
Update) using a fused Triton kernel for the expensive N‑N einsum:

    out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}

The kernel computes a block‑wise matrix multiplication for each (batch, channel)
pair, i.e. a GEMM with the second operand transposed, which matches the
einsum above.  All other lightweight operations (LayerNorm, Linear projections,
sigmoids, masking, final LayerNorm + Linear) are performed with PyTorch
functions.  The kernel works for arbitrary batch size (1‑2), sequence length
(≤ 1024) and hidden dimension (≤ 384) and runs on an H100 with Triton 3.3.1.

The returned tensor has shape `[bs, seq_len, seq_len, dim]` and dtype `float32`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: block‑wise matmul (left @ rightᵀ) for the TriMul einsum.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_einsum_kernel(
    left_ptr,          # const float*    left   [B, N, N, H]
    right_ptr,         # const float*   right  [B, N, N, H]
    out_ptr,           # float*         out    [B, N, N, H]
    B, N, H,           # shape parameters
    stride_left_batch, stride_left_i, stride_left_k, stride_left_c,
    stride_right_batch, stride_right_i, stride_right_k, stride_right_c,
    stride_out_batch, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
    using block‑wise GEMM with the second operand transposed.
    The kernel is launched with a 3‑D grid:
        (i_tile, j_tile, batch * channel)
    """

    # ------------------------------------------------------------------
    # 1) Program‑ID decomposition (batch & hidden dimension)
    # ------------------------------------------------------------------
    pid_bc = tl.program_id(2)                # 0 … B*H‑1
    b = pid_bc // H                           # batch index
    h = pid_bc % H                            # hidden‑channel index

    pid_i = tl.program_id(0)                 # i‑tile
    pid_j = tl.program_id(1)                 # j‑tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # 2) Create index vectors for the current tile
    # ------------------------------------------------------------------
    i_range = i0 + tl.arange(0, BLOCK_M)      # [i0, i0+BLOCK_M)
    j_range = j0 + tl.arange(0, BLOCK_N)      # [j0, j0+BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # accumulator in FP32 (very small hidden_dim → no overflow concerns)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 3) Loop over the reduction dimension k in BLOCK_K steps
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for k_block in range(0, num_k_blocks):
        k0 = k_block * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        mask_k = k_range < N

        # --------------------------------------------------------------
        # Load a tile from `left` : shape (BLOCK_M, BLOCK_K)
        #   left[b, i, k, h]
        # --------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_batch
            + i_range[:, None] * stride_left_i
            + k_range[None, :] * stride_left_k
            + h * stride_left_c
        )
        a = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # Load a tile from `right` *transposed* : shape (BLOCK_K, BLOCK_N)
        #   we want right[b, j, k, h] but we index it as (k, j) to obtain
        #   the transposed view required for GEMM.
        # --------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_batch
            + k_range[:, None] * stride_right_k
            + j_range[None, :] * stride_right_i
            + h * stride_right_c
        )
        b_tile = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # GEMM: acc += a @ b_tile
        # --------------------------------------------------------------
        acc += tl.dot(a, b_tile)

    # ------------------------------------------------------------------
    # 4) Write the result back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_batch
        + i_range[:, None] * stride_out_i
        + j_range[None, :] * stride_out_j
        + h * stride_out_c
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# Helper: LayerNorm that uses the provided weight/bias tensors.

def _layer_norm(x, weight, bias, eps=1e-5):
    """Thin wrapper around torch.nn.functional.layer_norm."""
    return F.layer_norm(x, x.shape[-1:], weight=weight, bias=bias, eps=eps)

# Main entry point required by the evaluator.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (AlphaFold‑3 “outgoing” update).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        - input_tensor : torch.Tensor[bs, N, N, dim]
        - mask_tensor  : torch.Tensor[bs, N, N]  (may be all‑ones)
        - weights_dict : mapping from layer names to torch tensors
        - config_dict  : must contain ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor[bs, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # 0) Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = _layer_norm(
        input_tensor,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj = F.linear(x, weights["left_proj.weight"])      # → [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])    # → [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # → [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 4) Apply optional mask and gating
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] → broadcast on hidden dim
        mask = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Ensure contiguous layout for the kernel
    left = left.contiguous()
    right = right.contiguous()

    # ------------------------------------------------------------------
    # 5) Triton kernel: compute einsum (N‑N accumulation)
    # ------------------------------------------------------------------
    out = torch.empty((bs, N, N, hidden_dim), dtype=dtype, device=device)

    # Block sizes – tuned for H100 (FP32 accumulation, modest hidden_dim)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Compute strides (torch uses row‑major layout)
    stride_l_b = left.stride(0)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_c = left.stride(3)

    stride_r_b = right.stride(0)
    stride_r_i = right.stride(1)   # this is the “j” dimension in the einsum
    stride_r_k = right.stride(2)
    stride_r_c = right.stride(3)

    stride_o_b = out.stride(0)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_c = out.stride(3)

    # 3‑D grid: (i‑tile, j‑tile, batch*channel)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        bs * hidden_dim,
    )
    tri_mul_einsum_kernel[grid](
        left,
        right,
        out,
        bs,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_c,
        stride_r_b,
        stride_r_i,
        stride_r_k,
        stride_r_c,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_c,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Output LayerNorm + gating
    # ------------------------------------------------------------------
    out_norm = _layer_norm(
        out,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to original `dim`
    # ------------------------------------------------------------------
    final = F.linear(out_norm, weights["to_out.weight"])  # [B,N,N,dim]

    # Guarantee FP32 output as required
    return final.float()
