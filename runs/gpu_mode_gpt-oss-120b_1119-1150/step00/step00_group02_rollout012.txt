def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # M, N, K are sizes
    # Compute block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Compute pointers for A and B base for this batch
    a_batch_ptr = A_ptr + batch_idx * stride_ab
    b_batch_ptr = B_ptr + batch_idx * stride_bb

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; eps = 1e-5

    # LayerNorm
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm:
    # shape of normalized dims = (dim,)
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )
    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))

    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply mask if provided
    if mask is not None:
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Now contraction
    # Permute to (B, hidden_dim, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Flatten batch and hidden dims
    B, H, N, _ = left_perm.shape
    total_batch = B * H
    left_flat = left_perm.view(total_batch, N, N)
    right_flat = right_perm.view(total_batch, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Get strides
    stride_al = left_flat.stride()
    stride_bl = right_flat.stride()
    stride_cl = out_flat.stride()

    # Compute kernel dims
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_batch,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Launch kernel
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_al[0],
        stride_ak=stride_al[2], # ?? Actually left_flat.shape is (total_batch, M, K) where M=N, K=N. Strides: stride_al[0] = stride between batches, stride_al[1] = stride between rows (M), stride_al[2] = stride between columns (K).
        stride_ab=stride_al[0],
        stride_bk=stride_bl[2],
        stride_bn=stride_bl[1],
        stride_bb=stride_bl[0],
        stride_cm=stride_cl[1],
        stride_cn=stride_cl[2],
        stride_cb=stride_cl[0],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K
    )

    # Restore shape
    out = out_flat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # [B,N,N,H]

    # LayerNorm on out (to_out_norm)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply out_gate
    out = out * out_gate

    # to_out linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,
    stride_bk, stride_bn, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for off range
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointer to the batch slice
    a_batch_ptr = A_ptr + batch_idx * stride_ab
    b_batch_ptr = B_ptr + batch_idx * stride_bb

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # A: shape (BLOCK_M, BLOCK_K)
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # B: shape (BLOCK_K, BLOCK_N) ; we load B as B[j,k] (i.e., row = j, col = k)
        b_ptrs = b_batch_ptr + (offs_n[None, :] * stride_bn) + (offs_k[:, None] * stride_bk)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)
    # Write out
    c_ptrs = C_ptr + batch_idx * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])

"""
TriMul (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Compute three linear projections of the normalized tensor:
   - left_proj  :  D → H
   - right_proj :  D → H
   - gating projections (left_gate, right_gate, out_gate) : D → H
   and apply a sigmoid to the gating projections.
3. Apply the optional pair‑wise mask (broadcast to H) to the left/right
   projections *before* gating, then multiply by the respective gates.
4. Perform the “outgoing” triangle multiplication:
        out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   This is a batched mat‑mul of shape (B·H, N, N) × (B·H, N, N)ᵀ.
   It is realised by a custom Triton kernel that loops over K in
   tiled fashion (BLOCK_M × BLOCK_N tiles, BLOCK_K reduction tiles).
5. Layer‑norm the result over the hidden dimension H, multiply by the
   previously computed out‑gate and finish with a linear projection
   back to the original dimension D.

The kernel works on float32 tensors (the most common precision for the
reference implementation) and uses a 64×64 tile size with a 64‑element
reduction tile, which is well suited to the H100 architecture.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K)
    stride_bk, stride_bn, stride_bb,   # B: (batch, N, K)   (we load B[j,k])
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # program IDs
    batch_idx = tl.program_id(0)      # combined batch = B * H
    pid_m    = tl.program_id(1)      # tile row
    pid_n    = tl.program_id(2)      # tile col

    # offsets for the current tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for the current batch
    a_batch_ptr = A_ptr + batch_idx * stride_ab
    b_batch_ptr = B_ptr + batch_idx * stride_bb

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in BLOCK_K sized chunks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # -------- load A tile (M x K) --------
        a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # -------- load B tile (K x N) --------
        # note: we want B[j,k] which means we load with j as the "row" dim
        b_ptrs = b_batch_ptr + (offs_n[None, :] * stride_bn) + (offs_k[:, None] * stride_bk)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # dot product (M x K) @ (K x N) -> (M x N)
        acc += tl.dot(a, b)

    # write back the result
    c_ptrs = C_ptr + batch_idx * stride_cb + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor [B, N, N, D] (float32)
        - mask_tensor  : torch.Tensor [B, N, N] (bool/float) or None
        - weights_dict : mapping of weight names to torch.Tensors
        - config_dict  : contains "dim", "hidden_dim", optional "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]          # D
    hidden_dim = config["hidden_dim"]  # H
    eps        = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (bias‑free) and gating
    # ------------------------------------------------------------------
    # Projections
    left  = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"]) # [B,N,N,H]

    # Optional mask (applied before gating)
    if mask is not None and not config.get("nomask", False):
        mask_exp = mask.to(dtype).unsqueeze(-1)       # [B,N,N,1]
        left  = left * mask_exp
        right = right * mask_exp

    # Gating values
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))  # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))   # [B,N,N,H]

    # Apply gates
    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3. Batched matrix multiplication: out = left @ rightᵀ  (outgoing)
    #    Shape after multiplication: [B,N,N,H] -> permute to [B*H, N, N]
    # ------------------------------------------------------------------
    # Permute to bring the hidden dimension into the batch axis
    left  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    B, H, N, _ = left.shape
    total_batch = B * H

    left_flat  = left.view(total_batch, N, N)
    right_flat = right.view(total_batch, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat, dtype=dtype, device=device)

    # Strides (in number of elements, not bytes)
    stride_ab, stride_am, stride_ak = left_flat.stride()
    stride_bb, stride_bn, stride_bk = right_flat.stride()
    stride_cb, stride_cm, stride_cn = out_flat.stride()

    # Tiling parameters – tuned for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    # Launch the Triton kernel
    _tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M=N,
        N=N,
        K=N,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_ab=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bb=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cb=stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # ------------------------------------------------------------------
    # 4. Post‑processing: layer‑norm over H, multiply by out_gate and final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    out = out * out_gate                     # broadcast over the hidden dim
    out = F.linear(out, weights["to_out.weight"])  # [B,N,N,D]

    return out
