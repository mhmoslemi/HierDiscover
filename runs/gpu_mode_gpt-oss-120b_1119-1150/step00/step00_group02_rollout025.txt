    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    dim = config["dim"] # same as input dim
    # norm

    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # mask
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate

    # prepare for matmul
    # permute to shape (bs, hidden_dim, seq_len, seq_len)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose right for multiplication
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    # cast to half precision for matmul
    left_h = left_perm.to(torch.float16)
    right_h = right_t.to(torch.float16)

    # fused batched matmul
    out_h = batched_matmul_triton(left_h, right_h)  # shape (bs*hidden, seq_len, seq_len) in half

    # reshape back
    out = out_h.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()  # shape (bs, seq_len, seq_len, hidden_dim)

    # to_out_norm

    out = out * out_gate

    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_hidden,
    M, N, K,
    stride_ah0, stride_ah1, stride_ah2,
    stride_bh0, stride_bh1, stride_bh2,
    stride_ch0, stride_ch1, stride_ch2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    # Compute batch and block m indices
    batch_idx = pid // num_blocks_m
    block_m = pid % num_blocks_m
    block_n = pid_n

    # Compute offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out-of-bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Pointers to A and B blocks
    a_ptrs = A_ptr + batch_idx * stride_ah0 + offs_m[:, None] * stride_ah1 + offs_k[None, :] * stride_ah2
    b_ptrs = B_ptr + batch_idx * stride_bh0 + offs_k[:, None] * stride_bh1 + offs_n[None, :] * stride_bh2

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load A and B
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (offs_k < K), other=0.0)
        b = tl.load(b_ptrs, mask=mask_n[None, :] & (offs_k[:, None] < K), other=0.0)

```python
import torch
import triton
import triton.language as tl

# Triton kernel for batched matrix multiplication (C = A @ B)
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    batch_hidden,
    M, N, K,
    stride_ah0, stride_ah1, stride_ah2,
    stride_bh0, stride_bh1, stride_bh2,
    stride_ch0, stride_ch1, stride_ch2,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute C = A @ B for many independent (M×K) @ (K×N) products.
    A, B, C have an outer batched dimension `batch_hidden`.
    Everything is in fp16 except the accumulator (fp32).
    """
    pid = tl.program_id(axis=0)   # combines batch index and block row
    pid_n = tl.program_id(axis=1) # block column

    # number of tiles along the M dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)

    # decode batch index and block row from pid
    batch_idx = pid // num_blocks_m
    block_m = pid % num_blocks_m
    block_n = pid_n

    # compute offsets within the tile
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # masks for OOB rows/cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # pointers to the beginning of the tile for this batch
    a_ptr = A_ptr + batch_idx * stride_ah0 + offs_m[:, None] * stride_ah1 + offs_k[None, :] * stride_ah2
    b_ptr = B_ptr + batch_idx * stride_bh0 + offs_k[:, None] * stride_bh1 + offs_n[None, :] * stride_bh2

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over the K dimension
    k_offset = 0
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k_offset + offs_k
        mask_k = cur_k < K

        a = tl.load(a_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)   # a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)

        # advance to next K‑tile
        a_ptr += BLOCK_K * stride_ah2
        b_ptr += BLOCK_K * stride_bh1
        k_offset += BLOCK_K

    # store result as fp16
    c = acc.to(tl.float16)
    c_ptr = C_ptr + batch_idx * stride_ch0 + offs_m[:, None] * stride_ch1 + offs_n[None, :] * stride_ch2
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])

def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper around `batched_matmul_kernel`.  A has shape (B*H, M, K) and
    B has shape (B*H, K, N).  Returns C of shape (B*H, M, N) in the same dtype as A/B.
    """
    assert A.shape[0] == B.shape[0]
    batch_hidden, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch_hidden, M, N), dtype=A.dtype, device=A.device)

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_hidden * triton.cdiv(M, BLOCK_M), triton.cdiv(N, BLOCK_N))

    batched_matmul_kernel[grid](
        A, B, C,
        batch_hidden,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C

def custom_kernel(data):
    """
    Triton‑accelerated forward of the “outgoing” TriMul module.
    The heavy pairwise multiplication (einsum over the third axis) is
    performed by a batched matmul kernel; all other operations are
    expressed with native PyTorch kernels for clarity.
    """
    input_tensor, mask, weights, config = data

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2. Linear projections + optional mask
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 3. Gating
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 4. Pairwise multiplication  (the O(N³) part)
    #    Compute out_i,j = sum_k left_i,k * right_j,k  for each hidden channel.
    #    Implemented as batched matmul C = A @ Bᵀ where
    #        A = left_perm   shape (B, hidden, N, N)
    #        Bᵀ = right_t    shape (B, hidden, N, N)
    # -----------------------------------------------------------------
    # bring the hidden dimension next to the batch dimension and make tensors contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # transpose the last two axes of right so that we can do a regular matmul
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()   # [B, hidden, N, N]

    # flatten batch and hidden dimension so the kernel sees a single leading dimension
    batch_hidden = bs * hidden_dim
    A = left_perm.view(batch_hidden, seq_len, seq_len).to(torch.float16)
    B = right_t.view(batch_hidden, seq_len, seq_len).to(torch.float16)

    # run the Triton kernel
    C = batched_matmul_triton(A, B)   # [batch_hidden, N, N] in fp16

    # reshape back to [B, N, N, hidden]
    out = C.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # -----------------------------------------------------------------
    # 5. Output layernorm, gating and final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)   # -> [B,N,N,dim]

    return out

TriMul (outgoing) forward pass with a Triton‑accelerated O(N³) core.

Algorithm
---------
1. Layer‑norm the input tensor on its feature dimension.
2. Linear projections produce `left` and `right` tensors of shape
   [B, N, N, H] (H = hidden_dim).  If a mask is present it is applied
   element‑wise before the gates.
3. Gating: sigmoid(Linear(x)) is multiplied into `left` and `right`.
4. Core operation:
   out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]   (einsum‑style)
   This is a batched matrix multiplication:
       C = A @ Bᵀ
   where A = left (B,H,N,N) and Bᵀ = right transposed on the last two axes.

5. Layer‑norm on the hidden dimension, element‑wise gating with `out_gate`,

                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        acc += tl.dot(a, b)            # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # move to the next K‑tile
        a_ptr += BLOCK_K * stride_ah2
        b_ptr += BLOCK_K * stride_bh1
        k_offset += BLOCK_K

    # write result (down‑cast to fp16)
    c = acc.to(tl.float16)
    c_ptr = (C_ptr + batch_idx * stride_ch0
             + offs_m[:, None] * stride_ch1
             + offs_n[None, :] * stride_ch2)
    tl.store(c_ptr, c, mask=mask_m[:, None] & mask_n[None, :])


def batched_matmul_triton(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Wrapper for `batched_matmul_kernel`.
    A : (B*H, M, K) fp16
    B : (B*H, K, N) fp16
    Returns C : (B*H, M, N) fp16
    """
    assert A.shape[0] == B.shape[0], "batch dimensions must match"
    batch_hidden, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "inner dimensions must match"

    C = torch.empty((batch_hidden, M, N), dtype=A.dtype, device=A.device)

    # Tile sizes chosen for good occupancy on H100 tensor cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (batch_hidden * triton.cdiv(M, BLOCK_M),  # combine batch & row‑tiles
            triton.cdiv(N, BLOCK_N))                # column tiles

    batched_matmul_kernel[grid](
        A, B, C,
        batch_hidden,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Public entry point used by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward of the outgoing TriMul module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask         : torch.Tensor or None, shape [B, N, N]
        - weights      : dict of torch.Tensors, containing all Linear/LayerNorm params
        - config       : dict, must contain `dim`, `hidden_dim`, and `nomask` flag
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    eps = 1e-5
    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1. Input LayerNorm
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )

    # --------------------------------------------------------------
    # 2. Linear projections (+ optional mask)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(x.dtype).unsqueeze(-1)        # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # --------------------------------------------------------------
    # 3. Gating
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 4. Core pairwise multiplication (highly optimized)
    # --------------------------------------------------------------
    # bring hidden dimension next to the batch dimension and make them contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # transpose the last two axes of `right` so we can use a regular GEMM
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()   # [B, H, N, N]

    # flatten batch & hidden dimensions → (B*H, N, N)
    bh = bs * hidden_dim
    A = left_perm.view(bh, seq_len, seq_len).to(torch.float16)   # (B*H, M, K)
    B = right_t.view(bh, seq_len, seq_len).to(torch.float16)    # (B*H, K, N)

    # batched matrix multiplication: C = A @ B
    C = batched_matmul_triton(A, B)                            # (B*H, M, N) fp16

    # reshape back to [B, N, N, H] and convert to fp32 for the following layers
    out = C.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # --------------------------------------------------------------
    # 5. Output LayerNorm, gating, final Linear
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
