@triton.jit
def triton_tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    batch_size, M, N, K,
    stride_left_batch, stride_left_m, stride_left_k,
    stride_right_batch, stride_right_k, stride_right_n,
    stride_out_batch, stride_out_m, stride_out_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # sentinel: mask for out-of-range:
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # Offsets for K dimension
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: left[batch_id, offs_m, offs_k]
        a_ptr = left_ptr + batch_id * stride_left_batch + offs_m[:,None] * stride_left_m + offs_k[None,:] * stride_left_k
        a = tl.load(a_ptr, mask=mask_m[:,None] & mask_k[None,:], other=0.0).to(tl.float32)
        # Load B tile: right_transposed[batch_id, offs_k, offs_n] 
        # Since we want right transposed, we use stride_right_k for K (which is original stride3 = 1)
        # and stride_right_n for N (original stride2 = N)
        b_ptr = right_ptr + batch_id * stride_right_batch + offs_k[:,None] * stride_right_k + offs_n[None,:] * stride_right_n
        b = tl.load(b_ptr, mask=mask_k[:,None] & mask_n[None,:], other=0.0).to(tl.float32)

        # Compute dot product for this tile
        acc += tl.dot(a, b)  # shape (BLOCK_M, BLOCK_N)
    # Write result
    out_ptrs = out_ptr + batch_id * stride_out_batch + offs_m[:,None] * stride_out_m + offs_n[None,:] * stride_out_n
    tl.store(out_ptrs, acc.to(tl.float16) if out_dtype == tl.float16 else acc, mask=mask_m[:,None] & mask_n[None,:])
```

    # right: (B, hidden, M, K) -> we need its transpose for matmul.

    assert K == M, "square matrices"

"""
TriMul (outgoing) implementation with a Triton matmul kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections → left/right (shape [B,N,N,H]) and three gate
   projections (left_gate, right_gate, out_gate) of shape [B,N,N,H].
3. Apply mask, sigmoid gates and element‑wise multiplication:
   left = left * left_gate * mask, right = right * right_gate * mask.
4. Compute the core O(N³·H) operation:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication for every hidden channel.
   We fuse the batched GEMM with a custom Triton kernel that
   processes all (B·H) batches in parallel.
5. Reshape the result back to [B,N,N,H] and apply a second
   LayerNorm (`to_out_norm`), then multiply by `out_gate`.
6. Final linear projection back to the original channel dimension C.

The heavy O(N³·H) work is performed by the Triton kernel, while the
layer‑norms, linear projections and sigmoid gates use PyTorch
functional calls.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM (A @ B^T) for many independent matrices.
# A and B are both square (N×N) and stored as fp16.  The kernel computes
# the result in fp32 (accumulator) and writes the output as fp32.
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,            # pointers
    M, N, K,                        # matrix sizes (M=N=K)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ---------------------------------------------------------------
    # Grid: (batch, block_row, block_col)
    # ---------------------------------------------------------------
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets for the output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over K dimension (the reduction axis)
    # ---------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A tile: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            a_ptr
            + batch_id * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + offs_k[None, :] * stride_a_k
        )
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            b_ptr
            + batch_id * stride_b_batch
            + offs_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float16)

        # Core dot product (uses tensor cores for fp16)
        acc += tl.dot(a, b)

    # Store the result tile (fp32)
    c_ptrs = (
        c_ptr
        + batch_id * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------
# Public entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor   [B, N, N, C]
        mask         : torch.Tensor   [B, N, N]  (may be None)
        weights      : dict of torch.Tensor containing all model parameters
        config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32, on the same device as input)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]      # H
    B, N, _, _ = input_tensor.shape       # input shape: [B, N, N, C]

    # ------------------------------------------------------------------
    # LayerNorm on the input (over the last dimension C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]      # (H, C)
    right_proj_weight = weights["right_proj.weight"]    # (H, C)

    left = F.linear(x, left_proj_weight)               # [B,N,N,H]
    right = F.linear(x, right_proj_weight)             # [B,N,N,H]

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]      # (H, C)
    right_gate_weight = weights["right_gate.weight"]    # (H, C)
    out_gate_weight = weights["out_gate.weight"]        # (H, C)

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))     # [B,N,N,H]

    # ------------------------------------------------------------------
    # Mask handling (broadcast to hidden dimension)
    # ------------------------------------------------------------------
    if mask is None or config.get("nomask", False):
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    mask = mask.to(dtype).unsqueeze(-1)   # shape [B,N,N,1]

    # Apply mask and gates (element‑wise)
    left = left * left_gate * mask
    right = right * right_gate * mask

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left  -> (B, H, N, N)   rows = i, cols = k
    #   right -> (B, H, N, N)   rows = j, cols = k  → we need its transpose
    # Both tensors are made contiguous and then reshaped to (B*H, N, N)
    # ------------------------------------------------------------------
    left_t = left.permute(0, 3, 1, 2).contiguous()      # [B, H, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous()    # [B, H, N, N]

    # Transpose the last two dimensions of right: we need (K, N) = (N, N) swapped
    right_t = right_t.transpose(-2, -1).contiguous()    # still [B, H, N, N]

    # Merge batch and hidden dimensions: (B*H, N, N)
    batch_hidden = B * hidden_dim
    left_mat = left_t.view(batch_hidden, N, N)
    right_mat = right_t.view(batch_hidden, N, N)

    # Convert to fp16 for the kernel (tensor‑core friendly)
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)

    # Output buffer (fp32, because later ops expect fp32)
    out_mat = torch.empty((batch_hidden, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides for the Triton kernel (all tensors are contiguous)
    # ------------------------------------------------------------------
    stride_a_batch, stride_a_m, stride_a_k = left_fp16.stride()
    stride_b_batch, stride_b_k, stride_b_n = right_fp16.stride()
    stride_c_batch, stride_c_m, stride_c_n = out_mat.stride()

    # Tuning parameters (found to be a good trade‑off on H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch_hidden,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    _batched_matmul_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_fp16,
        c_ptr=out_mat,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # Post‑processing: second LayerNorm, out_gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # Apply out_gate (broadcasted over the hidden dimension)
    out = out * out_gate

    to_out_weight = weights["to_out.weight"]      # shape (C, H)
    out = F.linear(out, to_out_weight)            # final shape [B, N, N, C]

    return out
