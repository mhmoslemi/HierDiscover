import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                           BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)

def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing TriMul (Triangular Multiplicative Update).
    The heavy pairwise multiplication is performed with a fused batched matrix multiplication
    using torch.matmul, while a small Triton kernel performs the final elementwise modulation
    (out_norm * out_gate). LayerNorms and linear projections are executed with PyTorch ops.
    Args:
        data: tuple (input, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # unpack inputs
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Ensure inputs are contiguous and proper dtype
    x = x.contiguous()
    mask = mask.contiguous() if mask is not None else None
    dtype = x.dtype
    device = x.device

    # --- 1. Input LayerNorm ---
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # LayerNorm over last dim (dim)
    x_norm = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5)

    # --- 2. Linear projections (left and right) ---
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)

    # --- 3. Apply mask (if present) ---
    if not nomask and mask is not None:
        # broadcast mask to last dim
        mask = mask.to(dtype)
        mask_exp = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --- 4. Gates (sigmoid of linear of x_norm) ---
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # --- 5. Batched matrix multiplication across hidden_dim ---
    # left: (B,N,N,hidden_dim) -> (B, hidden_dim, N, K) where K=N
    # right: (B,N,N,hidden_dim) -> (B, hidden_dim, K, N) (transpose K and N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()
    # torch.matmul performs batch matmul on the last two dims
    out_perm = torch.matmul(left_perm, right_perm)  # (B, hidden_dim, N, N)
    # Back to (B, N, N, hidden_dim)
    out = out_perm.permute(0, 2, 3, 1).contiguous()

    # --- 6. Output LayerNorm ---
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out, (hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias,
                            eps=1e-5)

    # --- 7. Elementwise modulation with out_gate (Triton) ---
    # out_norm and out_gate share the same shape
    out_gated = torch.empty_like(out_norm)
    total_elements = out_norm.numel()
    BLOCK_SIZE = 8192  # reasonable tile size for H100
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    elementwise_mul_kernel[grid](
        out_norm,
        out_gate,
        out_gated,
        total_elements,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --- 8. Final linear projection to original dim ---
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_gated, to_out_weight)  # shape [B,N,N,dim]

    return output

"""
TriMul (outgoing) forward pass with a small Triton kernel.

Algorithm
---------
1. Layer‑Norm the input tensor over the last channel dimension (`dim`).
2. Project the normalized tensor to a hidden dimension (`hidden_dim`) with two
   linear layers (`left_proj` and `right_proj`).
3. (Optional) Apply the pairwise mask – broadcasted over the hidden dimension.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from the
   same normalized tensor.
5. Fuse the gates: `left *= left_gate`, `right *= right_gate`.
6. Perform the core triangular multiplicative update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is realised as a batched matrix multiplication:
      left_perm  = left .permute(0,3,1,2)  # (B, H, I, K)
      right_perm = right.permute(0,3,2,1) # (B, H, K, J)
      out_perm   = torch.matmul(left_perm, right_perm)  # (B, H, I, J)
      out        = out_perm.permute(0,2,3,1)            # (B, I, J, H)
7. Apply a second Layer‑Norm over the hidden dimension.
8. Multiply the normalized result by `out_gate` using a Triton element‑wise
   kernel (the only required Triton component).
9. Final linear projection back to the original channel dimension (`dim`).

All heavy linear operations are performed with PyTorch (cuBLAS) for maximal
throughput; the Triton kernel only handles the final element‑wise modulation.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _elemwise_mul_kernel(a_ptr, b_ptr, c_ptr, size,
                         BLOCK_SIZE: tl.constexpr):
    """
    c = a * b  (element‑wise) for `size` elements.
    All tensors are assumed to be contiguous 1‑D views.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = offs < size
    a = tl.load(a_ptr + offs, mask=mask)
    b = tl.load(b_ptr + offs, mask=mask)
    tl.store(c_ptr + offs, a * b, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul operation.

    Args:
        data: tuple (input_tensor, mask_tensor, weight_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask_tensor : torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weight_dict : dict containing the model weights:
                * "norm.weight", "norm.bias"
                * "left_proj.weight", "right_proj.weight"
                * "left_gate.weight", "right_gate.weight", "out_gate.weight"
                * "to_out_norm.weight", "to_out_norm.bias"
                * "to_out.weight"
            - config_dict : dict with keys "dim", "hidden_dim", optional "nomask"
    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ensure contiguity / correct dtype
    x = x.contiguous()
    dtype = x.dtype
    device = x.device

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over channel dimension `dim`
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=1e-5
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections to hidden_dim
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_w)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Apply pairwise mask (if present)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # broadcast mask over the hidden dimension
        mask = mask.to(dtype).contiguous()
        mask_exp = mask.unsqueeze(-1)       # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates from the *unmasked* normalized tensor
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    # ------------------------------------------------------------------
    # 5) Fuse gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul: batched matmul across hidden_dim
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape for batched matmul: (B, H, I, K) @ (B, H, K, J)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, I, K)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, K, J)
    out_perm = torch.matmul(left_perm, right_perm)      # (B, H, I, J)
    out = out_perm.permute(0, 2, 3, 1).contiguous()    # (B, I, J, H)

    # ------------------------------------------------------------------
    # 7) Second LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w,
        bias=to_out_norm_b, eps=1e-5
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 8) Element‑wise multiplication with `out_gate` using Triton
    # ------------------------------------------------------------------
    out_gate = out_gate.contiguous()
    out_norm = out_norm.contiguous()
    out_gated = torch.empty_like(out_norm)

    total_elems = out_norm.numel()
    BLOCK_SIZE = 8192  # appropriate for H100; can be tuned
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),)
    _elemwise_mul_kernel[grid](
        out_norm,
        out_gate,
        out_gated,
        total_elems,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # ------------------------------------------------------------------
    # 9) Final projection back to original channel dimension `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"]
    output = F.linear(out_gated, to_out_w)  # (B, N, N, dim)

    return output
