```
@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_b, stride_i, stride_k, stride_d,
                  BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
                  BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
    pid = tl.program_id(0)  # tile index across output dimensions (i and j)
    num_pid_m = tl.cdiv(N, BLOCK_M)
    num_pid_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_pid_n
    pid_n = pid % num_pid_n

    # offsets for i and j
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    for b in range(B):
        # offsets for batch
        batch_offset = b * stride_b

        # accumulator for each d block
        for d_start in range(0, D, BLOCK_D):
            d_end = tl.minimum(d_start + BLOCK_D, D)
            cur_block_D = d_end - d_start
            # accumulator per d slice
            acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

            # loop over K dimension
            for k in range(0, N, BLOCK_K):
                cur_K = tl.minimum(BLOCK_K, N - k)
                # offsets for k
                offs_k = k + tl.arange(0, BLOCK_K)

                # Load left: shape (BLOCK_M, cur_K, BLOCK_D)
                left_ptrs = left_ptr + batch_offset + offs_i[:, None, None] * stride_i + offs_k[None, :, None] * stride_k + (d_start + tl.arange(0, BLOCK_D))[None, None, :] * stride_d
                left = tl.load(left_ptrs, mask=mask_i[:, None, None] & (offs_k < N)[:, None] & (tl.arange(0, BLOCK_D) < cur_block_D)[None, None, :], other=0.0).to(tl.float32)

                # Load right transposed (K,N -> shape cur_K, BLOCK_N, BLOCK_D)
                # Offsets: k dimension outer, j dimension inner
                right_ptrs = right_ptr + batch_offset + (d_start + tl.arange(0, BLOCK_D))[None, None, :] * stride_d + offs_k[:, None, None] * stride_k + offs_j[None, :, None] * stride_j
                right = tl.load(right_ptrs, mask=(offs_k < N)[:, None, None] & mask_j[None, :] & (tl.arange(0, BLOCK_D) < cur_block_D)[None, None, :], other=0.0).to(tl.float32)

                # Now compute outer product contribution for each d slice.
                # left_slice: (BLOCK_M, cur_K)
                # right_slice: (cur_K, BLOCK_N)
                # We can use tl.dot for each d:
                for d_off in range(cur_block_D):
                    # left_d: (BLOCK_M, cur_K)
                    left_d = left[:, :, d_off]  # shape (M,K)
                    # right_d: (cur_K, BLOCK_N)
                    right_d = right[:, :, d_off]  # shape (K,N)
                    acc[:, :, d_off] += tl.dot(left_d, right_d, trans_b=False)  # left (M,K) * right (K,N) = (M,N)
            # store results
            out_ptrs = out_ptr + batch_offset + offs_i[:, None, None] * stride_i + offs_j[None, :, None] * stride_j + (d_start + tl.arange(0, BLOCK_D))[None, None, :] * stride_d
            tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :] & (tl.arange(0, BLOCK_D) < cur_block_D)[None, None, :])

    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    device = input_tensor.device
    # layernorm x_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # left projection
    left = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right = torch.nn.functional.linear(x, weights['right_proj.weight'])
    if not config.get('nomask', False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f
    # gates
    left_gate = torch.nn.functional.linear(x, weights['left_gate.weight']).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights['right_gate.weight']).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights['out_gate.weight']).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # TriMul contraction: custom Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    B, N, _, D = left.shape  # D = hidden_dim

    out = torch.empty_like(left)  # shape B,N,N,D

    # define block size constants
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 8

    for b in range(B):

        left_b = left[b]  # shape N,N,D
        right_b = right[b]
        out_b = out[b]

        # Compute strides in elements
        stride_i = left_b.stride(0)
        stride_k = left_b.stride(1)
        stride_d = left_b.stride(2)

        stride_oi = out_b.stride(0)
        stride_oj = out_b.stride(1)
        stride_od = out_b.stride(2)

        grid = ( (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N )
        trimul_kernel[grid](
            left_b,
            right_b,
            out_b,
            N, D,

            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,

"""
Trident‑Mul (TriMul) “outgoing” forward kernel.

The core of the module is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where `left` and `right` are the projected & gated pair‑wise tensors.
The contraction is an N×N matrix multiplication for each hidden channel `d`,
and we implement it with a custom Triton kernel that tiles the i, j and k
dimensions while iterating over the hidden dimension in small blocks.
All other operations (LayerNorm, Linear projections, gating and final
projection) are performed with native PyTorch functions – only the O(N³)
contraction is off‑loaded to Triton.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left,          # *fp32  [N, N, D]  (batch slice)
    right,         # *fp32  [N, N, D]  (batch slice)
    out,           # *fp32  [N, N, D]  (batch slice)
    N, D,          # int32
    stride_li, stride_lk, stride_ld,   # int64 strides of left/right (i, k, d)
    stride_oi, stride_oj, stride_od,   # int64 strides of out   (i, j, d)
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Perform the sum‑over‑k for each output tile (i‑block, j‑block) and a
    block of hidden channels d.  The algorithm is:
        for d in blocks(D):
            acc = 0
            for k in blocks(N):
                left_tile  = left[i, k, d]   # (M, K, d_block)
                right_tile = right[j, k, d]  # (N, K, d_block) -> transposed to (K, N)
                acc += dot(left_tile[..., d], right_tile[..., d])
            write acc to out[i, j, d]
    """
    pid_i = tl.program_id(0)   # tile along the first sequence axis (i)
    pid_j = tl.program_id(1)   # tile along the second sequence axis (j)

    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # indices of the current tile
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # number of blocks in the channel dimension
    num_d_blocks = tl.cdiv(D, BLOCK_D)

    for d_block in range(num_d_blocks):
        d0 = d_block * BLOCK_D
        # d indices for this block
        offs_d = d0 + tl.arange(0, BLOCK_D)
        mask_d = offs_d < D

        # accumulator for this (i,j) tile and d‑block
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # iterate over the reduction dimension k
        num_k_blocks = tl.cdiv(N, BLOCK_K)
        for k_block in range(num_k_blocks):
            k0 = k_block * BLOCK_K
            offs_k = k0 + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # ----- load left[i, k, d] -------------------------------------------------
            # shape: (M, K, Dblock)
            left_ptrs = (
                left
                + offs_i[:, None, None] * stride_li
                + offs_k[None, :, None] * stride_lk
                + offs_d[None, None, :] * stride_ld
            )
            left_tile = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )
            left_tile = left_tile.to(tl.float32)

            # ----- load right[j, k, d] transposed as (k, j, d) ------------------------
            # shape: (K, N, Dblock)   (k is outer, j inner)
            right_ptrs = (
                right
                + offs_k[:, None, None] * stride_lk        # k dimension
                + offs_j[None, :, None] * stride_li        # j dimension uses the same stride as left's i‑axis
                + offs_d[None, None, :] * stride_ld
            )
            right_tile = tl.load(
                right_ptrs,
                mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :],
                other=0.0,
            )
            right_tile = right_tile.to(tl.float32)

            # ----- dot for each hidden channel in the block -------------------------
            for d in range(BLOCK_D):
                # left_d : (M, K)   right_d : (K, N)
                left_d = left_tile[:, :, d]
                right_d = right_tile[:, :, d]
                acc[:, :, d] += tl.dot(left_d, right_d)

        # ----- write the accumulated block to the output ---------------------------
        out_ptrs = (
            out
            + offs_i[:, None, None] * stride_oi
            + offs_j[None, :, None] * stride_oj
            + offs_d[None, None, :] * stride_od
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :] & mask_d[None, None, :],
        )


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim]
            - mask  : torch.Tensor [B, N, N] (may be ignored when config["nomask"]=True)
            - weights: dict of model weights (see keys in the docstring)
            - config : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # 1. LayerNorm on the input (dim -> dim)
    # -------------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # -------------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -------------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # -------------------------------------------------------------------------
    # 3. Optional mask (applied before gating)
    # -------------------------------------------------------------------------
    if not nomask:
        # mask is bool / uint8 ; cast to same dtype as left/right
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gating signals (sigmoid) and gated multiplication
    # -------------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 5. TriMul contraction with Triton (adds O(N³·hidden_dim) work)
    # -------------------------------------------------------------------------
    # Ensure contiguous layout for correct stride extraction
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, D = left.shape   # D == hidden_dim
    assert D == hidden_dim, "hidden dimension mismatch"

    # Output tensor (will hold the contracted result)
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # Tile sizes – empirically good for H100 / hidden_dim ≤ 384
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    BLOCK_D = 8

    for b in range(B):
        # slice out a single batch element (shape [N,N,hidden_dim])
        left_b = left[b]
        right_b = right[b]
        out_b = out[b]

        # strides in element units (not bytes)
        stride_li = left_b.stride(0)   # i‑axis stride (N*D)
        stride_lk = left_b.stride(1)   # k‑axis stride (D)
        stride_ld = left_b.stride(2)   # d‑axis stride (1)

        stride_oi = out_b.stride(0)    # i‑axis stride (N*D)
        stride_oj = out_b.stride(1)    # j‑axis stride (D)
        stride_od = out_b.stride(2)    # d‑axis stride (1)

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,   # number of i‑tiles
            (N + BLOCK_N - 1) // BLOCK_N,   # number of j‑tiles
        )

        trimul_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            D,
            stride_li,
            stride_lk,
            stride_ld,
            stride_oi,
            stride_oj,
            stride_od,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # -------------------------------------------------------------------------
    # 6. Output LayerNorm + gating
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    # -------------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])

    return out
