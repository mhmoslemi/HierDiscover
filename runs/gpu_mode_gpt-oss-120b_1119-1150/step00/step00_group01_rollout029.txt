def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    import torch.nn.functional as F
    
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # LayerNorm: use weight, bias
    # Compute mean and var:
    x = input_tensor
    B, N, _, _ = x.shape

    # compute mean/var
    # Use torch functions for speed
    mean = x.mean(-1, keepdim=True)
    var = x.var(-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # Linear projections
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    if not config.get("nomask", True):
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

             (N + BLOCK_J - 1) // BLOCK_J,

        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        N, hidden_dim,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D,
    )

    out = out.to(torch.float32)

    # to_out_norm
    out_norm = (out - out.mean(-1, keepdim=True)) / torch.sqrt(out.var(-1, unbiased=False, keepdim=True) + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Apply out_gate
    out_norm = out_norm * out_gate

    # Final linear
    out_final = F.linear(out_norm, weights["to_out.weight"])

    return out_final

import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    N, hidden_dim,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_b = tl.program_id(2)

    # offsets for i and j
    i_offsets = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask for out-of-bounds
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # allocate accumulator for each d block
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over the k dimension
    for k_base in range(0, N, BLOCK_K):
        # offsets for k within the block
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        # mask for k
        k_mask = k_offsets < N

        # Load left block: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = (left_ptr
                     + pid_b * stride_left_b
                     + i_offsets[:, None, None] * stride_left_i
                     + k_offsets[None, :, None] * stride_left_k
                     + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d)

        # Apply mask for i and k and d
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None], other=0.0)
        # left is float16; cast to float32
        left = left.to(tl.float32)

        # Load right block: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = (right_ptr
                     + pid_b * stride_right_b
                     + j_offsets[:, None, None] * stride_right_j
                     + k_offsets[None, :, None] * stride_right_k
                     + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d)

        right = tl.load(right_ptrs, mask=j_mask[:, None, None] & k_mask[None, :, None], other=0.0)
        right = right.to(tl.float32)

        # compute contribution for each k:
        # left_exp shape (BLOCK_I, BLOCK_K, 1, BLOCK_D)
        # right_exp shape (1, BLOCK_J, BLOCK_K, BLOCK_D)
        # Keep broadcasting: we can reshape left to (BLOCK_I, BLOCK_K, 1, BLOCK_D) via .reshape?
        # We can simply broadcast via newaxis:
        # left[:, :, None, :] * right[None, :, :, :]
        # However, left shape is (I,K,D) and right shape is (J,K,D). To broadcast to (I,J,K,D):
        # we need left_exp = left[:, :, None, :]  -> shape (I,K,1,D)
        # right_exp = right[None, :, :, :] -> shape (1,J,K,D)
        # Multiply left_exp * right_exp yields shape (I,J,K,D).
        # Then sum along axis=2 (K)
        prod = left[:, :, None, :] * right[None, :, :, :]  # shape (I,J,K,D)
        # sum over k dimension (axis=2)
        acc += tl.sum(prod, axis=2)  # result shape (I,J,D)

    # Store accumulator to out
    d_offsets = tl.arange(0, BLOCK_D)
    out_ptrs = (out_ptr
                + pid_b * stride_out_b
                + i_offsets[:, None, None] * stride_out_i
                + j_offsets[None, :, None] * stride_out_j
                + d_offsets[None, None, :] * stride_out_d)

    tl.store(out_ptrs, acc.to(tl.float16), mask=i_mask[:, None, None] & j_mask[None, :, None])

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel performing the 4d einsum reduction
@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[b,i,j,:] = sum_k left[b,i,k,:] * right[b,j,k,:].

    left: [B, N, N, D]
    right: [B, N, N, D]
    out: [B, N, N, D]
    Tile sizes: BLOCK_M x BLOCK_N output tile, reduction over K in BLOCK_K steps, D is processed in a single vector (BLOCK_D == hidden_dim).
    """
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute i, j indices for this tile
    i_offsets = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offsets = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # boolean masks for out-of-bound indices
    i_mask = i_offsets < N
    j_mask = j_offsets < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # iterate over K dimension
    for k_base in range(0, N, BLOCK_K):
        k_offsets = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_offsets < N

        # Load a tile from left: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        a_ptrs = (left_ptr + pid_b * stride_left_b
                  + i_offsets[:, None, None] * stride_left_i
                  + k_offsets[None, :, None] * stride_left_k
                  + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d)
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None, None] & k_mask[None, :, None],
                    other=0.0).to(tl.float32)

        # Load a tile from right: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        b_ptrs = (right_ptr + pid_b * stride_right_b
                  + j_offsets[:, None, None] * stride_right_j
                  + k_offsets[None, :, None] * stride_right_k
                  + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d)
        b = tl.load(b_ptrs,
                    mask=j_mask[:, None, None] & k_mask[None, :, None],
                    other=0.0).to(tl.float32)

        # Loop over the inner K dimension to accumulate outer products.
        # This uses broadcasting: a[i, kk, :] (1) * b[j, kk, :] (1) -> (i,j,d)
        for kk in range(0, BLOCK_K):
            a_vec = a[:, kk, :]                # (BLOCK_M, D)
            b_vec = b[:, kk, :]                # (BLOCK_N, D)
            # broadcast to (BLOCK_M, BLOCK_N, D)
            acc += a_vec[:, None, :] * b_vec[None, :, :]

    # Write the accumulator to the output tensor
    d_offsets = tl.arange(0, BLOCK_D)
    out_ptrs = (out_ptr + pid_b * stride_out_b
                + i_offsets[:, None, None] * stride_out_i
                + j_offsets[None, :, None] * stride_out_j
                + d_offsets[None, None, :] * stride_out_d)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None, None] & j_mask[None, :, None])

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul module.
    The function performs LayerNorm, linear projections, gating, an
    optimized Triton kernel for the triangular multiplicative update,
    a second LayerNorm + gating, and a final linear projection.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack inputs
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1. LayerNorm on the last dimension (dim)
    # -----------------------------------------------------------------
    # Compute mean & variance manually to avoid extra kernels
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias) to hidden dimension
    # -----------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (mask shape [bs,N,N])
    # -----------------------------------------------------------------
    if not config.get("nomask", True):
        # ensure mask is float for multiplication
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4. Gates via sigmoid(linear(x))
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triangular multiplicative update (einsum) using Triton
    # -----------------------------------------------------------------
    # Cast to half-precision for the kernel (triton works best with fp16)
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # Allocate output tensor for the contraction
    out_h = torch.empty_like(left_h, dtype=torch.float16)  # shape [bs,N,N,hidden_dim]

    # Determine tile sizes. Trade‑off between register pressure and occupancy.
    BLOCK_M = 8   # rows (i)
    BLOCK_N = 8   # cols (j)
    BLOCK_K = 16  # reduction chunk
    BLOCK_D = hidden_dim  # full hidden dimension as a vector register

    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             bs)

    # launch the kernel
    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Convert back to float32 for the rest of the pipeline
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension + out gate
    # -----------------------------------------------------------------
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]

    # Apply output gate (broadcasted)
    out_norm = out_norm * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to original dimension
    # -----------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The module implements the “outgoing” triangular multiplicative update
described in AlphaFold‑3.  Given an input tensor `x` of shape
[B, N, N, C] the computation is:

    x̂   = LayerNorm(x)                                 # over C
    L    = Linear(x̂, w_left)                           # → [B,N,N,H]
    R    = Linear(x̂, w_right)                          # → [B,N,N,H]
    L   *= mask (optional)   ;   R   *= mask
    gL   = sigmoid(Linear(x̂, w_left_gate))
    gR   = sigmoid(Linear(x̂, w_right_gate))
    gO   = sigmoid(Linear(x̂, w_out_gate))
    L   *= gL                                            # gating
    R   *= gR

    # core O(N³·H) operation:
    #   out[b,i,j,:] = Σ_k  L[b,i,k,:] * R[b,j,k,:]       (element‑wise product)
    # This contraction is performed by a custom Triton kernel that tiles the
    # i and j dimensions and reduces over k.  The hidden dimension H is kept
    # as a vector register (no extra tiling) – H is 128 in all test cases.

    out  = LayerNorm(out)                               # over H
    out  *= gO                                            # out‑gate
    out  = Linear(out, w_out)                            # → [B,N,N,C]

All operations are in float32 except the large O(N³) contraction,
which works in float16 internally for speed and is cast back to float32
before the final layers.  The kernel is written for H100 (SM 90) and
uses BLOCK_M × BLOCK_N output tiles (8×8) and a BLOCK_K reduction
chunk of 16.  This balances register pressure and occupancy while
keeping the implementation simple and correct.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    N, hidden_dim,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    left/right/out have shape [B, N, N, D] where D = hidden_dim.
    The kernel processes an (i,j) tile of size BLOCK_M×BLOCK_N.
    The reduction over k is done in chunks of BLOCK_K.
    The hidden dimension is kept as a vector register of length BLOCK_D
    (BLOCK_D must be equal to hidden_dim – 128 in our tests).
    """
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # ----- tile coordinates -------------------------------------------------
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # ----- accumulator (float32 for accuracy) -------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k -------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        k_off = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # Load a block from left: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        a_ptrs = (left_ptr
                  + pid_b * stride_left_b
                  + i_off[:, None, None] * stride_left_i
                  + k_off[None, :, None] * stride_left_k
                  + tl.arange(0, BLOCK_D)[None, None, :] * stride_left_d)
        a = tl.load(a_ptrs,
                    mask=i_mask[:, None, None] & k_mask[None, :, None],
                    other=0.0).to(tl.float32)

        # Load a block from right: shape (BLOCK_N, BLOCK_K, BLOCK_D)
        b_ptrs = (right_ptr
                  + pid_b * stride_right_b
                  + j_off[:, None, None] * stride_right_j
                  + k_off[None, :, None] * stride_right_k
                  + tl.arange(0, BLOCK_D)[None, None, :] * stride_right_d)
        b = tl.load(b_ptrs,
                    mask=j_mask[:, None, None] & k_mask[None, :, None],
                    other=0.0).to(tl.float32)

        # Accumulate outer‑product for each k within the block.
        # a[i, kk, :] has shape (BLOCK_M, D)
        # b[j, kk, :] has shape (BLOCK_N, D)
        # Broadcasting yields (BLOCK_M, BLOCK_N, D)
        for kk in range(0, BLOCK_K):

            b_vec = b[:, kk, :]                # (BLOCK_N, D)
            acc += a_vec[:, None, :] * b_vec[None, :, :]

    # ----- write back --------------------------------------------------------
    d_off = tl.arange(0, BLOCK_D)
    out_ptrs = (out_ptr
                + pid_b * stride_out_b
                + i_off[:, None, None] * stride_out_i
                + j_off[None, :, None] * stride_out_j
                + d_off[None, None, :] * stride_out_d)

    tl.store(out_ptrs,
             acc.to(tl.float16),
             mask=i_mask[:, None, None] & j_mask[None, :, None])

        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, C]  (float32)
            mask         : [B, N, N]      (bool or float, may be omitted)
            weights      : dict of model parameters
            config       : dict containing 'dim' and 'hidden_dim' and 'nomask'

    Returns:
        Tensor of shape [B, N, N, C] (float32)
    """
    # ----------------------------------------------------------------------
    # Unpack arguments
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm (over C)
    # ----------------------------------------------------------------------
    mean = input_tensor.mean(-1, keepdim=True)
    var = input_tensor.var(-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * weights["norm.weight"] + weights["norm.bias"]

    # ----------------------------------------------------------------------
    # 2. Linear projections to hidden dimension (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3. Optional binary mask (mask shape [B, N, N])
    # ----------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask may be bool; convert to float32 for multiplication
        mask_f = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_f
        right = right * mask_f

    # ----------------------------------------------------------------------
    # 4. Gating (sigmoid(linear(x)))
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5. Triangular multiplicative update – heavy O(N³·H) part
    #    Executed in float16 inside a Triton kernel.
    # ----------------------------------------------------------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Tile sizes – tuned for H100, hidden_dim is 128 → BLOCK_D = 128.
    BLOCK_M = 8   # rows (i)
    BLOCK_N = 8   # cols (j)
    BLOCK_K = 16  # reduction chunk
    BLOCK_D = hidden_dim  # vector length per output element

    grid = ((N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            B)

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        left_h.stride(0), left_h.stride(1), left_h.stride(2), left_h.stride(3),
        right_h.stride(0), right_h.stride(1), right_h.stride(2), right_h.stride(3),
        out_h.stride(0), out_h.stride(1), out_h.stride(2), out_h.stride(3),
        N, hidden_dim,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Back to float32 for the remaining layers
    out = out_h.to(torch.float32)

    # ----------------------------------------------------------------------
    # 6. Second LayerNorm over hidden dimension + output gate
    # ----------------------------------------------------------------------
    out_mean = out.mean(-1, keepdim=True)
    out_var = out.var(-1, unbiased=False, keepdim=True)
    out_norm = (out - out_mean) / torch.sqrt(out_var + eps)
    out_norm = out_norm * weights["to_out_norm.weight"] + weights["to_out_norm.bias"]
    out_norm = out_norm * out_gate

    # ----------------------------------------------------------------------
    # 7. Final linear projection back to original dimension
    # ----------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])

    return output
