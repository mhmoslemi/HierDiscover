def matmul_transpose(A: torch.Tensor, B: torch.Tensor):
   assert A.shape == B.shape
   batch, M, K = A.shape
   _, N, K2 = B.shape
   assert K2 == K
   # Ensure contiguous.
   A = A.contiguous()
   B = B.contiguous()
   C = torch.empty((batch, M, N), device=A.device, dtype=A.dtype) # we want possibly float32
   # compute strides
   stride_am, stride_ak = A.stride(1), A.stride(2)  # (M stride, K stride)
   stride_ab = A.stride(0)  # batch stride
   stride_bn, stride_bk = B.stride(1), B.stride(2)
   stride_bb = B.stride(0)
   stride_cm, stride_cn = C.stride(1), C.stride(2)
   stride_cb = C.stride(0)
   # constants
   BLOCK_M = 64
   BLOCK_N = 64
   BLOCK_K = 64
   grid = (batch * ((M + BLOCK_M - 1)//BLOCK_M) * ((N + BLOCK_N - 1)//BLOCK_N),)
   matmul_transpose_kernel[grid](
       A.data_ptr(), B.data_ptr(), C.data_ptr(),
       M, N, K,
       stride_am, stride_ak, stride_ab,
       stride_bn, stride_bk, stride_bb,
       stride_cm, stride_cn, stride_cb,
       BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
   )
   return C

@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_ams, stride_ak, stride_ab,
    stride_bns, stride_bk, stride_bb,
    stride_cms, stride_cns, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid = tl.program_id(axis=0)
    # total number of blocks per batch
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    total_blocks = num_blocks_m * num_blocks_n
    batch_idx = pid // total_blocks
    block_idx = pid % total_blocks
    pid_m = block_idx // num_blocks_n
    pid_n = block_idx % num_blocks_n

    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_base = a_ptr + batch_idx * stride_ab
    b_base = b_ptr + batch_idx * stride_bb
    c_base = c_ptr + batch_idx * stride_cb

    # pointers
    a_ptrs = a_base + offs_m[:, None] * stride_ams + offs_k[None, :] * stride_ak
    b_ptrs = b_base + offs_n[None, :] * stride_bns + offs_k[:, None] * stride_bk

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load
        a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (offs_k[None, :] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_n[None, :] < N) & (offs_k[:, None] < K), other=0.0)
        # compute
        acc += tl.dot(a, b)   # result shape (BLOCK_M, BLOCK_N)
        # increment
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk
    # store
    c_ptrs = c_base + offs_m[:, None] * stride_cms + offs_n[None, :] * stride_cns
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

"""
TriMul (outgoing) – Triton‑accelerated implementation

The TriMul operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and computes

    X̂ = LayerNorm(X)
    L  = (X̂·W_left)  ⊙ σ(X̂·W_left_gate)  ⊙ M                (B,N,N,H)
    R  = (X̂·W_right) ⊙ σ(X̂·W_right_gate) ⊙ M                (B,N,N,H)

    O  = Σ_k L_{i,k,:} · R_{j,k,:}  = (L @ Rᵀ)   (B,N,N,H)

    Ô = LayerNorm(O) ⊙ σ(X̂·W_out_gate)
    Y  = Ô·W_out                                            (B,N,N,C)

where H = hidden_dim, M is the optional binary mask (broadcast over H),
σ is the sigmoid and “·” denotes a linear projection (no bias).
The core N³ operation is the batched matrix multiplication
   C_b = A_b @ B_bᵀ   with  A_b,R_b ∈ ℝ^{N×N}.  We fuse this
step using a custom Triton kernel that processes the (B·H) independent
matmul tasks in one launch.

The implementation below:
* uses `torch.nn.functional.layer_norm` and `F.linear` for the
  inexpensive per‑element operations,
* packs the (B·hidden_dim) independent GEMMs into a single Triton
  kernel (`matmul_transpose_kernel`) that computes C = A @ Bᵀ,
* keeps the final tensors in FP32 (the kernel accumulates in FP32
  even when inputs are FP16).

The kernel launches with modest block sizes (64×64×64) which give
high occupancy on an H100 while fitting in registers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ   where
#   A : (batch, M, K)   (row‑major, stride_ams, stride_ak)
#   B : (batch, N, K)   (row‑major, stride_bns, stride_bk)
#   C : (batch, M, N)
# ----------------------------------------------------------------------
@triton.jit
def matmul_transpose_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K) strides
    stride_bn, stride_bk, stride_bb,   # B: (batch, N, K) strides
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    # ------------------------------------------------------------------
    # 1) map program id to a unique (batch, block_m, block_n) tile
    # ------------------------------------------------------------------
    pid = tl.program_id(axis=0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    tiles_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // tiles_per_batch
    tile_idx  = pid % tiles_per_batch
    pid_m = tile_idx // num_blocks_n
    pid_n = tile_idx % num_blocks_n

    # ------------------------------------------------------------------
    # 2) compute offsets for the current tile
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # base pointers for this batch
    a_base = a_ptr + batch_idx * stride_ab
    b_base = b_ptr + batch_idx * stride_bb
    c_base = c_ptr + batch_idx * stride_cb

    # pointers to the start of the tile (M × K) and (N × K)
    a_ptrs = a_base + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
    # Note: B is stored as (N, K); we need Bᵀ → (K, N) for dot
    b_ptrs = b_base + offs_n[None, :] * stride_bn + offs_k[:, None] * stride_bk

    # ------------------------------------------------------------------
    # 3) accumulate over K
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Load tiles, mask out‑of‑bounds entries
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_n[None, :] < N) & (offs_k[:, None] < K),
                    other=0.0)
        # Promote to FP32 for accumulation
        a = a.to(tl.float32)
        b = b.to(tl.float32)
        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_N)

        # Advance K‑dimension pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # ------------------------------------------------------------------
    # 4) write back
    # ------------------------------------------------------------------
    c_ptrs = c_base + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    mask = (offs_m[:, None] < M) & (offs_n[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)


def triton_matmul_transpose(A: torch.Tensor,
                            B: torch.Tensor,
                            *,
                            BLOCK_M: int = 64,
                            BLOCK_N: int = 64,
                            BLOCK_K: int = 64) -> torch.Tensor:
    """
    Batched matmul C = A @ Bᵀ   (A,B shape: (B, M, K) , (B, N, K))
    Returns C shape (B, M, N).  A and B must have identical batch size.
    """
    assert A.shape[0] == B.shape[0]
    assert A.shape[2] == B.shape[2]    # K must match
    batch, M, K = A.shape
    _, N, _ = B.shape

    A = A.contiguous()
    B = B.contiguous()
    C = torch.empty((batch, M, N), device=A.device, dtype=torch.float32)

    # Strides (in elements, not bytes)
    stride_am, stride_ak = A.stride(1), A.stride(2)
    stride_ab = A.stride(0)

    stride_bn, stride_bk = B.stride(1), B.stride(2)
    stride_bb = B.stride(0)

    stride_cm, stride_cn = C.stride(1), C.stride(2)
    stride_cb = C.stride(0)

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_programs = batch * num_blocks_m * num_blocks_n
    grid = (total_programs,)

    matmul_transpose_kernel[grid](
        A.data_ptr(),
        B.data_ptr(),
        C.data_ptr(),
        M,
        N,
        K,
        stride_am,
        stride_ak,
        stride_ab,
        stride_bn,
        stride_bk,
        stride_bb,
        stride_cm,
        stride_cn,
        stride_cb,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # H100 prefers 4–8 warps per block
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator (AlphaFold3 style).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (binary 0/1)
        - weights      : dict of model parameters (see code)
        - config       : dict containing at least “dim” and “hidden_dim”

    Returns
    -------
    torch.Tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # unpack & basic shapes
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps        = 1e-5

    # ------------------------------------------------------------------
    # LayerNorm over the channel dimension C
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gates
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------
    # Apply mask (if provided) – mask is broadcast on the hidden dim
    # ------------------------------------------------------------------
    mask = mask.to(dtype).unsqueeze(-1)               # (B, N, N, 1)
    left  = left_proj * mask * left_gate              # (B, N, N, H)
    right = right_proj * mask * right_gate

    # ------------------------------------------------------------------
    # Pack hidden dimension into the batch axis for the GEMM fusion
    # ------------------------------------------------------------------
    # (B, N, N, H) -> (B, H, N, N)
    left  = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B, H, N, _ = left.shape
    batch_h = B * H
    left_flat  = left.view(batch_h, N, N)   # (B·H, N, N)
    right_flat = right.view(batch_h, N, N)  # (B·H, N, N)

    # ------------------------------------------------------------------
    # Core N³ multiplication via the custom Triton kernel
    # ------------------------------------------------------------------
    out_flat = triton_matmul_transpose(left_flat, right_flat)

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Final LayerNorm on the hidden dimension + output gating
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)
    out = out * out_gate

    # ------------------------------------------------------------------
    # Linear projection back to original dimension C
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])

    return output
