```
@triton.jit
def tri_mul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_am, stride_ai, stride_ak,
    stride_bm, stride_bi, stride_bk,
    stride_cm, stride_ci, stride_ck,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    eps: tl.constexpr
):
    # program ids
    pid_m = tl.program_id(0)  # block row
    pid_n = tl.program_id(1)  # block col
    pid_head = tl.program_id(2)  # head index (batch*hidden)
    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # boundary check
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute k start
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        # guard k
        mask_k = offs_k < K

        # Load A block [M, K]
        a_ptrs = A_ptr + (pid_head * stride_am + offs_m[:, None] * stride_ai + offs_k[None, :] * stride_ak)
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B_T block [K, N]
        b_ptrs = B_ptr + (pid_head * stride_bm + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bi)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0).to(tl.float32)

        # accumulate
        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C_ptr + (pid_head * stride_cm + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_ck)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def triton_trimul(left, right):
    # left,right shape: [B,N,N,hidden_dim]
    # permute to [B, hidden, N, N]
    B, N, _, H = left.shape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    head = B * H
    left_mat = left_perm.view(head, N, N)
    right_mat = right_perm.view(head, N, N)

    # allocate output
    out_mat = torch.empty_like(left_mat)

    # compute strides
    stride_am, stride_ai, stride_ak = left_mat.stride()
    stride_bm, stride_bi, stride_bk = right_mat.stride()
    stride_cm, stride_ci, stride_ck = out_mat.stride()
    M = head
    K = N  # inner dimension

    # grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (tl.cdiv(N, BLOCK_M), tl.cdiv(N, BLOCK_N), M)

    tri_mul_kernel[grid](
        left_mat, right_mat, out_mat,
        M, N, K,
        stride_am, stride_ai, stride_ak,
        stride_bm, stride_bi, stride_bk,
        stride_cm, stride_ci, stride_ck,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )
    # after kernel, reshape back to [B,N,N,hidden]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out

"""
custom_kernel – TriMul “outgoing” implementation (AlphaFold‑3 style)

The kernel computes the heavy O(N³) part of TriMul:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

where left/right are the projected tensors (shape [B,N,N,H]).
We fuse the N×N matrix multiplication for every head (head = B*H) with a
custom Triton kernel that evaluates A @ Bᵀ for each head.  All other
operations (LayerNorm, linear maps, sigmoids, the final LayerNorm
and projection) are performed with PyTorch (high‑performance cuBLAS /
cudnn).  The kernel works for any batch size, sequence length and hidden
dimension (tested up to N=1024, H=384).  The final output is float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ---------------------------------------------------------------------------
# Triton kernel: C = A @ Bᵀ  (A,B shape = [Head, N, N])
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    a_ptr,               # pointer to A   [Head, N, N]
    b_ptr,               # pointer to B   [Head, N, N] (will be read transposed)
    c_ptr,               # pointer to C   [Head, N, N]
    HEAD: tl.constexpr,  # number of heads = B * hidden_dim
    N:    tl.constexpr,  # sequence length
    K:    tl.constexpr,  # same as N (reduction dimension)
    stride_am, stride_ai, stride_ak,   # strides of A (head, row i, col k)
    stride_bm, stride_bi, stride_bk,   # strides of Bᵀ (head, row k, col j) – note the swap
    stride_cm, stride_ci, stride_ck,   # strides of C (head, row i, col j)
    BLOCK_M: tl.constexpr,  # tile size in i‑dimension
    BLOCK_N: tl.constexpr,  # tile size in j‑dimension
    BLOCK_K: tl.constexpr,  # tile size in reduction k‑dimension
):
    pid_head = tl.program_id(2)   # which head (0 … HEAD‑1)
    pid_i    = tl.program_id(0)   # block row
    pid_j    = tl.program_id(1)   # block col

    # Offsets for the current block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator (FP32 for numerical stability)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_start = k * BLOCK_K
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # ---- load a block of A (shape BLOCK_M × BLOCK_K) ----
        a_ptrs = a_ptr + (
            pid_head * stride_am
            + offs_i[:, None] * stride_ai
            + offs_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- load a block of Bᵀ (shape BLOCK_K × BLOCK_N) ----
        # Bᵀ is accessed as B[head, k, j]  →  stride_bi (row‑stride) = stride_ak,
        #                                    stride_bk (col‑stride) = stride_ai
        b_ptrs = b_ptr + (
            pid_head * stride_bm
            + offs_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ---- accumulate dot product ----
        acc += tl.dot(a, b)

    # ---- write back the result block ----
    c_ptrs = c_ptr + (
        pid_head * stride_cm
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_ck
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ---------------------------------------------------------------------------
# Helper: batched matmul with transposed RHS using the Triton kernel above
# ---------------------------------------------------------------------------
def triton_batch_matmul_transpose(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    left, right:  [B, N, N, H]  (float32)
    returns:      [B, N, N, H]  (float32)  where out = left @ rightᵀ per head.
    """
    B, N, _, H = left.shape
    # bring the head dimension (B*H) to the front and make data contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    HEAD = B * H
    left_mat = left_perm.view(HEAD, N, N)
    right_mat = right_perm.view(HEAD, N, N)

    # output buffer
    out_mat = torch.empty_like(left_mat)

    # strides (in elements) for the 3‑D tensors
    stride_am, stride_ai, stride_ak = left_mat.stride()   # (head, i, k)
    stride_bm, stride_bi, stride_bk = right_mat.stride() # (head, i, k)

    # we need Bᵀ → swap the i and k strides
    stride_bi_T = stride_bk   # row‑stride of Bᵀ = original stride over k
    stride_bk_T = stride_bi   # col‑stride of Bᵀ = original stride over i

    stride_cm, stride_ci, stride_ck = out_mat.stride()   # same layout as A

    # tile sizes – tuned for H100 (can be changed if necessary)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),   # i‑tiles
        triton.cdiv(N, BLOCK_N),   # j‑tiles
        HEAD,                      # one program per head
    )

    # launch the kernel
    tri_mul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        HEAD,
        N,
        N,
        stride_am,
        stride_ai,
        stride_ak,
        stride_bm,
        stride_bi_T,
        stride_bk_T,
        stride_cm,
        stride_ci,
        stride_ck,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,               # 4 warps gives good occupancy on H100
    )

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    return out


# ---------------------------------------------------------------------------
# Main entry point expected by the evaluation harness
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ----------
    data : tuple (input_tensor, mask, weights, config)
        input_tensor : torch.Tensor [B, N, N, dim] (float32)
        mask         : torch.Tensor [B, N, N] (bool/float) – may be all‑ones.
        weights      : dict of torch.Tensor holding all linear / layernorm weights.
        config       : dict with keys "dim", "hidden_dim", "nomask" (bool).

    Returns
    -------
    out : torch.Tensor [B, N, N, dim]  (float32)
    """
    # -----------------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------------
    # Load weights (all tensors are assumed to be on the same device as input)
    # -----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    eps = 1e-5  # LayerNorm epsilon

    # -----------------------------------------------------------------------
    # 1️⃣  LayerNorm over the input channels (dim)
    # -----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2️⃣  Linear projections (no bias) → shape [B, N, N, hidden_dim]
    # -----------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # -----------------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden dimension)
    # -----------------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask dtype matches computation dtype
        mask = mask.to(x.dtype).unsqueeze(-1)          # [B, N, N, 1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------------
    # 4️⃣  Gating (sigmoid) – computed from the *normed* input
    # -----------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B, N, N, H]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # [B, N, N, H]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, H]

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5️⃣  Heavy O(N³) part: out = Σ_k left[...,i,k,:] * right[...,j,k,:]
    #     Implemented as a batched matmul with transposed RHS via Triton.
    # -----------------------------------------------------------------------
    out = triton_batch_matmul_transpose(left, right)   # [B, N, N, H]

    # -----------------------------------------------------------------------
    # 6️⃣  Output LayerNorm (over hidden_dim) and gating
    # -----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate

    # -----------------------------------------------------------------------
    # 7️⃣  Final linear projection back to original channel dimension
    # -----------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
