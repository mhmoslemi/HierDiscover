    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # Extract weights
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    # we assume all these are float32 (or maybe fp16). We'll convert to same dtype as input.

    # Apply first layernorm via Triton kernel for speed.
    # The kernel returns normalized tensor.
    # We'll implement a wrapper that calls the kernel.
    # We need to allocate output.
    x_norm = layernorm_triton(input_tensor, norm_weight, norm_bias, eps)
    # shape same as input.

    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply mask if present and not nomask
    if mask_tensor is not None:

        mask_exp = mask_tensor[..., None]
        left = left * mask_exp
        right = right * mask_exp

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Compute triple multiplication
    # Use batched matmul: reshape
    B, N, _, _ = left.shape
    # left shape B,N,N,hidden_dim
    # Permute
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_T = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)

    out_flat = torch.bmm(left_flat, right_T)  # shape (B*hidden_dim, N, N)

def layernorm_triton(x, weight, bias, eps=1e-5):
    # x: torch.Tensor (B,N,N,D)
    # weight,bias same dtype as x
    B,N,N,D = x.shape

    # compute strides
    stride_b, stride_i, stride_j, stride_d = x.stride()
    out = torch.empty_like(x)

    # define block sizes
    BLOCK_M = 64 # number of i per block
    BLOCK_N = 64 # number of j per block
    BLOCK_D = 128 # hidden_dim block (must match compile-time constant)

    # Determine grid
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
           )  # for each batch will be called separately

    # Loop over batch
    for b in range(B):
        # compute base pointer offsets (avoid using program_id for batch)
        # We can pass per-batch offset to kernel using a pointer offset:
        input_ptr = x.data_ptr() + b * stride_b * x.element_size()
        output_ptr = out.data_ptr() + b * stride_b * out.element_size()
        # call triton
        layernorm_kernel[grid](
            input_ptr, output_ptr,
            weight.data_ptr(), bias.data_ptr(),
            N, D,
            stride_in_i=stride_i,
            stride_in_j=stride_j,
            stride_in_d=stride_d,

@triton.jit
def layernorm_kernel(
    inp_ptr, out_ptr,
    weight_ptr, bias_ptr,
    N, D,
    stride_in_i, stride_in_j, stride_in_d,
    stride_out_i, stride_out_j, stride_out_d,
    eps: tl.constexpr,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # compute base indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # define ranges
    i_idx = i_base + tl.arange(0, BLOCK_M)
    j_idx = j_base + tl.arange(0, BLOCK_N)

    # mask for limits
    i_mask = i_idx < N
    j_mask = j_idx < N
    mask_ij = i_mask[:, None] & j_mask[None, :]

    # Initialize sums
    sum = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ones vector
    ones = tl.full((BLOCK_D,), 1.0, dtype=tl.float32)

    # Loop over d dimension
    for d_start in range(0, D, BLOCK_D):
        # d indices vector
        d_idx = tl.arange(0, BLOCK_D) + d_start
        d_mask = d_idx < D

        # compute pointers
        # i_idx and j_idx as shape (BLOCK_M, 1) and (1, BLOCK_N)
        # base offset
        ptrs = inp_ptr + i_idx[:, None] * stride_in_i + j_idx[None, :] * stride_in_j + d_idx[None, None, :] * stride_in_d
        # mask
        mask = mask_ij[:, :, None] & d_mask[None, None, :]
        x = tl.load(ptrs, mask=mask, other=0.0) # shape (BLOCK_M, BLOCK_N, BLOCK_D)
        sum += tl.dot(x, ones) # shape (BLOCK_M, BLOCK_N)
        sum_sq += tl.dot(x * x, ones)

    mean = sum / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # Second pass: write output normalized * weight + bias
    for d_start in range(0, D, BLOCK_D):
        d_idx = tl.arange(0, BLOCK_D) + d_start
        d_mask = d_idx < D
        # load x again
        ptrs = inp_ptr + i_idx[:, None] * stride_in_i + j_idx[None, :] * stride_in_j + d_idx[None, None, :] * stride_in_d
        mask = mask_ij[:, :, None] & d_mask[None, None, :]
        x = tl.load(ptrs, mask=mask, other=0.0)

        # normalize
        y = (x - mean[:, :, None]) * inv_std[:, :, None]
        # load weight and bias
        w = tl.load(weight_ptr + d_idx * 1, mask=d_mask, other=0.0) # shape (BLOCK_D,)
        b = tl.load(bias_ptr + d_idx * 1, mask=d_mask, other=0.0)

        y = y * w[None, None, :] + b[None, None, :]

        # store
        out_ptrs = out_ptr + i_idx[:, None] * stride_out_i + j_idx[None, :] * stride_out_j + d_idx[None, None, :] * stride_out_d
        tl.store(out_ptrs, y, mask=mask)

custom_kernel
==============

Implements the “outgoing” TriMul operator from AlphaFold3 using a mix of
PyTorch and Triton.  The heavy N³ reduction (`einsum('...i k d,...j k d -> ...i j d')`)
is performed with a very fast batched GEMM (`torch.bmm`) after the tensors have
been projected to the hidden dimension.  The two LayerNorms required by the
module (`norm` and `to_out_norm`) are written in Triton so that at least part
of the work is done on‑device without falling back to PyTorch’s implementation.

Algorithm
---------

1. **LayerNorm (input)**
   * Compute mean / variance across the last dimension `dim` for each
     `(batch, i, j)` entry with a custom Triton kernel.
   * Apply the learned scale (`norm.weight`) and bias (`norm.bias`).

2. **Linear projections & gates**
   * `left  = Linear(x, left_proj.weight)`   –>  shape `[B,N,N,hidden]`
   * `right = Linear(x, right_proj.weight)`
   * Compute three sigmoid‑gates (`left_gate`, `right_gate`, `out_gate`) with
     `Linear(..., *_gate.weight)`.

3. **Mask & gating**
   * If a mask is supplied, multiply `left` and `right` by `mask[...,None]`.
   * Apply element‑wise gating: `left *= left_gate`, `right *= right_gate`.

4. **TriMul core (`einsum`)**
   * Reshape the tensors so that the hidden dimension becomes an extra batch
     dimension:
     ```
     left_flat  = left.permute(0,3,1,2).reshape(B*hidden,N,N)   # (B*H,M,K)
     right_flat = right.permute(0,3,2,1).reshape(B*hidden,N,N)  # (B*H,K,N)
     ```
   * Perform a batched matrix multiplication:
     `out_flat = torch.bmm(left_flat, right_flat)`.
   * Reshape back to `[B,N,N,hidden]`.

5. **Second LayerNorm (hidden)**
   * Run the same Triton LayerNorm kernel on `out` using
     `to_out_norm.weight` / `bias`.

6. **Output gating & final projection**
   * Multiply by `out_gate` (sigmoid) and apply the final linear
     `Linear(..., to_out.weight)` to obtain the result `[B,N,N,dim]`.

All heavy arithmetic stays in `float32`; the kernel works for both
`float32` and `float16` (values are loaded as `float32`).  The kernel is
parameterised with compile‑time tile sizes that fit the problem constraints
(`dim` and `hidden_dim` are multiples of 128).

The implementation supports the full set of configurations described in the

with or without masks, and either normal or Cauchy input distributions).

    j0 = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_valid = i0 < N
    j_valid = j0 < N
    ij_valid = i_valid[:, None] & j_valid[None, :]

    # Broadcast i / j to 2‑D shape (BLOCK_M, BLOCK_N)
    i_off = i0[:, None] * stride_in_i
    j_off = j0[None, :] * stride_in_j

    # ------------------------------------------------------------------
    # First pass: compute mean and variance across D
    # ------------------------------------------------------------------
    sum_val = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    sum_sq = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    ones = tl.full((BLOCK_D,), 1.0, dtype=tl.float32)

    # Loop over the channel dimension in BLOCK_D sized chunks
    for d_start in range(0, D, BLOCK_D):
        d_off = tl.arange(0, BLOCK_D) + d_start
        d_valid = d_off < D

        # Pointer to the (i,j,d) values
        ptr = inp_ptr + i_off[:, None] + j_off[None, :] + d_off[None, None] * stride_in_d
        mask = ij_valid[:, :, None] & d_valid[None, None, :]

        x = tl.load(ptr, mask=mask, other=0.0)          # (M, N, D)
        sum_val += tl.dot(x, ones)                     # reduce over D
        sum_sq += tl.dot(x * x, ones)

    mean = sum_val / D
    var = sum_sq / D - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized values multiplied by weight & bias
    # ------------------------------------------------------------------
    for d_start in range(0, D, BLOCK_D):
        d_off = tl.arange(0, BLOCK_D) + d_start
        d_valid = d_off < D

        ptr_in = inp_ptr + i_off[:, None] + j_off[None, :] + d_off[None, None] * stride_in_d
        mask = ij_valid[:, :, None] & d_valid[None, None, :]

        x = tl.load(ptr_in, mask=mask, other=0.0)

        # Normalise
        y = (x - mean[:, :, None]) * inv_std[:, :, None]

        # Load weight & bias (1‑D tensors)
        w = tl.load(weight_ptr + d_off * 1, mask=d_valid, other=0.0)
        b = tl.load(bias_ptr + d_off * 1, mask=d_valid, other=0.0)

        y = y * w[None, None, :] + b[None, None, :]

        # Store result
        ptr_out = out_ptr + i_off[:, None] + j_off[None, :] + d_off[None, None] * stride_out_d
        tl.store(ptr_out, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of a 4‑D tensor
    (B, N, N, D) using the Triton kernel defined above.
    """
    assert x.dim() == 4, "LayerNorm expects a 4‑D tensor"
    B, N, _, D = x.shape
    # Tile sizes – chosen to respect the 1024‑thread limit per program
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_D = 128               # D is always a multiple of 128 (128 or 384)

    # Output tensor (same layout & dtype)
    out = torch.empty_like(x)

    # Strides (in number of elements, not bytes)
    s_b, s_i, s_j, s_d = x.stride()
    s_ob, s_oi, s_oj, s_od = out.stride()

    # Grid over i and j tiles (batch dimension is handled in Python)
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (grid_i, grid_j)

    weight = weight.contiguous()
    bias = bias.contiguous()

    for b in range(B):
        # Slice the b‑th batch (view is still contiguous)
        inp_slice = x[b]
        out_slice = out[b]

        layernorm_kernel[grid](
            inp_ptr=inp_slice,
            out_ptr=out_slice,
            weight_ptr=weight,
            bias_ptr=bias,
            N=N,
            D=D,
            stride_in_i=s_i,
            stride_in_j=s_j,
            stride_in_d=s_d,
            stride_out_i=s_oi,
            stride_out_j=s_oj,
            stride_out_d=s_od,
            eps=eps,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_D=BLOCK_D,
            num_warps=4,
        )
    return out


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor : torch.Tensor [B, N, N, dim]
            - mask         : torch.Tensor [B, N, N]  (may be None)
            - weights      : dict of model parameters (torch.Tensor)
            - config       : dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract weight tensors (make them contiguous and same dtype as input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"].contiguous()
    norm_bias = weights["norm.bias"].contiguous()

    left_proj_weight = weights["left_proj.weight"].contiguous()
    right_proj_weight = weights["right_proj.weight"].contiguous()

    left_gate_weight = weights["left_gate.weight"].contiguous()
    right_gate_weight = weights["right_gate.weight"].contiguous()
    out_gate_weight = weights["out_gate.weight"].contiguous()

    to_out_norm_weight = weights["to_out_norm.weight"].contiguous()
    to_out_norm_bias = weights["to_out_norm.bias"].contiguous()
    to_out_weight = weights["to_out.weight"].contiguous()

    # ------------------------------------------------------------------
    # 1) First LayerNorm (across `dim`)
    # ------------------------------------------------------------------
    x = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=eps)  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    # The Linear layers have no bias (bias=False in the reference code)
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 3) Gating signals
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 4) Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # Expand mask to broadcast over the hidden dimension
        mask_exp = mask_tensor[..., None]            # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 5) Apply gate multiplicatively
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Core TriMul:   out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    #-------------------------------------------------------------------
    # Reshape so that the hidden dimension becomes an extra batch dimension.
    B, N, _, _ = left.shape  # B, N, N, hidden
    # (B, hidden, N, N) → (B*hidden, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # L: (B*H, M, K)
    # right must be transposed on the last two dims before reshaping:
    right_flat = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N) # R: (B*H, K, N)

    # Batched matrix multiplication (highly optimized cuBLAS kernel)
    out_flat = torch.bmm(left_flat, right_flat)                        # (B*H, M, N)

    # Restore original layout: (B, N, N, hidden)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Second LayerNorm (across `hidden_dim`)
    # ------------------------------------------------------------------
    out = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 8) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return output
