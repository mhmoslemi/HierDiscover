def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_am, stride_ak, stride_aa,   # perhaps
                          stride_bk, stride_bn, stride_bb,
                          stride_cm, stride_cn, stride_cc,
                          BLOCK_M: tl.constexpr,
                          BLOCK_N: tl.constexpr,
                          BLOCK_K: tl.constexpr):
    pid = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    batch = pid // (tl.cdiv(M, BLOCK_M))
    block_m = pid % tl.cdiv(M, BLOCK_M)
    block_n = pid_n

    # Compute start offsets
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = A_ptr + batch * stride_aa + offs_m[:,None] * stride_am + offs_k[None,:] * stride_ak

"""
TriMul (outgoing) – fast Triton implementation of the core AlphaFold3 operation.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]).
2. Linear projections (no bias) → `left` and `right` of shape [B, N, N, H].
3. Optional mask (broadcast on the channel dim) applied to both tensors.
4. Compute three sigmoid gates from the same normalized input:
   `left_gate`, `right_gate`, `out_gate` (shape [B, N, N, H]).
5. Fuse gates via pointwise multiplication:
   `left *= left_gate`, `right *= right_gate`.
6. The heavy “TriMul” part:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   is a batched matrix multiplication over the sequence dimension.
   - Permute to [B, H, N, N] and reshape to a single batch dimension
     `B*H`.  The operation becomes a set of independent GEMMs
     (N×N)·(N×N) → (N×N) for each hidden channel.
   - A custom Triton kernel (`batched_matmul_kernel`) processes all
     `(B*H)` batches in one launch, tiling the output with
     BLOCK_M×BLOCK_N tiles and looping over K in BLOCK_K steps.
7. Reshape the result back to [B, N, N, H].
8. Apply a second LayerNorm (`to_out_norm`), multiply with `out_gate`,
   and finally a linear projection back to the original channel size C.

The kernel only handles the O(N³) multiplication; all other ops are
performed with PyTorch (still on‑GPU) for simplicity and correctness.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix‑multiply (C = A @ B) for many independent batches
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix dimensions (all = seq_len)
    stride_am, stride_ak, stride_ab,   # A: (batch, M, K) strides
    stride_bk, stride_bn, stride_bb,   # B: (batch, K, N) strides
    stride_cm, stride_cn, stride_cb,   # C: (batch, M, N) strides
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute many independent GEMMs C = A @ B."""
    pid = tl.program_id(0)        # encodes batch * block_m
    pid_n = tl.program_id(1)      # block_n

    # number of tiles along M dimension
    num_blocks_m = tl.cdiv(M, BLOCK_M)

    batch_id = pid // num_blocks_m
    block_m = pid % num_blocks_m
    block_n = pid_n

    # offsets within the matrices
    offs_m = block_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = block_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in blocks
    for k in range(0, K, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # pointer arithmetic for A and B tiles
        a_ptrs = A_ptr + batch_id * stride_ab \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = B_ptr + batch_id * stride_bb \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # load tiles (with OOB masking)
        a = tl.load(a_ptrs,
                    mask=(offs_m[:, None] < M) & (offs_k[None, :] < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & (offs_n[None, :] < N),
                    other=0.0)

        # dot‑product accumulation
        acc += tl.dot(a, b)

    # write back the result tile
    c_ptrs = C_ptr + batch_id * stride_cb \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc,
             mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    TriMul forward (outgoing) implementation.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C]
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)
        - weights      : dict of torch.Tensors (model parameters)
        - config       : dict containing "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # unpack
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5  # LayerNorm epsilon (same as nn.LayerNorm default)

    # ------------------------------------------------------------------
    # load weights (all are on the same device as the input)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm on input
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)     # [B, N, N, H]
    right = F.linear(x, right_proj_weight)   # [B, N, N, H]

    # ------------------------------------------------------------------
    # 3. Apply mask (if any). Mask is broadcast on the channel dim.
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask dtype matches the projected tensors
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # shape [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Compute gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 5. Fuse gates with projections
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6. TriMul: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as many independent GEMMs with Triton.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    batch_hidden = B * hidden_dim

    # permute to (B, H, N, N) – left stays (i,k), right needs (k,j)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # [B, H, N, N] (k, j)

    # collapse the batch*hidden dimensions → (B*H, N, N)
    left_flat = left_perm.view(batch_hidden, N, N)
    right_flat = right_perm.view(batch_hidden, N, N)

    # output buffer
    out_flat = torch.empty_like(left_flat)

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # number of tiles
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N

    # program ID 0 encodes (batch * block_m), so we need batch_hidden * num_blocks_m entries
    grid = (batch_hidden * num_blocks_m, num_blocks_n)

    # launch
    batched_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(1),  # stride_am (row stride)
        left_flat.stride(2),  # stride_ak (col stride)
        left_flat.stride(0),  # stride_ab (batch stride)
        right_flat.stride(1), # stride_bk
        right_flat.stride(2), # stride_bn
        right_flat.stride(0), # stride_bb
        out_flat.stride(1),   # stride_cm
        out_flat.stride(2),   # stride_cn
        out_flat.stride(0),   # stride_cb
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,          # modest warp count; H100 handles this well
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to [B, N, N, H]
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # 8. Output LayerNorm, gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    output = F.linear(out, to_out_weight)  # back to [B, N, N, C]

    return output
```
