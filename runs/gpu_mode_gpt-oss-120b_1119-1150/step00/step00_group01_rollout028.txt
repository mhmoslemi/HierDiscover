    pid_i = tl.program_id(1)  # i tile
    pid_k = tl.program_id(2)  # k tile

    k_start = pid_k * BLOCK_K

def mask_gate_kernel(
    proj_ptr, gate_ptr, mask_ptr, out_ptr,
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_k = tl.program_id(2)

    i = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    k = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    d = tl.arange(0, BLOCK_D)

    mask_i = i < N
    mask_k = k < N
    mask_d = d < H

    # Compute pointers
    # broadcasted mask across d dimension; we will load mask as shape (BLOCK_I, BLOCK_K)
    # Use broadcasting
    proj_ptrs = proj_ptr + pid_b * stride_proj_b + i[:, None] * stride_proj_i + k[None, :] * stride_proj_k
    gate_ptrs = gate_ptr + pid_b * stride_gate_b + i[:, None] * stride_gate_i + k[None, :] * stride_gate_k
    mask_ptrs = mask_ptr + pid_b * stride_mask_b + i[:, None] * stride_mask_i + k[None, :] * stride_mask_k

    # Load base slices (no d)
    proj_base = tl.load(proj_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0) # shape (BLOCK_I, BLOCK_K)
    gate_base = tl.load(gate_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
    mask_val = tl.load(mask_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

    # Compute per hidden dimension: we need to multiply each element with the corresponding hidden slice.
    # We'll iterate over d offset
    for i_d in range(0, BLOCK_D):
        if i_d >= H:
            break
        h_off = i_d
        # pointer for this hidden slice:
        proj_ptrs_h = proj_ptr + pid_b * stride_proj_b + i[:, None] * stride_proj_i + k[None, :] * stride_proj_k + h_off * stride_proj_h
        gate_ptrs_h = gate_ptr + pid_b * stride_gate_b + i[:, None] * stride_gate_i + k[None, :] * stride_gate_k + h_off * stride_gate_h
        out_ptrs_h = out_ptr + pid_b * stride_out_b + i[:, None] * stride_out_i + k[None, :] * stride_out_k + h_off * stride_out_h

        proj_h = tl.load(proj_ptrs_h, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    bs, N, _, _ = input_tensor.shape # but shape is [bs, seq_len, seq_len, dim]; check dims.

    # Ensure contiguous
    x = input_tensor
    # Apply LayerNorm
    x = torch.nn.functional.layer_norm(x, (dim,), weight=weights['norm.weight'], bias=weights['norm.bias'])
    # Linear projections
    left_proj = torch.nn.functional.linear(x, weights['left_proj.weight'])
    right_proj = torch.nn.functional.linear(x, weights['right_proj.weight'])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['left_gate.weight']))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['right_gate.weight']))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights['out_gate.weight']))

    # Apply mask and gates using Triton kernel
    device = x.device
    # if mask is None then use ones
    if mask is None or config.get('nomask', False):
        mask_tensor = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask_tensor = mask
    # Ensure mask shape matches
    mask_tensor = mask_tensor.to(x.dtype)

    # Allocate tensors for left and right after gating
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Determine block sizes (choose to fit H100)
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_D = 8  # hidden slice size
    # compile the kernel
    import triton
    import triton.language as tl

    @triton.jit
    def mask_gate_kernel(proj_ptr, gate_ptr, mask_ptr, out_ptr,
                         B, N, H,
                         stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
                         stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
                         stride_mask_b, stride_mask_i, stride_mask_k,
                         stride_out_b, stride_out_i, stride_out_k, stride_out_h,
                         BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr):
        """Apply element-wise mask and gate to projection."""
        pid_b = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_k = tl.program_id(2)

        i_base = pid_i * BLOCK_I
        k_base = pid_k * BLOCK_K

        i = i_base + tl.arange(0, BLOCK_I)
        k = k_base + tl.arange(0, BLOCK_K)
        d = tl.arange(0, BLOCK_D)

        # range checks
        mask_i = i < N
        mask_k = k < N
        mask_d = d < H

        # broadcast mask_i & mask_k for loads
        active = mask_i[:, None] & mask_k[None, :]

        # Load mask (scalar per i,k) and broadcast across D
        mask_ptrs = mask_ptr + pid_b * stride_mask_b + i[:, None] * stride_mask_i + k[None, :] * stride_mask_k
        mask_val = tl.load(mask_ptrs, mask=active, other=0.0)  # shape (BLOCK_I, BLOCK_K)
        mask_val = mask_val[..., None]  # (BLOCK_I, BLOCK_K, 1)

        # Build pointer tensors for proj, gate, out
        proj_ptrs = proj_ptr + pid_b * stride_proj_b + i[:, None, None] * stride_proj_i + k[None, :, None] * stride_proj_k + d[None, None, :] * stride_proj_h
        gate_ptrs = gate_ptr + pid_b * stride_gate_b + i[:, None, None] * stride_gate_i + k[None, :, None] * stride_gate_k + d[None, None, :] * stride_gate_h
        out_ptrs = out_ptr + pid_b * stride_out_b + i[:, None, None] * stride_out_i + k[None, :, None] * stride_out_k + d[None, None, :] * stride_out_h

        # Load
        proj = tl.load(proj_ptrs, mask=active[:, :, None] & mask_d[None, None, :], other=0.0)
        gate = tl.load(gate_ptrs, mask=active[:, :, None] & mask_d[None, None, :], other=0.0)

        # Compute
        result = proj * gate * mask_val

        # Store
        tl.store(out_ptrs, result, mask=active[:, :, None] & mask_d[None, None, :])

    # Launch kernel for left
    grid = (bs, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_K - 1) // BLOCK_K)
    mask_gate_kernel[grid](
        left_proj,
        left_gate,
        mask_tensor,
        left,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_proj_b=left_proj.stride(0),
        stride_proj_i=left_proj.stride(1),
        stride_proj_k=left_proj.stride(2),
        stride_proj_h=left_proj.stride(3),
        stride_gate_b=left_gate.stride(0),
        stride_gate_i=left_gate.stride(1),
        stride_gate_k=left_gate.stride(2),
        stride_gate_h=left_gate.stride(3),
        stride_mask_b=mask_tensor.stride(0),
        stride_mask_i=mask_tensor.stride(1),
        stride_mask_k=mask_tensor.stride(2),
        stride_out_b=left.stride(0),
        stride_out_i=left.stride(1),
        stride_out_k=left.stride(2),
        stride_out_h=left.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    # Right kernel
    mask_gate_kernel[grid](
        right_proj,
        right_gate,
        mask_tensor,
        right,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_proj_b=right_proj.stride(0),
        stride_proj_i=right_proj.stride(1),
        stride_proj_k=right_proj.stride(2),
        stride_proj_h=right_proj.stride(3),
        stride_gate_b=right_gate.stride(0),
        stride_gate_i=right_gate.stride(1),
        stride_gate_k=right_gate.stride(2),
        stride_gate_h=right_gate.stride(3),
        stride_mask_b=mask_tensor.stride(0),
        stride_mask_i=mask_tensor.stride(1),
        stride_mask_k=mask_tensor.stride(2),
        stride_out_b=right.stride(0),
        stride_out_i=right.stride(1),
        stride_out_k=right.stride(2),
        stride_out_h=right.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Now compute contraction
    # Permute to bring hidden dim in front and reshape to batch*hidden
    B = bs
    H = hidden_dim
    # we may want to ensure contiguous after permute
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # reshape to (B*H, N, N)
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)
    # matmul with transpose on right (k dimension)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)
    # reshape back to (B, N, N, H)
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # apply out_gate
    out_gated = out_norm * out_gate

    # final linear projection
    final_out = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return final_out

"""
custom_kernel - TriMul "outgoing" implementation (AlphaFold3).

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,C]).
2. Compute three linear projections of the normalized tensor:
      left_proj , right_proj  → (B,N,N,H)
      left_gate , right_gate , out_gate → (B,N,N,H)  (followed by sigmoid)
3. Fuse mask (if present), left‑gate and right‑gate with a small Triton kernel:
      left  = left_proj  * left_gate  * mask
      right = right_proj * right_gate * mask
4. Perform the pair‑wise contraction
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batch of H independent N×N matrix multiplications.
   This step is carried out with a highly‑optimized PyTorch batched GEMM
   (`torch.bmm`) on the H100.
5. Apply a second LayerNorm over the hidden dimension, multiply by the
   `out_gate` and finally linearly project back to the original channel size.
   The output has shape [B,N,N,C].

Only the mask‑gate fusion is implemented in Triton; the heavy N³ contraction
uses cuBLAS via `torch.bmm`, giving sub‑millisecond runtimes for the
target configurations.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# --------------------------------------------------------------------------- #
# Triton kernel: fuse mask, gate and projection (element‑wise product)
# --------------------------------------------------------------------------- #
@triton.jit
def _mask_gate_fuse_kernel(
    proj_ptr,          # float32*  left_proj / right_proj
    gate_ptr,          # float32*  left_gate / right_gate
    mask_ptr,          # float32*  mask (B,N,N)
    out_ptr,           # float32*  result tensor (B,N,N,H)
    B, N, H,
    stride_proj_b, stride_proj_i, stride_proj_k, stride_proj_h,
    stride_gate_b, stride_gate_i, stride_gate_k, stride_gate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    stride_out_b, stride_out_i, stride_out_k, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    """Apply element‑wise mask and gate to the projection.

    Each program works on a tile (BLOCK_I × BLOCK_K) of the (i,k) plane
    for a single batch element and processes up to BLOCK_D hidden channels
    at a time (the hidden dimension is iterated in registers).
    """
    pid_b = tl.program_id(0)   # batch index
    pid_i = tl.program_id(1)   # i‑tile
    pid_k = tl.program_id(2)   # k‑tile

    i0 = pid_i * BLOCK_I
    k0 = pid_k * BLOCK_K

    # 1‑D offsets inside the tile
    i = i0 + tl.arange(0, BLOCK_I)          # (BLOCK_I,)
    k = k0 + tl.arange(0, BLOCK_K)          # (BLOCK_K,)
    d = tl.arange(0, BLOCK_D)               # (BLOCK_D,)

    # Boundary masks
    mask_i = i < N
    mask_k = k < N
    mask_d = d < H

    # Broadcast i/k masks to (BLOCK_I, BLOCK_K) matrix
    active_ik = mask_i[:, None] & mask_k[None, :]

    # ------------------------------------------------------------------- #
    # Load mask value (scalar per i,k) and broadcast over hidden channels
    # ------------------------------------------------------------------- #
    mask_ptrs = (mask_ptr
                 + pid_b   * stride_mask_b
                 + i[:, None] * stride_mask_i
                 + k[None, :] * stride_mask_k)
    mask_val = tl.load(mask_ptrs, mask=active_ik, other=0.0)  # (BLOCK_I, BLOCK_K)
    mask_val = mask_val[:, :, None]                         # (BLOCK_I, BLOCK_K, 1)

    # ------------------------------------------------------------------- #
    # Build pointer tensors for proj, gate and output (3‑D block)
    # ------------------------------------------------------------------- #
    # (i, k, d) tiled pointer arithmetic – broadcasting works because strides are scalars
    proj_ptrs = (proj_ptr
                 + pid_b * stride_proj_b
                 + i[:, None, None] * stride_proj_i
                 + k[None, :, None] * stride_proj_k
                 + d[None, None, :] * stride_proj_h)

    gate_ptrs = (gate_ptr
                 + pid_b * stride_gate_b
                 + i[:, None, None] * stride_gate_i
                 + k[None, :, None] * stride_gate_k
                 + d[None, None, :] * stride_gate_h)

    out_ptrs  = (out_ptr
                 + pid_b * stride_out_b
                 + i[:, None, None] * stride_out_i
                 + k[None, :, None] * stride_out_k
                 + d[None, None, :] * stride_out_h)

    # ------------------------------------------------------------------- #
    # Load projection and gate, apply mask & gate, store result
    # ------------------------------------------------------------------- #
    load_mask = active_ik[:, :, None] & mask_d[None, None, :]
    proj = tl.load(proj_ptrs, mask=load_mask, other=0.0)   # (BLOCK_I, BLOCK_K, BLOCK_D)
    gate = tl.load(gate_ptrs, mask=load_mask, other=0.0)   # (BLOCK_I, BLOCK_K, BLOCK_D)

    result = proj * gate * mask_val                       # broadcasted mask

    tl.store(out_ptrs, result, mask=load_mask)


# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim]
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    bs, N, _, _ = input_tensor.shape  # input shape [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 1. Layer‑norm over the last dimension
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------- #
    # 2. Linear projections + gates
    # ------------------------------------------------------------------- #
    left_proj = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden_dim]
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # same shape
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # ------------------------------------------------------------------- #
    # 3. Fuse mask + gates with a Triton kernel (skip if no mask)
    # ------------------------------------------------------------------- #
    if nomask or mask_tensor is None:
        mask = torch.ones((bs, N, N), dtype=x.dtype, device=device)
    else:
        mask = mask_tensor.to(dtype=x.dtype, device=device)

    # Allocate output containers for the fused tensors
    left = torch.empty_like(left_proj)
    right = torch.empty_like(right_proj)

    # Choose tiling parameters (tuned for H100)
    BLOCK_I = 32    # tiles along the i dimension
    BLOCK_K = 32    # tiles along the k dimension
    BLOCK_D = 8     # number of hidden channels processed per tile

    grid = (bs,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_K - 1) // BLOCK_K)

    # Left side
    _mask_gate_fuse_kernel[grid](
        left_proj,
        left_gate,
        mask,
        left,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_proj_b=left_proj.stride(0),
        stride_proj_i=left_proj.stride(1),
        stride_proj_k=left_proj.stride(2),
        stride_proj_h=left_proj.stride(3),
        stride_gate_b=left_gate.stride(0),
        stride_gate_i=left_gate.stride(1),
        stride_gate_k=left_gate.stride(2),
        stride_gate_h=left_gate.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_k=mask.stride(2),
        stride_out_b=left.stride(0),
        stride_out_i=left.stride(1),
        stride_out_k=left.stride(2),
        stride_out_h=left.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # Right side
    _mask_gate_fuse_kernel[grid](
        right_proj,
        right_gate,
        mask,
        right,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_proj_b=right_proj.stride(0),
        stride_proj_i=right_proj.stride(1),
        stride_proj_k=right_proj.stride(2),
        stride_proj_h=right_proj.stride(3),
        stride_gate_b=right_gate.stride(0),
        stride_gate_i=right_gate.stride(1),
        stride_gate_k=right_gate.stride(2),
        stride_gate_h=right_gate.stride(3),
        stride_mask_b=mask.stride(0),
        stride_mask_i=mask.stride(1),
        stride_mask_k=mask.stride(2),
        stride_out_b=right.stride(0),
        stride_out_i=right.stride(1),
        stride_out_k=right.stride(2),
        stride_out_h=right.stride(3),
        BLOCK_I=BLOCK_I,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------- #
    # 4. Pair‑wise contraction (einsum) – performed as batched GEMM
    # ------------------------------------------------------------------- #
    # Rearrange tensors to (B*H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    left_mat = left_perm.view(bs * hidden_dim, N, N)          # (B*H, N, N)
    right_mat = right_perm.view(bs * hidden_dim, N, N)        # (B*H, N, N)

    # Multiply left with the transpose of right over the ‘k’ axis
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5. Second LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    out = out * out_gate                     # broadcast over hidden dim
    output = F.linear(out, weights["to_out.weight"])   # → [B,N,N,dim]

    return output
