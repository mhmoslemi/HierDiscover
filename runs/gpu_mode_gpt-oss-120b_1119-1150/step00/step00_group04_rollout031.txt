    # final linear

@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, D,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch
    pid_i = tl.program_id(0)  # block i
    pid_j = tl.program_id(1)  # block j

    # compute start indices
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # index vectors
    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)   # will be offset within each D block

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Output accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        # generate k indices
        off_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = off_k < N

        # left pointer offsets: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_off = (pid_b * stride_left_b
                    + offs_i[:, None, None] * stride_left_i
                    + off_k[None, :, None] * stride_left_k
                    + offs_d[None, None, :] * stride_left_d)

        # right pointer offsets: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_off = (pid_b * stride_right_b
                     + offs_j[:, None, None] * stride_right_j
                     + off_k[None, :, None] * stride_right_k
                     + offs_d[None, None, :] * stride_right_d)

        # load with mask; combine masks across i/k/d
        mask_left = mask_i[:, None, None] & mask_k[None, :, None]  # shape (BLOCK_I, BLOCK_K, 1)
        mask_left = mask_left & (offs_d[None, None, :] < D)  # broadcast, but D limit constant can be omitted as D known

        mask_right = mask_j[:, None, None] & mask_k[None, :, None]  # shape (BLOCK_J, BLOCK_K, 1)
        mask_right = mask_right & (offs_d[None, None, :] < D)

        left = tl.load(left_ptr + left_off, mask=mask_left, other=0.0) # shape (BLOCK_I, BLOCK_K, BLOCK_D)
        right = tl.load(right_ptr + right_off, mask=mask_right, other=0.0) # shape (BLOCK_J, BLOCK_K, BLOCK_D)

        # compute contribution: expand dims to broadcast across i/j
        # left_exp: (BLOCK_I, 1, BLOCK_K, BLOCK_D)
        # right_exp: (1, BLOCK_J, BLOCK_K, BLOCK_D)
        left_exp = left[:, None, :, :]  # broadcast over j
        right_exp = right[None, :, :, :]  # broadcast over i

        # product
        prod = left_exp * right_exp  # shape (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_D)

        # sum over k
        acc += tl.sum(prod, axis=2)  # reduce over K, result (BLOCK_I, BLOCK_J, BLOCK_D)

    # write out
    out_off = (pid_b * stride_out_b
               + offs_i[:, None, None] * stride_out_i
               + offs_j[None, :, None] * stride_out_j
               + offs_d[None, None, :] * stride_out_d)

    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & (offs_d[None, None, :] < D)
    tl.store(out_ptr + out_off, acc, mask=mask_out)

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask
    mask_un = mask.unsqueeze(-1)  # shape [B,N,N,1]
    left = left_proj * mask_un
    right = right_proj * mask_un

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute triangular multiplication via Triton kernel
    out = triton_tri_mul(left, right)

    # Final normalization and gating
    to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) implementation for AlphaFold3 style pairwise updates.

The heavy part of the forward pass is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which we compute with a custom Triton kernel.  The kernel
tiles the i‑, j‑ and d‑dimensions (i and j are run‑time dimensions,
d is the hidden_dim) and reduces over the k‑dimension in blocks.
All other operations (LayerNorm, Linear projections, gating, final
normalisation and linear) are performed with torch.nn.functional for
convenience.  The Triton kernel works in fp32 and is written to handle
any batch size (B∈{1,2}), sequence length N up to 1024 and hidden_dim
that is a multiple of BLOCK_D (we use BLOCK_D=8 which covers the
tested hidden_dim=128).  The kernel is launched with a 3‑D grid
(B, ceil(N/BLOCK_I), ceil(N/BLOCK_J)).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# -------------------------------------------------------------------------
# Triton kernel: computes the contraction Σ_k left[...,i,k,d] * right[...,j,k,d]
# -------------------------------------------------------------------------
@triton.jit
def _tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, D,
    # output strides
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # left strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # right strides
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)               # batch index
    pid_i = tl.program_id(0)               # i‑tile index
    pid_j = tl.program_id(1)               # j‑tile index

    # ---------------------------------------------------------------------
    # 1) Compute start offsets for the current tile
    # ---------------------------------------------------------------------
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)        # [BLOCK_I]
    offs_j = j_start + tl.arange(0, BLOCK_J)        # [BLOCK_J]

    mask_i = offs_i < N
    mask_j = offs_j < N

    # D may not be a multiple of BLOCK_D – we mask the tail.
    num_d_blocks = tl.cdiv(D, BLOCK_D)

    # ---------------------------------------------------------------------
    # 2) Loop over the hidden dimension (D) in small sub‑tiles
    # ---------------------------------------------------------------------
    for d_block in range(0, num_d_blocks):
        d_start = d_block * BLOCK_D
        offs_d = d_start + tl.arange(0, BLOCK_D)           # [BLOCK_D]
        mask_d = offs_d < D

        # accumulator for this (i,j,d) sub‑tile
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

        # -----------------------------------------------------------------
        # 3) Reduce over k in blocks of size BLOCK_K
        # -----------------------------------------------------------------
        for k_block in range(0, tl.cdiv(N, BLOCK_K)):
            k_start = k_block * BLOCK_K
            offs_k = k_start + tl.arange(0, BLOCK_K)        # [BLOCK_K]
            mask_k = offs_k < N

            # ----- left tile: shape (BLOCK_I, BLOCK_K, BLOCK_D) -----
            left_offset = (
                pid_b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_d[None, None, :] * stride_left_d
            )
            left_mask = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            left = tl.load(left_ptr + left_offset, mask=left_mask, other=0.0)   # fp32

            # ----- right tile: shape (BLOCK_J, BLOCK_K, BLOCK_D) -----
            right_offset = (
                pid_b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_d[None, None, :] * stride_right_d
            )
            right_mask = mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            right = tl.load(right_ptr + right_offset, mask=right_mask, other=0.0)   # fp32

            # ----- broadcast and accumulate across k -----
            # left : (BLOCK_I, 1, BLOCK_K, BLOCK_D)
            # right: (1, BLOCK_J, BLOCK_K, BLOCK_D)
            left_b  = left[:, None, :, :]      # (I,1,K,D)
            right_b = right[None, :, :, :]     # (1,J,K,D)
            # element‑wise product then sum over K
            acc += tl.sum(left_b * right_b, axis=2)   # (I,J,D)

        # -----------------------------------------------------------------
        # 4) Write the computed sub‑tile back to the output tensor
        # -----------------------------------------------------------------
        out_offset = (
            pid_b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_d[None, None, :] * stride_out_d
        )
        store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
        tl.store(out_ptr + out_offset, acc, mask=store_mask)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based implementation of:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    left/right shape: [B, N, N, D]   (contiguous, float32)
    Returns tensor of same shape and dtype.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, D = left.shape

    out = torch.empty_like(left)

    # -----------------------------------------------------------------
    # Tiling configuration (tuned for H100; fits into register budget)
    # -----------------------------------------------------------------
    BLOCK_I = 8
    BLOCK_J = 8
    BLOCK_K = 32
    BLOCK_D = 8                     # D must be a multiple of BLOCK_D (128 works)

    # Compute launch grid
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,     # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,     # j‑tiles
        B                                 # batch dimension
    )

    # Strides (in elements, not bytes)
    stride_out = out.stride()
    stride_left = left.stride()
    stride_right = right.stride()

    _tri_mul_kernel[grid](
        out,
        left,
        right,
        B, N, D,
        # output strides
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        # left strides
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        # right strides
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,                     # enough warps to keep the SM busy
    )
    return out


# -------------------------------------------------------------------------
# Public entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor : [B, N, N, dim] (float32)
            - mask         : [B, N, N] (bool or 0/1 float)
            - weights      : dict of model parameters (see below)
            - config       : dict with keys “dim” and “hidden_dim”

    Returns:
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Mask handling – unsqueeze to broadcast over hidden_dim
    # -----------------------------------------------------------------
    if mask is None:
        mask = torch.ones(
            (input_tensor.shape[0], input_tensor.shape[1], input_tensor.shape[2]),
            dtype=input_tensor.dtype,
            device=input_tensor.device,
        )
    mask = mask.to(input_tensor.dtype)
    mask = mask.unsqueeze(-1)          # shape [B,N,N,1]
    left = left * mask
    right = right * mask

    # -----------------------------------------------------------------
    # 4) Gating (sigmoid on linear transforms of the normalized input)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core contraction via Triton
    # -----------------------------------------------------------------
    out = _tri_mul_triton(left, right)                # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # 6) Final LayerNorm, gating and linear projection back to dim
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])     # back to [B,N,N,dim]

    return out
