def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M : tl.constexpr, BLOCK_N : tl.constexpr, BLOCK_K : tl.constexpr, BLOCK_D : tl.constexpr
):
    pid = tl.program_id(0)
    batch_id = tl.program_id(1)
    d_block_id = tl.program_id(2)

    num_blocks_n = tl.cdiv(N, BLOCK_N)
    i_block = pid // num_blocks_n
    j_block = pid % num_blocks_n

    # Offsets
    i = i_block * BLOCK_M + tl.arange(0, BLOCK_M)
    j = j_block * BLOCK_N + tl.arange(0, BLOCK_N)
    d = d_block_id * BLOCK_D + tl.arange(0, BLOCK_D)
    
    # Masks
    mask_i = i < N
    mask_j = j < N
    mask_d = d < D

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # Loop over k
    for k_start in range(0, N, BLOCK_K):
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Compute pointers, using broadcasting
        # left: shape (BLOCK_M, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + batch_id * stride_left_b \
            + i[:, None, None] * stride_left_i \
            + k[None, :, None] * stride_left_k \
            + d[None, None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # right: we want shape (BLOCK_K, BLOCK_N, BLOCK_D)
        right_ptrs = right_ptr + batch_id * stride_right_b \
            + k[:, None, None] * stride_right_k \
            + j[None, :, None] * stride_right_j \
            + d[None, None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        # Compute dot product across k dimension for each d. Since tl.dot works on 2D matrices, we'll compute for each d.
        # Use a small loop over d to avoid excessive register use.
        for di in range(0, BLOCK_D):
            # If d out-of-range, we could skip
            # Use mask_d[di] to condition
            # Extract slice
            left_slice = left[:, :, di]   # (BLOCK_M, BLOCK_K)
            right_slice = right[:, :, di] # (BLOCK_K, BLOCK_N)
            acc[:, :, di] += tl.dot(left_slice, right_slice)
    
    # Store result
    out_ptrs = out_ptr + batch_id * stride_out_b \
        + i[:, None, None] * stride_out_i \
        + j[None, :, None] * stride_out_j \
        + d[None, None, :] * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :])

"""
TriMul (outgoing) forward implementation with Triton fusion.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Project the normalized tensor to a hidden dimension `H` with four linear layers:
   - left_proj, right_proj (no bias)
   - left_gate, right_gate, out_gate   (followed by sigmoid)
3. Apply the optional pair mask (shape [B,N,N]) and the element‑wise gates:
       left  = left_proj * mask * left_gate
       right = right_proj * mask * right_gate
4. Core TriMul operation:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   The sum‑over‑k is a batched matrix multiplication per hidden channel.
   This step is performed by a custom Triton kernel that tiles the (i,j) plane
   and the hidden dimension `H`.  The kernel loads blocks of `left` and `right`
   in FP16, performs a dot‑product over the K‑dimension (the sequence axis),
   accumulates in FP32 and writes the result to a FP32 output tensor.
5. Layer‑norm the intermediate result over the hidden dimension, multiply by
   `out_gate`, and project back to the original channel dimension with a final
   linear layer.
6. Return the tensor of shape [B, N, N, C].

The kernel works for any batch size (B≤2), sequence length N∈{128,…,1024},
input dimension C (named `dim` in the config) and hidden dimension H
(`hidden_dim`).  It also properly respects the `nomask` flag.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # problem size
    N, H,
    # strides for left  [b, i, k, h]
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right [b, j, k, h] (j is the second dimension of the tensor)
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    # strides for output [b, i, j, h]
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].

    The kernel tiles the (i, j) plane with BLOCK_M × BLOCK_N and the hidden
    dimension with BLOCK_H.  K is tiled with BLOCK_K.  Accumulation is done in
    FP32 regardless of the input precision.
    """
    pid = tl.program_id(0)          # combined i‑tile and j‑tile index
    batch = tl.program_id(1)        # batch dimension (B ≤ 2)
    h_block = tl.program_id(2)      # hidden‑dimension block index

    # number of j‑tiles
    num_tiles_j = tl.cdiv(N, BLOCK_N)

    # derive i‑tile and j‑tile coordinates
    i_tile = pid // num_tiles_j
    j_tile = pid %  num_tiles_j

    # start offsets for the current tile
    i_start = i_tile * BLOCK_M
    j_start = j_tile * BLOCK_N
    h_start = h_block * BLOCK_H

    # range vectors (may contain out‑of‑bounds entries which we later mask)
    offs_i = i_start + tl.arange(0, BLOCK_M)
    offs_j = j_start + tl.arange(0, BLOCK_N)
    offs_h = h_start + tl.arange(0, BLOCK_H)

    # bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator: (BLOCK_M, BLOCK_N, BLOCK_H) in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the K dimension in BLOCK_K sized chunks
    # ----------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        # range vector for the K dimension
        offs_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load a block of `left`  → shape (BLOCK_M, BLOCK_K, BLOCK_H)
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + batch * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + offs_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # Load a block of `right` transposed in the K dimension:
        #   we want shape (BLOCK_K, BLOCK_N, BLOCK_H) i.e. (k, j, h)
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + batch * stride_right_b
            + offs_k[:, None, None] * stride_right_k
            + offs_j[None, :, None] * stride_right_j
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # For each hidden‑channel slice perform a matmul: (M,K) × (K,N)
        # ------------------------------------------------------------------
        for h in range(0, BLOCK_H):
            # skip completely masked channel slices early
            if not mask_h[h]:
                continue
            left_slice = left[:, :, h]   # (BLOCK_M, BLOCK_K)
            right_slice = right[:, :, h] # (BLOCK_K, BLOCK_N)
            acc[:, :, h] += tl.dot(left_slice, right_slice)

    # ----------------------------------------------------------------------
    # Store the accumulated block back to memory (apply final bounds mask)
    # ----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + batch * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Entry point for the TriMul outgoing kernel.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C] (float32)
        - mask         : torch.Tensor of shape [B, N, N] (bool/float)
        - weights      : dict of all model parameters (torch.Tensor)
        - config       : dict containing at least `dim`, `hidden_dim`,
                         optional `nomask` flag

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Model weights (no bias on the projection layers)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension C
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2) Linear projections (float32)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating + optional mask
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    if not nomask:
        # mask: [B, N, N] → broadcast to hidden dimension
        m = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left_proj * m * left_gate
        right = right_proj * m * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Core TriMul (FP16 for bandwidth, accumulate in FP32)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    B, N, _, _ = left_fp16.shape
    out_fp32 = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=left_fp16.device)

    # Block sizes – tuned for H100 Tensor Cores
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64
    BLOCK_H = 8   # must divide hidden_dim (or be padded by mask)

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_h = (hidden_dim + BLOCK_H - 1) // BLOCK_H
    grid = (grid_i * grid_j, B, grid_h)

    trimul_einsum_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        N,
        hidden_dim,
        # left strides
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
        # right strides (j‑dim is the second axis)
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
        # output strides
        out_fp32.stride(0),
        out_fp32.stride(1),
        out_fp32.stride(2),
        out_fp32.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 5) Post‑processing: layer‑norm over hidden_dim, gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_fp32, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # back to dim

    return out
