    import triton.language as tl

"""
TriMul (outgoing) forward pass with a Triton kernel for the N³ contraction.

The algorithm follows the PyTorch reference:

1. Layer‑normalize the 4‑D input (shape [B, N, N, C]).
2. Linear projections to a hidden dimension H (no bias):
      left = x·W_left,  right = x·W_right
3. Optional pair‑wise mask (broadcast over H) and element‑wise gates
      left  = left * mask * sigmoid(x·W_left_gate)
      right = right* mask * sigmoid(x·W_right_gate)
4. Core operation (the heavy O(N³) part):
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is implemented as a fused Triton kernel that treats each
   (batch, hidden) pair as an independent matrix‑multiply
   (N×N)·(N×N)ᵀ.  The kernel tiles the i‑ and j‑axes (64×64) and
   iterates over k in 64‑element blocks, using FP16 loads and Tensor‑Core
   dot products while accumulating in FP32.
5. Layer‑normalize the intermediate tensor over the hidden dimension,
   apply the output gate (sigmoid(x·W_out_gate)), and project back to
   the original channel size C with a final linear layer.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # ptr to left tensor (fp16)
    right_ptr,         # ptr to right tensor (fp16)
    out_ptr,           # ptr to output tensor (fp32)
    B, N, H,           # batch, sequence length, hidden dim
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_k, stride_right_j, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Triton implementation of
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left/right have shape [B,N,N,H] and out has the same shape.
    The kernel launches a 3‑D grid:
        pid0 → i‑tile, pid1 → j‑tile, pid2 → (batch, hidden) pair.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_hd = tl.program_id(2)                # combines batch and hidden index

    b = pid_hd // H                           # batch index
    h = pid_hd % H                            # hidden‑channel index

    # i‑ and j‑coordinates of the tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the K dimension (the summed axis)
    for kk in range(0, tl.cdiv(N, BLOCK_K)):
        k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ---------- load left tile ----------
        # shape (BLOCK_M, BLOCK_K)  -> left[b, i, k, h]
        left_offset = (
            b * stride_left_b +
            i[:, None] * stride_left_i +
            k[None, :] * stride_left_k +
            h * stride_left_d
        )
        left = tl.load(left_ptr + left_offset,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0)               # fp16 load

        # ---------- load right tile (transposed) ----------
        # we need right[b, j, k, h]ᵀ  → shape (BLOCK_K, BLOCK_N)
        right_offset = (
            b * stride_right_b +
            k[:, None] * stride_right_k +   # k as first dimension
            j[None, :] * stride_right_j +   # j as second dimension
            h * stride_right_d
        )
        right = tl.load(right_ptr + right_offset,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0)              # fp16 load

        # FP16 dot product, accumulate in FP32
        acc += tl.dot(left, right).to(tl.float32)

    # ---------- write the result ----------
    out_offset = (
        b * stride_out_b +
        i[:, None] * stride_out_i +
        j[None, :] * stride_out_j +
        h * stride_out_d
    )
    tl.store(out_ptr + out_offset,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.
    Arguments:
        data = (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] on the same device as the input.
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                # input / output channel size C
    hidden_dim = config["hidden_dim"]  # projection size H

    # ensure tensors are on CUDA (the tests guarantee this)
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the channel dimension
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,                # default eps of torch.nn.LayerNorm
    )   # shape: [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (dim → hidden_dim) – no bias
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # [B,N,N,H]  fp32
    right = F.linear(x, weights["right_proj.weight"])    # [B,N,N,H]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask is [B,N,N]; broadcast to [B,N,N,1] and to hidden_dim later
        mask_exp = mask.unsqueeze(-1).to(x.dtype)       # float32
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"])) # [B,N,N,H]
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,H]

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 for the heavy contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6. Allocate output buffer for the contraction: float32
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    out = torch.empty(B, N, N, hidden_dim,
                      dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # 7. Compute the O(N³) sum with Triton
    # ------------------------------------------------------------------
    # Tile sizes – chosen to give good occupancy on an H100.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides (in number of elements, not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    trimul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,
        stride_r_b,
        stride_r_k,          # stride for k dimension in right tensor
        stride_r_i,          # stride for j dimension in right tensor (original dim 1)
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate                               # broadcast over hidden_dim
    out = F.linear(out, weights["to_out.weight"])      # back to [B,N,N,dim]

    return out
```
