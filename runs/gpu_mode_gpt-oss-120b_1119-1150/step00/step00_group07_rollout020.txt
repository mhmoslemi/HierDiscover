"""
TriMul (outgoing) – Triton implementation
========================================

This kernel computes the core “triangle multiplicative update” used in
AlphaFold‑3 (outgoing version):

    out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]

where `left` and `right` are the gated, masked projections of the input
tensor `x ∈ ℝ^{B×N×N×C}` and `h` indexes the hidden dimension
(`hidden_dim`).  The implementation fuses the O(N³) contraction for each
(hidden, batch) pair using a Triton matmul‑like kernel.  All other
operations (LayerNorm, Linear projections, gates, final LayerNorm and
Linear) are performed with regular PyTorch kernels.

The kernel works on half‑precision inputs (to reduce memory traffic) and
accumulates in FP32 for numerical stability.  The final result is cast
back to FP32 for the remaining post‑processing steps.
"""

import torch
import triton
import triton.language as tl

# ---------------------------------------------------------------------------
# Triton kernel: one program per (batch, hidden) pair
# ---------------------------------------------------------------------------
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,            # pointers
    B, N,                                   # runtime sizes
    stride_left_b, stride_left_i, stride_left_k, stride_left_c,
    stride_right_b, stride_right_j, stride_right_k, stride_right_c,
    stride_out_b, stride_out_i, stride_out_j, stride_out_c,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    HIDDEN_DIM: tl.constexpr,
):
    """
    Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]

    Grid
    ----
    (program_id(0)) = batch * hidden_dim
    (program_id(1)) = i‑tile
    (program_id(2)) = j‑tile
    """
    # -----------------------------------------------------------------------
    # Decode program ids → batch index b and hidden channel h
    # -----------------------------------------------------------------------
    pid_bc = tl.program_id(0)                      # 0 … B*HIDDEN_DIM-1
    b = pid_bc // HIDDEN_DIM
    h = pid_bc % HIDDEN_DIM

    # -----------------------------------------------------------------------
    # Tile offsets for the (i,j) output matrix
    # -----------------------------------------------------------------------
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_offset = i_base + tl.arange(0, BLOCK_I)
    j_offset = j_base + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # -----------------------------------------------------------------------
    # Accumulator (FP32)
    # -----------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # -----------------------------------------------------------------------
    # Loop over the reduction dimension k
    # -----------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k_offset = k + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # -------------------------------------------------------------------
        # Load left[b,i,k,h]  → shape (BLOCK_I, BLOCK_K)
        # -------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_k
            + h * stride_left_c
        )
        mask_l = i_mask[:, None] & k_mask[None, :]
        left = tl.load(left_ptrs, mask=mask_l, other=tl.float16(0)).to(tl.float32)

        # -------------------------------------------------------------------
        # Load right[b,j,k,h] → shape (BLOCK_J, BLOCK_K)
        # -------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + j_offset[:, None] * stride_right_j
            + k_offset[None, :] * stride_right_k
            + h * stride_right_c
        )
        mask_r = j_mask[:, None] & k_mask[None, :]
        right = tl.load(right_ptrs, mask=mask_r, other=tl.float16(0)).to(tl.float32)

        # -------------------------------------------------------------------
        # Compute partial matmul: Σₖ left[i,k] * right[j,k]
        # (right is transposed inside tl.dot)
        # -------------------------------------------------------------------
        acc += tl.dot(left, right, trans_b=True)   # (BLOCK_I, BLOCK_J)

    # -----------------------------------------------------------------------
    # Store result (cast back to FP16 because output tensor is half‑precision)
    # -----------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + h * stride_out_c
    )
    mask_o = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_o)


# ---------------------------------------------------------------------------
# Entry point called by the evaluator
# ---------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Arguments
    ---------
    data : tuple(input, mask, weights, config)
        input   – Tensor[B, N, N, C]  (float32, on CUDA)
        mask    – Tensor[B, N, N] or None (float32/bool)
        weights – dict with all Linear/LayerNorm parameters
        config  – dict containing at least ``dim`` and ``hidden_dim``
    Returns
    -------
    out : Tensor[B, N, N, C] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # -----------------------------------------------------------------------
    # 1) LayerNorm on the input
    # -----------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # -----------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dim)
    # -----------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype=input_tensor.dtype, device=device).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------------
    # 4) Gates (sigmoid(Linear(x_norm)))
    # -----------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x_norm, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------------
    # 5) Cast to FP16 for the heavy O(N³) contraction
    # -----------------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()
    B, N, _, _ = left_fp16.shape

    # Output buffer (half‑precision)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16, device=device)

    # -----------------------------------------------------------------------
    # 6) Build strides for Triton
    # -----------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_c = left_fp16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_c = right_fp16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_c = out_fp16.stride()

    # -----------------------------------------------------------------------
    # 7) Launch kernel
    # -----------------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        B * hidden_dim,               # program_id(0) → batch × hidden
        triton.cdiv(N, BLOCK_I),      # program_id(1) → i‑tiles
        triton.cdiv(N, BLOCK_J),      # program_id(2) → j‑tiles
    )

    tri_mul_outgoing_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_c,
        stride_r_b,
        stride_r_i,          # right_j stride == right_i stride
        stride_r_k,
        stride_r_c,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_c,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        HIDDEN_DIM=hidden_dim,
    )

    # Back to FP32 for the remaining post‑processing
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------------
    # 8) Post‑processing: LayerNorm, gate, final Linear
    # -----------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
