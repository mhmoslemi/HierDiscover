def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # Normalization
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Mask if provided (if mask not None & not empty)
    if mask is not None:
        mask = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask
        right = right * mask
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate

    # Bring to shape for kernel: (B,N,N,hidden_dim) -> (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    # Permute and reshape
    left_flat = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # compute strides for kernel
    stride_lb, stride_li, stride_lk = left_flat.stride()
    stride_rb, stride_ri, stride_rk = right_flat.stride()
    stride_ob, stride_oi, stride_ok = out_flat.stride()

    # kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = ( (N + BLOCK_M - 1) // BLOCK_M,
             (N + BLOCK_N - 1) // BLOCK_N,
             B * hidden_dim )
    tri_mul_forward_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B, N, hidden_dim,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_ri, stride_rk,
        stride_ob, stride_oi, stride_ok,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K)

    # reshape output back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1) # shape B,N,N,H

    # final norm, gate multiplication, linear projection
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)
    out_gated = out_norm * out_gate
    out_final = torch.nn.functional.linear(out_gated, weights["to_out.weight"])
    return out_final

"""
TriMul (outgoing) forward pass implemented with a custom Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,C]).
2. Linear projections `left` / `right` (dim→hidden_dim) and their sigmoid gates.
   Optionally apply a binary mask.
3. Fuse the projection and gate by element‑wise multiplication.
4. Compute the core multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batched matrix multiplication across the sequence dimension.
   We reshape the tensors to (B·hidden_dim, N, N) and run a Triton kernel
   that performs tiled GEMM for each (batch·channel) slice.
5. Reshape result back to [B,N,N,hidden_dim].
6. Apply a second LayerNorm, the output gate, and a final linear projection
   back to the original channel dimension `dim`.

The heavy `O(N³)` work (step 4) is performed on‑GPU in the Triton kernel,
while the surrounding linear / norm / sigmoid operations use PyTorch
functional calls (they are already highly optimized). Mixed‑precision is
supported – the kernel works in float32 but can be changed to float16
if desired.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_forward_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # meta‑parameters
    B, N, H,
    # strides (in elements)
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_ok,
    # block sizes (compile‑time constants)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute, for each batch‑channel slice (b_h = b*H + h),

        out[b_h, i, j] = Σ_k left[b_h, i, k] * right[b_h, j, k]

    where left/right have shape (B*H, N, N) and the sum is over the
    shared dimension `k`.  The kernel tiles the i‑, j‑ and k‑axes
    (M, N, K) for efficient use of shared memory.
    """
    pid_bh = tl.program_id(2)          # combined batch * hidden index
    pid_i  = tl.program_id(0)          # tile over output rows   (i dimension)
    pid_j  = tl.program_id(1)          # tile over output cols   (j dimension)

    # offsets within the N‑dimensional axes
    i_off = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_off = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    i_mask = i_off < N
    j_mask = j_off < N

    # ------------------------------------------------------------------
    #   Base pointers for the current (batch, hidden) slice
    # ------------------------------------------------------------------
    # All three tensors share the same batch‑stride (N²) after reshaping
    batch_offset = pid_bh * stride_lb
    left_base  = left_ptr  + batch_offset
    right_base = right_ptr + batch_offset
    out_base   = out_ptr   + batch_offset

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    #   Loop over the k‑dimension in BLOCK_K chunks
    # ------------------------------------------------------------------
    for k_start in range(0, N, BLOCK_K):
        k_off = k_start + tl.arange(0, BLOCK_K)
        k_mask = k_off < N

        # left block  (M x K)  – indexed as (i, k)
        left_ptrs = left_base + i_off[:, None] * stride_li + k_off[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0,
                       dtype=tl.float32)

        # right block (K x N) – we need the transpose (k, j) to match TL.dot
        right_ptrs = right_base + k_off[:, None] * stride_ri + j_off[None, :] * stride_rk
        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0,
                        dtype=tl.float32)

        # matrix multiplication for this k‑tile
        acc += tl.dot(left, right)   # (M,K) @ (K,N) → (M,N)

    # ------------------------------------------------------------------
    #   Write the result back
    # ------------------------------------------------------------------
    out_ptrs = out_base + i_off[:, None] * stride_oi + j_off[None, :] * stride_ok
    tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    eps = 1e-5  # LayerNorm epsilon (same as PyTorch default)

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left  = F.linear(x, left_proj_weight)   # (dim→hidden_dim)
    right = F.linear(x, right_proj_weight)  # (dim→hidden_dim)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(dtype).unsqueeze(-1)           # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute sigmoid gates and apply them
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core multiplicative update (batched GEMM) via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # reshape to (B*hidden_dim, N, N) for the kernel
    #   left : [B,N,N,hidden_dim] → [B, hidden_dim, N, N] → (B*hidden_dim,N,N)
    left_flat  = left.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # allocate output tensor
    out_flat = torch.empty_like(left_flat)

    # strides (in elements) for the flattened tensors
    stride_lb, stride_li, stride_lk = left_flat.stride()   # (batch, i, k)
    stride_rb, stride_ri, stride_rk = right_flat.stride()
    stride_ob, stride_oi, stride_ok = out_flat.stride()

    # tiling parameters – chosen to give good occupancy on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim,                        # combined batch×hidden
    )

    tri_mul_forward_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_ob,
        stride_oi,
        stride_ok,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        # force 32‑bit arithmetic (default)
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, output gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # (hidden_dim → dim), no bias

    return out
```
