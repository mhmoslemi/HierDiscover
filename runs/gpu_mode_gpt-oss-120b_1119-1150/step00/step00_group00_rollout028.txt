def custom_kernel(data):
    import torch, triton, triton.language as tl

    input_tensor, mask, weights, config = data
    batch_size, seq_len, _, dim = input_tensor.shape # dim = input last dimension

    hidden_dim = config['hidden_dim']
    dim = config['dim'] # For output dim?

    # Load weight tensors:
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # layernorm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape: B,N,N,hidden_dim
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gating linear:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # Apply mask and gates using Triton kernel
    B, N, _, _ = left.shape
    H = hidden_dim

    # ensure tensors contiguous
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    if mask is not None:
        mask = mask.contiguous()
    # allocate gated tensors
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)

    # Strides
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h = left_gate.stride()
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h = right_gate.stride()
    if mask is not None:
        stride_mask_b, stride_mask_i, stride_mask_k = mask.stride()
    else:
        stride_mask_b = stride_mask_i = stride_mask_k = 0

    # compile kernel launch
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_k = (N + BLOCK_K - 1) // BLOCK_K
    num_hblocks = (H + BLOCK_H - 1) // BLOCK_H
    grid_z = B * num_hblocks

    has_mask = (mask is not None) and (not config.get('nomask', True))

    @triton.jit

        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr,

        stride_right_b, stride_right_i, stride_right_k, stride_right_h,

        else:

            right_val = right_val * right_gate_val

        right_out_ptrs = right_out_ptr + b * stride_right_b + i_offset[:, None, None] * stride_right_i + k_offset[None, :, None] * stride_right_k + d_offset[None, None, :] * stride_right_h

        tl.store(right_out_ptrs, right_val, mask=mask_ikd)

        right_ptr=right.data_ptr(),

        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask.data_ptr() if has_mask else 0,
        left_out_ptr=left_gated.data_ptr(),
        right_out_ptr=right_gated.data_ptr(),
        stride_left_b=stride_left_b, stride_left_i=stride_left_i, stride_left_k=stride_left_k, stride_left_h=stride_left_h,
        stride_right_b=stride_right_b, stride_right_i=stride_right_i, stride_right_k=stride_right_k, stride_right_h=stride_right_h,
        stride_lgate_b=stride_lgate_b, stride_lgate_i=stride_lgate_i, stride_lgate_k=stride_lgate_k, stride_lgate_h=stride_lgate_h,
        stride_rgate_b=stride_rgate_b, stride_rgate_i=stride_rgate_i, stride_rgate_k=stride_rgate_k, stride_rgate_h=stride_rgate_h,
        stride_mask_b=stride_mask_b, stride_mask_i=stride_mask_i, stride_mask_k=stride_mask_k,
        B=B, N=N, H=H,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        HAS_MASK=1 if has_mask else 0
    )

    # Now compute output via batched matmul
    # Cast to fp16 for efficient matmul
    left_h = left_gated.to(torch.float16)
    right_h = right_gated.to(torch.float16)

    # reshape for batched matmul
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)  # shape (B*H, N, N)
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)  # shape (B*H, N, N)

    # Compute batched matmul: left @ right^T
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)
    # reshape back
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # (B, N, N, H)
    out = out.to(torch.float32)  # back to float32

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # apply out_gate (shape B,N,N,H)
    out_norm = out_norm * out_gate

    # final projection
    out_final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape (B,N,N,dim)

    return out_final

"""
TriMul (outgoing) – fast Triton‑fused implementation

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape [B,N,N,D]).
2. Project `x` to a hidden dimension `H` with two bias‑free linear layers:
   * left  = x @ W_left   (shape [B,N,N,H])
   * right = x @ W_right  (shape [B,N,N,H])
3. Compute three gate tensors (sigmoid(Linear(x))) of shape [B,N,N,H]:
   * left_gate , right_gate , out_gate
4. A custom Triton kernel fuses the element‑wise mask (if present) and the
   gates onto `left` and `right`:
   left_out  = left  * left_gate  * mask
   right_out = right * right_gate * mask
   The kernel works on 3‑D tiles (i,k,H) and processes the hidden dimension
   in small vector chunks (`BLOCK_H`).  It also vectorizes across the `i`
   and `k` axes.
5. Convert the gated tensors to FP16 and reshape them to (B*H,N,N) so we
   can perform a batched matrix multiplication:
       out_fp16 = (left_out_flat) @ (right_out_flat).T
   This is exactly the Einstein sum
       out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
   and uses cuBLAS (via torch.bmm) which maps to Tensor‑cores on H100.
6. Cast back to FP32, apply a second LayerNorm (over the hidden dim),
   multiply by `out_gate`, and finally linearly project back to the original
   dimension `D`.
7. Return the result tensor of shape [B,N,N,D].

Only the mask‑and‑gate fusion is implemented in Triton; all large matrix
multiplications are delegated to cuBLAS for maximal throughput.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def _mask_gate_fusion_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    # strides for the 4‑D tensors (B,N,N,H)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_k, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_k, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_k,
    B, N, H,
    BLOCK_I: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    HAS_MASK: tl.constexpr
):
    # 3‑D launch grid: i‑tile, k‑tile, (batch * hidden‑tile)
    pid_i = tl.program_id(0)
    pid_k = tl.program_id(1)
    pid_z = tl.program_id(2)

    # Derive batch index and hidden‑tile index from pid_z
    num_hblocks = tl.cdiv(H, BLOCK_H)
    b = pid_z // num_hblocks
    h_block = pid_z % num_hblocks

    # Offsets for the three spatial dimensions
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    k_off = pid_k * BLOCK_K + tl.arange(0, BLOCK_K)
    h_off = h_block * BLOCK_H + tl.arange(0, BLOCK_H)

    i_mask = i_off < N
    k_mask = k_off < N
    h_mask = h_off < H
    # combined mask for (i,k,h) loading/storing
    ikh_mask = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]

    # ------------------------------------------------------------------
    # Load left/right tensors and their gates
    # ------------------------------------------------------------------
    left_ptrs = (left_ptr
                 + b * stride_left_b
                 + i_off[:, None, None] * stride_left_i
                 + k_off[None, :, None] * stride_left_k
                 + h_off[None, None, :] * stride_left_h)
    right_ptrs = (right_ptr
                  + b * stride_right_b
                  + i_off[:, None, None] * stride_right_i
                  + k_off[None, :, None] * stride_right_k
                  + h_off[None, None, :] * stride_right_h)

    lgate_ptrs = (left_gate_ptr
                  + b * stride_lgate_b
                  + i_off[:, None, None] * stride_lgate_i
                  + k_off[None, :, None] * stride_lgate_k
                  + h_off[None, None, :] * stride_lgate_h)
    rgate_ptrs = (right_gate_ptr
                  + b * stride_rgate_b
                  + i_off[:, None, None] * stride_rgate_i
                  + k_off[None, :, None] * stride_rgate_k
                  + h_off[None, None, :] * stride_rgate_h)

    left   = tl.load(left_ptrs,   mask=ikh_mask, other=0.0)
    right  = tl.load(right_ptrs,  mask=ikh_mask, other=0.0)
    lgate  = tl.load(lgate_ptrs,  mask=ikh_mask, other=0.0)
    rgate  = tl.load(rgate_ptrs,  mask=ikh_mask, other=0.0)

    if HAS_MASK:
        # mask has shape (B,N,N) – broadcast over H
        mask_ptrs = (mask_ptr
                     + b * stride_mask_b
                     + i_off[:, None] * stride_mask_i
                     + k_off[None, :] * stride_mask_k)
        mask_tile = tl.load(mask_ptrs,
                            mask=i_mask[:, None] & k_mask[None, :],
                            other=0.0)
        mask_tile = mask_tile[:, :, None]          # (BLOCK_I, BLOCK_K, 1)
        left = left * lgate * mask_tile
        right = right * rgate * mask_tile
    else:
        left = left * lgate
        right = right * rgate

    # ------------------------------------------------------------------
    # Store the gated tensors
    # ------------------------------------------------------------------
    left_out_ptrs = (left_out_ptr
                     + b * stride_left_b
                     + i_off[:, None, None] * stride_left_i
                     + k_off[None, :, None] * stride_left_k
                     + h_off[None, None, :] * stride_left_h)
    right_out_ptrs = (right_out_ptr
                      + b * stride_right_b
                      + i_off[:, None, None] * stride_right_i
                      + k_off[None, :, None] * stride_right_k
                      + h_off[None, None, :] * stride_right_h)

    tl.store(left_out_ptrs,  left,  mask=ikh_mask)
    tl.store(right_out_ptrs, right, mask=ikh_mask)


def custom_kernel(data):
    """Forward pass of the “outgoing” TriMul operator with Triton fusion."""
    import torch

    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]                 # input & output channel dimension
    hidden_dim = config["hidden_dim"]   # H
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------
    # Fetch model weights
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # --------------------------------------------------------------
    # Layer‑norm on the input
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B,N,N,D]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,H]
    right = torch.nn.functional.linear(x, right_proj_weight)  # [B,N,N,H]

    # --------------------------------------------------------------
    # Gate tensors (sigmoid(Linear(x)))
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # --------------------------------------------------------------
    # Optional mask handling
    # --------------------------------------------------------------
    # `mask` is either None / all‑ones or a real binary mask.
    # Convert to float32 for the Triton kernel.
    if mask is not None and not nomask:
        mask = mask.to(torch.float32).contiguous()
        has_mask = True
    else:
        # dummy tensor – pointer will never be dereferenced when HAS_MASK=0
        mask = torch.empty(0, device=input_tensor.device, dtype=torch.float32)
        has_mask = False

    # --------------------------------------------------------------
    # Triton kernel: fuse mask, left/right gates onto the projected tensors
    # --------------------------------------------------------------
    B, N, _, _ = left.shape
    H = hidden_dim

    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    left_out  = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # Strides (in element units, not bytes)
    sL_b, sL_i, sL_k, sL_h = left.stride()
    sR_b, sR_i, sR_k, sR_h = right.stride()
    sLG_b, sLG_i, sLG_k, sLG_h = left_gate.stride()
    sRG_b, sRG_i, sRG_k, sRG_h = right_gate.stride()
    sM_b, sM_i, sM_k = mask.stride() if has_mask else (0, 0, 0)

    # Tile sizes – tuned for H100 FP16 tensor‑core matmul later
    BLOCK_I = 32
    BLOCK_K = 32
    BLOCK_H = 32  # vector width for the hidden dimension

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_k = (N + BLOCK_K - 1) // BLOCK_K
    grid_h = (H + BLOCK_H - 1) // BLOCK_H
    grid_z = B * grid_h

    _mask_gate_fusion_kernel[
        (grid_i, grid_k, grid_z)
    ](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        left_gate_ptr=left_gate.data_ptr(),
        right_gate_ptr=right_gate.data_ptr(),
        mask_ptr=mask.data_ptr(),
        left_out_ptr=left_out.data_ptr(),
        right_out_ptr=right_out.data_ptr(),
        stride_left_b=sL_b, stride_left_i=sL_i, stride_left_k=sL_k, stride_left_h=sL_h,
        stride_right_b=sR_b, stride_right_i=sR_i, stride_right_k=sR_k, stride_right_h=sR_h,
        stride_lgate_b=sLG_b, stride_lgate_i=sLG_i, stride_lgate_k=sLG_k, stride_lgate_h=sLG_h,
        stride_rgate_b=sRG_b, stride_rgate_i=sRG_i, stride_rgate_k=sRG_k, stride_rgate_h=sRG_h,
        stride_mask_b=sM_b, stride_mask_i=sM_i, stride_mask_k=sM_k,
        B=B, N=N, H=H,
        BLOCK_I=BLOCK_I, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H,
        HAS_MASK=1 if has_mask else 0
    )

    # --------------------------------------------------------------
    # Batched matrix multiplication across the sequence dimension
    #      out[b,i,j,h] = Σ_k left_out[b,i,k,h] * right_out[b,j,k,h]
    # --------------------------------------------------------------
    # Cast to FP16 for Tensor‑core speed‑up
    left_h = left_out.to(torch.float16)
    right_h = right_out.to(torch.float16)

    # Reshape to (B*H, N, N) → treat each hidden slice as an independent matrix
    left_flat = left_h.permute(0, 3, 1, 2).reshape(B * H, N, N)      # [B*H, N, N]
    right_flat = right_h.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B*H, N, N]

    # Compute left @ right^T using cuBLAS (torch.bmm)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))      # [B*H, N, N]

    # Restore the original 4‑D layout and cast back to FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).to(torch.float32)  # [B,N,N,H]

    # --------------------------------------------------------------
    # Second LayerNorm over the hidden dimension, out_gate scaling,
    # and final projection back to `dim`
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
