def tri_mul_outgoing_kernel(
    A, B, C,
    BATCH,
    N,
    H,
    stride_abl, stride_al, stride_ak, stride_ad,
    stride_bbl, stride_bi, stride_bk, stride_bd,
    stride_cbl, stride_ci, stride_cj, stride_cd,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch * hidden
    batch_idx = pid_b // H
    hidden_idx = pid_b % H
    
    pid_m = tl.program_id(0)  # row block
    pid_n = tl.program_id(1)  # col block
    
    # offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute base pointers for this batch/hidden
    a_ptr = A + batch_idx * stride_abl + hidden_idx * stride_ad
    b_ptr = B + batch_idx * stride_bbl + hidden_idx * stride_bd
    c_ptr = C + batch_idx * stride_cbl + hidden_idx * stride_cd

    mask_n = offs_n < N

        # compute actual K offsets within block
        k_offs = k_base + offs_k

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # Retrieve weight tensors (already on device)
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"] 

    # LayerNorm (per last dim)
    # Compute mean and var across dim
    # Use float32 for stability
    x = input_tensor
    # shape: (bs, N, N, dim)
    # compute mean across last dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x = (x - mean) * torch.rsqrt(var + eps)
    x = x * norm_w + norm_b

    # Linear projections (weights shape hidden_dim x dim)
    # Cast to half for speed
    x_h = x.to(torch.float16)
    left = torch.nn.functional.linear(x_h, left_proj_weight.to(torch.float16))
    right = torch.nn.functional.linear(x_h, right_proj_weight.to(torch.float16))

    # Mask
    if mask is not None:
        mask_h = mask.to(torch.float16).unsqueeze(-1)  # shape (bs,N,N,1)
    else:
        mask_h = torch.ones((bs, N, N, 1), dtype=torch.float16, device=x.device)
    left = left * mask_h
    right = right * mask_h

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_h, left_gate_weight.to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_h, right_gate_weight.to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_h, out_gate_weight.to(torch.float16)))

    left = left * left_gate
    right = right * right_gate

    # TriMul: compute out = sum_k left[i,k,d] * right[j,k,d]
    # Using Triton kernel
    # allocate out tensor as half
    out = torch.empty_like(left, dtype=torch.float16)

    # Determine strides for kernel
    stride_al = left.stride(1)  # stride for i dimension (next after batch)
    stride_ak = left.stride(2)  # stride for k dimension
    stride_ad = left.stride(3)  # hidden dim stride (should be 1)
    stride_abl = left.stride(0) # batch stride

    stride_bi = right.stride(1)
    stride_bk = right.stride(2)
    stride_bd = right.stride(3)
    stride_bbl = right.stride(0)

    stride_ci = out.stride(1)
    stride_cj = out.stride(2)
    stride_cd = out.stride(3)
    stride_cbl = out.stride(0)

    # compute grid
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_b = bs * hidden_dim

    # Launch kernel
    tri_mul_outgoing_kernel[grid_m, grid_n, grid_b](
        left, right, out,
        bs, N, hidden_dim,
        stride_abl, stride_al, stride_ak, stride_ad,
        stride_bbl, stride_bi, stride_bk, stride_bd,
        stride_cbl, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )
    # After kernel, out shape (bs,N,N,hidden_dim) in half
    out = out.to(torch.float32)

    mean_o = out.mean(dim=-1, keepdim=True)

    # Final linear projection to dim
    out = torch.nn.functional.linear(out, to_out_weight)  # weight shape [dim, hidden_dim]
    return out

"""
TriMul (outgoing) forward pass – Triton‑accelerated version.

Algorithm
---------
1. Layer‑norm the pair tensor `x` over the last channel (dim).
2. Compute left/right linear projections and three gating vectors (left, right,
   out) using the provided weights (bias‑free linear layers + sigmoid).
3. Apply the pair mask (if any) and element‑wise gates to the projected tensors.
4. For every batch `b` and hidden channel `h` compute
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is exactly a batched matrix multiplication of
   `left[...,h] @ right[...,h].transpose(-2,-1)`.  The heavy O(N³) work is
   performed by a custom Triton GEMM kernel that processes
   (B·hidden) independent N×N multiplications using Tensor‑Core tiles.
   The kernel works in FP16 for the inputs, accumulates in FP32 and stores the
   result back in FP16.
5. Apply a second Layer‑Norm over the hidden dimension, multiply by the
   `out_gate`, and finally linearly project back to the original channel size.
All tensors are kept on the GPU; the kernel receives explicit strides so the
original memory layout `[B, N, N, H]` is used without any costly reshapes.

The implementation is deliberately mixed‑precision (FP16 inside the kernel,
FP32 for the small normalisation / final linear steps) to hit the target
runtime on an H100 while staying numerically stable.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: batched GEMM for the TriMul contraction
# ----------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    A, B, C,                     # pointers
    BATCH, N, H,                 # scalars
    stride_abl, stride_al, stride_ak, stride_ad,   # A strides
    stride_bbl, stride_bi, stride_bk, stride_bd,   # B strides
    stride_cbl, stride_ci, stride_cj, stride_cd,   # C strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute for each (batch, hidden) pair:
        C[b,i,j,h] = Σ_k A[b,i,k,h] * B[b,j,k,h]
    A, B, C are in layout [B, N, N, H] (row‑major, hidden last).
    The kernel processes a tile of size BLOCK_M×BLOCK_N for a single
    (b, h) pair.  Reduction over K uses BLOCK_K.
    """
    # --------------------------------------------------------------
    # Identify which (batch, hidden) we are processing.
    # program_id(2) enumerates all (batch * hidden) combinations.
    # --------------------------------------------------------------
    pid = tl.program_id(2)
    batch_idx = pid // H                      # integer division
    hidden_idx = pid - batch_idx * H          # remainder

    # --------------------------------------------------------------
    # Tile indices for rows (i) and columns (j)
    # --------------------------------------------------------------
    pid_m = tl.program_id(0)                  # tile row
    pid_n = tl.program_id(1)                  # tile col

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # --------------------------------------------------------------
    # Bounds for i / j dimensions
    # --------------------------------------------------------------
    mask_m = offs_m < N
    mask_n = offs_n < N

    # --------------------------------------------------------------
    # Base pointers for this (batch, hidden) slice.
    # --------------------------------------------------------------
    a_ptr = A + batch_idx * stride_abl + hidden_idx * stride_ad
    b_ptr = B + batch_idx * stride_bbl + hidden_idx * stride_bd
    c_ptr = C + batch_idx * stride_cbl + hidden_idx * stride_cd

    # --------------------------------------------------------------
    # Accumulator in FP32 (higher precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in tiles.
    # --------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        # Global K indices for this tile
        k_base = k * BLOCK_K
        k_offs = k_base + offs_k
        mask_k = k_offs < N

        # ---- Load A block (M × K) ---------------------------------
        a_ptrs = a_ptr + (offs_m[:, None] * stride_al) + (k_offs[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)          # cast to FP32

        # ---- Load B block (K × N) ---------------------------------
        # B is accessed as [k, j] -> rows = K, cols = N
        b_ptrs = b_ptr + (k_offs[:, None] * stride_bk) + (offs_n[None, :] * stride_bi)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # ---- Tile‑wise matrix multiply -----------------------------
        acc += tl.dot(a, b)                                   # (M,N) += (M,K)·(K,N)

    # --------------------------------------------------------------
    # Store the result tile back to C, casting to FP16.
    # --------------------------------------------------------------
    c_ptrs = c_ptr + (offs_m[:, None] * stride_ci) + (offs_n[None, :] * stride_cj)
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])


# ----------------------------------------------------------------------------- #
# Entrypoint: custom_kernel
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.
    `data` is a tuple (input, mask, weights, config) where:
        - input : torch.Tensor of shape [B, N, N, dim]
        - mask  : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights: dict with all model parameters (float32 tensors)
        - config : dict containing `dim`, `hidden_dim`, `layernorm_eps`
                   and the boolean `nomask`.
    Returns a tensor of shape [B, N, N, dim] (float32).
    """
    # --------------------------------------------------------------------- #
    # Unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # Load model weights (all are float32 on the device)
    # --------------------------------------------------------------------- #
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # --------------------------------------------------------------------- #
    # 1) Initial LayerNorm over the channel dimension (`dim`)
    # --------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)                     # still FP32

    # --------------------------------------------------------------------- #
    # 2) Cast to FP16 for the heavy compute part
    # --------------------------------------------------------------------- #
    x_h = x.to(torch.float16)

    # --------------------------------------------------------------------- #
    # Helper to perform bias‑free linear projections in FP16
    # --------------------------------------------------------------------- #
    def linear_fp16(inp, weight):
        # inp: (B,N,N,dim) -> flatten -> linear -> reshape back
        B_, N_, _, _ = inp.shape
        inp_flat = inp.view(-1, dim)                     # (B*N*N, dim)
        out_flat = F.linear(inp_flat, weight)            # (B*N*N, hidden_dim)
        return out_flat.view(B_, N_, N_, hidden_dim)

    # --------------------------------------------------------------------- #
    # 3) Projections
    # --------------------------------------------------------------------- #
    left = linear_fp16(x_h, left_proj_w.to(torch.float16))
    right = linear_fp16(x_h, right_proj_w.to(torch.float16))

    # --------------------------------------------------------------------- #
    # 4) Apply pair mask (if present)
    # --------------------------------------------------------------------- #
    if nomask:
        mask_h = torch.ones((B, N, N, 1), dtype=torch.float16, device=input_tensor.device)
    else:
        mask_h = mask.to(torch.float16).unsqueeze(-1)   # (B,N,N,1)
    left = left * mask_h
    right = right * mask_h

    # --------------------------------------------------------------------- #
    # 5) Gating vectors (sigmoid of linear projections)
    # --------------------------------------------------------------------- #
    def gate_fp16(inp, weight):
        # returns (B,N,N,hidden_dim) in FP16
        B_, N_, _, _ = inp.shape
        inp_flat = inp.view(-1, dim)
        out_flat = F.linear(inp_flat, weight)          # no bias
        out = torch.sigmoid(out_flat)                  # sigmoid works in FP16
        return out.view(B_, N_, N_, hidden_dim)

    left_gate = gate_fp16(x_h, left_gate_w.to(torch.float16))
    right_gate = gate_fp16(x_h, right_gate_w.to(torch.float16))
    out_gate = gate_fp16(x_h, out_gate_w.to(torch.float16))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 6) TriMul contraction via the Triton kernel
    # --------------------------------------------------------------------- #
    # Allocate output buffer (FP16) – same layout as left/right
    out = torch.empty_like(left, dtype=torch.float16)

    # Strides in units of elements (not bytes) – obtained from the tensors
    stride_abl = left.stride(0)    # batch stride
    stride_al  = left.stride(1)    # i (row) stride
    stride_ak  = left.stride(2)    # k (col) stride
    stride_ad  = left.stride(3)    # hidden stride (should be 1)

    stride_bbl = right.stride(0)
    stride_bi  = right.stride(1)
    stride_bk  = right.stride(2)
    stride_bd  = right.stride(3)

    stride_cbl = out.stride(0)
    stride_ci  = out.stride(1)
    stride_cj  = out.stride(2)
    stride_cd  = out.stride(3)

    # Tile sizes – tuned for H100 Tensor‑Core (FP16) performance
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid dimensions
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N
    grid_b = B * hidden_dim  # one launch instance per (batch, hidden)

    # Launch the kernel
    tri_mul_outgoing_kernel[(grid_m, grid_n, grid_b)](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_abl, stride_al, stride_ak, stride_ad,
        stride_bbl, stride_bi, stride_bk, stride_bd,
        stride_cbl, stride_ci, stride_cj, stride_cd,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Back to FP32 for the rest of the pipeline
    out = out.to(torch.float32)

    # --------------------------------------------------------------------- #
    # 7) Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # --------------------------------------------------------------------- #
    # 8) Apply the output gate
    # --------------------------------------------------------------------- #
    out = out * out_gate

    # --------------------------------------------------------------------- #
    # 9) Final linear projection back to `dim`
    # --------------------------------------------------------------------- #
    # Flatten to (B*N*N, hidden_dim) for the linear layer
    out_flat = out.view(-1, hidden_dim)
    final = F.linear(out_flat, to_out_w)                       # (B*N*N, dim)
    final = final.view(B, N, N, dim)                           # reshape to original shape

    return final
