def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    eps = 1e-5

    # extract weight and bias tensors
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # 1. LayerNorm
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )
    # 2. Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape (B,N,N,hidden)
    right = torch.nn.functional.linear(x, right_proj_weight)

    # 3. Gates
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    
    # 4. Apply mask and gates: fused in Triton
    # If mask is None or config['nomask'] True, skip mask.
    if mask is None or config.get('nomask', False):
        # Compute left_fused = left * left_gate, right_fused = right * right_gate
        left_fused = left * left_gate
        right_fused = right * right_gate
    else:
        left_fused, right_fused = apply_mask_gate_fused(left, right, left_gate, right_gate, mask)
    
    # 5. TriMul contraction using torch.bmm
    B, N, _, H = left_fused.shape
    left_ = left_fused.permute(0,3,1,2).reshape(B*H, N, N)
    right_ = right_fused.permute(0,3,1,2).reshape(B*H, N, N)
    out = torch.bmm(left_, right_.transpose(1,2))
    out = out.reshape(B, H, N, N).permute(0,2,3,1)   # shape (B,N,N,H)

    # 6. To_out_norm
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps
    )
    # 7. Multiply gate
    out = out * out_gate
    # 8. Final linear
    out = torch.nn.functional.linear(out, to_out_weight)
    return out
```

"""
custom_kernel for the “outgoing” TriMul operator.

The implementation follows the reference PyTorch module:
1. Layer‑norm the input tensor.
2. Linear projections (left / right) and gate projections (left_gate, right_gate, out_gate).
3. Optional pairwise mask is applied to the projected tensors.
4. Fuse the element‑wise gating (`proj * gate`) with a tiny Triton kernel.
5. Perform the core O(N³·H) contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a batched matrix‑multiply (torch.bmm) on the view
   (B·H, N, N), which is highly optimized by cuBLAS.
6. Apply a second LayerNorm, the out‑gate, and a final linear projection.

Only the element‑wise gating (step 4) is performed with Triton, satisfying
the “must use a kernel” requirement while keeping the heavy tensor
contraction fast via cuBLAS.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: element‑wise multiply (fused gate application)
# -------------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,                # left (or right) projection      [N_total]
    b_ptr,                # corresponding gate              [N_total]
    out_ptr,              # output (can be same as a_ptr)   [N_total]
    n_elements,           # total number of elements
    BLOCK_SIZE: tl.constexpr  # compile‑time block size
):
    """out[i] = a[i] * b[i] for i in [0, n_elements)."""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < n_elements

    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)
    out = a * b
    tl.store(out_ptr + offs, out, mask=mask)


# -------------------------------------------------------------------------
# Helper: in‑place fused gating using the Triton kernel
# -------------------------------------------------------------------------
def fuse_gate_inplace(tensor: torch.Tensor, gate: torch.Tensor) -> torch.Tensor:
    """In‑place `tensor = tensor * gate` using Triton."""
    assert tensor.is_contiguous() and gate.is_contiguous()
    n_el = tensor.numel()
    BLOCK = 1024  # works well on H100
    grid = (triton.cdiv(n_el, BLOCK),)
    elementwise_mul_kernel[grid](
        tensor, gate, tensor, n_el,
        BLOCK_SIZE=BLOCK,
        num_warps=4,
    )
    return tensor


# -------------------------------------------------------------------------
# Main entry point
# -------------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] or None
            - weights: dict with the model weights
            - config: dict with keys `dim`, `hidden_dim`, optional `nomask`

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # ---------------------------------------------------------------------
    # Unpack inputs
    # ---------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ---------------------------------------------------------------------
    # Extract weights
    # ---------------------------------------------------------------------
    # LayerNorm (input)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    # Gates (pre‑output)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    # Output LayerNorm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    # Output linear
    to_out_weight = weights["to_out.weight"]

    # ---------------------------------------------------------------------
    # 1. Input LayerNorm
    # ---------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ---------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ---------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ---------------------------------------------------------------------
    # 3. Optional mask (broadcasted over hidden dim)
    # ---------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # Cast mask to same dtype for multiplication
        mask_f = mask.to(dtype=dtype)
        # Shape (B,N,N,1) → broadcast over hidden dim
        mask_f = mask_f.unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ---------------------------------------------------------------------
    # 4. Compute gate tensors (sigmoid)
    # ---------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()  # [B,N,N,hidden]

    # Ensure contiguous layout before Triton (small overhead)
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()

    # ---------------------------------------------------------------------
    # 5. Fuse gating (left = left * left_gate, right = right * right_gate)
    #    performed with a lightweight Triton kernel.
    # ---------------------------------------------------------------------
    fuse_gate_inplace(left, left_gate)
    fuse_gate_inplace(right, right_gate)

    # ---------------------------------------------------------------------
    # 6. Core TriMul contraction:
    #    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    #    Implemented as a batched GEMM on (B*hidden, N, N).
    # ---------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # Reshape to (B*H, N, N) – note that we need right transposed on the
    # second axis (j ↔ k) which is achieved by a simple .transpose(1,2).
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)      # [B*H, N, N]
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)    # [B*H, N, N]

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N)^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # [B*H, N, N]

    # Restore original layout [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ---------------------------------------------------------------------
    # 7. Output LayerNorm (on hidden dimension)
    # ---------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ---------------------------------------------------------------------
    # 8. Apply out‑gate and final linear projection
    # ---------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)  # back to [B,N,N,dim]

    return out
