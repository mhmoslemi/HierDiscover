@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
):
    # program ids
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_g = tl.program_id(2)

    # compute base offset for group (b,h)
    b = pid_g // H
    h = pid_g % H

    # Compute starting offsets for i and j blocks
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # pointer to left base for this group
    left_base = left_ptr + b * stride_bl + h * stride_hl

    # pointer to right base for this group
    right_base = right_ptr + b * stride_br + h * stride_hr

    # pointer to output base
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        k0 = k * BLOCK_K
        # offsets within left and right
        left_offset = left_base + i0 * stride_il + k0 * stride_kl
        right_offset = right_base + k0 * stride_kr + j0 * stride_ir  # note swapped for transpose

        # load blocks (with mask for out-of-bound)
        left_mask = (i0 + tl.arange(0, BLOCK_M) < N)[:, None] & (k0 + tl.arange(0, BLOCK_K) < N)[None, :]
        right_mask = (k0 + tl.arange(0, BLOCK_K) < N)[:, None] & (j0 + tl.arange(0, BLOCK_N) < N)[None, :]

        left_block = tl.load(left_offset, mask=left_mask, other=0.0).to(tl.float32)
        right_block = tl.load(right_offset, mask=right_mask, other=0.0).to(tl.float32)

        # compute dot: we need to compute left_block (M x K) @ right_block.T? Actually right_block loaded as KxN (k rows, j cols) thanks to stride order.
        # The right_block currently shape (BLOCK_K, BLOCK_N) but each row is k0 + offset; columns correspond to j.
        # dot expects (M,K) and (K,N) -> (M,N)
        acc += tl.dot(left_block, right_block)

    # store result
    out_offset = out_base + i0 * stride_io + j0 * stride_jo
    # mask for output out-of-bound
    out_mask = (i0 + tl.arange(0, BLOCK_M) < N)[:, None] & (j0 + tl.arange(0, BLOCK_N) < N)[None, :]
    tl.store(out_offset, acc, mask=out_mask)

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Get program IDs.
    pid_i = tl.program_id(0)  # block row
    pid_j = tl.program_id(1)  # block col
    pid_g = tl.program_id(2)  # batch * hidden group

    # Determine batch and hidden index.
    h = pid_g % H
    b = pid_g // H

    # Compute offsets.
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Base pointers for this (b,h)
    left_base = left_ptr + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base = out_ptr + b * stride_bo + h * stride_ho

    # Prepare index vectors
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # Initialize accumulator.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_tiles):
        k0 = kk * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)

        # Compute masks
        mask_i = i < N
        mask_j = j < N
        mask_k = k < N

        # Load left tile (M x K)
        left_ptrs = left_base + i[:, None] * stride_il + k[None, :] * stride_kl
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile (K x N) (transposed layout)
        right_ptrs = right_base + k[:, None] * stride_kr + j[None, :] * stride_ir
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Accumulate

"""
TriMul (outgoing) forward pass.

The operation works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} :

1. Layer‑normalize X.
2. Project to a hidden dimension H:
        L = X·W_left      (B,N,N,H)
        R = X·W_right     (B,N,N,H)
3. (Optionally) mask L and R.
4. Compute gated projections:
        L ← L * σ(X·W_left_gate)
        R ← R * σ(X·W_right_gate)
5. Core “TriMul” reduction:
        out[b,i,j,h] = Σ_k L[b,i,k,h] * R[b,j,k,h]
   This is a batched matrix multiplication for every hidden channel.
   It is implemented with a custom Triton kernel that performs
   a GEMM on the (i,k) and (j,k) slices, accumulating in FP32.
6. Layer‑normalize the result, apply an output gate σ(X·W_out_gate)
   and a final linear projection back to C.
All heavy arithmetic (step 5) runs on the GPU via Triton,
while the surrounding linear layers, layer‑norms and sigmoids use
standard PyTorch primitives.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel for the “TriMul” reduction.
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,               # float16*  [B,N,N,H]
    right_ptr,              # float16*  [B,N,N,H]
    out_ptr,                # float32*  [B,N,N,H]
    B: tl.int32,            # batch size
    N: tl.int32,            # sequence length
    H: tl.int32,            # hidden dimension
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_M: tl.constexpr,  # tile rows   (i dimension)
    BLOCK_N: tl.constexpr,  # tile cols   (j dimension)
    BLOCK_K: tl.constexpr,  # tile reduction (k dimension)
):
    """Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]"""
    pid_i = tl.program_id(0)               # row‑tile index
    pid_j = tl.program_id(1)               # col‑tile index
    pid_g = tl.program_id(2)               # group = batch*hidden

    # ------------------------------------------------------------------
    # Resolve batch `b` and hidden channel `h` from the flat group id.
    # ------------------------------------------------------------------
    h = pid_g % H
    b = pid_g // H

    # ------------------------------------------------------------------
    # Starting offsets for the current tile.
    # ------------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Base pointers for this (b, h) pair.
    left_base  = left_ptr  + b * stride_bl + h * stride_hl
    right_base = right_ptr + b * stride_br + h * stride_hr
    out_base   = out_ptr   + b * stride_bo + h * stride_ho

    # Vector of row / col / reduction indices.
    i = i0 + tl.arange(0, BLOCK_M)
    j = j0 + tl.arange(0, BLOCK_N)

    # ------------------------------------------------------------------
    # Accumulator in FP32.
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles needed to cover the N dimension.
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_tiles):
        k0 = kk * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)

        # Masks for out‑of‑bounds accesses.
        mask_i = i < N          # (BLOCK_M,)
        mask_j = j < N          # (BLOCK_N,)
        mask_k = k < N          # (BLOCK_K,)

        # --------------------------------------------------------------
        # Load a BLOCK_M×BLOCK_K tile from left (i‑k plane).
        # --------------------------------------------------------------
        left_ptrs = left_base + i[:, None] * stride_il + k[None, :] * stride_kl
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)          # cast to FP32 for accumulation

        # --------------------------------------------------------------
        # Load a BLOCK_K×BLOCK_N tile from right (k‑j plane).
        # Note the transposition: we index right as (k, j) to obtain
        # the needed product left[i,k] * right[j,k].
        # --------------------------------------------------------------
        right_ptrs = right_base + k[:, None] * stride_kr + j[None, :] * stride_ir
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # --------------------------------------------------------------
        # Accumulate the inner product over k.
        # --------------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # --------------------------------------------------------------
    # Write the result back to global memory (still FP32).
    # --------------------------------------------------------------
    out_ptrs = out_base + i[:, None] * stride_io + j[None, :] * stride_jo
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Public entry point expected by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Helper: LayerNorm (no bias handling done manually, we reuse torch)
    # ------------------------------------------------------------------
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # → [B,N,N,dim], dtype=float32

    # ------------------------------------------------------------------
    # Linear projections to hidden dimension
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (mask may be None if nomask=True)
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        mask_exp = mask[..., None].to(left.dtype)   # broadcast over hidden
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # Gating projections (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy TriMul kernel (memory‑bound part)
    # ------------------------------------------------------------------
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Output buffer (FP32 accumulation, final type)
    out_fp32 = torch.empty((bs, seq_len, seq_len, hidden_dim),
                           dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Compute strides (in elements) for the kernel arguments
    # ------------------------------------------------------------------
    stride_bl, stride_il, stride_kl, stride_hl = (
        left_fp16.stride(0),
        left_fp16.stride(1),
        left_fp16.stride(2),
        left_fp16.stride(3),
    )
    stride_br, stride_ir, stride_kr, stride_hr = (
        right_fp16.stride(0),
        right_fp16.stride(1),
        right_fp16.stride(2),
        right_fp16.stride(3),
    )
    stride_bo, stride_io, stride_jo, stride_ho = (
        out_fp32.stride(0),
        out_fp32.stride(1),
        out_fp32.stride(2),
        out_fp32.stride(3),
    )

    # ------------------------------------------------------------------
    # Triton grid configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64   # rows tile
    BLOCK_N = 64   # cols tile
    BLOCK_K = 32   # reduction tile (k dimension)

    grid = (
        triton.cdiv(seq_len, BLOCK_M),
        triton.cdiv(seq_len, BLOCK_N),
        bs * hidden_dim,          # one Triton program per (batch, hidden) pair
    )

    # Launch the kernel
    trimul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp32,
        bs,
        seq_len,
        hidden_dim,
        stride_bl,
        stride_il,
        stride_kl,
        stride_hl,
        stride_br,
        stride_ir,
        stride_kr,
        stride_hr,
        stride_bo,
        stride_io,
        stride_jo,
        stride_ho,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: to_out_norm, output gate, final linear projection
    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension
    out = F.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply output gate
    out = out * out_gate

    # Final linear mapping back to `dim`
    output = F.linear(out, weights["to_out.weight"])

    return output
