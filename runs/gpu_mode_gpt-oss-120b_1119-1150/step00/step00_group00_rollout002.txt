    left_ptr, right_ptr, out_ptr,

    stride_right_b, stride_right_i, stride_right_k, stride_right_h,

    b = hb // H

"""
custom_kernel - Triton‑accelerated TriMul (outgoing) forward pass.

The heavy part of TriMul is the cubic‑order tensor contraction

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

which is equivalent to a batch of matrix multiplications
    left[...,d] @ right[...,d].T  for every hidden channel d.
A custom Triton kernel performs this batched GEMM on the
    (B × hidden_dim) independent matrices of shape (N, N),
while the surrounding LayerNorm / linear / gating operations stay
in pure PyTorch for simplicity.

The kernel tiles the output in (BLOCK_M, BLOCK_N) tiles and iterates
over K in BLOCK_K steps.  The batch‑and‑head dimension is collapsed
into a single program‑id axis, so a launch grid of
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*hidden_dim)
covers the whole computation.

All tensors are expected on the GPU; the kernel works with FP16
(for lower memory traffic) and accumulates in FP32 to keep numerical
accuracy.  The final result is cast back to FP32 before the remaining
LayerNorm, gating and output projection.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------
@triton.jit
def _triple_mul_gemm_kernel(
    left_ptr,        # const * left  [B, N, N, H]  (i, k)
    right_ptr,       # const * right [B, N, N, H]  (j, k) – we read as (k, j)
    out_ptr,         # * out   [B, N, N, H]  (i, j)
    B, N, H,         # problem sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes, for every (b, h):
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    where left[i,k] and right[j,k] share the k dimension.
    The computation is performed as a GEMM: left @ (right)^T.
    """
    pid_i = tl.program_id(0)        # block row (i)
    pid_j = tl.program_id(1)        # block col (j)
    pid_hb = tl.program_id(2)       # combined batch‑head index

    # De‑compose batch and hidden‑dim indices
    hb = pid_hb
    b = hb // H
    h = hb % H

    # Tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Bounds checks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Base address for this (b, h) slice
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        cur_k = k + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ---- left tile (M, K) ---------------------------------
        # left[b, i, k, h]
        ptrs_l = left_ptr + base_l \
                 + offs_i[:, None] * stride_l_i \
                 + cur_k[None, :] * stride_l_k
        left = tl.load(ptrs_l,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=tl.float16(0.0)).to(tl.float32)

        # ---- right tile, transposed to shape (K, N) ----------
        # we need right[b, j, k, h] -> treat as (k, j)
        ptrs_r = right_ptr + base_r \
                 + cur_k[:, None] * stride_r_k \
                 + offs_j[None, :] * stride_r_i   # stride_r_i indexes the “j” dimension
        right = tl.load(ptrs_r,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=tl.float16(0.0)).to(tl.float32)

        # ---- Accumulate -------------------------------------------------
        acc += tl.dot(left, right)          # (M,K) @ (K,N) -> (M,N)

    # Store the result (cast back to FP16 for the output tensor)
    ptrs_o = out_ptr + base_o \
             + offs_i[:, None] * stride_o_i \
             + offs_j[None, :] * stride_o_j
    tl.store(ptrs_o, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])

    # 1) Input LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Gating signals (sigmoid)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, left_gate_weight).sigmoid()
    right_gate = F.linear(x, right_gate_weight).sigmoid()
    out_gate = F.linear(x, out_gate_weight).sigmoid()

    # ------------------------------------------------------------------
    # 4) Apply mask (if provided) before gating – matches reference
    # ------------------------------------------------------------------
    if mask is not None and mask.numel() > 0:
        # mask: [B,N,N] -> broadcast to [B,N,N,1]
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 5) Apply gates
    # ------------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Cubic contraction via Triton (left @ right^T for each hidden head)
    # ------------------------------------------------------------------
    # Cast to FP16 for the kernel (lower memory traffic)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h, dtype=torch.float16)

    # Strides (in elements) for the three tensors
    # Shape = [B, N, N, H]; contiguous layout => strides:
    # (B, i, k, h)
    s_l = left_h.stride()
    s_r = right_h.stride()
    s_o = out_h.stride()

    stride_l_b, stride_l_i, stride_l_k, stride_l_h = s_l
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = s_r
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = s_o

    # Tile sizes – tuned for H100, can be adjusted if needed
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    # Launch the Triton kernel
    _triple_mul_gemm_kernel[grid](
        left_h,
        right_h,
        out_h,
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 has many warps; 8 is a good default
    )

    # Back to FP32 for the remaining ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 7) Output LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)

    # ------------------------------------------------------------------
    # 8) Apply final gate and linear projection back to dim
    # ------------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # bias is None in the reference

    return out
