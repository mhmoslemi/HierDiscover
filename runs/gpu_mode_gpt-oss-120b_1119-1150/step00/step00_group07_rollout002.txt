@triton.jit
def batched_gemm_kernel(A, B, C,
                        M, N, K,
                        stride_am, stride_ak, stride_ab,
                        stride_bk, stride_bn, stride_bb,
                        stride_cm, stride_cn, stride_cb,
                        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Block start offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute masks for final store
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k
        # Compute offsets for A and B
        a_ptrs = A + batch_id * stride_ab + offs_m[:, None] * stride_am + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        b_ptrs = B + batch_id * stride_bb + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk + offs_n[None, :] * stride_bn

        # Load A and B tiles
        a = tl.load(a_ptrs,
                    mask=(mask_m[:, None] & (cur_k + tl.arange(0, BLOCK_K) < K)),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=((cur_k + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :]),
                    other=0.0)

        # Compute dot product
        acc += tl.dot(a, b)

    # Write back results
    c_ptrs = C + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

"""
custom_kernel: Triton‑accelerated forward pass of the “outgoing” TriMul operation.

Algorithm
---------
1. Layer‑normalize the input tensor `x` (shape B×N×N×dim) using the provided
   LN weights/bias.
2. Compute four linear projections on the normalized tensor:
   * left_proj, right_proj  – shape B×N×N×hidden_dim
   * left_gate, right_gate, out_gate – apply sigmoid after the linear maps.
3. (Optional) mask the left/right projections.
4. Fuse the element‑wise gating:
      left  = left_proj  * left_gate
      right = right_proj * right_gate
5. Compute the core TriMul contraction
      out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   This is a batched matrix‑multiply for each hidden‑dim channel.
   We reshape the tensors to (B*hidden_dim, N, N) and launch a custom
   Triton GEMM kernel that accumulates in fp32 and stores fp16.
6. Reshape the result back to B×N×N×hidden_dim.
7. Apply a second LayerNorm (`to_out_norm`), multiply by `out_gate`,
   and finally a linear projection back to `dim`.
8. Cast the result to fp32 (the model’s default dtype) and return.

The heavy O(N³·hidden_dim) contraction is carried out by the Triton kernel,
while all surrounding ops use native PyTorch kernels (already highly
optimised). Mixed‑precision (fp16) is used throughout to keep memory bandwidth
low; accumulation inside the kernel is performed in fp32 to preserve numerical
stability.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the einsum contraction
# ----------------------------------------------------------------------
@triton.jit
def _batched_gemm_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix dimensions
    stride_am, stride_ak, stride_ab,   # A strides: (batch,M,K)
    stride_bk, stride_bn, stride_bb,   # B strides: (batch,K,N)
    stride_cm, stride_cn, stride_cb,   # C strides: (batch,M,N)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C[b] = A[b] @ B[b] for all batches."""
    batch_id = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets of the current block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, K, BLOCK_K):
        cur_k = k
        # Compute the start pointers for this k‑block
        a_ptrs = (
            A
            + batch_id * stride_ab
            + offs_m[:, None] * stride_am
            + (cur_k + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        )
        b_ptrs = (
            B
            + batch_id * stride_bb
            + (cur_k + tl.arange(0, BLOCK_K))[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load with boundary guards
        a = tl.load(
            a_ptrs,
            mask=(mask_m[:, None] & (cur_k + tl.arange(0, BLOCK_K) < K)),
            other=0.0,
        )
        b = tl.load(
            b_ptrs,
            mask=((cur_k + tl.arange(0, BLOCK_K) < K)[:, None] & mask_n[None, :]),
            other=0.0,
        )
        # Accumulate dot product (fp16 * fp16 -> fp32)
        acc += tl.dot(a, b)

    # Write result back to C
    c_ptrs = C + batch_id * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c = acc.to(tl.float16)
    tl.store(c_ptrs, c, mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation using a custom Triton kernel
    for the core contraction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape [B, N, N, dim]
            - mask: torch.Tensor or None, shape [B, N, N] (binary mask)
            - weights: dict of torch.Tensors (model parameters)
            - config: dict containing at least "dim", "hidden_dim",
                      optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # expected torch.float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to cast a weight tensor to fp16 on the correct device
    # ------------------------------------------------------------------
    def _to_fp16(t):
        return t.to(device=device, dtype=torch.float16)

    # ------------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_weight = _to_fp16(weights["norm.weight"])
    norm_bias = _to_fp16(weights["norm.bias"])
    # Cast input to fp16 for the heavy work; LN can stay in fp16 safely
    x = input_tensor.to(torch.float16)
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias) and gating scalars
    # ------------------------------------------------------------------
    left_proj = F.linear(x, _to_fp16(weights["left_proj.weight"]))
    right_proj = F.linear(x, _to_fp16(weights["right_proj.weight"]))

    left_gate = F.linear(x, _to_fp16(weights["left_gate.weight"])).sigmoid()
    right_gate = F.linear(x, _to_fp16(weights["right_gate.weight"])).sigmoid()
    out_gate = F.linear(x, _to_fp16(weights["out_gate.weight"])).sigmoid()

    # ------------------------------------------------------------------
    # Apply mask if present and required
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask shape [B, N, N] -> [B, N, N, 1] broadcast over hidden_dim
        mask_fp16 = mask.to(torch.float16).unsqueeze(-1)
        left_proj = left_proj * mask_fp16
        right_proj = right_proj * mask_fp16

    # ------------------------------------------------------------------
    # Element‑wise gating
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left shape = [B, N, N, hidden_dim]

    # Rearrange to (B, hidden_dim, N, N) for easy flattening
    left_perm = left.permute(0, 3, 1, 2).contiguous()      # (B, D, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()    # (B, D, N, N) with k↔j

    batch_dim = B * hidden_dim
    left_flat = left_perm.view(batch_dim, N, N)   # (B*D, M=N, K=N)
    right_flat = right_perm.view(batch_dim, N, N) # (B*D, K=N, N)

    # Allocate output buffer for the contraction
    out_flat = torch.empty_like(left_flat)  # fp16

    # ------------------------------------------------------------------
    # Triton GEMM launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    _batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(1),   # stride_am (M‑step)
        left_flat.stride(2),   # stride_ak (K‑step)
        left_flat.stride(0),   # stride_ab (batch step)
        right_flat.stride(1),  # stride_bk (K‑step)
        right_flat.stride(2),  # stride_bn (N‑step)
        right_flat.stride(0),  # stride_bb (batch step)
        out_flat.stride(1),    # stride_cm (M‑step)
        out_flat.stride(2),    # stride_cn (N‑step)
        out_flat.stride(0),    # stride_cb (batch step)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape contraction result back to original layout
    # ------------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,D)

    # ------------------------------------------------------------------
    # Second LayerNorm, gating and final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = _to_fp16(weights["to_out_norm.weight"])
    to_out_norm_bias = _to_fp16(weights["to_out_norm.bias"])

        out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps

    out = out * out_gate

    # Final linear back to `dim`
    out = F.linear(out, _to_fp16(weights["to_out.weight"]))

    # Return as float32 (model’s default dtype)
    return out.to(torch.float32)
