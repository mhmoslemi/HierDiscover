def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]; hidden_dim = config["hidden_dim"]; 
    # maybe config may contain other fields.

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # flatten last dim size dim
    x = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # linear projections and gates
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # compute projections; note weight shape (out_features, in_features)
    # Use torch.nn.functional.linear for multi-dim
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    # mask
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # broadcast
        left = left * mask_exp
        right = right * mask_exp

    # apply left/right gates
    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    # Prepare output
    B, N, _, _ = left.shape
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=x.device)

    # call Triton kernel
    trimul_outgoing_kernel(...)
    
    # post norm and linear
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # shape (B,N,N,dim)
    return out

"""
custom_kernel:  Triton‑accelerated “outgoing” TriMul (Triangular Multiplicative Update)

The heavyweight part of TriMul is the three‑way sum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
For each hidden feature `d` this is exactly a matrix multiplication
    (left_d) @ (right_d)^T   (shape N×N).

The implementation:
* LayerNorm, linear projections and gating are performed with regular
  PyTorch (highly optimized and easy to write).
* The N³ reduction is carried out by a custom Triton kernel that
  processes a tile of the (i,j) output plane for a given batch and
  hidden‑feature `d`.  The kernel computes a batched GEMM with
  half‑precision inputs and a float‑32 accumulator.
* Finally the result is LayerNorm‑normalized, gated, and projected back
  to the original `dim` using PyTorch.

The kernel works for any batch size ≤2, any sequence length up to 1024,
and hidden dimensions that fit in the given configuration.
"""

import triton
import triton.language as tl
import torch
from typing import Tuple, Dict


# --------------------------------------------------------------
# Triton kernel: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
# --------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # half *: [B, N, N, H]   (i,k) layout
    right_ptr,               # half *: [B, N, N, H]   (j,k) layout
    out_ptr,                 # float *: [B, N, N, H]  (i,j) output
    B, N, H,                 # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,   # left strides
    stride_r_b, stride_r_j, stride_r_k, stride_r_h,   # right strides
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,   # out strides
    BLOCK_M: tl.constexpr,   # tile size in i‑dimension
    BLOCK_N: tl.constexpr,   # tile size in j‑dimension
    BLOCK_K: tl.constexpr,   # tile size in k‑dimension
):
    """Batched matmul over the sequence dimension for every hidden head."""
    pid0 = tl.program_id(0)          # combined (batch, head) index
    pid1 = tl.program_id(1)          # i‑tile
    pid2 = tl.program_id(2)          # j‑tile

    # -----------------------------------------------------------
    # decode batch and hidden dimension from pid0
    # -----------------------------------------------------------
    b = pid0 // H
    h = pid0 % H

    # -----------------------------------------------------------
    # tile offsets
    # -----------------------------------------------------------
    offs_i = pid1 * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid2 * BLOCK_N + tl.arange(0, BLOCK_N)

    # bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in float32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------
    # k‑loop
    # -----------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        offs_k = k_tile * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load A = left[b, i, k, h]   shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        a_ptr = (left_ptr
                 + b * stride_l_b
                 + offs_i[:, None] * stride_l_i
                 + offs_k[None, :] * stride_l_k
                 + h * stride_l_h)
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)               # half

        # ------------------------------------------------------------------
        # Load B^T = right[b, j, k, h]   shape (BLOCK_K, BLOCK_N)
        # we load it already transposed (k as rows, j as columns)
        # ------------------------------------------------------------------
        b_ptr = (right_ptr
                 + b * stride_r_b
                 + offs_k[:, None] * stride_r_k
                 + offs_j[None, :] * stride_r_j
                 + h * stride_r_h)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)               # half

        # ------------------------------------------------------------------
        # Accumulate dot product over k
        # ------------------------------------------------------------------
        acc += tl.dot(a, b)                # (BLOCK_M, BLOCK_N) float32

    # -----------------------------------------------------------
    # Store the result to out[b,i,j,h]
    # -----------------------------------------------------------
    out_ptrs = (out_ptr
                + b * stride_o_b
                + offs_i[:, None] * stride_o_i
                + offs_j[None, :] * stride_o_j
                + h * stride_o_h)
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# --------------------------------------------------------------
# Python entry point
# --------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor,
                              torch.Tensor,
                              Dict[str, torch.Tensor],
                              Dict]) -> torch.Tensor:
    """
    Forward pass of the outgoing TriMul operation.

    Arguments
    ----------
    data :
        (input, mask, weights, config)
        - input : torch.Tensor of shape [B, N, N, dim]
        - mask  : torch.Tensor of shape [B, N, N]  (may be ``None``)
        - weights : dict of model parameters
        - config  : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device as input)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]          # H in the paper

    device = input_tensor.device
    dtype = input_tensor.dtype                # typically float32

    # ------------------------------------------------------------------
    # LayerNorm on the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B,N,N,dim]  float32

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_w)        # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # Mask handling (if a mask is supplied and ``nomask`` is False)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask is boolean or 0/1, broadcast over the feature dimension
        mask_f = mask.to(dtype).unsqueeze(-1)                # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gating (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half‑precision for the N³ kernel (reduces memory bandwidth)
    # ------------------------------------------------------------------
    left_h  = left.contiguous().to(torch.float16)
    right_h = right.contiguous().to(torch.float16)

    B, N, _, _ = left_h.shape

    # ------------------------------------------------------------------
    # Allocate output tensor for the kernel (float32 accumulator)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Compute strides (in element units) – required for the Triton kernel
    # ------------------------------------------------------------------
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_h.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_h = right_h.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out.stride()

    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100 (values are good defaults)
    # ------------------------------------------------------------------
    BLOCK_M = 64   # i‑dimension tile
    BLOCK_N = 64   # j‑dimension tile
    BLOCK_K = 32   # k‑dimension tile

    # Number of tiles in each dimension
    grid_m = (N + BLOCK_M - 1) // BLOCK_M
    grid_n = (N + BLOCK_N - 1) // BLOCK_N

    # A single grid dimension encodes both batch and hidden‐dim
    grid = (B * hidden_dim, grid_m, grid_n)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    trimul_outgoing_kernel[grid](
        left_h.data_ptr(),
        right_h.data_ptr(),
        out.data_ptr(),
        B, N, hidden_dim,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_j, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 comfortable with 8 warps per block
    )

    # ------------------------------------------------------------------
    # Post‑processing: LayerNorm over hidden_dim, gating, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w       = weights["to_out.weight"]   # (dim, hidden_dim)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate                         # gating
    out = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return out
