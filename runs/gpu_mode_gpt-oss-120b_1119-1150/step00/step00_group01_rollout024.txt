def custom_kernel(data):
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl
    
    # Unpack data
    input_tensor, mask_tensor, weights, config = data
    # Ensure device
    device = input_tensor.device

    # dims
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    
    # Helper: layer norm
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # normalize along last dim
    x = input_tensor
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias
    
    # Linear projections without bias
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]
    
    # Left and right projections
    left = F.linear(x_norm, left_proj_weight)  # [B,N,N,hidden_dim]
    right = F.linear(x_norm, right_proj_weight)
    
    # Apply mask if provided
    if mask_tensor is not None:
        # assume mask shape [B,N,N] with dtype float or bool
        if mask_tensor.dtype != torch.float32:
            mask = mask_tensor.to(torch.float32)
        else:
            mask = mask_tensor
        mask = mask.unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask
    
    # Gates
    left_gate = F.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = F.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = F.linear(x_norm, out_gate_weight).sigmoid()
    
    # Apply gates
    left = left * left_gate
    right = right * right_gate

    left = left.to(torch.float16)
    right = right.to(torch.float16)

    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N) where dims (i,k)
    right_perm = right.permute(0, 3, 2, 1).contiguous() # (B, H, N, N) where dims (k,i)
    
    # Merge batch and heads
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Compute strides
    stride_left_batch, stride_left_m, stride_left_k = left_mat.stride()

    stride_out_batch, stride_out_m, stride_out_n = out_mat.stride()

    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), BH)
    
    @triton.jit
    def tri_mul_gemm_kernel(
        left_ptr, right_ptr, out_ptr,
        N,
        stride_l_batch, stride_l_m, stride_l_k,
        stride_r_batch, stride_r_k, stride_r_n,
        stride_o_batch, stride_o_m, stride_o_n,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

        pid_n = tl.program_id(1)
        pid_z = tl.program_id(2)
        
        # Offsets
        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)

            acc += tl.dot(a, b)

        stride_out_batch, stride_out_m, stride_out_n,

    )

    # Final linear mapping to dim
    out_final = F.linear(out_norm, to_out_weight)  # shape [B,N,N,dim]
    
    return out_final

"""
TriMul (outgoing) forward implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B, N, N, C]) using the provided
   `norm.weight` and `norm.bias`.
2. Linear projections (no bias):
      left  = x_norm @ left_proj.weight   (→ hidden_dim)
      right = x_norm @ right_proj.weight
3. Optional mask (broadcasted on the last dimension) is applied to both
   projections.
4. Gate values are obtained with sigmoid‑gated linear layers:
      left_gate  = sigmoid(x_norm @ left_gate.weight)
      right_gate = sigmoid(x_norm @ right_gate.weight)
      out_gate   = sigmoid(x_norm @ out_gate.weight)
   The gated tensors become:
      left  = left  * left_gate
      right = right * right_gate
5. The core cubic operation is
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is a batched matrix multiply per hidden channel:
      out_d = left_d @ right_d.T
   This is realised by a custom Triton kernel that works on FP16
   (Tensor‑core) data while accumulating in FP32 for numerical stability.
6. The result (still in FP16) is reshaped back to [B, N, N, hidden_dim] and
   cast to FP32.
7. Apply a second Layer‑Norm (`to_out_norm`) on the hidden dimension,
   multiply element‑wise by `out_gate`, and finally linearly project back
   to the original feature dimension (`to_out.weight`).

The kernel fuses the whole N³ contraction across all hidden channels and
batches, yielding a substantial speed‑up over a naïve PyTorch `einsum`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul contraction
# --------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    left_ptr, right_ptr, out_ptr,
    N,
    stride_l_batch, stride_l_m, stride_l_k,
    stride_r_batch, stride_r_k, stride_r_n,
    stride_o_batch, stride_o_m, stride_o_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute for a single (batch*head) slice:
       out = left @ right.T   where left  [M, K] and right [K, N].
       All tensors are FP16; accumulator is FP32.
    """
    pid_m = tl.program_id(0)          # block row
    pid_n = tl.program_id(1)          # block column
    pid_z = tl.program_id(2)          # combined batch*head index

    # ------------------------------------------------------------------
    # Offsets for the block of rows/cols we are responsible for
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator (FP32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k in range(0, num_k_tiles):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load A_tile: shape (BLOCK_M, BLOCK_K)
        a_ptr = left_ptr + pid_z * stride_l_batch \
                + offs_m[:, None] * stride_l_m \
                + offs_k[None, :] * stride_l_k
        a = tl.load(a_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # Load B_tile (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptr = right_ptr + pid_z * stride_r_batch \
                + offs_k[:, None] * stride_r_k \
                + offs_n[None, :] * stride_r_n
        b = tl.load(b_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # Multiply‑accumulate (Tensor‑core)
        acc += tl.dot(a, b)   # a: (M,K), b: (K,N) → result (M,N)

    # ------------------------------------------------------------------
    # Write result back (cast to FP16)
    # ------------------------------------------------------------------
    c_ptr = out_ptr + pid_z * stride_o_batch \
            + offs_m[:, None] * stride_o_m \
            + offs_n[None, :] * stride_o_n
    tl.store(c_ptr, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

# Entry point used by the evaluation harness
# --------------------------------------------------------------

    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # Unpack arguments
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device

    # ------------------------------------------------------------------
    # Extract shapes & config
    # ------------------------------------------------------------------
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Layer‑norm on the last dimension (C)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # shape: [B, N, N, dim]  (float32)

    # ------------------------------------------------------------------
    # Linear projections (FP32)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # → [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Optional mask (broadcasted on the last dimension)
    # ------------------------------------------------------------------
    if mask_tensor is not None and not config.get("nomask", False):
        # Ensure mask is float32 on the same device
        mask = mask_tensor.to(dtype=torch.float32, device=device)
        mask = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates (still FP32)
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy cubic contraction
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left  → [B, hidden_dim, N, N]   (i,k)
    #   right → [B, hidden_dim, N, N]   (k,j)  (already transposed)
    # ------------------------------------------------------------------
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B, H, N, N) with k as first axis

    # Merge batch and head dimensions → (B*H, N, N)
    BH = B * hidden_dim
    left_mat  = left_perm.view(BH, N, N)
    right_mat = right_perm.view(BH, N, N)

    # Allocate output in FP16
    out_mat = torch.empty_like(left_mat, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Strides (convert to Python ints for Triton)
    # ------------------------------------------------------------------
    s_l_batch, s_l_m, s_l_k = left_mat.stride()
    s_r_batch, s_r_k, s_r_n = right_mat.stride()
    s_o_batch, s_o_m, s_o_n = out_mat.stride()

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # number of row blocks
        (N + BLOCK_N - 1) // BLOCK_N,          # number of column blocks
        BH                                      # one block per (batch, head)
    )

    # Launch the kernel
    tri_mul_gemm_kernel[grid](
        left_mat, right_mat, out_mat,
        N,
        s_l_batch, s_l_m, s_l_k,
        s_r_batch, s_r_k, s_r_n,
        s_o_batch, s_o_m, s_o_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,
        num_stages=2,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim] and cast to FP32
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # Second Layer‑norm on hidden dimension, gating and final projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    # Apply the output gate (already FP32)
    out_norm = out_norm * out_gate

    # Final linear projection back to `dim`
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_weight)   # shape [B, N, N, dim]

    return output
```
