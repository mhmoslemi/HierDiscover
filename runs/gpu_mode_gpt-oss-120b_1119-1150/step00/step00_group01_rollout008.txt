import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A, B, C,
                          M, N, K,
                          stride_ab, stride_ak, stride_ai,
                          stride_bb, stride_bk, stride_bj,
                          stride_cb, stride_ci, stride_cj,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k

        a_ptrs = A + pid_batch * stride_ab + offs_m[:, None] * stride_ai + k_offset[None, :] * stride_ak
        b_ptrs = B + pid_batch * stride_bb + k_offset[:, None] * stride_bk + offs_n[None, :] * stride_bj

        # Load with conditional masking
        a = tl.load(a_ptrs, mask=(offs_m < M) & (k_offset < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k_offset < K) & (offs_n < N), other=0.0)

        acc += tl.dot(a, b)

    # Write back
    c_ptrs = C + pid_batch * stride_cb + offs_m[:, None] * stride_ci + offs_n[None, :] * stride_cj
    mask = (offs_m < M) & (offs_n < N)
    tl.store(c_ptrs, acc, mask=mask)

"""
Optimized outgoing TriMul forward pass.

- LayerNorm and linear projections are performed with PyTorch.
- The core O(N^3) pairwise reduction, 
  out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d},
  is rewritten as a batched matrix multiplication 
  left_{i,k} @ right_{k,j} for each batch and hidden channel.
- This batched GEMM is implemented with a Triton kernel
  that processes many (batch * hidden) matrices in parallel,
  using block‑wise loads and accumulators. The kernel works
  for arbitrary sequence lengths up to 1024 and hidden
  dimensions up to 384. All tensors are kept in float32,
  and the final result is returned as float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(
    A, B, C,                     # pointers
    M, N, K,                     # matrix sizes
    stride_A_batch, stride_A_M, stride_A_K,
    stride_B_batch, stride_B_K, stride_B_N,
    stride_C_batch, stride_C_M, stride_C_N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Compute a batch of GEMMs:  C[b] = A[b] @ B[b] where
       A : (batch, M, K), B : (batch, K, N), C : (batch, M, N)."""
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k

        a_ptrs = A + pid_batch * stride_A_batch \
                  + offs_m[:, None] * stride_A_M \
                  + k_offset[None, :] * stride_A_K
        b_ptrs = B + pid_batch * stride_B_batch \
                  + k_offset[:, None] * stride_B_K \
                  + offs_n[None, :] * stride_B_N

        a = tl.load(a_ptrs,
                    mask=(offs_m < M) & (k_offset < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(k_offset < K) & (offs_n < N),
                    other=0.0)

        acc += tl.dot(a, b)

    c_ptrs = C + pid_batch * stride_C_batch \
                + offs_m[:, None] * stride_C_M \
                + offs_n[None, :] * stride_C_N
    mask = (offs_m < M) & (offs_n < N)
    tl.store(c_ptrs, acc, mask=mask)


def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched matrix multiplication using Triton.
    A: (B, M, K)
    B: (B, K, N)
    Returns C: (B, M, N)
    """
    assert A.is_contiguous() and B.is_contiguous()
    B_batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2

    C = torch.empty((B_batch, M, N), dtype=A.dtype, device=A.device)

    # strides in elements
    stride_A_batch, stride_A_M, stride_A_K = A.stride()
    stride_B_batch, stride_B_K, stride_B_N = B.stride()
    stride_C_batch, stride_C_M, stride_C_N = C.stride()

    # Tunable block sizes – work well for the target N ≤ 1024
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B_batch,
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_A_batch,
        stride_A_M,
        stride_A_K,
        stride_B_batch,
        stride_B_K,
        stride_B_N,
        stride_C_batch,
        stride_C_M,
        stride_C_N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module (AlphaFold3).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # same shape

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid) and gating of the projected tensors
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B, N, N, hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    Implemented as a batched GEMM: (i,k) @ (k,j) for each (b,d)
    # ------------------------------------------------------------------
    # Change layout to (B, hidden_dim, N, N) so that the hidden
    # dimension can be folded into the batch axis for the kernel.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # (B, H, N, N)

    # Transpose the last two axes of right so that we have (k, j)
    right_t = right_perm.transpose(-2, -1).contiguous()   # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden into one logical batch for the kernel
    left_flat = left_perm.view(B * H, N, N)
    right_flat = right_t.view(B * H, N, N)

    out_flat = batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Restore the original layout
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 6. Post‑processing (LayerNorm, output gate, final linear)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )  # [B, N, N, hidden_dim]

    # Apply the output gate
    out = out_norm * out_gate

    # Final linear projection back to `dim`
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out

"""
Optimized outgoing TriMul forward pass (AlphaFold3).

The module consists of:
1) LayerNorm on the input.
2) Two linear projections (left/right) and three gating linear layers.
3) An optional mask applied to the projected tensors.
4) The core O(N³) “triangle multiplicative update”:
       out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
   which we rewrite as a batched matrix multiplication
   (i,k) @ (k,j) for each hidden channel d.
   This GEMM is performed by a custom Triton kernel that processes many
   (batch × hidden) matrices in parallel using block‑wise loads, a
   float32 accumulator and tl.dot.  The kernel works for sequence lengths
   up to 1024 and hidden dimensions up to 384.
5) A second LayerNorm, an output gate, and a final linear projection.

All tensors remain on the GPU; the heavy N³ work is fully fused in Triton.
The final output dtype is float32.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    A, B, C,                # pointers to tensors
    M, N, K,                # matrix sizes
    stride_A_batch, stride_A_M, stride_A_K,
    stride_B_batch, stride_B_K, stride_B_N,
    stride_C_batch, stride_C_M, stride_C_N,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute C[b] = A[b] @ B[b] where
        A : (batch, M, K)
        B : (batch, K, N)
        C : (batch, M, N)
    The kernel is fully block‑wise and uses a float32 accumulator.
    """
    pid_batch = tl.program_id(0)          # which (batch * hidden) matrix
    pid_m = tl.program_id(1)              # block row
    pid_n = tl.program_id(2)              # block column

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension in tiles
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K + offs_k

        a_ptrs = A + pid_batch * stride_A_batch \
                  + offs_m[:, None] * stride_A_M \
                  + k_offset[None, :] * stride_A_K
        b_ptrs = B + pid_batch * stride_B_batch \
                  + k_offset[:, None] * stride_B_K \
                  + offs_n[None, :] * stride_B_N

        a = tl.load(a_ptrs,
                    mask=(offs_m < M) & (k_offset < K),
                    other=0.0)
        b = tl.load(b_ptrs,
                    mask=(k_offset < K) & (offs_n < N),
                    other=0.0)

        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N)

    # write result
    c_ptrs = C + pid_batch * stride_C_batch \
                + offs_m[:, None] * stride_C_M \
                + offs_n[None, :] * stride_C_N
    mask = (offs_m < M) & (offs_n < N)
    tl.store(c_ptrs, acc, mask=mask)


def batched_matmul(A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
    """
    Batched GEMM using the Triton kernel above.
    A : (B, M, K)   contiguous float32
    B : (B, K, N)   contiguous float32
    Returns C : (B, M, N)
    """
    assert A.is_contiguous() and B.is_contiguous()
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, "Inner dimensions must match."

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    # strides in element units (not bytes)
    stride_A_batch, stride_A_M, stride_A_K = A.stride()
    stride_B_batch, stride_B_K, stride_B_N = B.stride()
    stride_C_batch, stride_C_M, stride_C_N = C.stride()

    # block sizes – empirically good for the target workloads
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        batch,                              # batch*hidden dimension
        (M + BLOCK_M - 1) // BLOCK_M,       # rows
        (N + BLOCK_N - 1) // BLOCK_N,       # cols
    )
    batched_matmul_kernel[grid](
        A,
        B,
        C,
        M,
        N,
        K,
        stride_A_batch,
        stride_A_M,
        stride_A_K,
        stride_B_batch,
        stride_B_K,
        stride_B_N,
        stride_C_batch,
        stride_C_M,
        stride_C_N,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul (AlphaFold3) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : [B, N, N, dim]   (float32)
        - mask_tensor  : [B, N, N]       (bool or float)
        - weights_dict : {
            "norm.weight": Tensor(dim),
            "norm.bias":   Tensor(dim),
            "left_proj.weight": Tensor(hidden_dim, dim),
            "right_proj.weight": Tensor(hidden_dim, dim),
            "left_gate.weight": Tensor(hidden_dim, dim),
            "right_gate.weight": Tensor(hidden_dim, dim),
            "out_gate.weight": Tensor(hidden_dim, dim),
            "to_out_norm.weight": Tensor(hidden_dim),
            "to_out_norm.bias":   Tensor(hidden_dim),
            "to_out.weight": Tensor(dim, hidden_dim)
          }
        - config_dict : {"dim": int, "hidden_dim": int, "nomask": bool, ...}

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = F.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x_norm, left_proj_weight)    # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid) and apply gates
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul: out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
    #    This is a batch of GEMMs: (i,k) @ (k,j) for each (b, d)
    # ------------------------------------------------------------------
    # Rearrange tensors to (batch, hidden, N, N) for easy flattening.
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Transpose right on the last two axes so that we have (k, j).
    right_t = right_perm.transpose(-2, -1).contiguous()  # (B, H, N, N)

    B, H, N, _ = left_perm.shape
    # Collapse batch and hidden dims → logical batch for the kernel.
    left_flat = left_perm.view(B * H, N, N)   # (B*H, N, N)
    right_flat = right_t.view(B * H, N, N)    # (B*H, N, N)

    out_flat = batched_matmul(left_flat, right_flat)  # (B*H, N, N)

    # Restore original layout [B, N, N, hidden_dim]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6. Post‑processing: second LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )                       # [B, N, N, hidden_dim]

    out = out * out_gate      # apply output gate

    out = F.linear(out, to_out_weight)  # final projection back to `dim`
    # shape: [B, N, N, dim]

    return out
