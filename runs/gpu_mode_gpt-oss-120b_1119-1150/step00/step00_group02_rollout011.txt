```python
def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape
    
    # Convert to half
    x = input_tensor
    # layernorm: manual compute
    # We'll use torch.nn.functional.layer_norm using weight and bias
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # We'll compute normalized x:
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    
    # Cast to half
    x_norm_h = x_norm.to(torch.float16)
    
    # Cast weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    
    # Compute all linear transforms
    left = torch.nn.functional.linear(x_norm_h, left_proj_w)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm_h, right_proj_w)
    left_gate = torch.nn.functional.linear(x_norm_h, left_gate_w)
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_w)
    out_gate = torch.nn.functional.linear(x_norm_h, out_gate_w)
    
    # Convert mask to half and expand
    mask_h = mask.to(torch.float16)  # shape [B,N,N]
    # Launch kernel to apply mask, gating:
    left, right, out_gate = apply_mask_and_gate_kernel(left, right, left_gate, right_gate, out_gate, mask_h)
    # After kernel we have left, right gated and masked (float16). out_gate also sigmoid.

    # Reshape for batched matmul
    B, N, N, H = left.shape

    # Permute to (B*H,N,N)
    left_mat = left.permute(0,3,1,2).reshape(B*H, N, N)  # (B*H,N,N)
    right_mat = right.permute(0,3,1,2).reshape(B*H, N, N) # (B*H,N,N)
    # Compute batched matmul left @ right^T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1,2))  # shape (B*H,N,N)
    out = out_mat.view(B, H, N, N).permute(0,2,3,1).contiguous()  # shape [B,N,N,H]
    
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    
    # Multiply with out_gate (which is sigmoid-ed)
    out = out * out_gate.to(out.dtype) # ensure dtype matches
    # final linear: to_out weight (dim, hidden_dim)
    to_out_weight = weights["to_out.weight"].to(out.dtype) # shape (dim, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,dim)
    # Cast back to float32
    return out.to(torch.float32)
```

```
@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_outgate_b, stride_outgate_i, stride_outgate_j, stride_outgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # compute linear index within total size
    total = B * N * N * H
    mask = offs < total
    # compute indices
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # offsets for pointers (int64)
    left_offset = b*stride_left_b + i*stride_left_i + j*stride_left_j + h*stride_left_h
    right_offset = b*stride_right_b + i*stride_right_i + j*stride_right_j + h*stride_right_h
    lgate_offset = b*stride_lgate_b + i*stride_lgate_i + j*stride_lgate_j + h*stride_lgate_h
    rgate_offset = b*stride_rgate_b + i*stride_rgate_i + j*stride_rgate_j + h*stride_rgate_h
    outgate_offset = b*stride_outgate_b + i*stride_outgate_i + j*stride_outgate_j + h*stride_outgate_h
    mask_offset = b*stride_mask_b + i*stride_mask_i + j*stride_mask_j

    left_val = tl.load(left_ptr + left_offset, mask=mask)
    right_val = tl.load(right_ptr + right_offset, mask=mask)
    lgate_val = tl.load(left_gate_ptr + lgate_offset, mask=mask)
```

```python
import torch
import triton
import triton.language as tl
import math

@triton.jit
def mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_outgate_b, stride_outgate_i, stride_outgate_j, stride_outgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # linear index mapping to dimensions
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # compute memory offsets
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h
    right_offset = b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h
    lgate_offset = b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h
    rgate_offset = b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h
    outgate_offset = b * stride_outgate_b + i * stride_outgate_i + j * stride_outgate_j + h * stride_outgate_h
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # load
    left_val = tl.load(left_ptr + left_offset, mask=mask)
    right_val = tl.load(right_ptr + right_offset, mask=mask)
    lgate_val = tl.load(left_gate_ptr + lgate_offset, mask=mask)
    rgate_val = tl.load(right_gate_ptr + rgate_offset, mask=mask)
    outgate_val = tl.load(out_gate_ptr + outgate_offset, mask=mask)
    mask_val = tl.load(mask_ptr + mask_offset, mask=mask)

    # sigmoid implementation
    sig = lambda x: 1.0 / (1.0 + tl.exp(-x))

    left_gate_sig = sig(lgate_val)
    right_gate_sig = sig(rgate_val)
    out_gate_sig = sig(outgate_val)

    # apply mask and gates
    left_new = left_val * mask_val * left_gate_sig
    right_new = right_val * mask_val * right_gate_sig

    tl.store(left_ptr + left_offset, left_new, mask=mask)
    tl.store(right_ptr + right_offset, right_new, mask=mask)
    tl.store(out_gate_ptr + outgate_offset, out_gate_sig, mask=mask)

def apply_mask_and_gate(left, right, left_gate, right_gate, out_gate, mask):
    """
    Apply binary mask and sigmoid gates to left/right tensors.
    This kernel fuses three elementwise ops into a single pass.
    All tensors are expected to be torch.float16 on the same device.
    """
    B, N, N2, H = left.shape
    assert N == N2
    total_elements = B * N * N * H
    BLOCK_SIZE = 1024  # tuneable
    grid = ( (total_elements + BLOCK_SIZE - 1) // BLOCK_SIZE, )
    # Launch kernel
    mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate, out_gate,
        mask,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE
    )
    # No return needed; tensors are modified in-place.

def custom_kernel(data):
    """
    Custom forward kernel for the TriMul (outgoing) operator.
    The heavy pairwise reduction is performed with a batched GEMM (torch.bmm)
    while a Triton kernel fuses mask application and gating.
    """
    torch.cuda.empty_cache()
    input_tensor, mask, weights, config = data
    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Ensure inputs are on the same CUDA device
    device = input_tensor.device

    # LayerNorm on the input (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # float32

    # Cast to half for the remaining compute
    x_norm_h = x_norm.to(torch.float16)

    # Cast model weights to half
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16)
    to_out_w = weights["to_out.weight"].to(torch.float16)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm_h, left_proj_w)   # [B,N,N,H]
    right = torch.nn.functional.linear(x_norm_h, right_proj_w)
    left_gate = torch.nn.functional.linear(x_norm_h, left_gate_w)
    right_gate = torch.nn.functional.linear(x_norm_h, right_gate_w)
    out_gate = torch.nn.functional.linear(x_norm_h, out_gate_w)

    # Handle mask: if mask is None or all ones, make a float16 tensor of ones
    if mask is None:
        mask_h = torch.ones(x_norm.shape[:-1], dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(torch.float16)

    # Fuse mask and gates using Triton
    apply_mask_and_gate(left, right, left_gate, right_gate, out_gate, mask_h)

    # At this point left/right/out_gate are ready
    # Reshape for batched matrix multiplication: (B*H,N,N)
    B, N, _, H = left.shape
    left_mat = left.permute(0, 3, 1, 2).reshape(B * H, N, N)   # (B*H,N,N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N) # (B*H,N,N)

    # Compute batched GEMM: left @ right^T per hidden channel
    # Torch's bmm works on float16 with Tensor cores on H100.
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))  # (B*H,N,N)

    # Restore original layout [B,N,N,H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # [B,N,N,H]

    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply out gate (already sigmoid-ed)
    out = out * out_gate

    # Final linear projection back to dim
    out = torch.nn.functional.linear(out, to_out_w)  # (B,N,N,dim)

    # Return float32 tensor as final output
    return out.to(torch.float32)
```

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (`x`) using the provided LayerNorm weight/bias.
2. Cast the normalized tensor to FP16 (half‑precision) for the bulk of the work.
3. Perform the five linear projections (left, right, left_gate, right_gate,
   out_gate) with FP16 weights (no bias) using `torch.nn.functional.linear`.
4. Apply the pairwise mask and sigmoid gates in a single Triton kernel.
   The kernel reads:
        left   = left   * mask * sigmoid(left_gate)
        right  = right  * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)
   The mask is broadcast over the hidden dimension.
5. Compute the O(N³) pairwise reduction with a batched GEMM:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is implemented as a strided‑batched `torch.bmm` on tensors of shape
   (B·hidden_dim, N, N) → (B·hidden_dim, N, N).
6. Convert the result back to FP32, apply the second LayerNorm (`to_out_norm`),
   multiply by the `out_gate`, and finally map back to the original feature
   dimension with the `to_out` linear projection.
7. Return the result as a float‑32 tensor of shape [B, N, N, dim].

Only the mask‑and‑gate fusion is performed in Triton; the expensive
matrix‑multiplication uses cuBLAS (`torch.bmm`) which is already highly
optimized on H100 GPUs.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr, out_gate_ptr,
    mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_outgate_b, stride_outgate_i, stride_outgate_j, stride_outgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    """
    For each element (b,i,j,h) compute:
        left  = left  * mask * sigmoid(left_gate)
        right = right * mask * sigmoid(right_gate)
        out_gate = sigmoid(out_gate)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # 4‑D index from linear offset
    idx = offs
    b = idx // (N * N * H)
    rem = idx % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    j = rem2 // H
    h = rem2 % H

    # Compute memory offsets (in elements)
    left_off   = b * stride_left_b   + i * stride_left_i   + j * stride_left_j   + h * stride_left_h
    right_off  = b * stride_right_b  + i * stride_right_i  + j * stride_right_j  + h * stride_right_h
    lgate_off  = b * stride_lgate_b  + i * stride_lgate_i  + j * stride_lgate_j  + h * stride_lgate_h
    rgate_off  = b * stride_rgate_b  + i * stride_rgate_i  + j * stride_rgate_j  + h * stride_rgate_h
    outgate_off= b * stride_outgate_b+ i * stride_outgate_i+ j * stride_outgate_j+ h * stride_outgate_h
    mask_off   = b * stride_mask_b   + i * stride_mask_i   + j * stride_mask_j

    # Load values (masked loads avoid out‑of‑bounds)
    left_val   = tl.load(left_ptr   + left_off,   mask=mask)
    right_val  = tl.load(right_ptr  + right_off,  mask=mask)
    lgate_val  = tl.load(left_gate_ptr + lgate_off, mask=mask)
    rgate_val  = tl.load(right_gate_ptr + rgate_off, mask=mask)
    outgate_val= tl.load(out_gate_ptr + outgate_off, mask=mask)
    mask_val   = tl.load(mask_ptr + mask_off,   mask=mask)

    # Sigmoid (x -> 1/(1+exp(-x))) – works on half precision
    sig = lambda x: 1.0 / (1.0 + tl.exp(-x))

    lgate_sig  = sig(lgate_val)
    rgate_sig  = sig(rgate_val)
    outgate_sig= sig(outgate_val)

    # Apply mask and gates
    left_new   = left_val   * mask_val * lgate_sig
    right_new  = right_val  * mask_val * rgate_sig

    # Store results back in‑place
    tl.store(left_ptr   + left_off,   left_new,   mask=mask)
    tl.store(right_ptr  + right_off,  right_new,  mask=mask)
    tl.store(out_gate_ptr + outgate_off, outgate_sig, mask=mask)


def _apply_mask_and_gate(left, right, left_gate, right_gate, out_gate, mask):
    """Launch the Triton kernel that fuses mask & sigmoid gating."""
    B, N, _, H = left.shape
    total_elems = B * N * N * H
    BLOCK = 1024  # a good trade‑off for H100
    grid = ( (total_elems + BLOCK - 1) // BLOCK, )

    _mask_gate_kernel[grid](
        left, right,
        left_gate, right_gate, out_gate,
        mask,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        right.stride(0), right.stride(1), right.stride(2), right.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        right_gate.stride(0), right_gate.stride(1), right_gate.stride(2), right_gate.stride(3),
        out_gate.stride(0), out_gate.stride(1), out_gate.stride(2), out_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK
    )
    # In‑place modification – nothing to return


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, dim] (float32)
        - mask         : torch.Tensor, shape [B, N, N] (bool/float)
        - weights      : dict with the model parameters (float32)
        - config       : dict with keys "dim", "hidden_dim"
    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    torch.cuda.empty_cache()  # clean any stale allocations

    # Unpack inputs
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    device = input_tensor.device

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]  float32

    # ------------------------------------------------------------------
    # 2. Cast to FP16 for all heavy arithmetic
    # ------------------------------------------------------------------
    x_h = x_norm.to(torch.float16)

    # FP16 weights
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)
    to_out_norm_w = weights["to_out_norm.weight"]   # keep FP32 for final LN
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]        # keep FP32 for final projection

    # ------------------------------------------------------------------
    # 3. Linear projections (no bias)
    # ------------------------------------------------------------------
    left      = torch.nn.functional.linear(x_h, left_proj_w)      # [B,N,N,hidden_dim]
    right     = torch.nn.functional.linear(x_h, right_proj_w)
    left_gate = torch.nn.functional.linear(x_h, left_gate_w)
    right_gate= torch.nn.functional.linear(x_h, right_gate_w)
    out_gate  = torch.nn.functional.linear(x_h, out_gate_w)

    # ------------------------------------------------------------------
    # 4. Mask handling (convert to FP16)
    # ------------------------------------------------------------------
    if mask is None:
        mask_h = torch.ones(x_norm.shape[:-1], dtype=torch.float16, device=device)
    else:
        mask_h = mask.to(torch.float16)

    # ------------------------------------------------------------------
    # 5. Fuse mask & sigmoid gating with Triton
    # ------------------------------------------------------------------
    _apply_mask_and_gate(left, right, left_gate, right_gate, out_gate, mask_h)

    # ``left_gate`` and ``right_gate`` are no longer needed; let Python free them.
    del left_gate, right_gate

    # ------------------------------------------------------------------
    # 6. Batched GEMM:   out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Equivalent to (B·hidden_dim) separate matmuls of size (N×N)
    # ------------------------------------------------------------------
    B, N, _, H = left.shape
    # reshape to (B·H, N, N) for torch.bmm
    left_mat  = left.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_mat = right.permute(0, 3, 1, 2).reshape(B * H, N, N)

    # batched matrix multiplication (FP16 → fast Tensor‑Core path)
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B·H, N, N)

    # Back‑to [B, N, N, hidden_dim] layout
    out_half = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 7. Remaining post‑processing (keep in FP32 for numerical stability)
    # ------------------------------------------------------------------
    out = out_half.to(torch.float32)               # cast back to FP32
    # LayerNorm on hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )
    # Apply the output gate (sigmoid already computed in the kernel)
    out = out * out_gate.to(torch.float32)

    # Final linear projection back to `dim`
    out = torch.nn.functional.linear(out, to_out_w)  # [B,N,N,dim]  float32

    return out
