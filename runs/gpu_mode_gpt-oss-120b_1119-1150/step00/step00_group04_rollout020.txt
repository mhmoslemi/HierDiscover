    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    H: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    offs_i = i_start + tl.arange(0, BLOCK_I)
    offs_j = j_start + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_I, BLOCK_J, H), dtype=tl.float32)

        # compute offsets for left

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The operation works on a 4‑D tensor x ∈ ℝ^{B×N×N×C} :

    1. Layer‑norm over the last dimension (C).
    2. Linear projections → left, right ∈ ℝ^{B×N×N×H}.
    3. Optional binary mask applied to left/right.
    4. Gating (sigmoid(linear(x))) applied element‑wise.
    5. Core “TriMul”:
           out[b,i,j,d] = Σ_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
       This is a batch of H independent N×N matrix multiplications.
       The reduction over k is performed by a Triton kernel; each
       thread computes one (i,j) output vector of length H.
    6. Layer‑norm over the hidden dimension, gating, final linear
       projection back to C.

Only the forward pass is required, and the final tensor is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,          # const * left  [B,N,N,H]
    right_ptr,         # const * right [B,N,N,H]
    out_ptr,           # * out [B,N,N,H]   (float32 accumulator)
    B, N, H,           # tensor dimensions (runtime constants)
    # strides for left
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    # strides for right
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    # strides for out
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    H_CONST: tl.constexpr,   # compile‑time hidden dimension (== H)
):
    """Compute out[b,i,j,:] = Σ_k left[b,i,k,:] * right[b,j,k,:].

    Each program instance (thread) is responsible for a single (i,j)
    pair of a given batch.  The accumulation is done in float32 while
    the inputs are read as float32 (they could be half‑precision as
    well).  The outer loops are compiled because B, N, H are passed
    as compile‑time constants when the kernel is launched.
    """
    pid_b = tl.program_id(2)   # batch dimension
    pid_i = tl.program_id(0)   # i dimension
    pid_j = tl.program_id(1)   # j dimension

    i = pid_i
    j = pid_j

    # Guard against over‑launch (grid may be larger than N)
    mask_i = i < N
    mask_j = j < N

    # Accumulator for the hidden dimension
    acc = tl.zeros([H_CONST], dtype=tl.float32)

    # -----------------------------------------------------------------
    # Loop over the k dimension (the “pairwise” axis)
    # -----------------------------------------------------------------
    for k in range(0, N):
        # Compute linear offsets inside the contiguous 4‑D tensors
        left_offset = pid_b * stride_left_b + i * stride_left_i + k * stride_left_k
        right_offset = pid_b * stride_right_b + j * stride_right_i + k * stride_right_k

        # Load the hidden‑dim vector for this (i,k) and (j,k)
        # tl.arange produces the offset for the H dimension
        h_idx = tl.arange(0, H_CONST)          # [0 .. H-1]

        # left_vec[x] = left[b, i, k, x]
        left_vec = tl.load(
            left_ptr + left_offset + h_idx * stride_left_h,
            mask=mask_i, other=0.0
        )
        # right_vec[x] = right[b, j, k, x]
        right_vec = tl.load(
            right_ptr + right_offset + h_idx * stride_right_h,
            mask=mask_j, other=0.0
        )

        # Element‑wise product and accumulate
        acc += left_vec * right_vec

    # -----------------------------------------------------------------
    # Write the result back to memory
    # -----------------------------------------------------------------
    out_offset = pid_b * stride_out_b + i * stride_out_i + j * stride_out_j
    out_ptrs = out_ptr + out_offset + h_idx * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i & mask_j)


def _tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Launch the Triton kernel that implements the core TriMul reduction.
    left / right   : [B, N, N, H]  (float32 or float16)
    returns out    : [B, N, N, H]  (float32)
    """
    assert left.shape == right.shape
    B, N, N2, H = left.shape
    assert N == N2

    # Prepare the output tensor (float32 accumulator)
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)

    # Strides in element units (not bytes)
    s_bl, s_il, s_kl, s_hl = left.stride()
    s_br, s_ir, s_kr, s_hr = right.stride()
    s_bo, s_io, s_jo, s_ho = out.stride()

    # Launch configuration: one thread per (i, j) pair
    grid = (N, N, B)   # (i tiles, j tiles, batch)

    tri_mul_outgoing_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        H,
        # left strides
        s_bl,
        s_il,
        s_kl,
        s_hl,
        # right strides
        s_br,
        s_ir,
        s_kr,
        s_hr,
        # out strides
        s_bo,
        s_io,
        s_jo,
        s_ho,
        H_CONST=H,          # compile‑time hidden dimension
        num_warps=4,        # typical choice for H100
    )
    return out


def custom_kernel(data):
    """
    Entry point used by the evaluation harness.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    # -----------------------------------------------------------------
    # Extract configuration
    # -----------------------------------------------------------------
    dim = config["dim"]               # input feature dimension C
    hidden_dim = config["hidden_dim"] # projection dimension H
    use_mask = not config.get("nomask", True)

    # -----------------------------------------------------------------
    # Layer‑norm over the input feature dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # -----------------------------------------------------------------
    # Linear projections (no bias) → left / right
    # -----------------------------------------------------------------
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Optional binary mask (shape [B,N,N]) – broadcast to hidden dim
    # -----------------------------------------------------------------
    if use_mask:
        # ensure mask is float (0.0 / 1.0)
        mask = mask.to(dtype=x.dtype, device=x.device)
        mask = mask.unsqueeze(-1)                # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # Gating layers (sigmoid(linear(x)))
    # -----------------------------------------------------------------
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Core TriMul reduction – executed on the GPU via Triton
    # -----------------------------------------------------------------
    # Keep tensors contiguous for efficient pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = _tri_mul_triton(left, right)   # float32 output, shape [B,N,N,hidden_dim]

    # -----------------------------------------------------------------
    # Output normalisation, gating and final linear projection
    # -----------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # Layer‑norm over hidden dimension
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )  # [B,N,N,hidden_dim]

    # Multiply by the output gate
    out = out * out_gate

    # Final linear back to original dimension
    out = F.linear(out, to_out_w)   # [B,N,N,dim]

    return out
