def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    N, D,
    stride_li, stride_lk, stride_ld,
    stride_ri, stride_rk, stride_rd,
    stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    d0 = pid_d * BLOCK_D

    # Masks for i,j,d
    i_mask = i0 + tl.arange(0, BLOCK_I) < N
    j_mask = j0 + tl.arange(0, BLOCK_J) < N
    d_mask = d0 + tl.arange(0, BLOCK_D) < D

    # Ranges
    i_range = i0 + tl.arange(0, BLOCK_I)
    j_range = j0 + tl.arange(0, BLOCK_J)
    d_range = d0 + tl.arange(0, BLOCK_D)

    # Accumulator matches (BLOCK_I, BLOCK_J, BLOCK_D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k_block in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k_block * BLOCK_K
        k_range = k0 + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # Load left block (i,k,d)
        left_ptrs = left_ptr + i_range[:, None, None] * stride_li + k_range[None, :, None] * stride_lk + d_range[None, None, :] * stride_ld
        left = tl.load(left_ptrs, mask=i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # Load right block transposed (k,j,d) to make dot possible
        right_ptrs = right_ptr + k_range[:, None, None] * stride_rk + j_range[None, :, None] * stride_ri + d_range[None, None, :] * stride_rd
        right = tl.load(right_ptrs, mask=k_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :], other=0.0)

        # For each channel in the block, compute dot.
        # We'll use a loop over d_idx.
        for d_idx in range(BLOCK_D):
            if not (d_mask[d_idx]):  # skip if out of bounds
                continue
            left_slice = left[:, :, d_idx]   # (I, K)
            right_slice = right[:, :, d_idx] # (K, J)
            # Compute matmul
            prod = tl.dot(left_slice, right_slice) # (I, J)
            acc[:, :, d_idx] += prod

    # Store result
    out_ptrs = out_ptr + i_range[:, None, None] * stride_oi + j_range[None, :, None] * stride_oj + d_range[None, None, :] * stride_od
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

def custom_kernel(data):
    # unpack
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # x shape [B, N, N, dim]
    B, N, N2, _ = x.shape
    assert N == N2

    # LayerNorm on dim, using weight and bias
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Mask
    if mask is not None:
        mask_f = mask.unsqueeze(-1).to(x.dtype)
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for kernel
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Prepare output buffer in half precision to match kernel dtype then cast back to float32 for remaining steps
    out_h = torch.empty_like(left_h)  # shape same as left

    # Get strides for left, right, out (elements)
    stride_lb = left_h.stride(0)
    stride_li = left_h.stride(1)
    stride_lk = left_h.stride(2)
    stride_ld = left_h.stride(3)
    stride_rb = right_h.stride(0)
    stride_ri = right_h.stride(1)
    stride_rk = right_h.stride(2)
    stride_rd = right_h.stride(3)
    stride_ob = out_h.stride(0)
    stride_oi = out_h.stride(1)
    stride_oj = out_h.stride(2)
    stride_od = out_h.stride(3)

    # Kernel launch grid
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = ( (N + BLOCK_I - 1)//BLOCK_I,
             (N + BLOCK_J - 1)//BLOCK_J,
             hidden_dim )   # each program for each channel

    trimul_outgoing_kernel[grid](
        left_h.data_ptr(),
        right_h.data_ptr(),
        out_h.data_ptr(),
        B,
        N,
        hidden_dim,
        stride_lb, stride_li, stride_lk, stride_ld,
        stride_rb, stride_ri, stride_rk, stride_rd,
        stride_ob, stride_oi, stride_oj, stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
        # maybe set REPEAT for unrolling? Not needed.
    )
    
    # Convert out back to float32
    out = out_h.to(torch.float32)

    # to_out_norm: LayerNorm on hidden_dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # out_gate (already computed earlier) multiply: shape [B,N,N,hidden_dim]
    out = out * out_gate

    # final linear projection to original dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # bias=False

    return out

import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_ri, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)  # channel index
    # Guard: ensure pid_d < D
    # Compute offsets
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    d = pid_d  # scalar channel

    # Masks for dimensions
    i_mask = i0 + tl.arange(0, BLOCK_I) < N
    j_mask = j0 + tl.arange(0, BLOCK_J) < N

    i_range = i0 + tl.arange(0, BLOCK_I)
    j_range = j0 + tl.arange(0, BLOCK_J)

    # Loop over batch dimension
    for b in range(B):
        # base pointers for this batch
        left_base = left_ptr + b * stride_lb
        right_base = right_ptr + b * stride_rb
        out_base = out_ptr + b * stride_ob

        # accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # Loop over k dimension
        for k_block in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = k_block * BLOCK_K
            k_range = k0 + tl.arange(0, BLOCK_K)
            k_mask = k_range < N

            # Load left block
            left_ptrs = left_base + i_range[:, None] * stride_li + k_range[None, :] * stride_lk + d * stride_ld
            left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)
            # Cast to float32 if needed
            left = left.to(tl.float32)

            # Load right block transposed (K,J)
            right_ptrs = right_base + k_range[:, None] * stride_rk + j_range[None, :] * stride_ri + d * stride_rd
            right = tl.load(right_ptrs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)
            right = right.to(tl.float32)

            # Dot product across K dimension.
            acc += tl.dot(left, right)  # (I,J)

        # Store output
        out_ptrs = out_base + i_range[:, None] * stride_oi + j_range[None, :] * stride_oj + d * stride_od
        tl.store(out_ptrs, acc, mask=i_mask[:, None] & j_mask[None, :])

"""
TriMul “outgoing’’ implementation with a fused Triton kernel.

The heavy contraction
    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]
is computed on‑GPU by a custom Triton kernel.  The kernel tiles the
(i, j) plane (BLOCK_I×BLOCK_J) and processes each hidden channel `d`
independently (one program per channel).  For every tile it iterates
over k in blocks of size BLOCK_K, loads the required slices of `left`
and `right` (the latter transposed on‑the‑fly), performs a matrix‑multiply
via `tl.dot`, and accumulates the partial result in FP32.  Batch size is
≤2, so the kernel loops over the batch dimension inside the kernel.

All other operations (LayerNorm, Linear projections, gating, mask
application, final LayerNorm and projection) are performed with
standard PyTorch functional calls.  The kernel works for any
batch size B∈{1,2}, sequence length N up to 1024 and hidden dimension
C up to 384 (the config may also request `nomask=True`, in which case the
mask is simply skipped).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,          # *const half*  [B,N,N,D]
    right_ptr,         # *const half*  [B,N,N,D]
    out_ptr,           # *mut   half*  [B,N,N,D]
    B,                 # batch size (runtime)
    N,                 # sequence length
    D,                 # hidden dimension
    stride_lb, stride_li, stride_lk, stride_ld,   # left strides
    stride_rb, stride_ri, stride_rk, stride_rd,   # right strides
    stride_ob, stride_oi, stride_oj, stride_od,   # out strides
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes the pairwise multiplicative update:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    Parameters are passed as element‑wise strides (not byte strides).
    The kernel iterates over the batch dimension internally (B ≤ 2).
    """
    pid_i = tl.program_id(0)                     # i‑tile
    pid_j = tl.program_id(1)                     # j‑tile
    pid_d = tl.program_id(2)                     # hidden channel (scalar)

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    d  = pid_d                                    # scalar channel index

    # -----------------------------------------------------------------
    # Masks for boundary handling
    i_mask = i0 + tl.arange(0, BLOCK_I) < N
    j_mask = j0 + tl.arange(0, BLOCK_J) < N
    d_mask = d < D                                 # always true for valid launch

    i_range = i0 + tl.arange(0, BLOCK_I)
    j_range = j0 + tl.arange(0, BLOCK_J)

    # -----------------------------------------------------------------
    # Loop over each batch element (B ≤ 2)
    for b in range(B):
        # Base pointers for this batch
        left_base  = left_ptr  + b * stride_lb
        right_base = right_ptr + b * stride_rb
        out_base   = out_ptr   + b * stride_ob

        # Accumulator for the (i,j) tile of this channel
        acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

        # -----------------------------------------------------------------
        # Loop over k in blocks
        for k_block in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = k_block * BLOCK_K
            k_range = k0 + tl.arange(0, BLOCK_K)
            k_mask = k_range < N

            # ---------------------------------------------------------
            # Load left: shape (BLOCK_I, BLOCK_K)
            left_offset = (
                left_base
                + i_range[:, None] * stride_li
                + k_range[None, :] * stride_lk
                + d * stride_ld
            )
            left = tl.load(
                left_offset,
                mask=i_mask[:, None] & k_mask[None, :],
                other=0.0,
            )
            # Cast to FP32 for accumulation
            left = left.to(tl.float32)

            # ---------------------------------------------------------
            # Load right transposed: shape (BLOCK_K, BLOCK_J)
            # (k is first dimension, j second) → ready for tl.dot
            right_offset = (
                right_base
                + k_range[:, None] * stride_rk
                + j_range[None, :] * stride_ri
                + d * stride_rd
            )
            right = tl.load(
                right_offset,
                mask=k_mask[:, None] & j_mask[None, :],
                other=0.0,
            )
            right = right.to(tl.float32)

            # ---------------------------------------------------------
            # Matrix‑multiply the two blocks and accumulate
            acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J)

        # -----------------------------------------------------------------
        # Store the completed tile
        out_offset = (
            out_base
            + i_range[:, None] * stride_oi
            + j_range[None, :] * stride_oj
            + d * stride_od
        )
        tl.store(out_offset, acc, mask=i_mask[:, None] & j_mask[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing Triangle‑Multiplicative‑Update (TriMul).

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weight_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] – the TriMul output.
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]                # input/ output channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension

    B, N, N2, _ = input_tensor.shape
    assert N == N2, "Input must be square in the pairwise dimensions."

    # -----------------------------------------------------------------
    # 1. LayerNorm on the input (per position pair)
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (mask may be None when nomask=True)
    if mask_tensor is not None:
        mask = mask_tensor.unsqueeze(-1).to(x.dtype)   # [B,N,N,1]
        left = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gating scalars (sigmoid of linear transforms)
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Triton kernel for the O(N³·C) contraction
    # Cast to half‑precision for bandwidth; the kernel accumulates in FP32.
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty_like(left_h)                      # [B,N,N,hidden_dim]

    # Strides in element units (torch provides element strides)
    stride_lb, stride_li, stride_lk, stride_ld = left_h.stride()
    stride_rb, stride_ri, stride_rk, stride_rd = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_od = out_h.stride()

    # Tile sizes – tuned for H100 (fits into shared memory and registers)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    # Grid: one program per (i‑tile, j‑tile, hidden channel)
    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        hidden_dim,                     # one program per channel d
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=8,                    # H100 works well with 8 warps
    )

    # Convert back to FP32 for the remaining layers
    out = out_h.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Post‑projection layer‑norm and gating
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # -----------------------------------------------------------------
    # 7. Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
