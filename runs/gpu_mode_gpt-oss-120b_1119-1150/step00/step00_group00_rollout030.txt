"""
TriMul “outgoing” implementation (AlphaFold3 style).

The operation consists of:
1. Layer‑Norm over the feature dimension.
2. Linear projections (left / right) + per‑pair mask.
3. Sigmoid gates (left / right / out) applied to the projected tensors.
4. Core multiplicative update:
       out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
   which is a batched GEMM: out = left @ rightᵀ for each hidden channel.
   This heavy reduction is performed by a custom Triton kernel.
5. Layer‑Norm over the hidden dimension, out‑gate multiplication and final
   linear projection back to the original feature dimension.

The kernel works on tensors of shape
   left  : [G, M, K]   (G = batch * hidden_dim, M = N, K = N)
   right : [G, K, N]   (right is already transposed on the inner pair‑wise axes)
   out   : [G, M, N]

The kernel loops over K in blocks and uses `tl.dot` for the inner
multiply‑accumulate.  All memory accesses are fully vectorised and the
accumulator is float32 while the inputs are stored in float16 to reduce
bandwidth.  The final result is cast back to float32 and reshaped to
[batch, N, N, hidden_dim] for the remaining post‑processing in PyTorch.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the TriMul core
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_gemm_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # problem size
    M, N, K,
    # strides for A (shape: [G, M, K])
    stride_a_g, stride_a_m, stride_a_k,
    # strides for B (shape: [G, K, N])
    stride_b_g, stride_b_k, stride_b_n,
    # strides for C (shape: [G, M, N])
    stride_c_g, stride_c_m, stride_c_n,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile col
    pid_g = tl.program_id(2)          # group = batch * hidden_dim

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for the output tile (out‑of‑bounds rows/cols)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Pointers to the beginning of the current group (batch * hidden)
    # ------------------------------------------------------------------
    a_base = a_ptr + pid_g * stride_a_g
    b_base = b_ptr + pid_g * stride_b_g
    c_base = c_ptr + pid_g * stride_c_g

    # ------------------------------------------------------------------
    # Accumulator
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Main K‑loop
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(0, num_k_blocks):
        # Global K indices for this block
        k_offset = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < K

        # Load a tile of shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_base
            + offs_m[:, None] * stride_a_m                # row offset
            + k_offset[None, :] * stride_a_k,              # column offset
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )   # half‑precision (float16) because a_ptr points to fp16 data

        # Load a tile of shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_base
            + k_offset[:, None] * stride_b_k                # row offset (k dimension)
            + offs_n[None, :] * stride_b_n,                # column offset (j dimension)
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )   # half‑precision

        # Multiply‑accumulate (inner product over K)
        acc += tl.cast(tl.dot(a, b), tl.float32)

    # ------------------------------------------------------------------
    # Write out the result
    # ------------------------------------------------------------------
    c = c_base \
        + offs_m[:, None] * stride_c_m \
        + offs_n[None, :] * stride_c_n
    tl.store(c, acc, mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Entrypoint called from the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape [batch, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, N, _, _ = input_tensor.shape

    # ------------------------------------------------------------------
    # Layer‑norm over the feature dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Apply optional mask (mask shape: [B, N, N])
    # ------------------------------------------------------------------
    if config.get("nomask", False):
        mask = torch.ones_like(left[..., :1], device=device, dtype=left.dtype)
    else:
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)          # [B, N, N, 1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating (sigmoid) – also no bias
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Convert to half‑precision for the kernel (memory/bandwidth saving)
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Prepare tensors for the batched GEMM:
    #   left  -> [B, H, N, N]   (B, hidden, i, k)
    #   right -> [B, H, N, N]   (B, hidden, k, j)   (already transposed)
    #   flatten to a single group dimension G = B * H
    # ------------------------------------------------------------------
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N] (i, k)
    right = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N] (k, j)

    G = bs * hidden_dim
    left_flat = left.view(G, N, N)     # [G, M, K]
    right_flat = right.view(G, N, N)   # [G, K, N]

    # Output buffer – float32 accumulator
    out_flat = torch.empty((G, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Kernel launch parameters
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # tiles along M (i)
        (N + BLOCK_N - 1) // BLOCK_N,   # tiles along N (j)
        G,                              # groups = batch * hidden_dim
    )

    # ------------------------------------------------------------------
    # Invoke Triton kernel
    # ------------------------------------------------------------------
    tri_mul_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,            # M
        N,            # N
        N,            # K
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1)  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Post‑processing: hidden‑dim layer‑norm, out‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )

    # Apply the out‑gate (shape already matches)
    out = out * out_gate

    # Final linear projection back to `dim`
    output = torch.nn.functional.linear(out, weights["to_out.weight"])  # bias is None

    return output
