"""
TriMul – outgoing triangle multiplicative update (AlphaFold3)

This implementation performs the forward pass of the outgoing TriMul
operation using a custom Triton kernel for the expensive 4‑D einsum:

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

The kernel treats each pair (batch, hidden) as an independent
N×N matrix multiplication (left @ rightᵀ) and computes all tiles in
FP16 with Tensor‑Core accumulation (FP32).  The surrounding linear
projections, LayerNorms and sigmoids are handled with standard PyTorch
operations.  The API matches the required `custom_kernel` signature.

Supported configs:
    B ∈ {1,2}, N ∈ {128,256,512,1024}, dim ∈ {128,384},
    hidden_dim ∈ {128,384,768}
Both masked and unmasked inputs are handled.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batch (B * hidden_dim) mat‑muls of shape N×N @ (N×N)ᵀ
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,               # *fp16  left  [B,N,N,H]
    right_ptr,              # *fp16  right [B,N,N,H]
    out_ptr,                # *fp16  out   [B,N,N,H]
    B, N, H,                # runtime scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Each program instance works on one (batch, hidden) pair
    and computes a BLOCK_M×BLOCK_N tile of the result matrix.
    """
    batch_hidden = tl.program_id(0)   # 0‑th axis: B*H different GEMMs
    pid_m = tl.program_id(1)          # tile row
    pid_n = tl.program_id(2)          # tile col

    # ------------------------------------------------------------------- #
    # Decode (batch, hidden) index
    # ------------------------------------------------------------------- #
    b = batch_hidden // H
    h = batch_hidden % H

    # ------------------------------------------------------------------- #
    # Starting coordinates of the current tile
    # ------------------------------------------------------------------- #
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # ------------------------------------------------------------------- #
    # Helper ranges and bounds
    # ------------------------------------------------------------------- #
    i_range = i_start + tl.arange(0, BLOCK_M)
    j_range = j_start + tl.arange(0, BLOCK_N)
    k_range = tl.arange(0, BLOCK_K)

    i_valid = i_range < N
    j_valid = j_range < N

    # ------------------------------------------------------------------- #
    # Base pointers that include batch and hidden offsets
    # ------------------------------------------------------------------- #
    base_l = b * stride_l_b + h * stride_l_h
    base_r = b * stride_r_b + h * stride_r_h
    base_o = b * stride_o_b + h * stride_o_h

    # ------------------------------------------------------------------- #
    # Accumulator (FP32 for precision)
    # ------------------------------------------------------------------- #
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # Loop over the reduction dimension k
    # ------------------------------------------------------------------- #
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_start = kb * BLOCK_K
        k_off = k_start + k_range
        k_valid = k_off < N

        # ----- Load a BLOCK_M×BLOCK_K tile from left (row‑major) -----
        a_ptrs = left_ptr + base_l \
                 + i_range[:, None] * stride_l_i \
                 + k_off[None, :] * stride_l_k
        a = tl.load(a_ptrs,
                    mask=i_valid[:, None] & k_valid[None, :],
                    other=0.0).to(tl.float32)

        # ----- Load a BLOCK_K×BLOCK_N tile from right (TRANSPOSED) -----
        #   right is accessed as (k, j) so that we form rightᵀ implicitly.
        b_ptrs = right_ptr + base_r \
                 + k_off[:, None] * stride_r_k \
                 + j_range[None, :] * stride_r_i
        b = tl.load(b_ptrs,
                    mask=k_valid[:, None] & j_valid[None, :],
                    other=0.0).to(tl.float32)

        # ----- Matrix‑multiply the two tiles -----
        acc += tl.dot(a, b)          # (M×K)·(K×N) → (M×N)

    # ------------------------------------------------------------------- #
    # Write the result tile back to global memory (FP16)
    # ------------------------------------------------------------------- #
    c_ptrs = out_ptr + base_o \
             + i_range[:, None] * stride_o_i \
             + j_range[None, :] * stride_o_j
    tl.store(c_ptrs, acc.to(tl.float16),
             mask=i_valid[:, None] & j_valid[None, :])

# --------------------------------------------------------------------------- #
# Entry‑point: custom_kernel
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            input_tensor : torch.Tensor [B, N, N, dim]
            mask         : torch.Tensor [B, N, N] or None
            weights      : dict of torch.Tensor containing all model weights
            config       : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # Unpack weights
    # ------------------------------------------------------------------- #
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=norm_weight,
                     bias=norm_bias)                     # [B,N,N,dim] FP32

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------- #
    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim] FP32
    right = F.linear(x, right_proj_weight)  # same shape

    # ------------------------------------------------------------------- #
    # 3) Optional mask (mask is float, shape [B,N,N])
    # ------------------------------------------------------------------- #
    if mask is not None and not config.get("nomask", False):
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 4) Gating (sigmoid)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left  * left_gate   # [B,N,N,hidden_dim] FP32
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 5) Convert to FP16 and launch Triton kernel for the Einstein sum
    # ------------------------------------------------------------------- #
    left_f16  = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    out_f16   = torch.empty_like(left_f16)  # will hold [B,N,N,hidden_dim] FP16

    B, N, _, H = left_f16.shape   # B, N, N, hidden_dim

    # Strides (in elements) for the three tensors
    stride_l_b, stride_l_i, stride_l_k, stride_l_h = left_f16.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_h = right_f16.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_h = out_f16.stride()

    # Tunable tile sizes – reasonable defaults for H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        B * H,                                 # batch * hidden_dim
        (N + BLOCK_M - 1) // BLOCK_M,          # tiles along i
        (N + BLOCK_N - 1) // BLOCK_N,          # tiles along j
    )

    tri_mul_outgoing_kernel[grid](
        left_f16, right_f16, out_f16,
        B, N, H,
        stride_l_b, stride_l_i, stride_l_k, stride_l_h,
        stride_r_b, stride_r_i, stride_r_k, stride_r_h,
        stride_o_b, stride_o_i, stride_o_j, stride_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 works well with 8 warps
        num_stages=2,
    )

    # Result back to FP32 for the remaining layers
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------- #
    # 6) Post‑multiplicative LayerNorm, gating and final projection
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)          # [B,N,N,hidden_dim] FP32

    out = out * out_gate                                 # apply per‑channel gate

    out = F.linear(out, to_out_weight)                   # back to original dim

    return out
