@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Determine batch id and block id.
    batch = tl.program_id(0) // (tl.cdiv(M,BLOCK_M) * tl.cdiv(N,BLOCK_N))
    pid = tl.program_id(0) % (tl.cdiv(M,BLOCK_M) * tl.cdiv(N,BLOCK_N))
    pid_m = pid // tl.cdiv(N,BLOCK_N)
    pid_n = pid % tl.cdiv(N,BLOCK_N)

    # Compute offsets for a and b (and c)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # stride for block in M dimension
    a_ptrs = a_ptr + batch * stride_ah + offs_m[:,None] * stride_am + (tl.arange(0, BLOCK_K)[None,:]) * stride_ak
    # b matrix is transposed: shape (K,N); we pass original shape but swapped strides
    b_ptrs = b_ptr + batch * stride_bh + (tl.arange(0, BLOCK_K)[:,None]) * stride_bk + offs_n[None,:] * stride_bn

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k blocks
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a and b
        a = tl.load(a_ptrs, mask=offs_m < M and (k*BLOCK_K + tl.arange(0,BLOCK_K)) < K, other=0.0)
        b = tl.load(b_ptrs, mask= (k*BLOCK_K + tl.arange(0,BLOCK_K))[:,None] < K and offs_n < N, other=0.0)  # but mask building tricky
        # Actually we need to compute k offsets:
        k_offset = k * BLOCK_K
        a_ptrs = a_ptr + batch * stride_ah + offs_m[:,None] * stride_am + (k_offset + tl.arange(0, BLOCK_K))[None,:] * stride_ak
        b_ptrs = b_ptr + batch * stride_bh + (k_offset + tl.arange(0, BLOCK_K))[:,None] * stride_bk + offs_n[None,:] * stride_bn

        a = tl.load(a_ptrs, mask=(offs_m < M)[:,None] & ((k_offset + tl.arange(0, BLOCK_K)) < K)[None,:], other=0.0)
        b = tl.load(b_ptrs, mask=((k_offset + tl.arange(0, BLOCK_K)) < K)[:,None] & (offs_n < N)[None,:], other=0.0)

        # accumulate
        acc += tl.dot(a, b)   # this expects a shape (BLOCK_M, BLOCK_K) and b shape (BLOCK_K, BLOCK_N)
    # store
    # compute final offsets for output
    c_ptrs = c_ptr + batch * stride_ch + offs_m[:,None] * stride_cm + offs_n[None,:] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16) if ... else acc)  # store as float16 to keep memory small; but later we want float32.

    device = input_tensor.device

    hidden_dim = config["hidden_dim"]

    # Project

    right = F.linear(x, weights["right_proj.weight"])

        # mask may be bool or float; convert to float

        left = left * mask_f
        right = right * mask_f

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"])) # shape B,N,N,hidden_dim
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    right = right * right_gate

    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # permute to (B, hidden_dim, N, N)
    left_perm = left_h.permute(0,3,1,2).contiguous()
    right_perm = right_h.permute(0,3,1,2).contiguous()

@triton.jit
def batched_matmul_kernel(a_ptr, b_ptr, c_ptr,
                          stride_ah, stride_am, stride_ak,
                          stride_bh, stride_bk, stride_bn,
                          stride_ch, stride_cm, stride_cn,
                          M, N, K,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    # Program ID should be a flat index across batch * grid_m * grid_n
    pid = tl.program_id(0)
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Compute k offsets across loop
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # Offsets for A
        a_ptrs = a_ptr + batch_id * stride_ah \
                 + (offs_m[:, None]) * stride_am \
                 + (k_offset + tl.arange(0, BLOCK_K))[None, :] * stride_ak
        # Offsets for B (transposed)
        b_ptrs = b_ptr + batch_id * stride_bh \
                 + (k_offset + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Mask to avoid out-of-bounds
        a_mask = (offs_m < M)[:, None] & ((k_offset + tl.arange(0, BLOCK_K)) < K)[None, :]
        b_mask = ((k_offset + tl.arange(0, BLOCK_K)) < K)[:, None] & (offs_n < N)[None, :]

def custom_kernel(data):
    """
    TriMul forward kernel: computes the outgoing version of TriMul as described in AlphaFold3.
    The heavy O(N^3) triple product is implemented as a batched half‑precision GEMM using Triton.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm on the last dimension
    x = F.layer_norm(input_tensor,
                     (dim,),
                     weight=weights["norm.weight"],
                     bias=weights["norm.bias"])

    # Linear projections (dim -> hidden_dim)
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Apply mask if present
    if not nomask:
        # ensure mask is float and broadcastable
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # Gating
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to half for the Triton GEMM
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    # Reshape for batched GEMM: (B, hidden_dim, N, N) -> (B*hidden_dim, N, N)
    B, N, _, _ = left_h.shape
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim
    left_flat = left_perm.view(batch_h, N, N)
    right_flat = right_perm.view(batch_h, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Compute strides (all int64)
    stride_a = left_flat.stride()
    stride_b = right_flat.stride()
    stride_c = out_flat.stride()

    stride_ah = stride_a[0]
    stride_am = stride_a[1]
    stride_ak = stride_a[2]

    # B is transposed inside the kernel: we need row‑stride = original column stride
    stride_bh = stride_b[0]
    stride_bk = stride_b[2]   # K‑dimension stride (original column)
    stride_bn = stride_b[1]   # N‑dimension stride (original row)

    stride_ch = stride_c[0]
    stride_cm = stride_c[1]
    stride_cn = stride_c[2]

    M = N
    K = N

    # Choose tile sizes – tuned for H100 fp16 GEMM
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n

        right_flat.data_ptr(),
        out_flat.data_ptr(),

"""
TriMul “outgoing” forward pass (AlphaFold‑3 style).

The expensive O(N³·H) contraction
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is implemented as a batched half‑precision GEMM using Triton.
All other linear / LayerNorm / sigmoid operations are performed with
torch.nn.functional for maximum CUDA library support.
The kernel processes one batch‑*‑head slice (shape (M,N) with M=N=seq_len)
at a time and stores the result in FP16; it is cast back to FP32 before
the final LayerNorm and projection.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    stride_ah, stride_am, stride_ak,
    stride_bh, stride_bk, stride_bn,
    stride_ch, stride_cm, stride_cn,
    M, N, K,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """Batched GEMM: C = A @ Bᵀ  (A,B : [batch, M, K] , Bᵀ interpreted via swapped strides)"""
    pid = tl.program_id(0)

    # grid layout -------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    batch_id = pid // (num_blocks_m * num_blocks_n)
    block_id = pid % (num_blocks_m * num_blocks_n)
    pid_m = block_id // num_blocks_n
    pid_n = block_id % num_blocks_n

    # tile offsets -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator (FP32) -----------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # k‑loop -------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_off = k * BLOCK_K

        # A tile (M×K)
        a_ptrs = a_ptr + batch_id * stride_ah \
                 + offs_m[:, None] * stride_am \
                 + (k_off + tl.arange(0, BLOCK_K))[None, :] * stride_ak

        # Bᵀ tile (K×N) – we treat the original B layout with swapped strides
        b_ptrs = b_ptr + batch_id * stride_bh \
                 + (k_off + tl.arange(0, BLOCK_K))[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # out‑of‑bounds masks
        a_mask = (offs_m < M)[:, None] & ((k_off + tl.arange(0, BLOCK_K)) < K)[None, :]
        b_mask = ((k_off + tl.arange(0, BLOCK_K)) < K)[:, None] & (offs_n < N)[None, :]

        a = tl.load(a_ptrs, mask=a_mask, other=0.0)
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        acc += tl.dot(a, b)               # (BLOCK_M, BLOCK_N)

    # write result -------------------------------------------------
    c_ptrs = c_ptr + batch_id * stride_ch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim]
        - mask         : torch.Tensor[B, N, N] (ignored if config["nomask"] is True)
        - weights      : dict of torch.Tensors containing all learned parameters
        - config       : dict with keys "dim", "hidden_dim", "nomask" (optional)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1️⃣  LayerNorm on the input (per‑position across the feature dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2️⃣  Linear projections   dim → hidden_dim   (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over the feature dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # -----------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5️⃣  Batched GEMM (the O(N³) core) using Triton
    #    Compute   out_h = left_h @ right_hᵀ   for each head h ∈ [H]
    # -----------------------------------------------------------------
    #   a) cast to FP16 and make layout [B, H, N, N] → [B*H, N, N]
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, _ = left_h.shape
    #   b) bring head dimension next to batch for a flat batch dimension
    left_perm = left_h.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right_h.permute(0, 3, 1, 2).contiguous()
    batch_h = B * hidden_dim

    left_flat = left_perm.view(batch_h, N, N)            # [B*H, N, N]
    right_flat = right_perm.view(batch_h, N, N)          # [B*H, N, N]

    out_flat = torch.empty_like(left_flat)               # [B*H, N, N] (FP16)

    #   c) stride extraction (all int64)
    stride_a = left_flat.stride()
    stride_b = right_flat.stride()
    stride_c = out_flat.stride()

    stride_ah, stride_am, stride_ak = stride_a          # (batch, M, K)
    # B is accessed transposed inside the kernel → swap the last two strides
    stride_bh = stride_b[0]
    stride_bk = stride_b[2]   # original column‑stride becomes row‑stride for Bᵀ
    stride_bn = stride_b[1]   # original row‑stride becomes column‑stride for Bᵀ

    stride_ch, stride_cm, stride_cn = stride_c          # (batch, M, N)

    M = N
    K = N

    #   d) block sizes – tuned for fp16 GEMM on H100
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    num_blocks_m = (M + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_h * num_blocks_m * num_blocks_n

    #   e) launch Triton kernel
    _batched_matmul_kernel[(total_blocks,)](
        left_flat.data_ptr(),
        right_flat.data_ptr(),
        out_flat.data_ptr(),
        stride_ah, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        M, N, K,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # 6️⃣  Reshape back to [B, N, N, H] (convert to FP32 for the rest)
    # -----------------------------------------------------------------
    out = out_flat.view(B, hidden_dim, N, N) \
                    .permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # -----------------------------------------------------------------
    # 7️⃣  Final LayerNorm on hidden dimension
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )

    # -----------------------------------------------------------------
    # 8️⃣  Apply output gate and project back to original dim
    # -----------------------------------------------------------------
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
