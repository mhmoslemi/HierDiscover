import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    B, N, H,
    stride_az, stride_ai, stride_ak, stride_ah,
    stride_bz, stride_bi, stride_bk, stride_bh,
    stride_cz, stride_ci, stride_cj, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bh = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch idx and channel idx
    b = pid_bh // H
    h = pid_bh % H

    # offsets
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask for i,j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_start in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k_start * BLOCK_K
        cur_k = k_base + offs_k
        mask_k = cur_k < N

        # pointers for a (left) and b (right transposed)
        a_ptrs = a_ptr + \
            b * stride_az + \
            h * stride_ah + \
            offs_i[:, None] * stride_ai + \
            cur_k[None, :] * stride_ak

        b_ptrs = b_ptr + \
            b * stride_bz + \
            h * stride_bh + \
            cur_k[:, None] * stride_bk + \
            offs_j[None, :] * stride_bi

        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        acc += tl.dot(a, b)

    # compute output pointer
    c_ptrs = c_ptr + \
        b * stride_cz + \
        h * stride_ch + \
        offs_i[:, None] * stride_ci + \
        offs_j[None, :] * stride_cj

    # store, casting back to FP16
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """Forward pass of the outgoing TriMul module using fused Triton GEMM.

    The kernel computes the core cubic term  out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    for every batch b and hidden channel h.  The rest of the
    TriMul pipeline (layer‑norms, linear projections, gates and the final linear) is
    performed with PyTorch for simplicity.  The heavy O(N³·H) part is accelerated
    by a custom Triton kernel.
    """
    import torch
    import torch.nn.functional as F
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ----- 1. LayerNorm over dim -----
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ----- 2. Linear projections (no bias) -----
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, left_proj_weight)
    right = F.linear(x, right_proj_weight)

    # ----- 3. Mask (optional) -----
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ----- 4. Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ----- 5. Apply gates -----
    left = left * left_gate
    right = right * right_gate

    # ----- 6. Cast to FP16 for the cubic term -----
    left = left.contiguous().to(torch.float16)

"""
TriMul (outgoing) forward pass with a fused Triton kernel.

Algorithm
---------
1. **Layer‑norm** over the last dimension of the input (`dim`).
2. **Linear projections** (`left_proj`, `right_proj`) produce tensors
   of shape `[B, N, N, H]` where `H = hidden_dim`.
3. Optional **mask** multiplication.
4. **Gates** (`left_gate`, `right_gate`, `out_gate`) are computed by a
   sigmoid‑activated linear layer and applied to the projected tensors.
5. The cubic term

       out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

   is evaluated with a custom Triton GEMM kernel.  The kernel treats each
   pair `(b, h)` as an independent matrix‑multiply of size `N×N` and
   runs a tiled `M×K·K×N` multiplication (block sizes 64×64×64).  All
   heavy work (O(N³·H) FLOPs) is therefore carried out on the GPU in
   FP16 with FP32 accumulation.
6. The result is cast back to FP32, **Layer‑norm** over the hidden
   dimension, multiplied by the `out_gate`, and finally projected back
   to the original `dim` with `to_out`.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,               # left  [B, N, N, H]  (FP16)
    b_ptr,               # right [B, N, N, H]  (FP16)
    c_ptr,               # out   [B, N, N, H]  (FP16)
    B, N, H,
    stride_az, stride_ai, stride_ak, stride_ah,   # left strides
    stride_bz, stride_bi, stride_bk, stride_bh,   # right strides
    stride_cz, stride_ci, stride_cj, stride_ch,   # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Batched GEMM for the TriMul cubic term.

    For every (batch, hidden) pair we compute
        C = A @ Bᵀ ,
    where A = left[b, i, k, h]  (M×K) and
          B = right[b, j, k, h] (N×K) → Bᵀ is (K×N).
    """
    # -------------------------------------------------------------------------
    # 1) program IDs → batch & hidden‑channel index, and tile indices
    # -------------------------------------------------------------------------
    pid_bh = tl.program_id(0)            # combined batch * hidden
    pid_m  = tl.program_id(1)            # i‑tile
    pid_n  = tl.program_id(2)            # j‑tile

    batch_idx   = pid_bh // H
    channel_idx = pid_bh % H

    # -------------------------------------------------------------------------
    # 2) offsets for the current tile
    # -------------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------------------
    # 3) loop over the reduction dimension k
    # -------------------------------------------------------------------------
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        cur_k = k_tile * BLOCK_K + offs_k          # absolute k positions
        mask_k = cur_k < N

        # ----- A (left) : shape (BLOCK_M, cur_K) -----
        a_ptrs = (
            a_ptr
            + batch_idx * stride_az
            + channel_idx * stride_ah
            + offs_i[:, None] * stride_ai
            + cur_k[None, :] * stride_ak
        )
        a = tl.load(
            a_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- B (right) transposed: shape (cur_K, BLOCK_N) -----
        # we load right[j, k] as (k, j) to obtain the transpose on‑the‑fly
        b_ptrs = (
            b_ptr
            + batch_idx * stride_bz
            + channel_idx * stride_bh
            + cur_k[:, None] * stride_bk
            + offs_j[None, :] * stride_bi
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ----- matrix multiply & accumulate -----
        acc += tl.dot(a, b)         # (BLOCK_M, BLOCK_N)

    # -------------------------------------------------------------------------
    # 4) write the result (cast back to FP16)
    # -------------------------------------------------------------------------
    c_ptrs = (
        c_ptr
        + batch_idx * stride_cz
        + channel_idx * stride_ch
        + offs_i[:, None] * stride_ci
        + offs_j[None, :] * stride_cj
    )
    tl.store(
        c_ptrs,
        acc.to(tl.float16),
        mask=mask_i[:, None] & mask_j[None, :],
    )


def custom_kernel(data):
    """Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be ignored)
        - weights      : dict of torch.Tensor containing all learned parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device).
    """
    # -------------------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -------------------------------------------------------------------------
    # model weights
    # -------------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # -------------------------------------------------------------------------
    # 1. Input layer‑norm over the last dimension
    # -------------------------------------------------------------------------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # -------------------------------------------------------------------------
    # 2. Linear projections (bias‑free, shape [..., hidden_dim])
    # -------------------------------------------------------------------------
    left  = F.linear(x, left_proj_weight)   # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    # -------------------------------------------------------------------------
    # 3. Optional mask application
    # -------------------------------------------------------------------------
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)      # [B, N, N, 1]
        left  = left * mask_f
        right = right * mask_f

    # -------------------------------------------------------------------------
    # 4. Gates (sigmoid) – still in FP32 for stability
    # -------------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # -------------------------------------------------------------------------
    # 5. Apply gates
    # -------------------------------------------------------------------------
    left  = left  * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 6. Cast to FP16 – this halves memory bandwidth for the cubic term
    # -------------------------------------------------------------------------
    left  = left.contiguous().to(torch.float16)
    right = right.contiguous().to(torch.float16)

    B, N, _, H = left.shape
    assert H == hidden_dim, "Hidden dimension mismatch."

    # -------------------------------------------------------------------------
    # 7. Allocate output for the cubic term (FP16)
    # -------------------------------------------------------------------------
    out_fp16 = torch.empty_like(left, dtype=torch.float16)

    # -------------------------------------------------------------------------
    # 8. Gather strides – required for pointer arithmetic inside the kernel
    # -------------------------------------------------------------------------
    stride_al_b, stride_al_i, stride_al_k, stride_al_h = left.stride()
    stride_br_b, stride_br_i, stride_br_k, stride_br_h = right.stride()
    stride_c_b, stride_c_i, stride_c_j, stride_c_h = out_fp16.stride()

    # -------------------------------------------------------------------------
    # 9. Triton launch configuration
    # -------------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * H,                                 # batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
    )

    # -------------------------------------------------------------------------
    # 10. Execute the fused GEMM kernel
    # -------------------------------------------------------------------------
    tri_mul_gemm_kernel[grid](
        left.data_ptr(),
        right.data_ptr(),
        out_fp16.data_ptr(),
        B, N, H,
        stride_al_b, stride_al_i, stride_al_k, stride_al_h,
        stride_br_b, stride_br_i, stride_br_k, stride_br_h,
        stride_c_b, stride_c_i, stride_c_j, stride_c_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 benefits from a higher warp count
    )
    torch.cuda.synchronize()   # ensure kernel completion before next ops

    # -------------------------------------------------------------------------
    # 11. Convert back to FP32 for the remaining (lighter) layers
    # -------------------------------------------------------------------------
    out = out_fp16.to(torch.float32)

    # -------------------------------------------------------------------------
    # 12. Second layer‑norm over the hidden dimension
    # -------------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )

    # -------------------------------------------------------------------------
    # 13. Apply the output gate (element‑wise)
    # -------------------------------------------------------------------------
    out = out * out_gate

    # -------------------------------------------------------------------------
    # 14. Final projection back to the original `dim`
    # -------------------------------------------------------------------------
    out = F.linear(out, to_out_weight)   # no bias, shape [B, N, N, dim]

    return out
