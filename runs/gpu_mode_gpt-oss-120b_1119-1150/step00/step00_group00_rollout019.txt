                          stride_Ab, stride_Ar, stride_Ac,

                          stride_Cb, stride_Cr, stride_Cc,

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

@triton.jit
def batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_Ab, stride_Am, stride_Ak,
    stride_Bb, stride_Bk, stride_Bn,
    stride_Cb, stride_Cm, stride_Cn,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound
    mask_m = offs_m < M
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        # compute offsets
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # load A block [M,K]
        a_ptrs = A_ptr + pid_b * stride_Ab + offs_m[:, None] * stride_Am + offs_k[None, :] * stride_Ak
        a = tl.load(a_ptrs, mask=(mask_m[:, None] & (offs_k < K)), other=0.0)
        # load B block [K,N]
        b_ptrs = B_ptr + pid_b * stride_Bb + offs_k[:, None] * stride_Bk + offs_n[None, :] * stride_Bn
        b = tl.load(b_ptrs, mask=((offs_k[:, None] < K) & mask_n[None, :]), other=0.0)
        # compute dot product (M,K) · (K,N) => (M,N)
        acc += tl.dot(a, b)

    # store result
    c_ptrs = C_ptr + pid_b * stride_Cb + offs_m[:, None] * stride_Cm + offs_n[None, :] * stride_Cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    bsz, N, _, C = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # LayerNorm over last dim C (dim)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Use torch.nn.functional.layer_norm
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Cast to fp16 for following operations
    x16 = x.to(torch.float16)

    # Linear projections
    left = torch.nn.functional.linear(x16, weights["left_proj.weight"].to(torch.float16))
    right = torch.nn.functional.linear(x16, weights["right_proj.weight"].to(torch.float16))

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x16, weights["left_gate.weight"].to(torch.float16)))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x16, weights["right_gate.weight"].to(torch.float16)))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x16, weights["out_gate.weight"].to(torch.float16)))

    # Apply mask if present and not nomask
    if not config.get("nomask", False):
        mask16 = mask[..., None].to(torch.float16)
        left = left * mask16
        right = right * mask16

    # Apply gating
    left = left * left_gate
    right = right * right_gate

    # Prepare for batched matrix multiplication: reshape to (B*hidden_dim, N, N)
    # Permute to bring hidden_dim next to batch
    left_reshaped = left.permute(0, 3, 1, 2).contiguous().view(bsz * hidden_dim, N, N)
    # Transpose the inner N dimensions of right (j,k) -> (k,j)
    right_t = right.permute(0, 3, 2, 1).contiguous().view(bsz * hidden_dim, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_reshaped)

    # Triton kernel parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Compute grid dimensions
    grid = (bsz * hidden_dim,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    # Launch kernel
    batched_matmul_kernel[grid](
        left_reshaped,
        right_t,
        out_flat,
        N, N, N,
        left_reshaped.stride(0), left_reshaped.stride(1), left_reshaped.stride(2),
        right_t.stride(0), right_t.stride(1), right_t.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(bsz, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # Convert to float32 for following LayerNorm & projection
    out = out.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Apply out gate
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # Final linear projection to dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          M, N, K,
                          stride_a_batch, stride_a_m, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_m, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    """Batched matrix multiplication C = A @ B, where A and B are 3‑D tensors
    with shape (batch, M, K) and (batch, K, N) respectively.

    The kernel is tiled with BLOCK_M × BLOCK_N output tiles and accumulates
    over BLOCK_K sized chunks of the K dimension. It works for arbitrary
    batch size and is used to implement the TriMul einsum
    `out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]`.
    """
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Coordinates of the current output tile
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in BLOCK_K steps
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)

        # Load a block of A (shape BLOCK_M × BLOCK_K)
        a_ptrs = A_ptr + pid_batch * stride_a_batch \
                         + offs_m[:, None] * stride_a_m \
                         + offs_k[None, :] * stride_a_k
        mask_a = mask_m[:, None] & (offs_k[None, :] < K)
        a = tl.load(a_ptrs, mask=mask_a, other=0.0)

        # Load a block of B (shape BLOCK_K × BLOCK_N)
        b_ptrs = B_ptr + pid_batch * stride_b_batch \
                         + offs_k[:, None] * stride_b_k \
                         + offs_n[None, :] * stride_b_n
        mask_b = (offs_k[:, None] < K) & mask_n[None, :]
        b = tl.load(b_ptrs, mask=mask_b, other=0.0)

        # Accumulate the product
        acc += tl.dot(a, b)

    # Write the tile back to C
    c_ptrs = C_ptr + pid_batch * stride_c_batch \
                     + offs_m[:, None] * stride_c_m \
                     + offs_n[None, :] * stride_c_n
    mask_c = mask_m[:, None] & mask_n[None, :]
    tl.store(c_ptrs, acc, mask=mask_c)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul operation.
    The function receives a tuple (input, mask, weights, config) and returns
    the transformed tensor with shape [bs, seq_len, seq_len, dim].

    Steps:
      1. Layer‑norm the input.
      2. Linear projections left/right and three gates.
      3. Apply optional mask and per‑gate masking.
      4. Compute the core einsum using a fused batched matmul kernel.
      5. Layer‑norm, gate, and final linear projection back to ``dim``.
    """
    # unpack
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------
    # 1) Input LayerNorm (over dim)
    # ------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # -------------------------------------------------
    # 2) Linear projections and gates (all in fp32)
    # -------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -------------------------------------------------
    # 3) Optional mask and gating
    # -------------------------------------------------
    if not config.get("nomask", False):
        mask_exp = mask.unsqueeze(-1).to(x.dtype)  # shape [bs, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------
    # 4) Core einsum using Triton (left @ right^T per hidden dim)
    # -------------------------------------------------
    # reshape to (batch * hidden_dim, N, N) where the hidden dim is
    # folded into the batch dimension to obtain a standard batched GEMM.
    left_ = left.permute(0, 3, 1, 2).contiguous().view(bs * hidden_dim, N, N)   # (B*H, N, N)
    # transpose the second N dimension of right to match the einsum pattern
    right_t = right.permute(0, 3, 2, 1).contiguous().view(bs * hidden_dim, N, N)  # (B*H, N, N)

    out_flat = torch.empty_like(left_)  # same dtype (float32)

    # Triton kernel launch parameters
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        bs * hidden_dim,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batched_matmul_kernel[grid](
        left_,
        right_t,
        out_flat,
        N,
        N,
        N,
        left_.stride(0),
        left_.stride(1),
        left_.stride(2),
        right_t.stride(0),
        right_t.stride(1),
        right_t.stride(2),
        out_flat.stride(0),
        out_flat.stride(1),
        out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion before moving on
    torch.cuda.synchronize()

    # Restore the original 4‑D layout: [bs, N, N, hidden_dim]
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -------------------------------------------------
    # 5) Final normalization, gating and projection
    # -------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate  # out_gate is already fp32
    out = F.linear(out, weights["to_out.weight"])

    return out

    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr

    Compute C = A @ B for a 3‑D batch of matrices.
    A : (batch, M, K)
    B : (batch, K, N)

    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        x,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [bs, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections and gates (no bias in any Linear)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [bs, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # 3) Optional mask and gating
    # ------------------------------------------------------------------

        eps=eps,
    )
    out = out * out_gate                     # gating (still float32)
