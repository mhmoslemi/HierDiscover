    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak,
    stride_bk, stride_bn,
    stride_cm, stride_cn,
    BLOCK_M, BLOCK_N, BLOCK_K
):
    pid = tl.program_id(axis=0)  # block ID
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)
    batch_hid = tl.program_id(1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    pid_m = pid // num_blocks_n
    pid_n = pid % num_blocks_n

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-range
    mask_m = offs_m < M
    mask_n = offs_n < N

    a_ptrs = A_ptr + batch_hid * stride_A_batch + offs_m[:, None] * stride_A_m + tl.arange(0, BLOCK_K)[None, :] * stride_A_k
    b_ptrs = B_ptr + batch_hid * stride_B_batch + tl.arange(0, BLOCK_K)[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (tl.arange(0, BLOCK_K)[None, :] < K), other=0.0, dtype=tl.float16)

        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    c_ptrs = C_ptr + batch_hid * stride_C_batch + offs_m[:, None] * stride_C_m + offs_n[None, :] * stride_C_n
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])

    input_tensor, mask, weights, config = data
    # shapes
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    device = input_tensor.device
    eps = 1e-5

    # Step 1: layernorm
    # using torch.nn.functional.layer_norm
    # shape: [B,N,N,D]
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = torch.nn.functional.layer_norm(input_tensor, (D,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Step 2: linear projections
    left_proj_weight = weights["left_proj.weight"]  # shape (H, D)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # Linear using torch.nn.functional.linear
    # output shape [B,N,N,H]
    # Using weight.T for linear expects weight (out_features, in_features)
    left = torch.nn.functional.linear(x_norm, left_proj_weight) # shape B,N,N,H
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if not config.get("nomask", True):
        # mask shape [B,N,N] -> unsqueeze(-1)
        mask_exp = mask.unsqueeze(-1)
        left = left * mask_exp
        right = right * mask_exp

    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    right_perm = right.permute(0, 3, 1, 2).contiguous()

    @triton.jit
    def tri_mul_matmul_kernel(A_ptr, B_ptr, C_ptr,
                              M, N, K,
                              stride_A_batch, stride_A_m, stride_A_k,
                              stride_B_batch, stride_B_k, stride_B_n,
                              stride_C_batch, stride_C_m, stride_C_n,
                              BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        pid = tl.program_id(0)
        batch_hid = tl.program_id(1)  # flattened batch*hidden
        num_blocks_m = tl.cdiv(M, BLOCK_M)
        num_blocks_n = tl.cdiv(N, BLOCK_N)
        pid_m = pid // num_blocks_n
        pid_n = pid % num_blocks_n

        offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_m = offs_m < M
        mask_n = offs_n < N

        # offsets for k loop
        offs_k = tl.arange(0, BLOCK_K)

        # base pointers
        a_ptrs = A_ptr + batch_hid * stride_A_batch + offs_m[:, None] * stride_A_m + offs_k[None, :] * stride_A_k
        b_ptrs = B_ptr + batch_hid * stride_B_batch + offs_k[:, None] * stride_B_k + offs_n[None, :] * stride_B_n

    out_norm = torch.nn.functional.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Apply out_gate
    out_gate = out_gate  # shape [B,N,N,H] already computed
    out_gated = out_norm * out_gate

    # Final linear projection to dim D
    # Use torch.nn.functional.linear with weight (dim, H)
    out_final = torch.nn.functional.linear(out_gated, to_out_weight)

"""
custom_kernel: Triton‑accelerated implementation of the “outgoing” TriMul module
(reference: AlphaFold3).

Algorithm
---------
1. Layer‑norm over the last dimension of the input tensor `x`  (torch implementation).
2. Linear projections (left/right) and their gates (torch implementation).
   * left = (x @ left_proj_weight.T)          → shape [B, N, N, H]
   * right = (x @ right_proj_weight.T)        → shape [B, N, N, H]
   * Apply optional binary mask.
   * Apply element‑wise sigmoid gates.
3. Core O(N³) “tri‑multiplicative” update:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is a batch of H independent matrix multiplications:
      out_d = left_d @ right_dᵀ   for each hidden dimension d.
   A custom Triton kernel computes all B·H matmuls in parallel, using
   block‑tiling (64×64) and half‑precision loads with FP32 accumulation.
4. Layer‑norm over the hidden dimension of `out` (torch implementation).
5. Gating with `out_gate` (element‑wise multiplication).
6. Final linear projection back to `dim` (torch implementation).

Only the heavy N³ part is performed in Triton; the remaining ops use
high‑performance Torch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: batched matmul C = A @ Bᵀ
#   A : (BH, M, K)   – left side,  M = N, K = N
#   B : (BH, N, K)   – right side (will be read as transposed)
#   C : (BH, M, N)   – result
# --------------------------------------------------------------------------- #
@triton.jit
def tri_mul_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    M, N, K,                     # matrix sizes (M=N_seq, N=N_seq, K=N_seq)
    stride_A_batch, stride_A_m, stride_A_k,
    stride_B_batch, stride_B_k, stride_B_n,
    stride_C_batch, stride_C_m, stride_C_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # ----------------- program ID handling ---------------------------------
    pid = tl.program_id(0)               # tile index in the (M,N) plane
    bh_id = tl.program_id(1)              # flattened batch*hidden index

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    pid_m = pid // num_blocks_n           # tile row
    pid_n = pid % num_blocks_n            # tile column

    # ----------------- tile offsets ----------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_m = offs_m < M
    mask_n = offs_n < N

    # k‑loop offsets (shared across all iterations)
    offs_k = tl.arange(0, BLOCK_K)

    # ----------------- pointer arithmetic -----------------------------------
    a_ptrs = A_ptr + bh_id * stride_A_batch \
                + offs_m[:, None] * stride_A_m \
                + offs_k[None, :] * stride_A_k

    b_ptrs = B_ptr + bh_id * stride_B_batch \
                + offs_k[:, None] * stride_B_k \
                + offs_n[None, :] * stride_B_n

    # ----------------- accumulation -----------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & (offs_k[None, :] < K),
                    other=0.0, dtype=tl.float16)
        b = tl.load(b_ptrs,
                    mask=(offs_k[:, None] < K) & mask_n[None, :],
                    other=0.0, dtype=tl.float16)

        # dot: (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)
        acc += tl.dot(a, b)

        # advance pointers to next K‑tile
        a_ptrs += BLOCK_K * stride_A_k
        b_ptrs += BLOCK_K * stride_B_k

    # ----------------- store result -----------------------------------------
    c_ptrs = C_ptr + bh_id * stride_C_batch \
                 + offs_m[:, None] * stride_C_m \
                 + offs_n[None, :] * stride_C_n

    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# --------------------------------------------------------------------------- #
# Main entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward pass for the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (float32 or bool)
            - weights: dict of tensors required by the module
            - config: dict containing at least:
                * "dim"          – input/output channel dimension
                * "hidden_dim"   – hidden dimension (c_z)
                * "nomask" (opt) – if True skip mask multiplication

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------- #
    # unpack
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    B, N, _, D = input_tensor.shape
    H = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm over channel dimension D
    # ------------------------------------------------------------------- #
    x = F.layer_norm(
        input_tensor,
        (D,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, D]

    # ------------------------------------------------------------------- #
    # 2) Linear projections + optional mask
    # ------------------------------------------------------------------- #
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,H]
    right = F.linear(x, weights["right_proj.weight"])  # [B,N,N,H]

    if not config.get("nomask", True):
        # mask is broadcasted over the hidden dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------- #
    # 3) Gating
    # ------------------------------------------------------------------- #
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,H]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate  # now both are [B,N,N,H]

    # ------------------------------------------------------------------- #
    # 4) Core TriMul (batched matmul) – Triton
    # ------------------------------------------------------------------- #
    # (a) bring hidden dim to batch dimension
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # merge batch & hidden: shape (BH, N, N)
    BH = B * H
    left_mat = left_perm.view(BH, N, N).contiguous()
    right_mat = right_perm.view(BH, N, N).contiguous()

    # half‑precision for the kernel (fast on H100)
    left_fp16 = left_mat.to(torch.float16)
    right_fp16 = right_mat.to(torch.float16)
    out_fp16 = torch.empty_like(left_fp16)

    # stride extraction (after view -> 3‑D tensor)
    stride_A_batch, stride_A_m, stride_A_k = left_fp16.stride()
    stride_B_batch, stride_B_row, stride_B_col = right_fp16.stride()
    # For Bᵀ we swap the row/col strides
    stride_B_k = stride_B_col          # original column stride
    stride_B_n = stride_B_row          # original row stride
    stride_C_batch, stride_C_m, stride_C_n = out_fp16.stride()

    # block sizes – tuned for H100 half‑precision math
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M) * triton.cdiv(N, BLOCK_N),  # tiles over M×N
        BH,                                                 # batch*hidden
    )
    tri_mul_matmul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        N,                     # M
        N,                     # N
        N,                     # K
        stride_A_batch, stride_A_m, stride_A_k,
        stride_B_batch, stride_B_k, stride_B_n,
        stride_C_batch, stride_C_m, stride_C_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # reshape back to (B, N, N, H) and cast to FP32
    out = out_fp16.to(torch.float32).view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5) Post‑matmul LayerNorm (over hidden dim H) + out‑gate
    # ------------------------------------------------------------------- #
    out = F.layer_norm(
        out,
        (H,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate

    # ------------------------------------------------------------------- #
    # 6) Final projection back to dim D
    # ------------------------------------------------------------------- #
    out = F.linear(out, weights["to_out.weight"])   # [B,N,N,D]

    return out
