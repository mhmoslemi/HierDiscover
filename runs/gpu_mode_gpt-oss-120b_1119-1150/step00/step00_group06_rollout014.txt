@triton.jit
def batch_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # compute pointers
    a_ptrs = a_ptr + pid_batch * stride_a_batch + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + pid_batch * stride_b_batch + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # load a block, maybe mask out-of-bounds
    a = tl.load(a_ptrs, mask=offs_m < M, other=0.0) # shape (BLOCK_M, BLOCK_K)
    b = tl.load(b_ptrs, mask=offs_n < N, other=0.0) # shape (BLOCK_K, BLOCK_N)

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # load a block for this k chunk
        a = tl.load(a_ptrs, mask=(offs_m < M) & (offs_k[:, None] < K), other=0.0)
        b = tl.load(b_ptrs, mask=(offs_n < N) & (offs_k[None, :] < K), other=0.0)
        # cast to float16 then dot -> produce float32 accumulation
        acc += tl.dot(a, b)  # note dot expects a: (BLOCK_M, BLOCK_K), b: (BLOCK_K, BLOCK_N)
        # move pointers
        a_ptrs += BLOCK_K * stride_a_k
        b_ptrs += BLOCK_K * stride_b_k

    # store result
    c_ptrs = c_ptr + pid_batch * stride_c_batch + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # Layernorm x = (x - mean)/sqrt(var+eps) * weight + bias
    x = input_tensor
    # compute mean & var across dim (last dim) per each element of B,N,N
    # Use torch.nn.functional.layer_norm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Apply mask if not nomask
    if not config.get("nomask", False):
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # reshape for batched GEMM:
    # left: [B,N,N,hidden_dim] -> permute to [B, hidden_dim, N, N] and flatten batch
    left_perm = left.permute(0,3,1,2).contiguous()
    right_perm = right.permute(0,3,1,2).contiguous()

    # reshape to (B*hidden_dim, N, N)
    left_flat = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # optional: cast to half
    left_half = left_flat.to(torch.float16)
    right_half = right_flat.to(torch.float16)

    # compute B_T = right_half.transpose(-2,-1)
    right_t_half = right_half.transpose(-2, -1)

    # allocate output tensor
    out_half = torch.empty_like(left_half, dtype=torch.float16)

    # compute batched matmul using triton
    batch = left_half.shape[0]
    M = N
    K = N
    # compute strides
    stride_a_batch = left_half.stride(0)
    stride_a_m = left_half.stride(1)
    stride_a_k = left_half.stride(2)
    stride_b_batch = right_t_half.stride(0)
    stride_b_k = right_t_half.stride(1) # because shape (batch, K, N) after transpose; stride for rows = stride after transpose
    stride_b_n = right_t_half.stride(2)
    stride_c_batch = out_half.stride(0)
    stride_c_m = out_half.stride(1)
    stride_c_n = out_half.stride(2)

    # Choose BLOCK sizes; we can tune based on N.

    # Determine grid
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32
    grid = (batch, (M + BLOCK_M - 1)//BLOCK_M, (N + BLOCK_N - 1)//BLOCK_N)

    batch_matmul_kernel[grid](
        a_ptr=left_half,
        b_ptr=right_t_half,
        c_ptr=out_half,
        M=M,
        N=N,
        K=K,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to float32
    out = out_half.to(dtype)

    # reshape and permute back to [B, N, N, hidden_dim]
    out = out.view(B, hidden_dim, N, N).permute(0,2,3,1).contiguous()

    # to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # apply out_gate
    out = out * out_gate

    # final linear projection
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)

    return out

"""
TriMul “outgoing” forward pass – high‑performance Triton implementation.

The core of the module is the three‑tensor einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For each batch and hidden channel d this is a dense matrix product:
    L_d (N×N) @ R_dᵀ (N×N) → O_d (N×N)

We fuse all batch × hidden‑channel GEMMs into a single Triton kernel.
The surrounding LayerNorms, linear projections and gating are performed
with PyTorch (they are cheap compared with the N³ work).  The GEMM uses
FP16 inputs → Tensor‑core accumulation in FP32 → FP16 output, which gives
the best throughput on an H100.

Key ideas
---------
* Collapse the batch (B) and hidden‑dim (H) into a single GEMM batch:
      (B·H) × N × N   matrices.
* Use `right.transpose(-2,-1)` so that the kernel can read the
  transposed matrix without an explicit copy – only the strides change.
* The Triton kernel loops over the K‑dimension in tiles (BLOCK_K) and
  accumulates with `tl.dot`, which maps to tensor cores.
* After the kernel we reshape / permute back to `[B, N, N, H]`,
  apply the final LayerNorm, gating and output projection.

The kernel works for any batch size (1‑2), sequence length up to 1024 and
hidden dimensions used in the tests (128).  It runs entirely on the GPU
and returns a `torch.float32` tensor as required.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batch_matmul_kernel(
    a_ptr,                     # A = left  (batch, M, K)   – FP16
    b_ptr,                     # B = rightᵀ (batch, K, N)   – FP16 (transposed view)
    c_ptr,                     # C = out   (batch, M, N)   – FP16 (accumulated as FP32)
    M, N, K,                   # matrix sizes (all = seq_len)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched GEMM: C = A @ B   (A: M×K, B: K×N) for many independent batches."""
    pid_batch = tl.program_id(0)          # 0‑th axis → batch*hidden
    pid_m = tl.program_id(1)              # tile over rows   (M)
    pid_n = tl.program_id(2)              # tile over cols   (N)

    # ---- block coordinates -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---- main K‑loop -------------------------------------------------------
    num_k_blocks = tl.cdiv(K, BLOCK_K)
    for kb in range(num_k_blocks):
        cur_k = kb * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # pointers for this tile
        a_ptrs = (
            a_ptr
            + pid_batch * stride_a_batch
            + offs_m[:, None] * stride_a_m
            + cur_k[None, :] * stride_a_k
        )
        b_ptrs = (
            b_ptr
            + pid_batch * stride_b_batch
            + cur_k[:, None] * stride_b_k
            + offs_n[None, :] * stride_b_n
        )

        # load with boundary masks
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)  # (BM, BK)
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)  # (BK, BN)

        # dot produces FP32 accumulation automatically (FP16 inputs)
        acc += tl.dot(a, b)

    # ---- write back --------------------------------------------------------
    c_ptrs = (
        c_ptr
        + pid_batch * stride_c_batch
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_m[:, None] & mask_n[None, :])


def custom_kernel(data):
    """
    Compute the forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask_tensor, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32)
            - mask_tensor : torch.Tensor [B, N, N] (float32 or bool)
            - weights     : dict of model parameters (all float32)
            - config      : dict containing at least
                * "dim"          – input channel dimension
                * "hidden_dim"   – projection dimension
                * "nomask"       – bool, whether mask is to be ignored

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ----------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ----------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ----------------------------------------------------------------------
    # 3) Optional mask (mask shape [B,N,N])
    # ----------------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.to(dtype).unsqueeze(-1)      # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ----------------------------------------------------------------------
    # 4) Gating (sigmoid of linear transforms)
    # ----------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))     # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 5) Prepare tensors for batched GEMM: (B·H, N, N)
    # ----------------------------------------------------------------------
    #   left  → (B, H, N, N)
    #   right → (B, H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right = right.permute(0, 3, 1, 2).contiguous()

    # Collapse batch & hidden dimension
    left_flat = left.view(-1, N, N)                # [B*H, N, N]
    right_flat = right.view(-1, N, N)              # [B*H, N, N]

    # Cast to FP16 for tensor‑core GEMM
    left_fp16 = left_flat.to(torch.float16)
    right_fp16 = right_flat.to(torch.float16)

    # Transpose right on the fly (no copy, only stride change)
    right_t_fp16 = right_fp16.transpose(-2, -1)    # view: (batch, K=N, N)

    # Output buffer (FP16, will be up‑cast later)
    out_fp16 = torch.empty_like(left_fp16, dtype=torch.float16)

    # ----------------------------------------------------------------------
    # 6) Launch the batched GEMM kernel
    # ----------------------------------------------------------------------
    batch = left_fp16.shape[0]                     # B * hidden_dim

    # strides for the three tensors (as int64)
    stride_a_batch = left_fp16.stride(0)
    stride_a_m = left_fp16.stride(1)
    stride_a_k = left_fp16.stride(2)

    stride_b_batch = right_t_fp16.stride(0)
    stride_b_k = right_t_fp16.stride(1)            # row‑stride of K dimension after transpose
    stride_b_n = right_t_fp16.stride(2)            # column‑stride (N)

    stride_c_batch = out_fp16.stride(0)
    stride_c_m = out_fp16.stride(1)
    stride_c_n = out_fp16.stride(2)

    # Tile sizes – tuned for H100 tensor‑cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        batch,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    batch_matmul_kernel[grid](
        a_ptr=left_fp16,
        b_ptr=right_t_fp16,
        c_ptr=out_fp16,
        M=N,
        N=N,
        K=N,
        stride_a_batch=stride_a_batch,
        stride_a_m=stride_a_m,
        stride_a_k=stride_a_k,
        stride_b_batch=stride_b_batch,
        stride_b_k=stride_b_k,
        stride_b_n=stride_b_n,
        stride_c_batch=stride_c_batch,
        stride_c_m=stride_c_m,
        stride_c_n=stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # enough warps for the chosen tile size
    )

    # ----------------------------------------------------------------------
    # 7) Restore the original layout and continue with the rest of the model
    # ----------------------------------------------------------------------
    out = out_fp16.to(dtype)                        # back to float32

    # reshape from (B*H, N, N) → (B, N, N, H)
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # final LayerNorm over hidden dimension
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # multiply by the output gating
    out = out * out_gate

    # final linear projection back to dim
    out = F.linear(out, weights["to_out.weight"])

    return out
