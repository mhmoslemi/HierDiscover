"""
TriMul outgoing implementation (forward only).

The module follows the reference PyTorch code:
1. Layer‑norm over the last dimension.
2. Linear projections (`left_proj`, `right_proj`) and three gates.
3. Optional masking of the projected tensors.
4. Core contraction:
       out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batched matrix multiplication
   for each hidden channel d:
       out[b,d] = left[b,d] @ right[b,d].T
   The batched GEMM is executed by a custom Triton kernel.
5. Output layer‑norm, multiplicative gating and final linear projection.

The Triton kernel works on the reshaped tensors of shape
    (B*hidden_dim, N, N)  →  (batch_h, M, K) and (batch_h, K, N)
and writes the result (batch_h, M, N) in float32.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (C = A @ B^T) where
# A,B have shape (batch, M, K) and (batch, K, N) respectively.
# ----------------------------------------------------------------------
@triton.jit
def batched_matmul_kernel(
    # pointers
    a_ptr, b_ptr, c_ptr,
    # sizes
    batch, M, N, K,
    # strides (in elements, NOT bytes)
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_batch = tl.program_id(0)                # batch index
    pid_m = tl.program_id(1)                    # block row index
    pid_n = tl.program_id(2)                    # block col index

    # ----------------------- index computation -----------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds rows / cols
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Base pointers for the current batch
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch

    # Pointers to the start of the M‑ and N‑tiles (K is handled inside the loop)
    a_tile_ptr = a_batch_ptr + offs_m[:, None] * stride_a_m
    b_tile_ptr = b_batch_ptr + offs_n[None, :] * stride_b_n

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    num_k_iter = tl.cdiv(K, BLOCK_K)
    for k_iter in range(0, num_k_iter):
        # Absolute K indices for this iteration
        cur_k = k_iter * BLOCK_K + offs_k               # shape (BLOCK_K,)
        mask_k = cur_k < K

        # Load A tile  (M x K)
        a_ptrs = a_tile_ptr + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)

        # Load B tile  (K x N)   (B is already transposed)
        b_ptrs = b_tile_ptr + cur_k[:, None] * stride_b_k
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

    # Write back the result
    c_ptrs = c_batch_ptr + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton GEMM.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32 or bool)
        - weights      : dict of torch.Tensor containing all linear / layernorm params
        - config       : dict with at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    (x, mask, weights, config) = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = x.device
    dtype = x.dtype

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]          # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]        # (hidden_dim, dim)

    # torch.nn.functional.linear expects weight of shape (out_features, in_features)
    left = F.linear(x, left_proj_w)    # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_w)  # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # Optional masking (mask shape: [B,N,N] -> broadcast to hidden_dim)
    # ------------------------------------------------------------------
    if not nomask:
        # ensure mask is float32 and has same device
        mask = mask.to(dtype=dtype, device=device)
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gates
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))    # later applied after to_out_norm

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to half for the heavy GEMM and reshape for Triton
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape: (B, hidden_dim, N, N) → (B*hidden_dim, N, N)
    B, N, _, _ = left.shape
    left = left.permute(0, 3, 1, 2).contiguous()          # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()        # (B, hidden_dim, N, N)

    left_mat = left.view(B * hidden_dim, N, N)            # (batch_h, M, K)
    # right needs to be transposed on its last two dimensions for the
    # A @ B^T formulation.
    right_T = right.permute(0, 1, 3, 2).contiguous()     # (B, hidden_dim, N, N)
    right_mat = right_T.view(B * hidden_dim, N, N)       # (batch_h, K, N)

    # ------------------------------------------------------------------
    # Allocate output tensor (float32 for accumulation)
    # ------------------------------------------------------------------
    out_mat = torch.empty((B * hidden_dim, N, N), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (in elements) for the three tensors
    # ------------------------------------------------------------------
    stride_A_batch = left_mat.stride(0)
    stride_A_m = left_mat.stride(1)
    stride_A_k = left_mat.stride(2)

    stride_B_batch = right_mat.stride(0)
    stride_B_k = right_mat.stride(1)   # dimension K
    stride_B_n = right_mat.stride(2)   # dimension N

    stride_C_batch = out_mat.stride(0)
    stride_C_m = out_mat.stride(1)
    stride_C_n = out_mat.stride(2)

    # ------------------------------------------------------------------
    # Triton launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                               # batch dimension
        (N + BLOCK_M - 1) // BLOCK_M,                 # M blocks
        (N + BLOCK_N - 1) // BLOCK_N,                 # N blocks
    )

    batched_matmul_kernel[grid](
        left_mat,
        right_mat,
        out_mat,
        B * hidden_dim, N, N, N,                     # batch, M, N, K (K==N)
        stride_A_batch, stride_A_m, stride_A_k,
        stride_B_batch, stride_B_k, stride_B_n,
        stride_C_batch, stride_C_m, stride_C_n,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Reshape back to original layout: (B, N, N, hidden_dim)
    # ------------------------------------------------------------------
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Output LayerNorm, gating, and final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b)
    out = out * out_gate                     # broadcast multiply with out_gate (float32)

    to_out_w = weights["to_out.weight"]      # (dim, hidden_dim)
    out = F.linear(out, to_out_w)            # final projection, bias is None

    return out
