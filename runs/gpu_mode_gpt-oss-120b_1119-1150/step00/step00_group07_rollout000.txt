def elementwise_mul_kernel(a_ptr, b_ptr, c_ptr, n_elements,

"""
TriMul “outgoing” implementation with a fused Triton element‑wise kernel.

Algorithm
---------
1. Layer‑norm the input 4‑D tensor (B, N, N, C).
2. Linear projections to a hidden dimension H (left/right) and three gating
   projections (left_gate, right_gate, out_gate). All linear layers have no bias.
3. Apply optional pairwise mask (broadcast over H).
4. Fuse the element‑wise products left*left_gate and right*right_gate using a
   custom Triton kernel (the only required GPU kernel – everything else uses
   PyTorch which already has highly‑optimized kernels).
5. The triangular multiplicative update reduces to a batched matrix‑multiply:
   for each hidden channel h,
   out[..., i, j, h] = Σ_k left[..., i, k, h] * right[..., j, k, h],
   i.e. out_h = left_h @ right_hᵀ.
   This is performed with a single torch.bmm on a reshaped view
   (B*H, N, N) → (B*H, N, N).
6. Layer‑norm over the hidden dimension, gate with out_gate (again using the
   Triton element‑wise kernel), and a final linear projection back to the
   original channel dimension C.
7. Return the result tensor of shape (B, N, N, C).

The Triton kernel `elementwise_mul_kernel` multiplies two tensors of identical
shape element‑wise, handling arbitrary sizes via a 1‑D grid.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication
# ----------------------------------------------------------------------
@triton.jit
def elementwise_mul_kernel(
    a_ptr,               # *pointer* to tensor A (contiguous)
    b_ptr,               # *pointer* to tensor B (contiguous)
    c_ptr,               # *pointer* to output tensor C (contiguous)
    size,                # total number of elements (int)
    BLOCK_SIZE: tl.constexpr,   # compile‑time constant, block size
):
    """C = A * B  (element‑wise)"""
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offsets = block_start + tl.arange(0, BLOCK_SIZE)

    mask = offsets < size
    a = tl.load(a_ptr + offsets, mask=mask)
    b = tl.load(b_ptr + offsets, mask=mask)
    tl.store(c_ptr + offsets, a * b, mask=mask)


def mul_triton(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """
    Element‑wise multiply two tensors of identical shape using the Triton kernel.
    The inputs are flattened to 1‑D contiguous tensors before launching.
    """
    assert a.shape == b.shape, "Shapes must match for element‑wise multiplication"
    # Ensure contiguous layout – Triton works with flat indexing.
    a_cont = a.contiguous()
    b_cont = b.contiguous()
    out = torch.empty_like(a_cont)

    numel = a_cont.numel()
    BLOCK_SIZE = 65536  # 64 K elements per program; works for FP32/FP16
    grid = lambda meta: (triton.cdiv(numel, meta["BLOCK_SIZE"]),)

    elementwise_mul_kernel[grid](
        a_cont,
        b_cont,
        out,
        numel,
        BLOCK_SIZE=BLOCK_SIZE,
    )
    return out.view_as(a)  # reshape back to original dimensions


# ----------------------------------------------------------------------
# Entry‑point: custom Triton‑accelerated TriMul forward pass
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape (B, N, N, C)
        - mask         : torch.Tensor of shape (B, N, N) – may be ignored
        - weights      : dict of model weight tensors
        - config       : dict with keys:
            * "dim"         – C (input/output channel dimension)
            * "hidden_dim"  – H (intermediate hidden dimension)
            * "nomask"      – bool, if True mask is not applied

    Returns
    -------
    torch.Tensor
        Tensor of shape (B, N, N, C) – the result of the TriMul operation.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Retrieve the weight tensors
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]

    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (C)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # shape: (B, N, N, C)

    # ------------------------------------------------------------------
    # 2) Linear projections to the hidden dimension H
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # (B, N, N, H)
    right = F.linear(x, right_proj_weight)  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over H)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape (B, N, N) → (B, N, N, 1) to broadcast over H
        mask_f = mask.to(dtype=left.dtype, device=left.device).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute gating values (sigmoid(linear(x)))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,H)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight)) # (B,N,N,H)
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 5) Fuse element‑wise gating with a Triton kernel
    # ------------------------------------------------------------------
    left = mul_triton(left, left_gate)   # left = left * left_gate
    right = mul_triton(right, right_gate)  # right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Triangular multiplicative update:
    #    out_h = left_h @ right_hᵀ   for each hidden channel h
    #    Achieved via a single batched matrix multiply after reshaping.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # left.shape == (B, N, N, H)
    # Merge batch and hidden dimension: (B*H, N, N)
    left_flat = left.reshape(B * hidden_dim, N, N)
    right_flat = right.reshape(B * hidden_dim, N, N)

    # Compute out_flat = left_flat @ right_flatᵀ   → (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore original 4‑D shape (B, N, N, H)
    out = out_flat.reshape(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 7) Output layer‑norm over hidden dimension H
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )  # (B, N, N, H)

    # ------------------------------------------------------------------
    # 8) Apply the output gate (again via Triton)
    # ------------------------------------------------------------------
    out = mul_triton(out, out_gate)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to C channels
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_weight)  # (B, N, N, C)

    return out
