def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr):
    
    idx = tl.program_id(0) * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    mask = idx < B * N * N * H
    
    # Compute coordinates
    # total elements = B * N * N * H
    # We can compute b = (idx // (N*N*H))
    # idx2 = idx % (N*N*H)
    # i = idx2 // (N*H)
    # idx3 = idx2 % (N*H)
    # j = idx3 // H
    # h = idx3 % H
    offset = idx
    tmp = offset // (N * N * H)
    b = tmp
    rem = offset - b * (N * N * H)
    i = rem // (N * H)
    rem2 = rem - i * (N * H)
    j = rem2 // H
    h = rem2 - j * H
    
    # Compute pointers:
    left_offset = b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h

    left = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

    # mask broadcast; each hidden element uses same mask value.
    # Multiply
    left = left * m
    right = right * m
    tl.store(left_ptr + left_offset, left, mask=mask)
    tl.store(right_ptr + right_offset, right, mask=mask)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Normalize input
    # Using weight and bias:
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # ensure weight/bias shape matches dim
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Compute projections and gates:
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    # Elementwise multiplication
    left = left_proj * left_gate
    right = right_proj * right_gate
    # Apply mask if needed
    if not config.get("nomask", True) and mask is not None:
        # Ensure mask shape broadcast
        mask = mask.to(dtype).unsqueeze(-1)  # shape (B,N,N,1)
        # Use Triton kernel to multiply left and right by mask (in-place)
        apply_mask_triton(left, right, mask.squeeze(-1))
    # Contraction:
    # Reshape for batched matmul
    # left: (B,N,N,H) -> (B*H,N,N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_perm.view(bs * hidden_dim, N, N)
    right_flat = right_perm.view(bs * hidden_dim, N, N)
    # Compute matmul left * right^T:
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    # Reshape back
    out = out_flat.view(bs, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # (B,N,N,H)
    # to_out_norm:
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # out_gate
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    out = out_norm * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

import torch
import triton
import triton.language as tl

def apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor):
    """Multiply left and right tensors elementwise by a broadcasted mask using a Triton kernel.

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape (B, N, N, H), will be modified in‑place.
    right : torch.Tensor
        Tensor of shape (B, N, N, H), will be modified in‑place.
    mask : torch.Tensor
        Tensor of shape (B, N, N). Should be of the same dtype as ``left``.
    """
    B, N, N2, H = left.shape
    assert N2 == N, "left must be 4‑D (B,N,N,H)"
    total = B * N * N * H

    # Strides in elements (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_mask = mask.stride()

    # Triton launch configuration
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    mask_mul_kernel[grid](
        left, right, mask,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_mask[0], stride_mask[1], stride_mask[2],
        BLOCK_SIZE=1024,
    )
    # Ensure kernel finishes before we continue
    torch.cuda.synchronize()


@triton.jit
def mask_mul_kernel(
    left_ptr, right_ptr, mask_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr
):
    """
    In‑place multiplication of ``left`` and ``right`` tensors by a broadcasted mask.
    Each thread processes one element (b,i,j,h) of the ``left``/``right`` tensors.
    """
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Compute coordinates (b,i,j,h) from a flat index.
    stride_NNH = N * N * H
    stride_NH = N * H

    b = tl.div(offs, stride_NNH)
    rem = offs - b * stride_NNH

    i = tl.div(rem, stride_NH)
    rem2 = rem - i * stride_NH

    j = tl.div(rem2, H)
    h = rem2 - j * H

    # Offsets into the three tensors
    left_offset = (
        b * stride_left_b
        + i * stride_left_i
        + j * stride_left_j
        + h * stride_left_h
    )
    right_offset = (
        b * stride_right_b
        + i * stride_right_i
        + j * stride_right_j
        + h * stride_right_h
    )
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # Load, multiply, store
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    m_val = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    left_val = left_val * m_val
    right_val = right_val * m_val

    tl.store(left_ptr + left_offset, left_val, mask=mask)
    tl.store(right_ptr + right_offset, right_val, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” operator.

    The implementation follows the reference PyTorch module:
        1. Layer‑normalise the input.
        2. Linear “left” and “right” projections + element‑wise sigmoid gates.
        3. (Optional) Apply a pairwise mask using a small Triton kernel.
        4. Compute the einsum ``... i k d, ... j k d -> ... i j d`` by
           reshaping the tensors and calling a batched matrix multiplication.
        5. A second layer‑norm, a sigmoid gate, and a final linear projection
           back to the original feature dimension.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be ignored)
        - weights      : dict of model weight tensors
        - config       : dict containing ``dim`` and ``hidden_dim`` and ``nomask`` flag

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----- 1. Input layer‑norm ------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # torch.layer_norm expects normalized_shape = (dim,)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ----- 2. Left / right projections + gates --------------------------------
    # Projection matrices (out_features = hidden_dim)
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates (sigmoid of a second linear)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    # Element‑wise combine projection and gate
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ----- 3. Optional mask (using Triton) ------------------------------------
    if not nomask and mask is not None:
        # Ensure mask dtype matches the tensors and is on the same device
        mask = mask.to(dtype=dtype, device=device)
        # Apply in‑place mask multiplication
        apply_mask_triton(left, right, mask)

    # ----- 4. Pairwise contraction (einsum) -----------------------------------
    # Reshape tensors to (B*H, N, N) for a batched matmul.
    # Permute to bring the hidden dimension next to batch.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    left_flat = left_t.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N)  # (B*H, N, N)

    # Batched matrix multiplication with the second operand transposed on the last two axes
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))  # (B*H, N, N)

    # Reshape back to (B, N, N, H)
    out_hidden = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)

"""
TriMul (outgoing) forward pass with a tiny Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B,N,N,dim]).
2. Compute two linear projections (left/right) and two sigmoid gates.
   left = (x @ W_left_proj) * sigmoid(x @ W_left_gate)
   right = (x @ W_right_proj) * sigmoid(x @ W_right_gate)
3. (Optional) apply a pair‑wise mask.  This is the only operation that
   runs inside a custom Triton kernel: we multiply the ``left`` and ``right``
   tensors element‑wise by a broadcasted mask of shape [B,N,N].
4. Perform the “einsum”:
      out_{b,i,j,d} = Σ_k left_{b,i,k,d} * right_{b,j,k,d}
   This is a batched matrix multiplication on the (i,k) and (j,k) axes.
   We reshape the tensors to (B*H, N, N) and call ``torch.bmm``.
5. Layer‑norm the intermediate result (hidden dimension H), apply a third
   sigmoid gate (computed from the normalized input), and finally a linear
   projection back to the original ``dim``.
6. Return the tensor of shape [B,N,N,dim].

Only step 3 is implemented in Triton; the heavy GEMM in step 4 uses
torch.bmm (highly optimized on H100).  The kernel works for any batch
size ≤ 2, any sequence length up to 1024 and hidden dimensions supported
by the reference model.
"""

import torch
import triton
import triton.language as tl


def apply_mask_triton(left: torch.Tensor, right: torch.Tensor, mask: torch.Tensor) -> None:
    """
    In‑place multiplication of ``left`` and ``right`` by ``mask`` using Triton.

    Parameters
    ----------
    left : torch.Tensor
        Tensor of shape (B, N, N, H) – will be modified in‑place.
    right : torch.Tensor
        Tensor of shape (B, N, N, H) – will be modified in‑place.
    mask : torch.Tensor
        Tensor of shape (B, N, N).  Must be the same dtype as ``left``.
    """
    B, N, N2, H = left.shape
    assert N2 == N, "left must be 4‑D (B,N,N,H)"
    total = B * N * N * H

    # Strides in *elements* (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_mask = mask.stride()

    # One‑dimensional grid – each thread processes one element of the flattened tensor
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK_SIZE"]),)

    mask_mul_kernel[grid](
        left,
        right,
        mask,
        B,
        N,
        H,
        stride_left[0],
        stride_left[1],
        stride_left[2],
        stride_left[3],
        stride_right[0],
        stride_right[1],
        stride_right[2],
        stride_right[3],
        stride_mask[0],
        stride_mask[1],
        stride_mask[2],
        BLOCK_SIZE=1024,
    )
    # Ensure the asynchronous kernel completes before we read the tensors again.
    torch.cuda.synchronize()


@triton.jit
def mask_mul_kernel(
    left_ptr,
    right_ptr,
    mask_ptr,
    B,
    N,
    H,
    stride_left_b,
    stride_left_i,
    stride_left_j,
    stride_left_h,
    stride_right_b,
    stride_right_i,
    stride_right_j,
    stride_right_h,
    stride_mask_b,
    stride_mask_i,
    stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    """Element‑wise ``left *= mask`` and ``right *= mask`` (broadcast over H)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # Compute (b,i,j,h) from the flat index.
    stride_NNH = N * N * H
    stride_NH = N * H

    b = tl.div(offs, stride_NNH)
    rem = offs - b * stride_NNH

    i = tl.div(rem, stride_NH)
    rem2 = rem - i * stride_NH

    j = tl.div(rem2, H)
    h = rem2 - j * H

    # Offsets into the three tensors.
    left_offset = (
        b * stride_left_b
        + i * stride_left_i
        + j * stride_left_j
        + h * stride_left_h
    )
    right_offset = (
        b * stride_right_b
        + i * stride_right_i
        + j * stride_right_j
        + h * stride_right_h
    )
    mask_offset = b * stride_mask_b + i * stride_mask_i + j * stride_mask_j

    # Load, multiply, store.
    left_val = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    right_val = tl.load(right_ptr + right_offset, mask=mask, other=0.0)
    m_val = tl.load(mask_ptr + mask_offset, mask=mask, other=0.0)

    left_val = left_val * m_val
    right_val = right_val * m_val

    tl.store(left_ptr + left_offset, left_val, mask=mask)
    tl.store(right_ptr + right_offset, right_val, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N]  (may be dummy when nomask=True)
        - weights      : dict containing all linear / layer‑norm parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim].
    """
    # --------------------------------------------------------------------- #
    # 1. Unpack arguments
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # --------------------------------------------------------------------- #
    # 2. Input layer‑norm
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # (B,N,N,dim)

    # --------------------------------------------------------------------- #
    # 3. Linear projections + sigmoid gates
    # --------------------------------------------------------------------- #
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))

    left = left_proj * left_gate          # (B,N,N,H)
    right = right_proj * right_gate       # (B,N,N,H)

    # --------------------------------------------------------------------- #
    # 4. Optional mask (handled by Triton)
    # --------------------------------------------------------------------- #
    if not nomask and mask is not None:
        # Cast mask to the same dtype as the tensors and keep it on the same device.
        mask = mask.to(dtype=dtype, device=device)
        apply_mask_triton(left, right, mask)

    # --------------------------------------------------------------------- #
    # 5. Pairwise contraction (einsum) via batched matmul
    # --------------------------------------------------------------------- #
    # Bring hidden dimension next to batch for a tidy reshape.
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, N, N)
    right_t = right.permute(0, 3, 1, 2).contiguous() # (B, H, N, N)

    # Collapse batch and hidden into a single dimension.
    left_flat = left_t.view(B * hidden_dim, N, N)    # (B*H, N, N)
    right_flat = right_t.view(B * hidden_dim, N, N) # (B*H, N, N)

    # GEMM: (B*H, N, N) @ (B*H, N, N)^T -> (B*H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Reshape back to (B,N,N,H)
    out_hidden = (
        out_flat.view(B, hidden_dim, N, N)
        .permute(0, 2, 3, 1)
        .contiguous()
    )  # (B, N, N, H)

    # --------------------------------------------------------------------- #
    # 6. Second layer‑norm, output gate and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )

    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    out_gated = out_norm * out_gate

    # Final linear back to the original feature dimension.
    out = torch.nn.functional.linear(out_gated, weights["to_out.weight"])

    return out
