def layernorm_kernel(
    x_ptr,
    y_ptr,
    weight_ptr,
    bias_ptr,
    epsilon,
    stride_batch,
    stride_i,
    stride_j,
    stride_c,
    dim,
    BLOCK_SIZE: tl.constexpr
):
    # program ids
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # compute base pointer for this (b,i,j) row
    base_ptr = x_ptr + b * stride_batch + i * stride_i + j * stride_j
    out_ptr = y_ptr + b * stride_batch + i * stride_i + j * stride_j

    sum_val = tl.float32(0.0)
    sum_sq = tl.float32(0.0)

    num_blocks = tl.cdiv(dim, BLOCK_SIZE)

"""
TriMul (outgoing) forward pass implemented with Triton for the LayerNorm steps.
The algorithm follows the reference PyTorch implementation:

1. Layer‑norm the 4‑D input tensor (over the last dimension).
2. Linear projections for left/right tensors and for three gates.
3. Optional element‑wise mask on left/right (mask not applied to gates).
4. Apply the left/right gates (sigmoid) to the projected tensors.
5. Compute the cubic contraction:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is realized as a batched matrix multiplication (B*H, N, N) @ (B*H, N, N).
6. Layer‑norm the result (over hidden dimension), multiply by the output gate,
   and a final linear projection back to the original feature dimension.

Only the expensive LayerNorms are written in Triton, while the remaining
operations use highly‑optimised PyTorch kernels (linear, sigmoid, bmm).  This
ensures correctness and competitive runtime on an H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _layernorm_kernel(
    x_ptr,                     # const *float (input)
    y_ptr,                     # *float (output)
    weight_ptr,                # const *float (γ)
    bias_ptr,                  # const *float (β)
    epsilon,                   # float32
    stride_batch,              # int (stride over dim 0)
    stride_i,                  # int (stride over dim 1)
    stride_j,                  # int (stride over dim 2)
    stride_c,                  # int (stride over dim 3, the channel dim)
    dim,                       # int (size of channel dimension)
    BLOCK_C: tl.constexpr,     # Compile‑time block size for the channel axis
):
    """
    Performs LayerNorm over the last dimension of a 4‑D tensor.
    Grid shape: (batch, seq_len, seq_len) → each program processes one (b,i,j) row.
    """
    # ----- program IDs -----
    b = tl.program_id(0)
    i = tl.program_id(1)
    j = tl.program_id(2)

    # ----- base pointers for the (b,i,j) row -----
    row_ptr = x_ptr + b * stride_batch + i * stride_i + j * stride_j
    out_ptr = y_ptr + b * stride_batch + i * stride_i + j * stride_j

    # ----- first pass: compute mean & variance -----
    sum_val = tl.float32(0.0)
    sum_sq  = tl.float32(0.0)

    # number of blocks needed to cover the channel dimension
    num_blocks = tl.cdiv(dim, BLOCK_C)

    for blk in range(0, num_blocks):
        col = blk * BLOCK_C + tl.arange(0, BLOCK_C)          # absolute channel index
        mask = col < dim

        # load a tile of the row (shape: [BLOCK_C])
        x = tl.load(row_ptr + col * stride_c, mask=mask, other=0.0)

        # reduction across the block (sum of values and sum of squares)
        sum_val += tl.sum(x)
        sum_sq  += tl.sum(x * x)

    mean = sum_val / dim
    var  = sum_sq / dim - mean * mean
    inv_std = 1.0 / tl.sqrt(var + epsilon)

    # ----- second pass: write normalized result with γ & β -----
    for blk in range(0, num_blocks):
        col = blk * BLOCK_C + tl.arange(0, BLOCK_C)
        mask = col < dim

        x = tl.load(row_ptr + col * stride_c, mask=mask, other=0.0)
        w = tl.load(weight_ptr + col, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr   + col, mask=mask, other=0.0)

        # (x-mean) * inv_std * γ + β
        y = (x - mean) * inv_std
        y = y * w + b_

        tl.store(out_ptr + col * stride_c, y, mask=mask)


def custom_kernel(data):
    """
    Executes the TriMul outgoing operation.
    Input:
        data = (x, mask, weights, config)
            x        : torch.Tensor[bs, L, L, dim] (float32)
            mask     : torch.Tensor[bs, L, L] (bool or 0/1) – may be unused
            weights  : dict of pre‑loaded weight tensors
            config   : dict with keys "dim", "hidden_dim", "nomask"
    Returns:
        torch.Tensor[bs, L, L, dim] – result of the forward pass.
    """
    # unpack
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    bs, L, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    epsilon = 1e-5
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1) LayerNorm over the input (over dim)
    # ----------------------------------------------------------------------
    x_norm = torch.empty_like(x)
    stride_b, stride_i, stride_j, stride_c = x.stride()  # element strides

    # launch kernel: one program per (b, i, j) entry
    grid = (bs, L, L)
    BLOCK_C = 128  # enough for dim up to 384

        x,
        x_norm,
        weights["norm.weight"],
        weights["norm.bias"],
        epsilon,
        stride_b,
        stride_i,
        stride_j,
        stride_c,
        dim,
        BLOCK_C=BLOCK_C,
    )

    # ----------------------------------------------------------------------
    # 2) Linear projections and gates (all done with torch for speed)
    # ----------------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B,L,L,H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B,L,L,H]

    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)               # [B,L,L,1]
        left  = left * mask_f
        right = right * mask_f

    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # 3) Core cubic contraction: out = Σ_k left[...,k,:] * right[...,k,:]

    right_flat = right.permute(0, 3, 2, 1).reshape(bs * hidden_dim, L, L)

    out_flat = torch.bmm(left_flat, right_flat)   # (B*H, L, L)

    # Reshape back to [B, L, L, H]
    out = out_flat.view(bs, hidden_dim, L, L).permute(0, 2, 3, 1)  # [B,L,L,H]

    # ----------------------------------------------------------------------
    # 4) LayerNorm over the hidden dimension, multiply by out_gate
    # ----------------------------------------------------------------------
    out_norm = torch.empty_like(out)
    stride_b_o, stride_i_o, stride_j_o, stride_c_o = out.stride()

    grid_out = (bs, L, L)
    BLOCK_CH = 128  # hidden_dim ≤ 384
    _layernorm_kernel[grid_out](
        out,
        out_norm,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        epsilon,
        stride_b_o,
        stride_i_o,
        stride_j_o,
        stride_c_o,
        hidden_dim,
        BLOCK_C=BLOCK_CH,
    )

    # Multiply by the output gate (broadcasted over the hidden dim)
    out_norm = out_norm * out_gate

    # ----------------------------------------------------------------------
    # 5) Final linear projection back to original dim
    # ----------------------------------------------------------------------
    output = F.linear(out_norm, weights["to_out.weight"])  # [B, L, L, dim]

    return output
```

**Explanation of key components**

* `_layernorm_kernel` – a Triton kernel that normalises a single (b,i,j) row.
  It performs two passes: first gathers the mean / variance using `tl.sum`,
  then writes the normalised values applying the learnable scale (`γ`) and
  shift (`β`).  The kernel works for any channel size (`dim` or `hidden_dim`)
  and respects arbitrary tensor strides.

* In `custom_kernel` the two LayerNorms (on the input and on the intermediate
  tensor) are executed by the Triton kernel, while all linear projections,
  sigmoids, masking, the cubic “TriMul” contraction and the final projection
  are performed with highly‑optimised PyTorch primitives.  The cubic contraction
  is rewritten as a batched GEMM (`torch.bmm`) which exploits the GPU’s
  matrix‑multiply engine and avoids the O(N³) Python loops.
