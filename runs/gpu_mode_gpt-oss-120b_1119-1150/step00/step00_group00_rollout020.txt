def triamul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # i block index
    pid_n = tl.program_id(axis=1)  # j block index
    pid_bc = tl.program_id(axis=2)  # batch * hidden

    num_blocks_m = tl.cdiv(N, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)

    # compute base offsets
    b = pid_bc // H
    d = pid_bc % H

    # compute i, j offset start indices
    off_i = pid * BLOCK_M
    off_j = pid_n * BLOCK_N

    # compute pointers base
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # Loop over K
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute offsets
        off_k = k * BLOCK_K

        # indices
        i = off_i + tl.arange(0, BLOCK_M)
        j = off_j + tl.arange(0, BLOCK_N)
        kk = off_k + tl.arange(0, BLOCK_K)

        # mask out-of-bounds
        i_mask = i < N
        j_mask = j < N
        k_mask = kk < N

        # load left block (M,K)
        left_offs = i[:, None] * stride_li + kk[None, :] * stride_lk

import triton
import triton.language as tl

@triton.jit
def triamul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bc = tl.program_id(2)

    # decode batch and hidden dim
    b = pid_bc // H
    d = pid_bc % H

    # offsets
    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # ptr bases for this (b,d)
    left_base = left_ptr + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base = out_ptr + b * stride_ob + d * stride_od

    # Create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        # range vectors
        i = i_offset + tl.arange(0, BLOCK_M)
        j = j_offset + tl.arange(0, BLOCK_N)
        kk = k_offset + tl.arange(0, BLOCK_K)

        i_mask = i < N
        j_mask = j < N
        k_mask = kk < N

        # load left block (M, K)
        left_offs = i[:, None] * stride_li + kk[None, :] * stride_lk
        left = tl.load(left_base + left_offs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        right_offs = kk[:, None] * stride_rk + j[None, :] * stride_rj
        right = tl.load(right_base + right_offs, mask=k_mask[:, None] & j_mask[None, :], other=0.0)

        acc += tl.dot(left, right)

    out_i = i_offset + tl.arange(0, BLOCK_M)
    out_j = j_offset + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype  # assume float32

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # layer norm 1
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask
    if not nomask:
        mask = mask_tensor.to(dtype).unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask

    # gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # N = seq_len

    # strides
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()
    out_fp16 = torch.empty_like(left_fp16)  # shape B,N,N,hidden_dim, dtype half
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # block sizes: tune
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim )
    )

    triamul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )
    # Wait for kernel
    torch.cuda.synchronize()

    # convert to float32 for subsequent layers
    out = out_fp16.to(torch.float32)

    # to_out_norm
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # gating
    out = out * out_gate
    # final linear
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

"""
High‑performance Triton implementation of the outgoing TriMul operation.

The kernel computes, for each batch b and hidden channel d,
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] .
We fuse this reduction into a tiled GEMM using `tl.dot`, while the
remaining linear layers, sigmoids and LayerNorms are performed in
PyTorch.  The kernel works on FP16 tensors (the heavy multiplication)
and accumulates in FP32 for numerics.
"""

import torch
import triton
import triton.language as tl

@triton.jit
def triamul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_ld,
    stride_rb, stride_rj, stride_rk, stride_rd,
    stride_ob, stride_oi, stride_oj, stride_od,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute the outer‑product reduction used in TriMul.

    The kernel is a batched matrix‑multiply over the reduction axis ``k``.
    Each program instance works on a tile (BLOCK_M, BLOCK_N) of the
    output matrix for a particular (batch, hidden) pair.
    """
    pid_i = tl.program_id(0)   # tile index along the i dimension
    pid_j = tl.program_id(1)   # tile index along the j dimension
    pid_bc = tl.program_id(2)  # packed (batch, hidden) index

    # decode batch and hidden channel
    b = pid_bc // H
    d = pid_bc % H

    # starting offsets for the current tile
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # base pointers that already include batch and hidden offsets
    left_base  = left_ptr  + b * stride_lb + d * stride_ld
    right_base = right_ptr + b * stride_rb + d * stride_rd
    out_base   = out_ptr   + b * stride_ob + d * stride_od

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over the reduction dimension k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k0 = k * BLOCK_K

        # index vectors
        i = i0 + tl.arange(0, BLOCK_M)
        j = j0 + tl.arange(0, BLOCK_N)
        kk = k0 + tl.arange(0, BLOCK_K)

        # out‑of‑bounds masks
        i_mask  = i < N
        j_mask  = j < N
        kk_mask = kk < N

        # -----------------------------------------------------------------
        # Load a (M, K) tile from ``left`` (i varies, k varies)
        # -----------------------------------------------------------------
        left_offs = i[:, None] * stride_li + kk[None, :] * stride_lk
        left_tile = tl.load(
            left_base + left_offs,
            mask=i_mask[:, None] & kk_mask[None, :],
            other=0.0
        )  # FP16

        # -----------------------------------------------------------------
        # Load a (K, N) tile from ``right`` (k varies, j varies)
        # -----------------------------------------------------------------
        right_offs = kk[:, None] * stride_rk + j[None, :] * stride_rj
        right_tile = tl.load(
            right_base + right_offs,
            mask=kk_mask[:, None] & j_mask[None, :],
            other=0.0
        )  # FP16

        # -----------------------------------------------------------------
        # Matrix‑multiply the two tiles.
        # tl.dot performs a fused FP16×FP16 -> FP32 dot product.
        # -----------------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)

    # -----------------------------------------------------------------
    # Write the result back, converting to FP16 on the fly.
    # -----------------------------------------------------------------
    out_i = i0 + tl.arange(0, BLOCK_M)
    out_j = j0 + tl.arange(0, BLOCK_N)

    out_i_mask = out_i < N
    out_j_mask = out_j < N

    out_offs = out_i[:, None] * stride_oi + out_j[None, :] * stride_oj
    tl.store(
        out_base + out_offs,
        acc.to(tl.float16),
        mask=out_i_mask[:, None] & out_j_mask[None, :]
    )

    """Forward pass of the “outgoing” TriMul module using a Triton kernel.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns:
        Tensor of shape (batch, seq_len, seq_len, dim) with dtype float32.
    """
    input_tensor, mask_tensor, weights, config = data
    # Configuration
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # -----------------------------------------------------------------
    # 2. Linear projections (float32)
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted over the hidden dim)
    # -----------------------------------------------------------------
    if not nomask:
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)  # [B,N,N,1]

        right = right * mask

    # -----------------------------------------------------------------

    # -----------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Cast to FP16 and launch Triton kernel for the einsum
    # -----------------------------------------------------------------
    left_fp16 = left.to(torch.float16).contiguous()
    right_fp16 = right.to(torch.float16).contiguous()

    B, N, _, _ = left_fp16.shape  # batch, seq_len, ..., hidden_dim

    # Prepare strides (in elements)
    stride_lb, stride_li, stride_lk, stride_ld = left_fp16.stride()
    stride_rb, stride_rj, stride_rk, stride_rd = right_fp16.stride()

    out_fp16 = torch.empty_like(left_fp16)  # will hold the result of the contraction
    stride_ob, stride_oi, stride_oj, stride_od = out_fp16.stride()

    # Block sizes – these are a good trade‑off on H100 (tensor‑core friendly)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Grid: (tiles_i, tiles_j, batch*hidden)
    grid = (
        ( (N + BLOCK_M - 1) // BLOCK_M,
          (N + BLOCK_N - 1) // BLOCK_N,
          B * hidden_dim )
    )

    # Launch kernel
    triamul_kernel[grid](
        left_fp16,
        right_fp16,
        out_fp16,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_ld,
        stride_rb,
        stride_rj,
        stride_rk,
        stride_rd,
        stride_ob,
        stride_oi,
        stride_oj,
        stride_od,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,          # H100 has many warps per SM
    )
    # Synchronise to make sure kernel finished
    torch.cuda.synchronize()

    # Convert back to FP32 for the rest of the pipeline
    out = out_fp16.to(torch.float32)

    # -----------------------------------------------------------------
    # 6. Second LayerNorm (hidden_dim)
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )

    # -----------------------------------------------------------------
    # 7. Out‑gate and final projection back to ``dim``
    # -----------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
