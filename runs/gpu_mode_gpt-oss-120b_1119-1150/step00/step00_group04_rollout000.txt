    norm_bias = weights["norm.bias"]

    # Projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])

"""
TriMul forward (outgoing) implementation with a fused Triton kernel.

Algorithm
---------
1. Layer‑norm the input tensor (over the hidden dimension `dim`).
2. Compute four linear projections:
      left  = x @ W_left   (shape [..., hidden_dim])
      right = x @ W_right
      left_gate  = sigmoid(x @ W_left_gate)
      right_gate = sigmoid(x @ W_right_gate)
      out_gate   = sigmoid(x @ W_out_gate)
3. Optionally apply an input mask (broadcast on the hidden dimension).
4. Gating:   left  = left  * left_gate
            right = right * right_gate
5. Compute the expensive cubic term:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is executed by a custom Triton kernel that tiles the
   `i` and `j` dimensions (BLOCK_M × BLOCK_N) and iterates over
   the `k` dimension in BLOCK_K steps, accumulating with `tl.dot`
   (the right operand is transposed, giving the required sum over `k`).
   The kernel processes a single `(batch, hidden)` pair per program
   and writes the result directly into the output tensor.
6. Layer‑norm the cubic term over `hidden_dim`.
7. Multiply by `out_gate` (broadcasted) and apply the final linear
   projection back to the original `dim`.

The kernel operates in FP32 (all tensors are kept in float32 for
numerical stability) but the tiling strategy (64×64×64) gives good
occupancy on an H100.  The rest of the work (layer‑norms,
projections, sigmoid and the final linear) is handled by PyTorch,
which is efficient for the small‑ish per‑element work.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel for the cubic (einsum) term
# ----------------------------------------------------------------------
@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b_l, stride_i_l, stride_k_l, stride_h_l,
    stride_b_r, stride_i_r, stride_k_r, stride_h_r,
    stride_b_o, stride_i_o, stride_j_o, stride_h_o,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].

    Grid:
        (ceil(N/BLOCK_M), ceil(N/BLOCK_N), B*H) -> (pid_i, pid_j, pid_z)
    pid_z encodes (batch, hidden) = (b, d).
    """
    pid_i = tl.program_id(0)          # tile over i (rows)
    pid_j = tl.program_id(1)          # tile over j (cols)
    pid_z = tl.program_id(2)          # combined (batch, hidden)

    b = pid_z // H                    # batch index
    d = pid_z % H                     # hidden channel index

    # ------------------------------------------------------------------
    # Tile offsets
    # ------------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator for the BLOCK_M × BLOCK_N output tile
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: shape (BLOCK_M, BLOCK_K) -----
        left_ptrs = (
            left_ptr
            + b * stride_b_l
            + d * stride_h_l
            + offs_i[:, None] * stride_i_l
            + offs_k[None, :] * stride_k_l
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- right tile: shape (BLOCK_N, BLOCK_K) -----
        right_ptrs = (
            right_ptr
            + b * stride_b_r
            + d * stride_h_r
            + offs_j[:, None] * stride_i_r
            + offs_k[None, :] * stride_k_r
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ----- Multiply‑accumulate over k -----
        # right is (BLOCK_N, BLOCK_K); we need its transpose (BLOCK_K, BLOCK_N)
        # `trans_b=True` performs that transpose internally.
        acc += tl.dot(left, right, trans_b=True)

    # ------------------------------------------------------------------
    # Write the result tile back to global memory
    # ------------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_b_o
        + d * stride_h_o
        + offs_i[:, None] * stride_i_o
        + offs_j[None, :] * stride_j_o
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :],
    )


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the TriMul output.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    # Dimensionality
    dim = config["dim"]                # input channel dimension
    hidden_dim = config["hidden_dim"]  # projection dimension (c_z)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    # Shape after projection: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if mask is not None and not config.get("nomask", False):
        # mask shape [B, N, N] -> [B, N, N, 1] for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype=left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cubic term via Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for stride correctness
    left = left.contiguous()
    right = right.contiguous()

    B, N, _, _ = left.shape  # shape is [B, N, N, hidden_dim]

    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Kernel launch configuration
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim,                 # (batch, hidden) dimension
    )

    _trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=hidden_dim,
        # strides for left tensor
        stride_b_l=left.stride(0),
        stride_i_l=left.stride(1),
        stride_k_l=left.stride(2),
        stride_h_l=left.stride(3),
        # strides for right tensor
        stride_b_r=right.stride(0),
        stride_i_r=right.stride(1),
        stride_k_r=right.stride(2),
        stride_h_r=right.stride(3),
        # strides for output tensor
        stride_b_o=out.stride(0),
        stride_i_o=out.stride(1),
        stride_j_o=out.stride(2),
        stride_h_o=out.stride(3),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,   # 4 warps works well on H100 for this shape
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: LayerNorm over hidden_dim, gating, final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # broadcast over (B, N, N, hidden_dim)

    # Final linear back to original `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
