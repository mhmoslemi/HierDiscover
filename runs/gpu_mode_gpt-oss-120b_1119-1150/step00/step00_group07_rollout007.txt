def custom_kernel(data):
    input_tensor, mask, weights, config = data

    # Unpack config
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)  # might not be used; but if mask is None or size zero?

    # Do LayerNorm on last dim: norm
    # Considering the input shape [B,N,N,dim], we need to compute mean/var across dim.
    # We'll use torch.nn.functional.layer_norm with normalized shape = (dim,).
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left_proj_weight = weights["left_proj.weight"]   # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Mask (if mask is not None and not nomask)
    if not config.get("nomask", True):
        # mask shape [B,N,N]; expand dimension
        mask_exp = mask[..., None]   # broadcast to hidden dim
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # Compute out = left @ right^T across dimension k for each channel
    # Combine batch and hidden dims for using our triton kernel.
    B, N, _, _ = left.shape
    # Flatten batch & hidden dims: treat each as separate batch in kernel?
    # We'll call tri_mul_kernel which expects pointers with strides.
    # Compute out shape = [B,N,N,hidden_dim] (as float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=left.device)

    # Determine strides
    # left is contiguous in memory? shape (B,N,N,H). In PyTorch default layout is contiguous with last dim fastest.
    # So strides: (N*N*H, N*H, H, 1)?? Actually memory ordering: for shape (B,N,N,H) with default contiguous, stride of H axis = 1, stride of second N dimension = H, stride of first N dimension = N*H, stride of batch = N*N*H.
    # So we can get strides using left.stride().
    stride_b, stride_i, stride_j, stride_h = left.stride()

    # Launch kernel grid: (B*hidden_dim, ceil(N/BLOCK_M), ceil(N/BLOCK_N))
    # Choose block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32  # tile size for K

    grid = (B * hidden_dim, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    import triton.language as tl

    # We'll define kernel within function or outside but within same block.

    tri_mul_kernel[grid](
        left, right, out,
        B, N, hidden_dim,
        stride_b, stride_i, stride_j, stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,

    # After kernel, out is shape [B,N,N,H] as required

Define kernel:

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_b, stride_i, stride_j, stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # compute batch index and hidden index
    bc = tl.program_id(0)
    b = bc // H
    h = bc % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute row and col offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for within range
    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # compute k offset
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for K dimension
        mask_k = offs_k < N

        # compute pointers for left tile [M, K]
        left_ptrs = left_ptr + b * stride_b + offs_m[:, None] * stride_i + offs_k[None, :] * stride_j + h * stride_h
        # broadcast mask: shape (M,K)
        left_mask = mask_m[:, None] & mask_k[None, :]

        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # Compute right transposed tile [K, N] where right[j,k] -> transposed as (k,j)
        # right_ptrs: base + b*stride_b + offs_n[None, :] * stride_i + offs_k[:, None] * stride_j + h*stride_h
        right_ptrs = right_ptr + b * stride_b + offs_n[None, :] * stride_i + offs_k[:, None] * stride_j + h * stride_h
        # mask for right tile: shape (K,N)
        right_mask = mask_k[:, None] & mask_n[None, :]

        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # Compute dot product accumulate
        # left_tile shape (M,K), right_tile shape (K,N)
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_ptr + b * stride_b + offs_m[:, None] * stride_i + offs_n[None, :] * stride_j + h * stride_h
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=out_mask)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    stride_b, stride_i, stride_j, stride_h,
    N, H,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Compute batch and hidden channel index
    bc = tl.program_id(0)
    # compute batch and hidden
    b = bc // H
    h = bc % H

    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # Accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile: shape (M, K)
        left_ptrs = left_ptr + b * stride_b + offs_m[:, None] * stride_i + offs_k[None, :] * stride_j + h * stride_h
        left_mask = mask_m[:, None] & mask_k[None, :]
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # right tile transposed: shape (K, N)
        # Use strides swapped so that we load transposed view
        right_ptrs = right_ptr + b * stride_b + offs_k[:, None] * stride_j + offs_n[None, :] * stride_i + h * stride_h
        right_mask = mask_k[:, None] & mask_n[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # dot
        acc += tl.dot(left_tile, right_tile)

    # Store result
    out_ptrs = out_ptr + b * stride_b + offs_m[:, None] * stride_i + offs_n[None, :] * stride_j + h * stride_h
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
custom_kernel
==============

Implements the *outgoing* TriMul operation from AlphaFold3.

Algorithm
---------
1. **LayerNorm** over the last dimension of the input tensor `x` using the
   provided `norm.weight` and `norm.bias`.
2. **Linear projections** (no bias) to a hidden dimension `H`:
   * `left = x @ left_proj.weight.T`
   * `right = x @ right_proj.weight.T`
3. If a mask is present (`nomask=False`) it is broadcast and applied to
   both `left` and `right`.
4. **Gating** – three sigmoids are computed from `x` using the gate weights.
   `left` and `right` are multiplied element‑wise by their respective gates.
5. **Fused pairwise multiplication** (the core of TriMul):
   For each hidden channel `h` and batch `b`

        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

   This is a batch of `B*H` matrix multiplications `M @ Mᵀ` with
   `M ∈ ℝ^{N×N}`.  The heavy inner‑loop is performed by a Triton kernel
   (`tri_mul_kernel`) that tiles the matrices in the `M`, `N` and `K`
   dimensions.
6. **Post‑processing**
   * Layer‑norm the result over the hidden dimension (`to_out_norm`).
   * Multiply by the `out_gate` sigmoid.
   * Final linear projection back to `dim` (`to_out`).

All tensors are kept on the GPU; the expensive `N³` reduction is the only
operation executed in Triton.  Float16 is used for the large intermediate
tensors to reduce memory bandwidth while accumulation is performed in
float32 for numerical stability.

The kernel runs on an H100 (triton 3.3.1) and targets the micro‑second
regime for the provided test configurations.
"""
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matmul left @ rightᵀ across the hidden dimension
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,               # *const float16*  (B,N,N,H)
    right_ptr,              # *const float16*  (B,N,N,H)
    out_ptr,                # *float32*        (B,N,N,H)
    stride_b, stride_i, stride_j, stride_h,  # strides in elements
    N, H,                                 # sequence length, hidden dim
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Computes for a fixed (batch, hidden) pair:
        out[i, j] = Σ_k left[i, k] * right[j, k]
    with tiling along M, N and K.
    """
    # ---------------------------------------------------------------
    # Identify the (batch, hidden) pair this program processes
    # ---------------------------------------------------------------
    bc = tl.program_id(0)               # 0‑th axis: B * H programs
    b = bc // H                         # batch index
    h = bc % H                          # hidden‑channel index

    pid_m = tl.program_id(1)            # tile row index
    pid_n = tl.program_id(2)            # tile col index

    # ---------------------------------------------------------------
    # Offsets for the current tile
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ---------------------------------------------------------------
    # Loop over the reduction dimension K
    # ---------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for kk in range(0, num_k_tiles):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # ----- left tile: shape (M, K) ---------------------------------
        # address: batch * stride_b + i * stride_i + k * stride_j + h * stride_h
        left_ptrs = left_ptr + b * stride_b \
                              + offs_m[:, None] * stride_i \
                              + offs_k[None, :] * stride_j \
                              + h * stride_h
        left_mask = mask_m[:, None] & mask_k[None, :]
        # Load FP16 → FP32
        left_tile = tl.load(left_ptrs, mask=left_mask, other=0.0).to(tl.float32)

        # ----- right tile (transposed): shape (K, N) -------------------
        # We need right[j, k] accessed as rightᵀ[k, j].
        # address: batch * stride_b + k * stride_j + j * stride_i + h * stride_h
        right_ptrs = right_ptr + b * stride_b \
                                 + offs_k[:, None] * stride_j \
                                 + offs_n[None, :] * stride_i \
                                 + h * stride_h
        right_mask = mask_k[:, None] & mask_n[None, :]
        right_tile = tl.load(right_ptrs, mask=right_mask, other=0.0).to(tl.float32)

        # ----- accumulate ------------------------------------------------
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) → (M,N)

    # ---------------------------------------------------------------
    # Write back the result
    # ---------------------------------------------------------------
    out_ptrs = out_ptr + b * stride_b \
                         + offs_m[:, None] * stride_i \
                         + offs_n[None, :] * stride_j \
                         + h * stride_h
    out_mask = mask_m[:, None] & mask_n[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N]  (may be None)
        - weights      : dict[str, torch.Tensor] with all linear/LN weights
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # Linear projections (no bias) – cast to float16 for bandwidth saving
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]    # (H, dim)
    right_proj_weight = weights["right_proj.weight"]   # (H, dim)

    left  = F.linear(x, left_proj_weight)   # [B,N,N,H]   FP32
    right = F.linear(x, right_proj_weight)  # [B,N,N,H]

    # Reduce memory pressure by using half precision for the large tensors
    left  = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # ------------------------------------------------------------------
    # Optional masking – broadcast to hidden dimension
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B,N,N] → [B,N,N,1] → broadcast
        mask_broadcast = mask.to(left.dtype).unsqueeze(-1)
        left  = left * mask_broadcast
        right = right * mask_broadcast

    # ------------------------------------------------------------------
    # Gating (sigmoid) – computed in FP32 for stability, then cast back
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    # Gates are based on the *float32* normalized input `x`
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,H] FP32
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # later used after to_out_norm

    # Apply gates (cast to match left/right dtype)
    left  = left  * left_gate.to(left.dtype)
    right = right * right_gate.to(right.dtype)

    # ------------------------------------------------------------------
    # Core TriMul: left @ rightᵀ for every hidden channel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape  # N is seq_len, confirm square
    # Prepare output tensor (float32 for the rest of the pipeline)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    stride_b, stride_i, stride_j, stride_h = left.stride()   # (B,N,N,H) layout

    # Block sizes – chosen heuristically
    if N >= 512:
        BLOCK_M, BLOCK_N, BLOCK_K = 64, 64, 64
    else:
        BLOCK_M, BLOCK_N, BLOCK_K = 32, 32, 32

    grid = (
        B * hidden_dim,                              # program_id(0) – batch × hidden
        (N + BLOCK_M - 1) // BLOCK_M,                # program_id(1) – tiles over rows
        (N + BLOCK_N - 1) // BLOCK_N                 # program_id(2) – tiles over cols
    )

    # Launch the Triton kernel
    tri_mul_kernel[grid](
        left, right, out,
        stride_b, stride_i, stride_j, stride_h,
        N, hidden_dim,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: layer‑norm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    # Multiply by the out_gate (already sigmoid‑ed)
    out_scaled = out_norm * out_gate

    # Final projection back to original dimension `dim`
    to_out_weight = weights["to_out.weight"]          # (dim, hidden_dim)
    output = F.linear(out_scaled, to_out_weight)     # [B,N,N,dim]

    return output
