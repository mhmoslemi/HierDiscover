```python
import triton
import triton.language as tl

@triton.jit
def trilu_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,  # H hidden dim
    stride_bl, stride_i, stride_k, stride_h,
    stride_br, stride_i_r, stride_k_r, stride_h_r,
    stride_out_bl, stride_i_out, stride_j_out, stride_h_out,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute batch index b and hidden index h
    b = pid_b // H
    h = pid_b % H

    # compute base pointer offsets
    base_l = b * stride_bl + h * stride_h
    base_r = b * stride_br + h * stride_h_r
    base_o = b * stride_out_bl + h * stride_h_out

    # offsets for i and j
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_m = offs_m < N
    mask_n = offs_n < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        # k may exceed N boundary
        mask_k = offs_k < N

        # pointers
        left_tile_ptr = left_ptr + base_l + offs_m[:, None] * stride_i + offs_k[None, :] * stride_k
        right_tile_ptr = right_ptr + base_r + offs_n[None, :] * stride_i_r + offs_k[:, None] * stride_k_r

        # load, note we need to mask edges
        left_tile = tl.load(left_tile_ptr, mask=mask_m[:, None] & mask_k[None, :], other=0.0)
        right_tile = tl.load(right_tile_ptr, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        # multiply-accumulate
        acc += tl.dot(left_tile, right_tile)  # left_tile (M,K) dot right_tile (K,N) -> (M,N)

        # advance k offsets automatically due to ptr updates intangible

    # store
    out_ptrs = out_ptr + base_o + offs_m[:, None] * stride_i_out + offs_n[None, :] * stride_j_out
    tl.store(out_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

"""
TriMul (outgoing) implementation with a custom Triton kernel.

The heavy part of the TriMul forward pass is the tensor contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

which is a batched matrix multiplication of shape (N×N)×(N×N) for each
hidden channel `d`.  The kernel below fuses the three nested loops into a
single tiled GEMM using Triton.  All LayerNorms and linear projections are
performed with PyTorch (they are already heavily optimized).  The kernel
processes each (batch, hidden) pair as a separate program; tiles of size
BLOCK_M × BLOCK_N accumulate the inner product over K in blocks of
BLOCK_K.  The result is accumulated in FP32 for numerical stability.
"""

import torch
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batch*hidden GEMM of (N,N) @ (N,N).T  → (N,N)
# -------------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    # pointers
    left_ptr, right_ptr, out_ptr,
    # dimensions
    batch, hidden, N,
    # strides for left  [b, i, k, h]
    stride_bl, stride_li, stride_lk, stride_lh,
    # strides for right [b, j, k, h] (same layout as left)
    stride_br, stride_rj, stride_rk, stride_rh,
    # strides for out   [b, i, j, h]
    stride_ob, stride_oi, stride_oj, stride_oh,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_bn = tl.program_id(0)               # batch * hidden index
    pid_m  = tl.program_id(1)               # tile over i (M)
    pid_n  = tl.program_id(2)               # tile over j (N)

    # -----------------------------------------------------------------
    # decode batch index b and hidden index h from the flat id
    # -----------------------------------------------------------------
    b = pid_bn // hidden
    h = pid_bn % hidden

    # base offsets for this (b,h) pair
    base_l = b * stride_bl + h * stride_lh
    base_r = b * stride_br + h * stride_rh
    base_o = b * stride_ob + h * stride_oh

    # -----------------------------------------------------------------
    # tile offsets in the M and N dimensions
    # -----------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < N
    mask_n = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -----------------------------------------------------------------
    # loop over K dimension (the contracted axis)
    # -----------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # --------- load a tile of left (M × K) -----------------------
        # offset = base_l + i*stride_li + k*stride_lk
        left_tile_ptr = left_ptr + base_l \
                        + offs_m[:, None] * stride_li \
                        + offs_k[None, :] * stride_lk
        left_tile = tl.load(left_tile_ptr,
                            mask=mask_m[:, None] & mask_k[None, :],
                            other=0.0)

        # --------- load a tile of right transposed (K × N) -----------
        # we need right[j,k]  →  row=k, col=j
        # offset = base_r + k*stride_rk + j*stride_rj
        right_tile_ptr = right_ptr + base_r \
                         + offs_k[:, None] * stride_rk \
                         + offs_n[None, :] * stride_rj
        right_tile = tl.load(right_tile_ptr,
                             mask=mask_k[:, None] & mask_n[None, :],
                             other=0.0)

        # --------- multiply‑accumulate --------------------------------
        # left_tile: (M,K), right_tile: (K,N)
        acc += tl.dot(left_tile, right_tile, out_dtype=tl.float32)

    # -----------------------------------------------------------------
    # write the result
    # -----------------------------------------------------------------
    out_ptrs = out_ptr + base_o \
               + offs_m[:, None] * stride_oi \
               + offs_n[None, :] * stride_oj
    tl.store(out_ptrs,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])

# -------------------------------------------------------------------------
# Helper that launches the kernel
# -------------------------------------------------------------------------
def launch_trimul(left: torch.Tensor, right: torch.Tensor, out: torch.Tensor,
                  BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 64):
    """
    left, right, out  –  contiguous tensors of shape [B, N, N, H]
    """
    B, N, _, H = left.shape
    total = B * H                     # number of (batch, hidden) GEMMs

    grid = (total,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    trimul_kernel[grid](
        left_ptr   = left,
        right_ptr  = right,
        out_ptr    = out,
        batch      = B,
        hidden     = H,
        N          = N,
        # strides for left
        stride_bl  = left.stride(0),
        stride_li  = left.stride(1),
        stride_lk  = left.stride(2),
        stride_lh  = left.stride(3),
        # strides for right (same layout)
        stride_br  = right.stride(0),
        stride_rj  = right.stride(1),
        stride_rk  = right.stride(2),
        stride_rh  = right.stride(3),
        # strides for out
        stride_ob  = out.stride(0),
        stride_oi  = out.stride(1),
        stride_oj  = out.stride(2),
        stride_oh  = out.stride(3),
        BLOCK_M    = BLOCK_M,
        BLOCK_N    = BLOCK_N,
        BLOCK_K    = BLOCK_K,
    )
    return out

    Arguments
    ---------
    data : Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]
        - input tensor   : [B, N, N, dim]   (float32, CUDA)
        - mask tensor    : [B, N, N]       (bool/float, CUDA) – may be ignored
        - weights dict   : model parameters (LayerNorm, Linear, etc.)
        - config dict    : contains "dim", "hidden_dim", and optional "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] on the same device.
    """
    # -----------------------------------------------------------------
    # unpack
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    dim          = config["dim"]
    hidden_dim   = config["hidden_dim"]
    nomask       = config.get("nomask", False)

    # -----------------------------------------------------------------
    # convenient aliases for weights
    # -----------------------------------------------------------------
    norm_w       = weights["norm.weight"]
    norm_b       = weights["norm.bias"]
    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    # -----------------------------------------------------------------
    # 1. Input LayerNorm
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )   # -> [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2. Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = torch.nn.functional.linear(x, left_proj_w)   # [B, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # -----------------------------------------------------------------
    # 3. Optional mask (broadcasted on hidden dimension)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is [B, N, N]; convert to same dtype as x and add trailing dim
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)  # [B,N,N,1]
        left  = left * mask
        right = right * mask

    # -----------------------------------------------------------------
    # 4. Gates (sigmoid of linear transforms)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5. Core contraction – custom Triton kernel
    # -----------------------------------------------------------------
    # allocate output tensor for the hidden channel result
    out_hidden = torch.empty_like(left)   # same dtype & device, shape [B,N,N,hidden_dim]
    launch_trimul(left, right, out_hidden,
                  BLOCK_M=64, BLOCK_N=64, BLOCK_K=64)

    # -----------------------------------------------------------------
    # 6. Output LayerNorm, gating and final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_w)   # -> [B, N, N, dim]

    return out
