def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    a_batch_stride, a_row_stride, a_col_stride,
    b_batch_stride, b_row_stride, b_col_stride,
    c_batch_stride, c_row_stride, c_col_stride,
    M, N, K,

    total_batch = tl.cdiv(M, BLOCK_M) * tl.cdiv(N, BLOCK_N)
    batch_idx = pid // total_batch
    pid_in_batch = pid % total_batch
    pid_m = pid_in_batch // tl.cdiv(N, BLOCK_N)
    pid_n = pid_in_batch % tl.cdiv(N, BLOCK_N)
    # Offsets

    col_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

def custom_kernel(data):
    """
    Triton implementation of the outgoing TriMul module.

    The core computation is:
        out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]

    where left/right are linear projections of the normalized input,
    masked and gated. The reduction over k is performed with a batched
    matrix multiplication kernel written in Triton, which treats each
    (batch, hidden_dim) slice as an independent matrix‑multiply
    of shape (seq_len, seq_len). The rest of the operations (LayerNorm,
    gating, final linear) are performed with PyTorch.
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # -----------------------------------------------------------------
    # LayerNorm on last dimension
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=1e-5)

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)   # -> (..., hidden_dim)
    right = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Mask handling
    # -----------------------------------------------------------------
    nomask = config.get("nomask", False)
    if not nomask:
        # mask shape [B,N,N]; unsqueeze to broadcast on channel dim
        mask_uns = mask.unsqueeze(-1).to(dtype)  # (B,N,N,1)
        left = left * mask_uns
        right = right * mask_uns

    # -----------------------------------------------------------------
    # Gates
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Convert to half for the heavy matmul
    # -----------------------------------------------------------------
    # permute to bring channel dimension as batch axis for matmul
    B, N, _, _ = left.shape  # batch, seq_len, seq_len, hidden_dim
    left = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous() # (B, hidden_dim, N, N)

    # Transpose the second operand so that we sum over k
    right_t = right.transpose(-1, -2).contiguous()  # (B, hidden_dim, N, N)

    # Flatten batch & channel into a single batch dimension for the kernel
    batch_channel = B * hidden_dim
    left_flat = left.reshape(batch_channel, N, N).to(torch.float16)
    right_flat = right_t.reshape(batch_channel, N, N).to(torch.float16)

    # Output buffer (float32 accumulator)
    out_flat = torch.empty((batch_channel, N, N), device=device, dtype=torch.float32)

    # -----------------------------------------------------------------
    # Launch Triton batched GEMM kernel
    # -----------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    a_batch_stride = left_flat.stride(0)
    a_row_stride   = left_flat.stride(1)
    a_col_stride   = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_row_stride   = right_flat.stride(1)
    b_col_stride   = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_row_stride   = out_flat.stride(1)
    c_col_stride   = out_flat.stride(2)

    # Compute grid size
    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_channel * num_blocks_m * num_blocks_n

    batched_matmul_kernel[total_blocks](
        left_flat, right_flat, out_flat,
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_row_stride, b_col_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        N, N, N,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # -----------------------------------------------------------------
    # Reshape back to original layout (B,N,N,hidden_dim)
    # -----------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # Output layer norm, gate, and final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=1e-5)

    out = out * out_gate

    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)   # shape (B,N,N,dim)

    return out

"""
TriMul (outgoing) – Triton‑accelerated core of AlphaFold‑3.

The heavy part of TriMul is the reduction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of matrix‑multiplications of shape (seq_len, seq_len) for each
hidden‑dim channel.  This kernel treats each (batch, hidden_dim) pair as an
independent GEMM and performs the reduction with a custom Triton kernel.
All the surrounding LayerNorms, linear projections and gating are done with
PyTorch (they are cheap compared to the O(N³·C) reduction).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,                     # base pointers
    a_batch_stride, a_row_stride, a_col_stride,
    b_batch_stride, b_row_stride, b_col_stride,
    c_batch_stride, c_row_stride, c_col_stride,
    M, N, K,                                 # problem sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: C[b] = A[b] @ B[b]  for many batches.
    A : (batch, M, K)      (half‑precision)
    B : (batch, K, N)      (half‑precision, already transposed in host code)
    C : (batch, M, N)      (float32 accumulator)
    """
    pid = tl.program_id(0)                     # global program id

    # --------------------------------------------------------------
    # Decode batch index and the (i,j) tile within that batch
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    num_blocks_n = tl.cdiv(N, BLOCK_N)
    blocks_per_batch = num_blocks_m * num_blocks_n

    batch_idx = pid // blocks_per_batch
    pid_in_batch = pid % blocks_per_batch
    pid_m = pid_in_batch // num_blocks_n
    pid_n = pid_in_batch % num_blocks_n

    # --------------------------------------------------------------
    # Tile offsets
    # --------------------------------------------------------------
    row_offset = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    col_offset = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # --------------------------------------------------------------
    # Accumulator (float32 for precision)
    # --------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        k_offset = kk * BLOCK_K

        # ---- A tile ------------------------------------------------
        a_row = row_offset[:, None]                                 # (BM,1)
        a_col = k_offset + tl.arange(0, BLOCK_K)[None, :]           # (1,BK)
        a_ptrs = (
            a_ptr
            + batch_idx * a_batch_stride
            + a_row * a_row_stride
            + a_col * a_col_stride
        )
        a_mask = (row_offset < M)[:, None] & (k_offset + tl.arange(0, BLOCK_K) < K)[None, :]
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)                # (BM,BK)

        # ---- B tile (already transposed) ---------------------------
        b_row = k_offset + tl.arange(0, BLOCK_K)[:, None]          # (BK,1)
        b_col = col_offset[None, :]                                 # (1,BN)
        b_ptrs = (
            b_ptr
            + batch_idx * b_batch_stride
            + b_row * b_row_stride
            + b_col * b_col_stride
        )
        b_mask = (k_offset + tl.arange(0, BLOCK_K) < K)[:, None] & (col_offset < N)[None, :]
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)                # (BK,BN)

        # ---- Dot product (BM x BK) @ (BK x BN) --------------------
        acc += tl.dot(a, b)   # acc is float32, a/b are fp16

    # --------------------------------------------------------------
    # Write back the result
    # --------------------------------------------------------------
    c_row = row_offset[:, None]
    c_col = col_offset[None, :]
    c_ptrs = (
        c_ptr
        + batch_idx * c_batch_stride
        + c_row * c_row_stride
        + c_col * c_col_stride
    )
    c_mask = (row_offset < M)[:, None] & (col_offset < N)[None, :]
    tl.store(c_ptrs, acc, mask=c_mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim]
        - mask         : torch.Tensor [B, N, N] (may be ignored if config["nomask"])
        - weights      : dict of named tensors (see code)
        - config       : dict containing at least "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor
        Processed tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # 1) Input LayerNorm (over last dim)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )   # [B, N, N, dim]

    # -----------------------------------------------------------------
    # 2) Linear projections (no bias)
    # -----------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # -> [..., hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # 3) Optional mask (broadcast on channel dim)
    # -----------------------------------------------------------------
    if not nomask:
        # mask is bool/float; cast to same dtype as activations
        mask_uns = mask.unsqueeze(-1).to(dtype)          # [B,N,N,1]
        left  = left * mask_uns
        right = right * mask_uns

    # -----------------------------------------------------------------
    # 4) Gates (sigmoid of linear projections)
    # -----------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core reduction with Triton (batched GEMM)
    # -----------------------------------------------------------------
    B, N, _, _ = left.shape  # B, seq_len, seq_len, hidden_dim

    # Move hidden_dim to a batch dimension for the GEMM
    left  = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden_dim, N, N)
    right = right.permute(0, 3, 1, 2).contiguous()  # (B, hidden_dim, N, N)

    # Transpose the second operand so that the reduction is over the
    # shared dimension k (the inner N).
    right_t = right.transpose(-1, -2).contiguous()  # (B, hidden_dim, N, N)

    # Flatten batch & channel together: each slice is a separate GEMM
    batch_chan = B * hidden_dim
    left_flat  = left.reshape(batch_chan, N, N).to(torch.float16)
    right_flat = right_t.reshape(batch_chan, N, N).to(torch.float16)

    # Output buffer (float32 accumulator)
    out_flat = torch.empty((batch_chan, N, N), device=device, dtype=torch.float32)

    # --------------------------------------------------------------
    # Kernel launch parameters
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    a_batch_stride = left_flat.stride(0)
    a_row_stride   = left_flat.stride(1)
    a_col_stride   = left_flat.stride(2)

    b_batch_stride = right_flat.stride(0)
    b_row_stride   = right_flat.stride(1)
    b_col_stride   = right_flat.stride(2)

    c_batch_stride = out_flat.stride(0)
    c_row_stride   = out_flat.stride(1)
    c_col_stride   = out_flat.stride(2)

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    total_blocks = batch_chan * num_blocks_m * num_blocks_n

    batched_matmul_kernel[total_blocks](
        left_flat,               # a_ptr
        right_flat,              # b_ptr (already transposed)
        out_flat,                # c_ptr
        a_batch_stride, a_row_stride, a_col_stride,
        b_batch_stride, b_row_stride, b_col_stride,
        c_batch_stride, c_row_stride, c_col_stride,
        N, N, N,                 # M, N, K  (all equal to seq_len)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # --------------------------------------------------------------
    # 6) Reshape back to (B, N, N, hidden_dim)
    # --------------------------------------------------------------
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # -----------------------------------------------------------------
    # 7) Output LayerNorm, gating and final Linear
    # -----------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])   # --> [B, N, N, dim]

    return out
