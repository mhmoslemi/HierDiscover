import torch
import triton
import triton.language as tl

def _elementwise_mul_kernel(x_ptr, y_ptr, out_ptr, N, BLOCK: tl.constexpr):
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)
    mask = offs < N

    # Load as same dtype as pointer (assume fp16 or fp32)
    x = tl.load(x_ptr + offs, mask=mask, other=0.0)
    y = tl.load(y_ptr + offs, mask=mask, other=0.0)
    # Compute in fp32 for numerical stability
    x_f32 = x.to(tl.float32)
    y_f32 = y.to(tl.float32)
    prod = (x_f32 * y_f32).to(x.dtype)   # store back in original dtype
    tl.store(out_ptr + offs, prod, mask=mask)

def triton_elementwise_mul(a: torch.Tensor, b: torch.Tensor) -> torch.Tensor:
    """Compute a * b elementwise using a Triton kernel.
    a, b must be contiguous, same shape, same dtype.
    Returns a new tensor with the same dtype and device."""
    assert a.shape == b.shape
    assert a.is_contiguous() and b.is_contiguous()
    out = torch.empty_like(a)
    total = a.numel()
    # Choose block size based on dtype: 1024 works for both fp16/fp32
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta['BLOCK']),)
    _elementwise_mul_kernel[grid](a, b, out, total, BLOCK=BLOCK)
    return out
```

    """
    Triton-accelerated forward pass of the TriMul (outgoing) module.
    ...
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # LayerNorm over dim
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # mask shape [B, N, N] -> broadcast to hidden_dim
        mask = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = triton_elementwise_mul(left, left_gate)
    right = triton_elementwise_mul(right, right_gate)

    # ----- heavy contraction (batched matmul) -----
    # reshape for bmm: left: [B, hidden, N, N]; right transposed: [B, hidden, N, N]
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # B, hidden, N, N
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # B, hidden, N, N

        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]

    out_norm = triton_elementwise_mul(out_norm, out_gate)

    # Final linear projection to original dim
    out_final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

"""
tri_mul_triton.py

Implementation of the “outgoing” TriMul operator (AlphaFold‑3) using a
combination of PyTorch and a small Triton kernel.  The heavy
contraction

    out[b, i, j, d] = Σₖ left[b, i, k, d] * right[b, j, k, d]

is performed as a batched GEMM (torch.bmm) after appropriate
permutations, while the element‑wise gate applications are fused in a
Triton kernel.  No backward pass is required.

Supported shapes:
    B ∈ {1,2},  N ∈ {128,256,512,1024},  dim ∈ {128,384},
    hidden_dim = 128

All tensors are assumed to be on the CUDA device already.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: element‑wise multiplication (used for gate * tensor)
# ----------------------------------------------------------------------
@triton.jit
def _elemwise_mul_kernel(
    a_ptr,          # Tensor A pointer (contiguous)
    b_ptr,          # Tensor B pointer (contiguous, same shape as A)
    out_ptr,        # Output pointer (contiguous)
    n_elements,     # total number of elements = A.numel()
    BLOCK: tl.constexpr,
):
    """Compute out = a * b element‑wise."""
    pid = tl.program_id(0)
    start = pid * BLOCK
    offs = start + tl.arange(0, BLOCK)                     # [0, BLOCK)
    mask = offs < n_elements

    # Load (auto‑detect dtype from the pointer – works for fp16 / fp32)
    a = tl.load(a_ptr + offs, mask=mask, other=0.0)
    b = tl.load(b_ptr + offs, mask=mask, other=0.0)

    # Compute in FP32 for a bit more stability, then cast back
    prod = (a.to(tl.float32) * b.to(tl.float32)).to(a.dtype)

    tl.store(out_ptr + offs, prod, mask=mask)


def triton_elemwise_mul(x: torch.Tensor, y: torch.Tensor) -> torch.Tensor:
    """Thin wrapper around the Triton element‑wise multiplication kernel.

    Arguments:
        x, y: contiguous tensors with identical shape and dtype.
    Returns:
        Tensor `x * y` (new tensor, same dtype/device as the inputs).
    """
    assert x.shape == y.shape, "Shapes must match for element‑wise mul"
    assert x.is_contiguous() and y.is_contiguous(), "Inputs must be contiguous"
    out = torch.empty_like(x)

    total = x.numel()
    # Block size tuned for H100 – 1024 works well for fp16/fp32
    BLOCK = 1024
    grid = lambda meta: (triton.cdiv(total, meta["BLOCK"]),)
    _elemwise_mul_kernel[grid](
        x, y, out, total, BLOCK=BLOCK
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul (“outgoing”) module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights_dict : dict with the following keys (all tensors are
                         on the same CUDA device as ``input_tensor``):
            * "norm.weight", "norm.bias"
            * "left_proj.weight", "right_proj.weight"
            * "left_gate.weight", "right_gate.weight", "out_gate.weight"
            * "to_out_norm.weight", "to_out_norm.bias"
            * "to_out.weight"
        - config_dict  : dict with at least ``dim`` and ``hidden_dim`` fields.

    Returns
    -------
    torch.Tensor
        The processed tensor of shape [B, N, N, dim].
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]   # always 128 in the benchmark

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # ensure mask is float and broadcastable
        mask_f = mask.to(dtype).unsqueeze(-1)            # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Compute gates (sigmoid of a linear projection)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    # ------------------------------------------------------------------
    # 5) Fuse gate multiplication with a tiny Triton kernel
    # ------------------------------------------------------------------
    left = triton_elemwise_mul(left, left_gate)    # left = left * left_gate
    right = triton_elemwise_mul(right, right_gate)  # right = right * right_gate

    # ------------------------------------------------------------------
    # 6) Heavy contraction:   out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched GEMM (torch.bmm) after suitable permutes.
    # ------------------------------------------------------------------
    # bring the hidden dimension to the second axis
    left_perm = left.permute(0, 3, 1, 2).contiguous()       # [B, hidden, N, N]
    # right needs its two sequence axes swapped before the matmul
    right_perm = right.permute(0, 3, 2, 1).contiguous()     # [B, hidden, N, N]

    # Merge batch and hidden dims → (B*hidden, N, N)
    bh = B * hidden_dim
    left_mat = left_perm.view(bh, N, N)
    right_mat = right_perm.view(bh, N, N)

    # Batched matrix multiply
    out_mat = torch.bmm(left_mat, right_mat)                # (B*hidden, N, N)

    # Reshape back to [B, N, N, hidden]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Output layer‑norm (over hidden_dim)
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 8) Multiply by the output gate (again via Triton)
    # ------------------------------------------------------------------
    out_norm = triton_elemwise_mul(out_norm, out_gate)

    # ------------------------------------------------------------------
    # 9) Final linear projection back to original dim
    # ------------------------------------------------------------------
    final_out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final_out
