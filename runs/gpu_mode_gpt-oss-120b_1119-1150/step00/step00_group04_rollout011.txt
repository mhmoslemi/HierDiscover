@triton.jit
def batched_matmul_kernel(A_ptr, B_ptr, C_ptr,
                          N,
                          stride_a_batch, stride_a_i, stride_a_k,
                          stride_b_batch, stride_b_k, stride_b_n,
                          stride_c_batch, stride_c_i, stride_c_n,
                          BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # Mask for out-of-bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # Load A block
        a_ptrs = A_ptr + pid_bh * stride_a_batch + offs_i[:, None] * stride_a_i + offs_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float16)

        # Load B block (B is already transposed)
        b_ptrs = B_ptr + pid_bh * stride_b_batch + offs_k[:, None] * stride_b_k + offs_j[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float16)

        # Compute partial product
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # Store result
    c_ptrs = C_ptr + pid_bh * stride_c_batch + offs_i[:, None] * stride_c_i + offs_j[None, :] * stride_c_n
    mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=mask)

"""
custom_kernel for the outgoing TriMul operation.

Algorithm
---------
1. Layer‑norm the input pairwise tensor (shape [B,N,N,C]).
2. Linear projections → left / right (size H) and three gating vectors.
3. (Optional) Apply the pairwise mask to left/right.
4. Multiply left/right by their respective gates.
5. Compute the “tri‑multiplicative” contraction:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   This is batched matrix‐multiply where for each (b,d) we perform
   left_{i,k} @ right_{k,j}ᵀ.  Implemented with a custom Triton kernel
   that tiles the A (i,k) and Bᵀ (k,j) blocks and accumulates in FP32.
6. Layer‑norm on the hidden dimension, gate by `out_gate`, and a final
   linear projection back to the original channel dimension.
7. Return a float‑32 tensor of shape [B,N,N,C].

The heavy N³ contraction is fused in Triton; all other ops use
standard PyTorch (Torch‑CUDA) kernels.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F

# Triton kernel: batched matrix multiply (A @ Bᵀ) for many (batch×head) matrices
# ----------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_i, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # program ids
    pid_bh = tl.program_id(0)            # batch * head index
    pid_i  = tl.program_id(1)            # tile row
    pid_j  = tl.program_id(2)            # tile col

    # tile coordinates
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over K‑dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # bounds
        mask_i = offs_i < N
        mask_j = offs_j < N
        mask_k = offs_k < N

        # ---- Load block of A (i,k) ----
        a_ptrs = (A_ptr
                  + pid_bh * stride_a_batch
                  + offs_i[:, None] * stride_a_i
                  + offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float16)

        # ---- Load block of Bᵀ (k,j) ----
        # B is already stored as (k, j) because we pre‑transposed it
        b_ptrs = (B_ptr
                  + pid_bh * stride_b_batch
                  + offs_k[:, None] * stride_b_k
                  + offs_j[None, :] * stride_b_n)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float16)

        # ---- Matrix‑multiply this tile ----
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # ---- Write the result tile ----
    c_ptrs = (C_ptr
              + pid_bh * stride_c_batch
              + offs_i[:, None] * stride_c_i
              + offs_j[None, :] * stride_c_n)
    store_mask = (offs_i[:, None] < N) & (offs_j[None, :] < N)
    tl.store(c_ptrs, acc, mask=store_mask)


def _batched_matmul(A: torch.Tensor, B: torch.Tensor,
                    BLOCK_M=128, BLOCK_N=128, BLOCK_K=32) -> torch.Tensor:
    """
    A, B : tensors of shape (P, N, N)   (contiguous, dtype=half)
    Returns C of shape (P, N, N) as FP32.
    """
    assert A.is_contiguous() and B.is_contiguous()
    P, N, _ = A.shape
    C = torch.empty_like(A, dtype=torch.float32)  # result in FP32

    # Strides after the view (P, N, N) – they are the same for A and B
    stride_batch = N * N
    stride_i = N          # row stride
    stride_k = 1          # column stride for A
    stride_k_B = N        # row stride for Bᵀ (k dimension)
    stride_n_B = 1        # column stride for Bᵀ (j dimension)

    grid = (P,
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N)

    _batched_matmul_kernel[grid](
        A_ptr=A,
        B_ptr=B,
        C_ptr=C,
        N=N,
        stride_a_batch=stride_batch,
        stride_a_i=stride_i,
        stride_a_k=stride_k,
        stride_b_batch=stride_batch,
        stride_b_k=stride_k_B,
        stride_b_n=stride_n_B,
        stride_c_batch=stride_batch,
        stride_c_i=stride_i,
        stride_c_n=stride_n_B,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    return C


# ----------------------------------------------------------------------
# Entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape (bs, seq_len, seq_len, dim) on the same device.
    """
    # ------------------------------------------------------------------
    # Unpack
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype_in = torch.float16          # use half for the heavy compute

    # ------------------------------------------------------------------
    # Model hyper‑parameters
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # Cast weights to the computation dtype where appropriate
    # (linear layers use FP16, LayerNorms keep FP32 for numerical stability)
    # ------------------------------------------------------------------
    norm_w   = weights["norm.weight"].to(dtype_in)
    norm_b   = weights["norm.bias"].to(dtype_in)
    left_w   = weights["left_proj.weight"].to(dtype_in)
    right_w  = weights["right_proj.weight"].to(dtype_in)
    left_gate_w  = weights["left_gate.weight"].to(dtype_in)
    right_gate_w = weights["right_gate.weight"].to(dtype_in)
    out_gate_w   = weights["out_gate.weight"].to(dtype_in)

    to_out_norm_w = weights["to_out_norm.weight"]          # keep FP32
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]               # keep FP32 (final projection)

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (over last dim = dim)
    # ------------------------------------------------------------------
    x = input_tensor.to(dtype_in)
    x = F.layer_norm(x,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → left / right
    # ------------------------------------------------------------------
    left  = F.linear(x, left_w)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_w)

    # ------------------------------------------------------------------
    # 3. Mask (if present)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        mask = mask_tensor.to(dtype_in).unsqueeze(-1)   # (B,N,N,1)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gate values (sigmoid) and application
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core TriMul contraction via Triton (batched matmul)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    B, N, _, H = left.shape          # (B,N,N,H)

    #   a) permute to (B, H, N, N) where A = (i,k) and Bᵀ = (k,j)
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # (B,H,N,N)
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # (B,H,N,N) – already transposed

    #   b) collapse the leading two dims → (P, N, N) where P = B*H
    P = B * H
    left_flat  = left_perm.view(P, N, N)
    right_flat = right_perm.view(P, N, N)

    #   c) batched matmul (FP16 * FP16 → FP32)
    out_flat = _batched_matmul(left_flat, right_flat,
                               BLOCK_M=128, BLOCK_N=128, BLOCK_K=32)

    #   d) restore the original layout (B,N,N,H) – now in FP32
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()   # (B,N,N,H)

    # ------------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    # ------------------------------------------------------------------
    # 7. Apply output gate and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate.to(out.dtype)          # gating
    output = F.linear(out, to_out_w)            # (B,N,N,dim) – stays FP32

    return output
