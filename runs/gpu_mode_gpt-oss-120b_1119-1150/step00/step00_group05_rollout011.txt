"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑normalize the input 4‑D tensor (shape [B,N,N,dim]).
2. Linear projections (left/right) and three gating linear layers
   (left_gate, right_gate, out_gate) are computed with the provided
   weights (no bias).  The gating tensors are passed through a sigmoid.
3. If a mask is present (config["nomask"] == False) a Triton kernel fuses
   the three operations
        left  = left  * mask * left_gate
        right = right * mask * right_gate
   in‑place.  When there is no mask the same multiplication is performed
   with a scalar mask of 1.
4. The core “tri‑mul” is
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batch of matrix multiplications:
        L = left .permute(0,3,1,2)   # [B, H, N, N]
        R = right.permute(0,3,1,2)  # [B, H, N, N]
        out = (L @ R.transpose(-2,-1)).permute(0,2,3,1)   # [B,N,N,H]
   The batch size for the GEMM is B*hidden_dim (≤256), and the multiplies
   are performed by the highly‑optimized cuBLAS kernel.
5. Apply a second LayerNorm (to_out_norm), multiply by the out‑gate tensor
   and finally a linear projection back to ``dim`` (to_out).

The only Triton kernel used is the mask‑gate fusion (step 3); the heavy
matrix multiplication is delegated to cuBLAS for best performance.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


@triton.jit
def apply_mask_and_gate_kernel(
    left_ptr,                # float32 *
    right_ptr,               # float32 *
    left_gate_ptr,           # float32 *
    right_gate_ptr,          # float32 *
    mask_ptr,                # float32 *
    total_elements,          # int64 scalar, = B*N*N*H
    hidden_dim,              # int64 scalar = H
    BLOCK_SIZE: tl.constexpr # compile‑time constant (threads per program)
):
    """
    In‑place multiplication:
        left  = left  * mask[pos] * left_gate
        right = right * mask[pos] * right_gate
    where ``pos = element_index // hidden_dim``.
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    active = offs < total_elements

    # Position index for the mask (flattened B*N*N)
    pos = offs // hidden_dim

    # Load data (masked loads keep out‑of‑bounds elements at 0)
    left      = tl.load(left_ptr      + offs, mask=active, other=0.0)
    right     = tl.load(right_ptr     + offs, mask=active, other=0.0)
    left_g    = tl.load(left_gate_ptr + offs, mask=active, other=0.0)
    right_g   = tl.load(right_gate_ptr+ offs, mask=active, other=0.0)
    mask_val  = tl.load(mask_ptr      + pos , mask=active, other=0.0)

    # Fuse mask and gates
    left_out  = left  * mask_val * left_g
    right_out = right * mask_val * right_g

    # Store the results back in‑place
    tl.store(left_ptr  + offs, left_out,  mask=active)
    tl.store(right_ptr + offs, right_out, mask=active)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Args:
        data: Tuple (input, mask, weights, config)
            - input : torch.Tensor [B, N, N, dim] (float32)
            - mask  : torch.Tensor [B, N, N] (float32, optional)
            - weights: dict of torch.Tensors containing all linear and LN params
            - config : dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections and gating (no bias)
    # ------------------------------------------------------------------
    left_proj  = F.linear(x, weights["left_proj.weight"])
    right_proj = F.linear(x, weights["right_proj.weight"])

    left_gate  = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = F.linear(x, weights["out_gate.weight"]).sigmoid()

    # ------------------------------------------------------------------
    # 3) Apply mask and gates (fused in‑place Triton kernel if mask is used)
    # ------------------------------------------------------------------
    use_mask = (mask is not None) and (not config.get("nomask", True))
    if use_mask:
        # Ensure mask is float32 on the same device, flatten to 1‑D
        mask_f = mask.to(dtype=torch.float32, device=input_tensor.device)
        mask_f = mask_f.contiguous().view(-1)   # [B*N*N]

        # Make sure all tensors are contiguous for the kernel
        left_proj   = left_proj.contiguous()
        right_proj  = right_proj.contiguous()
        left_gate   = left_gate.contiguous()
        right_gate  = right_gate.contiguous()

        total_elems = left_proj.numel()               # B * N * N * hidden_dim
        BLOCK_SIZE = 1024                              # reasonable occupancy

        grid = lambda meta: (triton.cdiv(total_elems, meta["BLOCK_SIZE"]),)

        apply_mask_and_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_f,
            total_elements=total_elems,
            hidden_dim=hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # After the kernel left_proj/right_proj already include mask*gate
        # No further multiplication needed.
    else:
        # No mask: just apply the gates (mask == 1)
        left_proj = left_proj * left_gate
        right_proj = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4) Tri‑multiplicative update (batched GEMM)
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B, H, N, N) for torch.bmm
    L = left_proj.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    R = right_proj.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Batched matrix multiplication: (B*H, N, N) @ (B*H, N, N).T
    out = torch.matmul(L, R.transpose(-2, -1))
    # back to original layout [B, N, N, H]
    out = out.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5) Output LayerNorm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
