@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    NUM_BLOCKS_I: tl.constexpr, NUM_BLOCKS_J: tl.constexpr
):
    pid = tl.program_id(0)
    # compute indices
    total_blocks = NUM_BLOCKS_I * NUM_BLOCKS_J
    b = pid // total_blocks
    pid_rem = pid % total_blocks
    i_bid = pid_rem // NUM_BLOCKS_J
    j_bid = pid_rem % NUM_BLOCKS_J

    i_offset = i_bid * BLOCK_I
    j_offset = j_bid * BLOCK_J

    # define ranges
    offs_i = i_offset + tl.arange(0, BLOCK_I)
    offs_j = j_offset + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    offs_h = tl.arange(0, BLOCK_H)

    # accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # load left block
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            cur_k[None, :, None] * stride_left_k + \
            offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # load right block
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None, None] * stride_right_j + \
            cur_k[None, :, None] * stride_right_k + \
            offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None], other=0.0).to(tl.float32)

        # Accumulate outer product across i,j for each h over K
        # loop over inner k dimension for reduction
        for kk in range(0, BLOCK_K):
            # check mask for this k index
            k_valid = mask_k[kk]
            # optionally skip if not valid
            if k_valid:
                l_slice = left[:, kk, :]  # (BLOCK_I, BLOCK_H)
                r_slice = right[:, kk, :]  # (BLOCK_J, BLOCK_H)

                # broadcasting
                l_exp = tl.broadcast_to(l_slice, (BLOCK_I, 1, BLOCK_H))
                r_exp = tl.broadcast_to(r_slice, (1, BLOCK_J, BLOCK_H))
                acc += l_exp * r_exp
            # else skip
    # store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        offs_h[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float32), mask=mask_i[:, None, None] & mask_j[None, :, None])
```

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    NUM_BLOCKS_I: tl.constexpr,
    NUM_BLOCKS_J: tl.constexpr,
    NUM_BLOCKS_H: tl.constexpr,
):
    pid = tl.program_id(0)
    total_ij = NUM_BLOCKS_I * NUM_BLOCKS_J
    b = pid // (total_ij * NUM_BLOCKS_H)
    rem = pid % (total_ij * NUM_BLOCKS_H)
    h_bid = rem // total_ij
    rem2 = rem % total_ij
    i_bid = rem2 // NUM_BLOCKS_J
    j_bid = rem2 % NUM_BLOCKS_J

    i0 = i_bid * BLOCK_I
    j0 = j_bid * BLOCK_J
    h0 = h_bid * BLOCK_H

    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # left block pointer
        left_ptrs = left_ptr \
            + b * stride_left_b \
            + offs_i[:, None, None] * stride_left_i \
            + cur_k[None, :, None] * stride_left_k \
            + offs_h[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float16)

        # right block pointer
        right_ptrs = right_ptr \
            + b * stride_right_b \
            + offs_j[:, None, None] * stride_right_j \
            + cur_k[None, :, None] * stride_right_k \
            + offs_h[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float16)

        # For each hidden channel in this block
        for h_idx in range(0, BLOCK_H):
            # check mask_h[h_idx] maybe else zero
            # l_slice shape: (BLOCK_I, BLOCK_K)
            l_slice = left[:, :, h_idx]
            r_slice = right[:, :, h_idx]
            # dot product: (BLOCK_I, BLOCK_J)
            acc[:, :, h_idx] += tl.dot(l_slice, r_slice, trans_b=True).to(tl.float32)
    # store
    out_ptrs = out_ptr \
        + b * stride_out_b \
        + offs_i[:, None, None] * stride_out_i \
        + offs_j[None, :, None] * stride_out_j \
        + offs_h[None, None, :] * stride_out_h
    tl.store(out_ptrs, acc, mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :])
```

"""
TriMul “outgoing” implementation for AlphaFold3 style pairwise updates.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape [B,N,N,D]).
2. Linear projections `left_proj` and `right_proj` (to hidden_dim H) and three
   sigmoid gates (`left_gate`, `right_gate`, `out_gate`).
3. Apply mask (if present) and gates:
        left  = left_proj(x)  * mask * left_gate
        right = right_proj(x) * mask * right_gate
4. Compute the heavy N³‐H tensor contraction
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   using a custom Triton kernel that processes a tile (i‑block, j‑block, h‑block)
   per program and accumulates with‐tensor‑core `tl.dot` on half‑precision data.
5. Layer‑norm the result, multiply by `out_gate`, and apply the final linear
   projection back to the original dimension D.

The kernel only handles the N³‑H contraction; all other operations use
standard PyTorch (they are negligible compared to the contraction).
The final output is float‑32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,                # *float16  [B,N,N,H]
    right_ptr,               # *float16  [B,N,N,H]
    out_ptr,                 # *float32  [B,N,N,H]

    B, N, H,                 # runtime values

    # strides (in elements) of the tensors
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,

    # compile‑time tile sizes
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
    BLOCK_H: tl.constexpr,
    NUM_BLOCKS_I: tl.constexpr,
    NUM_BLOCKS_J: tl.constexpr,
    NUM_BLOCKS_H: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a tiled outer‑product + reduction over k.
    """
    # ----------- compute indices from the 1‑D program id ------------
    total_ij = NUM_BLOCKS_I * NUM_BLOCKS_J
    pid = tl.program_id(0)

    b = pid // (total_ij * NUM_BLOCKS_H)                      # batch index
    pid_rem = pid % (total_ij * NUM_BLOCKS_H)
    h_bid = pid_rem // total_ij                               # hidden‑block index
    pid_rem = pid_rem % total_ij
    i_bid = pid_rem // NUM_BLOCKS_J
    j_bid = pid_rem % NUM_BLOCKS_J

    # start offsets for this tile
    i0 = i_bid * BLOCK_I
    j0 = j_bid * BLOCK_J
    h0 = h_bid * BLOCK_H

    # range vectors
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)
    offs_h = h0 + tl.arange(0, BLOCK_H)

    # boundary masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator for this tile  (float32 to avoid overflow)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # ------------------- loop over the reduction dimension k -------------------
    for k_start in range(0, N, BLOCK_K):
        cur_k = k_start + tl.arange(0, BLOCK_K)
        mask_k = cur_k < N

        # ----- load left block ------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None, None] * stride_left_i
            + cur_k[None, :, None] * stride_left_k
            + offs_h[None, None, :] * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # -> half

        # ----- load right block ------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_j[:, None, None] * stride_right_j
            + cur_k[None, :, None] * stride_right_k
            + offs_h[None, None, :] * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        )  # -> half

        # ----- compute outer‑product for each hidden channel in this block -----
        # left  : (BLOCK_I, BLOCK_K, BLOCK_H)
        # right : (BLOCK_J, BLOCK_K, BLOCK_H)
        for h_idx in range(0, BLOCK_H):
            # slice for channel h_idx
            l_h = left[:, :, h_idx]   # (BLOCK_I, BLOCK_K)
            r_h = right[:, :, h_idx]  # (BLOCK_J, BLOCK_K)

            # matrix multiply over k (left * right^T) -> (BLOCK_I, BLOCK_J)
            # trans_b=True treats r_h as (BLOCK_K, BLOCK_J)
            prod = tl.dot(l_h, r_h, trans_b=True)  # result is half, accumulate as float32
            acc[:, :, h_idx] += prod.to(tl.float32)

    # --------------------- store the result ---------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None, None] * stride_out_i
        + offs_j[None, :, None] * stride_out_j
        + offs_h[None, None, :] * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] after the TriMul update.
    """
    # unpack arguments ---------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("layernorm_eps", 1e-5)

    # ----------- helper: LayerNorm (affine) ---------------------------------
    def layer_norm(tensor, weight, bias):
        mean = tensor.mean(dim=-1, keepdim=True)
        var = ((tensor - mean) ** 2).mean(dim=-1, keepdim=True)
        normed = (tensor - mean) / torch.sqrt(var + eps)
        return normed * weight + bias

    # ---------- 1️⃣ input LayerNorm -----------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x_norm = layer_norm(x, norm_weight, norm_bias)                     # [B,N,N,D]

    # ---------- 2️⃣ linear projections ---------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])               # [B,N,N,H]
    right = F.linear(x_norm, weights["right_proj.weight"])             # [B,N,N,H]

    # ---------- 3️⃣ optional mask --------------------------------------------
    if mask is not None:
        # mask shape: [B,N,N] – broadcast on hidden dim
        mask_f = mask.unsqueeze(-1).type_as(left)                     # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ---------- 4️⃣ gating ---------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ---------- 5️⃣ cast to half for the heavy contraction --------------------
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, N2, H = left_h.shape
    assert N == N2 and H == hidden_dim

    # output buffer for the contraction (float32 is required by the rest)
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # ---------- 6️⃣ launch Triton kernel ------------------------------------
    # tile sizes (tune for H100 – these values give good occupancy)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # must divide hidden_dim in the grid (remainder handled by mask)

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    num_h = (H + BLOCK_H - 1) // BLOCK_H
    grid = (B * num_i * num_j * num_h,)

    # strides in number of elements (Triton expects int64)
    s_lb, s_li, s_lk, s_lh = left_h.stride()
    s_rb, s_rj, s_rk, s_rh = right_h.stride()
    s_ob, s_oi, s_oj, s_oh = out.stride()

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        H,
        s_lb,
        s_li,
        s_lk,
        s_lh,
        s_rb,
        s_rj,
        s_rk,
        s_rh,
        s_ob,
        s_oi,
        s_oj,
        s_oh,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        NUM_BLOCKS_I=num_i,
        NUM_BLOCKS_J=num_j,
        NUM_BLOCKS_H=num_h,
        num_warps=4,            # enough warps for the matmul
    )

    # ---------- 7️⃣ post‑processing LayerNorm --------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = ((out - mean_o) ** 2).mean(dim=-1, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias

    # ---------- 8️⃣ apply out_gate and final linear -------------------------
    out_norm = out_norm * out_gate
    to_out_weight = weights["to_out.weight"]
    output = F.linear(out_norm, to_out_weight)    # [B,N,N,D]

    return output
