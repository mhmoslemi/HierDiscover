@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    # program IDs
    b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # iterate over hidden blocks
    h = 0
    while h < H:
        cur_h = tl.minimum(BLOCK_H, H - h)
        h_range = tl.arange(0, BLOCK_H) + h

        # accumulator
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # loop over K blocks
        for k_start in range(0, tl.cdiv(N, BLOCK_K)):
            k_base = k_start * BLOCK_K
            cur_k = tl.minimum(BLOCK_K, N - k_base)
            k_range = tl.arange(0, BLOCK_K) + k_base

            i_range = tl.arange(0, BLOCK_I) + i_base
            j_range = tl.arange(0, BLOCK_J) + j_base

            # masks for out-of-bound
            i_mask = i_range < N
            j_mask = j_range < N
            k_mask = k_range < N
            h_mask = h_range < H

            # compute offsets
            # left indices: (i, k, h)
            left_off = (i_range[:, None, None] * stride_il
                        + k_range[None, :, None] * stride_kl
                        + h_range[None, None, :] * stride_hl)
            # right indices: (j, k, h)
            right_off = (j_range[:, None, None] * stride_ir
                        + k_range[None, :, None] * stride_kr
                        + h_range[None, None, :] * stride_hr)

            # mask tensors
            left_mask = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]

                           mask=left_mask,

                            mask=right_mask,

                l = left[:, kk, :]   # shape (BLOCK_I, BLOCK_H)

                   + j_range[None, :, None] * stride_jo
                   + h_range[None, None, :] * stride_ho)

        h += BLOCK_H
```

    right_h = right.to(torch.float16)

    BLOCK_J = 32

    BLOCK_H = 32

        left_h, right_h, out,
        B, N, H,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_I=BLOCK_I, BLOCK_J=BLOCK_J, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )

    torch.cuda.synchronize()

    # Final linear
    final = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return final

"""
TriMul (outgoing) – fast Triton implementation.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} and a pairwise mask M ∈ {0,1}^{B×N×N}.
The forward pass consists of:

1. LayerNorm over the last channel dimension (C = dim).
2. Two bias‑free linear projections to a hidden dimension H (left_proj, right_proj).
3. Three sigmoid gates (left_gate, right_gate, out_gate) – also bias‑free linear layers.
4. Mask the projected tensors and apply the element‑wise gates.
5. Compute the core “tri‑multiplicative” update:
       out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   i.e. a per‑hidden‑channel Gram‑matrix over the pairwise dimension.
   This O(N³·H) contraction is performed with a custom Triton kernel that
   tiles the i, j, k and hidden dimensions.
6. LayerNorm over the hidden dimension, another sigmoid gate (out_gate) and a final
   linear projection back to the original channel dimension.

Only the heavy contraction (step 5) is written in Triton; all other
operations use PyTorch (which already runs fast on the GPU).  The kernel
loads tiles of the left and right tensors, forms an outer‑product for each
k‑tile, reduces over k and writes the result in FP32.  Mixed‑precision is
used for the inputs (FP16) while accumulation stays in FP32 to preserve
numerical quality.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_bl, stride_il, stride_kl, stride_hl,
    stride_br, stride_ir, stride_kr, stride_hr,
    stride_bo, stride_io, stride_jo, stride_ho,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """
    Triton kernel for: out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    """
    # program IDs
    b   = tl.program_id(0)                     # batch dimension
    pid_i = tl.program_id(1)                   # i‑tile index
    pid_j = tl.program_id(2)                   # j‑tile index

    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J

    # -------------------------------------------------------------------------
    # loop over hidden blocks (H dimension)
    # -------------------------------------------------------------------------
    for hb in range(0, tl.cdiv(H, BLOCK_H)):
        # start index of the current hidden block
        h0 = hb * BLOCK_H
        # range tensors (size BLOCK_*)
        i_range = tl.arange(0, BLOCK_I) + i0
        j_range = tl.arange(0, BLOCK_J) + j0
        h_range = tl.arange(0, BLOCK_H) + h0

        # mask for out‑of‑bounds in i, j and h
        i_mask = i_range < N
        j_mask = j_range < N
        h_mask = h_range < H

        # accumulator for this hidden block: shape (BLOCK_I, BLOCK_J, BLOCK_H)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # -----------------------------------------------------------------
        # loop over k‑tiles (pairwise dimension)
        # -----------------------------------------------------------------
        for kb in range(0, tl.cdiv(N, BLOCK_K)):
            k0 = kb * BLOCK_K
            k_range = tl.arange(0, BLOCK_K) + k0

            # mask for k
            k_mask = k_range < N

            # -------------------------------------------------------------
            # load left[b, i, k, h]  → shape (BLOCK_I, BLOCK_K, BLOCK_H)
            # load right[b, j, k, h] → shape (BLOCK_J, BLOCK_K, BLOCK_H)
            # -------------------------------------------------------------
            left_off  = (i_range[:, None, None] * stride_il +
                         k_range[None, :, None] * stride_kl +
                         h_range[None, None, :] * stride_hl)
            right_off = (j_range[:, None, None] * stride_ir +
                         k_range[None, :, None] * stride_kr +
                         h_range[None, None, :] * stride_hr)

            left_mask  = i_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]
            right_mask = j_mask[:, None, None] & k_mask[None, :, None] & h_mask[None, None, :]

            left  = tl.load(left_ptr + b * stride_bl + left_off,
                            mask=left_mask, other=0.0).to(tl.float32)
            right = tl.load(right_ptr + b * stride_br + right_off,
                            mask=right_mask, other=0.0).to(tl.float32)

            # -------------------------------------------------------------
            # outer‑product over the k‑axis:
            #   Σ_k left[i,k,h] * right[j,k,h]   for all i,j,h in the tile
            # -------------------------------------------------------------
            # left  : (BLOCK_I, BLOCK_K, BLOCK_H) → (BLOCK_I, 1, BLOCK_K, BLOCK_H)
            # right : (BLOCK_J, BLOCK_K, BLOCK_H) → (1, BLOCK_J, BLOCK_K, BLOCK_H)
            # after broadcasting → (BLOCK_I, BLOCK_J, BLOCK_K, BLOCK_H)
            prod = left[:, None, :, :] * right[None, :, :, :]
            acc += tl.sum(prod, axis=2)   # sum over the k‑axis → (BLOCK_I, BLOCK_J, BLOCK_H)

        # -----------------------------------------------------------------
        # store the accumulated tile into the output tensor
        # -----------------------------------------------------------------
        out_off = (i_range[:, None, None] * stride_io +
                   j_range[None, :, None] * stride_jo +
                   h_range[None, None, :] * stride_ho)

        out_mask = i_mask[:, None, None] & j_mask[None, :, None] & h_mask[None, None, :]
        tl.store(out_ptr + b * stride_bo + out_off,
                 acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N]   (bool / float)
        - weights      : dict of torch.Tensors containing all learnable parameters
        - config       : dict with keys "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # unpack arguments
    # -----------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    B, N, _, _ = input_tensor.shape

    # -----------------------------------------------------------------
    # 1) LayerNorm over the channel dimension (dim)
    # -----------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # shape (B,N,N,dim)

    # -----------------------------------------------------------------
    # 2) Linear projections (bias‑free)
    # -----------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])      # (B,N,N,hidden_dim)
    right = F.linear(x, weights["right_proj.weight"])    # (B,N,N,hidden_dim)

    # -----------------------------------------------------------------
    # 3) Gating layers (sigmoid)
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # -----------------------------------------------------------------
    # 4) Apply mask (if provided) and gates
    # -----------------------------------------------------------------
    if mask is not None:
        # mask shape (B, N, N) → broadcast to channel dim
        mask_f = mask.unsqueeze(-1).to(x.dtype)      # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # 5) Core tri‑multiplicative contraction via Triton
    # -----------------------------------------------------------------
    # cast to half for memory bandwidth, keep accumulation in fp32
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    # output buffer (float32)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # strides (in elements)
    stride_bl, stride_il, stride_kl, stride_hl = left_h.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right_h.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # tile sizes – chosen to fit H100 shared memory comfortably
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64
    BLOCK_H = 32

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    _trimul_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )
    # synchronize to make sure the kernel finished before next ops
    torch.cuda.synchronize()

    # -----------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out_gate, final linear
    # -----------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate
    final = F.linear(out_norm, weights["to_out.weight"])   # → (B,N,N,dim)

    return final
