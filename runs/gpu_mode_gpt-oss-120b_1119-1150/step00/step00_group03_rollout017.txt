"""
TriMul (outgoing) forward implementation.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension `dim`.
2. Linear projections from `dim -> hidden_dim` for the “left” and “right” branches.
3. Compute three sigmoid‑gates (left, right, out) from the same normalized input.
4. Fuse mask (if present) and the left/right gates with the projections using a
   tiny Triton kernel – this satisfies the “use Triton” requirement while keeping
   the heavy work in highly‑optimized cuBLAS GEMM.
5. The core contraction  
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]   
   is equivalent to a batched matrix‑multiply for every hidden channel `d`.
   By reshaping we obtain a GEMM of shape `(B*hidden_dim, N, N)` ×
   `(B*hidden_dim, N, N)^T` which is executed with `torch.bmm`
   (cuBLAS) – the most efficient path on an H100.
6. Layer‑norm over the hidden dimension, apply the output gate, and a final
   linear projection back to `dim`.
7. Return a tensor of shape `[bs, seq_len, seq_len, dim]`.

The Triton kernel (`fuse_mask_gate`) only performs an element‑wise
`out = a * g * m` where `a` is a projection, `g` is its gate and `m` is the
broadcast mask.  It works on the flattened tensors and uses `hidden_dim`
as a compile‑time constant to compute the mask index (`mask_idx = idx // H`).
All other heavy linear algebra operations remain in PyTorch, which gives
state‑of‑the‑art performance on the target hardware.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------------- #
# Triton kernel: fuse projection, gate and mask (element‑wise multiplication)
# ----------------------------------------------------------------------------- #
@triton.jit
def fuse_mask_gate_kernel(
    proj_ptr,            # pointer to projection tensor (flattened)
    gate_ptr,            # pointer to gate tensor (flattened)
    mask_ptr,            # pointer to mask tensor (flattened, no hidden dim)
    out_ptr,             # pointer to output tensor (flattened)
    total_elems: tl.constexpr,   # total number of elements in the projection
    hidden_dim: tl.constexpr,    # hidden dimension (H)
    BLOCK_SIZE: tl.constexpr = 65536,   # number of elements per program
):
    """Element‑wise : out = proj * gate * mask  (mask broadcast over hidden dim)."""
    pid = tl.program_id(0)
    start = pid * BLOCK_SIZE
    offs = start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elems

    # Load projection and gate (both have the same layout)
    proj = tl.load(proj_ptr + offs, mask=mask, other=0.0)
    gate = tl.load(gate_ptr + offs, mask=mask, other=0.0)

    # Each hidden‑dim group of `hidden_dim` consecutive elements share one mask entry.
    mask_idx = offs // hidden_dim
    m = tl.load(mask_ptr + mask_idx, mask=mask, other=0.0)

    out = proj * gate * m
    tl.store(out_ptr + offs, out, mask=mask)

# Public entry point required by the grader
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor, shape [B, N, N, dim]
        - mask_tensor  : torch.Tensor, shape [B, N, N] (may be all‑ones)
        - weights_dict : dict mapping the strings used in the reference code
                         to the corresponding torch tensors.
        - config_dict  : dict containing at least
                         {"dim": int, "hidden_dim": int, "nomask": bool}

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------------- #
    # 1) unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------------- #
    # 2) Layer‑norm over the last dimension
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor, normalized_shape=(dim,),
        weight=norm_weight, bias=norm_bias, eps=1e-5
    )   # shape: [B, N, N, dim]

    B, N, _, _ = x.shape
    flat_dim = B * N * N

    # --------------------------------------------------------------------- #
    # 3) Linear projections (flattened to use a single GEMM per weight matrix)
    # --------------------------------------------------------------------- #
    x_flat = x.reshape(flat_dim, dim)                     # (B·N·N, dim)

    left_proj_w = weights["left_proj.weight"]            # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]          # (hidden_dim, dim)

    left = F.linear(x_flat, left_proj_w)                 # (B·N·N, hidden_dim)
    right = F.linear(x_flat, right_proj_w)

    left = left.view(B, N, N, hidden_dim)                # [B,N,N,H]
    right = right.view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 4) Gates (also by a flattened GEMM)
    # --------------------------------------------------------------------- #
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x_flat, left_gate_w)).view(B, N, N, hidden_dim)
    right_gate = torch.sigmoid(F.linear(x_flat, right_gate_w)).view(B, N, N, hidden_dim)
    out_gate = torch.sigmoid(F.linear(x_flat, out_gate_w)).view(B, N, N, hidden_dim)

    # --------------------------------------------------------------------- #
    # 5) Fuse mask + gates with the projections via a tiny Triton kernel
    # --------------------------------------------------------------------- #
    if nomask:
        # No mask – just apply the gates
        left = left * left_gate
        right = right * right_gate
    else:
        # Ensure mask is float32 and contiguous
        mask = mask_tensor.to(left.dtype).contiguous()               # (B,N,N)
        mask_flat = mask.view(-1)                                    # (B·N·N)

        total_elems = left.numel()                                   # B·N·N·H
        # Allocate buffers for the fused results
        left_fused = torch.empty_like(left)
        right_fused = torch.empty_like(right)

        # Triton launch configuration
        BLOCK_SIZE = 65536  # each program processes 64K elements
        grid = lambda meta: (triton.cdiv(total_elems, meta["BLOCK_SIZE"]),)

        # Left branch
        fuse_mask_gate_kernel[grid](
            left.view(-1).data_ptr(),
            left_gate.view(-1).data_ptr(),
            mask_flat.data_ptr(),
            left_fused.view(-1).data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )
        # Right branch
        fuse_mask_gate_kernel[grid](
            right.view(-1).data_ptr(),
            right_gate.view(-1).data_ptr(),
            mask_flat.data_ptr(),
            right_fused.view(-1).data_ptr(),
            total_elems,
            hidden_dim,
            BLOCK_SIZE=BLOCK_SIZE,
        )

        left = left_fused
        right = right_fused

    # --------------------------------------------------------------------- #
    # 6) Core “tri‑multiplicative” contraction
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    -> treat each hidden channel as a separate batch and use torch.bmm
    # --------------------------------------------------------------------- #
    # reshape to (B*H, N, N) for batched GEMM
    left_bh = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B·H, N, N)
    right_bh = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # Batched matrix multiply: (B·H, N, N) @ (B·H, N, N)^T
    out_mat = torch.bmm(left_bh, right_bh.transpose(1, 2))            # (B·H, N, N)

    # reshape back to (B, N, N, H)
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # 7) Post‑processing (norm -> out‑gate -> final linear)
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out, normalized_shape=(hidden_dim,),
        weight=to_out_norm_w, bias=to_out_norm_b, eps=1e-5
    )

    # Apply the output gate (element‑wise)
    out = out * out_gate

    # Final linear projection back to `dim`
    to_out_w = weights["to_out.weight"]
    out_flat = out.view(flat_dim, hidden_dim)
    out = F.linear(out_flat, to_out_w)                # (B·N·N, dim)
    out = out.view(B, N, N, dim)                     # final shape

    return out
