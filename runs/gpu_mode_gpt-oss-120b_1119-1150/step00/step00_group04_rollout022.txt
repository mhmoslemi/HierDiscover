def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # Norm: using weight and bias
    eps = 1e-5
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)
    # Compute gates & projections
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    # Apply mask if needed
    if not nomask:
        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left_proj = left_proj * mask_f
        right_proj = right_proj * mask_f

    # Apply gate
    left = left_proj * left_gate
    right = right_proj * right_gate

    # Compute tri-mul: out_ij_d = sum_k left_i_k_d * right_j_k_d
    # Perform batched matmul per channel using Triton

    # Rearrange tensors to [B, H, N, N] and contiguous
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right for multiplication
    right_T = right_perm.transpose(2, 3).contiguous()

    B, H, N, _ = left_perm.shape
    total_bc = B * H

    left_flat = left_perm.view(total_bc, N, N)
    right_flat = right_T.view(total_bc, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Kernel launch
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (total_bc,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N, N, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Restore shape [B,N,N,H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # Apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Multiply by out_gate
    out = out * out_gate

    # Final linear projection
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

"""
TriMul forward pass (outgoing version) – Triton‑accelerated core.

Algorithm
---------
1️⃣ Layer‑normalize the 4‑D input tensor `x` (shape [B, N, N, C]).
2️⃣ Project `x` to a hidden space (`hidden_dim`) with two linear layers
   (`left_proj`, `right_proj`).  Compute per‑position gates
   (`left_gate`, `right_gate`, `out_gate`) via a sigmoid‑activated linear.
3️⃣ Apply mask (if present) and fuse the projected values with their gates:
   `left = left_proj * left_gate * mask` and analogously for `right`.
4️⃣ Core “TriMul” step:
      out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
   This is a batched matrix multiplication across the sequence dimension
   (i·k) × (k·j) for every hidden channel `h`.  We implement it with a
   custom Triton kernel that processes one `(batch, hidden)` pair per
   program and tiles the i/j dimensions.
5️⃣ Layer‑normalize the result (`to_out_norm`), apply the output gate and
   a final linear projection back to the original channel size `C`.

The Triton kernel handles the O(N³·hidden_dim) reduction efficiently
by tiling the matrix‑multiply and accumulating in FP32 while loading in
FP32 (or FP16 if desired).  All other lightweight operations are kept in
PyTorch for clarity.

The function `custom_kernel` receives a tuple
`(input_tensor, mask, weights, config)` and returns the processed
tensor of shape `[B, N, N, C]`.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication for the TriMul core.
# --------------------------------------------------------------
@triton.jit
def triton_tri_mul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers: left, right^T, output
    M, N, K,                           # M = N = seq_len, K = seq_len
    stride_ab, stride_am, stride_ak,   # a: batch, row (i), col (k)
    stride_bb, stride_bk, stride_bn,   # b: batch, row (k), col (j)
    stride_cb, stride_cm, stride_cn,   # c: batch, row (i), col (j)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(0)           # which (batch, hidden) pair
    pid_m    = tl.program_id(1)           # tile row   (i)
    pid_n    = tl.program_id(2)           # tile col   (j)

    # ----- tile offsets & masks -------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    m_mask = offs_m < M
    n_mask = offs_n < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ----- loop over the reduction dimension K ---------------------------------
    for k_tile in range(0, tl.cdiv(K, BLOCK_K)):
        start_k = k_tile * BLOCK_K
        offs_k = start_k + tl.arange(0, BLOCK_K)
        k_mask = offs_k < K

        # Pointers for this tile
        a_tile_ptr = (
            a_ptr
            + batch_idx * stride_ab
            + offs_m[:, None] * stride_am
            + offs_k[None, :] * stride_ak
        )
        b_tile_ptr = (
            b_ptr
            + batch_idx * stride_bb
            + offs_k[:, None] * stride_bk
            + offs_n[None, :] * stride_bn
        )

        # Load with out‑of‑bounds masking (read 0.0 where mask is false)
        a = tl.load(a_tile_ptr, mask=m_mask[:, None] & k_mask[None, :], other=0.0)
        b = tl.load(b_tile_ptr, mask=k_mask[:, None] & n_mask[None, :], other=0.0)

        # Accumulate the dot product
        acc += tl.dot(a, b)   # both a,b are (BLOCK_M, BLOCK_K) and (BLOCK_K, BLOCK_N)

    # ----- write back -----------------------------------------------------------
    c_tile_ptr = (
        c_ptr
        + batch_idx * stride_cb
        + offs_m[:, None] * stride_cm
        + offs_n[None, :] * stride_cn
    )
    tl.store(c_tile_ptr, acc, mask=m_mask[:, None] & n_mask[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, same device as input).
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (C == dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gating (no bias in all Linear layers)
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right_proj = torch.nn.functional.linear(x, weights["right_proj.weight"])

    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )  # [B, N, N, hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcasted on the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> [B, N, N, 1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # Fuse gate with projection
    left = left_proj * left_gate   # [B,N,N,hidden_dim]
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul: out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]
    #    Implemented as batched matmul (left @ right.T) per hidden channel.
    # ------------------------------------------------------------------
    # Rearrange to [B, hidden_dim, N, N] for convenient tiling
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous() # [B, H, N, N]

    # Transpose the last two dims of right -> shape (B, H, K=N, J=N)
    right_T = right_perm.transpose(2, 3).contiguous()   # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    total_bc = B * H  # flattened batch × hidden dimension

    # View as batch of matrices: [total_bc, N, N]
    left_flat = left_perm.view(total_bc, N, N)
    right_flat = right_T.view(total_bc, N, N)

    # Allocate output buffer
    out_flat = torch.empty_like(left_flat)

    # Tile sizes – chosen to fit H100 shared memory comfortably.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        total_bc,
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
    )

    triton_tri_mul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        N,                     # M
        N,                     # N
        N,                     # K
        left_flat.stride(0),   # stride_ab (batch)
        left_flat.stride(1),   # stride_am (row i)
        left_flat.stride(2),   # stride_ak (col k)
        right_flat.stride(0),  # stride_bb (batch)
        right_flat.stride(1),  # stride_bk (row k)
        right_flat.stride(2),  # stride_bn (col j)
        out_flat.stride(0),    # stride_cb (batch)
        out_flat.stride(1),    # stride_cm (row i)
        out_flat.stride(2),    # stride_cn (col j)
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Output LayerNorm, gating, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
