@triton.jit
def gating_kernel(
    left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr, out_gate_out_ptr,
    total_elem,
    H: tl.constexpr,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    block_start = pid * BLOCK_SIZE
    offs = block_start + tl.arange(0, BLOCK_SIZE)
    mask = offs < total_elem
    # compute mask index
    mask_idx = offs // H
    # loads
    l = tl.load(left_ptr + offs, mask=mask)
    r = tl.load(right_ptr + offs, mask=mask)
    lg = tl.load(left_gate_ptr + offs, mask=mask)
    rg = tl.load(right_gate_ptr + offs, mask=mask)
    og = tl.load(out_gate_ptr + offs, mask=mask)
    m = tl.load(mask_ptr + mask_idx, mask=mask)
    # sigmoid
    sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))
    l_gate = sigmoid(lg)
    r_gate = sigmoid(rg)
    o_gate = sigmoid(og)
    # apply mask (mask shape broadcast)
    l_out = l * m * l_gate
    r_out = r * m * r_gate
    # store
    tl.store(left_out_ptr + offs, l_out, mask=mask)
    tl.store(right_out_ptr + offs, r_out, mask=mask)
    tl.store(out_gate_out_ptr + offs, o_gate, mask=mask)

    hidden_dim = config["hidden_dim"]
    eps = 1e-5 (from layernorm default)

    # Extract weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]

        m = tl.load(mask_ptr + mask_idx, mask=mask)

        # sigmoid
        sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))

        r_gate = sigmoid(rg)
        o_gate = sigmoid(og)

        l_out = l * m * l_gate
        r_out = r * m * r_gate

        tl.store(left_out_ptr + offs, l_out, mask=mask)
        tl.store(right_out_ptr + offs, r_out, mask=mask)
        tl.store(out_gate_out_ptr + offs, o_gate, mask=mask)

    total_elements = left.numel()
    BLOCK_SIZE = 1024  # choose 1024 for enough parallelism
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)
    gating_kernel[grid](
        left, right, left_gate, right_gate, out_gate,
        mask,
        left_gated, right_gated, out_gate_sig,
        total_elem=total_elements,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Now compute core multiplication: left_gated @ right_gated.T per hidden dim
    B, N, N2, H = left_gated.shape
    assert N == N2
    # Reshape for batched matmul
    left_reshape = left_gated.permute(0, 3, 1, 2).reshape(B * H, N, N)
    right_reshape = right_gated.permute(0, 3, 2, 1).reshape(B * H, N, N)  # transpose second dim

    # Perform batched matrix multiplication
    out_mat = torch.bmm(left_reshape, right_reshape)  # shape (B*H, N, N)

    # Reshape back to [B,N,N,hidden_dim]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # Apply to_out_norm: layernorm over hidden_dim
    out_normed = torch.nn.functional.layer_norm(
        out, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=1e-5
    )

    # Apply out_gate sigmoid gating (already computed as out_gate_sig)
    out_gated = out_normed * out_gate_sig

    # Final linear projection to dim
    output = torch.nn.functional.linear(out_gated, to_out_weight)  # shape [B,N,N,dim] (weight shape dim x hidden_dim)

    return output

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, seq_len, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # gather weights (they are 2D for linear; 1D for layernorm)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm on input (dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # Linear projections
    left = torch.nn.functional.linear(x, left_proj_weight)  # shape [bs, N, N, hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # Gates (pre-sigmoid)
    left_gate = torch.nn.functional.linear(x, left_gate_weight)
    right_gate = torch.nn.functional.linear(x, right_gate_weight)
    out_gate = torch.nn.functional.linear(x, out_gate_weight)

    # Ensure contiguous and same dtype
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # Prepare mask (handle nomask)
    if mask is None:
        mask = torch.ones((bs, seq_len, seq_len), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype).contiguous()

    # Allocate outputs for gating
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)
    out_gate_sig = torch.empty_like(out_gate)

    # Triton kernel for gating and mask
    @triton.jit
    def gating_kernel(
        left_ptr, right_ptr, left_gate_ptr, right_gate_ptr, out_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr, out_gate_out_ptr,
        total_elems,
        H: tl.constexpr,
        BLOCK_SIZE: tl.constexpr):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elems

        # Load values
        l = tl.load(left_ptr + offs, mask=mask)
        r = tl.load(right_ptr + offs, mask=mask)
        lg = tl.load(left_gate_ptr + offs, mask=mask)
        rg = tl.load(right_gate_ptr + offs, mask=mask)
        og = tl.load(out_gate_ptr + offs, mask=mask)

        # Compute mask broadcast index
        mask_idx = offs // H
        m = tl.load(mask_ptr + mask_idx, mask=mask)

        # Sigmoid (approximate)
        sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))
        l_gate = sigmoid(lg)
        r_gate = sigmoid(rg)
        o_gate = sigmoid(og)

        # Apply mask and gating
        l_out = l * m * l_gate
        r_out = r * m * r_gate

        # Store
        tl.store(left_out_ptr + offs, l_out, mask=mask)
        tl.store(right_out_ptr + offs, r_out, mask=mask)
        tl.store(out_gate_out_ptr + offs, o_gate, mask=mask)

    total_elem = left.numel()
    BLOCK_SIZE = 1024
    grid = (triton.cdiv(total_elem, BLOCK_SIZE),)
    gating_kernel[grid](
        left, right, left_gate, right_gate, out_gate,
        mask,
        left_gated, right_gated, out_gate_sig,
        total_elems=total_elem,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # Core matrix multiplication per hidden dim
    B, N, _, H = left_gated.shape
    left_mat = left_gated.permute(0, 3, 1, 2).reshape(B * H, N, N)  # (B*H, N, N)
    right_mat = right_gated.permute(0, 3, 2, 1).reshape(B * H, N, N)  # (B*H, N, N)

    # Batched matmul
    out_mat = torch.bmm(left_mat, right_mat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1)

    # Apply second LayerNorm (hidden_dim)
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # Multiply by out_gate (sigmoid already computed)
    out = out * out_gate_sig

    # Final linear projection to dim
    output = torch.nn.functional.linear(out, to_out_weight)  # No bias per definition

    return output

"""
TriMul “outgoing” forward (AlphaFold‑3) – Triton‑fused gating.

The kernel fuses the three element‑wise operations that appear after the
linear projections:
* broadcasting the pairwise mask,
* sigmoid gating (left_gate, right_gate, out_gate),
* multiplication of the mask and the gates with the projected tensors.

The heavy part of the operator – the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
– is expressed as a batched matrix multiplication
(left @ rightᵀ) per hidden dimension using torch.bmm, which is already
highly optimized on H100.  The remaining LayerNorms and linear layers are
performed with torch.nn.functional helpers.

The implementation works for any batch size (1‑2), sequence length up to 1024,
and arbitrary `dim` / `hidden_dim` as used in the test configurations.
"""
import torch
import triton
import triton.language as tl


def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (or None when nomask is True)
            - weights: dict of model parameters
            - config: dict containing at least "dim" and "hidden_dim"

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------------- #
    # unpack inputs
    # --------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------------- #
    # retrieve weights (all are stored without bias unless otherwise noted)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]        # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]   # (hidden_dim,)
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]             # (dim, hidden_dim)

    # --------------------------------------------------------------------- #
    # LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------------- #
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # Linear projections (no bias)
    # --------------------------------------------------------------------- #
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # --------------------------------------------------------------------- #
    # Gate pre‑activations (still linear, sigmoid will be applied later)
    # --------------------------------------------------------------------- #
    left_gate = torch.nn.functional.linear(x, left_gate_weight)
    right_gate = torch.nn.functional.linear(x, right_gate_weight)
    out_gate = torch.nn.functional.linear(x, out_gate_weight)

    # --------------------------------------------------------------------- #
    # Ensure tensors are contiguous for the Triton kernel
    # --------------------------------------------------------------------- #
    left = left.contiguous()
    right = right.contiguous()
    left_gate = left_gate.contiguous()
    right_gate = right_gate.contiguous()
    out_gate = out_gate.contiguous()

    # --------------------------------------------------------------------- #
    # Prepare mask (if the caller passed ``None`` we treat it as all‑ones)
    # --------------------------------------------------------------------- #
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype).contiguous()   # shape [B,N,N]

    # --------------------------------------------------------------------- #
    # Allocate the tensors that will hold the gated results
    # --------------------------------------------------------------------- #
    left_gated = torch.empty_like(left)
    right_gated = torch.empty_like(right)
    out_gate_sig = torch.empty_like(out_gate)   # sigmoid(out_gate)

    # --------------------------------------------------------------------- #
    # Triton kernel: fuse mask, sigmoid gates and element‑wise multiplication
    # --------------------------------------------------------------------- #
    @triton.jit
    def gating_kernel(
        left_ptr, right_ptr,
        left_gate_ptr, right_gate_ptr,
        out_gate_ptr,
        mask_ptr,
        left_out_ptr, right_out_ptr, out_gate_out_ptr,
        total_elems,
        H: tl.constexpr,          # hidden_dim (compile‑time constant)
        BLOCK_SIZE: tl.constexpr, # program block size
    ):
        pid = tl.program_id(0)
        offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
        mask = offs < total_elems

        # Load raw data
        l  = tl.load(left_ptr + offs, mask=mask)
        r  = tl.load(right_ptr + offs, mask=mask)
        lg = tl.load(left_gate_ptr + offs, mask=mask)
        rg = tl.load(right_gate_ptr + offs, mask=mask)
        og = tl.load(out_gate_ptr + offs, mask=mask)

        # Broadcast mask from (B,N,N) to (B,N,N,H)
        mask_idx = offs // H                 # integer division – repeats each mask entry H times
        m = tl.load(mask_ptr + mask_idx, mask=mask)

        # Sigmoid implementation (identical to torch.sigmoid for fp32/fp16)
        sigmoid = lambda x: 1.0 / (1.0 + tl.exp(-x))
        l_gate = sigmoid(lg)
        r_gate = sigmoid(rg)
        o_gate = sigmoid(og)

        # Apply mask and gates
        l_out = l * m * l_gate
        r_out = r * m * r_gate

        # Store results
        tl.store(left_out_ptr + offs, l_out, mask=mask)
        tl.store(right_out_ptr + offs, r_out, mask=mask)
        tl.store(out_gate_out_ptr + offs, o_gate, mask=mask)

    total_elems = left.numel()                     # B * N * N * hidden_dim
    BLOCK_SIZE = 1024                               # safe block size for H100
    grid = (triton.cdiv(total_elems, BLOCK_SIZE),) # 1‑D launch grid

    gating_kernel[grid](
        left, right,
        left_gate, right_gate,
        out_gate,
        mask,
        left_gated, right_gated, out_gate_sig,
        total_elems=total_elems,
        H=hidden_dim,
        BLOCK_SIZE=BLOCK_SIZE,
    )

    # --------------------------------------------------------------------- #
    # Core tensor contraction:  out = Σ_k left[...,i,k,:] * right[...,j,k,:]
    # Implemented as a batched matmul (left @ rightᵀ) per hidden dimension.
    # --------------------------------------------------------------------- #
    # Reshape to (B*hidden_dim, N, N) where the inner matrices are (i,k) and (k,j)
    left_mat = left_gated.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, N, N)
    right_mat = right_gated.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N) # (B*H, N, N) with axes (k,j)

    # Batched matrix multiplication (highly tuned in cuBLAS)
    out_mat = torch.bmm(left_mat, right_mat)  # shape (B*H, N, N)

    # Restore the original 4‑D layout: [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------------- #
    # Second LayerNorm over the hidden dimension
    # --------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # --------------------------------------------------------------------- #
    # Apply the output gate (already sigmoid‑ed by the kernel)
    # --------------------------------------------------------------------- #
    out = out * out_gate_sig

    # --------------------------------------------------------------------- #
    # Final linear projection back to the original dimension (dim)
    # --------------------------------------------------------------------- #
    output = torch.nn.functional.linear(out, to_out_weight)  # bias is None by design

    return output
