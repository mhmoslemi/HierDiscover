We'll create kernel:

```
@triton.jit
def batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    batch_dim, M, N, K,
    stride_am, stride_ak, stride_ah,
    stride_bk, stride_bn, stride_bh,
    stride_cm, stride_cn, stride_ch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # compute offset for this batch
    a_batch_ptr = a_ptr + batch * stride_ah
    b_batch_ptr = b_ptr + batch * stride_bh
    c_batch_ptr = c_ptr + batch * stride_ch

    # Compute block offsets.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out of bounds
    mask_m = offs_m < M
    mask_n = offs_n < N

    # offsets for a and b
    a_ptrs = a_batch_ptr + (offs_m[:, None] * stride_am + tl.arange(0, BLOCK_K)[None, :] * stride_ak)
    b_ptrs = b_batch_ptr + (tl.arange(0, BLOCK_K)[:, None] * stride_bk + offs_n[None, :] * stride_bn)

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        a = tl.load(a_ptrs, mask=mask_m[:, None] & (k * BLOCK_K + tl.arange(0,BLOCK_K) < K), other=0.0)
        b = tl.load(b_ptrs, mask=(k * BLOCK_K + tl.arange(0,BLOCK_K)[:, None] < K) & mask_n[None, :], other=0.0)
        acc += tl.dot(a, b)

        # increment pointers
        a_ptrs += BLOCK_K * stride_ak
        b_ptrs += BLOCK_K * stride_bk

    # write back
    c_ptrs = c_batch_ptr + (offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

def layer_norm(x, weight, bias, eps=1e-5):
    # x: [..., C]
    # weight: (C,)
    # bias: (C,)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    if weight is not None:
        x_norm = x_norm * weight
    if bias is not None:
        x_norm = x_norm + bias
    return x_norm
```

Ensuring weight and bias must be broadcastable: x_norm shape [..., C]; weight shape [C]; they broadcast correctly.

"""
custom_kernel: Triton‑accelerated forward pass of AlphaFold3's “outgoing” TriMul.

Algorithm
---------
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections and gate‑generating projections (bias‑free) are applied
   to the flattened tensor, producing tensors of shape [B, N, N, H].
3. An optional mask (broadcast to the hidden dimension) zeroes out
   irrelevant positions both before and after gating.
4. The heavy O(N³·H) operation
       out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]
   is expressed as a batched matrix multiplication
       out_h = left_h @ right_hᵀ
   for every batch‑hidden pair (B·H).  This is implemented in a custom
   Triton kernel that (i) loads tiles of the two matrices in half‑precision,
   (ii) accumulates in FP32 for numerical stability, and (iii) writes the
   result back as FP32.
5. The result is layer‑normed (hidden dimension), multiplied element‑wise
   with the output gate, and finally projected back to the original channel
   dimension.
6. The final tensor has shape [B, N, N, C] and dtype torch.float32.

The kernel works for any batch size ≤2, sequence length up to 1024,
feature dimension C (e.g. 128/384) and hidden dimension H (e.g. 128).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,          # pointers
    M, N, K,                      # matrix sizes (M×K)·(K×N) -> (M×N)
    stride_am, stride_ak, stride_ah,   # a strides (row, col, batch)
    stride_bk, stride_bn, stride_bh,   # b strides (row=K, col=N, batch)
    stride_cm, stride_cn, stride_ch,   # c strides (row, col, batch)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM:  out[b] = a[b] @ b[b]ᵀ
    a: (B, M, K)  (half)
    b: (B, K, N)  (half)   – already transposed w.r.t the original right tensor
    c: (B, M, N)  (float32)
    """
    batch = tl.program_id(0)            # which (batch·hidden) pair
    pid_m = tl.program_id(1)            # block-row
    pid_n = tl.program_id(2)            # block-col

    # Compute the start offsets of the block.
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers for the current batch.
    a_batch = a_ptr + batch * stride_ah
    b_batch = b_ptr + batch * stride_bh
    c_batch = c_ptr + batch * stride_ch

    # Bounds checks.
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32.
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension.
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load A (M×K tile) – cast to FP32 for accumulation.
        a_ptrs = a_batch + (offs_m[:, None] * stride_am
                            + cur_k[None, :] * stride_ak)
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B (K×N tile, already transposed) – cast to FP32.
        b_ptrs = b_batch + (cur_k[:, None] * stride_bk
                            + offs_n[None, :] * stride_bn)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Accumulate.
        acc += tl.dot(a, b)

    # Write back the result.
    c_ptrs = c_batch + (offs_m[:, None] * stride_cm
                        + offs_n[None, :] * stride_cn)
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])


def _layer_norm(x, weight, bias, eps: float = 1e-5):
    """Layer‑norm over the last dimension (C)."""
    # x: (..., C)
    mean = x.mean(dim=-1, keepdim=True)
    var = ((x - mean) ** 2).mean(dim=-1, keepdim=True)
    inv = torch.rsqrt(var + eps)
    y = (x - mean) * inv
    if weight is not None:
        y = y * weight
    if bias is not None:
        y = y + bias
    return y

        - mask: torch.Tensor [B, N, N] (bool/float) – optional
        - weights: dict of torch.Tensor containing all linear and norm parameters
        - config: dict with at least ``hidden_dim`` (int) and optionally ``dim``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, C] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # Mask handling
    # ------------------------------------------------------------------
    if mask is None:
        mask = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask = mask.to(dtype)

    # ------------------------------------------------------------------
    # Layer‑norm on the input
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = _layer_norm(input_tensor, norm_w, norm_b)           # [B,N,N,C]

    # ------------------------------------------------------------------
    # Linear + gate projections (bias‑free)
    # ------------------------------------------------------------------
    # Flatten the spatial dimensions for a single batched linear call.
    x_flat = x.view(-1, C)                                 # [(B·N·N), C]

    # Projections
    left_proj = F.linear(x_flat, weights["left_proj.weight"]).view(B, N, N, hidden_dim)
    right_proj = F.linear(x_flat, weights["right_proj.weight"]).view(B, N, N, hidden_dim)

    # Gate‑generating projections
    left_gate = torch.sigmoid(F.linear(x_flat, weights["left_gate.weight"]).view(B, N, N, hidden_dim))
    right_gate = torch.sigmoid(F.linear(x_flat, weights["right_gate.weight"]).view(B, N, N, hidden_dim))
    out_gate = torch.sigmoid(F.linear(x_flat, weights["out_gate.weight"]).view(B, N, N, hidden_dim))

    # Apply mask (broadcast on the hidden dim)
    mask_exp = mask.unsqueeze(-1)   # [B,N,N,1]
    left = left_proj * mask_exp
    right = right_proj * mask_exp

    # Apply element‑wise gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Heavy O(N³·H) computation – Triton GEMM (left @ rightᵀ)
    # ------------------------------------------------------------------
    # Permute to (B, H, N, N) and make the right tensor transposed.
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, N, N) – indices (i,k)
    right_perm_T = right.permute(0, 3, 2, 1).contiguous()    # (B, H, N, N) – indices (k,j)

    # Collapse batch and hidden dimensions → (B·H, N, N)
    BH = B * hidden_dim
    left_mat = left_perm.view(BH, N, N)
    right_mat_T = right_perm_T.view(BH, N, N)

    # Use half‑precision for the GEMM inputs (memory / bandwidth saving).
    left_half = left_mat.to(torch.float16)
    right_half_T = right_mat_T.to(torch.float16)

    # Output buffer (FP32 accumulator → final float32 tensor)
    out_float = torch.empty((BH, N, N), dtype=torch.float32, device=device)

    # Strides (in element counts, not bytes)
    stride_am = left_half.stride(1)   # row‑stride (M dimension)
    stride_ak = left_half.stride(2)   # col‑stride (K dimension)
    stride_ah = left_half.stride(0)   # batch‑stride

    stride_bk = right_half_T.stride(1)   # K‑stride (row of B)
    stride_bn = right_half_T.stride(2)   # N‑stride (col of B)
    stride_bh = right_half_T.stride(0)   # batch‑stride

    stride_cm = out_float.stride(1)
    stride_cn = out_float.stride(2)
    stride_ch = out_float.stride(0)

    # Tuning parameters – chosen to fit H100 tensor‑core tile sizes.
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        BH,                               # batch * hidden
        (N + BLOCK_M - 1) // BLOCK_M,     # rows
        (N + BLOCK_N - 1) // BLOCK_N,     # cols
    )

    _batched_matmul_kernel[grid](
        left_half,
        right_half_T,
        out_float,
        N, N, N,                         # M, N, K (all equal to seq‑len)
        stride_am, stride_ak, stride_ah,
        stride_bk, stride_bn, stride_bh,
        stride_cm, stride_cn, stride_ch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,                     # good occupancy on H100
    )

    # Reshape back to (B,N,N,H)
    out = out_float.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # Post‑GEMM processing: to_out_norm, out_gate, final projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = _layer_norm(out, to_out_norm_w, to_out_norm_b)

    out = out * out_gate                        # gating

    # Final linear projection back to C channels.
    out_flat = out.view(-1, hidden_dim)
    final = F.linear(out_flat, weights["to_out.weight"]).view(B, N, N, C)

    return final
```
