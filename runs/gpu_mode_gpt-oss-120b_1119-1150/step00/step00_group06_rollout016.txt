@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_a_batch, stride_a_row, stride_a_col,
    stride_b_batch, stride_b_row, stride_b_col,
    stride_c_batch, stride_c_row, stride_c_col,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)
    # compute batch idx and row block
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid0 // num_blocks_m
    block_i = pid0 % num_blocks_m
    block_j = pid1

    # avoid out-of-range batches
    if batch_idx >= BATCH:
        return

    # row and col offsets
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for rows and cols
    mask_i = offs_i < M
    mask_j = offs_j < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # load A sub-matrix
        a_ptrs = A_ptr + batch_idx * stride_a_batch + offs_i[:, None] * stride_a_row + offs_k[None, :] * stride_a_col
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load B sub-matrix
        b_ptrs = B_ptr + batch_idx * stride_b_batch + offs_k[:, None] * stride_b_row + offs_j[None, :] * stride_b_col
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product accumulation
        acc += tl.dot(a, b)

    # Store results
    c_ptrs = C_ptr + batch_idx * stride_c_batch + offs_i[:, None] * stride_c_row + offs_j[None, :] * stride_c_col
    tl.store(c_ptrs, acc.to(tl.float32), mask=mask_i[:, None] & mask_j[None, :])
```

Implementation details for kernel:

When we pass `left_flat` and `right_flat` which are torch tensors of dtype torch.float16, the pointer is contiguous. However, Triton expects pointer types as `tl.pointer_type(dtype)`; they will be `tl.float16` type automatically from load.

Implementation of kernel:

```
@triton.jit
def batch_matmul_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Ar, stride_Ac,
    stride_Bb, stride_Br, stride_Bc,
    stride_Cb, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid0 = tl.program_id(0)
    pid1 = tl.program_id(1)

    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid0 // num_blocks_m
    block_i = pid0 % num_blocks_m
    block_j = pid1

    if batch_idx >= BATCH:
        return

    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A submatrix (M x K)
        a_ptrs = A_ptr + batch_idx * stride_Ab + offs_i[:, None] * stride_Ar + offs_k[None, :] * stride_Ac
        a = tl.load(a_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load B submatrix (K x N)
        b_ptrs = B_ptr + batch_idx * stride_Bb + offs_k[:, None] * stride_Br + offs_j[None, :] * stride_Bc
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(a, b)

    # Store result
    c_ptrs = C_ptr + batch_idx * stride_Cb + offs_i[:, None] * stride_Cr + offs_j[None, :] * stride_Cc
    tl.store(c_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul custom kernel (outgoing version).

The forward pass consists of:
1. LayerNorm on the input tensor (shape [B,N,N,D]).
2. Two linear projections (left/right) from D → H (hidden_dim) without bias.
3. Optional mask (broadcasted over the hidden dimension).
4. Multiplicative gating (sigmoid of three more linear layers).
5. Core O(N³·H) operation:
       out_{b,i,j,h} = Σ_k left_{b,i,k,h} * right_{b,j,k,h}
   which is a batched matrix multiplication L_h @ R_hᵀ for each hidden
   slice h.  This is implemented with a Triton kernel that computes a
   GEMM for each (batch, hidden) pair.
6. Second LayerNorm + output gating + final linear projection back to D.

Only the heavy O(N³·H) multiplication is done in Triton; all other
operations use PyTorch (they are cheap compared to the GEMM).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched GEMM A @ Bᵀ where B is provided already transposed
# (i.e. B has shape [batch, K, N] and we treat its rows as K, cols as N).
# ----------------------------------------------------------------------
@triton.jit
def batched_gemm_kernel(
    A_ptr, B_ptr, C_ptr,
    BATCH, M, N, K,
    stride_Ab, stride_Ar, stride_Ac,
    stride_Bb, stride_Br, stride_Bc,
    stride_Cb, stride_Cr, stride_Cc,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute C[b] = A[b] @ B[b]ᵀ for each batch."""
    pid0 = tl.program_id(0)          # encodes batch index and block-i
    pid1 = tl.program_id(1)          # block-j

    # --------------------------------------------------------------
    # Map pid0 -> (batch, block_i)
    # --------------------------------------------------------------
    num_blocks_m = tl.cdiv(M, BLOCK_M)
    batch_idx = pid0 // num_blocks_m
    block_i = pid0 % num_blocks_m
    block_j = pid1

    # Out‑of‑range guard (in case grid is slightly larger)
    if batch_idx >= BATCH:
        return

    # --------------------------------------------------------------
    # Offsets for the current tile
    # --------------------------------------------------------------
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    mask_i = offs_i < M
    mask_j = offs_j < N

    # Accumulator in FP32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension in blocks of size BLOCK_K
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a_ptrs = (
            A_ptr
            + batch_idx * stride_Ab
            + offs_i[:, None] * stride_Ar
            + offs_k[None, :] * stride_Ac
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)

        # Load B block (already transposed): shape (BLOCK_K, BLOCK_N)
        b_ptrs = (
            B_ptr
            + batch_idx * stride_Bb
            + offs_k[:, None] * stride_Br
            + offs_j[None, :] * stride_Bc
        )
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0).to(tl.float32)

        # Multiply‑accumulate
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Store the result tile back to C
    # --------------------------------------------------------------
    c_ptrs = (
        C_ptr
        + batch_idx * stride_Cb
        + offs_i[:, None] * stride_Cr
        + offs_j[None, :] * stride_Cc
    )
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module using a Triton batched GEMM.

    Args:
        data: tuple consisting of
            - input_tensor (torch.Tensor)      : shape [B, N, N, D]
            - mask_tensor  (torch.Tensor) or None: shape [B, N, N]
            - weights       (dict)              : model weights (see keys below)
            - config        (dict)              : contains "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, D] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    if len(data) == 4:
        input_tensor, mask_tensor, weights, config = data
    else:  # fallback if mask omitted
        input_tensor, weights, config = data
        mask_tensor = None

    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, D = input_tensor.shape
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to fetch weight tensors (all should already be on the same device)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (affine)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        mask = mask_tensor.to(dtype=dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid of linear layers, no bias)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core O(N³·H) multiplication using Triton
    # ------------------------------------------------------------------
    # Cast to float16 for bandwidth‑limited GEMM
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to (B, hidden, N, N) where the last two dimensions are
    # (row, col) for left and (col, row) for right.  The right operand is
    # transposed so that a simple GEMM gives the required sum over k.
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # (B, H, i, k)
    right_perm = right.permute(0, 3, 2, 1).contiguous()       # (B, H, k, j)

    # Collapse batch & hidden into a single GEMM batch dimension
    BHD = B * hidden_dim
    left_flat = left_perm.view(BHD, N, N)    # (BHD, i, k)
    right_flat = right_perm.view(BHD, N, N)  # (BHD, k, j)

    # Allocate output buffer (float16, will be promoted later)
    out_flat = torch.empty_like(left_flat, dtype=torch.float16, device=device)

    # Strides for the three tensors (batch, row, col)
    stride_Ab, stride_Ar, stride_Ac = left_flat.stride()
    stride_Bb, stride_Br, stride_Bc = right_flat.stride()
    stride_Cb, stride_Cr, stride_Cc = out_flat.stride()

    # Convert strides to Python ints for Triton
    stride_Ab, stride_Ar, stride_Ac = int(stride_Ab), int(stride_Ar), int(stride_Ac)
    stride_Bb, stride_Br, stride_Bc = int(stride_Bb), int(stride_Br), int(stride_Bc)
    stride_Cb, stride_Cr, stride_Cc = int(stride_Cb), int(stride_Cr), int(stride_Cc)

    # Block sizes (tuned for H100 / float16)
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    num_blocks_m = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_n = (N + BLOCK_N - 1) // BLOCK_N
    # Grid: (total_batches * num_blocks_m, num_blocks_n)
    grid = (BHD * num_blocks_m, num_blocks_n)

    batched_gemm_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        BHD, N, N, N,
        stride_Ab, stride_Ar, stride_Ac,
        stride_Bb, stride_Br, stride_Bc,
        stride_Cb, stride_Cr, stride_Cc,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,  # typical value for fp16 GEMM on H100
    )

    # Reshape back to [B, N, N, hidden_dim] and cast to float32 for the rest
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous().to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm, output gating, final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate  # element‑wise broadcast over hidden dimension
    out = F.linear(out, to_out_weight)  # back to original dim

    return out
