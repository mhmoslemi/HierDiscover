import torch
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(lhs_ptr, rhs_ptr, out_ptr,
                  B, N, D,
                  stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_d,
                  stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_d,
                  stride_out_b, stride_out_i, stride_out_j, stride_out_d,
                  BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    """Compute out[b,i,j,d] = sum_k lhs[b,i,k,d] * rhs[b,j,k,d]

    The kernel iterates over blocks of i,j and accumulates over k.
    """
    pid0 = tl.program_id(0)  # combined batch * hidden_dim dimension
    pid_i = tl.program_id(1)  # i block index
    pid_j = tl.program_id(2)  # j block index

    # split batch and hidden dimension
    b = pid0 // D
    d = pid0 % D

    # Offsets for i and j
    i_offsets = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offsets = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_offsets < N
    mask_j = j_offsets < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Number of k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k_blocks):
        k_offsets = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offsets < N

        # Load lhs block: shape (BLOCK_I, BLOCK_K)
        lhs_ptrs = (lhs_ptr
                    + b * stride_lhs_b
                    + d * stride_lhs_d
                    + i_offsets[:, None] * stride_lhs_i
                    + k_offsets[None, :] * stride_lhs_k)
        lhs = tl.load(lhs_ptrs,
                      mask=mask_i[:, None] & mask_k[None, :],
                      other=0.0)

        # Load rhs block as (BLOCK_K, BLOCK_J) so that we can do dot(lhs, rhs)
        rhs_ptrs = (rhs_ptr
                    + b * stride_rhs_b
                    + d * stride_rhs_d
                    + k_offsets[:, None] * stride_rhs_k
                    + j_offsets[None, :] * stride_rhs_j)
        rhs = tl.load(rhs_ptrs,
                      mask=mask_k[:, None] & mask_j[None, :],
                      other=0.0)

        # Cast to FP32 for accumulation
        lhs = lhs.to(tl.float32)
        rhs = rhs.to(tl.float32)

        acc += tl.dot(lhs, rhs)

    # Store result
    out_ptrs = (out_ptr
                + b * stride_out_b
                + d * stride_out_d
                + i_offsets[:, None] * stride_out_i
                + j_offsets[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """TriMul “outgoing” forward pass implemented with a fused Triton kernel.

    The implementation follows the PyTorch reference:
        1. LayerNorm on the last dimension (dim).
        2. Linear projections to hidden_dim (left/right) and gate values.
        3. Apply mask (if provided) and gates.
        4. Execute a Triton kernel that computes, for each hidden channel d,
           out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
        5. LayerNorm on hidden_dim, multiply by an output gate and a final linear
           projection back to dim.

    Arguments:
        data: Tuple (input_tensor, mask, weights, config).
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32).
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None.
            - weights: dict of torch.Tensors containing model parameters.
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool).

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32).
    """
    import math
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ----------------------------------------------------------------------
    # 1. LayerNorm on dim
    # ----------------------------------------------------------------------
    eps = 1e-5
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor,
                                       (dim,),
                                       weight=norm_weight,
                                       bias=norm_bias,
                                       eps=eps)

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate linear layers
    # ----------------------------------------------------------------------
    # Projection weights
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ----------------------------------------------------------------------
    # 3. Mask (if any) and gates
    # ----------------------------------------------------------------------
    if not nomask:
        # mask is expected shape [B,N,N]; ensure float type and broadcast
        mask_f = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 4. Triton kernel for the N^3 multiplication (per hidden channel)
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Cast to half for faster GEMM inside Triton
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()
    # Prepare output buffer
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=device)

    # Strides (int64)
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_d = left_f16.stride()
    stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_d = right_f16.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Block sizes – tuned for H100; you may adjust if needed
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64   # Must divide well, but we handle tail with masks

    # Grid dimensions
    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    total_bd = B * hidden_dim  # combined batch and hidden-channel dimension

    grid = (total_bd, num_i, num_j)

    trimul_kernel[grid](
        left_f16,
        right_f16,
        out,
        B,
        N,
        hidden_dim,
        stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_d,
        stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # Convert back to fp32 for subsequent layers
    out_fp32 = out.to(torch.float32)

    # ----------------------------------------------------------------------
    # 5. Post‑processing: to_out_norm, output gate, final linear
    # ----------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(out_fp32,
                                              (hidden_dim,),
                                              weight=to_out_norm_w,
                                              bias=to_out_norm_b,
                                              eps=eps)
    # Apply output gate (shape [B,N,N,hidden_dim])
    out_norm = out_norm * out_gate

    # Final linear projection back to dim
    to_out_w = weights["to_out.weight"]
    result = torch.nn.functional.linear(out_norm, to_out_w)   # [B,N,N,dim]

    return result

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------
# Triton kernel
# --------------------------------------------------------------
@triton.jit
def trimul_kernel(
    lhs_ptr, rhs_ptr, out_ptr,
    B, N, D,
    stride_lhs_b, stride_lhs_i, stride_lhs_k, stride_lhs_d,
    stride_rhs_b, stride_rhs_j, stride_rhs_k, stride_rhs_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Compute
       out[b, i, j, d] = Σ_k lhs[b, i, k, d] * rhs[b, j, k, d]

    Each program works on a tile of (i, j) for a single (b, d) pair.
    """
    pid0 = tl.program_id(0)          # combined batch * hidden_dim index
    pid_i = tl.program_id(1)         # i‑tile index
    pid_j = tl.program_id(2)         # j‑tile index

    # split the combined index into batch and hidden‑channel
    b = pid0 // D
    d = pid0 % D

    # tile offsets in the i / j dimensions
    i_off = pid_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_off = pid_j * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = i_off < N
    mask_j = j_off < N

    # accumulator in FP32 (higher precision than fp16)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over the k dimension in blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kk in range(num_k_blocks):
        k_off = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_off < N

        # ---- load lhs[i, k] -------------------------------------------------
        # shape of the loaded block: (BLOCK_I, BLOCK_K)
        lhs = tl.load(
            lhs_ptr
            + b * stride_lhs_b
            + d * stride_lhs_d
            + i_off[:, None] * stride_lhs_i
            + k_off[None, :] * stride_lhs_k,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        )

        # ---- load rhs[j, k] as (BLOCK_K, BLOCK_J) --------------------------
        # we transpose on‐the‑fly so that a dot product can be used.
        rhs = tl.load(
            rhs_ptr
            + b * stride_rhs_b
            + d * stride_rhs_d
            + k_off[:, None] * stride_rhs_k
            + j_off[None, :] * stride_rhs_j,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        )

        # cast to FP32 for the dot product
        lhs = lhs.to(tl.float32)
        rhs = rhs.to(tl.float32)

        # accumulate: (BLOCK_I, BLOCK_K) • (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(lhs, rhs)

    # ---- write the result ----------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + d * stride_out_d
        + i_off[:, None] * stride_out_i
        + j_off[None, :] * stride_out_j
    )
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    """
    TriMul “outgoing” forward pass (no gradient).

    Implements the PyTorch reference:
      1. LayerNorm over the last dimension (dim).
      2. Linear projections (left/right) and three gate linear layers.
      3. Optional mask application.
      4. Triton kernel that evaluates
            out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d].
      5. LayerNorm over the hidden dimension, multiply by an output gate,
         and a final linear projection back to `dim`.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) – may be dummy.
        - weights      : dict of torch.Tensors containing all model parameters.
        - config       : dict with keys
            * "dim"         – input/output channel size
            * "hidden_dim"  – internal hidden dimension
            * "nomask"      – bool, True when mask should be ignored.

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32) – the forward output.
    """
    import math

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    inp, mask, weights, cfg = data
    device = inp.device
    dtype = inp.dtype

    dim = cfg["dim"]
    hidden_dim = cfg["hidden_dim"]
    nomask = cfg.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the `dim` axis
    # ------------------------------------------------------------------
    eps = 1e-5
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        inp, (dim,), weight=norm_w, bias=norm_b, eps=eps
    )  # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate projections
    # ------------------------------------------------------------------

    right_proj_w = weights["right_proj.weight"]

    out_gate_w = weights["out_gate.weight"]

        left = left * mask_f
        right = right * mask_f

    # gate values (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Triton kernel: Σ_k left[...,k,d] * right[...,k,d]
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # block sizes – tuned for H100, can be adjusted if needed
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 64

    num_i = (N + BLOCK_I - 1) // BLOCK_I
    num_j = (N + BLOCK_J - 1) // BLOCK_J
    total_bd = B * hidden_dim            # combined batch × hidden‑channel axis

    grid = (total_bd, num_i, num_j)

    trimul_kernel[grid](
        left_f16,
        right_f16,
        out_f16,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_d,
        s_r_b,
        s_r_j,
        s_r_k,
        s_r_d,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # Convert back to FP32 for the rest of the pipeline
    out = out_f16.to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Post‑processing: hidden‑dim LayerNorm, output gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out_norm = torch.nn.functional.layer_norm(

        bias=to_out_norm_b,

    to_out_w = weights["to_out.weight"]
    result = torch.nn.functional.linear(out_norm, to_out_w)  # [B, N, N, dim]

    return result
