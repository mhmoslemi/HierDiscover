    X_ptr, Y_ptr, weight_ptr, bias_ptr,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    B, N, D,

    # if pid >= B*N*N: return (skip)

    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N
    # Compute base offset:
    x_ptr = X_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_ptr = Y_ptr + b * stride_yb + i * stride_yi + j * stride_yj
    # Compute mean / variance across D dimension.
    # Loop over D in chunks
    sum = tl.zeros([1], dtype=tl.float32)
    sum2 = tl.zeros([1], dtype=tl.float32)
    for off in range(0, D, BLOCK_D):
        cur_block = tl.minimum(D - off, BLOCK_D)
        mask = tl.arange(0, BLOCK_D) < cur_block
        # load
        x = tl.load(x_ptr + off * stride_xc + tl.arange(0, BLOCK_D) * stride_xc, mask=mask, other=0.0).to(tl.float32)

        sum += tl.sum(x, axis=0)
        sum2 += tl.sum(x * x, axis=0)
    mean = sum / D
    var = sum2 / D - mean * mean
    rstd = 1.0 / tl.sqrt(var + eps)
    # Load again to store normalized values
    for off in range(0, D, BLOCK_D):
        cur_block = tl.minimum(D - off, BLOCK_D)
        mask = tl.arange(0, BLOCK_D) < cur_block

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # 1) layernorm input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=1e-5)
    # 2) compute linear projections: left and right
    left_proj_weight = weights["left_proj.weight"]  # shape (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]
    # linear: x @ left_proj_weight.T
    # Use torch.nn.functional.linear
    left = torch.nn.functional.linear(x, left_proj_weight)  # weight shape (out_features, in_features)
    right = torch.nn.functional.linear(x, right_proj_weight)
    # 3) mask if not none
    if mask is not None:
        mask_ = mask.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_
        right = right * mask_
    # 4) compute gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x, out_gate_weight).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # 5) batched matmul across sequence dimension per hidden_dim
    B, N, N2, H = left.shape
    # reshape: (B, H, N, N)
    left_reshaped = left.permute(0, 3, 1, 2).contiguous()
    right_reshaped = right.permute(0, 3, 1, 2).contiguous()
    left_flat = left_reshaped.view(B * H, N, N)
    right_flat = right_reshaped.view(B * H, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1)  # shape (B,N,N,H)
    # 6) to_out_norm (LayerNorm over hidden_dim)
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=1e-5)
    # 7) out gating
    out = out * out_gate
    # 8) final linear to_out (hidden_dim -> dim)
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

"""
TriMul "outgoing" kernel (AlphaFold3) – forward only.

Algorithm
---------
1. Layer‑norm the input tensor `x` (shape B×N×N×C) using a custom Triton kernel.
2. Compute left/right linear projections (C → H) and the three gating projections
   (C → H).  The gates are passed through a sigmoid.
3. Apply optional pairwise mask and element‑wise gate multiplication.
4. For every hidden dimension `h` compute the matrix product
      out_h = left_h @ right_hᵀ   (B·H independent N×N GEMMs)
   which implements the Einstein sum
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h].
   This is performed with a batched torch.bmm (highly tuned cuBLAS).
5. Layer‑norm over the hidden dimension using the same Triton kernel.
6. Multiply by the output gate, then a final linear projection back to C.

Only the layer‑norms are implemented in Triton (the heavy N³ reduction is
delegated to a batched GEMM).  The kernel works for any batch size `B` (1‑2),
any sequence length `N` up to 1024 and channel dimensions `C`/`H` up to 384.
The final output is float32 on the same device as the input.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton layer‑norm kernel (generic over channel dimension D)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr,               # *const pointer to input  (B,N,N,D)
    y_ptr,               # *mut   pointer to output (B,N,N,D)
    weight_ptr,          # *const pointer to weight (D,)
    bias_ptr,            # *const pointer to bias   (D,)
    stride_xb, stride_xi, stride_xj, stride_xc,   # input strides
    stride_yb, stride_yi, stride_yj, stride_yc,   # output strides
    B, N, D,                                 # runtime dimensions
    eps: tl.constexpr = 1e-5,
    BLOCK_D: tl.constexpr = 128,
):
    """
    Compute per‑vector layer‑norm across the last dimension.
    One program (program_id) processes a single (b,i,j) vector.
    """
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ---- decode (b,i,j) from linear pid ----
    b = pid // (N * N)
    ij = pid % (N * N)
    i = ij // N
    j = ij % N

    # ---- base pointers for this vector ----
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ---- first pass: compute mean & variance ----
    sum_val = tl.zeros([1], dtype=tl.float32)
    sum_sq = tl.zeros([1], dtype=tl.float32)

    for offs in range(0, D, BLOCK_D):
        cur = tl.minimum(D - offs, BLOCK_D)
        mask = tl.arange(0, BLOCK_D) < cur
        idx = offs + tl.arange(0, BLOCK_D)
        x = tl.load(x_base + idx * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq  += tl.sum(x * x, axis=0)

    mean = sum_val / D
    var  = sum_sq / D - mean * mean
    rstd = tl.rsqrt(var + eps)   # 1 / sqrt(var + eps)

    # ---- second pass: write normalized output ----
    for offs in range(0, D, BLOCK_D):
        cur = tl.minimum(D - offs, BLOCK_D)
        mask = tl.arange(0, BLOCK_D) < cur
        idx = offs + tl.arange(0, BLOCK_D)

        x = tl.load(x_base + idx * stride_xc, mask=mask, other=0.0).to(tl.float32)

        w = tl.load(weight_ptr + idx, mask=mask, other=0.0)
        b_ = tl.load(bias_ptr + idx, mask=mask, other=0.0)

        y = ((x - mean) * rstd) * w + b_
        tl.store(y_base + idx * stride_yc, y, mask=mask)


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5) -> torch.Tensor:
    """
    Wrapper for the Triton layer‑norm kernel.
    Arguments:
        x     – input tensor of shape (B,N,N,D) (contiguous, float32)
        weight, bias – 1‑D tensors of length D (float32)
    Returns:
        Normalized tensor of same shape and dtype as `x`.
    """
    assert x.is_contiguous()
    B, N, N2, D = x.shape
    assert N == N2, "input must be square in the two middle dims"
    y = torch.empty_like(x)

    # Strides are expressed in number of elements (not bytes)
    stride_xb, stride_xi, stride_xj, stride_xc = x.stride()
    stride_yb, stride_yi, stride_yj, stride_yc = y.stride()

    total_vectors = B * N * N
    grid = (total_vectors,)

    # Choose a block size that evenly covers the channel dimension.
    # 128 works for D up to 384 (3 iterations max).
    layernorm_kernel[grid](
        x, y,
        weight, bias,
        stride_xb, stride_xi, stride_xj, stride_xc,
        stride_yb, stride_yi, stride_yj, stride_yc,
        B, N, D,
        eps=eps,
        BLOCK_D=128,
        num_warps=4,
    )
    return y


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operator.
    Input data is a tuple:
        (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape (B, N, N, dim) on the same device as the input.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype
    # ------------------------------------------------------------------
    # 1) Input layer‑norm (over `dim`)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = layernorm_triton(input_tensor, norm_weight, norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 2) Linear projections (C → H) and gates (C → H)
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (H, C)
    right_proj_weight = weights["right_proj.weight"]  # (H, C)

    left = torch.nn.functional.linear(x, left_proj_weight)   # (B,N,N,H)
    right = torch.nn.functional.linear(x, right_proj_weight) # (B,N,N,H)

    # ------------------------------------------------------------------
    # 3) Apply pairwise mask (if provided)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: (B,N,N) → (B,N,N,1) for broadcasting
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left  = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and gated multiplication
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]   # (H, C)
    right_gate_weight = weights["right_gate.weight"] # (H, C)
    out_gate_weight   = weights["out_gate.weight"]   # (H, C)

    left_gate  = torch.nn.functional.linear(x, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x, out_gate_weight).sigmoid()

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core N³ operation via batched GEMM:
    #    out_h = left_h @ right_hᵀ   for each hidden dim h
    # ------------------------------------------------------------------
    B, N, _, H = left.shape  # H == hidden_dim
    # reshape to (B, H, N, N) → (B*H, N, N) for batched matmul
    left_perm  = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_flat  = left_perm.view(B * H, N, N)
    right_flat = right_perm.view(B * H, N, N)

    # out_flat: (B*H, N, N) = left_flat @ right_flatᵀ
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, H)
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6) Post‑projection layer‑norm (over hidden_dim)
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = layernorm_triton(out, to_out_norm_weight, to_out_norm_bias, eps=1e-5)

    # ------------------------------------------------------------------
    # 7) Output gating and final linear projection (H → C)
    # ------------------------------------------------------------------
    out = out * out_gate
    to_out_weight = weights["to_out.weight"]   # (C, H)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B,N,N,C)

    return out
