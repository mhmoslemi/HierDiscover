def matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,  # a batch stride (the stride between batches)
    stride_bk, stride_bj, stride_bb,
    stride_cm, stride_cn, stride_cb,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(0)  # for M blocks
    pid_n = tl.program_id(1)  # for N blocks
    pid_batch = tl.program_id(2)  # for batch

    # compute offsets into M and N
    offs_m = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        # compute offsets for A and B
        k_base = k * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # a_ptr offsets: (batch_stride * pid_batch) + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak)
        a_offset = pid_batch * stride_ab + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        # b_ptr offsets: (batch_stride * pid_batch) + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bj)
        b_offset = pid_batch * stride_bb + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj

        a = tl.load(a_ptr + a_offset, mask=(offs_m < M)[:, None] & (offs_k < K)[None, :], other=0.0)
        b = tl.load(b_ptr + b_offset, mask=(offs_k < K)[:, None] & (offs_n < N)[None, :], other=0.0)

        # accumulate using dot
        acc += tl.dot(a, b)  # both fp16? Actually a,b are half; but to avoid overflow use float
    # write back
    c_offset = pid_batch * stride_cb + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    tl.store(c_ptr + c_offset, acc.to(tl.float16), mask=(offs_m < M)[:, None] & (offs_n < N)[None, :])
```

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl
    # parse inputs
    x, mask, weights, config = data
    bs, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    device = x.device
    # LayerNorm on x
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    # compute mean and var across dim
    # Using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)
    # Mask (if any)
    nomask = config.get("nomask", True)
    if not nomask:

        mask_f = mask.to(x.dtype).unsqueeze(-1)  # shape (bs, N, N, 1)
        left = left * mask_f
        right = right * mask_f
    # Gating
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()

    right = right * right_gate

    left = left.permute(0, 3, 1, 2).contiguous()

    def triton_matmul_kernel(

        M, N, K,
        stride_am, stride_ak, stride_ab,
        stride_bk, stride_bj, stride_bb,
        stride_cm, stride_cn, stride_cb,

        BLOCK_K: tl.constexpr

"""
custom_kernel – TriMul “outgoing” forward pass.

Algorithm
---------
1. Layer‑norm the input tensor `x` over the last dimension (dim) using the
   provided `norm.{weight,bias}`.
2. Linear projections (no bias) to a hidden dimension `hidden_dim`:
   * left  = x_norm @ left_proj.weight.T
   * right = x_norm @ right_proj.weight.T
3. (Optional) apply a binary mask of shape [B, N, N] by broadcasting
   over the hidden dimension.
4. Compute element‑wise sigmoid gates and apply them:
   * left  *= sigmoid(x_norm @ left_gate.weight.T)
   * right *= sigmoid(x_norm @ right_gate.weight.T)
5. **Core cubic operation** – for every hidden channel d and batch b,
   compute a matrix product
       out_{b,d} = left_{b,d} @ right_{b,d}.T
   where left/right are N×N matrices.  This is implemented with a
   custom Triton kernel that tiles the multiplication (BLOCK_M×BLOCK_N×BLOCK_K)
   and accumulates in FP32 while loading/storing in FP16.
6. Reshape the result back to [B, N, N, hidden_dim].
7. Layer‑norm over the hidden dimension using `to_out_norm.{weight,bias}`.
8. Multiply by an output gate `sigmoid(x_norm @ out_gate.weight.T)`.
9. Final linear projection back to `dim` using `to_out.weight`.
10. Return the tensor of shape [B, N, N, dim] (float32).

The Triton kernel performs the batched matrix multiplication for step 5,
while all other operations stay in PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------
# Triton kernel: batched matmul C = A @ Bᵀ
# A : (batch, M, K) row‑major
# B : (batch, N, K) row‑major, accessed as Bᵀ via swapped strides
# C : (batch, M, N) row‑major
# ------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_ab,    # A: row‑stride, col‑stride, batch‑stride
    stride_bk, stride_bj, stride_bb,    # Bᵀ: k‑stride (col‑stride of original B), j‑stride (row‑stride of original B), batch‑stride
    stride_cm, stride_cn, stride_cb,    # C: row‑stride, col‑stride, batch‑stride
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)                # block index in M‑dimension
    pid_n = tl.program_id(1)                # block index in N‑dimension
    pid_b = tl.program_id(2)                # batch index

    # Offsets for the output block
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Pointers shifted to the current batch
    a_batch_offset = pid_b * stride_ab
    b_batch_offset = pid_b * stride_bb
    c_batch_offset = pid_b * stride_cb

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction axis K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K
        offs_k = cur_k + tl.arange(0, BLOCK_K)

        # Load A tile (M×K)
        a_mask = (offs_m < M)[:, None] & (offs_k < K)[None, :]
        a_ptrs = a_ptr + a_batch_offset + offs_m[:, None] * stride_am + offs_k[None, :] * stride_ak
        a = tl.load(a_ptrs, mask=a_mask, other=0.0)

        # Load Bᵀ tile (K×N) – note swapped strides give us Bᵀ without a copy
        b_mask = (offs_k < K)[:, None] & (offs_n < N)[None, :]
        b_ptrs = b_ptr + b_batch_offset + offs_k[:, None] * stride_bk + offs_n[None, :] * stride_bj
        b = tl.load(b_ptrs, mask=b_mask, other=0.0)

        # Matrix‑multiply‑accumulate (a: M×K, b: K×N → M×N)
        acc += tl.dot(a, b)      # result accumulated in FP32
    # Store C tile back to memory (cast to FP16)
    c_ptrs = c_ptr + c_batch_offset + offs_m[:, None] * stride_cm + offs_n[None, :] * stride_cn
    c_mask = (offs_m < M)[:, None] & (offs_n < N)[None, :]
    tl.store(c_ptrs, acc.to(tl.float16), mask=c_mask)


def custom_kernel(data):
    """
    TriMul forward (outgoing) – expects
        data = (input, mask, weights, config)
    Returns a tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1) LayerNorm over the last dimension
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        mask_f = mask.to(x.dtype).unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating (sigmoid) and element‑wise gating
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate   = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core cubic operation: batched matmul per hidden channel
    #    left/right: [B, N, N, hidden_dim] → move hidden_dim to batch axis
    # ------------------------------------------------------------------
    # Convert to half precision before reshaping to save memory
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Shape → (B, hidden_dim, N, N) and make contiguous
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()

    B2, H, M, K = left.shape          # B2 = bs, H = hidden_dim, M = K = N
    batch = B2 * H

    # Views of shape (batch, N, N)
    left_3d = left.view(batch, M, K)
    right_3d = right.view(batch, M, K)

    # Output buffer
    out_3d = torch.empty_like(left_3d, dtype=torch.float16, device=device)

    # ------------------------------------------------------------------
    # Launch Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    # Strides of the 3‑D tensors (they are row‑major, contiguous)
    stride_a_batch = left_3d.stride(0)   # = N * N
    stride_am      = left_3d.stride(1)   # = N   (row‑stride)
    stride_ak      = left_3d.stride(2)   # = 1   (col‑stride)

    # B will be accessed as Bᵀ → swap k/j strides
    stride_b_batch = right_3d.stride(0)   # = N * N
    stride_bk = 1                         # stride of k in Bᵀ (original column stride)
    stride_bj = K                         # stride of j in Bᵀ (original row stride = N)

    # Output strides
    stride_c_batch = out_3d.stride(0)    # = N * N
    stride_cm      = out_3d.stride(1)    # = N
    stride_cn      = out_3d.stride(2)    # = 1

    grid = (
        triton.cdiv(N, BLOCK_M),   # blocks over rows (i)
        triton.cdiv(N, BLOCK_N),   # blocks over cols (j)
        batch,                     # one block set per batched matrix
    )

    _batched_matmul_kernel[grid](
        left_3d,
        right_3d,
        out_3d,
        N, N, N,                     # M, N, K
        stride_am, stride_ak, stride_a_batch,
        stride_bk, stride_bj, stride_b_batch,
        stride_cm, stride_cn, stride_c_batch,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6) Reshape back to [B, N, N, hidden_dim] (float32)
    # ------------------------------------------------------------------
    out = out_3d.to(torch.float32).view(B2, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 7) Final LayerNorm over the hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 8) Apply the output gate
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 9) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
