    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,

    pid_n = tl.program_id(1)
    pid_z = tl.program_id(2)

    b = pid_z // H
    h = pid_z % H

    hidden_offset = h * stride_left_h

TriMul (outgoing) – a high‑performance Triton implementation.

The kernel computes, for each batch b and hidden channel h,
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

which is exactly the einsum:
    out = torch.einsum('... i k d, ... j k d -> ... i j d', left, right)

Implementation details
----------------------
1.  Layer‑norm on the input dimension (dim) – done in PyTorch.
2.  Linear projections (left/right) and gated linear projections – PyTorch.
3.  Optional binary mask (broadcasted on the hidden dim) – PyTorch.
4.  The heavy O(N³) pair‑wise multiplication is fused into a
    Triton GEMM where the right operand is accessed transposed,
    i.e. we compute a batched matmul `L @ Rᵀ` for every hidden channel.
5.  Second Layer‑norm, out‑gate, and final linear projection – PyTorch.

The Triton kernel tiles the output matrix with
    BLOCK_M × BLOCK_N tiles and iterates over the K‑dimension in
    BLOCK_K steps.  All accumulations are in fp32, while input/
    output tensors are stored in fp16 to reduce memory traffic.

@triton.jit
def _triton_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Each program instance computes a (BLOCK_M × BLOCK_N) tile of the
    output matrix for a single (batch, hidden) pair.
    """
    pid_m = tl.program_id(0)          # tile row
    pid_n = tl.program_id(1)          # tile column
    pid_z = tl.program_id(2)          # combined batch * hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices
    # ------------------------------------------------------------------
    b = pid_z // H
    h = pid_z % H

    # Base offsets for the selected batch and hidden channel
    batch_offset = b * stride_left_b
    hidden_offset = h * stride_left_h     # stride_*_h == 1 for a contiguous layout

    # ------------------------------------------------------------------
    # Compute tile coordinates
    # ------------------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the reduction (k) dimension
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        offs_k = k + tl.arange(0, BLOCK_K)

        # Load left tile  – shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + batch_offset + hidden_offset \
            + offs_i[:, None] * stride_left_i \
            + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs,
                       mask=(offs_i[:, None] < N) & (offs_k[None, :] < N),
                       other=0.0,
                       dtype=tl.float16)
        left = left.to(tl.float32)

        # Load right tile transposed – shape (BLOCK_K, BLOCK_N)
        # We need right[k, j] which is the transposed view of the original
        # right[b, j, k, h].
        right_ptrs = right_ptr + batch_offset + hidden_offset \
            + offs_k[:, None] * stride_right_k \
            + offs_j[None, :] * stride_right_i
        right = tl.load(right_ptrs,
                        mask=(offs_k[:, None] < N) & (offs_j[None, :] < N),
                        other=0.0,
                        dtype=tl.float16)
        right = right.to(tl.float32)

        # Accumulate the matrix product for this K‑block
        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # ------------------------------------------------------------------
    # Store the result (convert back to fp16)
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + batch_offset + hidden_offset \
        + offs_i[:, None] * stride_out_i \
        + offs_j[None, :] * stride_out_j
    tl.store(out_ptrs, acc.to(tl.float16),
             mask=mask_i[:, None] & mask_j[None, :])


def _triton_outgoing(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Wrapper that launches the Triton kernel.
    `left` and `right` must be contiguous, have the same shape
    [B, N, N, H] and be of dtype torch.float16.
    Returns a fp16 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left, dtype=torch.float16, memory_format=torch.contiguous_format)

    # Strides (Python ints)
    sl = left.stride()
    sr = right.stride()
    so = out.stride()

    # Tile sizes – tuned for H100 (can be adjusted if needed)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # rows
        (N + BLOCK_N - 1) // BLOCK_N,   # cols
        B * H,                          # batch * hidden
    )

    _triton_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=sl[0],
        stride_left_i=sl[1],
        stride_left_k=sl[2],
        stride_left_h=sl[3],
        stride_right_b=sr[0],
        stride_right_i=sr[1],
        stride_right_k=sr[2],
        stride_right_h=sr[3],
        stride_out_b=so[0],
        stride_out_i=so[1],
        stride_out_j=so[2],
        stride_out_h=so[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,
        num_stages=2,
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (bool or 0/1) – may be ignored
        - weights      : dict of model parameters (torch.Tensors)
        - config       : dict with keys:
            * "dim"          – input channel size
            * "hidden_dim"   – projection size
            * "nomask"       – bool, if True the mask is ignored

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # First LayerNorm (over the last dimension)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])          # [B,N,N,hidden]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional binary mask (broadcasted on the hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask: [B, N, N] → [B, N, N, 1] → broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # Gated linear transformations
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel: pairwise multiplication over the second sequence axis
    # ------------------------------------------------------------------
    # Cast to fp16 for bandwidth‑limited kernel
    left_fp16 = left.contiguous().to(torch.float16)
    right_fp16 = right.contiguous().to(torch.float16)

    out_fp16 = _triton_outgoing(left_fp16, right_fp16)    # [B,N,N,hidden] fp16
    out = out_fp16.to(torch.float32)                     # back to fp32 for the rest

    # ------------------------------------------------------------------
    # Second LayerNorm, out‑gate, and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,),
                       weight=weights["to_out_norm.weight"],
                       bias=weights["to_out_norm.bias"],
                       eps=eps)

    out = out * out_gate            # apply the output gate

    # Final linear projection back to `dim`
    out = F.linear(out, weights["to_out.weight"])

    return out
