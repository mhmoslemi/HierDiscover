@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_n1, stride_left_n2, stride_left_h,
    stride_right_b, stride_right_n1, stride_right_n2, stride_right_h,
    stride_out_b, stride_out_n1, stride_out_n2, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
   pid_i = tl.program_id(0)
   pid_j = tl.program_id(1)
   pid_z = tl.program_id(2)  # batch * hidden block idx

   # compute block coordinates
   i0 = pid_i * BLOCK_M
   j0 = pid_j * BLOCK_N
   batch_idx = pid_z // (tl.cdiv(H, BLOCK_H))
   h_block = (pid_z % (tl.cdiv(H, BLOCK_H))) * BLOCK_H

   # offsets
   i = i0 + tl.arange(0, BLOCK_M)
   j = j0 + tl.arange(0, BLOCK_N)
   h = h_block + tl.arange(0, BLOCK_H)

   # mask out-of-bounds
   mask_i = i < N
   mask_j = j < N
   mask_h = h < H

   # initialize accumulator
   acc = tl.zeros((BLOCK_H, BLOCK_M, BLOCK_N), dtype=tl.float32)

   # Iterate over k
   for k_offset in range(0, N, BLOCK_K):
       # k tile valid range
       k = k_offset + tl.arange(0, BLOCK_K)
       mask_k = k < N

       # left: shape (BLOCK_H, BLOCK_M, BLOCK_K)
       left_ptrs = left_ptr + \
           batch_idx * stride_left_b + \
           i[:,None,None] * stride_left_n1 + \
           k[None,None,:] * stride_left_n2 + \
           h[:,None,None] * stride_left_h
       left = tl.load(left_ptrs, mask=mask_i[:,None,None] & mask_k[None,None,:] & mask_h[:,None,None], other=0.0).to(tl.float32)

       # right: right shape (BLOCK_H, BLOCK_N, BLOCK_K)
       right_ptrs = right_ptr + \
           batch_idx * stride_right_b + \
           j[:,None,None] * stride_right_n1 + \
           k[None,None,:] * stride_right_n2 + \
           h[:,None,None] * stride_right_h
       right = tl.load(right_ptrs, mask=mask_j[:,None,None] & mask_k[None,None,:] & mask_h[:,None,None], other=0.0).to(tl.float32)

       # compute outer product per h: left[h,i,k] * right[h,j,k]
       # left_expand: (BLOCK_H, BLOCK_M, BLOCK_K) -> (BLOCK_H, BLOCK_M, 1, BLOCK_K)
       # but we want sum over k -> produce (BLOCK_H, BLOCK_M, BLOCK_N)
       # compute broadcasting:
       left_exp = left[:, :, :, None]  # (H, M, K, 1)
       right_exp = right[:, :, None, :] # (H, N, 1, K)
       # Multiply and sum over K dimension:
       acc += tl.sum(left_exp * right_exp, axis=3)  # sum over K dimension

def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_z = tl.program_id(2)

    # compute block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N

    # compute batch and hidden block
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    batch_idx = pid_z // num_h_blocks
    h_block = (pid_z % num_h_blocks) * BLOCK_H

    # generate index ranges
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    h = h_block + tl.arange(0, BLOCK_H)

    # masks for out-of-bounds
    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulate

    # loop over k
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # pointers for left
        left_ptrs = left_ptr + \
            batch_idx * stride_left_b + \
            i[:, None, None] * stride_left_i + \
            k[None, :, None] * stride_left_k + \

def custom_kernel(data):
    """
    Triton-accelerated implementation of the outgoing TriMul operator from AlphaFold3.
    ...
    """
    import torch
    import triton
    import triton.language as tl

    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # Layernorm on input tensor
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Mask
    if not config.get("nomask", True) and mask is not None:
        mask = mask.to(x.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to half for faster Triton compute
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape
    # allocate output (float32)
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)

    # Compute strides
    s_left_b, s_left_i, s_left_k, s_left_h = left_h.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right_h.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    # Kernel block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16
    
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((hidden_dim + BLOCK_H - 1) // BLOCK_H)
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B, N, hidden_dim,
        s_left_b, s_left_i, s_left_k, s_left_h,
        s_right_b, s_right_i, s_right_k, s_right_h,
        s_out_b, s_out_i, s_out_j, s_out_h,

        bias=weights["to_out_norm.bias"]
    )

    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

"""
TriMul (outgoing) – Triton‐accelerated forward pass.

The implementation follows the reference PyTorch module:
1. Layer‑norm the input tensor (shape [B, N, N, C]).
2. Linear projections → left, right (shape [B, N, N, H]).
3. Optional binary mask (broadcasted on the last dim).
4. Gated activations (sigmoid of linear layers) are applied elementwise.
5. Heavy N³ contraction:
      out[b, i, j, h] = Σₖ left[b, i, k, h] * right[b, j, k, h]
   is performed by a custom Triton kernel that tiles the (i, j) plane and the
   hidden dimension H.  All arithmetic inside the kernel is in float32 while
   the inputs are loaded as float16 for higher throughput on H100.
6. Layer‑norm on the contracted tensor, gating with out_gate,
   and final linear projection back to the original channel dimension.

Only the contraction (step 5) is off‑loaded to Triton, satisfying the
requirement to implement “part of the operations” in a kernel.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # const float16 *
    right_ptr,              # const float16 *
    out_ptr,                # float32 *
    B, N, H,                # int64 scalars
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute   out[b,i,j,h] = Σₖ left[b,i,k,h] * right[b,j,k,h]   .
    The kernel tiles the output (i,j) plane (BLOCK_M×BLOCK_N) and a block of
    hidden channels (BLOCK_H).  For each k‑tile we load a slice of left and
    right, perform a broadcasted multiplication and reduce over k.
    """
    pid_i = tl.program_id(0)            # tile index in i dimension
    pid_j = tl.program_id(1)            # tile index in j dimension
    pid_z = tl.program_id(2)            # combined batch & hidden‑block index

    # ----- block start indices -----
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    num_h_blocks = tl.cdiv(H, BLOCK_H)
    b = pid_z // num_h_blocks                     # batch index
    h_block = (pid_z % num_h_blocks) * BLOCK_H    # hidden‑block start

    # ----- index vectors -----
    i = i0 + tl.arange(0, BLOCK_M)                # [M]
    j = j0 + tl.arange(0, BLOCK_N)                # [N]
    h = h_block + tl.arange(0, BLOCK_H)           # [H]

    mask_i = i < N
    mask_j = j < N
    mask_h = h < H

    # accumulator for the output tile (M×N×H)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # ----- loop over k dimension -----
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k0 = k_tile * BLOCK_K
        k = k0 + tl.arange(0, BLOCK_K)            # [K]
        mask_k = k < N

        # ----- load left[b, i, k, h] -----
        left_ptrs = (
            left_ptr
            + b * stride_l_b
            + i[:, None, None] * stride_l_i
            + k[None, :, None] * stride_l_k
            + h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)    # shape (M, K, H)

        # ----- load right[b, j, k, h] -----
        right_ptrs = (
            right_ptr
            + b * stride_r_b
            + j[:, None, None] * stride_r_i
            + k[None, :, None] * stride_r_k
            + h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)    # shape (N, K, H)

        # ----- contraction over k -----
        # left:  (M, K, H) → (M, K, H, 1)
        # right: (N, K, H) → (1, N, K, H)
        # broadcast → (M, N, K, H) → sum over K → (M, N, H)
        acc += tl.sum(left[..., None] * right[None, ...], axis=2)

    # ----- store result -----
    out_ptrs = (
        out_ptr
        + b * stride_o_b
        + i[:, None, None] * stride_o_i
        + j[None, :, None] * stride_o_j
        + h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


def custom_kernel(data):
    """
    Triton‑based forward pass for the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, C] (float32)
        - mask         : torch.Tensor [B, N, N] (bool/float) or None
        - weights      : dict of torch.Tensor containing model parameters
        - config       : dict with keys ``dim``, ``hidden_dim``, ``nomask``

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    import math

    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, C]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on last dim)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear layers, no bias)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy contraction via Triton kernel
    # ------------------------------------------------------------------
    # Cast to half‑precision for the kernel (still exact for most values)
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()

    B, N, _, H = left_h.shape                 # H == hidden_dim
    out = torch.empty_like(left_h, dtype=torch.float32, device=device)  # [B, N, N, H]

    # strides for the contiguous tensors (in element units, not bytes)
    s_l_b, s_l_i, s_l_k, s_l_h = left_h.stride()
    s_r_b, s_r_i, s_r_k, s_r_h = right_h.stride()
    s_o_b, s_o_i, s_o_j, s_o_h = out.stride()

    # kernel tiling parameters – chosen to fit H100 registers and shared memory
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * ((hidden_dim + BLOCK_H - 1) // BLOCK_H),
    )

    trimul_outgoing_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_l_b,
        s_l_i,
        s_l_k,
        s_l_h,
        s_r_b,
        s_r_i,
        s_r_k,
        s_r_h,
        s_o_b,
        s_o_i,
        s_o_j,
        s_o_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
    )

    # ------------------------------------------------------------------
    # 6. Post‑processing: norm, gating, final linear projection
    # ------------------------------------------------------------------
    out_norm = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out_norm = out_norm * out_gate
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
