def layernorm_kernel(
    src_ptr, dst_ptr,
    weight_ptr, bias_ptr,
    N, C,
    stride_src_row, stride_src_col,  # stride across leading dims to element start
    stride_dst_row, stride_dst_col,
    BLOCK_C: tl.constexpr,
    eps: tl.constexpr
):

    pid = tl.program_id(0)

    dst_row_ptr = dst_ptr + pid * stride_dst_row

    offs = tl.arange(0, BLOCK_C)

    # x: torch.Tensor shape (..., C) contiguous
    # output = (x - mean) / sqrt(var + eps) * weight + bias
    # x may be float32; we can use float16 for speed but final output must be float32
    # We'll compile with BLOCK_C = 128 or next multiple of 32.
    # Determine grid size: total = x.shape[0]; blocks needed = ceil_div(total, 1) (since each program processes one row).
    # We'll launch kernel with program_id across rows.

    N = x.shape[0]
    C = x.shape[1]
    # allocate output same dtype as x
    out = torch.empty_like(x)

    # Choose BLOCK_SIZE for C: maybe 128 or 64, whichever divides C.
    BLOCK_C = 64 if C % 64 == 0 else 32 # we can use compile-time constants: bc = tl.constexpr

    layernorm_kernel[grid](

        weight, bias,
        N, C,
        x.stride(0), x.stride(1),
        out.stride(0), out.stride(1),
        BLOCK_C=128,
        eps=eps,
    )
    return out

    dim = config["dim"]

    x = input_tensor.contiguous()
    mask = mask_tensor
    if mask is not None:
        mask = mask.contiguous()
    else:

        mask = torch.ones(x.shape[:3], dtype=x.dtype, device=x.device)

    # Norm
    x_norm = triton_layernorm(x, weights["norm.weight"], weights["norm.bias"], eps)  

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask
    if not config.get("nomask", False):
        # mask shape [B, N, N], unsqueeze

        right = right * mask_unsq

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()

def layernorm_kernel(
    src_ptr, dst_ptr,
    weight_ptr, bias_ptr,
    N, C,
    stride_src_row, stride_src_col,
    stride_dst_row, stride_dst_col,
    stride_w, stride_b,
    BLOCK_SIZE: tl.constexpr,
    eps: tl.constexpr,
):
    pid = tl.program_id(0)
    mask_pid = pid < N
    # compute base ptrs
    src_row_ptr = src_ptr + pid * stride_src_row
    dst_row_ptr = dst_ptr + pid * stride_dst_row

    # Compute sum and sum_sq
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

def triton_layernorm(x, weight, bias, eps=1e-5):
    # x: torch.Tensor shape (..., C)
    # flatten x to (N, C)
    orig_shape = x.shape
    C = x.shape[-1]
    x_flat = x.reshape(-1, C).contiguous()
    N = x_flat.shape[0]
    out_flat = torch.empty_like(x_flat)
    # compute strides as ints
    # The memory layout is contiguous; stride0 = C, stride1 = 1
    stride0 = x_flat.stride(0)
    stride1 = x_flat.stride(1)

    # weight and bias often 1D contiguous
    w = weight.contiguous()
    b = bias.contiguous()
    weight_ptr = w
    bias_ptr = b
    stride_w = w.stride(0)  # 1
    stride_b = b.stride(0)  # 1

    # compile-time block size
    BLOCK_SIZE = 128  # some multiple, ensure covers up to 384, we loop over blocks.

    grid = (N,)
    layernorm_kernel[grid](
        x_flat, out_flat,
        weight_ptr, bias_ptr,
        N, C,
        stride0, stride1,
        stride0, stride1,
        stride_w, stride_b,
        BLOCK_SIZE=BLOCK_SIZE,
        eps=eps,
        num_warps=4,
    )
    return out_flat.view(orig_shape)

def custom_kernel(data):
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    # if mask is none or config nomask True, ignore mask
    # Ensure input is contiguous
    x = input_tensor.contiguous()
    # LayerNorm (input)
    x_norm = triton_layernorm(x, weights["norm.weight"], weights["norm.bias"], eps)
    # Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])
    # Mask
    if (mask_tensor is not None) and not config.get("nomask", False):
        mask = mask_tensor.contiguous().to(x.dtype)
        mask = mask.unsqueeze(-1)  # shape B,N,N,1
        left = left * mask
        right = right * mask
    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()
    left = left * left_gate
    right = right * right_gate
    # Compute multiplication
    # Permute to (B, hidden, N, N)
    left_bc = left.permute(0, 3, 1, 2).contiguous()
    right_bc = right.permute(0, 3, 1, 2).contiguous()
    out_bc = torch.matmul(left_bc, right_bc.transpose(-2, -1))
    out = out_bc.permute(0, 2, 3, 1).contiguous()
    # to_out_norm
    out_norm = triton_layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)
    out_norm = out_norm * out_gate
    # final projection
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])
    return output

"""
TriMul outgoing implementation.

Algorithm:
1. Apply LayerNorm to the input tensor (last dim = dim) using a Triton kernel.
2. Compute four linear transforms on the normalized tensor:
   - left_proj, right_proj (hidden_dim outputs)
   - left_gate, right_gate, out_gate (hidden_dim outputs), each passed through sigmoid.
3. Multiply left/right by mask (if provided) and by their gates.
4. Compute the pairwise product:
   out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}.
   This is implemented as a batched matrix multiplication per hidden dimension.
5. Apply a second LayerNorm (over hidden_dim) using a Triton kernel, then multiply
   by out_gate.
6. Project back to the original dimension with a final linear layer.
All heavy reductions (layer‑norm) are performed on GPU with Triton; matrix
multiplications use the highly‑optimized cuBLAS kernels via `torch.matmul`.
"""

import torch
import triton
import triton.language as tl

# ------------------------------------------------------------------
# Triton LayerNorm kernel (works for any C, loops over C in BLOCK_SIZE chunks)
# ------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    src_ptr, dst_ptr,               # input / output pointers
    weight_ptr, bias_ptr,           # 1‑D weight & bias (size C)
    N, C,                           # number of rows, channel dim
    stride_src_row, stride_src_col, # strides for input
    stride_dst_row, stride_dst_col, # strides for output
    stride_w, stride_b,             # strides for weight / bias (normally 1)
    BLOCK_SIZE: tl.constexpr,       # compile‑time block size for the C‑dim
    eps: tl.constexpr,              # epsilon for numerical stability
):
    pid = tl.program_id(0)
    # guard against over‑launch
    if pid >= N:
        return

    # pointers to the beginning of the current row
    src_row_ptr = src_ptr + pid * stride_src_row
    dst_row_ptr = dst_ptr + pid * stride_dst_row

    # ------------------------------------------------------------------
    # First pass: compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

    offset = 0
    while offset < C:
        cur_offs = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur_offs < C
        x = tl.load(src_row_ptr + cur_offs * stride_src_col,
                    mask=mask, other=0.0, dtype=tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq_val += tl.sum(x * x, axis=0)
        offset += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq_val / C - mean * mean
    # clamp variance to avoid negative values due to FP error
    var = tl.maximum(var, 0.0)
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: write normalized value and apply affine parameters
    # ------------------------------------------------------------------
    offset = 0
    while offset < C:
        cur_offs = offset + tl.arange(0, BLOCK_SIZE)
        mask = cur_offs < C
        x = tl.load(src_row_ptr + cur_offs * stride_src_col,
                    mask=mask, other=0.0, dtype=tl.float32)
        y = (x - mean) * inv_std

        w = tl.load(weight_ptr + cur_offs * stride_w,
                    mask=mask, other=0.0, dtype=tl.float32)
        b = tl.load(bias_ptr + cur_offs * stride_b,
                    mask=mask, other=0.0, dtype=tl.float32)

        y = y * w + b
        tl.store(dst_row_ptr + cur_offs * stride_dst_col, y,
                 mask=mask)
        offset += BLOCK_SIZE

def triton_layernorm(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Layer‑norm over the last dimension using a Triton kernel.
    Input `x` is expected to be contiguous and have at least 2 dimensions.
    """
    if x.ndim < 2:
        raise ValueError("LayerNorm needs at least two dimensions")
    C = x.shape[-1]
    x_flat = x.view(-1, C).contiguous()
    N = x_flat.shape[0]

    out_flat = torch.empty_like(x_flat)

    # strides in elements (not bytes)
    stride_src_row = x_flat.stride(0)
    stride_src_col = x_flat.stride(1)
    stride_dst_row = out_flat.stride(0)
    stride_dst_col = out_flat.stride(1)

    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    # Tunable block size – 128 works for all supported channel sizes
    BLOCK_SIZE = 128
    grid = (N,)

    layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        N,
        C,
        stride_src_row,
        stride_src_col,
        stride_dst_row,
        stride_dst_col,
        stride_w,
        stride_b,
        BLOCK_SIZE=BLOCK_SIZE,
        eps=eps,
        num_warps=4,
    )
    return out_flat.view_as(x)

# ------------------------------------------------------------------
# Entry point expected by the evaluation harness
# ------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (dtype float32).
    """
    input_tensor, mask_tensor, weights, config = data

    # ------------------------------------------------------------------
    # Configuration & constants
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", False) and mask_tensor is not None

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (over dim)
    # ------------------------------------------------------------------
    # Input is [B, N, N, dim]
    x_norm = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections + gated masks
    # ------------------------------------------------------------------
    # `torch.nn.functional.linear` expects weight of shape (out_feat, in_feat)
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if use_mask:
        # mask shape [B, N, N] -> broadcast to [B, N, N, 1]
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # Gating linear layers
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Pairwise multiplication: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    This is a batched matrix multiplication for each hidden channel.
    # ------------------------------------------------------------------
    # Transform to (B, hidden, N, N) so that we can call torch.matmul
    left_bc = left.permute(0, 3, 1, 2).contiguous()   # (B, hidden, N, N)
    right_bc = right.permute(0, 3, 1, 2).contiguous() # (B, hidden, N, N)

    # MatMul over the third dimension (N) – result (B, hidden, N, N)
    out_bc = torch.matmul(left_bc, right_bc.transpose(-2, -1))
    # Back to original layout (B, N, N, hidden)
    out = out_bc.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Second LayerNorm (over hidden_dim) + out‑gate
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 5) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
```

"""
TriMul (outgoing) forward pass – H100‑optimized implementation.

Algorithm
---------
1. Layer‑norm over the last dimension of the input tensor using a custom Triton
   kernel (no need for a separate reduction step – the kernel computes mean,
   variance, and the affine transform in one pass).
2. Four linear transformations of the normalized tensor are performed with
   `torch.nn.functional.linear` (high‑performance cuBLAS GEMM):
      * left_proj, right_proj – produce the hidden representation.
      * left_gate, right_gate, out_gate – produce gating vectors that are passed
        through a sigmoid.
3. Optional pair‑wise mask (`[B, N, N]`) is applied to *left* and *right* before
   gating.
4. Gated tensors are reshaped to `(B, hidden, N, N)` and the core TriMul
   operation is evaluated as a batched matrix multiplication:
        out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
   which is equivalent to `torch.matmul(left, rightᵀ)`.
5. A second Layer‑norm (over the hidden dimension) is applied with the same
   Triton kernel, then multiplied element‑wise by `out_gate`.
6. A final linear projection maps the hidden dimension back to `dim`.

Only the Layer‑Norm reductions are coded in Triton; all large matrix
multiplications use the highly‑optimized cuBLAS kernels via `torch.matmul`.
The implementation works for any batch size (1 or 2), sequence length up to
1024, and the channel dimensions required by the test suite.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: LayerNorm over the last dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    src_ptr, dst_ptr,                # input / output pointers
    weight_ptr, bias_ptr,            # 1‑D weight & bias (size C)
    N, C,                            # N rows, C channels
    stride_src_row, stride_src_col,  # strides for the input tensor
    stride_dst_row, stride_dst_col,  # strides for the output tensor
    stride_w, stride_b,              # strides for weight / bias (normally 1)
    BLOCK_SIZE: tl.constexpr,        # block size for the channel dimension
    eps: tl.constexpr,               # epsilon for numeric stability
):
    pid = tl.program_id(0)
    if pid >= N:
        return

    # ------------------------------------------------------------------
    # Pointers to the start of the current row
    # ------------------------------------------------------------------
    src_row_ptr = src_ptr + pid * stride_src_row
    dst_row_ptr = dst_ptr + pid * stride_dst_row

    # ------------------------------------------------------------------
    # First pass – compute mean and variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

    off = 0
    while off < C:
        cur_offs = off + tl.arange(0, BLOCK_SIZE)
        mask = cur_offs < C
        x = tl.load(src_row_ptr + cur_offs * stride_src_col,
                    mask=mask, other=0.0, dtype=tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq_val += tl.sum(x * x, axis=0)
        off += BLOCK_SIZE

    mean = sum_val / C
    var = sum_sq_val / C - mean * mean
    var = tl.maximum(var, 0.0)          # guard against tiny negative values
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass – write normalized + affine transformed values
    # ------------------------------------------------------------------
    off = 0
    while off < C:
        cur_offs = off + tl.arange(0, BLOCK_SIZE)
        mask = cur_offs < C
        x = tl.load(src_row_ptr + cur_offs * stride_src_col,
                    mask=mask, other=0.0, dtype=tl.float32)
        y = (x - mean) * inv_std

        w = tl.load(weight_ptr + cur_offs * stride_w,
                    mask=mask, other=0.0, dtype=tl.float32)
        b = tl.load(bias_ptr + cur_offs * stride_b,
                    mask=mask, other=0.0, dtype=tl.float32)

        y = y * w + b
        tl.store(dst_row_ptr + cur_offs * stride_dst_col, y, mask=mask)
        off += BLOCK_SIZE


def triton_layernorm(x: torch.Tensor,
                    weight: torch.Tensor,
                    bias: torch.Tensor,
                    eps: float = 1e-5) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of ``x`` using the Triton kernel.
    Arguments:
        x:    Tensor of shape (..., C)   (must be float32)
        weight, bias: 1‑D tensors of length C
        eps: numerical epsilon
    Returns:
        Tensor of the same shape as ``x`` (float32).
    """
    if x.ndim < 2:
        raise ValueError("LayerNorm requires at least 2 dimensions")
    C = x.shape[-1]

    # flatten all leading dimensions – each row will be processed by a program
    x_flat = x.view(-1, C).contiguous()
    N = x_flat.shape[0]
    out_flat = torch.empty_like(x_flat)

    # stride values (in elements, not bytes)
    stride_src_row = x_flat.stride(0)
    stride_src_col = x_flat.stride(1)
    stride_dst_row = out_flat.stride(0)
    stride_dst_col = out_flat.stride(1)

    weight = weight.contiguous()
    bias = bias.contiguous()
    stride_w = weight.stride(0)
    stride_b = bias.stride(0)

    BLOCK_SIZE = 128                     # chosen to cover all supported C
    grid = (N,)

    layernorm_kernel[grid](
        x_flat,
        out_flat,
        weight,
        bias,
        N,
        C,
        stride_src_row,
        stride_src_col,
        stride_dst_row,
        stride_dst_col,
        stride_w,
        stride_b,
        BLOCK_SIZE=BLOCK_SIZE,
        eps=eps,
        num_warps=4,
    )
    return out_flat.view_as(x)


# ----------------------------------------------------------------------
# Entry point used by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    input_tensor, mask_tensor, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    use_mask = not config.get("nomask", False) and mask_tensor is not None

    # ------------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x_norm = triton_layernorm(
        input_tensor.contiguous(),
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2) Linear projections and gating (cuBLAS GEMM)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if use_mask:
        mask = mask_tensor.to(x_norm.dtype).unsqueeze(-1)   # (B, N, N, 1)
        left = left * mask
        right = right * mask

    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 3) Core TriMul: out_{i,j,d} = Σ_k left_{i,k,d} * right_{j,k,d}
    #    Implemented as a batched matrix multiplication per hidden channel.
    # ------------------------------------------------------------------
    # reshape to (B, hidden, N, N) for torch.matmul
    left_bc = left.permute(0, 3, 1, 2).contiguous()
    right_bc = right.permute(0, 3, 1, 2).contiguous()

    # (B, hidden, N, N) × (B, hidden, N, N)ᵀ → (B, hidden, N, N)
    out_bc = torch.matmul(left_bc, right_bc.transpose(-2, -1))

    # back to the original layout (B, N, N, hidden)
    out = out_bc.permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 4) Second LayerNorm over the hidden dimension, then apply out_gate
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
    )
    out_norm = out_norm * out_gate

    # ------------------------------------------------------------------
    # 5) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

    return output
