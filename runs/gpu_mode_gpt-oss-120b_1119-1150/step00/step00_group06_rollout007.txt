def custom_kernel(data):
    input_tensor, mask, weights, config = data

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"] (maybe config["hidden_dim"] else None)

    # Get weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # LayerNorm x
    eps = 1e-5
    x_norm = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # Linear projections (no bias)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape B,N,N,hidden
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Apply mask if present
    if mask is not None:
        # Convert mask to float dtype
        mask_f = mask.to(left.dtype).unsqueeze(-1)  # shape B,N,N,1
        # Apply mask via Triton kernel for efficiency
        left, right = apply_mask_triton(left, right, mask_f)
    else:
        # No mask: pass through
        pass

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Core multiplication: out = left @ right^T per hidden channel, using Triton GEMM
    # Permute to bring hidden_dim first for batch processing
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # shape (B,H,N,N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose right (swap last two dims) for matmul: shape (B,H,N,N) -> (B,H,N,N) (swap N,N)
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()  # shape (B,H,N,N) same but after permute

    # Reshape to (batch*hidden, N, N)
    B, H, M, K = left_perm.shape
    # Ensure M==K==N
    assert M == K == N

    # Flatten batch*hidden
    left_flat = left_perm.view(B*H, N, N)
    right_flat = right_t.view(B*H, N, N)

    # Output buffer
    out_flat = torch.empty_like(left_flat)  # same dtype, shape (B*H,N,N)

    # Launch Triton kernel for batched GEMM
    # Compute grid shape
    # Blocks per M and N
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    # Determine number of blocks
    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N
    grid = (num_m_blocks * num_n_blocks, B*H)  # axis0: tile, axis1: batch

    # Compute strides
    # For left_flat: shape (B*H, N, N) contiguous (batch, row, col)
    stride_ab = left_flat.stride(0)  # batch stride
    stride_am = left_flat.stride(1)  # row stride
    stride_ak = left_flat.stride(2)  # col stride

    # For right_flat: shape (B*H, N, N) as transposed (K,N)
    # Since we pre-transposed, we want B_mat = right_flat shape (B*H, K=N, N). So row dimension = K, col dimension = N
    stride_bh = right_flat.stride(0)
    stride_bk = right_flat.stride(1)  # K stride
    stride_bn = right_flat.stride(2)  # N stride

    # For out_flat: shape (B*H,N,N)

"""
TriMul “outgoing” implementation (AlphaFold‑3 style).

The kernel follows the PyTorch reference:
1. LayerNorm over the last channel.
2. Linear projections (left/right) and gated linear layers (left/right/out).
3. Apply an optional pair‑wise mask.
4. Core operation:  out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d] .
   This is a batched GEMM where each hidden channel (d) is an independent
   matrix‑multiply (N×N)·(N×N)ᵀ.  The heavy computation is performed by a
   Triton GEMM (`bmm_kernel`) that processes every (batch * hidden) matrix
   in FP32 and stores the result back.
5. LayerNorm on the hidden dimension, gated by `out_gate`, and a final
   linear projection back to the original channel size.
All tensor arguments are assumed to be CUDA tensors.
"""

import torch
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton batched GEMM: C = A @ Bᵀ  (A,B shape: [B*H, N, N])
# ----------------------------------------------------------------------
@triton.jit
def bmm_kernel(
    A_ptr, B_ptr, C_ptr,               # pointers
    M, N, K,                           # matrix sizes (M=N, K=N)
    stride_ab, stride_am, stride_ak,   # strides for A (batch, row, col)
    stride_bh, stride_bk, stride_bn,   # strides for B (batch, row, col) – B is already transposed
    stride_ch, stride_cm, stride_cn,   # strides for C (batch, row, col)
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
    num_warps: tl.constexpr,
):
    """
    Batched matrix multiplication for a batch of (M×K) @ (K×N) matrices.
    Grid dimensions: (num_m_blocks, num_n_blocks, batch * hidden).
    """
    pid_m = tl.program_id(0)            # block index in M dimension
    pid_n = tl.program_id(1)            # block index in N dimension
    batch = tl.program_id(2)            # which (batch * hidden) matrix

    # --------------------------------------------------------------
    # Block offsets
    # --------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # base pointers for this batch
    a_ptr = A_ptr + batch * stride_ab
    b_ptr = B_ptr + batch * stride_bh
    c_ptr = C_ptr + batch * stride_ch

    # accumulator in FP32 (works for both FP16/FP32 inputs)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # Loop over K dimension
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Load A block: shape (BLOCK_M, BLOCK_K)
        a = tl.load(
            a_ptr + (offs_m[:, None] * stride_am) + (offs_k[None, :] * stride_ak),
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )
        # Load B block (already transposed): shape (BLOCK_K, BLOCK_N)
        b = tl.load(
            b_ptr + (offs_k[:, None] * stride_bk) + (offs_n[None, :] * stride_bn),
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )
        # Accumulate the dot-product
        acc += tl.dot(a, b)

    # --------------------------------------------------------------
    # Write back result
    # --------------------------------------------------------------
    c = acc.to(tl.float32)
    tl.store(
        c_ptr + (offs_m[:, None] * stride_cm) + (offs_n[None, :] * stride_cn),
        c,
        mask=mask_m[:, None] & mask_n[None, :],
    )


# ----------------------------------------------------------------------
# Main entry point
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor,
                               Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the TriMul “outgoing” operator.
    Arguments:
        data = (input, mask, weights, config)
            input   – [B, N, N, C]  (C = dim)
            mask    – [B, N, N]  (optional, ignored if config['nomask']=True)
            weights – dict with all linear / norm parameters
            config  – dict with keys: dim, hidden_dim, nomask (bool)
    Returns:
        Tensor of shape [B, N, N, dim] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Load weights (all are 2‑D tensors; biases may be present)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    eps = 1e-5

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the channel dimension
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        x, (dim,), weight=norm_weight, bias=norm_bias, eps=eps
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional pair‑wise mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape -> [B, N, N, 1] to broadcast over hidden dim
        mask_f = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Compute gates (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core multiplication: out = Σₖ left[...,i,k,d] * right[...,j,k,d]
    #    Perform as batched GEMM with Triton.
    # ------------------------------------------------------------------
    # Move hidden dimension to second axis for easy batching:
    #   (B, N, N, H) -> (B, H, N, N)
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    # Transpose the last two axes of right for the “right‑transpose” needed by the einsum
    right_t = right_perm.permute(0, 1, 3, 2).contiguous()

    B, H, N, _ = left_perm.shape  # _ == N
    # Flatten batch*hidden into a single GEMM batch dimension
    batch_hidden = B * H

    A = left_perm.view(batch_hidden, N, N)       # [BH, N, N]
    Bmat = right_t.view(batch_hidden, N, N)      # [BH, N, N]  (already transposed)

    # Output buffer
    C = torch.empty_like(A)

    # --------------------------------------------------------------
    # Triton launch configuration
    # --------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32
    num_warps = 4

    num_m_blocks = (N + BLOCK_M - 1) // BLOCK_M
    num_n_blocks = (N + BLOCK_N - 1) // BLOCK_N

    grid = (num_m_blocks, num_n_blocks, batch_hidden)

    # Strides (in elements) for the flattened tensors
    stride_ab = A.stride(0)
    stride_am = A.stride(1)
    stride_ak = A.stride(2)

    stride_bh = Bmat.stride(0)
    stride_bk = Bmat.stride(1)
    stride_bn = Bmat.stride(2)

    stride_ch = C.stride(0)
    stride_cm = C.stride(1)
    stride_cn = C.stride(2)

    # ------------------------------------------------------------------
    # Invoke Triton GEMM kernel
    # ------------------------------------------------------------------
    bmm_kernel[grid](
        A, Bmat, C,
        N, N, N,
        stride_ab, stride_am, stride_ak,
        stride_bh, stride_bk, stride_bn,
        stride_ch, stride_cm, stride_cn,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=num_warps,
    )

    # ------------------------------------------------------------------
    # Reshape back to the original layout: [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out_hidden = C.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 6️⃣ Hidden‑dim LayerNorm, out‑gate, final linear projection
    # ------------------------------------------------------------------
    out_hidden = torch.nn.functional.layer_norm(
        out_hidden,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )
    out_hidden = out_hidden * out_gate
    out = torch.nn.functional.linear(out_hidden, to_out_weight)   # → [B, N, N, dim]

    return out
```
