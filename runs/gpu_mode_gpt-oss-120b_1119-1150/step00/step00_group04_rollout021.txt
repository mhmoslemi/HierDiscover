```
@triton.jit
def fuse_gates_mask_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    stride_lgate_b, stride_lgate_i, stride_lgate_j, stride_lgate_h,
    stride_rgate_b, stride_rgate_i, stride_rgate_j, stride_rgate_h,
    stride_mask_b, stride_mask_i, stride_mask_j,
    BLOCK_SIZE: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute flat offset
    total = B * N * N * H
    offset = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    # compute mask for OOB
    mask = offset < total
    # convert offset to indices b,i,j,h
    # using integer division
    # Compute b = offset // (N*N*H)
    # remainder offset2 = offset % (N*N*H)
    # i = offset2 // (N*H)
    # remainder offset3 = offset2 % (N*H)
    # j = offset3 // H
    # h = offset3 % H
    # since H can be up to 384 etc, it's fine.
    # Provide in integer
    off = offset
    b = off // (N * N * H)
    off = off - b * (N * N * H)
    i = off // (N * H)
    off = off - i * (N * H)
    j = off // H
    h = off - j * H

    # compute pointers offsets
    left = tl.load(left_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h, mask=mask)
    right = tl.load(right_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h, mask=mask)
    left_gate = tl.load(left_gate_ptr + b * stride_lgate_b + i * stride_lgate_i + j * stride_lgate_j + h * stride_lgate_h, mask=mask)
    right_gate = tl.load(right_gate_ptr + b * stride_rgate_b + i * stride_rgate_i + j * stride_rgate_j + h * stride_rgate_h, mask=mask)
    # load mask values (no hidden dimension)
    mask_val = tl.load(mask_ptr + b * stride_mask_b + i * stride_mask_i + j * stride_mask_j, mask=mask)
    # mask is broadcast along hidden dimension
    left_out = left * mask_val * left_gate
    right_out = right * mask_val * right_gate
    tl.store(left_out_ptr + b * stride_left_b + i * stride_left_i + j * stride_left_j + h * stride_left_h, left_out, mask=mask)
    tl.store(right_out_ptr + b * stride_right_b + i * stride_right_i + j * stride_right_j + h * stride_right_h, right_out, mask=mask)
```

```
@triton.jit
def fuse_masks_and_gates_kernel(
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    block_size: tl.constexpr,
):
    pid = tl.program_id(0)
    offs = pid * block_size + tl.arange(0, block_size)
    mask = offs < B * N * N * H
    # compute mask offset (divide by H)
    mask_offs = offs // H

    # load values
    left = tl.load(left_ptr + offs, mask=mask)
    right = tl.load(right_ptr + offs, mask=mask)
    left_gate = tl.load(left_gate_ptr + offs, mask=mask)
    right_gate = tl.load(right_gate_ptr + offs, mask=mask)
    m = tl.load(mask_ptr + mask_offs, mask=mask)

    left_out = left * m * left_gate
    right_out = right * m * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)
```

"""
TriMul outgoing kernel (forward only) using half precision and a Triton kernel
to fuse mask and gate multiplication. The rest of the computation (linear
projections, batched GEMM, and layer-norm) uses PyTorch primitives which are
already highly optimized on the H100.

Algorithm:
1. LayerNorm over the last dimension of the input tensor.
2. Linear projections for left/right (+ gates) in float16.
3. If a mask is supplied, a Triton kernel multiplies left/right with the
   mask and with their respective sigmoid gates in a single pass.
   Otherwise the same multiplication is done with a simple element‑wise op.
4. For each hidden dimension we compute a matrix multiplication
   `out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]`.
   This is implemented as a single batched GEMM with batch size B*hidden_dim.
5. LayerNorm over the hidden dimension, followed by an output gate.
6. Final linear projection back to the original feature dimension.
All heavy arithmetic is performed in float16; the final result is returned
as float32.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def _fuse_mask_gate_kernel(
    left_in_ptr, right_in_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    B, N, H,
    block_size: tl.constexpr,
):
    """
    Fuse `mask`, `left_gate` and `right_gate` into `left` and `right`.
    All tensors are assumed to be contiguous and stored in row‑major order.
    The layout for the 4‑D tensors is (b, i, j, h) with stride_h == 1,
    stride_j == H, stride_i == N*H, stride_b == N*N*H.
    """
    pid = tl.program_id(0)
    offs = pid * block_size + tl.arange(0, block_size)
    total = B * N * N * H
    mask = offs < total

    # index into the mask tensor (which lacks the hidden dimension)
    mask_offs = offs // H

    # Load values (use 0.0 as the out‑of‑bounds value – will be masked later)
    left = tl.load(left_in_ptr + offs, mask=mask, other=tl.float16(0.0))
    right = tl.load(right_in_ptr + offs, mask=mask, other=tl.float16(0.0))
    left_gate = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate = tl.load(right_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    m = tl.load(mask_ptr + mask_offs, mask=mask, other=tl.float16(0.0))

    left_out = left * m * left_gate
    right_out = right * m * right_gate

    tl.store(left_out_ptr + offs, left_out, mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the “outgoing” TriMul operator.

    Arguments
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N] (float32/float16) or None
        - weights      : dict of torch.Tensors (float32) containing all model parameters
        - config       : dict with keys ``dim``, ``hidden_dim`` and ``nomask`` (bool)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Layer‑norm over the last dimension (float32 -> half)
    # ------------------------------------------------------------------
    # compute moments in float32 for numerical stability
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)

    # apply learned scale & bias
    norm_weight = weights["norm.weight"].to(torch.float16).view(1, 1, 1, -1)
    norm_bias = weights["norm.bias"].to(torch.float16).view(1, 1, 1, -1)
    x_norm = (x_norm.to(torch.float16) * norm_weight) + norm_bias  # half precision

    # ------------------------------------------------------------------
    # 2. Linear projections + gate logits (all in float16)
    # ------------------------------------------------------------------
    # cast weight matrices once
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    # proj
    left_proj = F.linear(x_norm, left_proj_w)   # (B,N,N,hidden_dim)
    right_proj = F.linear(x_norm, right_proj_w)

    # gates (sigmoid)
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_w))

    B, N, _, _ = left_proj.shape  # N is seq_len

    # ------------------------------------------------------------------
    # 3. Fuse mask (if present) and gate multiplication with a Triton kernel
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        # ensure mask is half‑precision and contiguous
        mask_h = mask.to(torch.float16).contiguous()
        left = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK = 1024
        grid = (triton.cdiv(B * N * N * hidden_dim, BLOCK),)

        _fuse_mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_h,
            left,
            right,
            B, N, hidden_dim,
            block_size=BLOCK,
        )
        # after the kernel `left` and `right` already contain the gated values
        del left_proj, right_proj, left_gate, right_gate, mask_h  # free memory
    else:
        # No mask – just multiply with the sigmoid gates
        left = left_proj * left_gate
        right = right_proj * right_gate
        if mask is not None and nomask:
            # mask present but we are instructed to ignore it – treat as all ones
            pass
        # free temporaries we no longer need
        del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 4. Batched matrix multiplication over the hidden dimension
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a single batched GEMM with batch = B*hidden_dim
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N)
    left_flat = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # matrix multiplication (half precision)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout: (B,N,N,hidden_dim)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output layer‑norm over the hidden dimension + output gate
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).view(1, 1, 1, -1)
    to_out_norm_b = weights["to_out_norm.bias"].to(torch.float16).view(1, 1, 1, -1)

    out_norm = out_norm * to_out_norm_w + to_out_norm_b
    out_norm = out_norm * out_gate  # apply output gate

    # ------------------------------------------------------------------
    # 6. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)
    result = F.linear(out_norm, to_out_w)  # (B,N,N,dim) half

    # Convert back to float32 as required by the interface
    return result.to(torch.float32)

"""
TriMul (outgoing) forward pass – Triton‑accelerated core.

Algorithm
---------
1. **Layer‐norm** over the last feature dimension of the input tensor.
2. **Linear projections** for the left/right streams and the three sigmoidal
   gates (all computed in float16 for speed).
3. If a mask is provided, a tiny Triton kernel fuses
   `mask * left_proj * left_gate` and `mask * right_proj * right_gate`
   into the final `left` and `right` tensors (otherwise a simple
   element‑wise multiplication is used).
4. The “tri‑multiplicative” contraction
   `out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]`
   is implemented as a single batched GEMM with batch size `B*hidden_dim`.
5. **Layer‑norm** over the hidden dimension, followed by the output gate.
6. **Linear** projection back to the original feature dimension.
   The result is returned as float32.

Only the mask‑gate fusion is performed with a custom Triton kernel; the
remaining heavy lifting (matrix multiplications, layer‑norm statistics,
sigmoids) uses highly‑optimised PyTorch kernels.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _fuse_mask_gate_kernel(
    left_in_ptr, right_in_ptr,          # [B,N,N,H] left/right projections
    left_gate_ptr, right_gate_ptr,      # [B,N,N,H] sigmoid gates
    mask_ptr,                           # [B,N,N]   binary mask (float16)
    left_out_ptr, right_out_ptr,        # [B,N,N,H] results after fusion
    B, N, H,                            # compile‑time constants
    block_size: tl.constexpr,
):
    """
    Fuse mask and sigmoid gates into the projected tensors.
    All tensors are assumed to be contiguous in row‑major order:
        (b, i, j, h)  with stride_h == 1,
        stride_j == H,
        stride_i == N * H,
        stride_b == N * N * H.
    """
    pid = tl.program_id(0)
    offs = pid * block_size + tl.arange(0, block_size)                # element offsets
    total = B * N * N * H
    mask = offs < total

    # broadcast mask across the hidden dimension:
    mask_offs = offs // H

    # load values (out‑of‑bounds elements are zero‑filled and will be masked)
    left        = tl.load(left_in_ptr   + offs, mask=mask, other=tl.float16(0.0))
    right       = tl.load(right_in_ptr  + offs, mask=mask, other=tl.float16(0.0))
    left_gate   = tl.load(left_gate_ptr + offs, mask=mask, other=tl.float16(0.0))
    right_gate  = tl.load(right_gate_ptr+ offs, mask=mask, other=tl.float16(0.0))
    m           = tl.load(mask_ptr      + mask_offs, mask=mask, other=tl.float16(0.0))

    left_out  = left  * m * left_gate
    right_out = right * m * right_gate

    tl.store(left_out_ptr  + offs, left_out,  mask=mask)
    tl.store(right_out_ptr + offs, right_out, mask=mask)


def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the TriMul “outgoing” module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config) where
        - input_tensor : torch.Tensor[B, N, N, dim] (float32)
        - mask         : torch.Tensor[B, N, N] (float32/float16) or None
        - weights      : dict of torch.Tensor (float32) with the model parameters
        - config       : dict containing at least ``dim``, ``hidden_dim`` and ``nomask``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    nomask      = config.get("nomask", False)
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input LayerNorm (float32 → float16)
    # ------------------------------------------------------------------
    # compute moments in float32 for stability
    mean = input_tensor.mean(dim=-1, keepdim=True)
    var  = input_tensor.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (input_tensor - mean) / torch.sqrt(var + eps)

    # apply learned scale & bias (weights are stored as float32)
    norm_w = weights["norm.weight"].to(torch.float16).view(1, 1, 1, -1)
    norm_b = weights["norm.bias"]  .to(torch.float16).view(1, 1, 1, -1)
    x_norm = (x_norm.to(torch.float16) * norm_w) + norm_b   # half‑precision tensor

    # ------------------------------------------------------------------
    # 2. Linear projections + sigmoid gates (all half‑precision)
    # ------------------------------------------------------------------
    # cast all weight matrices once
    left_proj_w   = weights["left_proj.weight"].to(torch.float16)
    right_proj_w  = weights["right_proj.weight"].to(torch.float16)
    left_gate_w   = weights["left_gate.weight"].to(torch.float16)
    right_gate_w  = weights["right_gate.weight"].to(torch.float16)
    out_gate_w    = weights["out_gate.weight"].to(torch.float16)

    # projections
    left_proj  = F.linear(x_norm, left_proj_w)   # (B, N, N, H)
    right_proj = F.linear(x_norm, right_proj_w)

    # gates (sigmoid)
    left_gate  = torch.sigmoid(F.linear(x_norm, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x_norm, out_gate_w))

    B, N, _, _ = left_proj.shape

    # ------------------------------------------------------------------
    # 3. Fuse mask and gates (Triton)  –  if a mask is present
    # ------------------------------------------------------------------
    if mask is not None and not nomask:
        mask_h = mask.to(torch.float16).contiguous()
        left  = torch.empty_like(left_proj)
        right = torch.empty_like(right_proj)

        BLOCK = 1024                       # tuneable; 1024 works well on H100
        grid = (triton.cdiv(B * N * N * hidden_dim, BLOCK),)

        _fuse_mask_gate_kernel[grid](
            left_proj,
            right_proj,
            left_gate,
            right_gate,
            mask_h,
            left,
            right,
            B, N, hidden_dim,
            block_size=BLOCK,
        )
        # free temporaries that are no longer needed
        del left_proj, right_proj, left_gate, right_gate, mask_h
    else:
        # No mask – simply multiply with the sigmoid gates
        left  = left_proj  * left_gate
        right = right_proj * right_gate
        # free the now‑unneeded intermediates
        del left_proj, right_proj, left_gate, right_gate

    # ------------------------------------------------------------------
    # 4. Tri‑multiplicative contraction via a batched GEMM
    #    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    # ------------------------------------------------------------------
    # reshape to (B*H, N, N) for a single bmm call
    left_flat  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_flat = right.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)

    # matrix multiplication (half‑precision)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # restore original layout: (B, N, N, H)
    out = out_flat.reshape(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 5. Output LayerNorm + output gate (all half‑precision)
    # ------------------------------------------------------------------
    mean_o = out.mean(dim=-1, keepdim=True)
    var_o  = out.var(dim=-1, unbiased=False, keepdim=True)
    out_norm = (out - mean_o) / torch.sqrt(var_o + eps)

    to_out_norm_w = weights["to_out_norm.weight"].to(torch.float16).view(1, 1, 1, -1)
    to_out_norm_b = weights["to_out_norm.bias"] .to(torch.float16).view(1, 1, 1, -1)

    out_norm = out_norm * to_out_norm_w + to_out_norm_b
    out_norm = out_norm * out_gate                     # apply the output gate

    # ------------------------------------------------------------------
    # 6. Final linear projection back to `dim`
    # ------------------------------------------------------------------
    to_out_w = weights["to_out.weight"].to(torch.float16)   # (dim, hidden_dim)
    result = F.linear(out_norm, to_out_w)                   # (B,N,N,dim) half

    # convert back to float32 as required by the caller
    return result.to(torch.float32)
