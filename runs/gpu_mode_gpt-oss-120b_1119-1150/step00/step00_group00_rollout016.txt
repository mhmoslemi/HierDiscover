```
def trimul_matmul_kernel(
    a_ptr,  # left tensor pointer
    b_ptr,  # right tensor pointer (interpreted as B_T)
    c_ptr,  # output tensor pointer
    BATCH, N,
    stride_a_batch, stride_a_i, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_j,
    stride_c_batch, stride_c_i, stride_c_j,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)
    
    # offsets for each block
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    
    # compute the actual pointers base for batch and channel
    a_batch_ptr = a_ptr + pid_batch * stride_a_batch
    b_batch_ptr = b_ptr + pid_batch * stride_b_batch
    c_batch_ptr = c_ptr + pid_batch * stride_c_batch
    
    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # block of K dimension
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # Load a block
        a_ptrs = a_batch_ptr + (offs_i[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs, mask= (offs_i[:, None] < N) & (offs_k[None, :] < N), other=0.0)
        # Load b block (B_T)
        b_ptrs = b_batch_ptr + (offs_k[:, None] * stride_b_k) + (offs_j[None, :] * stride_b_j)
        b = tl.load(b_ptrs, mask= (offs_k[:, None] < N) & (offs_j[None, :] < N), other=0.0)
        # compute dot product across k
        acc += tl.dot(a, b)
        # advance pointers not needed if using offsets each iteration
    # store
    c_ptrs = c_batch_ptr + (offs_i[:, None] * stride_c_i) + (offs_j[None, :] * stride_c_j)
```

```
import triton
import triton.language as tl

@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    N,
    hidden_dim,
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,

):

    pid_j = tl.program_id(1)
    pid_batch = tl.program_id(2)
    # decode batch and channel
    b_idx = pid_batch // hidden_dim
```

The *outgoing* Triangle‑Multiplicative‑Update (TriMul) performs

```
out[b,i,j,d] = Σ_k left[b,i,k,d] · right[b,j,k,d]          (1)
```

where `left` and `right` are linear projections of the normalized input
and are gated element‑wise.  
Equation (1) is a batched matrix multiplication for every hidden channel `d`

```
left_d[b]   :  (N × N)   – rows are i, columns are k
right_d[b]  :  (N × N)   – rows are j, columns are k
out_d[b]    = left_d[b] @ right_d[b].T
```

Thus we need **B × hidden_dim** independent matrix multiplies of size `N×N`.
The dominant cost is the `einsum` above, which we implement with a custom
Triton kernel that:

* works on `float16` (half) for high throughput,
* accumulates in `float32`,
* treats the hidden channel as part of the batch dimension (`pid_z`),
* respects the non‑contiguous layout of the 4‑D tensors
  (`[B,N,N,hidden]` → strides `(N²·H, N·H, H, 1)`).

All other operations (layer‑norms, linear layers, masks, gates) are
performed with regular PyTorch kernels, which are already highly optimized.
Only the expensive `einsum` is off‑loaded to Triton.

```python
"""
custom_kernel – highly‑optimized forward pass of the outgoing TriMul
operator (AlphaFold‑3).  

The kernel performs the N³ contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by treating each hidden channel `d` as an extra batch dimension and
executing a fused batched GEMM in half precision (fp16) with
float32 accumulation.  All surrounding linear layers, layer‑norms,
mask handling and gating are done in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched GEMM for the contraction (1)
# ----------------------------------------------------------------------
@triton.jit
def _trimul_matmul_kernel(
    a_ptr,                      # left  (fp16)
    b_ptr,                      # right (fp16) – used as transposed view
    c_ptr,                      # out   (fp16)

    N,                          # sequence length (int)
    H,                          # hidden dimension (int)

    # strides of the 4‑D tensors (in element units)
    stride_a_batch, stride_a_i, stride_a_k, stride_a_d,
    stride_b_batch, stride_b_k, stride_b_j, stride_b_d,
    stride_c_batch, stride_c_i, stride_c_j, stride_c_d,

    BLOCK_M: tl.constexpr,      # tile size along rows   (i)
    BLOCK_N: tl.constexpr,      # tile size along cols   (j)
    BLOCK_K: tl.constexpr,      # tile size along reduction (k)
):
    pid_i = tl.program_id(0)     # block‑row index
    pid_j = tl.program_id(1)     # block‑col index
    pid_z = tl.program_id(2)     # combined (batch, hidden) index

    # --------------------------------------------------------------
    # decode batch and hidden channel from pid_z
    # --------------------------------------------------------------
    batch_id = pid_z // H
    hid_id   = pid_z %  H

    # base pointers for this (batch, hidden) slice
    a_base = a_ptr + batch_id * stride_a_batch + hid_id * stride_a_d
    b_base = b_ptr + batch_id * stride_b_batch + hid_id * stride_b_d
    c_base = c_ptr + batch_id * stride_c_batch + hid_id * stride_c_d

    # --------------------------------------------------------------
    # tile coordinates
    # --------------------------------------------------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)   # i‑axis
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)   # j‑axis
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # --------------------------------------------------------------
    # loop over the reduction dimension k
    # --------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)

        # ----- load a block of left (M × K) -----
        a_ptrs = a_base + (offs_i[:, None] * stride_a_i) + (offs_k[None, :] * stride_a_k)
        a = tl.load(a_ptrs,
                    mask=(mask_i[:, None] & (offs_k[None, :] < N)),
                    other=0.0,
                    dtype=tl.float16)

        # ----- load a block of rightᵀ (K × N) -----
        # right is accessed as transposed: row = k, col = j
        b_ptrs = b_base + (offs_k[:, None] * stride_b_k) + (offs_j[None, :] * stride_b_j)
        b = tl.load(b_ptrs,
                    mask=((offs_k[:, None] < N) & mask_j[None, :]),
                    other=0.0,
                    dtype=tl.float16)

        # ----- matrix multiply (M×K)·(K×N) → (M×N) -----
        acc += tl.dot(a, b, out_dtype=tl.float32)

    # --------------------------------------------------------------
    # write back result (fp16)
    # --------------------------------------------------------------
    c_ptrs = c_base + (offs_i[:, None] * stride_c_i) + (offs_j[None, :] * stride_c_j)
    tl.store(c_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

        * input_tensor : torch.Tensor, shape [B, N, N, dim]
        * mask_tensor  : torch.Tensor, shape [B, N, N] (may be ignored)
        * weights_dict : dict of torch.Tensors containing all model weights
        * config_dict  : dict with keys
            - "dim" (int)
            - "hidden_dim" (int)
            - "nomask" (bool) – if True, skip mask multiplication

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on the same device)
    """
    # ------------------------------------------------------------------
    # unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # convenience shortcuts for the weight tensors
    # ------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]

    left_proj_w  = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]

    left_gate_w  = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w   = weights["out_gate.weight"]

    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w      = weights["to_out.weight"]

    eps = 1e-5  # LayerNorm epsilon

    # ------------------------------------------------------------------
    # 1) Input LayerNorm
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_w,
                     bias=norm_b,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    # shape -> [B, N, N, hidden_dim]
    left  = F.linear(x, left_proj_w)    # (hidden_dim, dim)
    right = F.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3) Optional mask (broadcast on hidden_dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape [B,N,N] → [B,N,N,1]
        mask = mask_tensor.unsqueeze(-1).type_as(left)
        left  = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4) Gating
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x, left_gate_w))
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_w))

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) The costly contraction – Triton GEMM
    # ------------------------------------------------------------------
    # cast to half for the kernel
    left_h  = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h   = torch.empty_like(left_h)          # [B,N,N,hidden_dim]

    # stride information (in element units, not bytes)
    s_a = left_h.stride()   # (B,N,N,hidden) -> (N²·H, N·H, H, 1)
    s_b = right_h.stride()
    s_c = out_h.stride()

    # tile sizes – tuned for H100 (fit into shared memory & registers)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )

    _trimul_matmul_kernel[grid](
        left_h,
        right_h,
        out_h,
        N,
        hidden_dim,
        s_a[0], s_a[1], s_a[2], s_a[3],   # A strides
        s_b[0], s_b[2], s_b[1], s_b[3],   # B strides (note: k‑row, j‑col)
        s_c[0], s_c[1], s_c[2], s_c[3],   # C strides
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # convert back to fp32 for the remaining layers
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 6) Output LayerNorm (over hidden_dim) + out‑gate
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_w,
                       bias=to_out_norm_b,
                       eps=eps)

    out = out * out_gate

    # ------------------------------------------------------------------
    # 7) Final linear projection back to `dim`
    # ------------------------------------------------------------------
    out = F.linear(out, to_out_w)   # (dim, hidden_dim)

    return out
```
