@triton.jit
def trimul_matmul_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_a_batch,
    stride_bk, stride_bn, stride_b_batch,
    stride_cm, stride_cn, stride_c_batch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # batch idx along (batch*H)
    # compute block row and col:
    block_i = tl.program_id(axis=1)  # i tile
    block_j = tl.program_id(axis=2)  # j tile
    # offsets:
    offs_i = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = block_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for kk in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        # load a tile
        a_ptrs = a_ptr + pid * stride_a_batch + offs_i[:, None] * stride_am + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid * stride_b_batch + offs_k[:, None] * stride_bk + offs_j[None, :] * stride_bn
        a = tl.load(a_ptrs, mask=(offs_i[:, None] < M) & (offs_k[None, :] < K), other=0.0).to(tl.float32)
        b = tl.load(b_ptrs, mask=(offs_k[:, None] < K) & (offs_j[None, :] < N), other=0.0).to(tl.float32)
        acc += tl.dot(a, b)
    # store
    c_ptrs = c_ptr + pid * stride_c_batch + offs_i[:, None] * stride_cm + offs_j[None, :] * stride_cn
    tl.store(c_ptrs, acc.to(tl.float16), mask=(offs_i[:, None] < M) & (offs_j[None, :] < N))

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim, hidden_dim = config["dim"], config["hidden_dim"]
    # Get device (should be same for all)
    device = input_tensor.device

    # LayerNorm parameters
    eps = 1e-5

    # Get weights as torch tensors (weights dict provided)
    # Flatten them? They are already correct shape
    # weights["norm.weight"] shape [dim]; bias shape [dim]
    # left_proj_weight shape [hidden_dim, dim]
    # right_proj_weight shape [hidden_dim, dim]
    # left_gate_weight shape [hidden_dim, dim]
    # right_gate_weight shape [hidden_dim, dim]
    # out_gate_weight shape [hidden_dim, dim]
    # to_out_norm_weight shape [hidden_dim]; bias shape [hidden_dim]
    # to_out_weight shape [dim, hidden_dim]

    # Step 1: Normalize input
    # compute mean across dim axis
    x = input_tensor
    # using torch.nn.functional.layer_norm
    x_norm = torch.nn.functional.layer_norm(
        x, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps
    )

    # Step 2: Projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Step 3: Apply mask if present
    if mask is not None:
        mask_ = mask.unsqueeze(-1).to(x_norm.dtype)
        left = left * mask_
        right = right * mask_
    # else mask unchanged.

    # Step 4: Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Cast to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Permute to bring hidden_dim to batch dimension and transposed right
    left_perm = left.permute(0, 3, 1, 2).contiguous()  # [B, H, N, K]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, K, N]

    B, H, N, K = left_perm.shape
    # reshape to (B*H, N, K) and (B*H, K, N)
    a = left_perm.view(B * H, N, K)
    b = right_perm.view(B * H, K, N)

    # call Triton GEMM kernel
    out_mat = triton_gemm(a, b)  # returns shape (B*H, N, N), dtype float16

    # reshape back
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # Step 5: to_out_norm
    # Use layer_norm on hidden_dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # Step 6: multiply out_gate
    out = out * out_gate  # broadcasting over batch/seq dims

    # Step 7: final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out

@triton.jit
def trimul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    M, N, K,
    stride_am, stride_ak, stride_abatch,
    stride_bk, stride_bn, stride_bbatch,
    stride_cm, stride_cn, stride_cbatch,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid = tl.program_id(axis=0)  # batch id
    pid_m = tl.program_id(axis=1)  # row tile
    pid_n = tl.program_id(axis=2)  # col tile

    # Compute offsets for rows/cols
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    # Mask for boundary
    mask_m = offs_m < M
    mask_n = offs_n < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

TriMul custom kernel (outgoing version)

Implements the forward pass of the TriMul module from AlphaFold3.
The heavy pairwise multiplication (einsum “… i k d, … j k d → … i j d”)
is executed with a fused Triton GEMM kernel that works on the hidden

    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

):
    """batched GEMM: C = A @ B   where
       A : [batch, M, K] (fp16)
       B : [batch, K, N] (fp16)
       C : [batch, M, N] (fp16)
    The batch dimension corresponds to (batch * hidden_dim) in TriMul.
    """
    pid = tl.program_id(axis=0)      # batch index
    pid_m = tl.program_id(axis=1)    # block row
    pid_n = tl.program_id(axis=2)    # block col

    # ------------------------------------------------------------------
    # Block offsets
    # ------------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # ------------------------------------------------------------------
    # Accumulator in FP32
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over K dimension
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = offs_k < K

        # Pointers to the current tiles of A and B
        a_ptrs = a_ptr + pid * stride_abatch \
                 + offs_m[:, None] * stride_am \
                 + offs_k[None, :] * stride_ak
        b_ptrs = b_ptr + pid * stride_bbatch \
                 + offs_k[:, None] * stride_bk \
                 + offs_n[None, :] * stride_bn

        # Load tiles (FP16) with boundary masking
        a = tl.load(a_ptrs,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0).to(tl.float32)
        b = tl.load(b_ptrs,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0).to(tl.float32)

        # Matrix‑multiply‑accumulate
        acc += tl.dot(a, b)

    # ------------------------------------------------------------------
    # Write back result (FP16)
    # ------------------------------------------------------------------
    c_ptrs = c_ptr + pid * stride_cbatch \
             + offs_m[:, None] * stride_cm \
             + offs_n[None, :] * stride_cn
    tl.store(c_ptrs,
             acc.to(tl.float16),
             mask=mask_m[:, None] & mask_n[None, :])

# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def _trimul_gemm(a: torch.Tensor, b: torch.Tensor,
                 block_m: int = 64, block_n: int = 64, block_k: int = 32) -> torch.Tensor:
    """
    Batched GEMM using Triton.
    a: [B, M, K] (fp16)
    b: [B, K, N] (fp16)
    Returns c: [B, M, N] (fp16)
    """
    assert a.is_contiguous() and b.is_contiguous()
    B, M, K = a.shape
    _, K2, N = b.shape
    assert K == K2

    c = torch.empty((B, M, N), device=a.device, dtype=torch.float16)

    # Strides (in elements, not bytes)
    stride_am, stride_ak, stride_ab = a.stride(1), a.stride(2), a.stride(0)
    stride_bk, stride_bn, stride_bb = b.stride(1), b.stride(2), b.stride(0)
    stride_cm, stride_cn, stride_cb = c.stride(1), c.stride(2), c.stride(0)

    grid = (B,
            (M + block_m - 1) // block_m,
            (N + block_n - 1) // block_n)

    _trimul_gemm_kernel[grid](
        a_ptr=a,
        b_ptr=b,
        c_ptr=c,
        M=M, N=N, K=K,
        stride_am=stride_am,
        stride_ak=stride_ak,
        stride_abatch=stride_ab,
        stride_bk=stride_bk,
        stride_bn=stride_bn,
        stride_bbatch=stride_bb,
        stride_cm=stride_cm,
        stride_cn=stride_cn,
        stride_cbatch=stride_cb,
        BLOCK_M=block_m,
        BLOCK_N=block_n,
        BLOCK_K=block_k,
    )
    return c

# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator using a fused Triton GEMM.
    Args:
        data: Tuple (input_tensor, mask, weights, config)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    eps = 1e-5

    # ------------------------------------------------------------------
    # Extract config
    # ------------------------------------------------------------------
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # ------------------------------------------------------------------
    # LayerNorm on the input (dim)
    # ------------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Linear projections (dim -> hidden_dim)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, H]
    right = F.linear(x_norm, weights["right_proj.weight"])  # [B, N, N, H]

    # ------------------------------------------------------------------
    # Optional mask (broadcast on the hidden dimension)
    # ------------------------------------------------------------------
    if mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1]
        mask_ = mask.to(left.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # Gating (sigmoid of Linear(dim -> hidden_dim))
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Convert to fp16 for the GEMM kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # Rearrange tensors so that the hidden dimension becomes a batch dim
    # and right is transposed on the two sequence axes.
    #   left  : [B, N, N, H] -> [B, H, N, K] where K = N
    #   right : [B, N, N, H] -> [B, H, K, N]
    # ------------------------------------------------------------------
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous() # [B, H, N, N]

    B, H, N, K = left_perm.shape
    # Flatten batch and hidden dimension: treat each (b, h) as a separate GEMM
    a = left_perm.view(B * H, N, K)          # [BH, M, K]   (M == N)
    b = right_perm.view(B * H, K, N)         # [BH, K, N]

    # ------------------------------------------------------------------
    # Core Triton GEMM – computes Σ_k left[...,k,d] * right[...,k,d]
    # ------------------------------------------------------------------
    c = _trimul_gemm(a, b)   # [BH, N, N]  fp16

    # Reshape back to [B, N, N, H]
    out = c.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B, N, N, H]

    # ------------------------------------------------------------------
    # LayerNorm on the hidden dimension
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # Apply the output gate (broadcast)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # Final linear projection (hidden_dim -> dim)
    # ------------------------------------------------------------------
    out = F.linear(out, weights["to_out.weight"])   # [B, N, N, dim]

    return out
