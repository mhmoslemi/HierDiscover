@triton.jit
def final_linear_kernel(
    gated_ptr, weight_ptr, out_ptr,
    B, N, H, D,
    stride_gated_b, stride_gated_i, stride_gated_j, stride_gated_h,
    stride_weight_D, stride_weight_H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_D: tl.constexpr, BLOCK_H: tl.constexpr, EPS: tl.constexpr
):
    pid0 = tl.program_id(0)  # index for each (b,i,j)
    pid1 = tl.program_id(1)  # block index for D dimension

    # compute b,i,j from pid0
    total_ij = N * N
    b = pid0 // total_ij
    ij = pid0 % total_ij
    i = ij // N
    j = ij % N

    # offsets for gated vector
    gated_base = gated_ptr + b * stride_gated_b + i * stride_gated_i + j * stride_gated_j
    # gated_base points to gated[ b,i,j,0 ] in memory; actual pointer for hidden dim.

    # compute d_offset
    d_off = pid1 * BLOCK_D + tl.arange(0, BLOCK_D)
    # mask d_off < D
    mask_d = d_off < D

    # Initialize accumulator:
    acc = tl.zeros((BLOCK_D,), dtype=tl.float32)

    # loop over hidden dimension in blocks
    for k in range(0, tl.cdiv(H, BLOCK_H)):
        h_off = k * BLOCK_H + tl.arange(0, BLOCK_H)  # shape BLOCK_H
        mask_h = h_off < H

        weight = tl.load(weight_ptr + d_off[:, None] * stride_weight_D + h_off[None, :] * stride_weight_H,
                         mask=mask_d[:, None] & mask_h[None, :], other=0.0)

    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))

    right = right * right_gate

    B, N, _, H = left.shape

    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 1, 2).contiguous()
    left_mat = left_perm.view(B * H, N, N)
    right_mat = right_perm.view(B * H, N, N)

    # Perform batch matmul: left * right^T
    # use torch.bmm
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()  # [B,N,N,H]

    final_out = torch.empty((B, N, N, dim), dtype=gated.dtype, device=gated.device)

    # final projection using Triton
    _final_linear_triton(gated, to_out_weight, final_out)

"""
TriMul “outgoing” implementation (forward only).

Algorithm
---------
1. Layer‑norm the input tensor `[B, N, N, C]`.
2. Compute four linear projections (left, right, left‑gate, right‑gate, out‑gate)
   using the provided weights.  The gate tensors are passed through a sigmoid.
3. Apply the optional mask (broadcast over the hidden dimension) and multiply
   the linear projections by their respective gates.
4. Contract over the sequence dimension `k`:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is a batched matrix multiplication (`left @ rightᵀ`) for every hidden
   channel `h`.  The implementation uses `torch.bmm` on a reshaped view.
5. Layer‑norm the result over the hidden dimension, then scale by `out_gate`.
6. Final linear projection back to the original feature dimension `C`.  The
   projection (`out_norm * out_gate`  ×  `to_out.weightᵀ`) is performed by a
   custom Triton kernel that fuses the reduction over the hidden dimension and
   the matrix‑vector multiply for each output feature.

Only the final projection is written in Triton; all other heavy‑weight
operations (layer‑norms, linear layers, and the `k`‑contraction) are carried out
by highly‑optimized PyTorch kernels.  This satisfies the requirement that at
least part of the work runs in a custom Triton kernel while keeping the overall
implementation correct and reasonably fast on an H100.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# --------------------------------------------------------------------------- #
# Triton kernel: final linear projection (fused hidden‑dim reduction)
# --------------------------------------------------------------------------- #
@triton.jit
def final_linear_kernel(
    gated_ptr,               # *float32  [B, N, N, H]   gated = out_norm * out_gate
    weight_ptr,              # *float32  [D, H]        final linear weight
    out_ptr,                 # *float32  [B, N, N, D]   output
    B, N, H, D,              # compile‑time scalar dimensions
    stride_gated_b, stride_gated_i, stride_gated_j, stride_gated_h,
    stride_weight_D, stride_weight_H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_D: tl.constexpr,   # tile size in output feature dimension
    BLOCK_H: tl.constexpr,   # tile size in hidden dimension (loop granularity)
):
    # ------------------------------------------------------------------- #
    # 1) Map program ids   pid0 -> (b,i,j)    pid1 -> block of D
    # ------------------------------------------------------------------- #
    pid0 = tl.program_id(0)          # flattened index for (b,i,j)
    pid1 = tl.program_id(1)          # block index for output feature dimension

    total_ij = N * N
    b = pid0 // total_ij
    ij = pid0 % total_ij
    i = ij // N
    j = ij % N

    # ------------------------------------------------------------------- #
    # 2) Compute output‑feature offsets for this block
    # ------------------------------------------------------------------- #
    d_start = pid1 * BLOCK_D
    d_offset = d_start + tl.arange(0, BLOCK_D)
    mask_d = d_offset < D

    # pointer to the start of the hidden‑dim vector for this (b,i,j)
    gated_base = gated_ptr + b * stride_gated_b + i * stride_gated_i + j * stride_gated_j

    # accumulator for the D‑values produced by this block
    acc = tl.zeros([BLOCK_D], dtype=tl.float32)

    # ------------------------------------------------------------------- #
    # 3) Loop over the hidden dimension in chunks of BLOCK_H
    # ------------------------------------------------------------------- #
    num_h_blocks = tl.cdiv(H, BLOCK_H)
    for hb in range(0, num_h_blocks):
        h_start = hb * BLOCK_H
        h_offset = h_start + tl.arange(0, BLOCK_H)
        mask_h = h_offset < H

        # Load the hidden‑dim slice (shape [BLOCK_H])
        gated = tl.load(gated_base + h_offset * stride_gated_h,
                        mask=mask_h, other=0.0)                # (BLOCK_H,)

        # Load the corresponding slice of the weight matrix (shape [BLOCK_D, BLOCK_H])
        w = tl.load(weight_ptr
                    + d_offset[:, None] * stride_weight_D   # D‑axis
                    + h_offset[None, :] * stride_weight_H,  # H‑axis
                    mask=mask_d[:, None] & mask_h[None, :],
                    other=0.0)                               # (BLOCK_D, BLOCK_H)

        # Accumulate: Σ_h gated[h] * w[d, h]
        # broadcasting works automatically: w * gated[None, :] -> (BLOCK_D, BLOCK_H)
        acc += tl.sum(w * gated, axis=1)

    # ------------------------------------------------------------------- #
    # 4) Store the result to the output tensor
    # ------------------------------------------------------------------- #
    out_ptrs = out_ptr + b * stride_out_b + i * stride_out_i + j * stride_out_j + d_offset * stride_out_d
    tl.store(out_ptrs, acc, mask=mask_d)


def _triton_final_linear(gated: torch.Tensor, weight: torch.Tensor, out: torch.Tensor):
    """
    Launch the Triton kernel that computes:
        out[b,i,j,d] = Σ_h gated[b,i,j,h] * weight[d, h]

    Arguments
    ---------
    gated : torch.Tensor
        Tensor of shape [B, N, N, H] (float32, contiguous).
    weight : torch.Tensor
        Tensor of shape [D, H] (float32, contiguous).
    out : torch.Tensor
        Pre‑allocated output tensor of shape [B, N, N, D].
    """
    B, N, N2, H = gated.shape
    D = weight.shape[0]
    assert N == N2, "Input must be square in the sequence dimensions."

    # stride information (in elements, not bytes)
    stride_gated_b, stride_gated_i, stride_gated_j, stride_gated_h = gated.stride()
    stride_w_d, stride_w_h = weight.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tile sizes – chosen to give good occupancy on H100
    BLOCK_D = 32
    BLOCK_H = 32

    grid = (B * N * N, (D + BLOCK_D - 1) // BLOCK_D)
    final_linear_kernel[grid](
        gated_ptr=gated,
        weight_ptr=weight,
        out_ptr=out,
        B=B, N=N, H=H, D=D,
        stride_gated_b=stride_gated_b,
        stride_gated_i=stride_gated_i,
        stride_gated_j=stride_gated_j,
        stride_gated_h=stride_gated_h,
        stride_weight_D=stride_w_d,
        stride_weight_H=stride_w_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_D=BLOCK_D,
        BLOCK_H=BLOCK_H,
        num_warps=8,
        num_stages=3,

        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, C]
        - mask         : torch.Tensor of shape [B, N, N] (float or bool)
        - weights      : dict of torch.Tensors with the model parameters
        - config       : dict containing at least ``dim`` and ``hidden_dim``

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, C] (float32, on the same device as the input)
    """
    # ------------------------------------------------------------------- #
    # Unpack inputs
    # ------------------------------------------------------------------- #
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, C = input_tensor.shape
    H = config["hidden_dim"]
    D = config["dim"]                # final output dimension (may equal C)

    # ------------------------------------------------------------------- #
    # Extract weights (already on the correct device)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight   = weights["left_proj.weight"]
    right_proj_weight  = weights["right_proj.weight"]
    left_gate_weight   = weights["left_gate.weight"]
    right_gate_weight  = weights["right_gate.weight"]
    out_gate_weight    = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # [D, H]

    # ------------------------------------------------------------------- #
    # 1) Input LayerNorm
    # ------------------------------------------------------------------- #
    x = F.layer_norm(input_tensor, (C,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------- #
    # 2) Linear projections (no bias) + optional mask
    # ------------------------------------------------------------------- #
    # shape: [B, N, N, H]
    left  = F.linear(x, left_proj_weight)   # left_proj
    right = F.linear(x, right_proj_weight)  # right_proj

    # mask handling – if ``mask`` is None or an empty tensor we treat it as all‑ones
    if mask is None or mask.numel() == 0:
        mask_exp = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_exp = mask.to(dtype)
    mask_exp = mask_exp.unsqueeze(-1)          # [B, N, N, 1]
    left  = left  * mask_exp
    right = right * mask_exp

    # ------------------------------------------------------------------- #
    # 3) Gates (sigmoid of linear projections)
    # ------------------------------------------------------------------- #
    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # Apply gates
    left  = left  * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # 4) Contraction over k (batched matrix multiply)
    # ------------------------------------------------------------------- #
    # reshape to (B*H, N, N) for batched GEMM
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    left_mat  = left_perm.view(B * H, N, N)              # (B*H, N, N)
    right_mat = right_perm.view(B * H, N, N)             # (B*H, N, N)

    # out_mat[b*h] = left_mat[b*h] @ right_mat[b*h].T
    out_mat = torch.bmm(left_mat, right_mat.transpose(1, 2))   # (B*H, N, N)

    # reshape back to [B, N, N, H]
    out = out_mat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------- #
    # 5) Hidden‑dim LayerNorm and scaling by out_gate
    # ------------------------------------------------------------------- #
    out = F.layer_norm(out, (H,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    gated = out * out_gate                       # [B, N, N, H]

    # ------------------------------------------------------------------- #
    # 6) Final projection to output dimension D (Triton kernel)
    # ------------------------------------------------------------------- #
    final = torch.empty((B, N, N, D), dtype=dtype, device=device)

    # launch the fused reduction + linear projection kernel
    _triton_final_linear(gated, to_out_weight, final)

    return final
