def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_ri, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
    
    pid = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # Compute batch and hidden dimension indices
    total = B * H
    b = pid // H
    h = pid % H
    # Actually pid is pid of first dimension representing total batch dimension; we have separate dims for i and j.
    # We'll treat program_id(0) as batch index.
    # get block indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    
    # Compute offsets for output
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    
    # Prepare mask for out-of-bounds i/j
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # Compute output pointer offsets (will be broadcast after k reduction)
    out_ptrs = out_ptr + b * stride_lb + h * stride_lh \
                + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    # Loop over k dimension
    for k in range(0, N, BLOCK_K):
        # compute k range
        k_base = k
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # load left tile (M x K)
        left_offsets = b * stride_lb + h * stride_lh \
                       + offs_i[:, None] * stride_li \
                       + offs_k[None, :] * stride_lk
        left = tl.load(left_ptr + left_offsets, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # left shape (BLOCK_M, BLOCK_K)
        
        # load right tile transposed (K x N)
        # We need right[j,k] but transposed => right_offsets for shape (BLOCK_K, BLOCK_N) with k index outer.
        right_offsets = b * stride_rb + h * stride_rh \
                        + offs_k[:, None] * stride_rk \
                        + offs_j[None, :] * stride_ri
        right = tl.load(right_ptr + right_offsets, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # right shape (BLOCK_K, BLOCK_N)
        
        # cast to fp32
        left_f = left.to(tl.float32)
        right_f = right.to(tl.float32)
        # accumulate
        acc += tl.dot(left_f, right_f)
    
    # Apply mask to store out of bound indexes
    # Where out-of-bounds i or j -> store 0 maybe (already zero)
    # Use mask for store
    mask_store = mask_i[:, None] & mask_j[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_store)

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B_total, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid = tl.program_id(0)  # batch id
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    
    # compute i and j base indices
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    
    # index vectors
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    
    # create mask for out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    
    # offset base for this batch
    batch_off = pid * stride_lb
    
    # out pointer base for i-j block
    out_ptrs = out_ptr + batch_off + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
    
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    
    for k in range(0, N, BLOCK_K):
        k_base = k
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N
        
        # left tile (M x K)
        left_ptrs = left_ptr + batch_off + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None,:], other=0.0).to(tl.float32)
        
        # right tile transposed (K x N) => load as (K, N)
        right_ptrs = right_ptr + batch_off + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        
        acc += tl.dot(left, right)
    
    # Write back result as fp16
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    bs, seq_len, _, _ = input_tensor.shape    # ignore _ for dim?

    # Get weights
    # e.g., weights["norm.weight"] is shape (dim,)
    # Use F.layer_norm
    import torch.nn.functional as F
    eps = 1e-5

    # LayerNorm x
    x_norm = F.layer_norm(input_tensor, (dim,), weight=weights["norm.weight"], bias=weights["norm.bias"], eps=eps)

    # Projections
    left = F.linear(x_norm, weights["left_proj.weight"])   # shape (bs, N, N, hidden_dim)
    right = F.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if not nomask
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # Gates:
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Convert to fp16 for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # reshape for kernel: bring hidden_dim to batch
    # left shape: (bs, N, N, hidden_dim)
    left_perm = left.permute(0, 3, 1, 2).contiguous()   # (bs, hidden_dim, N, N)
    right_perm = right.permute(0, 3, 1, 2).contiguous()

    # reshape to (bs * hidden_dim, N, N)
    left_reshaped = left_perm.view(-1, seq_len, seq_len)
    right_reshaped = right_perm.view(-1, seq_len, seq_len)

    # Prepare output tensor
    out_reshaped = torch.empty_like(left_reshaped, dtype=torch.float16)  # same shape (batch_total, N, N)

    # Launch Triton kernel
    import triton
    import triton.language as tl

    # Block sizes
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32  # can be same as 32

    batch_total = left_reshaped.shape[0]
    N = seq_len

    # Compute grid dimensions
    grid = (batch_total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Get strides in elements
    stride_lb, stride_li, stride_lk = left_reshaped.stride()
    stride_rb, stride_ri, stride_rk = right_reshaped.stride()
    stride_ob, stride_oi, stride_oj = out_reshaped.stride()

    # Call kernel
    trimul_einsum_kernel[grid](
        left_reshaped,
        right_reshaped,
        out_reshaped,
        batch_total, N,
        stride_lb, stride_li, stride_lk,
        stride_rb, stride_ri, stride_rk,
        stride_ob, stride_oi, stride_oj,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K
    )

    # Reshape back to original shape (bs, N, N, hidden_dim)
    out = out_reshaped.view(bs, hidden_dim, seq_len, seq_len).permute(0, 2, 3, 1).contiguous()
    # Convert to float32
    out = out.to(torch.float32)

    # Apply to_out_norm
    out = F.layer_norm(out, (hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"], eps=eps)

    # Apply out_gate
    out = out * out_gate

    # Final linear
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Optimized TriMul forward pass using Triton.

We compute:
    out_{b,i,j} = Σ_k left_{b,i,k} * right_{b,j,k}
where left and right are projected + gated tensors of shape (B,N,N,H).
The contraction over k is a batched matrix multiplication left @ right^T for each
hidden dimension H. We fuse this O(N^3) operation in a Triton kernel, while
the remaining linear / layer‑norm / gating operations are performed in PyTorch.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B_total, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out = left @ right^T for each batch (where batch encodes both
    original batch and hidden dimension). The input tensors have shape
        left  : (B_total, N, N)
        right : (B_total, N, N)
    and the output has the same shape.
    """
    batch_idx = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Block start indices
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    # Offsets for the i/j dimensions
    offs_i = i0 + tl.arange(0, BLOCK_M)
    offs_j = j0 + tl.arange(0, BLOCK_N)

    # Masks for edge handling
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Pointer offset for the current batch
    batch_off = batch_idx * stride_lb   # left/right/out share the same batch stride

    # Output pointer
    out_ptrs = out_ptr + batch_off + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj

    # Accumulator in FP32 for precision
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k
    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile (M x K)
        left_ptrs = left_ptr + batch_off + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # Load right tile transposed (K x N) -> we load using (k, j) layout
        right_ptrs = right_ptr + batch_off + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Accumulate dot product
        acc += tl.dot(left, right)

    # Write back result in FP16 (the downstream code converts to FP32)

def custom_kernel(data):
    """
    Forward pass of the TriMul module.
    Arguments:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)    
    """
    # unpack arguments
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (shape [B, N, N, dim])
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # (B, N, N, H)
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: (B, N, N) -> broadcast to (B, N, N, 1)
        mask_ = mask.to(x.dtype).unsqueeze(-1)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------------
    # 4. Gating (Sigmoid of linear transforms of x)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 and prepare for Triton kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Move hidden dimension to the batch dimension:
    #   (B, N, N, H) -> (B, H, N, N) -> (B*H, N, N)
    left = left.permute(0, 3, 1, 2).contiguous()
    right = right.permute(0, 3, 1, 2).contiguous()
    B, H, N, _ = left.shape  # note: H == hidden_dim
    batch_total = B * H
    left = left.view(batch_total, N, N)
    right = right.view(batch_total, N, N)

    # Output buffer for the kernel
    out_flat = torch.empty_like(left)  # FP16, shape (B*H, N, N)

    # ------------------------------------------------------------------
    # 6. Triton kernel launch (batched matmul left @ right^T)
    # ------------------------------------------------------------------
    # Tile sizes – tuned for H100; fits comfortably in registers.
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32

    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Strides in element (not byte) units
    stride_lb, stride_li, stride_lk = left.stride()
    stride_rb, stride_ri, stride_rk = right.stride()
    stride_ob, stride_oi, stride_oj = out_flat.stride()

    trimul_einsum_kernel[grid](
        left,
        right,
        out_flat,
        batch_total,
        N,
        stride_lb,
        stride_li,
        stride_lk,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_ob,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 7. Reshape back to (B, N, N, H) and cast to FP32
    # ------------------------------------------------------------------
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()
    out = out.to(torch.float32)

    # ------------------------------------------------------------------
    # 8. Final LayerNorm, gating, and linear projection to original dim
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out

"""
Optimized TriMul forward pass (AlphaFold3 "outgoing" version).

The core of the TriMul operator is the O(N³) contraction
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]
which can be seen as a batched matrix multiplication
    left_h[b] @ right_h[b].T    for each hidden channel h.
We fuse this contraction in a Triton kernel while all other
operations (LayerNorm, linear projections, gating) are performed
with PyTorch.  The kernel works on tensors of shape
    (B*H, N, N)  (batch dimension encodes both the original batch
    and the hidden channel).  The result is reshaped back to
    (B, N, N, H) and the remaining post‑processing is applied.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B_total, N,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Batched matmul kernel: out = left @ right.T

    Shapes (all in elements):
        left : (B_total, N, N)
        right: (B_total, N, N)
        out  : (B_total, N, N)

    The multiplication is performed for each batch independently and
    accumulates in FP32 for numerical stability.
    """
    batch_idx = tl.program_id(0)      # which (batch, hidden) pair
    pid_i = tl.program_id(1)         # i‑tile index
    pid_j = tl.program_id(2)         # j‑tile index

    # -------------------------------------------------------------
    # Tile start indices
    # -------------------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N

    offs_i = i0 + tl.arange(0, BLOCK_M)   # (BLOCK_M,)
    offs_j = j0 + tl.arange(0, BLOCK_N)   # (BLOCK_N,)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # -------------------------------------------------------------
    # Base pointer for the current batch
    # -------------------------------------------------------------
    batch_off = batch_idx * stride_lb     # same stride for left/right/out

    # -------------------------------------------------------------
    # Output pointer (M×N tile)
    # -------------------------------------------------------------
    out_ptrs = out_ptr + batch_off + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj

    # -------------------------------------------------------------
    # Accumulator (FP32)
    # -------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # -------------------------------------------------------------
    # Reduction over k
    # -------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k0 = k
        offs_k = k0 + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left tile : (M, K)
        left_ptrs = left_ptr + batch_off + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # right tile transposed : (K, N)  (load as (K, N) to match tl.dot)
        right_ptrs = right_ptr + batch_off + offs_k[:, None] * stride_rk + offs_j[None, :] * stride_ri
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        acc += tl.dot(left, right)   # (M, K) @ (K, N) -> (M, N)

    # -------------------------------------------------------------
    # Store result (back to FP16)
    # -------------------------------------------------------------
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
