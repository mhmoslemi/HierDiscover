@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,  # total number of matrices M = B*hidden_dim
    stride_lm, stride_li, stride_lk,
    stride_rm, stride_ri, stride_rk,
    stride_outm, stride_outi, stride_outj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # Program ID: each program handles a tile of output (i,j) for a given matrix m.
    pid = tl.program_id(0)
    # compute grid dims:
    total_blocks_i = tl.cdiv(N, BLOCK_M)
    total_blocks_j = tl.cdiv(N, BLOCK_N)
    total_blocks_m = M
    # pid to (m, block_i, block_j)
    m = pid // (total_blocks_i * total_blocks_j)
    pid_ij = pid % (total_blocks_i * total_blocks_j)
    block_i = pid_ij // total_blocks_j
    block_j = pid_ij % total_blocks_j

    # offsets for i/j dimensions
    i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask out-of-bounds
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        # mask for k
        mask_k = k_offset < N

        # compute pointers
        left_ptrs = left_ptr + m * stride_lm + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        # For right, we need transposed: indices (k,j) -> i index = k_offset, k index = j_offset
        right_ptrs = right_ptr + m * stride_rm + k_offset[:, None] * stride_ri + j_offset[None, :] * stride_rk

        # load; use float16 or float32. Input likely float32. We'll casting to float32 for accumulation.
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product
        acc += tl.dot(left, right)  # left: (BLOCK_M, BLOCK_K), right: (BLOCK_K, BLOCK_N) => (BLOCK_M, BLOCK_N)

    # Write out result

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # LayerNorm: we must manually compute stats? Use torch.nn.functional.layer_norm
    # shape normalized_shape = (dim,)
    # To be safe: compute mean & var across last dimension using torch.mean, torch.var

    x_norm = torch.nn.functional.layer_norm(input_tensor, normalized_shape=(dim,), weight=weights["norm.weight"], bias=weights["norm.bias"])

    # Linear projections
    left = torch.nn.functional.linear(x_norm, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    # Apply mask if present
    if mask is not None:
        # mask shape [B,N,N]; unsqueeze to broadcast to dim of channel
        mask_exp = mask.unsqueeze(-1).to(dtype)
        left = left * mask_exp
        right = right * mask_exp

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, weights["out_gate.weight"]))

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute contraction using Triton kernel
    B, N, N2, C2 = left.shape
    assert N == N2 and C2 == hidden_dim

    # Rearrange to (B * hidden_dim, N, N)
    left_reshape = left.permute(0, 3, 1, 2).contiguous()
    right_reshape = right.permute(0, 3, 1, 2).contiguous()
    M = B * hidden_dim
    left_reshape = left_reshape.view(M, N, N)
    right_reshape = right_reshape.view(M, N, N)

    # Prepare output tensor
    out_reshape = torch.empty_like(left_reshape)

    # Triton kernel constants
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    import triton
    import triton.language as tl

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        M, N,
        stride_lm, stride_li, stride_lk,
        stride_rm, stride_ri, stride_rk,
        stride_outm, stride_outi, stride_outj,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)
        # Compute grid dims
        num_blocks_i = tl.cdiv(N, BLOCK_M)
        num_blocks_j = tl.cdiv(N, BLOCK_N)
        # total blocks per matrix = num_blocks_i * num_blocks_j
        blocks_per_mat = num_blocks_i * num_blocks_j
        # Determine which matrix (m) and block indices
        m = pid // blocks_per_mat
        pid_ij = pid % blocks_per_mat
        block_i = pid_ij // num_blocks_j
        block_j = pid_ij % num_blocks_j

        # Offsets
        i_offset = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
        j_offset = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_i = i_offset < N
        mask_j = j_offset < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # Loop over k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_offset < N

            # Load left tile (i,k)
            left_ptrs = left_ptr + m * stride_lm + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
            left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

            # Load right tile transposed (k,j)
            right_ptrs = right_ptr + m * stride_rm + k_offset[:, None] * stride_ri + j_offset[None, :] * stride_rk
            right_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

            # Matrix multiply accumulation
            acc += tl.dot(left_tile, right_tile)

        # Write result
        out_ptrs = out_ptr + m * stride_outm + i_offset[:, None] * stride_outi + j_offset[None, :] * stride_outj
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # Determine grid size
    num_blocks_i = (N + BLOCK_M - 1) // BLOCK_M
    num_blocks_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (M * num_blocks_i * num_blocks_j,)

    # Launch kernel
    tri_mul_kernel[grid](
        left_ptr=left_reshape,
        right_ptr=right_reshape,
        out_ptr=out_reshape,
        M=M,
        N=N,
        stride_lm=left_reshape.stride(0),
        stride_li=left_reshape.stride(1),
        stride_lk=left_reshape.stride(2),
        stride_rm=right_reshape.stride(0),
        stride_ri=right_reshape.stride(1),
        stride_rk=right_reshape.stride(2),
        stride_outm=out_reshape.stride(0),
        stride_outi=out_reshape.stride(1),
        stride_outj=out_reshape.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # Reshape result back to original shape
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # to_out_norm
    out_norm = torch.nn.functional.layer_norm(out, normalized_shape=(hidden_dim,), weight=weights["to_out_norm.weight"], bias=weights["to_out_norm.bias"])

    # Gating with out_gate
    out_norm = out_norm * out_gate

    # final linear projection
    out = torch.nn.functional.linear(out_norm, weights["to_out.weight"])

"""
TriMul (outgoing) forward pass implemented with a fused Triton kernel.

The operation consists of:
1. Layer‑norm over the last channel dimension.
2. Linear projections (left / right) and three sigmoid gates.
3. Optional masking of the projected tensors.
4. A three‑way contraction:
        out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
   which is equivalent to a batch of matrix‑multiplications
   (one for every (batch, hidden_dim) pair). This O(N³) kernel is
   tiled over the i‑ and j‑axes and reduces over k using `tl.dot`.
5. A second Layer‑norm, out‑gate modulation and a final linear projection.

Only the heavy contraction (step 4) is performed on the GPU with Triton;
the remaining element‑wise work uses PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F

def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
            - input_tensor: torch.Tensor of shape [B, N, N, C] (C = dim)
            - mask_tensor : torch.Tensor of shape [B, N, N] or None.
            - weights_dict : mapping str -> torch.Tensor containing all model
                              parameters required by the reference implementation.
            - config_dict  : must contain "dim", "hidden_dim" and optionally "nomask".

    Returns:
        torch.Tensor of shape [B, N, N, dim] (the TriMul output).
    """
    # ----------------------------------------------------------------------
    # unpack inputs
    # ----------------------------------------------------------------------
    x, mask, weights, config = data
    device = x.device
    dtype = x.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----------------------------------------------------------------------
    # 1. Input LayerNorm
    # ----------------------------------------------------------------------
    # torch.nn.functional.layer_norm expects weight/bias of shape [C]
    x = F.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # ----------------------------------------------------------------------
    # 2. Linear projections (no bias) and gate computation
    # ----------------------------------------------------------------------
    left  = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])  # [B, N, N, hidden_dim]

    # optional mask (mask shape [B,N,N] -> broadcast to channel dim)
    if (mask is not None) and (not nomask):
        mask_exp = mask.unsqueeze(-1).to(dtype)      # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # gate tensors (sigmoid after linear)
    left_gate  = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))  # [B,N,N,hidden_dim]
    out_gate   = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))    # [B,N,N,hidden_dim]

    # apply gates
    left  = left * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 3. Heavy 3‑way contraction computed with Triton
    # ----------------------------------------------------------------------
    B, N, _, _ = left.shape          # N is the pairwise sequence length

    # reshape to (M, N, N) where M = B * hidden_dim
    left_t = left.permute(0, 3, 1, 2).contiguous()   # [B, hidden_dim, N, N]
    right_t = right.permute(0, 3, 1, 2).contiguous() # [B, hidden_dim, N, N]

    M = B * hidden_dim
    left_t  = left_t.view(M, N, N)
    right_t = right_t.view(M, N, N)

    out_t = torch.empty_like(left_t)   # [M, N, N]

    # --------------------------------------------------------------
    # Triton kernel: batched mat‑mul with transposed RHS (i·k · k·j)
    # --------------------------------------------------------------
    BLOCK_M = 128   # tile size for the i‑axis
    BLOCK_N = 128   # tile size for the j‑axis
    BLOCK_K = 32    # reduction tile size

    @triton.jit
    def tri_mul_kernel(
        left_ptr, right_ptr, out_ptr,
        M, N,
        stride_lm, stride_li, stride_lk,
        stride_rm, stride_ri, stride_rk,
        stride_outm, stride_outi, stride_outj,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    ):
        pid = tl.program_id(0)

        # number of tiles per matrix
        num_i = tl.cdiv(N, BLOCK_M)
        num_j = tl.cdiv(N, BLOCK_N)
        tiles_per_mat = num_i * num_j

        # decode program id -> (matrix m, tile i, tile j)
        m = pid // tiles_per_mat
        pid_ij = pid % tiles_per_mat
        block_i = pid_ij // num_j
        block_j = pid_ij % num_j

        # tile offsets (i‑axis, j‑axis)
        i_off = block_i * BLOCK_M + tl.arange(0, BLOCK_M)
        j_off = block_j * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_i = i_off < N
        mask_j = j_off < N

        # accumulator in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over the reduction dimension k
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            k_off = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = k_off < N

            # left tile: (i, k)
            left_ptrs = (left_ptr +
                         m * stride_lm +
                         i_off[:, None] * stride_li +
                         k_off[None, :] * stride_lk)
            left_tile = tl.load(left_ptrs,
                                mask=mask_i[:, None] & mask_k[None, :],
                                other=0.0).to(tl.float32)

            # right tile transposed: we need (k, j) so we swap i/k when loading
            right_ptrs = (right_ptr +
                          m * stride_rm +
                          k_off[:, None] * stride_ri +   # i index = k
                          j_off[None, :] * stride_rk)    # k index = j
            right_tile = tl.load(right_ptrs,
                                 mask=mask_k[:, None] & mask_j[None, :],
                                 other=0.0).to(tl.float32)

            # matrix multiply‑accumulate
            acc += tl.dot(left_tile, right_tile)   # (BLOCK_M, BLOCK_N)

        # write the result
        out_ptrs = (out_ptr +
                    m * stride_outm +
                    i_off[:, None] * stride_outi +
                    j_off[None, :] * stride_outj)
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

    # launch configuration
    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    grid = (M * num_i * num_j,)

    # stride arguments (int64)
    stride_lm, stride_li, stride_lk = left_t.stride()
    stride_rm, stride_ri, stride_rk = right_t.stride()
    stride_outm, stride_outi, stride_outj = out_t.stride()

    tri_mul_kernel[grid](
        left_ptr=left_t,
        right_ptr=right_t,
        out_ptr=out_t,
        M=M,
        N=N,
        stride_lm=int(stride_lm),
        stride_li=int(stride_li),
        stride_lk=int(stride_lk),
        stride_rm=int(stride_rm),
        stride_ri=int(stride_ri),
        stride_rk=int(stride_rk),
        stride_outm=int(stride_outm),
        stride_outi=int(stride_outi),
        stride_outj=int(stride_outj),
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_t.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # 4. Second LayerNorm, out‑gate and final projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
