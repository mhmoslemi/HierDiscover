@triton.jit
def trilu_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J
    d_start = pid_d * BLOCK_D

    i_offset = i_start + tl.arange(0, BLOCK_I)
    j_offset = j_start + tl.arange(0, BLOCK_J)
    d_offset = d_start + tl.arange(0, BLOCK_D)

    # Masks for out-of-range
    i_mask = i_offset < N
    j_mask = j_offset < N
    d_mask = d_offset < H

    # allocate accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # Loop over k dimension
    for k in range(0, N):
        # get left tile for i,k
        left_ptrs = left_ptr + i_offset[:, None] * stride_left_i + k * stride_left_k + d_offset[None, :] * stride_left_d
        left_tile = tl.load(left_ptrs, mask=i_mask[:, None] & d_mask[None, :], other=0.0) # shape (BLOCK_I, BLOCK_D)
        # right tile for j,k
        right_ptrs = right_ptr + j_offset[:, None] * stride_right_i + k * stride_right_k + d_offset[None, :] * stride_right_d
        right_tile = tl.load(right_ptrs, mask=j_mask[:, None] & d_mask[None, :], other=0.0) # shape (BLOCK_J, BLOCK_D)

        # broadcast left and right
        # left_tile_exp: (BLOCK_I, 1, BLOCK_D)
        # right_tile_exp: (1, BLOCK_J, BLOCK_D)
        left_exp = tl.broadcast_to(left_tile[:, None, :], (BLOCK_I, BLOCK_J, BLOCK_D))
        right_exp = tl.broadcast_to(right_tile[None, :, :], (BLOCK_I, BLOCK_J, BLOCK_D))
        # accumulate
        acc += left_exp * right_exp

    # store result
    out_ptrs = out_ptr + i_offset[:, None, None] * stride_out_i + j_offset[None, :, None] * stride_out_j + d_offset[None, None, :] * stride_out_d
    tl.store(out_ptrs, acc, mask=i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

def triton_trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_d,
    stride_right_i, stride_right_k, stride_right_d,
    stride_out_i, stride_out_j, stride_out_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_D: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    i_idx = i_base + tl.arange(0, BLOCK_I)
    j_idx = j_base + tl.arange(0, BLOCK_J)
    d_idx = d_base + tl.arange(0, BLOCK_D)

    # mask out-of-range
    i_mask = i_idx < N
    j_mask = j_idx < N
    d_mask = d_idx < H

    # allocate accumulator in float32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N):
        # left slice shape (BLOCK_I, BLOCK_D)
        left_ptrs = left_ptr + i_idx[:, None] * stride_left_i + k * stride_left_k + d_idx[None, :] * stride_left_d
        left = tl.load(left_ptrs, mask=i_mask[:, None] & d_mask[None, :], other=0.0).to(tl.float32)
        # right slice shape (BLOCK_J, BLOCK_D)
        right_ptrs = right_ptr + j_idx[:, None] * stride_right_i + k * stride_right_k + d_idx[None, :] * stride_right_d
        right = tl.load(right_ptrs, mask=j_mask[:, None] & d_mask[None, :], other=0.0).to(tl.float32)

        left_exp = left[:, None, :]  # shape (BLOCK_I, 1, BLOCK_D)
        right_exp = right[None, :, :]  # shape (1, BLOCK_J, BLOCK_D)

    out_ptrs = out_ptr + i_idx[:, None, None] * stride_out_i + j_idx[None, :, None] * stride_out_j + d_idx[None, None, :] * stride_out_d

TriMul (outgoing) kernel implementation with Triton.

The heavy part of TriMul is the einsum
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a 3‑D reduction over the sequence dimension k while keeping the hidden
dimension d.  The kernel tiles the (i,j) plane and the hidden channel d,
loads a vector of length d for each (i,k) and (j,k) pair, and performs a
rank‑1 update:
    out[i,j,d] += left[i,k,d] * right[j,k,d]
The accumulation is done in float32 for numerical stability.  All other
operations (layer‑norms, linear projections, sigmoids, masking) are carried
out in PyTorch.

The implementation works for batch sizes 1‑2, sequence lengths up to 1024,
and hidden dimensions 128‑384.  It respects the optional mask (controlled by
config["nomask"]).  The final output has shape [B, N, N, dim] and dtype
torch.float32.
"""

import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel for the core einsum reduction
# ---------------------------------------------------------
@triton.jit
def triton_trimul_kernel(
    left_ptr,          # *float16
    right_ptr,         # *float16
    out_ptr,           # *float32
    N,                 # sequence length (runtime)
    H,                 # hidden dimension (runtime)
    stride_l_i, stride_l_k, stride_l_d,   # left strides (elements)
    stride_r_i, stride_r_k, stride_r_d,   # right strides (elements)
    stride_o_i, stride_o_j, stride_o_d,   # output strides (elements)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    """
    Compute:
        out[i, j, d] = Σ_k left[i, k, d] * right[j, k, d]
    for a single batch (the caller passes a pointer already offset to the batch).
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    i_idx = i_base + tl.arange(0, BLOCK_I)
    j_idx = j_base + tl.arange(0, BLOCK_J)
    d_idx = d_base + tl.arange(0, BLOCK_D)

    i_mask = i_idx < N
    j_mask = j_idx < N
    d_mask = d_idx < H

    # accumulator in fp32
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ---------------------------------------------------------
    # Loop over the reduction dimension k (seq_len)
    # ---------------------------------------------------------
    for k in range(0, N):
        # ------------ load left[i, k, d] -------------
        left_off = (
            i_idx[:, None] * stride_l_i   # i dimension
            + k * stride_l_k               # k dimension (scalar for this iteration)
            + d_idx[None, :] * stride_l_d  # d dimension
        )
        left = tl.load(left_ptr + left_off,
                       mask=(i_mask[:, None] & d_mask[None, :]),
                       other=0.0).to(tl.float32)          # (BLOCK_I, BLOCK_D)

        # ------------ load right[j, k, d] ------------
        right_off = (
            j_idx[:, None] * stride_r_i
            + k * stride_r_k
            + d_idx[None, :] * stride_r_d
        )
        right = tl.load(right_ptr + right_off,
                        mask=(j_mask[:, None] & d_mask[None, :]),
                        other=0.0).to(tl.float32)         # (BLOCK_J, BLOCK_D)

        # Broadcast to (BLOCK_I, BLOCK_J, BLOCK_D) and accumulate
        # left:  (BLOCK_I, 1, BLOCK_D)
        # right: (1, BLOCK_J, BLOCK_D)
        left_exp = tl.reshape(left, (BLOCK_I, 1, BLOCK_D))
        right_exp = tl.reshape(right, (1, BLOCK_J, BLOCK_D))
        acc += left_exp * right_exp

    # ---------------------------------------------------------
    # Store the result
    # ---------------------------------------------------------
    out_off = (
        i_idx[:, None, None] * stride_o_i
        + j_idx[None, :, None] * stride_o_j
        + d_idx[None, None, :] * stride_o_d
    )
    tl.store(out_ptr + out_off,
             acc,
             mask=(i_mask[:, None, None] &
                   j_mask[None, :, None] &
                   d_mask[None, None, :]))


# ---------------------------------------------------------
# Entry point called by the autograder
# ---------------------------------------------------------
def custom_kernel(data):
    """
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)
        - input_tensor : torch.Tensor  [B, N, N, dim]
        - mask_tensor  : torch.Tensor  [B, N, N] (bool/float)
        - weights_dict : mapping name -> torch.Tensor
        - config_dict  : contains "dim", "hidden_dim", "nomask" (bool)

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # -----------------------------------------------------------------
    # Unpack inputs
    # -----------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # -----------------------------------------------------------------
    # LayerNorm on input (dim last)
    # -----------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias)

    # -----------------------------------------------------------------
    # Linear projections (no bias)
    # -----------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_proj = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right_proj = F.linear(x, right_proj_weight)

    # -----------------------------------------------------------------
    # Gating linear layers + sigmoid
    # -----------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # -----------------------------------------------------------------
    # Apply mask (if any) and gating
    # -----------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] -> broadcast to [B, N, N, 1]
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left_proj * mask * left_gate
        right = right_proj * mask * right_gate
    else:
        left = left_proj * left_gate
        right = right_proj * right_gate

    # Cast to fp16 for the Triton kernel (memory bandwidth)
    left = left.to(torch.float16).contiguous()
    right = right.to(torch.float16).contiguous()

    # -----------------------------------------------------------------
    # Allocate output tensor for the kernel (fp32 accumulator)
    # -----------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # Strides in element units (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # -----------------------------------------------------------------
    # Triton launch parameters
    # -----------------------------------------------------------------
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_D = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        (hidden_dim + BLOCK_D - 1) // BLOCK_D,
    )

    # Launch one kernel per batch (B ≤ 2)
    for b in range(B):
        left_ptr = left[b].data_ptr()
        right_ptr = right[b].data_ptr()
        out_ptr = out[b].data_ptr()
        triton_trimul_kernel[grid](
            left_ptr,
            right_ptr,
            out_ptr,
            N,
            hidden_dim,
            stride_l_i, stride_l_k, stride_l_d,
            stride_r_i, stride_r_k, stride_r_d,
            stride_o_i, stride_o_j, stride_o_d,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_D=BLOCK_D,
            # compile-time constants
        )

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm on hidden, gating, final linear
    # -----------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = F.layer_norm(out,
                            normalized_shape=(hidden_dim,),
                            weight=to_out_norm_weight,
                            bias=to_out_norm_bias)

    # Apply the output gate (broadcasted over the last hidden dim)
    out = out_norm * out_gate  # out_gate is float32, broadcasting works

    # Final linear projection back to original dim
    to_out_weight = weights["to_out.weight"]
    out = F.linear(out, to_out_weight)  # shape [B, N, N, dim]

    return out
