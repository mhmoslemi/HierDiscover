@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
   b_idx = tl.program_id(0) # batch*hidden index
   pid_m = tl.program_id(1)
   pid_n = tl.program_id(2)

   # compute batch and hidden index
   b = b_idx // H
   h = b_idx % H

   # define offsets
   i_offset = pid_m * BLOCK_M
   j_offset = pid_n * BLOCK_N

   # If i_offset >= N or j_offset >= N: exit early?
   # Use boundary checks via mask.

   # pointer base offsets
   left_ptr_base = left_ptr + b * stride_lb + h * stride_lh
   right_ptr_base = right_ptr + b * stride_rb + h * stride_rh
   out_ptr_base = out_ptr + b * stride_ob + h * stride_oh

   # allocate accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

   # K loop
   for k in range(0, tl.cdiv(N, BLOCK_K)):
       k_offset = k * BLOCK_K
       # compute i,k offsets for left
       offs_i = i_offset + tl.arange(0, BLOCK_M)
       offs_k = k_offset + tl.arange(0, BLOCK_K)
       # left: shape [BLOCK_M, BLOCK_K]
       left_ptrs = left_ptr_base + offs_i[:, None] * stride_li + offs_k[None, :] * stride_lk
       # mask for OOB: i < N, k < N
       mask_i = offs_i < N
       mask_k = offs_k < N
       left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)

       # right: shape [BLOCK_N, BLOCK_K]
       offs_j = j_offset + tl.arange(0, BLOCK_N)
       right_ptrs = right_ptr_base + offs_j[:, None] * stride_rj + offs_k[None, :] * stride_rk
       mask_j = offs_j < N
       right = tl.load(right_ptrs, mask=mask_j[:, None] & mask_k[None, :], other=0.0)

       # dot product over K
       acc += tl.dot(left, right)

   # store result
   # compute output pointers: shape [BLOCK_M, BLOCK_N]
   offs_i = i_offset + tl.arange(0, BLOCK_M)
   offs_j = j_offset + tl.arange(0, BLOCK_N)
   out_ptrs = out_ptr_base + offs_i[:, None] * stride_oi + offs_j[None, :] * stride_oj
   mask_out = (offs_i[:, None] < N) & (offs_j[None, :] < N)
   tl.store(out_ptrs, acc.to(tl.float16) if out_ptr.dtype == tl.float16 else acc, mask=mask_out)

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    batch_idx = tl.program_id(0)
    pid_m = tl.program_id(1)
    pid_n = tl.program_id(2)

    # Compute batch and hidden indexes
    b = batch_idx // H
    h = batch_idx % H

    # Compute base pointers for this batch+h
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # Offsets for output block
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # Compute indices
        i = i_start + tl.arange(0, BLOCK_M)
        j = j_start + tl.arange(0, BLOCK_N)
        kk = k_start + tl.arange(0, BLOCK_K)

        # Bounds
        mask_i = i < N
        mask_j = j < N
        mask_k = kk < N

        # Load left block (M x K)
        left_offset = i[:, None] * stride_li + kk[None, :] * stride_lk
        left = tl.load(left_base + left_offset, mask=mask_i[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # Load right block (N x K)
        right_offset = j[:, None] * stride_rj + kk[None, :] * stride_rk
        right = tl.load(right_base + right_offset, mask=mask_j[:, None] & mask_k[None, :], other=0.0, dtype=tl.float16)

        # dot product over K
        acc += tl.dot(left, right)  # results in float32

    # Store results
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N
    out_offset = i[:, None] * stride_oi + j[None, :] * stride_oj

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h]
    left/right are half, out is float32.
    Batch dimension is flattened as (b * H + h).
    """
    batch_idx = tl.program_id(0)  # 0 .. B*H-1
    pid_m = tl.program_id(1)      # block row index (i)
    pid_n = tl.program_id(2)      # block col index (j)

    # Unpack batch and hidden indices
    b = batch_idx // H
    h = batch_idx % H

    # Base pointers for this (b,h)
    left_base = left_ptr + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base = out_ptr + b * stride_ob + h * stride_oh

    # Starting coordinates of the current tile
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Accumulator in fp32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # Indices for the tile
        i = i_start + tl.arange(0, BLOCK_M)
        j = j_start + tl.arange(0, BLOCK_N)
        kk = k_start + tl.arange(0, BLOCK_K)

        # Boundary masks
        mask_i = i < N
        mask_j = j < N
        mask_k = kk < N

        # Load left block [M, K] (half precision)
        left_off = i[:, None] * stride_li + kk[None, :] * stride_lk
        left = tl.load(left_base + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)

        # Load right block [N, K] (half precision)
        right_off = j[:, None] * stride_rj + kk[None, :] * stride_rk
        right = tl.load(right_base + right_off,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0, dtype=tl.float16)

        # Accumulate dot product over K (result in fp32)
        acc += tl.dot(left, right)

    # Store the result tile into output (fp32)
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N
    out_off = i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_base + out_off, acc, mask=mask_i[:, None] & mask_j[None, :])
    

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the "outgoing" TriMul operator.
    The heavy pairwise multiplication (einsum over the third axis) is
    computed with a fused batched matrix multiplication kernel.
    All other linear, gating and layer‑norm steps are performed in PyTorch.
    """
    # Unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    # Config parameters
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # Retrieve weight tensors
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ---------- 1. LayerNorm over the last dimension ----------
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ---------- 2. Linear projections ----------
    # Linear layers have weight shape (out_features, in_features)
    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight) # [B,N,N,hidden]

    # ---------- 3. Optional masking ----------
    if not nomask and mask is not None:
        # Broadcast mask over hidden dim
        mask_unsq = mask.to(dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_unsq
        right = right * mask_unsq

    # ---------- 4. Gating ----------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ---------- 5. Compute the core multiplication with Triton ----------
    # Work in FP16 for the kernel (memory bandwidth reduction)
    left_half = left.to(torch.float16)
    right_half = right.to(torch.float16)

    B, N, _, _ = left_half.shape  # N = seq_len, hidden_dim = hidden_dim

    # Allocate output tensor (float32 accumulation)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides (in element units)
    l_stride_b, l_stride_i, l_stride_k, l_stride_h = left_half.stride()
    r_stride_b, r_stride_j, r_stride_k, r_stride_h = right_half.stride()
    o_stride_b, o_stride_i, o_stride_j, o_stride_h = out.stride()

    # Kernel block sizes – tuned for H100, but generic enough
    BLOCK_M = 64   # i dimension tile
    BLOCK_N = 64   # j dimension tile
    BLOCK_K = 32   # reduction dimension tile

    # Grid: (batch*hidden, n_blocks_i, n_blocks_j)
    batch_total = B * hidden_dim
    grid = (batch_total,
            triton.cdiv(N, BLOCK_M),
            triton.cdiv(N, BLOCK_N))

    # Launch kernel
    tri_mul_kernel[grid](
        left_half, right_half, out,
        B, N, hidden_dim,
        l_stride_b, l_stride_i, l_stride_k, l_stride_h,
        r_stride_b, r_stride_j, r_stride_k, r_stride_h,
        o_stride_b, o_stride_i, o_stride_j, o_stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ---------- 6. Output layer‑norm ----------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # ---------- 7. Gated output ----------
    out = out * out_gate

    # ---------- 8. Final linear projection ----------
    out = F.linear(out, to_out_weight)   # shape [B,N,N,dim]

    return out

"""
TriMul (outgoing) forward pass with Triton acceleration.

The heavy part of the TriMul operator is the 4‑D einsum:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

For every hidden dimension `d` this is a standard matrix multiplication
    out_d = left_d @ right_dᵀ   (shape N×N)
with `left_d` and `right_d` of shape (N, N).  We fuse the three
dimensions (batch, hidden) into a single batch index and implement a
batched GEMM in Triton.  All other steps (LayerNorm, linear projections,
gating, final projection) are performed with PyTorch for simplicity.

The kernel works in FP16 for the two input tensors (lower bandwidth) and
accumulates in FP32 for numerical stability.  The final result is stored
as FP32 and fed back to the remaining PyTorch layers.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # half  * (B, N, N, H)
    right_ptr,              # half  * (B, N, N, H)
    out_ptr,                # fp32  * (B, N, N, H)
    B, N, H,
    stride_lb, stride_li, stride_lk, stride_lh,
    stride_rb, stride_rj, stride_rk, stride_rh,
    stride_ob, stride_oi, stride_oj, stride_oh,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """
    Compute
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    The batch dimension is flattened as (b * H + h).  `left` and `right`
    are stored as FP16, accumulation is performed in FP32.
    """
    batch_idx = tl.program_id(0)      # 0 … B*H-1
    pid_m     = tl.program_id(1)      # block row (i)
    pid_n     = tl.program_id(2)      # block col (j)

    # De‑flatten batch / hidden indices
    b = batch_idx // H
    h = batch_idx % H

    # Base pointers for this (b, h)
    left_base  = left_ptr  + b * stride_lb + h * stride_lh
    right_base = right_ptr + b * stride_rb + h * stride_rh
    out_base   = out_ptr   + b * stride_ob + h * stride_oh

    # Tile origin in the output matrix
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension K
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # Coordinate vectors
        i   = i_start + tl.arange(0, BLOCK_M)
        j   = j_start + tl.arange(0, BLOCK_N)
        kk  = k_start + tl.arange(0, BLOCK_K)

        # Boundary masks
        mask_i = i < N
        mask_j = j < N
        mask_k = kk < N

        # ---- Load left block [M, K] -------------------------------------------------
        left_off = i[:, None] * stride_li + kk[None, :] * stride_lk
        left = tl.load(left_base + left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0, dtype=tl.float16)

        # ---- Load right block [N, K] ------------------------------------------------
        right_off = j[:, None] * stride_rj + kk[None, :] * stride_rk
        right = tl.load(right_base + right_off,
                        mask=mask_j[:, None] & mask_k[None, :],
                        other=0.0, dtype=tl.float16)

        # ---- Dot‑product over K (accumulate in FP32) ---------------------------------
        acc += tl.dot(left, right)

    # ---- Store the output tile -------------------------------------------------------
    i = i_start + tl.arange(0, BLOCK_M)
    j = j_start + tl.arange(0, BLOCK_N)
    mask_i = i < N
    mask_j = j < N
    out_off = i[:, None] * stride_oi + j[None, :] * stride_oj
    tl.store(out_base + out_off, acc, mask=mask_i[:, None] & mask_j[None, :])

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))   # [B, N, N, hidden_dim]

    left  = left  * left_gate
    right = right * right_gate

    # ----------------------------------------------------------------------
    # 7️⃣  Core O(N³) multiplication – Triton kernel
    # ----------------------------------------------------------------------
    # Cast to FP16 for the kernel (lower memory bandwidth)
    left_h  = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left_h.shape   # N == seq_len

    # Output buffer (FP32 accumulation)
    out = torch.empty((B, N, N, hidden_dim),
                      dtype=torch.float32,
                      device=device)

    # Strides expressed in element counts (not bytes)
    lsb, lsi, lsk, lsh = left_h.stride()
    rsb, rsj, rsk, rsh = right_h.stride()
    osb, osi, osj, osh = out.stride()

    # Tile sizes – chosen to fit H100 shared‑memory / register budget
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (batch*hidden, n_i_tiles, n_j_tiles)
    batch_total = B * hidden_dim
    grid = (
        batch_total,
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
    )

    # Launch
    tri_mul_kernel[grid](
        left_h, right_h, out,
        B, N, hidden_dim,
        lsb, lsi, lsk, lsh,
        rsb, rsj, rsk, rsh,
        osb, osi, osj, osh,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
    )

    # ----------------------------------------------------------------------
    # 8️⃣  Output LayerNorm (over hidden_dim)
    # ----------------------------------------------------------------------
    out = F.layer_norm(out,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # ----------------------------------------------------------------------
    # 9️⃣  Gated output & final linear projection back to ``dim``
    # ----------------------------------------------------------------------
    out = out * out_gate                       # element‑wise gating
    out = F.linear(out, to_out_weight)          # [B, N, N, dim]

    return out
```
