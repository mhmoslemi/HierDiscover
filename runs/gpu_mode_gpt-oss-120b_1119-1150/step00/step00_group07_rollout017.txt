@triton.jit
def tri_mul_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_abe, stride_a_m, stride_a_k,
    stride_bbe, stride_b_k, stride_b_n,
    stride_cbe, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    batch_idx = tl.program_id(axis=2) # each batch slice
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    # compute offsets
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    a_ptrs = a_ptr + batch_idx * stride_abe + offs_m[:, None] * stride_a_m + offs_k[None, :] * stride_a_k
    b_ptrs = b_ptr + batch_idx * stride_bbe + offs_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n

    # load
    a = tl.load(a_ptrs, mask=offs_m < M, other=0.0)
    # b will be loaded in loop over k
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k in range(0, tl.cdiv(K, BLOCK_K)):
       # compute current offsets
       k_offset = k * BLOCK_K
       cur_k = offs_k + k_offset
       a_ptrs = a_ptr + batch_idx * stride_abe + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k
       b_ptrs = b_ptr + batch_idx * stride_bbe + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
       a = tl.load(a_ptrs, mask=(offs_m[:, None] < M) & (cur_k[None, :] < K), other=0.0)
       b = tl.load(b_ptrs, mask=(cur_k[:, None] < K) & (offs_n[None, :] < N), other=0.0)
       acc += tl.dot(a, b)
    # store result
    c_ptrs = c_ptr + batch_idx*stride_cbe + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=(offs_m[:, None] < M) & (offs_n[None, :] < N))

```
@triton.jit
def tux_gemm_kernel(
    a_ptr, b_ptr, c_ptr,
    BATCH, M, N, K,
    stride_abe, stride_a_m, stride_a_k,
    stride_bbe, stride_b_k, stride_b_n,
    stride_cbe, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_batch = tl.program_id(axis=2)
    pid_m = tl.program_id(axis=0)
    pid_n = tl.program_id(axis=1)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # load A tile
        a_ptrs = a_ptr + pid_batch * stride_abe + offs_m[:, None] * stride_a_m + cur_k[None, :] * stride_a_k
        a = tl.load(a_ptrs, mask=mask_m[:, None] & mask_k[None, :], other=0.0)

        # load B_T tile (original B shape; using strides for transposed view)
        b_ptrs = b_ptr + pid_batch * stride_bbe + cur_k[:, None] * stride_b_k + offs_n[None, :] * stride_b_n
        b = tl.load(b_ptrs, mask=mask_k[:, None] & mask_n[None, :], other=0.0)

        acc += tl.dot(a, b)

    # store
    c_ptrs = c_ptr + pid_batch * stride_cbe + offs_m[:, None] * stride_c_m + offs_n[None, :] * stride_c_n
    tl.store(c_ptrs, acc, mask=mask_m[:, None] & mask_n[None, :])
```

    right = right * right_gate

"""
TriMul forward pass (outgoing version) with a custom Triton kernel.

The kernel implements the heavy N³ reduction:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by performing a batched matrix multiplication
    out_slice = left_slice @ right_slice.T
for each (batch, hidden) slice.  Layers‑norms, linear projections,
gating and the final projection are carried out with PyTorch; the
core O(N³) work is accelerated in half‑precision on H100 using Triton.
"""

import torch
import triton
import triton.language as tl
import torch.nn.functional as F


@triton.jit
def tri_mul_gemm_kernel(
    a_ptr,               # left  [batch, M, K]   (float16)
    b_ptr,               # right [batch, N, K]   (float16) – will be read as K×N (transposed)
    c_ptr,               # out   [batch, M, N]   (float32)
    BATCH, M, N, K,      # dimensions
    # strides (in elements, not bytes)
    stride_abe, stride_a_m, stride_a_k,
    stride_bbe, stride_b_k, stride_b_n,
    stride_cbe, stride_c_m, stride_c_n,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Batched GEMM: for each batch index b,
        C[b] = A[b] @ B[b].T
    where A, B are (M, K) and (N, K) respectively.
    Accumulation is performed in FP32, inputs are FP16.
    """
    pid_batch = tl.program_id(axis=2)          # batch index
    pid_m = tl.program_id(axis=0)              # block row
    pid_n = tl.program_id(axis=1)              # block col

    # Offsets within the M and N dimensions
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_m = offs_m < M
    mask_n = offs_n < N

    # Accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K dimension in chunks of BLOCK_K
    for k in range(0, tl.cdiv(K, BLOCK_K)):
        cur_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = cur_k < K

        # Load a tile of A (M×K)
        a_ptrs = (
            a_ptr
            + pid_batch * stride_abe
            + offs_m[:, None] * stride_a_m
            + cur_k[None, :] * stride_a_k
        )
        a = tl.load(
            a_ptrs,
            mask=mask_m[:, None] & mask_k[None, :],
            other=0.0,
        )  # fp16

        # Load a tile of Bᵀ (K×N).  B is stored as (N,K); we reinterpret it as (K,N)
        b_ptrs = (
            b_ptr
            + pid_batch * stride_bbe
            + cur_k[:, None] * stride_b_k   # row‑stride = original col‑stride
            + offs_n[None, :] * stride_b_n   # col‑stride = original row‑stride
        )
        b = tl.load(
            b_ptrs,
            mask=mask_k[:, None] & mask_n[None, :],
            other=0.0,
        )  # fp16

        # Dot product: (BLOCK_M×BLOCK_K)·(BLOCK_K×BLOCK_N) → (BLOCK_M×BLOCK_N)
        acc += tl.dot(a, b)   # result accumulated in FP32

    # Store the result
    c_ptrs = (
        c_ptr
        + pid_batch * stride_cbe
        + offs_m[:, None] * stride_c_m
        + offs_n[None, :] * stride_c_n
    )
    tl.store(
        c_ptrs,
        acc,
        mask=mask_m[:, None] & mask_n[None, :],
    )


def custom_kernel(data):
    """
    TriMul forward pass (outgoing) using a fused Triton kernel for the
    N³ reduction.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim]
            - mask: torch.Tensor of shape [B, N, N] (may be all‑ones)
            - weights: dict of model parameters (see code)
            - config: dict containing "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=eps,
    )  # [B, N, N, dim]   FP32

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on the hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        m = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * m
        right = right * m

    # ------------------------------------------------------------------
    # 4. Gating
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))  # later

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Cast to FP16 and reshape for the batched GEMM kernel
    # ------------------------------------------------------------------
    B, N, _, _ = input_tensor.shape
    # Move hidden dimension to second axis and flatten batch*hidden
    left = left.to(torch.float16).permute(0, 3, 1, 2).contiguous()   # [B, hidden, N, N]
    right = right.to(torch.float16).permute(0, 3, 1, 2).contiguous()

    total_slices = B * hidden_dim
    left = left.view(total_slices, N, N)
    right = right.view(total_slices, N, N)

    # ------------------------------------------------------------------
    # 6. Allocate output buffer (FP32 accumulation)
    # ------------------------------------------------------------------
    out = torch.empty(total_slices, N, N, dtype=torch.float32, device=left.device)

    # ------------------------------------------------------------------
    # 7. Launch the Triton kernel
    # ------------------------------------------------------------------
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # blocks in M dimension
        (N + BLOCK_N - 1) // BLOCK_N,   # blocks in N dimension
        total_slices,                   # one program per (batch, hidden) slice
    )

    # Strides in elements (not bytes)
    stride_a_batch, stride_a_m, stride_a_k = left.stride()
    stride_b_batch, stride_b_n, stride_b_k = right.stride()
    stride_c_batch, stride_c_m, stride_c_n = out.stride()

    tri_mul_gemm_kernel[grid](
        left,
        right,
        out,
        total_slices,
        N,
        N,
        N,
        stride_a_batch,
        stride_a_m,
        stride_a_k,
        stride_b_batch,
        stride_b_k,   # row‑stride for K (original col‑stride)
        stride_b_n,   # col‑stride for N (original row‑stride)
        stride_c_batch,
        stride_c_m,
        stride_c_n,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 8. Reshape back to [B, N, N, hidden_dim]
    # ------------------------------------------------------------------
    out = out.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()  # FP32

    # ------------------------------------------------------------------
    # 9. Output layer‑norm, out‑gate and final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=eps,
    )
    out = out * out_gate
    out = F.linear(out, weights["to_out.weight"])

    return out
