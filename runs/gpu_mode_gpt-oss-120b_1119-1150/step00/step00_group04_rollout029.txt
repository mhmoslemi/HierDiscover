    pid_j = tl.program_id(1)

```python
"""
TriMul forward kernel using Triton to accelerate the pairwise
multiplication and reduction over the sequence dimension.
The kernel computes:
    out[b,i,j,d] = sum_{k=0}^{N-1} left[b,i,k,d] * right[b,j,k,d]
for each batch `b` and hidden dimension `d`.  All other operations
(LayerNorm, linear projections, sigmoid gates) are performed with
PyTorch.  The kernel works for arbitrary batch size (1‑2), sequence
lengths up to 1024 and hidden dimensions that are multiples of
`BLOCK_D` (a fallback with masking handles the remainder).
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute the TriMul contraction for a tile.

    Each program instance (block) computes a TILE of shape
    (BLOCK_M, BLOCK_N, BLOCK_D) in the output tensor.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    # ----- tile offsets ---------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N
    d_base = pid_d * BLOCK_D

    # index vectors within the tile
    offs_i = i_base + tl.arange(0, BLOCK_M)
    offs_j = j_base + tl.arange(0, BLOCK_N)
    offs_d = d_base + tl.arange(0, BLOCK_D)

    # out‑of‑bounds masks for each axis
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < H

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----------------------------------------------------------------------
    # Loop over the reduction dimension k
    # ----------------------------------------------------------------------
    num_k = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k):
        k_base = k_tile * BLOCK_K
        # iterate over the inner K axis (compile‑time loop)
        for kk in range(BLOCK_K):
            k = k_base + kk
            mask_k = k < N

            # ------ load left[i, k, d] ------------------------------------
            left_ptrs = left_ptr + \
                (offs_i[:, None] * stride_left_i) + \
                (k * stride_left_k) + \
                (offs_d[None, :] * stride_left_h)
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k & mask_d[None, :],
                other=0.0,
                dtype=tl.float32)

            # ------ load right[j, k, d] -----------------------------------
            right_ptrs = right_ptr + \
                (offs_j[:, None] * stride_right_i) + \
                (k * stride_right_k) + \
                (offs_d[None, :] * stride_right_h)
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None] & mask_k & mask_d[None, :],
                other=0.0,
                dtype=tl.float32)

            # ------ outer‑product accumulation ------------------------------
            # left : (BLOCK_M, BLOCK_D)   -> (M, 1, D)
            # right: (BLOCK_N, BLOCK_D)   -> (1, N, D)
            acc += left[:, None, :] * right[None, :, :]

    # ----------------------------------------------------------------------
    # Store the result
    # ----------------------------------------------------------------------
    out_ptrs = out_ptr + \
        (offs_i[:, None, None] * stride_out_i) + \
        (offs_j[None, :, None] * stride_out_j) + \
        (offs_d[None, None, :] * stride_out_h)

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    )

def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant) using a Triton
    kernel for the O(N³·H) contraction.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict mapping string names to torch.Tensor containing the model weights
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # unpack inputs
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # Linear projections
    # ------------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]  # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"] # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)

    # ------------------------------------------------------------------
    # Apply mask (if provided).  mask is broadcast on the hidden dimension.
    # ------------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones((bs, N, N), dtype=input_tensor.dtype, device=device)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B,N,N,1]
    left = left * mask
    right = right * mask

    # ------------------------------------------------------------------
    # Gating pathways
    # ------------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Triton kernel for the 3‑D contraction
    # ------------------------------------------------------------------
    # Prepare output tensor
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim], same dtype as left

    # Constants for the kernel
    BLOCK_M = 16   # i‑axis tile size
    BLOCK_N = 16   # j‑axis tile size
    BLOCK_D = 32   # hidden‑dim tile size
    BLOCK_K = 4    # inner‑k tile size (compile‑time constant)

    # Compute grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    launch_grid = (grid_i, grid_j, grid_d)

    # Launch one kernel per batch (bs ≤ 2)
    for b in range(bs):
        # Pointers for this batch
        left_ptr  = left[b].contiguous()
        right_ptr = right[b].contiguous()
        out_ptr   = out[b].contiguous()

        # Strides (in elements, not bytes)
        stride_left_i, stride_left_k, stride_left_h = left_ptr.stride()
        stride_right_i, stride_right_k, stride_right_h = right_ptr.stride()

"""
TriMul (outgoing) forward pass.

This implementation follows the reference PyTorch module but
accelerates the cubic contraction

    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]

with a Triton kernel.  All surrounding operations
(LayerNorm, linear projections, sigmoid gates) are executed in
PyTorch.  The kernel works for batch sizes 1‑2, sequence lengths up
to 1024 and hidden dimensions that are multiples of the tile size
(`BLOCK_D`).  Tail handling is performed with masks, so non‑multiple
sizes are also supported.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    stride_left_i, stride_left_k, stride_left_h,
    stride_right_i, stride_right_k, stride_right_h,
    stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Tile‐wise computation of the TriMul contraction.

    Each program instance computes a tile of shape
    (BLOCK_M, BLOCK_N, BLOCK_D) in the output tensor.
    """
    pid_i = tl.program_id(0)   # i‑tile
    pid_j = tl.program_id(1)   # j‑tile
    pid_d = tl.program_id(2)   # hidden‑dim tile

    # ----- tile base indices -------------------------------------------------
    i0 = pid_i * BLOCK_M
    j0 = pid_j * BLOCK_N
    d0 = pid_d * BLOCK_D

    # ----- index vectors inside the tile ------------------------------------
    off_i = i0 + tl.arange(0, BLOCK_M)
    off_j = j0 + tl.arange(0, BLOCK_N)
    off_d = d0 + tl.arange(0, BLOCK_D)

    mask_i = off_i < N
    mask_j = off_j < N
    mask_d = off_d < H

    # accumulator (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # ----- reduction over k --------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_base = k_tile * BLOCK_K
        # inner loop over the K dimension (compile‑time unrolled)
        for kk in range(BLOCK_K):
            k = k_base + kk
            mask_k = k < N

            # left[i, k, d]  -> shape (BLOCK_M, BLOCK_D)
            left_ptrs = left_ptr + \
                (off_i[:, None] * stride_left_i) + \
                (k * stride_left_k) + \
                (off_d[None, :] * stride_left_h)
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None] & mask_k & mask_d[None, :],
                other=0.0,
                dtype=tl.float32)

            # right[j, k, d] -> shape (BLOCK_N, BLOCK_D)
            right_ptrs = right_ptr + \
                (off_j[:, None] * stride_right_i) + \
                (k * stride_right_k) + \
                (off_d[None, :] * stride_right_h)
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None] & mask_k & mask_d[None, :],
                other=0.0,
                dtype=tl.float32)

            # outer‑product accumulation: (M,1,D) * (1,N,D) -> (M,N,D)
            acc += left[:, None, :] * right[None, :, :]

    # ----- write the tile -----------------------------------------------------
    out_ptrs = out_ptr + \
        (off_i[:, None, None] * stride_out_i) + \
        (off_j[None, :, None] * stride_out_j) + \
        (off_d[None, None, :] * stride_out_h)

    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    )


def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) module.
    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [B, N, N, dim]
        - mask         : torch.Tensor of shape [B, N, N] (may be all‑ones)
        - weights      : dict of torch.Tensors – model parameters
        - config       : dict with keys "dim" and "hidden_dim"
    Returns
    -------
    torch.Tensor of shape [B, N, N, dim]
    """
    # --------------------------------------------------------------
    # unpack and basic shapes
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    # --------------------------------------------------------------
    # 1) Input LayerNorm over the channel dimension `dim`
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # shape [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden_dim, dim)

    left  = F.linear(x, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_weight)

    # --------------------------------------------------------------
    # 3) Apply mask (if any) – broadcast on hidden dimension
    # --------------------------------------------------------------
    if mask is None or mask.numel() == 0:
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=device)
    mask = mask.to(input_tensor.dtype).unsqueeze(-1)   # [B, N, N, 1]
    left  = left * mask
    right = right * mask

    # --------------------------------------------------------------
    # 4) Gating pathways
    # --------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5) Triton kernel for the O(N³·H) contraction
    # --------------------------------------------------------------
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # Tile sizes – tuned for H100 (register‑friendly)
    BLOCK_M = 16   # i‑axis
    BLOCK_N = 16   # j‑axis
    BLOCK_D = 32   # hidden‑dim axis
    BLOCK_K = 4    # inner reduction chunk (compile‑time)

    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    grid_d = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    launch_grid = (grid_i, grid_j, grid_d)

    # launch one kernel per batch element (B ≤ 2)
    for b in range(B):
        left_b  = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b   = out[b].contiguous()

        # strides in elements (not bytes)
        s_li, s_lk, s_lh = left_b.stride()
        s_ri, s_rk, s_rh = right_b.stride()
        s_oi, s_oj, s_oh = out_b.stride()

        trimul_kernel[launch_grid, num_warps=8](
            left_b.data_ptr(),
            right_b.data_ptr(),
            out_b.data_ptr(),
            N,
            hidden_dim,
            s_li, s_lk, s_lh,
            s_ri, s_rk, s_rh,
            s_oi, s_oj, s_oh,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )

    # --------------------------------------------------------------
    # 6) Post‑processing: hidden‑dim LayerNorm, out‑gate, final Linear
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]   # (dim, hidden_dim)

    # hidden‑dim LayerNorm
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )

    # apply out gate
    out = out * out_gate

    # final projection back to `dim`
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
