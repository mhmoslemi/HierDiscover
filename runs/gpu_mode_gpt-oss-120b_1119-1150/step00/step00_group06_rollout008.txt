def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    @triton.jit
    def trimul_matmul_kernel(left_ptr, right_ptr, out_ptr,
                             B, H, N,
                             stride_left_b, stride_left_i, stride_left_k, stride_left_h,
                             stride_right_b, stride_right_i, stride_right_k, stride_right_h,
                             stride_out_b, stride_out_i, stride_out_j, stride_out_h,
                             BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr):
        # mat_id encodes (b, h)
        mat_id = tl.program_id(0)
        pid_i = tl.program_id(1)
        pid_j = tl.program_id(2)

        b = mat_id // H
        h = mat_id % H

        # base pointers for this matrix
        left_base = left_ptr + b * stride_left_b + h * stride_left_h
        right_base = right_ptr + b * stride_right_b + h * stride_right_h
        out_base = out_ptr + b * stride_out_b + h * stride_out_h

        # tile offsets
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

        mask_i = offs_i < N
        mask_j = offs_j < N

        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # loop over k dimension
        for k in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = k * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile [M, K]
            left_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
            left_tile = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :],
                                other=0.0, dtype=tl.float16).to(tl.float32)

            # right tile transposed: we need shape [K, N] where rows are k, cols are j.
            right_ptrs = right_base + (offs_k[:, None] * stride_right_i) + (offs_j[None, :] * stride_right_k)
            right_tile_T = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :],
                                   other=0.0, dtype=tl.float16).to(tl.float32)

            # accumulate dot product
            acc += tl.dot(left_tile, right_tile_T)

        # write out
        out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
custom_kernel: Fast Triton implementation of the “outgoing” TriMul operator.

Algorithm
---------
1. Layer‑norm the input tensor over the feature dimension *dim*.
2. Compute four linear projections of the normalized tensor:
   * left_proj , right_proj   – hidden dimension (no bias)
   * left_gate , right_gate , out_gate – sigmoid‑gated (no bias)
3. Optional masking of *left_proj*/*right_proj*.
4. Fuse gating:   left = left_proj * left_gate
                 right = right_proj * right_gate
   (Both tensors are cast to FP16 to reduce memory traffic.)
5. Compute the core tensor contraction:
      out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   This is exactly a batched matrix‑multiply
   ( left[...,h] @ right[...,h].T ) for every hidden channel *h*.
   It is implemented in Triton with a 3‑D grid:
     * program_id(0) → (batch, hidden) pair
     * program_id(1) → tile of the output‑row dimension *i*
     * program_id(2) → tile of the output‑column dimension *j*
   The kernel loads FP16 tiles, promotes them to FP32 for the dot
   product, and writes the result back in FP32.
6. Layer‑norm the result over the hidden dimension, multiply by *out_gate*,
   and apply the final linear projection back to *dim*.

Only the expensive O(N³) contraction is performed on the GPU with Triton;
all other element‑wise operations use native PyTorch primitives.
"""

def custom_kernel(data):
    import torch
    import triton
    import triton.language as tl

    # ------------------------------------------------------------------
    # Triton kernel: batched GEMM for the TriMul contraction.
    # ------------------------------------------------------------------
    @triton.jit
    def trimul_matmul_kernel(
        left_ptr, right_ptr, out_ptr,
        B, H, N,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
    ):
        # (b, h) index for the current matrix
        mat_id = tl.program_id(0)
        pid_i = tl.program_id(1)   # tile along i (rows)
        pid_j = tl.program_id(2)   # tile along j (cols)

        b = mat_id // H
        h = mat_id % H

        # base pointers for the (b,h) slice
        left_base = left_ptr + b * stride_left_b + h * stride_left_h
        right_base = right_ptr + b * stride_right_b + h * stride_right_h
        out_base = out_ptr + b * stride_out_b + h * stride_out_h

        # tile offsets
        offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
        offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
        mask_i = offs_i < N
        mask_j = offs_j < N

        # accumulator in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

        # iterate over the reduction dimension k
        for kk in range(0, tl.cdiv(N, BLOCK_K)):
            offs_k = kk * BLOCK_K + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left tile   : shape (M, K)  -> left[i, k, h]
            left_ptrs = left_base + (offs_i[:, None] * stride_left_i) + (offs_k[None, :] * stride_left_k)
            left_tile = tl.load(left_ptrs,
                                mask=mask_i[:, None] & mask_k[None, :],
                                other=0.0, dtype=tl.float16).to(tl.float32)

            # right tile transposed: shape (K, N)  -> right_T[k, j, h] = right[j, k, h]
            right_ptrs = right_base + (offs_k[:, None] * stride_right_k) + (offs_j[None, :] * stride_right_i)
            right_tile_T = tl.load(right_ptrs,
                                   mask=mask_k[:, None] & mask_j[None, :],
                                   other=0.0, dtype=tl.float16).to(tl.float32)

            # dot‑product accumulation
            acc += tl.dot(left_tile, right_tile_T)

        # write the result back (FP32)
        out_ptrs = out_base + (offs_i[:, None] * stride_out_i) + (offs_j[None, :] * stride_out_j)
        tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[:, None])

    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # Load model parameters
    # ------------------------------------------------------------------
    # layer‑norm on the input feature dimension
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    # linear projections (no bias)
    w_left_proj = weights["left_proj.weight"]       # (hidden_dim, dim)
    w_right_proj = weights["right_proj.weight"]
    # gates (no bias)
    w_left_gate = weights["left_gate.weight"]
    w_right_gate = weights["right_gate.weight"]
    w_out_gate = weights["out_gate.weight"]
    # output path
    w_to_out_norm = weights["to_out_norm.weight"]
    b_to_out_norm = weights["to_out_norm.bias"]
    w_to_out = weights["to_out.weight"]            # (dim, hidden_dim)

    # ------------------------------------------------------------------
    # 1. Input layer‑norm
    # ------------------------------------------------------------------
    # Normalise over the last dimension (feature dim)
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_w,
        bias=norm_b,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2. Linear projections + gates
    # ------------------------------------------------------------------
    left_proj = torch.nn.functional.linear(x, w_left_proj)   # [B,N,N,hidden]
    right_proj = torch.nn.functional.linear(x, w_right_proj)

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, w_left_gate))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, w_right_gate))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, w_out_gate))

    # ------------------------------------------------------------------
    # 3. Optional mask (mask shape [B,N,N])
    # ------------------------------------------------------------------
    if mask_tensor is not None and not nomask:
        mask = mask_tensor.to(dtype=x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left_proj = left_proj * mask
        right_proj = right_proj * mask

    # ------------------------------------------------------------------
    # 4. Apply gates
    # ------------------------------------------------------------------
    left = left_proj * left_gate
    right = right_proj * right_gate

    # ------------------------------------------------------------------
    # 5. Heavy O(N³) contraction via Triton
    # ------------------------------------------------------------------
    # Cast to FP16 to reduce bandwidth; accumulation will be FP32.
    left_f16 = left.to(torch.float16).contiguous()
    right_f16 = right.to(torch.float16).contiguous()

    # Allocate output tensor for the contraction (FP32)
    out_fp32 = torch.empty(
        (bs, N, N, hidden_dim),
        dtype=torch.float32,
        device=input_tensor.device,
    )

    # Strides (in element counts)
    lsb, lsi, lsk, lsh = left_f16.stride()
    rsb, rsi, rsk, rsh = right_f16.stride()
    osb, osi, osk, osh = out_fp32.stride()

    # Tunable block sizes (empirically good for H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: one program per (batch, hidden) pair and per tile in i/j.
    grid = (
        bs * hidden_dim,               # program_id(0) → (b, h)
        triton.cdiv(N, BLOCK_M),       # program_id(1) → i‑tile
        triton.cdiv(N, BLOCK_N),       # program_id(2) → j‑tile
    )

    trimul_matmul_kernel[grid](
        left_f16,
        right_f16,
        out_fp32,
        bs,
        hidden_dim,
        N,
        lsb,
        lsi,
        lsk,
        lsh,
        rsb,
        rsi,
        rsk,
        rsh,
        osb,
        osi,
        osk,       # stride over the *j* axis (original k axis)
        osh,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # 6. Output layer‑norm, gating and final projection
    # ------------------------------------------------------------------
    # Layer‑norm over the hidden dimension
    out_norm = torch.nn.functional.layer_norm(
        out_fp32,
        normalized_shape=(hidden_dim,),
        weight=w_to_out_norm,
        bias=b_to_out_norm,
    )  # [B,N,N,hidden]

    # Apply the output gate
    out_norm = out_norm * out_gate

    # Final linear back to original feature dimension (no bias)
    final = torch.nn.functional.linear(out_norm, w_to_out)  # [B,N,N,dim]

    return final
```
