    B, N, D,

"""
TriMul (outgoing) forward pass implemented with a tiny Triton kernel.

Algorithm
---------
1. Layer‑normalize the input tensor `x` over the channel dimension `dim`.
2. Compute linear projections for the left/right values and for the three gates.
   All linear layers are bias‑free and are performed with `torch.nn.functional.linear`.
3. Apply the pairwise mask and the left/right gates element‑wise.
   This cheap element‑wise work is fused into a custom Triton kernel
   (mask * left_proj * left_gate , mask * right_proj * right_gate).
4. Contract over the third sequence index `k`:
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   The contraction is a batched matrix‑multiply; we reshape the tensors to
   (B·hidden_dim, N, N) and call `torch.bmm`.
5. Layer‑normalize the intermediate result over the hidden dimension,
   multiply by the output gate, and apply the final linear projection back
   to the original `dim` channels.

Only the element‑wise mask‑and‑gate step is written in Triton; the heavy
matrix multiplication is delegated to cuBLAS via `torch.bmm`, which is
highly optimized on H100.  The implementation works for both masked and
unmasked inputs and supports the shapes required by the test suite.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _mask_gate_kernel(
    # pointers
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    out_left_ptr, out_right_ptr,
    # tensor shapes
    B, N, D,
    # strides
    stride_left_b, stride_left_i, stride_left_j, stride_left_d,
    stride_gate_b, stride_gate_i, stride_gate_j, stride_gate_d,
    stride_mask_b, stride_mask_i, stride_mask_j,
    # compile‑time block size
    BLOCK_SIZE: tl.constexpr,
):
    """
    Apply mask and gates:
        out_left  = left  * mask * left_gate
        out_right = right * mask * right_gate
    The kernel processes BLOCK_SIZE elements per program.
    """
    pid = tl.program_id(0)
    offs = tl.arange(0, BLOCK_SIZE)
    idx = pid * BLOCK_SIZE + offs
    total = B * N * N * D
    mask = idx < total

    # decompose linear index into (b,i,j,d)
    d = idx % D
    idx_div = idx // D
    j = idx_div % N
    idx_div = idx_div // N
    i = idx_div % N
    b = idx_div // N

    # compute flat offsets for each tensor
    off_left = (b * stride_left_b + i * stride_left_i
                + j * stride_left_j + d * stride_left_d)
    off_right = (b * stride_left_b + i * stride_left_i
                 + j * stride_left_j + d * stride_left_d)
    off_lgate = (b * stride_gate_b + i * stride_gate_i
                 + j * stride_gate_j + d * stride_gate_d)
    off_rgate = (b * stride_gate_b + i * stride_gate_i
                 + j * stride_gate_j + d * stride_gate_d)
    off_mask = (b * stride_mask_b + i * stride_mask_i + j * stride_mask_j)

    # load (convert to float32 for arithmetic)
    left = tl.load(left_ptr + off_left, mask=mask, other=0.0).to(tl.float32)
    right = tl.load(right_ptr + off_right, mask=mask, other=0.0).to(tl.float32)
    lgate = tl.load(left_gate_ptr + off_lgate, mask=mask, other=0.0).to(tl.float32)
    rgate = tl.load(right_gate_ptr + off_rgate, mask=mask, other=0.0).to(tl.float32)
    mval = tl.load(mask_ptr + off_mask, mask=mask, other=0.0).to(tl.float32)

    # apply mask and gates
    out_l = left * mval * lgate
    out_r = right * mval * rgate

    # store results (cast back to the original dtype)
    tl.store(out_left_ptr + off_left, out_l.to(left_ptr.dtype), mask=mask)
    tl.store(out_right_ptr + off_right, out_r.to(right_ptr.dtype), mask=mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ----------------------------------------------------------------------
    # Unpack inputs
    # ----------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # input_tensor: [B, N, N, dim]
    # mask:          [B, N, N]  (may be None if nomask)
    # weights: dict of torch tensors
    # config:  {'dim': int, 'hidden_dim': int, 'nomask': bool, ...}

    device = input_tensor.device
    dtype = input_tensor.dtype

    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ----------------------------------------------------------------------
    # LayerNorm over the channel dimension (dim)
    # ----------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # ----------------------------------------------------------------------
    # Linear projections (bias‑free)
    # ----------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # shape: [B, N, N, hidden_dim]
    left_proj = F.linear(x, left_proj_weight)          # left projection
    right_proj = F.linear(x, right_proj_weight)        # right projection
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    # ----------------------------------------------------------------------
    # Prepare mask (broadcast to hidden_dim) – optional
    # ----------------------------------------------------------------------
    if config.get("nomask", False) or mask is None:
        mask_tensor = torch.ones((B, N, N), dtype=dtype, device=device)
    else:
        mask_tensor = mask.to(dtype)

    # Allocate output tensors for the gated values
    left_gated = torch.empty_like(left_proj)
    right_gated = torch.empty_like(right_proj)

    # ----------------------------------------------------------------------
    # Triton kernel: apply mask and gates element‑wise
    # ----------------------------------------------------------------------
    BLOCK_SIZE = 128
    total_elements = B * N * N * hidden_dim
    grid = (triton.cdiv(total_elements, BLOCK_SIZE),)

    # Strides for the tensors (as Python ints)
    s_left = left_proj.stride()
    s_gate = left_gate.stride()
    s_mask = mask_tensor.stride()

    _mask_gate_kernel[grid](
        left_proj,
        right_proj,
        left_gate,
        right_gate,
        mask_tensor,
        left_gated,
        right_gated,
        B,
        N,
        hidden_dim,
        s_left[0], s_left[1], s_left[2], s_left[3],
        s_gate[0], s_gate[1], s_gate[2], s_gate[3],
        s_mask[0], s_mask[1], s_mask[2],
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,
    )

    # ----------------------------------------------------------------------
    # Contract over the third sequence index (k) using batched matmul
    # ----------------------------------------------------------------------
    # reshape to (B*hidden_dim, N, N) for both left and right
    left_flat = left_gated.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)
    right_flat = right_gated.permute(0, 3, 1, 2).contiguous().view(B * hidden_dim, N, N)

    # out_flat shape: (B*hidden_dim, N, N)
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # reshape back to (B, N, N, hidden_dim)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ----------------------------------------------------------------------
    # Second LayerNorm (over hidden_dim), gating, final linear projection
    # ----------------------------------------------------------------------
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias, eps=eps)

    out = out * out_gate  # apply output gate

    # final linear maps back to original dimension `dim`
    out = F.linear(out, to_out_weight)

    return out
```
