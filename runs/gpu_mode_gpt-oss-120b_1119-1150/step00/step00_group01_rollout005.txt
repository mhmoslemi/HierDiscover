"""
custom_kernel for the “outgoing” Triangle Multiplicative Update (TriMul).

The kernel fuses the costly N×N×N tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
by a batched matrix‑multiply implemented in Triton.
All other Tensor‑wise operations (LayerNorm, linear projections,
gate sigmoids, masking) are done with PyTorch for simplicity.
The implementation works for any batch size (1‑2), sequence length
up to 1024 and hidden dimension 128 (as in AlphaFold‑3), and falls
back to the same result as the reference PyTorch module.
"""

import math
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# -------------------------------------------------------------------------
# Triton kernel: batched matrix multiplication (A @ B) across many batches
# -------------------------------------------------------------------------
@triton.jit
def _batched_matmul_kernel(
    a_ptr, b_ptr, c_ptr,               # pointers
    M, N, K,                           # matrix dimensions
    stride_a_batch, stride_a_m, stride_a_k,
    stride_b_batch, stride_b_k, stride_b_n,
    stride_c_batch, stride_c_m, stride_c_n,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute C = A @ B for many independent batches.
    A : (batch, M, K)
    B : (batch, K, N)
    C : (batch, M, N)
    """
    pid_m = tl.program_id(0)   # block index in M dimension
    pid_n = tl.program_id(1)   # block index in N dimension
    pid_b = tl.program_id(2)   # batch (including hidden‑dim) index

    # ---------------------------------------------------------------
    # Compute offsets for this block
    # ---------------------------------------------------------------
    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # Masks for out‑of‑bounds loads
    mask_m = offs_m < M
    mask_n = offs_n < N
    mask_k = offs_k < K

    # ----------------------------------------------------------------
    # Pointers to the first tile of the current batch
    # ----------------------------------------------------------------
    a_base = a_ptr + pid_b * stride_a_batch
    b_base = b_ptr + pid_b * stride_b_batch
    c_base = c_ptr + pid_b * stride_c_batch

    # ----------------------------------------------------------------
    # Load tiles of A (M×K) and B (K×N) and accumulate
    # ----------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    for _ in range(0, tl.cdiv(K, BLOCK_K)):
        # Pointers for this tile of K
        a_tile_ptr = a_base + (offs_m[:, None] * stride_a_m
                               + (offs_k[None, :] + _) * BLOCK_K * stride_a_k)
        b_tile_ptr = b_base + ((offs_k[:, None] + _) * BLOCK_K * stride_b_k
                               + offs_n[None, :] * stride_b_n)

        a = tl.load(a_tile_ptr,
                    mask=mask_m[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_tile_ptr,
                    mask=mask_k[:, None] & mask_n[None, :],
                    other=0.0)

        # Accumulate matrix product of the tile
        acc += tl.dot(a, b)   # (BLOCK_M, BLOCK_K) @ (BLOCK_K, BLOCK_N) → (BLOCK_M, BLOCK_N)

        # Advance K offsets for the next iteration
        # (the stride updates are folded into the loop index via the +_ term)

    # ----------------------------------------------------------------
    # Write the result
    # ----------------------------------------------------------------
    c_tile_ptr = c_base + (offs_m[:, None] * stride_c_m
                           + offs_n[None, :] * stride_c_n)
    tl.store(c_tile_ptr,
             acc,
             mask=mask_m[:, None] & mask_n[None, :])


def _batched_matmul_triton(A: torch.Tensor, B: torch.Tensor,
                           BLOCK_M: int = 64, BLOCK_N: int = 64, BLOCK_K: int = 32) -> torch.Tensor:
    """
    Wrapper for the Triton batched matmul kernel.
    A : (batch, M, K)   B : (batch, K, N)   →   C : (batch, M, N)
    """
    assert A.is_cuda and B.is_cuda, "Triton kernel requires CUDA tensors"
    batch, M, K = A.shape
    _, K2, N = B.shape
    assert K == K2, f"Inner dimensions must match, got {K} vs {K2}"

    C = torch.empty((batch, M, N), dtype=A.dtype, device=A.device)

    grid = (
        (M + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        batch,
    )

    _batched_matmul_kernel[grid](
        A, B, C,
        M, N, K,
        A.stride(0), A.stride(1), A.stride(2),
        B.stride(0), B.stride(1), B.stride(2),
        C.stride(0), C.stride(1), C.stride(2),
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=4,  # 128 threads per warp is a good default for fp32
    )
    # Synchronize to make sure the kernel finishes before we use C
    torch.cuda.synchronize()
    return C

    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # --------------------------------------------------------------
    # 1) Input layer‑norm
    # --------------------------------------------------------------
    x_norm = F.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x_norm, left_proj_weight)   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, right_proj_weight)  # same shape

    # --------------------------------------------------------------
    # 3) Optional masking (mask is broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask_tensor is not None:
        # make mask the same dtype as the projections and add a trailing dim
        mask = mask_tensor.to(left.dtype).unsqueeze(-1)   # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # --------------------------------------------------------------
    # 4) Compute gating values (sigmoid of linear projections of x_norm)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, left_gate_weight))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x_norm, out_gate_weight))    # used later

    # --------------------------------------------------------------
    # 5) Apply gates
    # --------------------------------------------------------------
    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 6) Core contraction: out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
    #    Implemented as a batched matmul in Triton.
    # --------------------------------------------------------------
    #   bring hidden_dim to a batch dimension:
    #   left : (B, N, N, H)  →  (B, H, N, N)  (i as rows, k as cols)
    #   right: (B, N, N, H)  →  (B, H, N, N)  (k as rows, j as cols)
    left_perm = left.permute(0, 3, 1, 2).contiguous()          # [B, H, N, N]
    right_perm = right.permute(0, 3, 2, 1).contiguous()        # [B, H, N, N]

    B, H, N, _ = left_perm.shape
    # flatten batch and hidden dimension so that we have a single batch axis
    left_flat = left_perm.view(B * H, N, N)    # (batch*H, M=N, K=N)
    right_flat = right_perm.view(B * H, N, N)  # (batch*H, K=N, N)

    # Triton batched matmul
    out_flat = _batched_matmul_triton(left_flat, right_flat)  # (B*H, N, N)

    # Reshape back to [B, N, N, H]
    out = out_flat.view(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 7) Output layer‑norm, gating and final linear projection
    # --------------------------------------------------------------
    out_norm = F.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out_norm = out_norm * out_gate   # broadcast over the hidden dimension

    # Final linear: maps hidden_dim → dim (no bias)
    output = F.linear(out_norm, to_out_weight)   # [B, N, N, dim]

    return output
```
