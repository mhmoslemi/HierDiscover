@triton.jit
def trimul_kernel(left_ptr, right_ptr, out_ptr,
                  B, N, D,
                  stride_l_b, stride_l_i, stride_l_k, stride_l_d,
                  stride_r_b, stride_r_i, stride_r_k, stride_r_d,
                  stride_o_b, stride_o_i, stride_o_j, stride_o_d,
                  BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bd = tl.program_id(2)

    # compute batch and hidden dim index
    b = pid_bd // D
    d = pid_bd % D

    # offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # range of indices
    i_range = i_base + tl.arange(0, BLOCK_I)
    j_range = j_base + tl.arange(0, BLOCK_J)

    # mask for out-of-range i,j
    i_mask = i_range < N
    j_mask = j_range < N

    # initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # iterate over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_base = k * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_range < N

        # left tile shape (BLOCK_I, BLOCK_K)
        left_ptrs = left_ptr + b * stride_l_b + d * stride_l_d \
                    + i_range[:, None] * stride_l_i \
                    + k_range[None, :] * stride_l_k
        left_mask = i_mask[:, None] & k_mask[None, :]
        left = tl.load(left_ptrs, mask=left_mask, other=0.0)

        # right tile transposed shape (BLOCK_K, BLOCK_J)
        # we load as (BLOCK_K, BLOCK_J) = k_range[:, None] * stride_r_k + j_range[None, :] * stride_r_i
        right_ptrs = right_ptr + b * stride_r_b + d * stride_r_d \
                    + k_range[:, None] * stride_r_k \
                    + j_range[None, :] * stride_r_i
        right_mask = k_mask[:, None] & j_mask[None, :]
        # load as (BLOCK_K, BLOCK_J)
        right = tl.load(right_ptrs, mask=right_mask, other=0.0)

        # dot product: left (BLOCK_I, BLOCK_K) @ right (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        acc += tl.dot(left, right)

    # Store result
    out_ptrs = out_ptr + b * stride_o_b + d * stride_o_d \
               + i_range[:, None] * stride_o_i \
               + j_range[None, :] * stride_o_j
    out_mask = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc, mask=out_mask)

"""
custom_kernel for the outgoing TriMul operation (AlphaFold3).

Algorithm
---------
1. Layer‑normalize the 4‑D input tensor `x` over its last dimension `dim`.
2. Compute two linear projections (`left_proj`, `right_proj`) from `dim` → `hidden_dim`.
3. (Optionally) mask the projected tensors.
4. Compute three sigmoid gates (`left_gate`, `right_gate`, `out_gate`) from `x`.
   Apply the left/right gates element‑wise to the masked projections.
5. Perform the cubic contraction
        out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   for every batch `b` and hidden channel `d`.
   This is a batched matrix multiplication (N×N)·(N×N)ᵀ per
   hidden channel and is implemented as a fused Triton kernel.
6. Layer‑normalize the result over `hidden_dim`, multiply by `out_gate`,
   and finally apply a linear projection back to `dim`.
All heavy arithmetic (step 5) is carried out in the custom Triton kernel;
the remaining cheap operations use PyTorch primitives.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def _trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_i, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Tri‐multiplicative update:
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]"""
    pid_i = tl.program_id(0)          # tile along i
    pid_j = tl.program_id(1)          # tile along j
    pid_bd = tl.program_id(2)         # combined batch * hidden‑dim index

    # ----- decode batch and hidden‑dim index -----
    b = pid_bd // D
    d = pid_bd % D

    # ----- compute absolute indices for this tile -----
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    i_idx = i_base + tl.arange(0, BLOCK_I)
    j_idx = j_base + tl.arange(0, BLOCK_J)

    i_valid = i_idx < N
    j_valid = j_idx < N

    # accumulator for the output tile (BLOCK_I × BLOCK_J)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ----- loop over the reduction dimension k -----
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        k_valid = k_idx < N

        # ----- load left tile  (i, k)  -----
        left_offset = (
            b * stride_l_b + d * stride_l_d
            + i_idx[:, None] * stride_l_i
            + k_idx[None, :] * stride_l_k
        )
        left_mask = i_valid[:, None] & k_valid[None, :]
        left_tile = tl.load(left_ptr + left_offset, mask=left_mask, other=0.0)

        # ----- load right tile transposed (k, j)  -----
        # we load as (k, j) so that tl.dot can treat it as (BLOCK_K, BLOCK_J)
        right_offset = (
            b * stride_r_b + d * stride_r_d
            + k_idx[:, None] * stride_r_k
            + j_idx[None, :] * stride_r_i
        )
        right_mask = k_valid[:, None] & j_valid[None, :]
        right_tile = tl.load(right_ptr + right_offset, mask=right_mask, other=0.0)

        # ----- matrix‑multiply the two sub‑tiles -----
        # left_tile : (BLOCK_I, BLOCK_K)
        # right_tile: (BLOCK_K, BLOCK_J)
        acc += tl.dot(left_tile, right_tile)

    # ----- write the result tile -----
    out_offset = (
        b * stride_o_b + d * stride_o_d
        + i_idx[:, None] * stride_o_i
        + j_idx[None, :] * stride_o_j
    )
    out_mask = i_valid[:, None] & j_valid[None, :]
    tl.store(out_ptr + out_offset, acc, mask=out_mask)


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul block.

    Args:
        data: (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim]
            - mask: torch.Tensor [B, N, N] (may be None or ignored when config["nomask"] is True)
            - weights: dict of torch.Tensors containing all model weights
            - config: dict with keys "dim", "hidden_dim", "nomask" (bool)

    Returns:
        torch.Tensor of shape [B, N, N, dim]
    """
    # -------------------------------------------------------------------------
    # unpack arguments
    # -------------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -------------------------------------------------------------------------
    # 1️⃣ Layer‑norm over the last dimension
    # -------------------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_w, bias=norm_b)

    # -------------------------------------------------------------------------
    # 2️⃣ Linear projections & gating
    # -------------------------------------------------------------------------
    # ---- projections ----
    left_proj_w = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_w = weights["right_proj.weight"]    # (hidden_dim, dim)

    left = F.linear(x, left_proj_w)    # [B, N, N, hidden_dim]
    right = F.linear(x, right_proj_w)  # [B, N, N, hidden_dim]

    # ---- optional mask (applied before gating) ----
    if not nomask:
        # mask may be bool or float; bring it to same dtype and broadcast
        mask = mask.to(dtype).unsqueeze(-1)          # [B, N, N, 1]
        left = left * mask
        right = right * mask

    # ---- gating ----
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(F.linear(x, left_gate_w))   # [B,N,N,hidden_dim]
    right_gate = torch.sigmoid(F.linear(x, right_gate_w))
    out_gate = torch.sigmoid(F.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # -------------------------------------------------------------------------
    # 3️⃣ Cubic contraction via Triton – the heavy part
    # -------------------------------------------------------------------------
    B, N, _, _ = left.shape
    # ensure contiguous layout for pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # [B, N, N, hidden_dim]

    # strides in number of elements (not bytes)
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_i, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # tuned tile sizes – work well on H100 for the target ranges
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
        B * hidden_dim,
    )

    _trimul_kernel[grid](
        left_ptr=left.data_ptr(),
        right_ptr=right.data_ptr(),
        out_ptr=out.data_ptr(),
        B=B,
        N=N,
        D=hidden_dim,
        stride_l_b=stride_l_b,
        stride_l_i=stride_l_i,
        stride_l_k=stride_l_k,
        stride_l_d=stride_l_d,
        stride_r_b=stride_r_b,
        stride_r_i=stride_r_i,
        stride_r_k=stride_r_k,
        stride_r_d=stride_r_d,
        stride_o_b=stride_o_b,
        stride_o_i=stride_o_i,
        stride_o_j=stride_o_j,
        stride_o_d=stride_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # -------------------------------------------------------------------------
    # 4️⃣ Post‑processing (LayerNorm, out‑gate, final projection)
    # -------------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    out = out * out_gate  # element‑wise gating

    to_out_w = weights["to_out.weight"]        # (dim, hidden_dim)
    output = F.linear(out, to_out_w)           # [B, N, N, dim]

    return output
