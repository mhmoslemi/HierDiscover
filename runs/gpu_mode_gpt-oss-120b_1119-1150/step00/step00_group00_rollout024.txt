    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)
    eps = 1e-5

    # LayerNorm on input
    # We define a helper function triton_layernorm

    # Projections
    # left_proj = F.linear(x_norm, weights["left_proj.weight"])

    right = torch.nn.functional.linear(x_norm, weights["right_proj.weight"])

    if not nomask and mask is not None:
        # mask shape [B,N,N]; unsqueeze to [B,N,N,1]; ensure same dtype (float)
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)  # broadcast
        left = left * mask_f
        right = right * mask_f
    else:
        # no mask: do nothing
        pass

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # compute out via batched matmul
    B, N, _, H = left.shape

    # permute and reshape for batched matmul
    left_perm = left.permute(0, 3, 1, 2).contiguous().reshape(B * H, N, N)
    right_perm = right.permute(0, 3, 2, 1).contiguous().reshape(B * H, N, N)
    out_b = torch.bmm(left_perm, right_perm)  # shape (B*H,N,N)
    out = out_b.reshape(B, H, N, N).permute(0, 2, 3, 1).contiguous()

    out_norm = triton_layernorm(out, weights["to_out_norm.weight"], weights["to_out_norm.bias"], eps)

    # gating
    out = out_norm * out_gate

    # final projection
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output

def layernorm_kernel(
    x_ptr, y_ptr,
    weight_ptr, bias_ptr,
    eps,
    B, N, C,
    stride_xb, stride_xi, stride_xj, stride_xc,
    stride_yb, stride_yi, stride_yj, stride_yc,
    stride_ww, stride_bb,
    BLOCK_SIZE: tl.constexpr):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # base pointers
    x_ptr_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_ptr_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # compute mean and var
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

    for offset in range(0, C, BLOCK_SIZE):
        offs = offset + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_ptr_base + offs * stride_xc, mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)
        sum_sq_val += tl.sum(x * x, axis=0)

    var = sum_sq_val / C - mean * mean

    for offset in range(0, C, BLOCK_SIZE):
        offs = offset + tl.arange(0, BLOCK_SIZE)

"""
TriMul (outgoing) implementation for AlphaFold3.
------------------------------------------------
The forward pass consists of three stages:

1. **Layer‑Norm on the input pair representation** – performed with a tiny
   Triton kernel that computes mean / variance over the last dimension and
   applies the learned scale and bias.

2. **Linear projections, gating and the pairwise multiplication** – realised
   with PyTorch `F.linear` (high‑performance cuBLAS kernels) and a batched
   matrix multiplication (`torch.bmm`).  The core “einsum” from the reference
   implementation becomes a batched GEMM over the sequence dimension.

3. **Output Layer‑Norm, gating and final projection** – the second Layer‑Norm
   is again executed by the Triton kernel, followed by an element‑wise
   multiplication with the output gate and a final linear projection.

Only the Layer‑Norm is written in Triton; the remaining heavy work is delegated
to PyTorch which already uses highly tuned kernels on the H100.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: Layer‑Norm over the last dimension (C)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, y_ptr,               # input / output tensors
    w_ptr, b_ptr,               # layer‑norm weight & bias (shape C)
    eps,                        # epsilon for numerical stability
    B, N, C,                    # B – batch, N – sequence length, C – channel dim (constexpr)
    stride_xb, stride_xi, stride_xj, stride_xc,   # strides for x
    stride_yb, stride_yi, stride_yj, stride_yc,   # strides for y
    stride_ww, stride_bb,                           # strides for weight / bias (normally 1)
    BLOCK_SIZE: tl.constexpr           # number of threads per program (>=32, power‑of‑2)
):
    pid = tl.program_id(0)
    total = B * N * N
    if pid >= total:
        return

    # ------------------------------------------------------------------
    # Decode (b, i, j) from linear program id
    # ------------------------------------------------------------------
    b = pid // (N * N)
    tmp = pid % (N * N)
    i = tmp // N
    j = tmp % N

    # Base pointers for this (b,i,j) entry
    x_base = x_ptr + b * stride_xb + i * stride_xi + j * stride_xj
    y_base = y_ptr + b * stride_yb + i * stride_yi + j * stride_yj

    # ------------------------------------------------------------------
    # First pass: compute mean & variance
    # ------------------------------------------------------------------
    sum_val = tl.float32(0.0)
    sum_sq_val = tl.float32(0.0)

    # Loop over the C dimension in BLOCK_SIZE‑chunks
    for off in range(0, C, BLOCK_SIZE):
        offs = off + tl.arange(0, BLOCK_SIZE)          # vector of offsets
        mask = offs < C                                 # guard for tail
        x = tl.load(x_base + offs * stride_xc,
                    mask=mask, other=0.0).to(tl.float32)
        sum_val += tl.sum(x, axis=0)                   # reduction across threads
        sum_sq_val += tl.sum(x * x, axis=0)

    C_f = tl.float32(C)
    mean = sum_val / C_f
    var = sum_sq_val / C_f - mean * mean
    inv_std = 1.0 / tl.sqrt(var + eps)

    # ------------------------------------------------------------------
    # Second pass: normalise, apply weight & bias, write out
    # ------------------------------------------------------------------
    for off in range(0, C, BLOCK_SIZE):
        offs = off + tl.arange(0, BLOCK_SIZE)
        mask = offs < C
        x = tl.load(x_base + offs * stride_xc,
                    mask=mask, other=0.0).to(tl.float32)
        w = tl.load(w_ptr + offs * stride_ww,
                    mask=mask, other=0.0).to(tl.float32)
        b_ = tl.load(b_ptr + offs * stride_bb,
                     mask=mask, other=0.0).to(tl.float32)

        y = (x - mean) * inv_std * w + b_
        tl.store(y_base + offs * stride_yc, y, mask=mask)


# ----------------------------------------------------------------------
# Helper: Triton based LayerNorm (float32)
# ----------------------------------------------------------------------
def triton_layernorm(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_sz: int = 128) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of ``x`` using Triton.
    All tensors must reside on the same CUDA device and be of dtype float32.
    """
    assert x.is_cuda and weight.is_cuda and bias.is_cuda
    B, N, _, C = x.shape
    y = torch.empty_like(x)

    # Strides (int64) – needed for pointer arithmetic inside the kernel
    sx = x.stride()
    sy = y.stride()
    sw = weight.stride(0)
    sb = bias.stride(0)

    # Grid: one program per (b,i,j) entry
    grid = (B * N * N,)

    layernorm_kernel[grid](
        x_ptr=x,
        y_ptr=y,
        w_ptr=weight,
        b_ptr=bias,
        eps=eps,
        B=B,
        N=N,
        C=C,
        stride_xb=sx[0],
        stride_xi=sx[1],
        stride_xj=sx[2],
        stride_xc=sx[3],
        stride_yb=sy[0],
        stride_yi=sy[1],
        stride_yj=sy[2],
        stride_yc=sy[3],
        stride_ww=sw,
        stride_bb=sb,
        BLOCK_SIZE=block_sz,
        num_warps=8,            # enough warps to keep the H100 busy
    )
    return y


# ----------------------------------------------------------------------
# Entry point required by the evaluator
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated TriMul (outgoing) forward pass.

    Args:
        data: tuple (input_tensor, mask_tensor, weights_dict, config_dict)
              - ``input_tensor`` : [B, S, S, C] float32
              - ``mask_tensor``  : [B, S, S] (may be None)
              - ``weights_dict`` : mapping of all model weights
              - ``config_dict``  : contains ``dim``, ``hidden_dim`` and optional ``nomask``
    Returns:
        output tensor of shape [B, S, S, dim] (float32)
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)          # True -> ignore mask

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm (Triton)
    # ------------------------------------------------------------------
    x_norm = triton_layernorm(
        input_tensor,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=1e-5,
        block_sz=128,
    )   # shape [B, S, S, C]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,S,S,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"]) # [B,S,S,hidden_dim]

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (broadcasted over the hidden dimension)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, S, S] → [B, S, S, 1] and cast to same dtype as the projections
        mask_f = mask.to(input_tensor.dtype).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid of linear layers)
    # ------------------------------------------------------------------
    left_gate = F.linear(x_norm, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x_norm, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x_norm, weights["out_gate.weight"]).sigmoid()

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core "TriMul" – batch matrix multiplication over the sequence axis
    # ------------------------------------------------------------------
    B, S, _, H = left.shape  # H == hidden_dim

    # reshape to (B*H, S, S) for a batched GEMM
    left_flat = left.permute(0, 3, 1, 2).contiguous().reshape(B * H, S, S)  # (i,k)
    # right needs its first two dimensions swapped before the GEMM (j,k)
    right_flat = right.permute(0, 3, 2, 1).contiguous().reshape(B * H, S, S)  # (k,j)

    # Batched matmul: (i,k) @ (k,j) → (i,j)
    out_flat = torch.bmm(left_flat, right_flat)                      # [B*H, S, S]
    out = out_flat.reshape(B, H, S, S).permute(0, 2, 3, 1).contiguous()  # [B,S,S,H]

    # ------------------------------------------------------------------
    # 6️⃣ Output LayerNorm (Triton) + output gate
    # ------------------------------------------------------------------
    out_norm = triton_layernorm(
        out,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=1e-5,
        block_sz=128,
    )
    out = out_norm * out_gate

    # ------------------------------------------------------------------
    # 7️⃣ Final linear projection back to original channel dimension
    # ------------------------------------------------------------------
    output = F.linear(out, weights["to_out.weight"])   # shape [B,S,S,dim]

    return output
