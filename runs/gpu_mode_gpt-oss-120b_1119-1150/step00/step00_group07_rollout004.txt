"""
TriMul (outgoing) forward pass – Triton‐accelerated version.

The operator works on a 4‑D tensor X ∈ ℝ^{B×N×N×C} :

1. Layer‑norm over the last dimension (C).
2. Linear projections → left, right ∈ ℝ^{B×N×N×H}
   (H = hidden_dim) together with three gating projections.
3. Multiply by an optional mask and by the sigmoid gates.
4. Core “tri‑mul”:
        out[b,i,j,h] = Σ_{k=0}^{N-1} left[b,i,k,h] * right[b,j,k,h]
   which is a batched matrix multiplication for each head h.
   This O(N³·H) contraction is performed by a custom Triton kernel.
5. Layer‑norm over H, gate with out_gate and a final linear projection
   back to C dimensions.

All heavy arithmetic (step 4) is done on the GPU via Triton, while the
surrounding tensor algebra uses PyTorch (layer‑norm, linear layers,
sigmoid, etc.). The implementation supports optional masking
(`nomask=False`) and works for the full range of the test configurations.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with the k‑dimension summed.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    # Pointers
    left_ptr, right_ptr, out_ptr,
    # Problem size
    N,
    # Strides for the three tensors (batch*head, i, k) and (batch*head, j, k)
    l_batch_stride, l_i_stride, l_k_stride,
    r_batch_stride, r_i_stride, r_k_stride,
    o_batch_stride, o_i_stride, o_j_stride,
    # Compile‑time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Computes for a given (batch*head) tile:
        out[i, j] = Σ_k left[i, k] * right_T[k, j]
    where right_T is the transpose of the inner‑dimension `k` of `right`.
    """
    pid = tl.program_id(0)

    # ------------------------------------------------------------------
    # Decode program ID → (batch_head, tile_i, tile_j)
    # ------------------------------------------------------------------
    num_i_tiles = tl.cdiv(N, BLOCK_I)
    num_j_tiles = tl.cdiv(N, BLOCK_J)
    tiles_per_batch = num_i_tiles * num_j_tiles

    batch_head = pid // tiles_per_batch
    tile_id    = pid - batch_head * tiles_per_batch
    tile_i     = tile_id // num_j_tiles
    tile_j     = tile_id - tile_i * num_j_tiles

    # ------------------------------------------------------------------
    # Compute absolute offsets for the current i‑ and j‑tiles
    # ------------------------------------------------------------------
    i_offset = tile_i * BLOCK_I + tl.arange(0, BLOCK_I)
    j_offset = tile_j * BLOCK_J + tl.arange(0, BLOCK_J)

    i_mask = i_offset < N
    j_mask = j_offset < N

    # accumulator in FP32 (the kernel works in float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension in blocks of size BLOCK_K
    # ------------------------------------------------------------------
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask   = k_offset < N

        # ----- left[i, k] -------------------------------------------------
        left_ptrs = (left_ptr
                     + batch_head * l_batch_stride
                     + i_offset[:, None] * l_i_stride
                     + k_offset[None, :] * l_k_stride)
        left = tl.load(left_ptrs,
                       mask=(i_mask[:, None] & k_mask[None, :]),
                       other=0.0)

        # ----- right_T[k, j]  (right transposed on the inner dimension) --
        #   right is stored as (batch_head, row=j, col=k)
        #   we need (k, j) → swap the role of i and k indices.
        right_T_ptrs = (right_ptr
                        + batch_head * r_batch_stride
                        + k_offset[:, None] * r_i_stride   # k plays the row role
                        + j_offset[None, :] * r_k_stride)  # j plays the column role
        right_T = tl.load(right_T_ptrs,
                          mask=(k_mask[:, None] & j_mask[None, :]),
                          other=0.0)

        # ----- matrix multiply for this k‑tile ----------------------------
        # left : (BLOCK_I, BLOCK_K)   , right_T : (BLOCK_K, BLOCK_J)
        acc += tl.dot(left, right_T)

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = (out_ptr
                + batch_head * o_batch_stride
                + i_offset[:, None] * o_i_stride
                + j_offset[None, :] * o_j_stride)
    tl.store(out_ptrs, acc, mask=(i_mask[:, None] & j_mask[None, :]))


# ----------------------------------------------------------------------
# Public entry point – the “custom kernel” used by the test harness.
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Args:
        data: Tuple (input_tensor, mask_tensor, weights_dict, config_dict)
    Returns:
        Tensor of shape [B, N, N, dim] – forward result of TriMul.
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim         = config["dim"]
    hidden_dim  = config["hidden_dim"]
    eps         = 1e-5  # LayerNorm epsilon – matches PyTorch default

    # ------------------------------------------------------------------
    # Load model weights (all are 2‑D except bias vectors)
    # ------------------------------------------------------------------
    norm_weight          = weights["norm.weight"]
    norm_bias            = weights["norm.bias"]
    left_proj_weight     = weights["left_proj.weight"]
    right_proj_weight    = weights["right_proj.weight"]
    left_gate_weight     = weights["left_gate.weight"]
    right_gate_weight    = weights["right_gate.weight"]
    out_gate_weight      = weights["out_gate.weight"]
    to_out_norm_weight   = weights["to_out_norm.weight"]
    to_out_norm_bias     = weights["to_out_norm.bias"]
    to_out_weight        = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the channel dimension (dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(input_tensor,
                     normalized_shape=(dim,),
                     weight=norm_weight,
                     bias=norm_bias,
                     eps=eps)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) and gating projections
    # ------------------------------------------------------------------
    left      = F.linear(x, left_proj_weight)          # [B,N,N,H]
    right     = F.linear(x, right_proj_weight)         # [B,N,N,H]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))

    # ------------------------------------------------------------------
    # 3. Apply optional mask and gates (elementwise)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask shape: [B, N, N]; expand to hidden dimension
        mask_f = mask.to(dtype).unsqueeze(-1)   # [B,N,N,1]
        left  = left * mask_f
        right = right * mask_f

    left  = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core TriMul (batched matrix multiplication over k)
    #    - reshape to (B*H, N, N) and run the Triton kernel
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Permute to (B, H, N, N) so that we can collapse B and H into a single batch axis
    left_perm  = left.permute(0, 3, 1, 2).contiguous()   # [B, H, N, N]
    right_perm = right.permute(0, 3, 1, 2).contiguous()  # [B, H, N, N]

    # Collapse the first two dimensions → (B*H, N, N)
    left_flat  = left_perm.view(-1, N, N)
    right_flat = right_perm.view(-1, N, N)

    # Output placeholder
    out_flat = torch.empty_like(left_flat)

    # Strides (in element units) for the flattened tensors
    l_batch_stride, l_i_stride, l_k_stride = left_flat.stride()
    r_batch_stride, r_i_stride, r_k_stride = right_flat.stride()
    o_batch_stride, o_i_stride, o_j_stride = out_flat.stride()

    # Block sizes – tuned for H100, can be adjusted if needed
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64

    # Number of tiles per (batch*head) slice
    num_i_tiles = (N + BLOCK_I - 1) // BLOCK_I
    num_j_tiles = (N + BLOCK_J - 1) // BLOCK_J
    total_tiles = (B * hidden_dim) * num_i_tiles * num_j_tiles

    # Launch Triton kernel (1‑D grid)
    tri_mul_kernel[total_tiles](
        left_flat,
        right_flat,
        out_flat,
        N,
        l_batch_stride, l_i_stride, l_k_stride,
        r_batch_stride, r_i_stride, r_k_stride,
        o_batch_stride, o_i_stride, o_j_stride,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,        # enough warps for this workload
    )

    # Reshape back to (B, N, N, H)
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm over hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    out = F.layer_norm(out,
                       normalized_shape=(hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias,
                       eps=eps)
    out = out * out_gate                     # apply the out gate
    out = F.linear(out, to_out_weight)       # [B,N,N,dim]

    return out
