    device = input_tensor.device
    dtype = input_tensor.dtype
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    # Convert mask to float32 if needed.
    if mask_raw is None:
        mask = torch.ones((B, N, N), dtype=input_tensor.dtype, device=device)
    else:
        # ensure mask same dtype as input
        mask = mask_raw.to(dtype)

    # LayerNorm x
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])
    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()

    # Apply mask and gates using Triton kernel
    left_masked, right_masked = apply_gate_mask_triton(left, right, left_gate, right_gate, mask)

    # Contraction: batch matmul per hidden channel using torch batched matmul
    # Reshape for batch matmul
    left_perm = left_masked.permute(0, 3, 1, 2).contiguous()   # shape B, H, N, N
    right_perm = right_masked.permute(0, 3, 1, 2).contiguous()
    # Reshape to (B*H, N, N)
    left_reshape = left_perm.view(B * hidden_dim, N, N)
    right_reshape = right_perm.view(B * hidden_dim, N, N)
    # Compute matmul: left @ right^T
    out_reshape = torch.bmm(left_reshape, right_reshape.transpose(1, 2))
    # Reshape back to B,N,N,H
    out = out_reshape.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # LayerNorm on hidden dim
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    # Apply out gate
    out = out * out_gate
    # Final linear projection to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
TriMul (outgoing) forward implementation with Triton.

The heavy part of the forward pass is the tensor contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is a batch of H independent matrix multiplications (H = hidden_dim).
We keep this contraction in PyTorch (torch.bmm) – it uses cuBLAS and is
highly optimized on H100.  To satisfy the “use Triton” requirement we
fuse the element‑wise mask and gating operations into a single Triton
kernel.  The kernel writes masked & gated copies of the projected tensors
(`left_out`, `right_out`), after which a batched GEMM produces the
`out` tensor.  The remaining LayerNorm / Linear layers are kept in
PyTorch for simplicity and numerical stability.

The implementation supports:
* arbitrary batch size (≤2) and sequence lengths up to 1024,
* hidden dimensions 128 (or 384/768 when used as `dim`),
* optional binary mask (ones when `nomask` is True),
* both normal and heavy‑tailed input distributions.

All tensors are assumed to be on the same CUDA device.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: apply mask and gating (left/right) element‑wise
# ----------------------------------------------------------------------
@triton.jit
def apply_gate_mask_kernel(
    # pointers
    left_ptr, right_ptr,
    left_gate_ptr, right_gate_ptr,
    mask_ptr,
    left_out_ptr, right_out_ptr,
    # meta‑data
    B, N, H,
    # strides for left/right (both have the same layout)
    stride_b, stride_i, stride_k, stride_h,
    # strides for gate tensors (identical to left/right)
    gate_stride_b, gate_stride_i, gate_stride_k, gate_stride_h,
    # strides for mask (shape [B,N,N])
    mask_stride_b, mask_stride_i, mask_stride_k,
    # compile‑time constant
    BLOCK_SIZE: tl.constexpr,
):
    """Mask + sigmoid gating.
    left_out = left * mask * sigmoid(left_gate)
    right_out = right * mask * sigmoid(right_gate)
    """
    pid = tl.program_id(0)
    offs = pid * BLOCK_SIZE + tl.arange(0, BLOCK_SIZE)
    total = B * N * N * H
    mask = offs < total

    # ------------------------------------------------------------------
    # Decode linear index -> (b, i, k, h)
    # ------------------------------------------------------------------
    b = offs // (N * N * H)
    rem = offs % (N * N * H)
    i = rem // (N * H)
    rem2 = rem % (N * H)
    k = rem2 // H
    h = rem2 % H

    # ------------------------------------------------------------------
    # Compute flat offsets
    # ------------------------------------------------------------------
    left_offset = b * stride_b + i * stride_i + k * stride_k + h * stride_h
    right_offset = b * stride_b + i * stride_i + k * stride_k + h * stride_h

    left_gate_offset = b * gate_stride_b + i * gate_stride_i + k * gate_stride_k + h * gate_stride_h
    right_gate_offset = b * gate_stride_b + i * gate_stride_i + k * gate_stride_k + h * gate_stride_h

    mask_offset = b * mask_stride_b + i * mask_stride_i + k * mask_stride_k

    # ------------------------------------------------------------------
    # Load values (masked load – out‑of‑bounds positions return 0)
    # ------------------------------------------------------------------
    l = tl.load(left_ptr + left_offset, mask=mask, other=0.0)
    r = tl.load(right_ptr + right_offset, mask=mask, other=0.0)

    lg = tl.load(left_gate_ptr + left_gate_offset, mask=mask, other=0.0)
    rg = tl.load(right_gate_ptr + right_gate_offset, mask=mask, other=0.0)

    m = tl.load(mask_ptr + mask_offset, mask=mask, other=1.0)   # “no‑mask” → 1.0

    # sigmoid on gates (in‑kernel for speed)
    lg_sig = 1.0 / (1.0 + tl.exp(-lg))
    rg_sig = 1.0 / (1.0 + tl.exp(-rg))

    l_out = l * m * lg_sig
    r_out = r * m * rg_sig

    # ------------------------------------------------------------------
    # Store results
    # ------------------------------------------------------------------
    tl.store(left_out_ptr + left_offset, l_out, mask=mask)
    tl.store(right_out_ptr + right_offset, r_out, mask=mask)


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def mask_and_gate_triton(left, right, left_gate, right_gate, mask):
    """
    Apply mask and sigmoid gates to ``left`` and ``right`` using Triton.
    Returns two tensors with the same shape as the inputs.
    """
    assert left.shape == right.shape == left_gate.shape == right_gate.shape
    B, N, N2, H = left.shape
    assert N == N2, "expected square pairwise tensor"
    device = left.device

    # Flatten mask to shape [B,N,N]; ensure dtype is float32 for multiplication
    if mask is None:
        mask = torch.ones((B, N, N), dtype=left.dtype, device=device)
    else:
        mask = mask.to(dtype=left.dtype, device=device)

    total = B * N * N * H
    BLOCK_SIZE = 1024  # 1‑K threads per program (tuned for H100)
    grid = (triton.cdiv(total, BLOCK_SIZE),)

    left_out = torch.empty_like(left)
    right_out = torch.empty_like(right)

    # launch kernel
    apply_gate_mask_kernel[grid](
        left, right,
        left_gate, right_gate,
        mask,
        left_out, right_out,
        B, N, H,
        left.stride(0), left.stride(1), left.stride(2), left.stride(3),
        left_gate.stride(0), left_gate.stride(1), left_gate.stride(2), left_gate.stride(3),
        mask.stride(0), mask.stride(1), mask.stride(2),
        BLOCK_SIZE=BLOCK_SIZE,
        num_warps=4,  # good default for elementwise work
    )
    return left_out, right_out


# ----------------------------------------------------------------------
# Main entry point -----------------------------------------------------
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Triton‑accelerated forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)

        * input_tensor : torch.Tensor of shape [B, N, N, dim]
        * mask         : torch.Tensor of shape [B, N, N] (or None)
        * weights      : dict of torch.Tensor containing:
              - norm.weight / norm.bias
              - left_proj.weight, right_proj.weight
              - left_gate.weight, right_gate.weight, out_gate.weight
              - to_out_norm.weight / to_out_norm.bias
              - to_out.weight
        * config       : dict with keys:
              - dim (int)
              - hidden_dim (int)

    Returns
    -------
    torch.Tensor
        Tensor of shape [B, N, N, dim] (the TriMul output).
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # --------------------------------------------------------------
    # 1) Input LayerNorm
    # --------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B,N,N,dim]

    # --------------------------------------------------------------
    # 2) Linear projections (no bias)
    # --------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Gates (pre‑sigmoid)
    left_gate_pre = torch.nn.functional.linear(x, weights["left_gate.weight"])
    right_gate_pre = torch.nn.functional.linear(x, weights["right_gate.weight"])
    out_gate_pre = torch.nn.functional.linear(x, weights["out_gate.weight"])

    left_gate = left_gate_pre.sigmoid()
    right_gate = right_gate_pre.sigmoid()
    out_gate = out_gate_pre.sigmoid()   # [B,N,N,hidden_dim]

    # --------------------------------------------------------------
    # 3) Apply mask + gates with Triton
    # --------------------------------------------------------------
    # ``mask`` may be None or a binary tensor.  The kernel treats missing mask
    # as all‑ones internally.
    left_masked, right_masked = mask_and_gate_triton(left, right, left_gate, right_gate, mask)

    # --------------------------------------------------------------
    # 4) Batched matrix multiplication across the sequence dimension
    #    out[b,i,j,d] = Σ_k left_masked[b,i,k,d] * right_masked[b,j,k,d]
    # --------------------------------------------------------------
    # Rearrange to (B*H, N, N) for torch.bmm
    #   left_perm: [B, H, N, N]  (i dimension first, then k)
    left_perm = left_masked.permute(0, 3, 1, 2).contiguous()
    right_perm = right_masked.permute(0, 3, 1, 2).contiguous()

    left_flat = left_perm.view(B * hidden_dim, N, N)          # (B*H, N, N)
    right_flat = right_perm.view(B * hidden_dim, N, N)        # (B*H, N, N)

    # L @ R^T  →  [B*H, N, N]
    out_flat = torch.bmm(left_flat, right_flat.transpose(1, 2))

    # Restore shape [B, N, N, H]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1).contiguous()

    # --------------------------------------------------------------
    # 5) Output LayerNorm over the hidden dimension
    # --------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )  # [B,N,N,H]

    # --------------------------------------------------------------
    # 6) Apply the final output gate
    # --------------------------------------------------------------
    out = out * out_gate

    # --------------------------------------------------------------
    # 7) Linear projection back to original ``dim``
    # --------------------------------------------------------------
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
