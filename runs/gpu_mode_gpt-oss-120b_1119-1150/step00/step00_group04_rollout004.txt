@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_n, stride_l_k,
    stride_r_m, stride_r_k, stride_r_n,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Get program ids
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)
    # compute offsets
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # define range for k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        # offsets within block
        k_base = k * BLOCK_K

        # Load left block
        left_i = i_base + tl.arange(0, BLOCK_I)
        left_k = k_base + tl.arange(0, BLOCK_K)
        # mask out-of-bounds for i,k
        left_mask_i = left_i < N
        left_mask_k = left_k < N

        left_ptrs = left_ptr + pid_m * stride_l_m \
            + left_i[:, None] * stride_l_n \
            + left_k[None, :] * stride_l_k

        left_block = tl.load(left_ptrs, mask=left_mask_i[:, None] & left_mask_k[None, :], other=0.0)

        # Load right block
        # right shape is (M, K, J)
        right_k = k_base + tl.arange(0, BLOCK_K)
        right_j = j_base + tl.arange(0, BLOCK_J)
        right_mask_k = right_k < N
        right_mask_j = right_j < N

        right_ptrs = right_ptr + pid_m * stride_r_m \
            + right_k[:, None] * stride_r_k \
            + right_j[None, :] * stride_r_n

        right_block = tl.load(right_ptrs, mask=right_mask_k[:, None] & right_mask_j[None, :], other=0.0)

        # Compute partial product
        # left_block shape (BLOCK_I, BLOCK_K)
        # right_block shape (BLOCK_K, BLOCK_J)
        acc += tl.dot(left_block, right_block)

    # Write back output
    # compute out pointers
    out_i = i_base + tl.arange(0, BLOCK_I)
    out_j = j_base + tl.arange(0, BLOCK_J)

    out_mask_i = out_i < N
    out_mask_j = out_j < N

    out_ptrs = out_ptr + pid_m * stride_o_m \
        + out_i[:, None] * stride_o_i \
        + out_j[None, :] * stride_o_j

    tl.store(out_ptrs, acc.to(tl.float16) if acc.dtype != tl.float16 else acc, mask=out_mask_i[:, None] & out_mask_j[None, :]) ?

def custom_kernel(data):
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # layernorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # more weight retrieval

    # x_norm
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, keepdim=True, unbiased=False)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left_proj = torch.nn.functional.linear(x_norm, left_proj_weight) # shape [B,N,N,hidden_dim]
    right_proj = torch.nn.functional.linear(x_norm, right_proj_weight)

    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B,N,N,1]
        left_proj = left_proj * mask_exp
        right_proj = right_proj * mask_exp

    # Gates
    left_gate = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left = left_proj * left_gate
    right = right_proj * right_gate

    # Rearrange for matmul
    B, N, _, _ = left.shape

    # Permute and reshape
    left_perm = left.permute(0, 3, 1, 2).contiguous()
    right_perm = right.permute(0, 3, 2, 1).contiguous()  # k before j

    # Flatten batch and hidden_dim
    M = B * hidden_dim
    left_flat = left_perm.view(M, N, N)
    right_flat = right_perm.view(M, N, N)

    # Allocate output
    out_flat = torch.empty_like(left_flat)

    # Triton kernel
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    grid = (M, (N + BLOCK_I - 1) // BLOCK_I, (N + BLOCK_J - 1) // BLOCK_J)
    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M,
        N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )
    # Reshape back
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)  # shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out_norm = (out - out.mean(dim=-1, keepdim=True)) / torch.sqrt(out.var(dim=-1, keepdim=True, unbiased=False) + eps)
    out_norm = out_norm * to_out_norm_weight + to_out_norm_bias
    out_norm = out_norm * out_gate

    # final projection
    result = torch.nn.functional.linear(out_norm, to_out_weight)

    return result

def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,
    M, N,
    stride_l_m, stride_l_n, stride_l_k,
    stride_r_m, stride_r_k, stride_r_n,
    stride_o_m, stride_o_i, stride_o_j,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    pid_m = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # Compute block offsets
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # Compute indexes for left (M,I,K)
        i_idx = i_start + tl.arange(0, BLOCK_I)
        k_idx = k_start + tl.arange(0, BLOCK_K)

        # mask out of bounds for i and k
        i_mask = i_idx < N
        k_mask = k_idx < N

        # Load left block

        left = tl.load(left_ptrs, mask=i_mask[:, None] & k_mask[None, :], other=0.0)

        # Compute indexes for right (M,K,J)
        j_idx = j_start + tl.arange(0, BLOCK_J)

        right_ptrs = right_ptr + pid_m * stride_r_m + k_idx[:, None] * stride_r_k + j_idx[None, :] * stride_r_n

    # Write out result
    out_i = i_start + tl.arange(0, BLOCK_I)
    out_j = j_start + tl.arange(0, BLOCK_J)
    i_out_mask = out_i < N
    j_out_mask = out_j < N

"""
custom_kernel for the “outgoing” TriMul operation (AlphaFold‑3).

Algorithm
---------
1. Layer‑norm the input tensor `x` over the channel dimension `dim`.
2. Linear projections   left = x·W_left   ,   right = x·W_right
3. Optional masking of the projected tensors.
4. Compute sigmoid‑gates from `x` and apply them:
        left  = left  * sigmoid(x·W_left_gate)
        right = right * sigmoid(x·W_right_gate)
5. Compute the bilinear reduction
        out[i,j] = Σ_k left[i,k] * right[j,k]
   for every batch and every hidden dimension.  This is a batched
   matrix‑multiply with shapes (M, N, N) × (M, N, N) → (M, N, N) where
   M = batch_size * hidden_dim.  The heavy work is performed by a
   Triton kernel that tiles the `i`, `j` and `k` axes.
6. Reshape the result back to [B, N, N, hidden_dim] and apply a second
   LayerNorm over the hidden dimension.
7. Apply the output gate (sigmoid) and a final linear projection back
   to `dim`.

The kernel works with any batch size (1‑2), sequence length up to 1024
and hidden dimension up to 128 (as required by the tests).  It uses
float32 accumulators (the inputs are float32) and stores the final
output in float32.  The rest of the operations are performed with
native PyTorch kernels, which are very fast for the remaining
element‑wise work.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched mat‑mul for the TriMul reduction.
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_matmul_kernel(
    left_ptr, right_ptr, out_ptr,          # base pointers
    M, N,                                 # M = B*hidden_dim, N = seq_len
    stride_l_m, stride_l_i, stride_l_k,   # left strides  (M, I, K)
    stride_r_m, stride_r_k, stride_r_j,   # right strides (M, K, J)
    stride_o_m, stride_o_i, stride_o_j,   # out strides   (M, I, J)
    BLOCK_I: tl.constexpr,
    BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_m = tl.program_id(0)          # batch*head index
    pid_i = tl.program_id(1)          # i‑tile index
    pid_j = tl.program_id(2)          # j‑tile index

    # Tile origins
    i_start = pid_i * BLOCK_I
    j_start = pid_j * BLOCK_J

    # Accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Loop over K dimension in tiles
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k * BLOCK_K

        # ---- load left block (M, I, K) ----
        i_idx = i_start + tl.arange(0, BLOCK_I)
        k_idx = k_start + tl.arange(0, BLOCK_K)

        i_mask = i_idx < N
        k_mask = k_idx < N

        left_ptrs = left_ptr \
            + pid_m * stride_l_m \
            + i_idx[:, None] * stride_l_i \
            + k_idx[None, :] * stride_l_k

        left = tl.load(left_ptrs,
                       mask=i_mask[:, None] & k_mask[None, :],
                       other=0.0)

        # ---- load right block (M, K, J) ----
        j_idx = j_start + tl.arange(0, BLOCK_J)

        j_mask = j_idx < N

        right_ptrs = right_ptr \
            + pid_m * stride_r_m \
            + k_idx[:, None] * stride_r_k \
            + j_idx[None, :] * stride_r_j

        right = tl.load(right_ptrs,
                        mask=k_mask[:, None] & j_mask[None, :],
                        other=0.0)

        # ---- accumulate dot product ----
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J)

    # ---- write tile to output ----
    i_out = i_start + tl.arange(0, BLOCK_I)
    j_out = j_start + tl.arange(0, BLOCK_J)

    i_out_mask = i_out < N
    j_out_mask = j_out < N

    out_ptrs = out_ptr \
        + pid_m * stride_o_m \
        + i_out[:, None] * stride_o_i \
        + j_out[None, :] * stride_o_j

    tl.store(out_ptrs,
             acc,
             mask=i_out_mask[:, None] & j_out_mask[None, :])


# ----------------------------------------------------------------------
# Python entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Output tensor of shape [bs, seq_len, seq_len, dim] (float32).
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    B, N, _, dim = x.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # extract weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]

    left_proj_weight  = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]

    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1) Layer‑norm over the channel dimension `dim`
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )   # shape [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2) Linear projections
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3) Optional mask (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        # cast mask to float32 and broadcast
        mask_f = mask.to(x.dtype).unsqueeze(-1)   # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gates (sigmoid)
    # ------------------------------------------------------------------
    left_gate  = torch.nn.functional.linear(x_norm, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_norm, right_gate_weight).sigmoid()
    out_gate   = torch.nn.functional.linear(x_norm, out_gate_weight).sigmoid()

    left  = left  * left_gate   # [B,N,N,hidden_dim]
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Bilinear reduction via Triton (einsum over k)
    # ------------------------------------------------------------------
    # reshape for batched GEMM: (B, H, N, N) → (M = B*H, N, N)
    left_t = left.permute(0, 3, 1, 2).contiguous()   # (B, H, i, k)
    right_t = right.permute(0, 3, 2, 1).contiguous() # (B, H, k, j)

    M = B * hidden_dim
    left_flat  = left_t.view(M, N, N)   # (M, I, K)
    right_flat = right_t.view(M, N, N)  # (M, K, J)
    out_flat   = torch.empty_like(left_flat)   # (M, I, J)

    # launch Triton kernel
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 64
    grid = (M,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    tri_mul_matmul_kernel[grid](
        left_flat,
        right_flat,
        out_flat,
        M, N,
        left_flat.stride(0), left_flat.stride(1), left_flat.stride(2),
        right_flat.stride(0), right_flat.stride(1), right_flat.stride(2),
        out_flat.stride(0), out_flat.stride(1), out_flat.stride(2),
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
    )

    # reshape back to [B, N, N, hidden_dim]
    out = out_flat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------------
    # 6) Second LayerNorm over hidden_dim
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )

    # ------------------------------------------------------------------
    # 7) Output gating and final linear projection
    # ------------------------------------------------------------------
    out = out * out_gate
    out = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
```
