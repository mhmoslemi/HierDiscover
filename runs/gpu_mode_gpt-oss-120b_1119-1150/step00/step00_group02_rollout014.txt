import triton
import triton.language as tl

@triton.jit
def tr_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    b_h = tl.program_id(0)   # combine batch and hidden
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # calculate batch and hidden from combined index
    b = b_h // H
    h = b_h % H

    # compute start indices
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # mask for out-of-bound on i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # offsets for output
    out_off = b * stride_out_b + (i_offset[:, None] * stride_out_i) + (j_offset[None, :] * stride_out_j) + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k_offset = tl.arange(0, BLOCK_K) + k_start
        mask_k = k_offset < N

        # load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + (i_offset[:, None] * stride_left_i) + (k_offset[None, :] * stride_left_k) + h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # load right tile transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b * stride_right_b + (j_offset[None, :] * stride_right_i) + (k_offset[:, None] * stride_right_k) + h * stride_right_h
        # This yields shape (BLOCK_K, BLOCK_N) when loaded with tl.load
        # However tl.load expects shape consistent with offsets shape; we compute offsets of shape (BLOCK_K, BLOCK_N)
        right = tl.load(right_ptrs, mask=mask_j[None, :] & mask_k[:, None], other=0.0).to(tl.float32)

        # right has shape (BLOCK_K, BLOCK_N). We need shape (BLOCK_K, BLOCK_N) to be used as B in dot (K,N)
        # left shape (BLOCK_M, BLOCK_K). So we can compute dot
        acc += tl.dot(left, right)

    # store result
    tl.store(out_off, acc, mask=mask_i[:, None] & mask_j[None, :])

```
Define the kernel:

```
@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    # program ids: 0 -> batch_hidden, 1->row tile, 2->col tile
    pid_batch_hidden = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute batch and hidden index
    b = pid_batch_hidden // H
    h = pid_batch_hidden % H

    # offsets for i and j within this tile
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks
    mask_i = i < N
    mask_j = j < N

    out_ptrs = out_ptr + b * stride_out_b + (i[:, None] * stride_out_i) + (j[None, :] * stride_out_j) + h * stride_out_h

    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # iterate over k tiles
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + (i[:, None] * stride_left_i) + (k[None, :] * stride_left_k) + h * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile transposed: shape (BLOCK_K, BLOCK_N)
        right_ptrs = right_ptr + b * stride_right_b + (j[None, :] * stride_right_i) + (k[:, None] * stride_right_k) + h * stride_right_h
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Multiply-accumulate
        acc += tl.dot(left, right)

    # Store result
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

Now we need to define wrapper:

```
def tr_mul_triton(left, right):
    """
    Compute out[b,i,j,h] = sum_k left[b,i,k,h] * right[b,j,k,h] using Triton.
    left/right: torch.Tensor of shape (B,N,N,H) and dtype float32.
    Returns torch.Tensor of same shape and dtype.
    """
    assert left.is_contiguous() and right.is_contiguous()
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # strides in elements
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # define block sizes (tunable)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (B * H, (N + BLOCK_M - 1) // BLOCK_M, (N + BLOCK_N - 1) // BLOCK_N)

    tri_mul_kernel[grid](
        left, right, out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        num_warps=8,
    )
    return out
```

```
def custom_kernel(data):
    """
    Triton-accelerated forward pass for the outgoing TriMul operation.
    Arguments:
        data: tuple (input_tensor, mask, weights, config)
    Returns:
        output tensor of shape [bs, seq_len, seq_len, dim].
    """
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]

    device = input_tensor.device
    dtype = input_tensor.dtype

    # LayerNorm on dim
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    # Normalize over last dimension (dim)
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Linear projections
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # Mask handling
    nomask = config.get("nomask", False)
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gates
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # Core multiplicative update using Triton
    out = tr_mul_triton(left, right)  # shape [B,N,N,hidden_dim]

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)

    # out_gate gating
    out = out * out_gate

    # final linear projection to dim
    output = F.linear(out, weights["to_out.weight"])
    return output

"""
TriMul (outgoing) – Triton‑accelerated forward pass

Given an input tensor x of shape [B, N, N, C] the module performs:
  1. LayerNorm on the channel dimension C.
  2. Two linear projections (left/right) from C → H (hidden_dim) without bias.
  3. Optional masking on the pair‑wise dimension.
  4. Three sigmoid gates (left, right, out) – each a C → H linear.
  5. Multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
     which is a batched matrix‑multiply of left (i×k) with rightᵀ (j×k).
     This O(N³·H) kernel is written in Triton, using half‑precision
     loads and tensor‑core fused dot products (fp16×fp16 → fp32).
  6. LayerNorm on the hidden dimension, gating with out_gate,
     and a final linear projection H → C.

The heavy *einsum* step (step 5) is off‑loaded to the Triton kernel
`tri_mul_kernel`.  All other operations are performed with PyTorch
functions (they are cheap compared to the N³ term).  The implementation
uses mixed precision (fp16 for the big tensor work) while the final
output is in fp32, matching the reference PyTorch module.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the core multiplicative update
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # *const half* [B, N, N, H]
    right_ptr,           # *const half* [B, N, N, H]
    out_ptr,             # *float32*   [B, N, N, H] – accumulator output
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """
    Compute:
        out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

    The kernel iterates over k‑tiles and performs a fused dot product
    (fp16×fp16 → fp32) for each hidden slice `h`.  Batches and hidden
    channels are fused into program‑id(0) to keep the launch dimensionality
    low.
    """

    # ------------------------------------------------------------------
    # 1) Identify which batch and hidden channel this program works on.
    # ------------------------------------------------------------------
    pid0 = tl.program_id(0)      # combined batch * hidden
    pid_i = tl.program_id(1)      # tile row   (i dimension)
    pid_j = tl.program_id(2)      # tile col   (j dimension)

    b = pid0 // H                 # batch index
    h = pid0 % H                  # hidden index

    # ------------------------------------------------------------------
    # 2) Compute start indices for the i/j tiles.
    # ------------------------------------------------------------------
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # 3) Prepare pointer to the output tile.
    # ------------------------------------------------------------------
    out_off = (
        b * stride_out_b
        + (i[:, None] * stride_out_i)
        + (j[None, :] * stride_out_j)
        + h * stride_out_h
    )
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # 4) Loop over k‑tiles.
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(num_k_tiles):
        k_start = k_tile * BLOCK_K
        k = k_start + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # left[b, i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        left_off = (
            b * stride_left_b
            + (i[:, None] * stride_left_i)
            + (k[None, :] * stride_left_k)
            + h * stride_left_h
        )
        left = tl.load(left_off,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0,
                       dtype=tl.float16)

        # right[b, j, k, h] transposed => shape (BLOCK_K, BLOCK_N)
        # i.e. we load right as (k, j) so that a dot product can be performed.
        right_off = (
            b * stride_right_b
            + (j[None, :] * stride_right_i)   # j index on the second‑dim stride
            + (k[:, None] * stride_right_k)   # k index on the third‑dim stride
            + h * stride_right_h
        )
        right = tl.load(right_off,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0,
                        dtype=tl.float16)

        # fused mul‑add: fp16×fp16 → fp32 accumulation
        acc += tl.dot(left, right)   # (M,K)·(K,N) → (M,N)

    # ------------------------------------------------------------------
    # 5) Write the result.
    # ------------------------------------------------------------------
    tl.store(out_off, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Batched multiplicative update:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]

    Arguments:
        left, right : torch tensors of shape (B, N, N, H) and dtype torch.float16
    Returns:
        out : torch.float32 tensor of the same shape.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    # Output tensor in float32 (accumulation precision)
    out = torch.empty_like(left, dtype=torch.float32)

    # Strides in *elements* (not bytes)
    stride_left = left.stride()
    stride_right = right.stride()
    stride_out = out.stride()

    # Tune these block sizes for the H100 – they fit well in shared memory
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        B * H,                                 # program_id(0)
        (N + BLOCK_M - 1) // BLOCK_M,          # program_id(1) – i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # program_id(2) – j‑tiles
    )

    tri_mul_kernel[grid](
        left,
        right,
        out,
        B, N, H,
        stride_left[0], stride_left[1], stride_left[2], stride_left[3],
        stride_right[0], stride_right[1], stride_right[2], stride_right[3],
        stride_out[0], stride_out[1], stride_out[2], stride_out[3],
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=8,      # H100 thrives with 8‑warps per block
    )
    return out


# ----------------------------------------------------------------------
# Entry point – the required custom_kernel function
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operation (AlphaFold3 style).

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor, shape [B, N, N, C] (float32)
        - mask         : torch.Tensor or None, shape [B, N, N] (bool/float)
        - weights      : dict of torch.Tensors containing all linear / LN weights
        - config       : dict with keys:
                "dim"        – input channel dimension C
                "hidden_dim" – hidden dimension H
                "nomask"     – True if the mask should be ignored

    Returns
    -------
    torch.Tensor
        Output of shape [B, N, N, C] (float32).
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1) Input LayerNorm (float32 for numerical stability)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # Cast to FP16 for the heavy part of the computation
    x = x.to(torch.float16)

    # ------------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"].to(torch.float16))
    right = F.linear(x, weights["right_proj.weight"].to(torch.float16))

    # ------------------------------------------------------------------
    # 3) Optional mask (applied after projection, before gating)
    # ------------------------------------------------------------------
    if not nomask and mask is not None:
        # mask: [B, N, N] -> [B, N, N, 1] and cast to half
        mask_f = mask.to(torch.float16).unsqueeze(-1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4) Gating networks (sigmoid after linear)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        F.linear(x, weights["left_gate.weight"].to(torch.float16))
    )
    right_gate = torch.sigmoid(
        F.linear(x, weights["right_gate.weight"].to(torch.float16))
    )
    out_gate = torch.sigmoid(
        F.linear(x, weights["out_gate.weight"].to(torch.float16))
    )

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5) Core multiplicative update using Triton
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for Triton pointer arithmetic
    left = left.contiguous()
    right = right.contiguous()
    out_hidden = tri_mul_triton(left, right)               # float32, shape [B,N,N,H]

    # ------------------------------------------------------------------
    # 6) Post‑processing: LayerNorm over hidden, gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = F.layer_norm(out_hidden,
                       (hidden_dim,),
                       weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    # out_gate was computed in FP16 – cast to FP32 for multiplication
    out = out * out_gate.to(torch.float32)

    # Final projection back to the original dimension (float32)
    output = F.linear(out, weights["to_out.weight"])

    return output
