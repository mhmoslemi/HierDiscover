"""
custom_kernel: Triton‑accelerated “outgoing” TriMul (Triangular Multiplicative Update).

Algorithm
---------
1. Layer‑norm the 4‑D tensor `x` over the last dimension.
2. Linear projections → `left`, `right` (shape [B,N,N,H]).
   Apply optional binary mask (broadcasted over the channel axis).
   Compute sigmoid gates from separate linear layers and fuse:
        left  = left  * left_gate  * mask
        right = right * right_gate * mask
3. Core O(N³·H) operation:
        out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   Implemented as a batched half‑precision matrix multiply
   `left @ rightᵀ` using a custom Triton kernel.  Each (b,h) pair is a
   separate program‑id dimension, while (i,j) tiling is expressed with the
   first two program‑ids.
4. Convert the half‑precision result back to fp32.
5. Layer‑norm the intermediate tensor (hidden dimension H),
   multiply with the output gate, and project back to `dim`.
6. Return the final tensor `[B, N, N, dim]` (float32).

The kernel fuses the K‑loop of the matrix multiply and handles out‑of‑bounds
tiles with masking, enabling high utilisation of H100 Tensor‑Cores.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


@triton.jit
def trmul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_lb, stride_li, stride_lk,
    stride_rb, stride_ri, stride_rk,
    stride_ob, stride_oi, stride_oj,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    """Batched matmul: out = left @ rightᵀ  (per batch & hidden channel)."""
    pid_i = tl.program_id(0)          # tile row   (i)
    pid_j = tl.program_id(1)          # tile col   (j)
    pid_bh = tl.program_id(2)         # combined batch & hidden channel

    batch = pid_bh // H                # batch index
    chan  = pid_bh % H                 # hidden channel index

    # Base pointers for this (batch, channel)
    left_base  = left_ptr  + batch * stride_lb + chan
    right_base = right_ptr + batch * stride_rb + chan
    out_base   = out_ptr   + batch * stride_ob + chan

    # Tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Out‑of‑bounds masks for i and j
    i_mask = i_offset < N
    j_mask = j_offset < N

    # Accumulator in fp32 for numerical stability
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Number of k‑tiles
    num_k = tl.cdiv(N, BLOCK_K)

    for kk in range(num_k):
        k_offset = kk * BLOCK_K + tl.arange(0, BLOCK_K)
        k_mask = k_offset < N

        # ----- Load left tile (M×K) -----
        # Layout: left[b, i, k, h]  -> strides (lb, li, lk, 1)
        left_ptrs = left_base + i_offset[:, None] * stride_li + k_offset[None, :] * stride_lk
        mask_left = i_mask[:, None] & k_mask[None, :]
        a = tl.load(left_ptrs, mask=mask_left, other=0.0)

        # ----- Load right tile transposed (K×N) -----
        # We need rightᵀ, i.e. right[b, j, k, h] accessed as (k, j)
        right_ptrs = right_base + k_offset[:, None] * stride_rk + j_offset[None, :] * stride_ri
        mask_right = k_mask[:, None] & j_mask[None, :]
        b = tl.load(right_ptrs, mask=mask_right, other=0.0)

        # Cast to fp32 and accumulate
        acc += tl.dot(a.to(tl.float32), b.to(tl.float32))

    # ----- Store result (convert back to fp16) -----
    out_ptrs = out_base + i_offset[:, None] * stride_oi + j_offset[None, :] * stride_oj
    mask_out = i_mask[:, None] & j_mask[None, :]
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_out)

    norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1. LayerNorm over the feature dimension (`dim`)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )  # [B, N, N, dim]  fp32

    # ------------------------------------------------------------------
    # 2. Linear projections + optional mask
    # ------------------------------------------------------------------
    # Linear layers (bias‑free)
    left = F.linear(x, left_proj_weight)    # [B, N, N, H]
    right = F.linear(x, right_proj_weight)  # [B, N, N, H]

    if not nomask:
        # mask: [B,N,N] -> [B,N,N,1] for broadcasting
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid) and elementwise multiplication
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³·H) contraction via Triton
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape

    # Convert to fp16 for Tensor‑Core speed
    left_h = left.to(torch.float16).contiguous()
    right_h = right.to(torch.float16).contiguous()
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left_h.device)

    # Strides in elements (as used by Triton)
    stride_lb, stride_li, stride_lk, stride_lc = left_h.stride()
    stride_rb, stride_ri, stride_rk, stride_rc = right_h.stride()
    stride_ob, stride_oi, stride_oj, stride_oc = out_h.stride()

    # Tile sizes – chosen to fit H100 Tensor‑Core resources
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * hidden_dim,               # (batch, hidden) programs
    )

    trmul_outgoing_kernel[grid](
        left_h,
        right_h,
        out_h,
        B,
        N,
        hidden_dim,
        stride_lb,
        stride_li,
        stride_lk,
        stride_rb,
        stride_ri,
        stride_rk,
        stride_ob,
        stride_oi,
        stride_oj,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )
    # Ensure kernel completion
    torch.cuda.synchronize()

    # Convert back to fp32 for the remaining lightweight ops
    out = out_h.to(torch.float32)

    # ------------------------------------------------------------------
    # 5. Post‑processing: LayerNorm → gate → final linear projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out = out * out_gate
    out = F.linear(out, to_out_weight)   # [B, N, N, dim]

    return out
