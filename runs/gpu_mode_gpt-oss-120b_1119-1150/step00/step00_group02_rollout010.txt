@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0) # block row i dimension
    pid_j = tl.program_id(1) # block col j dimension
    pid_bh = tl.program_id(2) # batch*hidden dimension

    # Compute batch and hidden indices
    b = pid_bh // H
    h = pid_bh % H

    # Compute start row/col offsets
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # boundary checks
    i_mask = i_base + tl.arange(0, BLOCK_M) < N
    j_mask = j_base + tl.arange(0, BLOCK_N) < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        # Compute offsets for k blocks
        k_base = k_iter * BLOCK_K

        # Load left tile (i,k)
        offs_i = i_base + tl.arange(0, BLOCK_M)
        offs_k = k_base + tl.arange(0, BLOCK_K)
        # apply mask
        mask_i = offs_i < N
        mask_k = offs_k < N

        left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)
        # left shape (BLOCK_M, BLOCK_K)

        # Load right tile transposed (k,j)
        # We want right[j,k] but load as (k,j) by swapping strides.
        # pointer offset: b*stride_right_b + h*stride_right_h + offs_k[:, None] * stride_right_j + offs_j[None, :] * stride_right_k
        offs_j = j_base + tl.arange(0, BLOCK_N)
        mask_j = offs_j < N

        right_ptrs_T = right_ptr + b * stride_right_b + h * stride_right_h + offs_k[:, None] * stride_right_j + offs_j[None, :] * stride_right_k
        right_T = tl.load(right_ptrs_T, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)
        # right_T shape (BLOCK_K, BLOCK_N)

        # Compute acc += left @ right_T
        acc += tl.dot(left, right_T)

    # Write result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i_base[:, None] * stride_out_i + j_base + tl.arange(0, BLOCK_N)[None, :] * stride_out_j
    # the above may need to incorporate stride_out_j and stride_out_i correctly.

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)

    # compute batch idx and hidden idx
    b = pid_bh // H
    h = pid_bh % H

    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    k = tl.arange(0, BLOCK_K)

    # bounds mask
    mask_i = i < N
    mask_j = j < N
    mask_k = k < N

    # pointers for base: left has shape [B,N,N,H]; left[i,k] for given batch b, hidden h
    # left_ptrs shape (BLOCK_M, BLOCK_K)
    left_ptrs = left_ptr + b * stride_left_b + h * stride_left_h + i[:, None] * stride_left_i + k[None, :] * stride_left_k
    # right transposed pointers: we need (k,j)
    right_ptrs_T = right_ptr + b * stride_right_b + h * stride_right_h + k[:, None] * stride_right_j + j[None, :] * stride_right_k

    # accumulate
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loops over k blocks
    # compute number of k blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(num_k_blocks):
        # compute k offset for this block
        k_base = kb * BLOCK_K
        cur_k = k_base + k
        # apply size mask
        cur_mask_k = cur_k < N

        # load left subblock
        left_blk = tl.load(left_ptr + b * stride_left_b + h * stride_left_h + i[:, None] * stride_left_i + cur_k[None, :] * stride_left_k,
                         mask=mask_i[:, None] & cur_mask_k[None, :], other=0.0).to(tl.float32)

        # load right transposed subblock
        right_blk_T = tl.load(right_ptr + b * stride_right_b + h * stride_right_h + cur_k[:, None] * stride_right_j + j[None, :] * stride_right_k,
                              mask=cur_mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # dot product accumulate: left_blk (M,K) * right_blk_T (K,N) -> (M,N)
        acc += tl.dot(left_blk, right_blk_T)

    # store result
    out_ptrs = out_ptr + b * stride_out_b + h * stride_out_h + i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

"""
TriMul (outgoing) forward implementation.

The operation consists of:
1. LayerNorm over the last dimension of the pairwise input tensor.
2. Two linear projections (left/right) to a hidden dimension.
3. Optional pairwise mask application.
4. Gating (sigmoid) applied element‑wise to the projected tensors.
5. A heavy O(N³) reduction:
   out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
   which is equivalent to a per‑hidden‑channel batched matrix multiplication
   left @ rightᵀ.  This reduction is performed with a custom Triton kernel
   that tiles the i and j dimensions and iterates over k.
6. LayerNorm over the hidden dimension, multiplication by an output gate,
   and a final linear projection back to the original dimension.

Only the reduction step is written in Triton; the surrounding linear /
layer‑norm / sigmoid operations use PyTorch for simplicity.
"""

import torch
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: per‑batch‑hidden‑channel tiled matmul (left @ rightᵀ)
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,                     # *Pointer to left  [B,N,N,H] (float32)
    right_ptr,                    # *Pointer to right [B,N,N,H] (float32)
    out_ptr,                      # *Pointer to output [B,N,N,H] (float32)
    B, N, H,                      # runtime constants
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """
    Compute out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
    using a classic tiled GEMM where the right matrix is accessed in transposed
    order to avoid explicit transposition.
    """
    pid_i = tl.program_id(0)          # block row   (i)
    pid_j = tl.program_id(1)          # block col   (j)
    pid_bh = tl.program_id(2)         # combined batch & hidden index

    # ------------------------------------------------------------------
    # Decode batch and hidden indices from the third dimension
    # ------------------------------------------------------------------
    b = pid_bh // H
    h = pid_bh % H

    # ------------------------------------------------------------------
    # Starting coordinates of the tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Index vectors for the current tile (i, j)
    # ------------------------------------------------------------------
    i = i_base + tl.arange(0, BLOCK_M)
    j = j_base + tl.arange(0, BLOCK_N)

    # Boundary masks for i / j (required for the final store)
    mask_i = i < N
    mask_j = j < N

    # ------------------------------------------------------------------
    # Accumulator (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over the k‑dimension in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)
        mask_k = k < N

        # ------------------------------------------------------------------
        # Load left[i, k, h]  -> shape (BLOCK_M, BLOCK_K)
        # ------------------------------------------------------------------
        left_ptrs = left_ptr + \
            b * stride_left_b + h * stride_left_h + \
            i[:, None] * stride_left_i + k[None, :] * stride_left_k
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0,
        ).to(tl.float32)                     # ensure fp32 accumulation

        # ------------------------------------------------------------------
        # Load right[j, k, h] in transposed order (k, j) -> shape (BLOCK_K, BLOCK_N)
        # The original layout is (b, j, k, h).  Swapping the strides for j and k
        # yields the transposed view without materialising it.
        # ------------------------------------------------------------------
        right_ptrs_T = right_ptr + \
            b * stride_right_b + h * stride_right_h + \
            k[:, None] * stride_right_k + j[None, :] * stride_right_i
        right_T = tl.load(
            right_ptrs_T,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0,
        ).to(tl.float32)

        # ------------------------------------------------------------------
        # Accumulate: (M,K) @ (K,N)  -> (M,N)
        # ------------------------------------------------------------------
        acc += tl.dot(left, right_T)

    # ------------------------------------------------------------------
    # Store the result tile, respecting the edge masks
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        b * stride_out_b + h * stride_out_h + \
        i[:, None] * stride_out_i + j[None, :] * stride_out_j
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Python entry point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor, shape (B, N, N, dim)
            - mask: torch.Tensor or None, shape (B, N, N) (bool / float)
            - weights: dict of parameter tensors
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        torch.Tensor of shape (B, N, N, dim)
    """
    # ------------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # Helper to fetch a weight (may already be on the correct device)
    # ------------------------------------------------------------------
    def get(name):
        return weights[name]

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = get("norm.weight")
    norm_bias = get("norm.bias")
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=1e-5,
    )

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections to hidden_dim (bias=False in the reference)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, get("left_proj.weight"))
    right = torch.nn.functional.linear(x, get("right_proj.weight"))

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (the mask is broadcast over the hidden dimension)
    # ------------------------------------------------------------------
    if not config.get("nomask", False) and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)               # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate = torch.nn.functional.linear(x, get("left_gate.weight")).sigmoid()
    right_gate = torch.nn.functional.linear(x, get("right_gate.weight")).sigmoid()
    out_gate = torch.nn.functional.linear(x, get("out_gate.weight")).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Core TriMul reduction – Triton kernel
    # ------------------------------------------------------------------
    # Ensure the tensors are contiguous and in float32 for the kernel.
    left = left.contiguous().to(torch.float32)
    right = right.contiguous().to(torch.float32)

    # Allocate output tensor for the hidden intermediate.
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Strides in element units (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out_hidden.stride()

    # Tiling parameters – chosen to fit well on H100 SMs.
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    # Grid: (i‑tiles, j‑tiles, batch*hidden)
    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,
        (N + BLOCK_N - 1) // BLOCK_N,
        B * hidden_dim,
    )
    tri_mul_kernel[grid](
        left,
        right,
        out_hidden,
        B,
        N,
        hidden_dim,
        stride_left_b,
        stride_left_i,
        stride_left_k,
        stride_left_h,
        stride_right_b,
        stride_right_i,          # ↔ original “i” dimension (we treat it as j here)
        stride_right_k,
        stride_right_h,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        num_warps=4,            # tuned for H100
    )

    # ------------------------------------------------------------------
    # 6️⃣ Post‑processing: layer‑norm over hidden_dim, output‑gate, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = get("to_out_norm.weight")
    to_out_norm_bias = get("to_out_norm.bias")
    out_norm = torch.nn.functional.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=1e-5,
    )
    out_norm = out_norm * out_gate  # apply output gate

    # Final linear projection back to the original dimension
    out = torch.nn.functional.linear(out_norm, get("to_out.weight"))

    return out
