@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)  # i tile
    pid_j = tl.program_id(1)  # j tile
    pid_bd = tl.program_id(2) # combined batch and hidden dim

    b = pid_bd // D
    d = pid_bd % D

    i_offset = pid_i * BLOCK_M
    j_offset = pid_j * BLOCK_N

    # Compute row and col ranges
    range_i = i_offset + tl.arange(0, BLOCK_M)
    range_j = j_offset + tl.arange(0, BLOCK_N)

    # Initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over K
    for k_iter in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_iter * BLOCK_K
        range_k = k_offset + tl.arange(0, BLOCK_K)

        # mask for valid indices
        mask_i = range_i < N
        mask_j = range_j < N
        mask_k = range_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K)
        left_ptrs = left_ptr + b * stride_left_b + d * stride_left_d + (range_i[:, None] * stride_left_i) + (range_k[None, :] * stride_left_k)
        left = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0).to(tl.float32)

        # Load right tile: shape (BLOCK_K, BLOCK_N) (transposed)
        right_ptrs = right_ptr + b * stride_right_b + d * stride_right_d + (range_k[:, None] * stride_right_k) + (range_j[None, :] * stride_right_j)
        right = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0).to(tl.float32)

        # Dot product
        acc += tl.dot(left, right)  # left (M,K) @ right (K,N) -> (M,N)
    
    # Store result
    out_ptrs = out_ptr + b * stride_out_b + d * stride_out_d + (range_i[:, None] * stride_out_i) + (range_j[None, :] * stride_out_j)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul "outgoing" implementation with Triton.

The forward pass consists of:
1. Layer‑Norm on the last dimension of the input.
2. Linear projections (left/right) and gated linear projections.
3. Optional masking of the projected tensors.
4. A cubic‑time tensor contraction:
      out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
   which is the core of the incoming/outgoing TriMul.
   This contraction is performed with a custom Triton kernel that
   treats the hidden dimension `d` as a separate program dimension and
   computes a blocked GEMM for every (i‑tile, j‑tile, batch, hidden).
5. A second Layer‑Norm on the hidden dimension, gated scaling and a
   final linear projection back to `dim`.

The kernel is tiled with BLOCK_M×BLOCK_N output tiles and a BLOCK_K
reduction tile.  It works for any batch size (≤2), sequence length
up to 1024 and hidden dimensions up to 384.  All computations are in
float32 (the kernel loads as float32; you may cast to fp16 before the
kernel for bandwidth‑limited scenarios).
"""

import torch
import triton
import triton.language as tl

# --------------------------------------------------------------------------- #
# Triton kernel: computes the outgoing TriMul contraction.
# --------------------------------------------------------------------------- #
@triton.jit
def _trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    # strides for left  [b, i, k, d]
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right [b, j, k, d]  (j is second dim, k third)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for out   [b, i, j, d]
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time block sizes
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr,
):
    """ Compute out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d] """
    pid_i = tl.program_id(0)   # tile over i (row)
    pid_j = tl.program_id(1)   # tile over j (col)
    pid_bd = tl.program_id(2)  # combined batch * hidden dimension

    # decode batch and hidden index
    b = pid_bd // D
    d = pid_bd % D

    # Starting offsets for this tile
    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # range vectors
    i_range = i_base + tl.arange(0, BLOCK_M)
    j_range = j_base + tl.arange(0, BLOCK_N)

    # accumulator for the tile (BLOCK_M x BLOCK_N)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over the reduction dimension k in blocks of size BLOCK_K
    num_k_blocks = tl.cdiv(N, BLOCK_K)
    for kb in range(0, num_k_blocks):
        k_base = kb * BLOCK_K
        k_range = k_base + tl.arange(0, BLOCK_K)

        # --- Load left tile: shape (BLOCK_M, BLOCK_K) ---
        # left[b, i, k, d]
        left_ptrs = (left_ptr +
                     b * stride_left_b +
                     d * stride_left_d +
                     (i_range[:, None] * stride_left_i) +
                     (k_range[None, :] * stride_left_k))
        mask_i = i_range < N
        mask_k = k_range < N
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None] & mask_k[None, :],
                       other=0.0).to(tl.float32)

        # --- Load right tile (transposed): shape (BLOCK_K, BLOCK_N) ---
        # right[b, j, k, d] ---> we need (k, j) order for dot
        right_ptrs = (right_ptr +
                      b * stride_right_b +
                      d * stride_right_d +
                      (k_range[:, None] * stride_right_k) +
                      (j_range[None, :] * stride_right_j))
        mask_j = j_range < N
        right = tl.load(right_ptrs,
                        mask=mask_k[:, None] & mask_j[None, :],
                        other=0.0).to(tl.float32)

        # Dot product: (M, K) @ (K, N) -> (M, N)
        acc += tl.dot(left, right)

    # --- Write the tile to output ---
    out_ptrs = (out_ptr +
                b * stride_out_b +
                d * stride_out_d +
                (i_range[:, None] * stride_out_i) +
                (j_range[None, :] * stride_out_j))
    mask_o = (i_range < N)[:, None] & (j_range < N)[None, :]
    tl.store(out_ptrs, acc, mask=mask_o)


# --------------------------------------------------------------------------- #
# Public entry point
# --------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Triton‑accelerated forward of the TriMul “outgoing” module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor [B, N, N, dim] (float32, on CUDA)
            - mask: torch.Tensor [B, N, N] (bool or float) or None
            - weights: dict of torch.Tensor containing all linear / ln weights
            - config: dict with keys "dim", "hidden_dim", possibly "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32, on CUDA)
    """
    # Unpack arguments
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------- #
    # Layer‑Norm on the input (last dimension = dim)
    # ------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------- #
    # Linear projections (no bias) and gating
    # ------------------------------------------------------------------- #
    # Projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # Optional mask (if `nomask` is True the mask may be all‑ones or omitted)
    if mask is not None:
        mask_f = mask.to(left.dtype).unsqueeze(-1)        # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # Gating (sigmoid)
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------- #
    # Triton kernel for the cubic contraction
    # ------------------------------------------------------------------- #
    B, N, _, _ = left.shape  # N = seq_len
    # Allocate output of the contraction (before final LN & linear)
    out = torch.empty((B, N, N, hidden_dim), dtype=left.dtype, device=device)

    # Strides (int64) – convert to Python ints for Triton
    l_str = left.stride()
    r_str = right.stride()
    o_str = out.stride()
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = map(int, l_str)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = map(int, r_str)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = map(int, o_str)

    # Tunable block sizes (empirically good for N up to 1024 on H100)
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,          # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,          # j‑tiles
        B * hidden_dim,                        # combined batch × hidden
    )

    _trimul_outgoing_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        D=hidden_dim,
        # strides for left
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        # strides for right
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        # strides for out
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------- #
    # Post‑processing: second LayerNorm, gated scaling, final linear
    # ------------------------------------------------------------------- #
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate  # gated scaling
    out = torch.nn.functional.linear(out, weights["to_out.weight"])  # [B,N,N,dim]

    return out
