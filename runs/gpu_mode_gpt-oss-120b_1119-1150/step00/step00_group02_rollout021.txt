    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)
    # compute offsets
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    total = num_i * num_j
    b = pid // total
    pid_rem = pid % total
    i_block = pid_rem // num_j
    j_block = pid_rem % num_j

    # compute offsets
    offs_i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Initialize accumulator
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

    # Loop over K dimension blocks
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k_base)
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block
        left_ptrs = left_ptr + \
            b * stride_left_b + \
            offs_i[:, None, None] * stride_left_i + \
            offs_k[None, :, None] * stride_left_k + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_left_h
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H), other=0.0)

        # Load right block
        right_ptrs = right_ptr + \
            b * stride_right_b + \
            offs_j[:, None, None] * stride_right_j + \
            offs_k[None, :, None] * stride_right_k + \
            tl.arange(0, BLOCK_H)[None, None, :] * stride_right_h
        right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & (tl.arange(0, BLOCK_H)[None, None, :] < H), other=0.0)

        # Compute matmul for each hidden channel
        for h in tl.static_range(0, BLOCK_H):
            # shape (BLOCK_I, cur_k) and (BLOCK_J, cur_k)
            left_h = left[:, :, h]  # (BLOCK_I, BLOCK_K)
            right_h = right[:, :, h]  # (BLOCK_J, BLOCK_K)
            # Perform dot product: left_h @ right_h.T
            # tl.dot expects shapes (M, K) and (N, K) when trans_b=True (makes B (K,N)). We'll compute:
            acc[:, :, h] += tl.dot(left_h, right_h, trans_b=True)
    # Store result
    out_ptrs = out_ptr + \
        b * stride_out_b + \
        offs_i[:, None, None] * stride_out_i + \
        offs_j[None, :, None] * stride_out_j + \
        tl.arange(0, BLOCK_H)[None, None, :] * stride_out_h

def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # compute block indices
    num_i = tl.cdiv(N, BLOCK_I)
    num_j = tl.cdiv(N, BLOCK_J)
    total_blocks_per_batch = num_i * num_j
    b = pid // total_blocks_per_batch
    block_id = pid % total_blocks_per_batch
    i_block = block_id // num_j
    j_block = block_id % num_j

    # base coordinates
    offs_i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)

    # mask for i and j
    mask_i = offs_i < N
    mask_j = offs_j < N

    # Loop over hidden dimension blocks
    for h_offset in range(0, tl.cdiv(H, BLOCK_H)):
        h_base = h_offset * BLOCK_H
        cur_H = tl.minimum(BLOCK_H, H - h_base)
        offs_h = h_base + tl.arange(0, BLOCK_H)
        mask_h = offs_h < H

        # accumulator for this tile (BLOCK_I, BLOCK_J, BLOCK_H)
        acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_H), dtype=tl.float32)

        # Loop over K blocks
        for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
            k_base = k_offset * BLOCK_K
            cur_K = tl.minimum(BLOCK_K, N - k_base)
            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # pointers for left and right
            left_ptrs = left_ptr + \
                b * stride_left_b + \
                offs_i[:, None, None] * stride_left_i + \
                offs_k[None, :, None] * stride_left_k + \
                offs_h[None, None, :] * stride_left_h
            left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

            right_ptrs = right_ptr + \
                b * stride_right_b + \
                offs_j[:, None, None] * stride_right_j + \
                offs_k[None, :, None] * stride_right_k + \
                offs_h[None, None, :] * stride_right_h
            right = tl.load(right_ptrs, mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0)

            # compute per-channel dot
            for h in tl.static_range(0, BLOCK_H):
                # skip out-of-range channel values
                if mask_h[h] == 0:
                    continue  # cannot use continue; need to use mask later
                acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h], trans_b=True)

        # store result
        out_ptrs = out_ptr + \
            b * stride_out_b + \
            offs_i[:, None, None] * stride_out_i + \
            offs_j[None, :, None] * stride_out_j + \
            offs_h[None, None, :] * stride_out_h

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    # LayerNorm on input
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Linear projections
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)
    right = torch.nn.functional.linear(x, right_proj_weight)
    # apply mask if provided
    if mask is not None:
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp
    # gates
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))
    left = left * left_gate
    right = right * right_gate

    # ensure contiguous for Triton
    left = left.contiguous()
    right = right.contiguous()
    # compute out = sum_k left[...,k,:] * right[...,k,:] using Triton
    out = trimul_forward(left, right)

    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    # out gate scaling
    out = out * out_gate
    # final linear
    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)
    return out

trilinear multiplication (TriMul) forward pass.

The heavy part of the operator is the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
over the sequence dimension *k*.  We implement it with a custom
Triton kernel that computes a batched matmul for every hidden
channel (d) using tiling on the *i*, *j*, *k* and hidden (d)
dimensions.  The surrounding LayerNorms, linear projections,
gates and the final output projection are performed with
regular PyTorch kernels.

The kernel works for batch size 1‑2, sequence length up to
1024 and hidden dimensions up to 384.  It respects an optional
pair‑wise mask (broadcasted to the hidden dimension) and returns a
float‑32 result of shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl

# Triton kernel for the triple product:   out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr,          # const float*   left  [B,N,N,H]
    right_ptr,         # const float*   right [B,N,N,H]
    out_ptr,           # float*         out   [B,N,N,H]
    B, N, H,           # int64 scalars  batch, seq_len, hidden
    # strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    pid = tl.program_id(0)

    # ---- block indices -------------------------------------------------
    num_i = tl.cdiv(N, BLOCK_I)          # blocks along the i‑axis
    num_j = tl.cdiv(N, BLOCK_J)          # blocks along the j‑axis
    blocks_per_batch = num_i * num_j
    b = pid // blocks_per_batch           # batch index
    pid_in_batch = pid % blocks_per_batch
    i_block = pid_in_batch // num_j
    j_block = pid_in_batch % num_j

    # ---- i / j coordinates --------------------------------------------
    offs_i = i_block * BLOCK_I + tl.arange(0, BLOCK_I)
    offs_j = j_block * BLOCK_J + tl.arange(0, BLOCK_J)
    mask_i = offs_i < N
    mask_j = offs_j < N

    for h_tile in range(0, tl.cdiv(H, BLOCK_H)):

        cur_h = tl.minimum(BLOCK_H, H - h_base)          # may be < BLOCK_H on the last tile

        mask_h = offs_h < H

            offs_k = k_base + tl.arange(0, BLOCK_K)
            mask_k = offs_k < N

            # left[b, i, k, h]
            left_ptrs = (
                left_ptr
                + b * stride_left_b
                + offs_i[:, None, None] * stride_left_i
                + offs_k[None, :, None] * stride_left_k
                + offs_h[None, None, :] * stride_left_h
            )
            left = tl.load(
                left_ptrs,
                mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )

            # right[b, j, k, h]
            right_ptrs = (
                right_ptr
                + b * stride_right_b
                + offs_j[:, None, None] * stride_right_j
                + offs_k[None, :, None] * stride_right_k
                + offs_h[None, None, :] * stride_right_h
            )
            right = tl.load(
                right_ptrs,
                mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
                other=0.0,
            )

            # ---- per‑channel matmul ------------------------------------
            #   left[i,k,h]  (BLOCK_I × cur_k)
            #   right[j,k,h] (BLOCK_J × cur_k)
            #   dot(left, right, trans_b=True) → (BLOCK_I × BLOCK_J)
            for h in tl.static_range(0, BLOCK_H):
                # The loads already zero‑pad out‑of‑range h values, so
                # we can safely compute the dot for every h.
                acc[:, :, h] += tl.dot(left[:, :, h], right[:, :, h], trans_b=True)

        # ---- write the tile back ---------------------------------------
        out_ptrs = (
            out_ptr
            + b * stride_out_b
            + offs_i[:, None, None] * stride_out_i
            + offs_j[None, :, None] * stride_out_j
            + offs_h[None, None, :] * stride_out_h
        )
        tl.store(
            out_ptrs,
            acc,
            mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
        )


# ----------------------------------------------------------------------
# Helper that launches the kernel
# ----------------------------------------------------------------------
def trimul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the TriMul contraction using `trimul_kernel`.
    Arguments
    ---------
    left, right : torch.Tensor
        Shape [B, N, N, H] (float32, contiguous, on CUDA).

    Returns
    -------
    out : torch.Tensor
        Same shape as inputs.
    """
    assert left.is_cuda and right.is_cuda
    B, N, _, H = left.shape

    # output buffer
    out = torch.empty_like(left)

    # strides (in elements)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # tile sizes – tuned for H100, fits in shared memory/registers
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_H = 8   # hidden‑dim tile; 8 ⇒ 32 KB per load (32 × 32 × 8)

    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    total_programs = B * grid_i * grid_j

    trimul_kernel[total_programs](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        H=H,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_H=BLOCK_H,
        num_warps=4,          # enough warps for the tile size on H100
    )
    return out


# ----------------------------------------------------------------------
# Entry point required by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul module (outgoing variant) using a Triton
    kernel for the O(N³·hidden) contraction.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor      shape [B, N, N, dim]
        - mask        : torch.Tensor      shape [B, N, N]  (may be all‑ones)
        - weights     : dict[str, torch.Tensor]  model parameters
        - config      : dict   must contain "dim" and "hidden_dim"

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32, on CUDA).
    """
    # unpack
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # ------------------------------------------------------------------
    # 1. Input LayerNorm
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor, (dim,), weight=norm_weight, bias=norm_bias
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections
    # ------------------------------------------------------------------
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------------
    # 3. Optional pair‑wise mask (broadcasted over hidden dim)
    # ------------------------------------------------------------------
    if not config.get("nomask", False):
        # mask is [B,N,N]; expand to hidden dim
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 4. Gates (sigmoid of linear transforms)
    # ------------------------------------------------------------------
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction (TriMul) – Triton kernel
    # ------------------------------------------------------------------
    # Ensure contiguous tensors for the Triton kernel
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_forward(left, right)   # shape [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 6. Output LayerNorm + gating + final linear projection
    # ------------------------------------------------------------------
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b
    )
    out = out * out_gate                       # gated residual
    to_out_w = weights["to_out.weight"]        # shape [dim, hidden]
    out = torch.nn.functional.linear(out, to_out_w)   # back to `dim`

    return out
