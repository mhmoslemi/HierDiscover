@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_left_m, stride_left_k, stride_left_h,
    stride_right_n, stride_right_k, stride_right_h,
    stride_out_m, stride_out_n, stride_out_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
    eps: tl.constexpr = 0.0
):
   pid = tl.program_id(0)
   num_pid_m = tl.cdiv(N, BLOCK_M)
   num_pid_n = tl.cdiv(N, BLOCK_N)
   pid_m = pid // num_pid_n
   pid_n = pid % num_pid_n

   pid_h = tl.program_id(1)  # second dimension for hidden.

   off_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
   off_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
   off_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

   # Mask OOB
   mask_m = off_m < N
   mask_n = off_n < N
   mask_h = off_h < H

   # initialize accumulator
   acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

   # Loop over K dimension
   for k in range(0, N, BLOCK_K):
       off_k = k + tl.arange(0, BLOCK_K)
       mask_k = off_k < N

       # left tile: shape (M,K,H)
       left_ptrs = left_ptr + off_m[:, None, None] * stride_left_m + off_k[None, :, None] * stride_left_k + off_h[None, None, :] * stride_left_h
       left = tl.load(left_ptrs, mask=mask_m[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

       # right tile: shape (N,K,H)
       right_ptrs = right_ptr + off_n[:, None, None] * stride_right_n + off_k[None, :, None] * stride_right_k + off_h[None, None, :] * stride_right_h
       right = tl.load(right_ptrs, mask=mask_n[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :], other=0.0).to(tl.float32)

       # accumulate: sum over k dimension: left[:,k,:] * right[:,k,:] across k
       # do elementwise multiply and sum across k
       # vectorized using broadcasting across M,N for each k index
       # we can use tl.broadcast_to within the loop:
       for kk in range(BLOCK_K):
           # compute mask for this k element
           mask_kk = mask_k[kk]
           # as if we only compute when in bounds
           # get vectors
           l = left[:, kk, :]   # (M,H)
           r = right[:, kk, :]  # (N,H)
           # broadcast and accumulate
           acc += tl.broadcast_to(l[:, None, :], (BLOCK_M, BLOCK_N, BLOCK_H)) * tl.broadcast_to(r[None, :, :], (BLOCK_M, BLOCK_N, BLOCK_H))

   # Store result
   out_ptrs = out_ptr + off_m[:, None, None] * stride_out_m + off_n[None, :, None] * stride_out_n + off_h[None, None, :] * stride_out_h

    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]

    # cast to fp16 for speed
    x_h = x.to(torch.float16)

    # Linear projections: using fp16 weights
    left_proj_weight = weights["left_proj.weight"].to(torch.float16)
    right_proj_weight = weights["right_proj.weight"].to(torch.float16)
    left_gate_weight = weights["left_gate.weight"].to(torch.float16)
    right_gate_weight = weights["right_gate.weight"].to(torch.float16)
    out_gate_weight = weights["out_gate.weight"].to(torch.float16)

    left = torch.nn.functional.linear(x_h, left_proj_weight)  # shape B,N,N,H
    right = torch.nn.functional.linear(x_h, right_proj_weight)

    right_gate = torch.nn.functional.linear(x_h, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_h, out_gate_weight).sigmoid()

        left = left * mask
        right = right * mask

    # Apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute pairwise multiplication via Triton kernel
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    # Strides for left/right/out
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)
    stride_r_j = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)
    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 4

    # compute grid dimensions

             ( (N + BLOCK_N - 1) // BLOCK_N ),

        right_ptr = right[b].contiguous()

"""
TriMul “outgoing” implementation for AlphaFold‑3.

The heavy O(N³·H) pairwise “einsum” 
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel that tiles the i‑, j‑ and h‑axes.
All other ops (LayerNorm, linear projections, sigmoids, masking) are kept in
plain PyTorch for readability and to reuse highly‑optimised cuBLAS kernels.
The kernel works on FP16 data to minimise bandwidth and accumulates in FP32
to preserve numerical stability. The final output is returned in FP32.
"""

import torch
import triton
import triton.language as tl
from torch.nn import functional as F


# ----------------------------------------------------------------------------- #
# Triton kernel
# ----------------------------------------------------------------------------- #
@triton.jit
def trimul_einsum_kernel(
    left_ptr,            # *fp16  [i, k, h]   (batch slice)
    right_ptr,           # *fp16  [j, k, h]   (batch slice)
    out_ptr,             # *fp32  [i, j, h]
    N, H,                # problem sizes
    stride_l_i, stride_l_k, stride_l_h,   # left strides (i, k, h)
    stride_r_j, stride_r_k, stride_r_h,   # right strides (j, k, h)
    stride_o_i, stride_o_j, stride_o_h,   # output strides (i, j, h)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr,
):
    """Compute  out[i,j,h] = Σ_k left[i,k,h] * right[j,k,h]   (FP16→FP32)."""

    pid_m = tl.program_id(0)          # tile index over i dimension
    pid_n = tl.program_id(1)          # tile index over j dimension
    pid_h = tl.program_id(2)          # tile index over hidden dimension

    # --- tile offsets ---------------------------------------------------------
    offs_i = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * BLOCK_H + tl.arange(0, BLOCK_H)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

    # --- loop over the reduction axis k --------------------------------------
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left[i,k,h]   shape (M, K, H)
        left_ptrs = (
            left_ptr
            + offs_i[:, None, None] * stride_l_i
            + offs_k[None, :, None] * stride_l_k
            + offs_h[None, None, :] * stride_l_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Load right[j,k,h]   shape (N, K, H)
        right_ptrs = (
            right_ptr
            + offs_j[:, None, None] * stride_r_j
            + offs_k[None, :, None] * stride_r_k
            + offs_h[None, None, :] * stride_r_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0,
        ).to(tl.float32)

        # Accumulate Σ_k left[i,k,h] * right[j,k,h]
        # Unroll the K‑loop manually for the small block (BLOCK_K ≤ 32)
        for kk in range(BLOCK_K):
            l = left[:, kk, :]                    # (M, H)
            r = right[:, kk, :]                   # (N, H)
            # broadcast to (M,N,H) and multiply
            acc += (
                tl.broadcast_to(l[:, None, :], (BLOCK_M, BLOCK_N, BLOCK_H))
                * tl.broadcast_to(r[None, :, :], (BLOCK_M, BLOCK_N, BLOCK_H))
            )

    # --- write back -----------------------------------------------------------
    out_ptrs = (
        out_ptr
        + offs_i[:, None, None] * stride_o_i
        + offs_j[None, :, None] * stride_o_j
        + offs_h[None, None, :] * stride_o_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
    )


# ----------------------------------------------------------------------------- #
# Entrypoint
# ----------------------------------------------------------------------------- #
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Args:
        data: tuple (input, mask, weights, config)
            - input : torch.Tensor[bs, N, N, dim]   (float32)
            - mask  : torch.Tensor[bs, N, N] (optional, 0/1)
            - weights: dict of all linear / LN parameters (float32)
            - config : dict with keys "dim" and "hidden_dim"

    Returns:
        torch.Tensor[bs, N, N, dim]  (float32)
    """
    input_tensor, mask_tensor, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    device = input_tensor.device

    # --------------------------------------------------------------------- #
    # 1. LayerNorm on the input (float32)
    # --------------------------------------------------------------------- #
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # --------------------------------------------------------------------- #
    # 2. Linear projections + gating (FP16 for speed)
    # --------------------------------------------------------------------- #
    x_h = x.to(torch.float16)

    # projection weights (FP16)
    left_proj_w = weights["left_proj.weight"].to(torch.float16)
    right_proj_w = weights["right_proj.weight"].to(torch.float16)
    left_gate_w = weights["left_gate.weight"].to(torch.float16)
    right_gate_w = weights["right_gate.weight"].to(torch.float16)
    out_gate_w = weights["out_gate.weight"].to(torch.float16)

    left = F.linear(x_h, left_proj_w)          # [B, N, N, H]  fp16
    right = F.linear(x_h, right_proj_w)       # [B, N, N, H]  fp16

    left_gate = F.linear(x_h, left_gate_w).sigmoid()   # fp16
    right_gate = F.linear(x_h, right_gate_w).sigmoid() # fp16
    out_gate = F.linear(x_h, out_gate_w).sigmoid()     # fp16

    # --------------------------------------------------------------------- #
    # 3. Apply optional mask and element‑wise gates (still FP16)
    # --------------------------------------------------------------------- #
    if mask_tensor is not None:
        # mask shape [B, N, N] → [B, N, N, 1] for broadcasting
        mask = mask_tensor.to(torch.float16).unsqueeze(-1)
        left = left * mask
        right = right * mask

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------------- #
    # 4. Heavy einsum via Triton (output accumulated in FP32)
    # --------------------------------------------------------------------- #
    B, N, _, H = left.shape
    out = torch.empty((B, N, N, H), device=device, dtype=torch.float32)

    # tile sizes – chosen to fit comfortably into registers on an H100
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 4   # hidden dimension is processed in small blocks

    # stride helpers (all tensors are contiguous in the last dimension)
    stride_l_i = left.stride(1)
    stride_l_k = left.stride(2)
    stride_l_h = left.stride(3)

    stride_r_j = right.stride(1)
    stride_r_k = right.stride(2)
    stride_r_h = right.stride(3)

    stride_o_i = out.stride(1)
    stride_o_j = out.stride(2)
    stride_o_h = out.stride(3)

    # launch one kernel per batch (B ≤ 2, negligible overhead)
    for b in range(B):
        left_b = left[b].contiguous()
        right_b = right[b].contiguous()
        out_b = out[b]

        grid = (
            (N + BLOCK_M - 1) // BLOCK_M,
            (N + BLOCK_N - 1) // BLOCK_N,
            (H + BLOCK_H - 1) // BLOCK_H,
        )
        trimul_einsum_kernel[grid](
            left_b,
            right_b,
            out_b,
            N,
            H,
            stride_l_i,
            stride_l_k,
            stride_l_h,
            stride_r_j,
            stride_r_k,
            stride_r_h,
            stride_o_i,
            stride_o_j,
            stride_o_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            BLOCK_H=BLOCK_H,
        )

    # --------------------------------------------------------------------- #
    # 5. Output LayerNorm, gate, and final linear projection
    # --------------------------------------------------------------------- #
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_w, bias=to_out_norm_b)

    # out_gate is still FP16 → cast to FP32 for the final multiplication
    out_gate_f32 = out_gate.to(torch.float32)
    out = out * out_gate_f32

    # final projection back to the original channel dimension
    to_out_w = weights["to_out.weight"]
    out = F.linear(out, to_out_w)

    return out
