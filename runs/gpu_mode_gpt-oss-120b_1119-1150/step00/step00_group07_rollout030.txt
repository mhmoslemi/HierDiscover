    N, H,
    stride_i, stride_k, stride_h,
    out_stride_i, out_stride_j, out_stride_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, VEC: tl.constexpr

    pid_n = tl.program_id(1)
    pid_h = tl.program_id(2)

    offs_n = pid_n * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_h = pid_h * VEC + tl.arange(0, VEC)

    mask_h = offs_h < H

    for k in range(0, N, BLOCK_K):
        # k offsets
        offs_k = k + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left tile: shape (BLOCK_M, BLOCK_K, VEC)

"""
TriMul (outgoing) fused implementation.

The core operation is the 4‑D “einsum”
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
where left/right are the linear‑projected, gated and masked tensors
of shape [B, N, N, H].  This is a batched matrix multiplication for every
hidden channel.  The kernel tiles the output (i,j) plane (BLOCK_M×BLOCK_N)
and the hidden dimension (VEC) while iterating over k in blocks (BLOCK_K).
For each k‑block we load a tile of left (i×k×VEC) and a transposed tile of
right (k×j×VEC) and perform a small GEMM per hidden slice using `tl.dot`,
accumulating in fp32.  The kernel works on a single batch; the Python wrapper
calls it for each batch element.  All other operations (LayerNorms,
Linear projections, gating, final LayerNorm and Linear) are performed with
PyTorch for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl
from typing import Tuple, Dict

# ----------------------------------------------------------------------
# Triton kernel that computes the heavy “einsum” part.
# ----------------------------------------------------------------------
@triton.jit
def trmul_kernel(
    left_ptr,            # *fp16   left  tile  [N, N, H]
    right_ptr,           # *fp16   right tile  [N, N, H]
    out_ptr,             # *fp32   output     [N, N, H]
    N, H,                # sizes
    stride_i, stride_k, stride_h,               # strides for left/right (i, k, h)
    out_stride_i, out_stride_j, out_stride_h,   # strides for output (i, j, h)
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, VEC: tl.constexpr
):
    """
    Compute out[i,j,:] = Σ_k left[i,k,:] * right[j,k,:]  (no batch dim)
    Each thread block processes a (BLOCK_M × BLOCK_N) tile of the (i,j)
    matrix and VEC hidden channels.  The computation is tiled over k.
    """
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_h = tl.program_id(2)

    # ------------------- offsets for this tile -------------------
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)          # (BLOCK_M,)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)          # (BLOCK_N,)
    offs_h = pid_h * VEC + tl.arange(0, VEC)                  # (VEC,)

    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_h = offs_h < H

    # accumulator in FP32, shape (BLOCK_M, BLOCK_N, VEC)
    acc = tl.zeros([BLOCK_M, BLOCK_N, VEC], dtype=tl.float32)

    # ------------------- loop over k in blocks -------------------
    for k_base in range(0, N, BLOCK_K):
        offs_k = k_base + tl.arange(0, BLOCK_K)              # (BLOCK_K,)
        mask_k = offs_k < N

        # ----- load left tile: shape (BLOCK_M, BLOCK_K, VEC) -----
        # left[i,k,h] with i from offs_i, k from offs_k, h from offs_h
        left_ptrs = left_ptr \
            + (offs_i[:, None, None] * stride_i) \
            + (offs_k[None, :, None] * stride_k) \
            + (offs_h[None, None, :] * stride_h)
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_h[None, None, :],
            other=0.0
        )                              # fp16, shape (BLOCK_M, BLOCK_K, VEC)

        # ----- load right tile transposed: shape (BLOCK_K, BLOCK_N, VEC) -----
        # we want right[j,k,h] -> treat as (k,j) for GEMM
        right_ptrs = right_ptr \
            + (offs_k[:, None, None] * stride_k) \
            + (offs_j[None, :, None] * stride_i) \
            + (offs_h[None, None, :] * stride_h)
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :],
            other=0.0
        )                              # fp16, shape (BLOCK_K, BLOCK_N, VEC)

        # ----- compute contribution for each hidden slice -----
        for v in range(VEC):          # VEC is constexpr → unrolled
            a = left[:, :, v]          # (BLOCK_M, BLOCK_K)
            b = right[:, :, v]         # (BLOCK_K, BLOCK_N)
            # promote to fp32 for accurate accumulation
            a_fp32 = a.to(tl.float32)
            b_fp32 = b.to(tl.float32)
            # GEMM on this slice
            acc[:, :, v] += tl.dot(a_fp32, b_fp32)   # (BLOCK_M, BLOCK_N)

    # ------------------- store result -------------------
    out_ptrs = out_ptr \
        + (offs_i[:, None, None] * out_stride_i) \
        + (offs_j[None, :, None] * out_stride_j) \
        + (offs_h[None, None, :] * out_stride_h)

    mask_out = mask_i[:, None, None] & mask_j[None, :, None] & mask_h[None, None, :]
    tl.store(out_ptrs, acc, mask=mask_out)


# ----------------------------------------------------------------------
# Python entry point that prepares inputs, launches the kernel and
# finishes the TriMul forward pass.
# ----------------------------------------------------------------------
def custom_kernel(data: Tuple[torch.Tensor, torch.Tensor, Dict[str, torch.Tensor], Dict]) -> torch.Tensor:
    """
    Forward pass of the “outgoing” TriMul module.

    Arguments
    ---------
    data : Tuple containing
        - input           : torch.Tensor [B, N, N, dim]
        - mask            : torch.Tensor [B, N, N] (may be all‑ones)
        - weights         : dict of model weights (LayerNorm, Linear etc.)
        - config          : dict with keys "dim", "hidden_dim", "nomask"

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", True)

    # ------------------------------------------------------------------
    # 1. LayerNorm on the input (last dimension = dim)
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    eps = 1e-5
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias) → hidden_dim
    # ------------------------------------------------------------------
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = F.linear(x, left_proj_weight)       # [B,N,N,hidden_dim]
    right = F.linear(x, right_proj_weight)     # [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 3. Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        mask = mask.unsqueeze(-1)               # [B,N,N,1]
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid)
    # ------------------------------------------------------------------
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]
    left_gate = torch.sigmoid(F.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(F.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core multiplication (heavy part) – Triton kernel
    #    We work in float16 for bandwidth, accumulate in float32.
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    device = left.device
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # Output buffer for the hidden‑dim result
    out_hidden = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Tiling parameters (tuned for H100 – 32×32 threads per block)
    BLOCK_M = 32          # i‑tile size
    BLOCK_N = 32          # j‑tile size
    BLOCK_K = 32          # reduction tile
    VEC = 8               # vector width for hidden dimension (must divide hidden_dim)

    # Compute grid dimensions
    grid = (
        triton.cdiv(N, BLOCK_M),
        triton.cdiv(N, BLOCK_N),
        triton.cdiv(hidden_dim, VEC),
    )

    # Strides for [N, N, H] tensors (contiguous layout)
    stride_i = left.stride(0)   # N * H
    stride_k = left.stride(1)   # H
    stride_h = left.stride(2)   # 1

    # Output strides (still contiguous)
    out_stride_i = out_hidden.stride(1)  # N * hidden_dim
    out_stride_j = out_hidden.stride(2)  # hidden_dim
    out_stride_h = out_hidden.stride(3)  # 1

    # Launch one kernel per batch element (B ≤ 2)
    for b in range(B):
        trmul_kernel[grid](
            left_ptr=left[b],
            right_ptr=right[b],
            out_ptr=out_hidden[b],
            N=N,
            H=hidden_dim,
            stride_i=stride_i,
            stride_k=stride_k,
            stride_h=stride_h,
            out_stride_i=out_stride_i,
            out_stride_j=out_stride_j,
            out_stride_h=out_stride_h,
            BLOCK_M=BLOCK_M,
            BLOCK_N=BLOCK_N,
            BLOCK_K=BLOCK_K,
            VEC=VEC,
        )

    # ------------------------------------------------------------------
    # 6. Final LayerNorm over hidden_dim and output projection back to dim
    # ------------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]   # shape (dim, hidden_dim)

    # LayerNorm on hidden dimension
    out = F.layer_norm(
        out_hidden,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden_dim]

    # Gated by out_gate (broadcasted on hidden dim)
    out = out * out_gate

    # Final linear projection back to original dim
    out = F.linear(out, to_out_weight)   # [B,N,N,dim]

    return out
