  BLOCK_N = tl.constexpr
  BLOCK_K = tl.constexpr
  BLOCK_D = tl.constexpr

  pid_j = tl.program_id(1)

  hidden_block_idx = pid_k % num_hidden_blocks

  i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
  j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
  d = hidden_block_idx * BLOCK_D + tl.arange(0, BLOCK_D)

  # mask out-of-bounds
  i_mask = i < N
  j_mask = j < N
  d_mask = d < hidden_dim

  left_ptr_base = left_ptr + batch_idx * stride_left_b
  right_ptr_base = right_ptr + batch_idx * stride_right_b
  out_ptr_base = out_ptr + batch_idx * stride_out_b

  # create accumulator
  acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

  # loop over K tiles
  for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
    k_base = k_tile * BLOCK_K
    k = k_base + tl.arange(0, BLOCK_K)

def custom_kernel(data):
    input_tensor, mask, weights, config = data
    # unpack dims
    dim = config["dim"]; hidden_dim = config["hidden_dim"]
    device = input_tensor.device
    dtype = input_tensor.dtype

    # layernorm on input
    # using weighted bias: manual
    # Compute mean and variance across last dim per element: using torch.nn.functional.layer_norm.
    # But we could call torch.nn.functional.layer_norm directly: torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)
    # Ensure to reshape config.

    norm_weight = weights["norm.weight"]; norm_bias = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]; right_proj_weight = weights["right_proj.weight"]; left_gate_weight = weights["left_gate.weight"]; right_gate_weight = weights["right_gate.weight"]; out_gate_weight = weights["out_gate.weight"]; to_out_norm_weight = weights["to_out_norm.weight"]; to_out_norm_bias = weights["to_out_norm.bias"]; to_out_weight = weights["to_out.weight"]

    # Input layernorm
    x = torch.nn.functional.layer_norm(input_tensor, input_tensor.shape[-1:], weight=norm_weight, bias=norm_bias)

    # Projections
    # x shape [B,N,N,dim]. We need to reshape to (B*N*N, dim) to apply linear weights.
    B, N, _, _ = x.shape
    x2 = x.reshape(-1, dim)  # shape [B*N*N, dim]
    left = torch.nn.functional.linear(x2, left_proj_weight)  # shape [B*N*N, hidden_dim]
    right = torch.nn.functional.linear(x2, right_proj_weight)

    left_gate = torch.nn.functional.linear(x2, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x2, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x2, out_gate_weight).sigmoid()

    # reshape back to [B,N,N,hidden_dim]
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # apply mask if provided and not nomask (config may include)
    # Mask shape [B,N,N] maybe bool or float; convert to same dtype
    mask_val = mask.to(dtype) if mask is not None else None
    if mask_val is not None:
        mask_val = mask_val.unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask_val
        right = right * mask_val

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # Compute out via Triton kernel
    # allocate output tensor
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # Get strides
    left_ptr = left
    right_ptr = right
    out_ptr = out

    # compute strides in elements
    # left.stride() returns in elements for each dim: (stride0, stride1, stride2, stride3)
    # Our pointer stride values for left: stride_left_b = left.stride(0); stride_left_i = left.stride(1); stride_left_k = left.stride(2); stride_left_d = left.stride(3)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # define grid
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 8   # tile size for k
    BLOCK_D = 8   # hidden dim per block

    num_i = (N + BLOCK_M - 1) // BLOCK_M
    num_j = (N + BLOCK_N - 1) // BLOCK_N
    num_hidden_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D

    grid = (num_i, num_j, B * num_hidden_blocks)

    # launch kernel
    tri_mul_outgoing_kernel[grid](
        left_ptr, right_ptr, out_ptr,
        B, N, hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_D=BLOCK_D
    )

    # apply to_out_norm
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias)
    out = out * out_gate  # gating after norm
    # final linear to dim
    out = out.view(-1, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)
    out = out.view(B, N, N, dim)

    return out

"""
TriMul (outgoing) implementation with Triton acceleration.

The heavy O(N³·C) operation in the forward pass is
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]                (1)

where `left` and `right` are the linearly projected (and gated) tensors.
In the Triton kernel each program instance computes a tile of the
output tensor (i‑range × j‑range) for a small block of hidden channels `d`.
For every k‑index we load the vectors
    left_vec  = left[b,i,k,:]   (shape M×D)
    right_vec = right[b,j,k,:]  (shape N×D)
and accumulate their outer product
    acc[i,j,d] += left_vec[i,d] * right_vec[j,d].

The kernel works on a single batch; the Python wrapper launches it for all
batches and all hidden‑channel blocks.  All other operations (LayerNorm,
Linear projections, gates and the final linear) are performed with
torch.nn.functional for simplicity.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_outgoing_kernel(
    left_ptr,                # *float32
    right_ptr,               # *float32
    out_ptr,                 # *float32
    B,                       # int64   (batch size, runtime)
    N,                       # int64   (sequence length, constexpr)
    hidden_dim,              # int64   (hidden channels, constexpr)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr,   # rows per tile
    BLOCK_N: tl.constexpr,   # cols per tile
    BLOCK_K: tl.constexpr,   # inner‑k unroll factor
    BLOCK_D: tl.constexpr,   # hidden‑channel block size
):
    """
    Triton kernel for the core TriMul contraction.
    Each program instance computes a (BLOCK_M × BLOCK_N × BLOCK_D) tile of the
    output tensor for one batch and one hidden‑channel block.
    """

    # --------------------------------------------------------------
    # program ids: block i, block j, combined (batch, hidden‑block)
    pid_i = tl.program_id(0)          # tile row index
    pid_j = tl.program_id(1)          # tile column index
    pid_hb = tl.program_id(2)         # batch * hidden‑block index

    # number of hidden blocks (compile‑time constant)
    num_hidden_blocks = tl.cdiv(hidden_dim, BLOCK_D)

    # decode batch index and hidden‑block index
    batch_idx = pid_hb // num_hidden_blocks
    hidden_block_idx = pid_hb % num_hidden_blocks

    # --------------------------------------------------------------
    # offsets within the tensor
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)                 # [M]
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)                 # [N]
    d_start = hidden_block_idx * BLOCK_D
    d = d_start + tl.arange(0, BLOCK_D)                         # [D]

    # out‑of‑bounds masks
    i_mask = i < N
    j_mask = j < N
    d_mask = d < hidden_dim

    # base pointers for this batch
    left_base  = left_ptr  + batch_idx * stride_left_b
    right_base = right_ptr + batch_idx * stride_right_b
    out_base   = out_ptr   + batch_idx * stride_out_b

    # accumulator for the tile (float32)
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # --------------------------------------------------------------
    # number of k‑tiles (compile‑time constant)
    num_k_tiles = tl.cdiv(N, BLOCK_K)

    # iterate over k tiles
    for tk in range(num_k_tiles):
        k_tile_start = tk * BLOCK_K
        # inner unrolled loop over the k elements inside the tile
        for kk in range(BLOCK_K):
            k = k_tile_start + kk
            k_valid = k < N

            # ------- load left[i, k, d]  (shape M×D) -----------------
            left_ptrs = (left_base
                         + i[:, None] * stride_left_i          # i‑dim
                         + k * stride_left_k                    # k‑dim
                         + d[None, :] * stride_left_d)          # d‑dim
            left_vec = tl.load(left_ptrs,
                               mask=i_mask[:, None] & d_mask[None, :] & k_valid,
                               other=0.0,  # padding for OOB
                               dtype=tl.float32)

            # ------- load right[j, k, d] (shape N×D) -----------------
            right_ptrs = (right_base
                          + j[:, None] * stride_right_j          # j‑dim
                          + k * stride_right_k                   # k‑dim
                          + d[None, :] * stride_right_d)         # d‑dim
            right_vec = tl.load(right_ptrs,
                                mask=j_mask[:, None] & d_mask[None, :] & k_valid,
                                other=0.0,
                                dtype=tl.float32)

            # ------- outer‑product and accumulate --------------------
            # left_vec:  (M, D) -> (M,1,D)
            # right_vec: (N, D) -> (1,N,D)
            acc += left_vec[:, None, :] * right_vec[None, :, :]

    # --------------------------------------------------------------
    # write back the tile
    out_ptrs = (out_base
                + i[:, None, None] * stride_out_i
                + j[None, :, None] * stride_out_j
                + d[None, None, :] * stride_out_d)

    tl.store(out_ptrs,
             acc,
             mask=i_mask[:, None, None] &
                  j_mask[None, :, None] &
                  d_mask[None, None, :])


def custom_kernel(data):
    """
    Python entry‑point that reproduces the forward pass of the TriMul module
    described in the AlphaFold3 paper.  All heavy O(N³·C) work (the einsum)
    is offloaded to the Triton kernel above.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor:  [B, N, N, C]  torch.Tensor
            - mask:         [B, N, N]    torch.Tensor (float or bool) or None
            - weights: dict of all model parameters (torch.Tensors)
            - config: dict with keys "dim", "hidden_dim", optionally "nomask"

    Returns:
        torch.Tensor of shape [B, N, N, dim]   (float32, on the same device)
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # 1️⃣ LayerNorm on the input (last dimension)
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    B, N, _, _ = x.shape

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (no bias)
    left_proj_weight = weights["left_proj.weight"]      # (hidden_dim, dim)
    right_proj_weight = weights["right_proj.weight"]    # (hidden_dim, dim)

    # reshape to (B*N*N, dim) for a single matmul call
    x_flat = x.view(-1, dim)                           # (B*N*N, dim)

    left = torch.nn.functional.linear(x_flat, left_proj_weight)   # (B*N*N, hidden_dim)
    right = torch.nn.functional.linear(x_flat, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Gates (sigmoid, no bias)
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.nn.functional.linear(x_flat, left_gate_weight).sigmoid()
    right_gate = torch.nn.functional.linear(x_flat, right_gate_weight).sigmoid()
    out_gate = torch.nn.functional.linear(x_flat, out_gate_weight).sigmoid()

    # reshape back to (B, N, N, hidden_dim)
    left = left.view(B, N, N, hidden_dim)
    right = right.view(B, N, N, hidden_dim)
    left_gate = left_gate.view(B, N, N, hidden_dim)
    right_gate = right_gate.view(B, N, N, hidden_dim)
    out_gate = out_gate.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------------
    # 4️⃣ Optional mask (if a mask tensor is supplied and not nomask)
    if not nomask and mask is not None:
        mask_f = mask.to(dtype).unsqueeze(-1)           # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 5️⃣ Apply gates before the heavy contraction
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 6️⃣ Heavy contraction with Triton
    # allocate output tensor (float32 for numerical stability)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # make sure the tensors are contiguous for simple stride handling
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # strides in element units
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tile sizes (tuned for H100; they are compile‑time constants)
    BLOCK_M = 32            # rows per program tile
    BLOCK_N = 32            # cols per program tile
    BLOCK_K = 8             # inner‑k unroll factor
    BLOCK_D = 8             # hidden‑channel block size

    # grid dimensions
    grid_i = (N + BLOCK_M - 1) // BLOCK_M
    grid_j = (N + BLOCK_N - 1) // BLOCK_N
    hidden_blocks = (hidden_dim + BLOCK_D - 1) // BLOCK_D
    grid_hb = B * hidden_blocks                     # batch × hidden‑block

    # launch kernel
    tri_mul_outgoing_kernel[
        (grid_i, grid_j, grid_hb)
    ](
        left,
        right,
        out,
        B,
        N,
        hidden_dim,
        stride_left_b, stride_left_i, stride_left_k, stride_left_d,
        stride_right_b, stride_right_j, stride_right_k, stride_right_d,
        stride_out_b, stride_out_i, stride_out_j, stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )

    # ------------------------------------------------------------------
    # 7️⃣ Post‑processing: to_out_norm, gating, final linear
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    to_out_weight = weights["to_out.weight"]          # (dim, hidden_dim)

    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                               # apply out‑gate after norm

    # final linear projection back to `dim`
    out = out.view(-1, hidden_dim)                     # (B*N*N, hidden_dim)
    out = torch.nn.functional.linear(out, to_out_weight)  # (B*N*N, dim)
    out = out.view(B, N, N, dim)

    return out
