```
import triton
import triton.language as tl

@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,
    B, N, H,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,
    stride_right_b, stride_right_i, stride_right_j, stride_right_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_bh = tl.program_id(2)
    # compute batch and hidden indices
    bh = pid_bh
    b = bh // H
    h = bh % H
    # offsets for i, j
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    # compute masks for out-of-bound i/j
    mask_i = i_offset < N
    mask_j = j_offset < N
    # initialize accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)
    # loop over k
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N
        # pointers for left (shape M x K)
        left_ptrs = left_ptr + b * stride_left_b + i_offset[:, None] * stride_left_i + k_offset[None, :] * stride_left_j + h * stride_left_h
        # load left values as fp16
        a = tl.load(left_ptrs, mask=mask_i[:, None] & mask_k[None, :], other=0.0)
        # pointers for right transposed (shape K x N)
        right_ptrs = right_ptr + b * stride_right_b + k_offset[:, None] * stride_right_i + j_offset[None, :] * stride_right_j + h * stride_right_h
        b_tile = tl.load(right_ptrs, mask=mask_k[:, None] & mask_j[None, :], other=0.0)
        # compute dot
        acc += tl.dot(a, b_tile)   # a: MxK, b_tile: KxN -> MxN
    # store result
    out_ptrs = out_ptr + b * stride_out_b + i_offset[:, None] * stride_out_i + j_offset[None, :] * stride_out_j + h * stride_out_h
    tl.store(out_ptrs, acc.to(tl.float16), mask=mask_i[:, None] & mask_j[None, :])
```

"""
TriMul “outgoing” implementation for AlphaFold3.

The kernel computes the core 4‑D tensor contraction
    out[b,i,j,d] = Σₖ left[b,i,k,d] * right[b,j,k,d]
where left/right are projected, gated pairwise representations.
Implementation details
----------------------
* LayerNorm and all linear / gating operations are performed with
  PyTorch (highly optimized, autograd not required).
* The expensive O(N³·H) contraction is realized as a batch of matrix
  multiplications C = A @ Bᵀ where A and B are (N×N) matrices per
  hidden channel.  A Triton kernel tiles the i, j, and k dimensions
  (BLOCK_M×BLOCK_N×BLOCK_K) and processes each hidden channel
  independently (grid dimension = B·H).  Input matrices are stored in
  FP16, the dot product accumulates in FP32, and the result is written
  as FP32.
* The kernel is launched with `grid = (⌈N/BM⌉, ⌈N/BN⌉, B*H)`.
* After the contraction we apply a second LayerNorm, the output gate,
  and a final linear projection back to the original feature dimension.
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: batched matrix multiplication with a transpose on the
# right operand.  The kernel works on tensors of shape
#    left , right : [B, N, N, H]   (H = hidden_dim)
# and writes    out      : [B, N, N, H].
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    out_ptr, left_ptr, right_ptr,               # pointers
    B, N, H,                                    # static sizes
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,   # output strides
    stride_left_b, stride_left_i, stride_left_j, stride_left_h,   # left strides
    stride_right_b, stride_right_i, stride_right_j, stride_right_h, # right strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr
):
    # Program IDs
    pid_i = tl.program_id(0)          # i‑tile index
    pid_j = tl.program_id(1)          # j‑tile index
    pid_bh = tl.program_id(2)         # combined batch‑hidden index

    # Decode batch index b and hidden index h
    bh = pid_bh
    b = bh // H
    h = bh % H

    # Tile offsets
    i_offset = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_offset = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # In‑bounds masks for i and j
    mask_i = i_offset < N
    mask_j = j_offset < N

    # Accumulator (FP32)
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # Loop over k dimension in blocks
    for k in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k * BLOCK_K + tl.arange(0, BLOCK_K)
        mask_k = k_offset < N

        # ---- Load left tile A (M × K) ---------------------------------
        # A[b, i, k, h]  with strides (b,i,k,h)
        a_ptrs = (
            left_ptr
            + b * stride_left_b
            + i_offset[:, None] * stride_left_i
            + k_offset[None, :] * stride_left_j
            + h * stride_left_h
        )
        a = tl.load(a_ptrs,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)          # FP16 -> will be up‑cast in dot

        # ---- Load right tile Bᵀ (K × N) -------------------------------
        # We need Bᵀ: B[b, k, j, h] (original B is [b,i,j,h])
        b_ptrs = (
            right_ptr
            + b * stride_right_b
            + k_offset[:, None] * stride_right_i
            + j_offset[None, :] * stride_right_j
            + h * stride_right_h
        )
        b_tile = tl.load(b_ptrs,
                         mask=mask_k[:, None] & mask_j[None, :],
                         other=0.0)        # FP16

        # ---- Matrix‑multiply tile ------------------------------------
        # a: (M, K)   b_tile: (K, N)   dot result: (M, N)
        acc += tl.dot(a, b_tile)        # accumulates in FP32

    # ---- Store result -------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + i_offset[:, None] * stride_out_i
        + j_offset[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(out_ptrs,
             acc,
             mask=mask_i[:, None] & mask_j[None, :])


# ----------------------------------------------------------------------
# Entry point: custom_kernel
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the TriMul “outgoing” operation.
    Arguments
    ---------
    data : tuple
        (input_tensor, mask_tensor, weights_dict, config_dict)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32, on the same device).
    """
    # ------------------------------------------------------------------
    # Unpack and prepare
    # ------------------------------------------------------------------
    input_tensor, mask_tensor, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]                     # input feature dimension
    hidden_dim = config["hidden_dim"]       # intermediate dimension
    nomask = config.get("nomask", True)    # True → ignore mask

    # ------------------------------------------------------------------
    # LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # Linear projections (no bias)
    # ------------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # ------------------------------------------------------------------
    # Optional mask (broadcast on hidden dim)
    # ------------------------------------------------------------------
    if not nomask:
        # mask shape: [B, N, N] → [B, N, N, 1] and cast to x.dtype
        mask = mask_tensor.to(x.dtype).unsqueeze(-1)
        left = left * mask
        right = right * mask

    # ------------------------------------------------------------------
    # Gating functions (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    ).to(left.dtype)                # cast to FP16 later
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    ).to(right.dtype)
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )                               # stay in FP32 for later multiplication

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # Cast to FP16 for the heavy O(N³·H) kernel
    # ------------------------------------------------------------------
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    B, N, _, _ = left.shape

    # ------------------------------------------------------------------
    # Allocate output buffer (FP32 accumulator)
    # ------------------------------------------------------------------
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=device)

    # ------------------------------------------------------------------
    # Strides (in elements, not bytes)
    # ------------------------------------------------------------------
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()
    stride_left_b, stride_left_i, stride_left_j, stride_left_h = left.stride()
    stride_right_b, stride_right_i, stride_right_j, stride_right_h = right.stride()

    # ------------------------------------------------------------------
    # Kernel launch configuration
    # ------------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        triton.cdiv(N, BLOCK_M),      # i‑tiles
        triton.cdiv(N, BLOCK_N),      # j‑tiles
        B * hidden_dim               # batch‑hidden dimension
    )

    # ------------------------------------------------------------------
    # Triton kernel invocation
    # ------------------------------------------------------------------
    tri_mul_kernel[grid](
        out,
        left,
        right,
        B,
        N,
        hidden_dim,
        stride_out_b,
        stride_out_i,
        stride_out_j,
        stride_out_h,
        stride_left_b,
        stride_left_i,
        stride_left_j,
        stride_left_h,
        stride_right_b,
        stride_right_i,
        stride_right_j,
        stride_right_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # ------------------------------------------------------------------
    # Post‑processing: second LayerNorm, output gate, final projection
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    # Apply the output gate (broadcasted automatically)
    out = out * out_gate

    # Final linear projection back to `dim` (no bias)
    out = torch.nn.functional.linear(out, weights["to_out.weight"])

    return out
