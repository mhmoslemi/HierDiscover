def layernorm_kernel(
    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, D,
    stride_x_m, stride_x_d,
    stride_out_m, stride_out_d,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid = tl.program_id(axis=0)
    pid_m = pid
    row_start = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    mask_m = row_start < M
    off_x_m = row_start[:, None] * stride_x_m
    off_out_m = row_start[:, None] * stride_out_m
    
    # Initialize accumulators for sum and sumsq
    sum = tl.zeros((BLOCK_M,), dtype=tl.float32)
    sumsq = tl.zeros((BLOCK_M,), dtype=tl.float32)
    
    for d in range(0, tl.cdiv(D, BLOCK_D)):

        mask_d = d_offset < D
        # load x block

        sum += tl.sum(x, axis=1)   # sum across D dimension? Actually x shape (BLOCK_M, BLOCK_D). We want per-row sum. Use tl.sum(x, axis=1) yields vector of length BLOCK_M.
        sumsq += tl.sum(x * x, axis=1)
    
    mean = sum / D

    import triton.language as tl
    input_tensor, mask, weights, config = data

    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)

    norm_bias = weights["norm.bias"]

    right_proj_weight = weights["right_proj.weight"]

    x_ptr, out_ptr,
    weight_ptr, bias_ptr,
    M, D,
    stride_x_m, stride_x_d,
    stride_out_m, stride_out_d,
    eps,
    BLOCK_M: tl.constexpr, BLOCK_D: tl.constexpr
):

    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)
    # avoid out-of-range
    mask_row = row < M

    # accumulators
    sum_val = tl.zeros([BLOCK_M], dtype=tl.float32)
    sumsq_val = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, D, BLOCK_D):
        cur_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D
        # compute pointers
        x_ptrs = x_ptr + row[:, None] * stride_x_m + cur_d[None, :] * stride_x_d
        x = tl.load(x_ptrs, mask=mask_row[:, None] & mask_d[None, :], other=0.0)

        cur_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D
        x_ptrs = x_ptr + row[:, None] * stride_x_m + cur_d[None, :] * stride_x_d

def layernorm_triton(x: torch.Tensor, weight: torch.Tensor, bias: torch.Tensor,
                    eps: float = 1e-5, block_d: int = 64) -> torch.Tensor:
    """
    Triton implementation of LayerNorm across the last dimension of ``x``.
    ``x`` must be contiguous and have shape [*, C]. The function returns a
    tensor of the same shape and dtype.
    """
    assert x.is_contiguous()
    M, D = x.shape[0], x.shape[1]
    out = torch.empty_like(x)

    stride_x_m, stride_x_d = x.stride()
    stride_out_m, stride_out_d = out.stride()

    # launch one program per row (BLOCK_M = 1)
    grid = ( (M + 1 - 1) // 1, )
    layernorm_kernel[grid](
        x, out,
        weight, bias,
        M, D,
        stride_x_m, stride_x_d,
        stride_out_m, stride_out_d,
        eps,
        BLOCK_M=1,
        BLOCK_D=block_d,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the "outgoing" TriMul module.
    Performs:
      * LayerNorm on the input (Triton)
      * Linear projections + gates (torch)
      * Optional mask application
      * Pairwise multiplicative update via batched GEMM (torch.bmm)
      * LayerNorm on the intermediate (Triton)
      * Final gating and projection back to ``dim``.
    Returns a tensor of shape [bs, seq_len, seq_len, dim].
    """
    import torch
    import torch.nn.functional as F

    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------
    # 1️⃣  Input LayerNorm (Triton)
    # ------------------------------------------------------------
    # ensure contiguous layout: shape [B, N, N, dim]
    x = input_tensor.contiguous()
    B, N, _, _ = x.shape

    # reshape to (M, dim) where M = B*N*N
    x_view = x.view(-1, dim)
    x_norm_view = layernorm_triton(
        x_view,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_d=64,
    )
    x_norm = x_norm_view.view(B, N, N, dim)

    # ------------------------------------------------------------
    # 2️⃣  Linear projections (no bias)
    # ------------------------------------------------------------
    left = F.linear(x_norm, weights["left_proj.weight"])   # [B,N,N,hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # ------------------------------------------------------------
    # 3️⃣  Optional mask (broadcast over hidden_dim)
    # ------------------------------------------------------------
    if not nomask:
        # mask may be bool or int; cast to same dtype
        mask_exp = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------
    # 4️⃣  Gating (sigmoid of linear)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5️⃣  TriMul core:  out[b,i,j] = sum_k left[b,i,k] * right[b,j,k]
    #    Implemented as batched GEMM with torch.bmm (cuBLAS)
    # ------------------------------------------------------------
    # reshape for batch GEMM: (B*hidden, N, N)
    left_perm = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)
    right_perm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)
    out_mat = torch.bmm(left_perm, right_perm)  # (B*hidden, N, N)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)

    # ------------------------------------------------------------
    # 6️⃣  Output LayerNorm (Triton) over hidden_dim
    # ------------------------------------------------------------
    out_view = out.view(-1, hidden_dim)
    out_norm_view = layernorm_triton(
        out_view,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_d=64,
    )
    out_norm = out_norm_view.view(B, N, N, hidden_dim)

    # ------------------------------------------------------------
    # 7️⃣  Output gating and final projection back to ``dim``
    # ------------------------------------------------------------
    out_norm = out_norm * out_gate
    final_out = F.linear(out_norm, weights["to_out.weight"])

    return final_out

"""
TriMul (outgoing) forward pass.

The implementation follows the reference PyTorch module but replaces the two
LayerNorm calls with custom Triton kernels.  The heavy N³ pairwise update is
expressed as a batched matrix multiplication (torch.bmm) which uses cuBLAS on
the H100 and is therefore very fast.  The overall flow is:

1. Layer‑norm over the last dimension of the input tensor (Triton).
2. Linear projections → left / right tensors (no bias).
3. Optional mask multiplication.
4. Gating: sigmoid(linear(x)) for left‑gate, right‑gate and out‑gate.
5. Apply gates to left/right.
6. Pairwise multiplicative update:
   out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
   Implemented as a batched GEMM (B·H, N, N) × (B·H, N, N) → (B·H, N, N).
7. Layer‑norm over the hidden dimension of the result (Triton).
8. Multiply by out‑gate and project back to the original ``dim`` with a linear.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton LayerNorm (mean / variance computed per row)
# ----------------------------------------------------------------------
@triton.jit
def layernorm_kernel(
    x_ptr, out_ptr,                # pointers to input / output tensors
    weight_ptr, bias_ptr,          # LayerNorm parameters (size = D)
    M, D,                          # M = number of rows, D = hidden size
    stride_x_m, stride_x_d,        # strides for input
    stride_out_m, stride_out_d,    # strides for output
    eps,                           # epsilon for numerical stability
    BLOCK_M: tl.constexpr,         # rows per program (set to 1)
    BLOCK_D: tl.constexpr,         # columns per load/store (tile size)
):

    row = pid * BLOCK_M + tl.arange(0, BLOCK_M)    # actual row index
    mask_row = row < M

    # ---------- first pass: compute mean & variance ----------
    sum_val  = tl.zeros([BLOCK_M], dtype=tl.float32)
    sumsq_val = tl.zeros([BLOCK_M], dtype=tl.float32)

    for d_start in range(0, D, BLOCK_D):
        cur_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D

        # load a (BLOCK_M, BLOCK_D) tile
        x = tl.load(
            x_ptr + row[:, None] * stride_x_m + cur_d[None, :] * stride_x_d,
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        )

    var  = sumsq_val / D - mean * mean

    for d_start in range(0, D, BLOCK_D):
        cur_d = d_start + tl.arange(0, BLOCK_D)
        mask_d = cur_d < D

            x_ptr + row[:, None] * stride_x_m + cur_d[None, :] * stride_x_d,
            mask=mask_row[:, None] & mask_d[None, :],
            other=0.0,
        )
        x_f = x.to(tl.float32)

        norm = (x_f - mean[:, None]) * rstd[:, None]   # (BLOCK_M, BLOCK_D)

        w = tl.load(weight_ptr + cur_d, mask=mask_d, other=0.0)
        b = tl.load(bias_ptr   + cur_d, mask=mask_d, other=0.0)

            mask=mask_row[:, None] & mask_d[None, :],
        )


def layernorm_triton(x: torch.Tensor,
                     weight: torch.Tensor,
                     bias: torch.Tensor,
                     eps: float = 1e-5,
                     block_d: int = 64) -> torch.Tensor:
    """
    Apply LayerNorm over the last dimension of ``x`` using the Triton kernel
    defined above.  ``x`` must be contiguous and have shape [*, C].
    Returns a tensor of identical shape and dtype.
    """
    assert x.is_contiguous(), "input to layernorm_triton must be contiguous"
    M, D = x.shape[0], x.shape[1]

    out = torch.empty_like(x)

    stride_x_m, stride_x_d = x.stride()
    stride_out_m, stride_out_d = out.stride()

    # One program per row (BLOCK_M = 1)
    grid = ((M + 1 - 1) // 1,)

    layernorm_kernel[grid](
        x, out,
        weight, bias,
        M, D,
        stride_x_m, stride_x_d,
        stride_out_m, stride_out_d,
        eps,
        BLOCK_M=1,
        BLOCK_D=block_d,
    )
    return out


# ----------------------------------------------------------------------
# Custom kernel entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor of shape [bs, seq_len, seq_len, dim]
        - mask          : torch.Tensor of shape [bs, seq_len, seq_len] (may be unused)
        - weights       : dict of torch.Tensors containing all linear/LayerNorm params
        - config        : dict with keys "dim", "hidden_dim", "nomask" (bool), and optional "eps"

    Returns
    -------
    torch.Tensor
        Output of shape [bs, seq_len, seq_len, dim]
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = config.get("eps", 1e-5)
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm (Triton)
    # --------------------------------------------------------------
    x = input_tensor.contiguous()                       # [B, N, N, dim]
    B, N, _, _ = x.shape
    x_view = x.view(-1, dim)                            # (M, dim)  where M = B*N*N
    x_norm_view = layernorm_triton(
        x_view,
        weights["norm.weight"],
        weights["norm.bias"],
        eps=eps,
        block_d=64,
    )
    x_norm = x_norm_view.view(B, N, N, dim)             # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left  = F.linear(x_norm, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x_norm, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask:
        # mask may be bool or int; ensure float dtype
        mask_exp = mask.unsqueeze(-1).to(left.dtype)      # [B, N, N, 1]
        left  = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating (sigmoid of a linear)
    # --------------------------------------------------------------
    left_gate  = torch.sigmoid(F.linear(x_norm, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x_norm, weights["right_gate.weight"]))
    out_gate   = torch.sigmoid(F.linear(x_norm, weights["out_gate.weight"]))

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. TriMul core:  out[b,i,j] = Σₖ left[b,i,k] * right[b,j,k]
    #    Implemented as a batched GEMM (cuBLAS) for speed.
    # --------------------------------------------------------------
    # reshape to (B*hidden, N, N) for bmm
    left_perm  = left.permute(0, 3, 1, 2).reshape(B * hidden_dim, N, N)   # (B*H, i, k)
    right_perm = right.permute(0, 3, 2, 1).reshape(B * hidden_dim, N, N)  # (B*H, k, j)
    out_mat = torch.bmm(left_perm, right_perm)                            # (B*H, i, j)

    # reshape back to [B, N, N, hidden_dim]
    out = out_mat.view(B, hidden_dim, N, N).permute(0, 2, 3, 1)           # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden dimension (Triton)
    # --------------------------------------------------------------
    out_cont = out.contiguous()
    out_view = out_cont.view(-1, hidden_dim)    # (M_out, hidden_dim) with M_out = B*N*N
    out_norm_view = layernorm_triton(
        out_view,
        weights["to_out_norm.weight"],
        weights["to_out_norm.bias"],
        eps=eps,
        block_d=64,
    )
    out_norm = out_norm_view.view(B, N, N, hidden_dim)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 7. Output gating + final linear projection back to dim
    # --------------------------------------------------------------
    out_norm = out_norm * out_gate
    final_out = F.linear(out_norm, weights["to_out.weight"])   # [B, N, N, dim]

    return final_out
