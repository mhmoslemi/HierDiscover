def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,
    N,  # seq_len
    stride_bL, stride_iL, stride_kL, stride_hL,
    stride_bR, stride_iR, stride_kR, stride_hR,
    stride_bO, stride_iO, stride_jO, stride_hO,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr, batch: tl.constexpr):
    pid_m = tl.program_id(0)  # block row i
    pid_n = tl.program_id(1)  # block col j
    pid_hb = tl.program_id(2) # batch*hidden
  
    # compute batch & hidden
    b = pid_hb // hidden_dim
    d = pid_hb % hidden_dim

    # base offsets
    offset_base = b * stride_bL + d * stride_hL  # same for left/right/out

    # compute block start positions
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # allocate accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k in range(0, N, BLOCK_K):
        # compute actual block size (handle tail)
        cur_k = tl.minimum(N - k, BLOCK_K)
        # offsets for left
        offs_i = i_start + tl.arange(0, BLOCK_M)
        offs_k = k + tl.arange(0, BLOCK_K)
        # mask for OOB
        mask_i = offs_i < N
        mask_k = offs_k < N
        # compute pointer for left
        left_ptrs = left_ptr + offset_base + offs_i[:, None] * stride_iL + offs_k[None, :] * stride_kL

        # mask for j

```python
def custom_kernel(data):
    x, mask, weights, config = data
    dim = config['dim']
    hidden_dim = config['hidden_dim']
    # load weights
    norm_weight = weights['norm.weight']
    norm_bias = weights['norm.bias']
    left_proj_weight = weights['left_proj.weight']
    right_proj_weight = weights['right_proj.weight']
    left_gate_weight = weights['left_gate.weight']
    right_gate_weight = weights['right_gate.weight']
    out_gate_weight = weights['out_gate.weight']
    to_out_norm_weight = weights['to_out_norm.weight']
    to_out_norm_bias = weights['to_out_norm.bias']
    to_out_weight = weights['to_out.weight']

    # LayerNorm over last dim
    eps = 1e-5
    # Compute mean and variance across dim
    mean = x.mean(dim=-1, keepdim=True)
    var = x.var(dim=-1, unbiased=False, keepdim=True)
    x_norm = (x - mean) / torch.sqrt(var + eps)
    x_norm = x_norm * norm_weight + norm_bias

    # Projections
    left = torch.nn.functional.linear(x_norm, left_proj_weight)  # shape [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # Optional mask
    if not config.get('nomask', True):
        # mask shape [B,N,N]
        mask = mask.to(left.dtype).unsqueeze(-1)  # shape [B,N,N,1]
        left = left * mask
        right = right * mask

    # Gates
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight)).to(left.dtype)
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight)).to(right.dtype)
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # cast to half for kernel
    left = left.to(torch.float16)
    right = right.to(torch.float16)

    # allocate out tensor (float32)
    out = torch.empty(left.shape, dtype=torch.float32, device=left.device)

    # compute strides
    stride_bl = left.stride(0)
    stride_il = left.stride(1)
    stride_kl = left.stride(2)
    stride_hl = left.stride(3)

    stride_br = right.stride(0)
    stride_ir = right.stride(1)
    stride_kr = right.stride(2)
    stride_hr = right.stride(3)

    stride_bo = out.stride(0)
    stride_io = out.stride(1)
    stride_jo = out.stride(2)
    stride_ho = out.stride(3)

    bs, seq_len, _, _ = left.shape

    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = ( (seq_len + BLOCK_M - 1) // BLOCK_M,
             (seq_len + BLOCK_N - 1) // BLOCK_N,
             bs * hidden_dim )

    trimul_out_kernel[grid](
        left, right, out,
        N=seq_len,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim
    )
    # kernel will compute out of shape [B,N,N,H] in float32

    # Apply output LayerNorm over hidden dim
    out = torch.nn.functional.layer_norm(out, (hidden_dim,),
                                         weight=to_out_norm_weight,
                                         bias=to_out_norm_bias,
                                         eps=eps)
    # Multiply by out_gate (float32)
    out = out * out_gate

    # Final linear projection to dim
    output = torch.nn.functional.linear(out, to_out_weight)
    return output
```

"""
TriMul (outgoing) fast implementation.

The heavy O(N³) part of TriMul:
    out[..., i, j, d] = Σ_k left[..., i, k, d] * right[..., j, k, d]

We treat each (batch, hidden) pair as an independent matrix multiplication
over the (i, j) axes with reduction over k.  This is exactly a batched GEMM.
A custom Triton kernel computes the batched GEMM in FP16 (Tensor‑Core)
while accumulating in FP32.  All other cheap operations (LayerNorm,
linear projections, gates) are performed with PyTorch.

The kernel is launched with a 3‑D grid:
    (i‑tile, j‑tile, batch*hidden_dim)
and maps the program‑id to the corresponding batch index `b` and hidden
channel `d`.  The kernel loads tiles of `left[b,i,k,d]` (M×K) and
`right[b,j,k,d]` (K×N) and accumulates the product into an FP32 tile.
After the kernel finishes, a LayerNorm over the hidden dimension,
an output gate, and a final linear projection produce the result
shape [B, N, N, dim].
"""

import torch
import triton
import triton.language as tl

# ----------------------------------------------------------------------
# Triton kernel: computes out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
# ----------------------------------------------------------------------
@triton.jit
def trimul_out_kernel(
    left_ptr, right_ptr, out_ptr,               # pointers
    N,                                          # sequence length (int)
    stride_bl, stride_il, stride_kl, stride_hl, # left strides
    stride_br, stride_ir, stride_kr, stride_hr, # right strides
    stride_bo, stride_io, stride_jo, stride_ho, # out strides
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    hidden_dim: tl.constexpr,
):
    pid_m = tl.program_id(0)        # block row (i)
    pid_n = tl.program_id(1)        # block col (j)
    pid_hb = tl.program_id(2)       # combined batch*hidden index

    # decode batch and hidden channel
    b = pid_hb // hidden_dim
    d = pid_hb % hidden_dim

    # base pointers for this (b,d)
    left_base  = left_ptr  + b * stride_bl + d * stride_hl
    right_base = right_ptr + b * stride_br + d * stride_hr
    out_base   = out_ptr   + b * stride_bo + d * stride_ho

    # tile origins
    i_start = pid_m * BLOCK_M
    j_start = pid_n * BLOCK_N

    # coordinate vectors
    i_idx = i_start + tl.arange(0, BLOCK_M)
    j_idx = j_start + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_i = i_idx < N
    mask_j = j_idx < N

    # accumulator in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # ------------------------------------------------------------------
    # reduction over k
    # ------------------------------------------------------------------
    for k in range(0, N, BLOCK_K):
        k_idx = k + tl.arange(0, BLOCK_K)
        mask_k = k_idx < N

        # ---- left tile (M x K) -------------------------------------------------
        left_ptrs = left_base + (i_idx[:, None] * stride_il + k_idx[None, :] * stride_kl)
        left_tile = tl.load(left_ptrs,
                            mask=mask_i[:, None] & mask_k[None, :],
                            other=0.0).to(tl.float32)

        # ---- right tile (K x N) ------------------------------------------------
        # treat right as (K rows, N columns) = right[b, j, k, d]  with k as row
        right_ptrs = right_base + (k_idx[:, None] * stride_kr + j_idx[None, :] * stride_ir)
        right_tile = tl.load(right_ptrs,
                             mask=mask_k[:, None] & mask_j[None, :],
                             other=0.0).to(tl.float32)

        # ---- accumulate ---------------------------------------------------------
        acc += tl.dot(left_tile, right_tile)   # (M,K)·(K,N) -> (M,N)

    # ------------------------------------------------------------------
    # write back the result
    # ------------------------------------------------------------------
    out_ptrs = out_base + (i_idx[:, None] * stride_io + j_idx[None, :] * stride_jo)
    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])

# ----------------------------------------------------------------------
# Public entry point
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Compute the outgoing TriMul operation.

    Parameters
    ----------
    data : tuple
        (input_tensor, mask, weights, config)
        - input_tensor : torch.Tensor [B, N, N, dim] (float32)
        - mask         : torch.Tensor [B, N, N]   (bool/int/float)
        - weights      : dict of model parameters (torch.Tensor)
        - config       : dict with keys "dim", "hidden_dim", "nomask", ...

    Returns
    -------
    torch.Tensor
        Output tensor of shape [B, N, N, dim] (float32)
    """
    # ------------------------------------------------------------------
    # unpack arguments
    # ------------------------------------------------------------------
    x, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    eps = 1e-5

    # ------------------------------------------------------------------
    # model weights
    # ------------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    to_out_weight      = weights["to_out.weight"]

    # ------------------------------------------------------------------
    # 1️⃣ Input LayerNorm
    # ------------------------------------------------------------------
    x_norm = torch.nn.functional.layer_norm(
        x,
        (dim,),
        weight=norm_weight,
        bias=norm_bias,
        eps=eps,
    )  # [B, N, N, dim]

    # ------------------------------------------------------------------
    # 2️⃣ Linear projections (float32)
    # ------------------------------------------------------------------
    left  = torch.nn.functional.linear(x_norm, left_proj_weight)   # [B,N,N,hidden]
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # ------------------------------------------------------------------
    # 3️⃣ Optional mask (applied before gating)
    # ------------------------------------------------------------------
    if not config.get("nomask", True):
        # mask is broadcast over the hidden dimension
        mask_f = mask.to(x.dtype).unsqueeze(-1)       # [B,N,N,1]
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4️⃣ Gating (sigmoid) – still in float32
    # ------------------------------------------------------------------
    left_gate  = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate   = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    # apply gates
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5️⃣ Cast to fp16 for the heavy O(N³) kernel
    # ------------------------------------------------------------------
    left  = left.to(torch.float16)
    right = right.to(torch.float16)

    # ------------------------------------------------------------------
    # 6️⃣ Allocate output buffer (float32)
    # ------------------------------------------------------------------
    out = torch.empty_like(left, dtype=torch.float32, device=left.device)  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 7️⃣ Triton kernel launch parameters
    # ------------------------------------------------------------------
    B, N, _, _ = left.shape
    # strides are in elements (not bytes)
    stride_bl, stride_il, stride_kl, stride_hl = left.stride()
    stride_br, stride_ir, stride_kr, stride_hr = right.stride()
    stride_bo, stride_io, stride_jo, stride_ho = out.stride()

    # block sizes – tuned for H100 Tensor‑Cores
    BLOCK_M = 128
    BLOCK_N = 128
    BLOCK_K = 32

    grid = (
        (N + BLOCK_M - 1) // BLOCK_M,   # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,   # j‑tiles
        B * hidden_dim,                 # batch × hidden
    )

    trimul_out_kernel[grid](
        left, right, out,
        N,
        stride_bl, stride_il, stride_kl, stride_hl,
        stride_br, stride_ir, stride_kr, stride_hr,
        stride_bo, stride_io, stride_jo, stride_ho,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K,
        hidden_dim=hidden_dim,
        # Triton launch kwargs for high performance on H100
        num_warps=8,
        num_stages=4,
    )

    # ------------------------------------------------------------------
    # 8️⃣ Output LayerNorm over hidden dimension
    # ------------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
        eps=eps,
    )  # [B,N,N,hidden]

    # ------------------------------------------------------------------
    # 9️⃣ Apply output gate (float32)
    # ------------------------------------------------------------------
    out = out * out_gate

    # ------------------------------------------------------------------
    # 🔟 Final linear projection back to original dimension
    # ------------------------------------------------------------------
    output = torch.nn.functional.linear(out, to_out_weight)   # [B,N,N,dim]

    return output
