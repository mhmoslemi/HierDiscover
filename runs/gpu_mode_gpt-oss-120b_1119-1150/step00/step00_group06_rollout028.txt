"""
Triton kernel for the core Tensor Contraction of the TriMul “outgoing” update.

Given
    left  : Tensor[B, N, N, H]   (output of left_proj + left_gate)
    right : Tensor[B, N, N, H]   (output of right_proj + right_gate)

the operator computes for every batch b, position i, j and hidden channel h:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

which is exactly a batched matrix‑multiply (Gram) for each hidden channel:
    out_h = left_h @ right_hᵀ   where left_h, right_h ∈ ℝ^{B×N×N}

The kernel tiles the output matrix with BLOCK_I × BLOCK_J blocks and
accumulates over K in steps of BLOCK_K.  Each program instance also
processes a single hidden channel (so the 2‑nd launch axis is B × H).
The implementation uses `tl.dot` for the inner (I×K)·(K×J) multiplication,
relying on the GPU’s WMMA/TF32 units for high throughput.  The rest of
the TriMul pipeline (LayerNorms, linear projections, sigmoids and the final
linear) is expressed in PyTorch for simplicity; only the O(N³) contraction
is off‑loaded to Triton.

The final output is a float‑32 tensor of shape [B, N, N, dim].
"""

    B, N, H,
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_j, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr
):
    # --------------------------------------------------------------
    #  Program IDs
    # --------------------------------------------------------------
    pid_i = tl.program_id(0)             # tile over i dimension
    pid_j = tl.program_id(1)             # tile over j dimension
    pid_hb = tl.program_id(2)            # combined batch * hidden channel

    # --------------------------------------------------------------
    #  Decode indices
    # --------------------------------------------------------------
    i0 = pid_i * BLOCK_I
    j0 = pid_j * BLOCK_J
    b = pid_hb // H                     # batch index
    h = pid_hb % H                      # hidden channel index

    # Offsets for the output tile
    offs_i = i0 + tl.arange(0, BLOCK_I)
    offs_j = j0 + tl.arange(0, BLOCK_J)

    mask_i = offs_i < N
    mask_j = offs_j < N

    # Accumulator for the output tile (float32)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # --------------------------------------------------------------
    #  Loop over K dimension in blocks of size BLOCK_K
    # --------------------------------------------------------------
    for k_base in range(0, N, BLOCK_K):
        cur_k = tl.minimum(BLOCK_K, N - k_base)
        offs_k = k_base + tl.arange(0, BLOCK_K)

        mask_k = offs_k < N

        # ------------------------------------------------------------------
        # Load left block: shape (BLOCK_I, BLOCK_K)
        #   address = b*stride_left_b + i*stride_left_i + k*stride_left_k + h*stride_left_h
        # ------------------------------------------------------------------
        left_ptrs = (
            left_ptr
            + b * stride_left_b
            + offs_i[:, None] * stride_left_i
            + offs_k[None, :] * stride_left_k
            + h * stride_left_h
        )
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None] & mask_k[None, :],
            other=0.0
        )  # (BLOCK_I, BLOCK_K)

        # ------------------------------------------------------------------
        # Load right block transposed: shape (BLOCK_K, BLOCK_J)
        #   we need rightᵀ so rows correspond to k, columns to j
        #   address = b*stride_right_b + k*stride_right_k + j*stride_right_j + h*stride_right_h
        # ------------------------------------------------------------------
        right_ptrs = (
            right_ptr
            + b * stride_right_b
            + offs_k[:, None] * stride_right_k
            + offs_j[None, :] * stride_right_j
            + h * stride_right_h
        )
        right = tl.load(
            right_ptrs,
            mask=mask_k[:, None] & mask_j[None, :],
            other=0.0
        )  # (BLOCK_K, BLOCK_J)

        # ------------------------------------------------------------------
        # Matrix multiply for this K‑tile and accumulate
        #   (BLOCK_I, BLOCK_K) @ (BLOCK_K, BLOCK_J) -> (BLOCK_I, BLOCK_J)
        # ------------------------------------------------------------------
        acc += tl.dot(left, right)

    # --------------------------------------------------------------
    #  Write the result back to global memory
    # --------------------------------------------------------------
    out_ptrs = (
        out_ptr
        + b * stride_out_b
        + offs_i[:, None] * stride_out_i
        + offs_j[None, :] * stride_out_j
        + h * stride_out_h
    )
    tl.store(
        out_ptrs,
        acc,
        mask=mask_i[:, None] & mask_j[None, :]
    )


# ----------------------------------------------------------------------
# Entry‑point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul operator implemented with a
    custom Triton kernel for the heavy contraction.

    Args:
        data: Tuple (input_tensor, mask, weights, config)
            - input_tensor: Tensor[bs, N, N, dim]  (float32)
            - mask: Tensor[bs, N, N] (bool or float, 0/1)
            - weights: dict of model weights (see code for exact keys)
            - config: dict with keys "dim", "hidden_dim", "nomask"

    Returns:
        Tensor[bs, N, N, dim] – the processed pair representation.
    """
    # ------------------------------------------------------------------
    # Unpack arguments
    # ------------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    bs, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ------------------------------------------------------------------
    # Helper to fetch a weight and ensure it is on the right device/dtype
    # ------------------------------------------------------------------
    def get(name):
        return weights[name].to(device=device, dtype=dtype)

    # ------------------------------------------------------------------
    # 1. LayerNorm over the last dimension (dim)
    # ------------------------------------------------------------------
    norm_weight = get("norm.weight")
    norm_bias   = get("norm.bias")
    x = F.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias)

    # ------------------------------------------------------------------
    # 2. Linear projections (no bias)
    # ------------------------------------------------------------------
    left_proj_weight  = get("left_proj.weight")   # (hidden_dim, dim)
    right_proj_weight = get("right_proj.weight")  # (hidden_dim, dim)

    left = F.linear(x, left_proj_weight)   # (B,N,N,hidden_dim)
    right = F.linear(x, right_proj_weight) # (B,N,N,hidden_dim)

    # ------------------------------------------------------------------
    # 3. Apply mask if required (mask shape: [B,N,N])
    # ------------------------------------------------------------------
    if not nomask:
        # mask broadcasted over hidden dimension
        mask_f = mask.to(dtype=dtype).unsqueeze(-1)   # (B,N,N,1)
        left = left * mask_f
        right = right * mask_f

    # ------------------------------------------------------------------
    # 4. Gating (sigmoid of linear projections)
    # ------------------------------------------------------------------
    left_gate_weight  = get("left_gate.weight")
    right_gate_weight = get("right_gate.weight")
    out_gate_weight   = get("out_gate.weight")

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # (B,N,N,hidden_dim)
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # (B,N,N,hidden_dim)
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # (B,N,N,hidden_dim)

    # Gated activations
    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 5. Core contraction using the Triton kernel
    # ------------------------------------------------------------------
    # Ensure tensors are contiguous for correct stride extraction
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)   # shape [B,N,N,hidden_dim]

    # Extract strides (in elements, not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_h = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_h = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_h = out.stride()

    # Tile sizes – chosen to fit comfortably in registers on H100
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32

    grid = (
        (N + BLOCK_I - 1) // BLOCK_I,          # i‑tiles
        (N + BLOCK_J - 1) // BLOCK_J,          # j‑tiles
        bs * hidden_dim                        # batch × hidden‑channel
    )

    trimul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=bs,
        N=N,
        H=hidden_dim,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_h=stride_left_h,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_h=stride_right_h,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_h=stride_out_h,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        num_warps=4,
    )

    # ------------------------------------------------------------------
    # 6. Post‑process: LayerNorm on hidden_dim, gating, final linear
    # ------------------------------------------------------------------
    to_out_norm_weight = get("to_out_norm.weight")
    to_out_norm_bias   = get("to_out_norm.bias")
    out = F.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight,
                       bias=to_out_norm_bias)

    out = out * out_gate                       # broadcast over hidden dim

    to_out_weight = get("to_out.weight")      # (dim, hidden_dim)
    output = F.linear(out, to_out_weight)     # final shape (B,N,N,dim)

    return output
