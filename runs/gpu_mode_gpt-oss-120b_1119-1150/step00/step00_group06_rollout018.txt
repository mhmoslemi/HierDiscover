        left_ptr, right_ptr, out_ptr,
        B, N, H,
        stride_left_b, stride_left_i, stride_left_k, stride_left_h,
        stride_right_b, stride_right_i, stride_right_k, stride_right_h,
        stride_out_b, stride_out_i, stride_out_j, stride_out_h,
        BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
):
    pid_bh = tl.program_id(0)  # up to B*H
    pid_i = tl.program_id(1)  # row tile
    pid_j = tl.program_id(2)  # col tile

    b = pid_bh // H
    h = pid_bh % H

    # compute tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)
    offs_k = tl.arange(0, BLOCK_K)

    # mask out-of-bounds
    mask_i = offs_i < N
    mask_j = offs_j < N

    # compute base pointers with b and h offsets:
    left_base = left_ptr + b * stride_left_b + h * stride_left_h
    right_base = right_ptr + b * stride_right_b + h * stride_right_h
    out_base = out_ptr + b * stride_out_b + h * stride_out_h

    # loads
    a_ptrs = left_base + offs_i[:, None] * stride_left_i + offs_k[None, :] * stride_left_k
    b_ptrs = right_base + offs_j[None, :] * stride_right_i + offs_k[:, None] * stride_right_k

"""
TriMul outgoing kernel (AlphaFold3 style) implemented with Triton.

The core operation is:
    out[b, i, j, h] = Σ_k left[b, i, k, h] * right[b, j, k, h]

where left/right are the projected (and gated) tensors of shape
[B, N, N, H] (H = hidden_dim). This is a batched matrix multiplication
(left @ rightᵀ) for each hidden head. The heavy N³ reduction is performed
in a fused Triton kernel for maximum throughput on H100 GPUs.
All other parts of the TriMul module (LayerNorms, linear projections,
gates, final norm/gate/linear) are handled in PyTorch.
"""

import torch
import triton
import triton.language as tl


@triton.jit
def tri_mul_kernel(
    left_ptr,               # const half*  (B,N,N,H)
    right_ptr,              # const half*  (B,N,N,H)
    out_ptr,                # float*       (B,N,N,H)
    B, N, H,                # int64 scalars
    stride_left_b, stride_left_i, stride_left_k, stride_left_h,
    stride_right_b, stride_right_i, stride_right_k, stride_right_h,
    stride_out_b, stride_out_i, stride_out_j, stride_out_h,
    BLOCK_M: tl.constexpr,  # rows of output tile (i dimension)
    BLOCK_N: tl.constexpr,  # cols of output tile (j dimension)
    BLOCK_K: tl.constexpr,  # reduction tile size (k dimension)
):
    # program IDs:
    #   pid_bh : combined batch*head index
    #   pid_i  : tile index along i (rows)
    #   pid_j  : tile index along j (cols)
    pid_bh = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # derive batch and head indices from combined pid
    b = pid_bh // H
    h = pid_bh % H

    # tile offsets
    offs_i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # masks for out‑of‑bounds rows / cols
    mask_i = offs_i < N
    mask_j = offs_j < N

    # accumulate in FP32
    acc = tl.zeros((BLOCK_M, BLOCK_N), dtype=tl.float32)

    # loop over k dimension
    for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
        k_offset = k_tile * BLOCK_K
        offs_k = k_offset + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # pointers to the current tiles of left and rightᵀ
        # left: [i, k]   -> stride_i for rows, stride_k for cols
        a_ptr = (left_ptr
                 + b * stride_left_b + h * stride_left_h
                 + offs_i[:, None] * stride_left_i
                 + offs_k[None, :] * stride_left_k)

        # rightᵀ: [k, j]   -> stride_k for rows, stride_i for cols
        b_ptr = (right_ptr
                 + b * stride_right_b + h * stride_right_h
                 + offs_k[:, None] * stride_right_k
                 + offs_j[None, :] * stride_right_i)

        # load tiles with proper masking
        a = tl.load(a_ptr,
                    mask=mask_i[:, None] & mask_k[None, :],
                    other=0.0)
        b = tl.load(b_ptr,
                    mask=mask_k[:, None] & mask_j[None, :],
                    other=0.0)

        # fused dot (BLOCK_M x BLOCK_K) * (BLOCK_K x BLOCK_N) -> (BLOCK_M x BLOCK_N)
        acc += tl.dot(a, b)

    # write the result back to global memory
    out_ptrs = (out_ptr
                + b * stride_out_b + h * stride_out_h
                + offs_i[:, None] * stride_out_i
                + offs_j[None, :] * stride_out_j)

    tl.store(out_ptrs, acc, mask=mask_i[:, None] & mask_j[None, :])


def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul module.

    Args:
        data: tuple (input_tensor, mask, weights, config)
            - input_tensor: torch.Tensor of shape [B, N, N, dim] (float32)
            - mask: torch.Tensor of shape [B, N, N] (bool or float) or None
            - weights: dict of model weight tensors
            - config: dict with keys "dim", "hidden_dim", "nomask" etc.

    Returns:
        torch.Tensor of shape [B, N, N, dim] (float32)
    """
    input_tensor, mask, weights, config = data

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # -----------------------------------------------------------------
    # LayerNorm over the input channels
    # -----------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=dim,
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
        eps=1e-5,
    )

    # -----------------------------------------------------------------
    # Linear projections (no bias) producing hidden_dim channels
    # -----------------------------------------------------------------
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # -----------------------------------------------------------------
    # Optional mask (broadcast along hidden_dim)
    # -----------------------------------------------------------------
    if not nomask and mask is not None:
        # ensure mask is float and broadcastable
        mask_exp = mask.to(input_tensor.dtype).unsqueeze(-1)  # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # -----------------------------------------------------------------
    # Gating (sigmoid), also hidden_dim sized
    # -----------------------------------------------------------------
    left_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["left_gate.weight"])
    )
    right_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["right_gate.weight"])
    )
    out_gate = torch.sigmoid(
        torch.nn.functional.linear(x, weights["out_gate.weight"])
    )

    left = left * left_gate
    right = right * right_gate

    # -----------------------------------------------------------------
    # Cast to half for the heavy N³ reduction kernel
    # -----------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)

    B, N, _, _ = left.shape
    # Output buffer for the kernel (FP32 accumulator)
    out = torch.empty((B, N, N, hidden_dim), dtype=torch.float32, device=input_tensor.device)

    # -----------------------------------------------------------------
    # Triton kernel launch configuration
    # -----------------------------------------------------------------
    BLOCK_M = 64
    BLOCK_N = 64
    BLOCK_K = 64

    grid = (
        B * hidden_dim,                          # combined batch*head dimension
        (N + BLOCK_M - 1) // BLOCK_M,            # i‑tiles
        (N + BLOCK_N - 1) // BLOCK_N,            # j‑tiles
    )

    # strides for the left/right tensors (as produced by torch.nn.functional.linear)
    # shape = [B, N, N, hidden_dim]
    s_left_b, s_left_i, s_left_k, s_left_h = left.stride()
    s_right_b, s_right_i, s_right_k, s_right_h = right.stride()
    s_out_b, s_out_i, s_out_j, s_out_h = out.stride()

    tri_mul_kernel[grid](
        left_h,
        right_h,
        out,
        B,
        N,
        hidden_dim,
        s_left_b,
        s_left_i,
        s_left_k,
        s_left_h,
        s_right_b,
        s_right_i,
        s_right_k,
        s_right_h,
        s_out_b,
        s_out_i,
        s_out_j,
        s_out_h,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
    )

    # -----------------------------------------------------------------
    # Post‑processing: LayerNorm on hidden_dim, gating, final linear
    # -----------------------------------------------------------------
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=hidden_dim,
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
        eps=1e-5,
    )
    out = out * out_gate  # apply output gate (float32)

    # final projection back to dim
    output = torch.nn.functional.linear(out, weights["to_out.weight"])

    return output
