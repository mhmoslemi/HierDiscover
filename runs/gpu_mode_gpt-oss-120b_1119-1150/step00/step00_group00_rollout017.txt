    input_tensor, mask, weights, config = data

    # parse dims
    bs, N, _, dim = input_tensor.shape
    hidden_dim = config['hidden_dim']
    # get weights: weights dictionary with .weight and .bias as needed.

    # compute layernorm for x
    # Use torch.nn.functional.layer_norm: compute out = (x - mean) / sqrt(var+eps) * weight + bias
    eps = 1e-6
    # Input shape (bs,N,N,dim)
    x = input_tensor
    # flatten last dim for layernorm: compute per element across dim. We'll use torch.nn.functional.layer_norm.
    x_norm = torch.nn.functional.layer_norm(x, normalized_shape=(dim,), weight=norm_weight, bias=norm_bias, eps=eps)

    # linear projections: left = torch.nn.functional.linear(x_norm, left_proj_weight) (bias None)
    left = torch.nn.functional.linear(x_norm, left_proj_weight)
    right = torch.nn.functional.linear(x_norm, right_proj_weight)

    # optionally mask: if mask is not None and not config['nomask']; multiply mask positions. mask shape [bs,N,N] (bool or float). The PyTorch code uses mask = mask.unsqueeze(-1), left=left*mask, right=right*mask.
    if mask is not None:
        mask_unsq = mask.unsqueeze(-1).to(left.dtype)
        left = left * mask_unsq
        right = right * mask_unsq

    # gates:
    left_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x_norm, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # Cast left/right to float16 for kernel
    left_fp16 = left.to(torch.float16)
    right_fp16 = right.to(torch.float16)

    # Allocate output tensor
    out = torch.empty_like(left_fp16)  # shape [bs,N,N,hidden_dim] dtype float16

    # Launch Triton kernel
    import triton
    import triton.language as tl

    # compute strides
    # left_fp16 is contiguous: stride0 = N*N*H, stride1 = N*H, stride2 = H, stride3 = 1
    stride_b = left_fp16.stride(0)
    stride_i = left_fp16.stride(1)
    stride_k = left_fp16.stride(2)
    stride_h = left_fp16.stride(3)

    # grid dims
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16  # tile of hidden dim
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), bs)

    # launch kernel
    trimul_kernel[grid](
        left_fp16, right_fp16, out,
        bs, N, hidden_dim,
        stride_l_b=stride_b, stride_l_i=stride_i, stride_l_k=stride_k, stride_l_h=stride_h,
        stride_r_b=stride_b, stride_r_i=stride_i, stride_r_k=stride_k, stride_r_h=stride_h,
        stride_o_b=stride_b, stride_o_i=stride_i, stride_o_j=stride_k, stride_o_h=stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )

    # out currently fp16; cast to float32
    out_fp32 = out.to(torch.float32)

    # Apply to_out_norm (layernorm over hidden_dim)
    out_norm = torch.nn.functional.layer_norm(out_fp32, normalized_shape=(hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)

    # Multiply by out_gate
    out_gate = out_gate  # shape [bs,N,N,hidden_dim]
    out_norm = out_norm * out_gate

    # linear to final dim
    final = torch.nn.functional.linear(out_norm, to_out_weight)  # shape [bs,N,N,dim]

    return final

import torch
import triton
import triton.language as tl

@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    pid_b = tl.program_id(2)
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # Compute base indices
    i = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    # Masks for i, j
    mask_i = i < N
    mask_j = j < N

    # Loop over hidden dim tiles
    for d_tile in range(0, tl.cdiv(H, BLOCK_H)):
        d_offset = d_tile * BLOCK_H + tl.arange(0, BLOCK_H)
        mask_d = d_offset < H

        # Accumulator (float32)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # Loop over k dimension
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k_base = k_tile * BLOCK_K
            for kk in range(0, BLOCK_K):
                k = k_base + kk
                mask_k = k < N

                    pid_b * stride_l_b + \
                    i[:, None] * stride_l_i + \

def custom_kernel(data):
    """
    Triton-accelerated forward pass of the outgoing TriMul operator.

    This kernel implements the heavy triple-sum operation
    out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    where left/right are the projected and gated pair representations.
    The surrounding LayerNorm, gating and linear layers are performed with
    PyTorch; only the O(N^3) core is fused in a Triton kernel.
    """
    input_tensor, mask, weights, config = data
    # extract dims
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6
    # LayerNorm
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(input_tensor, (dim,), weight=norm_weight, bias=norm_bias, eps=eps)
    # Linear projections (no bias)
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])
    # Apply mask if present
    if not config.get("nomask", False):
        # mask shape [B,N,N] -> broadcast
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_
    # Gating
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, weights["out_gate.weight"]))
    left = left * left_gate
    right = right * right_gate
    # Cast to half for kernel
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    # Allocate output in half
    out_h = torch.empty((B, N, N, hidden_dim), dtype=torch.float16, device=left.device)
    # Compute strides
    stride_b = left_h.stride(0)
    stride_i = left_h.stride(1)   # stride for i dimension
    stride_k = left_h.stride(2)   # stride for k dimension (contiguous over hidden dim)
    stride_h = left_h.stride(3)   # stride for hidden dim (1)
    # Kernel launch config
    BLOCK_M = 32
    BLOCK_N = 32
    BLOCK_K = 32
    BLOCK_H = 16
    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)
    trimul_outgoing_kernel[grid](
        left_h, right_h, out_h,
        B, N, hidden_dim,
        stride_l_b=stride_b, stride_l_i=stride_i, stride_l_k=stride_k, stride_l_h=stride_h,
        stride_r_b=stride_b, stride_r_i=stride_i, stride_r_k=stride_k, stride_r_h=stride_h,
        stride_o_b=stride_b, stride_o_i=stride_i, stride_o_j=stride_k, stride_o_h=stride_h,
        BLOCK_M=BLOCK_M, BLOCK_N=BLOCK_N, BLOCK_K=BLOCK_K, BLOCK_H=BLOCK_H
    )
    # Convert back to fp32 for remaining ops
    out = out_h.to(torch.float32)
    # to_out_norm
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(out, (hidden_dim,), weight=to_out_norm_weight, bias=to_out_norm_bias, eps=eps)
    # Multiply by out_gate
    out = out * out_gate
    # Final linear
    final = torch.nn.functional.linear(out, weights["to_out.weight"])
    return final

    left_ptr, right_ptr, out_ptr,

    pid_j = tl.program_id(1)

    # Iterate over hidden tiles

"""
TriMul (outgoing) – Triton‑accelerated forward pass.

The heavy O(N³·H) computation
    out[b,i,j,h] = Σ_k left[b,i,k,h] * right[b,j,k,h]
is performed by a custom Triton kernel.  All surrounding
operations (LayerNorm, linear projections, gates) are
implemented with PyTorch for simplicity.  The kernel works
for arbitrary batch size (1‑2), sequence length (≤1024) and
hidden dimension (≤384).  It tiles the i‑, j‑ and hidden‑
dimensions and accumulates in float32 before writing the
result in float16 (later cast back to float32 for the remaining
layers).

The kernel launch grid:
    (ceil(N/BLOCK_M), ceil(N/BLOCK_N), batch)
with BLOCK_M/N = 16, BLOCK_K = 32 and BLOCK_H = 16 – a
register‑friendly configuration that fits comfortably on an
H100.
"""

import torch
import triton
import triton.language as tl


# ------------------------------------------------------------
# Triton kernel: core triple‑sum (the "einsum" in the PyTorch code)
# ------------------------------------------------------------
@triton.jit
def trimul_outgoing_kernel(
    left_ptr,               # fp16 * [B,N,N,H]
    right_ptr,              # fp16 * [B,N,N,H]
    out_ptr,                # fp16 * [B,N,N,H]
    B, N, H,                # runtime sizes
    stride_l_b, stride_l_i, stride_l_k, stride_l_h,
    stride_r_b, stride_r_i, stride_r_k, stride_r_h,
    stride_o_b, stride_o_i, stride_o_j, stride_o_h,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_H: tl.constexpr
):
    """Compute for a single (b,i,j) tile:
       out[b,i_start:i_end, j_start:j_end, h_tile] =
          Σ_k left[b,i_start:i_end, k, h_tile] *
                right[b,j_start:j_end, k, h_tile] .
    The loop over hidden dimension is tiled (BLOCK_H).
    """
    # program ids
    pid_b = tl.program_id(2)                     # batch
    pid_i = tl.program_id(0)                     # i‑tile
    pid_j = tl.program_id(1)                     # j‑tile

    # i / j coordinates for this tile
    i_range = pid_i * BLOCK_M + tl.arange(0, BLOCK_M)
    j_range = pid_j * BLOCK_N + tl.arange(0, BLOCK_N)

    mask_i = i_range < N
    mask_j = j_range < N

    # ------------------------------------------------------------------
    # Loop over hidden‑dimension tiles (BLOCK_H)
    # ------------------------------------------------------------------
    for d_tile in range(0, tl.cdiv(H, BLOCK_H)):
        d_start = d_tile * BLOCK_H
        d_range = d_start + tl.arange(0, BLOCK_H)
        mask_d = d_range < H

        # accumulator in float32: (M, N, H_tile)
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_H), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Loop over the K dimension (the summed axis)
        # ------------------------------------------------------------------
        for k_tile in range(0, tl.cdiv(N, BLOCK_K)):
            k_start = k_tile * BLOCK_K
            for kk in range(0, BLOCK_K):               # unrolled, BLOCK_K is constexpr
                k = k_start + kk
                mask_k = k < N

                # ------------------------------
                # Load left[b,i,k,:]  -> (M, H_tile)
                # ------------------------------
                left_off = (
                    pid_b * stride_l_b +
                    i_range[:, None] * stride_l_i +
                    k * stride_l_k +
                    d_range[None, :] * stride_l_h
                )
                left = tl.load(left_off,
                               mask=mask_i[:, None] & mask_k & mask_d,
                               other=0.0).to(tl.float32)      # (M, H_tile)

                # ------------------------------
                # Load right[b,j,k,:] -> (N, H_tile)
                # ------------------------------
                right_off = (
                    pid_b * stride_r_b +
                    j_range[:, None] * stride_r_i +
                    k * stride_r_k +
                    d_range[None, :] * stride_r_h
                )
                right = tl.load(right_off,
                                mask=mask_j[:, None] & mask_k & mask_d,
                                other=0.0).to(tl.float32)      # (N, H_tile)

                # ------------------------------
                # Outer‑product accumulate:
                #   acc[i, j, h] += left[i, h] * right[j, h]
                # ------------------------------
                acc += left[:, None, :] * right[None, :, :]   # (M, N, H_tile)

        # --------------------------------------------------------------
        # Store the completed tile (cast back to fp16)
        # --------------------------------------------------------------
        out_off = (
            pid_b * stride_o_b +
            i_range[:, None, None] * stride_o_i +
            j_range[None, :, None] * stride_o_j +
            d_range[None, None, :] * stride_o_h
        )
        tl.store(out_off,
                 acc.to(tl.float16),
                 mask=mask_i[:, None, None] &
                      mask_j[None, :, None] &
                      mask_d[None, None, :])


# ------------------------------------------------------------
# Entry point required by the evaluation harness
# ------------------------------------------------------------
def custom_kernel(data):
    """Forward pass of the outgoing TriMul operator.

    Arguments
    ---------
    data : tuple
        (input_tensor, mask, weights, config)

    Returns
    -------
    torch.Tensor
        Tensor of shape [bs, seq_len, seq_len, dim] (float32)
    """
    # ------------------------------------------------------------
    # Unpack inputs
    # ------------------------------------------------------------
    input_tensor, mask, weights, config = data
    # Shapes
    B, N, _, dim = input_tensor.shape
    hidden_dim = config["hidden_dim"]
    eps = 1e-6

    # ------------------------------------------------------------
    # Load weights
    # ------------------------------------------------------------
    norm_w = weights["norm.weight"]
    norm_b = weights["norm.bias"]
    left_proj_w = weights["left_proj.weight"]
    right_proj_w = weights["right_proj.weight"]
    left_gate_w = weights["left_gate.weight"]
    right_gate_w = weights["right_gate.weight"]
    out_gate_w = weights["out_gate.weight"]
    to_out_norm_w = weights["to_out_norm.weight"]
    to_out_norm_b = weights["to_out_norm.bias"]
    to_out_w = weights["to_out.weight"]

    # ------------------------------------------------------------
    # 1) LayerNorm over the channel dimension
    # ------------------------------------------------------------
    x = torch.nn.functional.layer_norm(
        input_tensor,
        (dim,),
        weight=norm_w,
        bias=norm_b,
        eps=eps,
    )  # [B,N,N,dim]

    # ------------------------------------------------------------
    # 2) Linear projections (no bias)
    # ------------------------------------------------------------
    left = torch.nn.functional.linear(x, left_proj_w)   # [B,N,N,hidden_dim]
    right = torch.nn.functional.linear(x, right_proj_w)

    # ------------------------------------------------------------
    # 3) Optional mask (broadcasted on the channel axis)
    # ------------------------------------------------------------
    if not config.get("nomask", False):
        # mask: [B,N,N] -> [B,N,N,1]
        mask_ = mask.unsqueeze(-1).type_as(left)
        left = left * mask_
        right = right * mask_

    # ------------------------------------------------------------
    # 4) Gating (sigmoid of a linear projection of the normalized input)
    # ------------------------------------------------------------
    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_w))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_w))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_w))

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------
    # 5) Cast to half‑precision and allocate output buffer
    # ------------------------------------------------------------
    left_h = left.to(torch.float16)
    right_h = right.to(torch.float16)
    out_h = torch.empty((B, N, N, hidden_dim),
                        dtype=torch.float16,
                        device=left_h.device)

    # ------------------------------------------------------------
    # 6) Compute strides (contiguous layout)
    # ------------------------------------------------------------
    stride_b = left_h.stride(0)   # B‑stride
    stride_i = left_h.stride(1)   # i‑stride (N*H)
    stride_k = left_h.stride(2)   # k‑stride (H)
    stride_h = left_h.stride(3)   # h‑stride (1)

    # ------------------------------------------------------------
    # 7) Triton kernel launch
    # ------------------------------------------------------------
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 32
    BLOCK_H = 16

    grid = (triton.cdiv(N, BLOCK_M),   # i‑tiles
            triton.cdiv(N, BLOCK_N),   # j‑tiles
            B)                         # batch

    trimul_outgoing_kernel[grid](
        left_h,          # left pointer
        right_h,         # right pointer
        out_h,           # output pointer
        B, N, hidden_dim,
        stride_b, stride_i, stride_k, stride_h,   # left strides
        stride_b, stride_i, stride_k, stride_h,   # right strides (identical)
        stride_b, stride_i, stride_k, stride_h,   # output strides (B,N,N,H)
        BLOCK_M, BLOCK_N, BLOCK_K, BLOCK_H
    )

    # ------------------------------------------------------------
    # 8) Remaining post‑processing in FP32
    # ------------------------------------------------------------
    out = out_h.to(torch.float32)                     # [B,N,N,H]

    # LayerNorm on the hidden dimension
    out = torch.nn.functional.layer_norm(
        out,
        (hidden_dim,),
        weight=to_out_norm_w,
        bias=to_out_norm_b,
        eps=eps,
    )

    # Apply out‑gate
    out = out * out_gate

    # Final linear projection back to the original channel dimension
    final = torch.nn.functional.linear(out, to_out_w)   # [B,N,N,dim]

    return final
