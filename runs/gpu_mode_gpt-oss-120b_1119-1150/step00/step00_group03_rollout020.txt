def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # compute base row/col coords
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # create range vectors
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_k_base = 0  # start of k loop
    # compute d offset vectors
    offs_d = tl.arange(0, BLOCK_D)

    # mask for out-of-bound i/j
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # initialize accumulators
    # shape (BLOCK_D, BLOCK_I, BLOCK_J)
    acc = tl.zeros((BLOCK_D, BLOCK_I, BLOCK_J), dtype=tl.float32)

    # loop over k dimension
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # Load left block: shape (BLOCK_I, BLOCK_K, BLOCK_D)

                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + offs_d[None, None, :] * stride_l_d
        left = tl.load(left_ptrs, mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :], other=0.0).to(tl.float32)

        for d in range(0, BLOCK_D):

            a = left[:, :, d]
            # shape (BLOCK_J, BLOCK_K)

            # compute dot across k, producing (BLOCK_I, BLOCK_J)
            # tl.dot expects (M,K) and (K,N). So need to transpose b.

            acc[d] = acc_d

    for d in range(0, BLOCK_D):

                   + offs_i[:, None] * stride_o_i \
                   + offs_j[None, :] * stride_o_j \
                   + (offs_d[d]) * stride_o_d
        # we need to broadcast acc[d] shape (BLOCK_I, BLOCK_J)
        # Use tl.store with mask for i/j/d
        tl.store(out_ptrs, acc[d], mask=mask_i[:, None] & mask_j[None, :])

def trimul_triton(left, right):

    B,N,N_,D = left.shape
    assert N == N_

    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)  # shape same as left (B,N,N,D)

import torch
import torch.nn.functional as F
import triton
import triton.language as tl

@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr
):
    '''
    Compute out[b,i,j,d] = sum_k left[b,i,k,d] * right[b,j,k,d]
    with tiling over i, j, d and k.
    '''
    pid_b = tl.program_id(0)
    pid_i = tl.program_id(1)
    pid_j = tl.program_id(2)

    # base coordinates
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    # coordinate vectors
    offs_i = i_base + tl.arange(0, BLOCK_I)
    offs_j = j_base + tl.arange(0, BLOCK_J)
    offs_d = tl.arange(0, BLOCK_D)

    # bounds
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # accumulator: shape (BLOCK_I, BLOCK_J, BLOCK_D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)
        mask_k = offs_k < N

        # left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + offs_d[None, None, :] * stride_l_d
        left = tl.load(left_ptrs,
                       mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                       other=0.0).to(tl.float32)

        # right: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        right_ptrs = right_ptr + pid_b * stride_r_b \
                     + offs_j[:, None, None] * stride_r_j \
                     + offs_k[None, :, None] * stride_r_k \
                     + offs_d[None, None, :] * stride_r_d
        right = tl.load(right_ptrs,
                        mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
                        other=0.0).to(tl.float32)

        # accumulate per d
        for d in range(0, BLOCK_D):
            a = left[:, :, d]                     # (BLOCK_I, BLOCK_K)
            b = right[:, :, d]                    # (BLOCK_J, BLOCK_K)
            # b needs to be transposed to (BLOCK_K, BLOCK_J) for tl.dot
            acc[:, :, d] += tl.dot(a, tl.trans(b))

    # Store result
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None, None] * stride_o_i \
               + offs_j[None, :, None] * stride_o_j \
               + offs_d[None, None, :] * stride_o_d

    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)

def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton implementation of the TriMul einsum.
    left/right: float32 tensors of shape [B, N, N, D] (D=hidden_dim)
    Returns tensor of same shape.
    """
    assert left.shape == right.shape
    B, N, N2, D = left.shape
    assert N == N2
    # Ensure contiguous
    left = left.contiguous()
    right = right.contiguous()
    out = torch.empty_like(left)

    # strides
    stride_l_b, stride_l_i, stride_l_k, stride_l_d = left.stride()
    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    # tile sizes (tuned for H100)
    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden_dim is a multiple of 8 for typical configs

    grid = (B,
            (N + BLOCK_I - 1) // BLOCK_I,
            (N + BLOCK_J - 1) // BLOCK_J)

    trimul_kernel[grid](
        left,
        right,
        out,
        B, N, D,
        stride_l_b, stride_l_i, stride_l_k, stride_l_d,
        stride_r_b, stride_r_j, stride_r_k, stride_r_d,
        stride_o_b, stride_o_i, stride_o_j, stride_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out

def custom_kernel(data):
    """
    Forward pass of the TriMul (outgoing) operator using Triton for the heavy
    pairwise multiplication.  The rest of the pipeline (LayerNorm, projections,
    gates and final linear) is performed with native PyTorch ops.
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the last dimension -----
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )

    # ----- Linear projections (no bias) -----
    # shape: [B, N, N, hidden_dim]
    left = F.linear(x, weights["left_proj.weight"])
    right = F.linear(x, weights["right_proj.weight"])

    # ----- Optional mask (broadcasted over hidden dimension) -----
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1)               # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gates (sigmoid) -----
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # ----- Core TriMul operation (batched outer product) -----
    # Result shape: [B, N, N, hidden_dim]
    out = trimul_triton(left, right)

    # ----- Output layer norm (hidden_dim) -----
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # ----- Final linear projection back to original dim -----
    result = F.linear(out, weights["to_out.weight"])

    return result

"""
TriMul (outgoing) forward pass.

The heavy part of the operation is the triple‑wise product
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
which is equivalent to a batch of matrix multiplications
    out_d = left_d @ right_d.T   (for each hidden channel d).

We implement this O(N³·D) contraction with a Triton kernel that tiles the
(i, j) output plane and the hidden dimension D while looping over the
summation axis k.  The rest of the TriMul module (LayerNorm, Linear
projections, gates and the final Linear) is performed with native PyTorch
ops for simplicity.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: batched outer‑product (einsum) over the k dimension
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_l_b, stride_l_i, stride_l_k, stride_l_d,
    stride_r_b, stride_r_j, stride_r_k, stride_r_d,
    stride_o_b, stride_o_i, stride_o_j, stride_o_d,
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """
    Compute  out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]

    Tiling:
        i dimension -> BLOCK_I
        j dimension -> BLOCK_J
        hidden d   -> BLOCK_D (inner loop)
        k dimension -> BLOCK_K (outer loop)
    """
    pid_b = tl.program_id(0)      # batch index
    pid_i = tl.program_id(1)      # i‑tile index
    pid_j = tl.program_id(2)      # j‑tile index

    # ------------------------------------------------------------------
    # Coordinate vectors for the current tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J

    offs_i = i_base + tl.arange(0, BLOCK_I)    # (BLOCK_I,)
    offs_j = j_base + tl.arange(0, BLOCK_J)    # (BLOCK_J,)
    offs_d = tl.arange(0, BLOCK_D)             # (BLOCK_D,)

    # Out‑of‑bounds masks
    mask_i = offs_i < N
    mask_j = offs_j < N
    mask_d = offs_d < D

    # Accumulator: shape (BLOCK_I, BLOCK_J, BLOCK_D)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # ------------------------------------------------------------------
    # Loop over k in tiles of size BLOCK_K
    # ------------------------------------------------------------------
    num_k_tiles = tl.cdiv(N, BLOCK_K)
    for k_tile in range(0, num_k_tiles):
        k_base = k_tile * BLOCK_K
        offs_k = k_base + tl.arange(0, BLOCK_K)   # (BLOCK_K,)
        mask_k = offs_k < N

        # --------------------------------------------------------------
        # Load a block of left: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        left_ptrs = left_ptr + pid_b * stride_l_b \
                    + offs_i[:, None, None] * stride_l_i \
                    + offs_k[None, :, None] * stride_l_k \
                    + offs_d[None, None, :] * stride_l_d
        left = tl.load(
            left_ptrs,
            mask=mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)   # (BLOCK_I, BLOCK_K, BLOCK_D)

        # --------------------------------------------------------------
        # Load a block of right: shape (BLOCK_J, BLOCK_K, BLOCK_D)
        # --------------------------------------------------------------
        right_ptrs = right_ptr + pid_b * stride_r_b \
                     + offs_j[:, None, None] * stride_r_j \
                     + offs_k[None, :, None] * stride_r_k \
                     + offs_d[None, None, :] * stride_r_d
        right = tl.load(
            right_ptrs,
            mask=mask_j[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :],
            other=0.0,
        ).to(tl.float32)   # (BLOCK_J, BLOCK_K, BLOCK_D)

        # --------------------------------------------------------------
        # Accumulate the inner product for each hidden slice d
        # --------------------------------------------------------------
        for d in range(0, BLOCK_D):
            a = left[:, :, d]                # (BLOCK_I, BLOCK_K)
            b = right[:, :, d]               # (BLOCK_J, BLOCK_K)
            # tl.dot expects (M,K) @ (K,N) -> (M,N)
            acc[:, :, d] += tl.dot(a, tl.trans(b))

    # ------------------------------------------------------------------
    # Store the result tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + pid_b * stride_o_b \
               + offs_i[:, None, None] * stride_o_i \
               + offs_j[None, :, None] * stride_o_j \
               + offs_d[None, None, :] * stride_o_d

    store_mask = mask_i[:, None, None] & mask_j[None, :, None] & mask_d[None, None, :]
    tl.store(out_ptrs, acc, mask=store_mask)


def trimul_triton(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Triton‑based implementation of the core einsum in TriMul.
    left/right: [B, N, N, D] (float32)
    Returns: out of the same shape and dtype.
    """

    B, N, N2, D = left.shape

    right = right.contiguous()
    out = torch.empty_like(left)

    stride_r_b, stride_r_j, stride_r_k, stride_r_d = right.stride()
    stride_o_b, stride_o_i, stride_o_j, stride_o_d = out.stride()

    BLOCK_I = 32
    BLOCK_J = 32
    BLOCK_K = 32
    BLOCK_D = 8   # hidden_dim is a multiple of 8 in all test configs

    grid = (
        B,
        (N + BLOCK_I - 1) // BLOCK_I,
        (N + BLOCK_J - 1) // BLOCK_J,
    )

    trimul_kernel[grid](
        left,
        right,
        out,
        B,
        N,
        D,
        stride_l_b,
        stride_l_i,
        stride_l_k,
        stride_l_d,

        stride_r_j,
        stride_r_k,
        stride_r_d,
        stride_o_b,
        stride_o_i,
        stride_o_j,
        stride_o_d,
        BLOCK_I=BLOCK_I,
        BLOCK_J=BLOCK_J,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
    )
    return out


# ----------------------------------------------------------------------
# Entry‑point used by the evaluation harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the outgoing TriMul operator.
    Arguments:
        data = (input_tensor, mask, weights, config)
            input_tensor : [B, N, N, dim]  (float32)
            mask         : [B, N, N]       (float32 or bool) – may be None
            weights      : dict of model weights
            config       : dict with keys "dim", "hidden_dim", "nomask" (optional)
    Returns:
        Tensor of shape [B, N, N, dim]
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # --------------------------------------------------------------
    # 1. Input LayerNorm over the last dimension (dim)
    # --------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        (dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )  # [B, N, N, dim]

    # --------------------------------------------------------------
    # 2. Linear projections (no bias)
    # --------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # [B, N, N, hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    # --------------------------------------------------------------
    # 3. Optional mask (broadcast over hidden_dim)
    # --------------------------------------------------------------
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1)  # [B, N, N, 1]
        left = left * mask_exp
        right = right * mask_exp

    # --------------------------------------------------------------
    # 4. Gating scalars (sigmoid)
    # --------------------------------------------------------------
    left_gate = torch.sigmoid(F.linear(x, weights["left_gate.weight"]))
    right_gate = torch.sigmoid(F.linear(x, weights["right_gate.weight"]))
    out_gate = torch.sigmoid(F.linear(x, weights["out_gate.weight"]))

    left = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # 5. Core TriMul contraction – Triton kernel
    # --------------------------------------------------------------
    out = trimul_triton(left, right)   # [B, N, N, hidden_dim]

    # --------------------------------------------------------------
    # 6. Output LayerNorm over hidden_dim and gating
    # --------------------------------------------------------------
    out = F.layer_norm(
        out,
        (hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    # --------------------------------------------------------------
    # 7. Final linear projection back to original dim
    # --------------------------------------------------------------
    result = F.linear(out, weights["to_out.weight"])

    return result
