@triton.jit
def trimul_outgoing_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, H,
    # strides
    left_batch_stride, left_i_stride, left_k_stride, left_h_stride,
    right_batch_stride, right_j_stride, right_k_stride, right_h_stride,
    out_batch_stride, out_i_stride, out_j_stride, out_h_stride,
    # block sizes as constexpr
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # Program IDs
    pid_b = tl.program_id(0)  # batch index (0..B-1)
    pid_i = tl.program_id(1)  # i block index
    pid_j = tl.program_id(2)  # j block index
    pid_d = tl.program_id(3)  # d block index (hidden)

    # Compute base offsets
    i_offset = pid_i * BLOCK_I
    j_offset = pid_j * BLOCK_J
    d_offset = pid_d * BLOCK_D

    # Compute out pointer base for this block
    out_ptrs = out_ptr + pid_b * out_batch_stride \
               + i_offset * out_i_stride \
               + j_offset * out_j_stride \
               + d_offset * out_h_stride

    # Initialize accumulator tile (BLOCK_I, BLOCK_J)
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Determine number of K blocks
    num_k_blocks = tl.cdiv(N, BLOCK_K)

    # Loops over k blocks
    for k_iter in range(num_k_blocks):
        k_offset = k_iter * BLOCK_K
        # Load left tile: shape (BLOCK_I, BLOCK_K, BLOCK_D)
        # For each i in i_range, each k in k_range, we need hidden slice d_offset..d_offset+BLOCK_D
        # Use tl.load with a mask.
        # We will load left as (BLOCK_I, BLOCK_K, BLOCK_D)
        # But for matmul we need left slice with shape (BLOCK_I, BLOCK_K) (since we will dot with right for each d). But we need to dot per d independently.
        # Approach: For each hidden within BLOCK_D, we will compute a matmul accumulation for that hidden dimension. We can loop over hidden offset within tile.
        # However more efficient: load left and right slices of shape (BLOCK_I, BLOCK_K) per hidden, and compute tl.dot for each hidden inside a loop.

        # We'll iterate over d_idx within tile: 0..BLOCK_D-1
        for d_idx in range(BLOCK_D):
            # compute pointer offsets for left and right for this hidden index
            left_ptrs = left_ptr \
                + pid_b * left_batch_stride \
                + (i_offset + tl.arange(0, BLOCK_I)) * left_i_stride \
                + (k_offset + tl.arange(0, BLOCK_K)) * left_k_stride \
                + (d_offset + d_idx) * left_h_stride
            # left block shape (BLOCK_I, BLOCK_K)
            left = tl.load(left_ptrs, mask=(i_offset + tl.arange(0, BLOCK_I) < N) & (k_offset + tl.arange(0, BLOCK_K) < N), other=0.0)

            right_ptrs = right_ptr \
                + pid_b * right_batch_stride \
                + (j_offset + tl.arange(0, BLOCK_J)) * right_j_stride \
                + (k_offset + tl.arange(0, BLOCK_K)) * right_k_stride \
                + (d_offset + d_idx) * right_h_stride
            right = tl.load(right_ptrs, mask=(j_offset + tl.arange(0, BLOCK_J) < N) & (k_offset + tl.arange(0, BLOCK_K) < N), other=0.0)

```python
import torch
import torch.nn.functional as F
import triton
import triton.language as tl

# Triton kernel
@triton.jit
def trimul_einsum_kernel(
    left_ptr, right_ptr, out_ptr,
    N, H,
    # strides for left: i, k, h
    stride_i, stride_k, stride_h,
    # strides for right: j, k, h
    stride_j, stride_k_r, stride_h_r,
    # strides for out: i_out, j_out, h_out
    stride_i_out, stride_j_out, stride_h_out,
    # compile-time block sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    """Compute out[i,j,d] = sum_k left[i,k,d] * right[j,k,d] for a single batch.
    left  shape: [N, N, H] (i, k, d)
    right shape: [N, N, H] (j, k, d)
    out   shape: [N, N, H] (i, j, d)
    """

    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)
    pid_d = tl.program_id(2)

    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_base = pid_d * BLOCK_D

    i_idx = i_base + tl.arange(0, BLOCK_I)
    j_idx = j_base + tl.arange(0, BLOCK_J)
    d_idx = d_base + tl.arange(0, BLOCK_D)

    i_mask = i_idx < N
    j_mask = j_idx < N
    d_mask = d_idx < H

    # accumulator for this tile (i, j, d)
    acc = tl.zeros((BLOCK_I, BLOCK_J, BLOCK_D), dtype=tl.float32)

    # number of k blocks
    num_k = tl.cdiv(N, BLOCK_K)

    for k_iter in range(num_k):
        k_base = k_iter * BLOCK_K
        k_idx = k_base + tl.arange(0, BLOCK_K)
        k_mask = k_idx < N

        # ---- Load a tile of left (i, k, d) ----
        left_ptrs = left_ptr \
            + (i_idx[:, None, None] * stride_i) \
            + (k_idx[None, :, None] * stride_k) \
            + (d_idx[None, None, :] * stride_h)
        left_tile = tl.load(left_ptrs,
                            mask = i_mask[:, None, None] & k_mask[None, :, None] & d_mask[None, None, :],
                            other = 0.0)

        # ---- Load a tile of right transposed (k, j, d) ----
        # We want layout (k, j, d) so that we can dot left (i,k) @ right (k,j)
        right_ptrs = right_ptr \
            + (j_idx[None, :, None] * stride_j) \
            + (k_idx[:, None, None] * stride_k_r) \
            + (d_idx[None, None, :] * stride_h_r)
        right_tile = tl.load(right_ptrs,
                             mask = j_mask[None, :, None] & k_mask[:, None, None] & d_mask[None, None, :],
                             other = 0.0)

        # ---- Compute partial products for each hidden slice ----
        # Both left_tile and right_tile have shape (BLOCK_I, BLOCK_K, BLOCK_D) and (BLOCK_K, BLOCK_J, BLOCK_D)
        for dh in range(BLOCK_D):
            # shape (BLOCK_I, BLOCK_K) and (BLOCK_K, BLOCK_J)
            left_mat = left_tile[:, :, dh]
            right_mat = right_tile[:, :, dh]
            # tl.dot returns a (BLOCK_I, BLOCK_J) matrix
            acc[:, :, dh] += tl.dot(left_mat, right_mat)

    # ---- Write back results ----
    out_ptrs = out_ptr \
        + (i_idx[:, None, None] * stride_i_out) \
        + (j_idx[None, :, None] * stride_j_out) \
        + (d_idx[None, None, :] * stride_h_out)

    tl.store(out_ptrs,
             acc,
             mask = i_mask[:, None, None] & j_mask[None, :, None] & d_mask[None, None, :])

def trimul_einsum(left, right):
    """
    Triton implementation of the contracted einsum
    out[i,j,d] = sum_k left[i,k,d] * right[j,k,d]
    left/right: [B, N, N, H] float32 (or float16)
    Returns out with the same dtype as inputs.
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Choose tile sizes (tune for H100, we use fixed values)
    BLOCK_I = 64
    BLOCK_J = 64
    BLOCK_K = 32
    BLOCK_D = 32

    # Ensure dimensions are divisible by block sizes by rounding up in grid
    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_d = (H + BLOCK_D - 1) // BLOCK_D

    for b in range(B):
        left_b = left[b]
        right_b = right[b]
        out_b = out[b]

        # Strides (in elements, not bytes)
        stride_i, stride_k, stride_h = left_b.stride()
        stride_j, stride_k_r, stride_h_r = right_b.stride()
        stride_i_out, stride_j_out, stride_h_out = out_b.stride()

        trimul_einsum_kernel[(grid_i, grid_j, grid_d)](
            left_ptr=left_b,
            right_ptr=right_b,
            out_ptr=out_b,
            N=N,
            H=H,
            stride_i=stride_i,
            stride_k=stride_k,
            stride_h=stride_h,
            stride_j=stride_j,
            stride_k_r=stride_k_r,
            stride_h_r=stride_h_r,
            stride_i_out=stride_i_out,
            stride_j_out=stride_j_out,
            stride_h_out=stride_h_out,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
            BLOCK_D=BLOCK_D,
        )
    return out

def custom_kernel(data):
    """
    TriMul 'outgoing' forward pass implemented with Triton for the
    heavy contraction. The rest of the operations (LayerNorm, Linear,
    gating, and final projection) are performed with PyTorch.
    """
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype = input_tensor.dtype

    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask = config.get("nomask", False)

    # ----- LayerNorm on the last dimension -----
    norm_weight = weights["norm.weight"]
    norm_bias = weights["norm.bias"]
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )

    # ----- Linear projections (no bias) -----
    left_proj_weight = weights["left_proj.weight"]
    right_proj_weight = weights["right_proj.weight"]
    left = torch.nn.functional.linear(x, left_proj_weight)   # shape [B,N,N,hidden]
    right = torch.nn.functional.linear(x, right_proj_weight)

    # ----- Apply mask if present -----
    if not nomask and mask is not None:
        mask_exp = mask.unsqueeze(-1).to(dtype)   # shape [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ----- Gating values (sigmoid) -----
    left_gate_weight = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight = weights["out_gate.weight"]

    left_gate = torch.sigmoid(torch.nn.functional.linear(x, left_gate_weight))
    right_gate = torch.sigmoid(torch.nn.functional.linear(x, right_gate_weight))
    out_gate = torch.sigmoid(torch.nn.functional.linear(x, out_gate_weight))

    left = left * left_gate
    right = right * right_gate

    # ----- Heavy contraction via Triton -----
    # Ensure tensor contiguity for the kernel
    left = left.contiguous()
    right = right.contiguous()
    out = trimul_einsum(left, right)   # shape [B,N,N,hidden]

    # ----- Output LayerNorm + gating + linear projection -----
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias = weights["to_out_norm.bias"]
    out = torch.nn.functional.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate   # broadcast over hidden_dim

    to_out_weight = weights["to_out.weight"]
    out = torch.nn.functional.linear(out, to_out_weight)  # final shape [B,N,N,dim]

    return out
```

"""
TriMul (outgoing) forward implementation.
-------------------------------------------------
The heavy part of the operator is the contraction

    out[b, i, j, d] = Σ_k left[b, i, k, d] * right[b, j, k, d]

where `left` and `right` are the gated linear projections of the
input tensor `[B, N, N, C]`.  The contraction is written as an
einsum `...i k d, ...j k d -> ...i j d`.  It is an O(N³·D) operation
and dominates the runtime.  A custom Triton kernel is used for this
contraction, while the surrounding LayerNorm, Linear and gating
operations are performed with PyTorch.

Kernel design
-------------
* The kernel processes a **single hidden channel `d`** at a time.
  The hidden dimension is mapped to the third program‑grid axis,
  therefore one Triton program (warp) handles a tile of the output
  `(i_tile, j_tile)` for a fixed hidden index.
* Each program works on a tile of size `(BLOCK_I, BLOCK_J)` (default
  `16×16`).  Inside the program the tile is stored in a 2‑D accumulator
  of type `float32`.
* The reduction over `k` is tiled (`BLOCK_K = 32`).  For each `k`‑tile
  the kernel loads two sub‑matrices:
      `left[i, k, d]`  shape `(BLOCK_I, BLOCK_K)`
      `right[j, k, d]` (transposed) shape `(BLOCK_K, BLOCK_J)`
  and accumulates `tl.dot(left_tile, right_tile)` into the accumulator.
* At the end the accumulator is written back to the output tensor.
* The kernel is launched once per batch; the batch loop is performed in
  Python because the batch size is at most 2.

The rest of the forward pass (LayerNorms, Linear projections,
gates, final projection) is straightforward PyTorch code.

"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel for the 3‑tensor contraction (one hidden channel per program)
# ----------------------------------------------------------------------
@triton.jit
def trimul_kernel_one_hidden(
    left_ptr,          # const float*  (N, N, D)  – left projection
    right_ptr,         # const float*  (N, N, D)  – right projection
    out_ptr,           # float*        (N, N, D)  – output buffer
    N,                 # int32    sequence length
    # strides for left (i,k,d)
    stride_i, stride_k, stride_h,
    # strides for right (j,k,d)
    stride_j, stride_k_r, stride_h_r,
    # strides for output (i,j,d)
    stride_i_out, stride_j_out, stride_h_out,
    # compile‑time tile sizes
    BLOCK_I: tl.constexpr, BLOCK_J: tl.constexpr, BLOCK_K: tl.constexpr,
):
    """Compute out[i,j,d] = Σ_k left[i,k,d] * right[j,k,d] for a single d."""
    pid_i = tl.program_id(0)          # tile row
    pid_j = tl.program_id(1)          # tile column
    pid_d = tl.program_id(2)          # hidden index (exact channel, not a block)

    # ------------------------------------------------------------------
    # Offsets for the current tile
    # ------------------------------------------------------------------
    i_base = pid_i * BLOCK_I
    j_base = pid_j * BLOCK_J
    d_idx  = pid_d                     # exact hidden channel (BLOCK_D = 1)

    i = i_base + tl.arange(0, BLOCK_I)      # (BLOCK_I,)
    j = j_base + tl.arange(0, BLOCK_J)      # (BLOCK_J,)

    i_mask = i < N
    j_mask = j < N

    # hidden offset (stride_h is usually 1 for a contiguous tensor)
    hidden_offset = d_idx * stride_h

    # ------------------------------------------------------------------
    # Accumulator for this tile (float32)
    # ------------------------------------------------------------------
    acc = tl.zeros((BLOCK_I, BLOCK_J), dtype=tl.float32)

    # Number of k‑tiles
    n_k = tl.cdiv(N, BLOCK_K)

    for kk in range(n_k):
        k_base = kk * BLOCK_K
        k = k_base + tl.arange(0, BLOCK_K)      # (BLOCK_K,)
        k_mask = k < N

        # --------------------------------------------------------------
        # Load left[i, k, d]
        # --------------------------------------------------------------
        left_ptrs = left_ptr + \
            (i[:, None] * stride_i) + \
            (k[None, :] * stride_k) + \
            hidden_offset
        left = tl.load(
            left_ptrs,
            mask=(i_mask[:, None] & k_mask[None, :]),
            other=0.0,
        )   # (BLOCK_I, BLOCK_K)

        # --------------------------------------------------------------
        # Load right[j, k, d] **transposed** as (k, j)
        # --------------------------------------------------------------
        right_ptrs = right_ptr + \
            (j[None, :] * stride_j) + \
            (k[:, None] * stride_k_r) + \
            hidden_offset
        right = tl.load(
            right_ptrs,
            mask=(j_mask[None, :] & k_mask[:, None]),
            other=0.0,
        )   # (BLOCK_K, BLOCK_J)

        # --------------------------------------------------------------
        # Multiply‑add over the K‑tile
        # --------------------------------------------------------------
        acc += tl.dot(left, right)   # (BLOCK_I, BLOCK_J)

    # ------------------------------------------------------------------
    # Write back the tile
    # ------------------------------------------------------------------
    out_ptrs = out_ptr + \
        (i[:, None] * stride_i_out) + \
        (j[None, :] * stride_j_out) + \
        hidden_offset

    tl.store(
        out_ptrs,
        acc,
        mask=i_mask[:, None] & j_mask[None, :],
    )


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel for the whole batch
# ----------------------------------------------------------------------
def trimul_contraction(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Contract left and right tensors using Triton.
    Shapes: left/right = [B, N, N, H]  (float32 or float16)
    Returns: out = [B, N, N, H] (same dtype)
    """
    assert left.shape == right.shape
    B, N, _, H = left.shape
    out = torch.empty_like(left)

    # Tile sizes – they fit comfortably into registers on an H100
    BLOCK_I = 16
    BLOCK_J = 16
    BLOCK_K = 32   # 32‑wide reduction works well with tensor cores

    # Number of tiles per dimension
    grid_i = (N + BLOCK_I - 1) // BLOCK_I
    grid_j = (N + BLOCK_J - 1) // BLOCK_J
    grid_d = H                       # one program per hidden channel

    # --------------------------------------------------------------
    # Launch one kernel per batch (B ≤ 2)
    # --------------------------------------------------------------
    for b in range(B):
        left_b = left[b]   # [N, N, H]
        right_b = right[b]
        out_b = out[b]

        # Strides in *elements* (not bytes)
        s_i, s_k, s_h = left_b.stride()          # (i, k, d)
        s_j, s_k_r, s_h_r = right_b.stride()     # (j, k, d)
        s_i_o, s_j_o, s_h_o = out_b.stride()    # (i, j, d)

        trimul_kernel_one_hidden[(grid_i, grid_j, grid_d)](
            left_ptr=left_b,
            right_ptr=right_b,
            out_ptr=out_b,
            N=N,
            stride_i=s_i,
            stride_k=s_k,
            stride_h=s_h,
            stride_j=s_j,
            stride_k_r=s_k_r,
            stride_h_r=s_h_r,
            stride_i_out=s_i_o,
            stride_j_out=s_j_o,
            stride_h_out=s_h_o,
            BLOCK_I=BLOCK_I,
            BLOCK_J=BLOCK_J,
            BLOCK_K=BLOCK_K,
        )
    return out


# ----------------------------------------------------------------------
# Main entry point required by the problem statement
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Forward pass of the “outgoing” TriMul module.
    Arguments
    ---------
    data : tuple (input, mask, weights, config)
        - input  : torch.Tensor [B, N, N, dim] (float32)
        - mask   : torch.Tensor [B, N, N]   (bool/int) or None
        - weights: dict of torch.Tensors containing all model parameters
        - config : dict with keys
            * dim        – input channel size
            * hidden_dim – hidden dimension used inside TriMul
            * nomask     – bool, true if the mask should be ignored

    Returns
    -------
    torch.Tensor of shape [B, N, N, dim] (float32)
    """
    # --------------------------------------------------------------
    # Unpack inputs
    # --------------------------------------------------------------
    input_tensor, mask, weights, config = data
    device = input_tensor.device
    dtype  = input_tensor.dtype

    dim        = config["dim"]
    hidden_dim = config["hidden_dim"]
    nomask     = config.get("nomask", False)

    # --------------------------------------------------------------
    # LayerNorm on the input (last dimension = dim)
    # --------------------------------------------------------------
    norm_weight = weights["norm.weight"]
    norm_bias   = weights["norm.bias"]
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=norm_weight,
        bias=norm_bias,
    )   # [B, N, N, dim]

    # --------------------------------------------------------------
    # Linear projections (no bias)
    # --------------------------------------------------------------
    left_proj_weight  = weights["left_proj.weight"]   # (hidden, dim)
    right_proj_weight = weights["right_proj.weight"]  # (hidden, dim)

    left  = F.linear(x, left_proj_weight)   # [B,N,N,hidden]
    right = F.linear(x, right_proj_weight)  # [B,N,N,hidden]

    # --------------------------------------------------------------
    # Optional mask (same broadcasting as the reference implementation)
    # --------------------------------------------------------------
    if (not nomask) and mask is not None:
        mask_b = mask.unsqueeze(-1).to(dtype)   # [B,N,N,1]
        left  = left * mask_b
        right = right * mask_b

    # --------------------------------------------------------------
    # Gating values (sigmoid of linear projections of the *original* x)
    # --------------------------------------------------------------
    left_gate_weight  = weights["left_gate.weight"]
    right_gate_weight = weights["right_gate.weight"]
    out_gate_weight   = weights["out_gate.weight"]

    left_gate  = torch.sigmoid(F.linear(x, left_gate_weight))   # [B,N,N,hidden]
    right_gate = torch.sigmoid(F.linear(x, right_gate_weight))  # [B,N,N,hidden]
    out_gate   = torch.sigmoid(F.linear(x, out_gate_weight))    # [B,N,N,hidden]

    left  = left * left_gate
    right = right * right_gate

    # --------------------------------------------------------------
    # Heavy contraction via Triton kernel
    # --------------------------------------------------------------
    # Ensure contiguous layout for the kernel
    left  = left.contiguous()
    right = right.contiguous()
    out   = trimul_contraction(left, right)        # [B,N,N,hidden]

    # --------------------------------------------------------------
    # Output LayerNorm, gating and final linear projection
    # --------------------------------------------------------------
    to_out_norm_weight = weights["to_out_norm.weight"]
    to_out_norm_bias   = weights["to_out_norm.bias"]
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=to_out_norm_weight,
        bias=to_out_norm_bias,
    )
    out = out * out_gate                           # broadcast over hidden dim

    to_out_weight = weights["to_out.weight"]      # (dim, hidden)
    out = F.linear(out, to_out_weight)            # [B,N,N,dim]

    return out

**Explanation of the implementation**

* **Layer‑norm & linear layers** – performed with `torch.nn.functional` for simplicity and to keep the
  code numerically identical to the reference module.
* **Mask handling** – if `nomask` is `False` and a mask tensor is supplied, it is broadcast to the
  hidden dimension and multiplied into `left` and `right`.
* **Gates** – computed with a sigmoid‑activated linear layer and fused element‑wise with the
  projected tensors.
* **Contraction** – the core `einsum` is replaced by `trimul_contraction`, which invokes the
  Triton kernel `trimul_kernel_one_hidden`.  
  The kernel processes one hidden channel at a time, using a tiled matrix‑multiply
  (`tl.dot`) over the `k` dimension.  Tile sizes `(BLOCK_I, BLOCK_J, BLOCK_K) = (16,16,32)`
  were chosen to fit comfortably into registers on an H100 while providing good
  occupancy.
* **Post‑processing** – the intermediate result is layer‑normalised, gated, and finally
  projected back to the original dimensionality with a linear layer.

The function `custom_kernel` complies with the required signature and can be
directly used as the forward pass of the TriMul module in AlphaFold‑3‑style
models.
