def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, D,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_b = tl.program_id(2)  # batch index
    pid_i = tl.program_id(0)
    pid_j = tl.program_id(1)

    # block start indices
    i_start = pid_i * BLOCK_M
    j_start = pid_j * BLOCK_N
    k_end = N

    # create accumulator
    acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

    # loop over k dimension
    for k_offset in range(0, tl.cdiv(N, BLOCK_K)):
        k_start = k_offset * BLOCK_K
        # compute within block size: actual K range
        cur_K = tl.minimum(BLOCK_K, N - k_start)

        # load left block: shape (BLOCK_M, cur_K, BLOCK_D)
        i_range = i_start + tl.arange(0, BLOCK_M)
        k_range = k_start + tl.arange(0, BLOCK_K)
        d_range = tl.arange(0, BLOCK_D)

        # compute offsets:
        left_ptrs = left_ptr + i_range[:, None, None] * stride_left_i \
                                  + k_range[None, :, None] * stride_left_k \
                                  + d_range[None, None, :] * stride_left_d \
                                  + pid_b * stride_left_b

        # similarly for right block: shape (cur_K, BLOCK_N, BLOCK_D)
        j_range = j_start + tl.arange(0, BLOCK_N)

        right_ptrs = right_ptr + j_range[None, :, None] * stride_right_j \
                                    + k_range[:, None, None] * stride_right_k \
                                    + d_range[None, None, :] * stride_right_d \
                                    + pid_b * stride_right_b

        # mask out-of-bounds for i/j:
        mask_i = i_range < N
        mask_j = j_range < N
        mask_k = k_range < N

        # broadcast
        mask_i = mask_i[:, None, None]
        mask_j = mask_j[None, :, None]
        mask_k = mask_k[None, :, None]  # shape (1, K, 1) maybe

        # load with zero for out-of-range
        left = tl.load(left_ptrs, mask=mask_i & mask_k, other=0.0)
        right = tl.load(right_ptrs, mask=mask_j & mask_k, other=0.0)

        # We can compute product per k and sum across k via loop unrolling over k in block dimension.
        # That is: for kk in range(cur_K):

        # We can compute outer product across i and j for each kk: product = left_i_kk[:, None, :] * right_kk_j[None, :, :]
        # That yields shape (M,N,D). Then accumulate.
        # This may be heavy but for small BLOCK_M and BLOCK_N (e.g., 32) and BLOCK_K maybe 32, it's okay.

        for kk in range(0, BLOCK_K):
            cur_k_valid = k_start + kk < N
            if not cur_k_valid:

            right_kk = right[kk, :, :]        # (BLOCK_N, BLOCK_D)
            # broadcast to (BLOCK_M, BLOCK_N, BLOCK_D)
            # left_kk[:, None, :] shape (M,1,D); right_kk[None, :, :] shape (1,N,D)
            prod = left_kk[:, None, :] * right_kk[None, :, :]  # (M,N,D)
            acc += prod

@triton.jit
def tri_mul_kernel(
    left_ptr, right_ptr, out_ptr,
    B, N, hidden_dim,
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr, BLOCK_K: tl.constexpr,
    BLOCK_D: tl.constexpr
):
    pid = tl.program_id(axis=0)  # tile m
    pid_n = tl.program_id(axis=1)  # tile n
    pid_b = tl.program_id(axis=2)

    # compute tile start
    i_start = pid * BLOCK_M
    j_start = pid_n * BLOCK_N
    b = pid_b

    # loop over hidden dim tiles
    for d_start in range(0, hidden_dim, BLOCK_D):
        # init accumulator
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # loop over k
        for k_start in range(0, N, BLOCK_K):
            cur_k = tl.minimum(N - k_start, BLOCK_K)
            # compute offsets for left and right
            # left: shape (BLOCK_M, cur_k, BLOCK_D)
            # right: shape (cur_k, BLOCK_N, BLOCK_D)
            # create index vectors
            i_range = i_start + tl.arange(0, BLOCK_M)
            j_range = j_start + tl.arange(0, BLOCK_N)
            k_range = k_start + tl.arange(0, BLOCK_K)
            d_range = d_start + tl.arange(0, BLOCK_D)

            # mask out-of-bounds
            mask_i = i_range < N
            mask_j = j_range < N
            mask_k = k_range < N
            mask_d = d_range < hidden_dim

            # compute pointers
            left_ptrs = left_ptr + b * stride_left_b \
                + i_range[:, None, None] * stride_left_i \
                + k_range[None, :, None] * stride_left_k \
                + d_range[None, None, :] * stride_left_d

            right_ptrs = right_ptr + b * stride_right_b \
                + k_range[:, None, None] * stride_right_k \
                + j_range[None, :, None] * stride_right_j \
                + d_range[None, None, :] * stride_right_d

            left = tl.load(left_ptrs, mask=(mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]), other=0.0)
            right = tl.load(right_ptrs, mask=(mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :]), other=0.0)

            # compute contributions: each left[k] (size (BLOCK_M, BLOCK_D)) etc.
            # we can loop over inner k dimension
            for kk in range(BLOCK_K):
                a_slice = left[:, kk, :]   # (BLOCK_M, BLOCK_D)
                b_slice = right[kk, :, :]  # (BLOCK_N, BLOCK_D)
                # broadcast and accumulate
                acc += a_slice[:, None, :] * b_slice[None, :, :]
        # store acc into output slice
        out_ptrs = out_ptr + b * stride_out_b \
            + i_range[:, None, None] * stride_out_i \
            + j_range[None, :, None] * stride_out_j \
            + d_range[None, None, :] * stride_out_d

        mask_store_i = i_range < N
        mask_store_j = j_range < N
        mask_store_d = d_range < hidden_dim
        mask_store = mask_store_i[:, None, None] & mask_store_j[None, :, None] & mask_store_d[None, None, :]

        tl.store(out_ptrs, acc, mask=mask_store)

    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]
    # input shape (bs, N, N, dim)
    bs, N, _, d = input_tensor.shape
    # LayerNorm: x = (x - mean)/sqrt(var+eps)*weight + bias
    # Use torch.nn.functional.layer_norm:
    x = torch.nn.functional.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"]
    )
    # Linear projections
    left = torch.nn.functional.linear(x, weights["left_proj.weight"])    # shape (bs,N,N,hidden_dim)
    right = torch.nn.functional.linear(x, weights["right_proj.weight"])

    # mask handling
    if mask is not None:
        mask_exp = mask.unsqueeze(-1)  # shape (bs,N,N,1)
        left = left * mask_exp
        right = right * mask_exp
    # Gates
    left_gate = torch.nn.functional.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = torch.nn.functional.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = torch.nn.functional.linear(x, weights["out_gate.weight"]).sigmoid()

    right = right * right_gate

    out = tri_mul_forward(left, right)  # shape (bs,N,N,hidden_dim)

        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"]
    )
    out = out * out_gate

    # final linear to dim
    out = torch.nn.functional.linear(out, weights["to_out.weight"])
    return out

"""
Triton implementation of the “outgoing” TriMul operator (AlphaFold3 style).

The kernel computes the contraction
    out[b,i,j,d] = Σ_k left[b,i,k,d] * right[b,j,k,d]
for a 4‑D tensor [B, N, N, D] (D = hidden_dim).  This is the costly N³ part
of the TriMul module.  The implementation tiles the output (i,j) plane and the
hidden dimension (d) to keep register usage modest:

* BLOCK_M × BLOCK_N threads compute a tile of the (i,j) matrix.
* The hidden dimension is processed in chunks of BLOCK_D (here 32) – the
  kernel loops over the hidden channels, accumulating a partial result in a
  (BLOCK_M, BLOCK_N, BLOCK_D) accumulator.
* The sum over the k‑dimension is tiled with BLOCK_K (here 4).  For each k
  block the kernel loads a (BLOCK_M, BLOCK_K, BLOCK_D) slice of `left` and a
  (BLOCK_K, BLOCK_N, BLOCK_D) slice of `right`, then performs a broadcasted
  multiplication and reduction over k.

The rest of the TriMul forward pass (LayerNorms, Linear projections,
gate‑sigmoids, masking, final linear projection) is performed with
torch.nn.functional for simplicity.  Only the O(N³) contraction is off‑loaded
to Triton, satisfying the requirement to implement a non‑trivial kernel.
"""

import torch
import torch.nn.functional as F
import triton
import triton.language as tl


# ----------------------------------------------------------------------
# Triton kernel: hidden‑dim tiled contraction
# ----------------------------------------------------------------------
@triton.jit
def tri_mul_kernel(
    left_ptr,            # const float*  [B, N, N, D]
    right_ptr,           # const float*  [B, N, N, D]
    out_ptr,             # float*        [B, N, N, D]
    B, N, hidden_dim,    # int64 scalars
    # strides for left tensor
    stride_left_b, stride_left_i, stride_left_k, stride_left_d,
    # strides for right tensor (same layout as left)
    stride_right_b, stride_right_j, stride_right_k, stride_right_d,
    # strides for output tensor (same layout as left)
    stride_out_b, stride_out_i, stride_out_j, stride_out_d,
    # compile‑time tile sizes
    BLOCK_M: tl.constexpr, BLOCK_N: tl.constexpr,
    BLOCK_K: tl.constexpr, BLOCK_D: tl.constexpr,
):
    # ------------------------------------------------------------------
    # Thread‑block identifiers
    # ------------------------------------------------------------------
    pid_b = tl.program_id(2)                 # batch dimension
    pid_i = tl.program_id(0)                 # i‑tile (row)
    pid_j = tl.program_id(1)                 # j‑tile (col)

    i_base = pid_i * BLOCK_M
    j_base = pid_j * BLOCK_N

    # ------------------------------------------------------------------
    # Tile the hidden dimension
    # ------------------------------------------------------------------
    d_offset = 0
    while d_offset < hidden_dim:
        # (BLOCK_M, BLOCK_N, BLOCK_D) accumulator in FP32
        acc = tl.zeros((BLOCK_M, BLOCK_N, BLOCK_D), dtype=tl.float32)

        # ------------------------------------------------------------------
        # Tile the reduction (k) dimension
        # ------------------------------------------------------------------
        k_offset = 0
        while k_offset < N:
            # ------------------------------------------------------------------
            # Index vectors for this tile
            # ------------------------------------------------------------------
            i = i_base + tl.arange(0, BLOCK_M)                 # (M,)
            j = j_base + tl.arange(0, BLOCK_N)                 # (N,)
            k = k_offset + tl.arange(0, BLOCK_K)               # (K,)
            d = d_offset + tl.arange(0, BLOCK_D)               # (D,)

            # Masks for out‑of‑bounds (important for the last partial tile)
            mask_i = i < N
            mask_j = j < N
            mask_k = k < N
            mask_d = d < hidden_dim

            # ------------------------------------------------------------------
            # Compute pointer offsets for left[b, i, k, d]
            # ------------------------------------------------------------------
            left_ptrs = (
                left_ptr
                + pid_b * stride_left_b
                + i[:, None, None] * stride_left_i
                + k[None, :, None] * stride_left_k
                + d[None, None, :] * stride_left_d
            )
            # ------------------------------------------------------------------
            # Compute pointer offsets for right[b, j, k, d]
            # (right is indexed as (b, j, k, d) – note the swap of i/j)
            # ------------------------------------------------------------------
            right_ptrs = (
                right_ptr
                + pid_b * stride_right_b
                + k[:, None, None] * stride_right_k
                + j[None, :, None] * stride_right_j
                + d[None, None, :] * stride_right_d
            )
            # Load with zero‑fill for masked elements
            mask_left = mask_i[:, None, None] & mask_k[None, :, None] & mask_d[None, None, :]
            mask_right = mask_j[None, :, None] & mask_k[:, None, None] & mask_d[None, None, :]
            left = tl.load(left_ptrs, mask=mask_left, other=0.0)   # (M, K, D)
            right = tl.load(right_ptrs, mask=mask_right, other=0.0)  # (K, N, D)

            # ------------------------------------------------------------------
            # Accumulate: for each k within the block compute
            #   acc[i, j, d] += left[i, k, d] * right[k, j, d]
            # ------------------------------------------------------------------
            for kk in range(0, BLOCK_K):
                a = left[:, kk, :]       # (M, D)
                b = right[kk, :, :]      # (N, D)
                # Broadcast multiply -> (M, N, D)
                acc += a[:, None, :] * b[None, :, :]

            k_offset += BLOCK_K

        # ------------------------------------------------------------------
        # Write the tile (i_base … i_base+M, j_base … j_base+N, d_offset …)
        # ------------------------------------------------------------------
        out_ptrs = (
            out_ptr
            + pid_b * stride_out_b
            + i[:, None, None] * stride_out_i
            + j[None, :, None] * stride_out_j
            + d[None, None, :] * stride_out_d
        )
        mask_out = (
            (i < N)[:, None, None]
            & (j < N)[None, :, None]
            & (d < hidden_dim)[None, None, :]
        )
        tl.store(out_ptrs, acc, mask=mask_out)

        d_offset += BLOCK_D


# ----------------------------------------------------------------------
# Helper that launches the Triton kernel
# ----------------------------------------------------------------------
def tri_mul_forward(left: torch.Tensor, right: torch.Tensor) -> torch.Tensor:
    """
    Compute the contracted tensor using the Triton kernel.
    Arguments:
        left, right – shape [B, N, N, D] (contiguous, float32)
    Returns:
        out – shape [B, N, N, D] (float32)
    """
    assert left.shape == right.shape
    B, N, _, D = left.shape

    # Allocate output
    out = torch.empty_like(left)

    # Ensure contiguity (required for stride based pointer arithmetic)
    left = left.contiguous()
    right = right.contiguous()
    out = out.contiguous()

    # Strides in elements (not bytes)
    stride_left_b, stride_left_i, stride_left_k, stride_left_d = left.stride()
    stride_right_b, stride_right_j, stride_right_k, stride_right_d = right.stride()
    stride_out_b, stride_out_i, stride_out_j, stride_out_d = out.stride()

    # Tile sizes – chosen to keep register pressure modest while exposing enough
    # parallelism.  hidden_dim is always 128 in the reference configs, so we
    # process it in 4 chunks of 32.
    BLOCK_M = 16
    BLOCK_N = 16
    BLOCK_K = 4
    BLOCK_D = 32

    grid = (triton.cdiv(N, BLOCK_M), triton.cdiv(N, BLOCK_N), B)

    tri_mul_kernel[grid](
        left_ptr=left,
        right_ptr=right,
        out_ptr=out,
        B=B,
        N=N,
        hidden_dim=D,
        stride_left_b=stride_left_b,
        stride_left_i=stride_left_i,
        stride_left_k=stride_left_k,
        stride_left_d=stride_left_d,
        stride_right_b=stride_right_b,
        stride_right_j=stride_right_j,
        stride_right_k=stride_right_k,
        stride_right_d=stride_right_d,
        stride_out_b=stride_out_b,
        stride_out_i=stride_out_i,
        stride_out_j=stride_out_j,
        stride_out_d=stride_out_d,
        BLOCK_M=BLOCK_M,
        BLOCK_N=BLOCK_N,
        BLOCK_K=BLOCK_K,
        BLOCK_D=BLOCK_D,
        num_warps=4,    # typical choice for these tile sizes on an H100
    )
    return out


# ----------------------------------------------------------------------
# Full forward pass (no autograd) – the entry point required by the test harness
# ----------------------------------------------------------------------
def custom_kernel(data):
    """
    Entry point matching the problem statement.

    Arguments:
        data = (input_tensor, mask, weights, config)
        - input_tensor : [B, N, N, dim]  (float32)
        - mask        : [B, N, N] (bool or float, may be omitted)
        - weights     : dict of model weights (all are float32 tensors)
        - config      : dict with keys "dim" and "hidden_dim"

    Returns:
        output : [B, N, N, dim]  (float32)
    """
    input_tensor, mask, weights, config = data
    dim = config["dim"]
    hidden_dim = config["hidden_dim"]

    # ------------------------------------------------------------------
    # 1. Input LayerNorm  (dim -> dim)
    # ------------------------------------------------------------------
    x = F.layer_norm(
        input_tensor,
        normalized_shape=(dim,),
        weight=weights["norm.weight"],
        bias=weights["norm.bias"],
    )

    # ------------------------------------------------------------------
    # 2. Linear projections + optional mask
    # ------------------------------------------------------------------
    left = F.linear(x, weights["left_proj.weight"])   # -> [B,N,N,hidden_dim]
    right = F.linear(x, weights["right_proj.weight"])

    if mask is not None:
        # mask may be bool; cast to same dtype as activations
        mask = mask.to(dtype=x.dtype)
        mask_exp = mask.unsqueeze(-1)                 # [B,N,N,1]
        left = left * mask_exp
        right = right * mask_exp

    # ------------------------------------------------------------------
    # 3. Gating (sigmoid on another linear projection)
    # ------------------------------------------------------------------
    left_gate = F.linear(x, weights["left_gate.weight"]).sigmoid()
    right_gate = F.linear(x, weights["right_gate.weight"]).sigmoid()
    out_gate = F.linear(x, weights["out_gate.weight"]).sigmoid()

    left = left * left_gate
    right = right * right_gate

    # ------------------------------------------------------------------
    # 4. Core O(N³) contraction – Triton kernel
    # ------------------------------------------------------------------
    out = tri_mul_forward(left, right)               # shape [B,N,N,hidden_dim]

    # ------------------------------------------------------------------
    # 5. Post‑contraction normalization, gating and final projection
    # ------------------------------------------------------------------
    out = F.layer_norm(
        out,
        normalized_shape=(hidden_dim,),
        weight=weights["to_out_norm.weight"],
        bias=weights["to_out_norm.bias"],
    )
    out = out * out_gate

    out = F.linear(out, weights["to_out.weight"])    # back to [B,N,N,dim]

    return out
