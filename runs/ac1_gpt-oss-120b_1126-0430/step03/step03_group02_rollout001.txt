    n = a.size

        alpha = alpha0 + alpha_inc * t / max_iters

        w = np.exp(alpha * (conv - max_val))
        p = w / w.sum()

"""
Propose a non‑negative step function that minimises

    O = 2·n·max(conv(a,a)) / (sum(a))²

The bound is scale‑invariant, so the search is performed on the simplex
{a≥0, Σa=1}.  The optimiser uses a projected exponentiated‑gradient
scheme.  At each iteration the sub‑gradient of the maximal convolution
value is approximated by a convex combination of

* the exact sub‑gradient (derived from the indices where the convolution
  attains its maximum),
* a smooth “soft‑max” gradient obtained by weighting all convolution
  entries with a softmax (controlled by a temperature parameter α).

The weight of the smooth term is increased linearly during the run,
allowing a fast coarse descent followed by a fine‑grained optimisation.
Multiple initialisations (previous best, the 1/√‑law, a random point) and
several sequence lengths are tried; the best sequence found before the
time‑budget expires is returned.
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x≥0, Σx=1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using an FFT of power‑of‑two size."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.fft(a, n=size)
    conv = np.fft.ifft(A * A).real[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Compute the bound 2·n·max(conv(a,a)) for a normalised vector (Σa=1).
    """
    n = a.size
    conv = _conv_fft(a)
    return 2.0 * n * np.max(conv)


# ----------------------------------------------------------------------
def _subgradient_exact(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """
    Exact sub‑gradient of max(conv(a,a)) w.r.t. a on the simplex.
    conv must be the full linear convolution a*a.
    """
    n = a.shape[0]
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # derivative of a_i * a_j appears twice (i↔j)
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
def _smooth_gradient(a: np.ndarray, conv: np.ndarray, alpha: float = 5.0) -> np.ndarray:
    """
    Gradient of a soft‑max approximation of max(conv).
    The weight vector is p_k ∝ exp(α·(conv_k – max(conv))).
    """
    n = a.shape[0]
    max_val = conv.max()
    # soft‑max weights (the maximum entry gets weight 1, others ≤1)
    w = np.exp(alpha * (conv - max_val))
    p = w / w.sum()                     # length 2n‑1, sums to 1

    # cross‑correlation a ↔ p  (g_i = 2 Σ_j a_j p_{i+j})
    a_rev = a[::-1]
    size = 1 << ((len(a_rev) + len(p) - 1).bit_length())
    A_rev = np.fft.fft(a_rev, size)
    P = np.fft.fft(p, size)
    cross = np.fft.ifft(A_rev * P).real          # length ≥ 3n‑2
    start = n - 1
    g = 2.0 * cross[start:start + n]            # exactly n entries
    return g


# ----------------------------------------------------------------------
def _exp_update(a: np.ndarray, g: np.ndarray, eta: float) -> np.ndarray:
    """
    Exponentiated‑gradient step (log‑space) followed by normalisation.
    """
    log_a = np.log(a + 1e-15) - eta * g
    log_a -= np.max(log_a)          # keep numbers moderate
    a_new = np.exp(log_a)
    a_new /= a_new.sum()
    return a_new


# ----------------------------------------------------------------------
def _optimize_combined(a0: np.ndarray,
                       max_iters: int,
                       eta0: float,
                       alpha: float,
                       deadline: float,
                       smooth_freq: int = 2) -> tuple[np.ndarray, float]:
    """
    Projected exponentiated‑gradient optimiser.
    The gradient is a convex combination of the exact sub‑gradient of the
    max‑convolution and a smooth “soft‑max” gradient.  The weight of the
    smooth part grows linearly from 0 to 1.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)
    n = a.size
    patience = max_iters // 20
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ---------- compute convolution ----------
        conv = _conv_fft(a)

        # ---------- exact sub‑gradient ----------
        g_exact = _subgradient_exact(a, conv)

        # ---------- smooth gradient (occasionally) ----------
        if (t % smooth_freq) == 0:
            g_smooth = _smooth_gradient(a, conv, alpha=alpha)
        else:
            g_smooth = None

        # weight for the smooth component (increase during optimisation)
        w_s = min(1.0, t / max_iters)

        if g_smooth is not None:
            g = (1.0 - w_s) * g_exact + w_s * g_smooth
        else:
            g = g_exact

        # ---------- exponentiated‑gradient update ----------
        eta = eta0 * (1.0 - t / max_iters)        # linear decay
        a = _exp_update(a, g, eta)

        # ---------- keep best ----------
        cur_score = _objective(a)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a.copy()
            no_improve = 0
            print(f"[opt] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random jitter to escape plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0

    return best_a, best_score

        if time.time() >= deadline:
            break
        print(f"[run] exploring length n={n}")

        # -------------------- build starting points --------------------
        starts = []

        # 1/√ law – analytically motivated
        idx = np.arange(1, n + 1, dtype=float)
        a_sqrt = 1.0 / np.sqrt(idx)
        a_sqrt /= a_sqrt.sum()
        starts.append(a_sqrt)

        # random interior point
        a_rand = np.random.rand(n)
        a_rand = _project_to_simplex(a_rand)
        starts.append(a_rand)

        # previous best if dimensions match
        if 'height_sequence_1' in globals():
            try:
                a_prev = np.asarray(height_sequence_1, dtype=float)
                if a_prev.size == n:
                    a_prev_norm = a_prev / a_prev.sum()
                    starts.append(a_prev_norm)
            except Exception:
                pass

        # -----------------------------------------------------------------
        # Optimisation parameters – we keep them modest to fit the time budget.
        max_iters = 12000                # per start
        eta0 = 0.75                      # initial learning rate for EG
        alpha = 6.0                      # soft‑max temperature
        smooth_every = 3                 # compute smooth gradient every 3 steps

        for i_start, a_start in enumerate(starts):
            if time.time() >= deadline:
                break
            init_score = _objective(a_start)
            print(f"[init {i_start}] n={n} start O={init_score:.6f}")

            a_opt, score_opt = _optimize_combined(a_start,
                                                  max_iters=max_iters,
                                                  eta0=eta0,
                                                  alpha=alpha,
                                                  deadline=deadline,
                                                  smooth_freq=smooth_every)

            print(f"[result {i_start}] n={n} optimized O={score_opt:.6f}")

            if score_opt < best_score:
                best_score = score_opt
                best_seq = a_opt.copy()
                best_n = n

    # ------------------------------------------------------------------
    # Fallback to the 1/√ law if nothing was found (unlikely).
    if best_seq is None:
        n = base_n
        idx = np.arange(1, n + 1, dtype=float)
        best_seq = (1.0 / np.sqrt(idx))
        best_seq /= best_seq.sum()
        best_score = _objective(best_seq)

    # ------------------------------------------------------------------
    # Final exact evaluation (scale‑invariant, so we keep sum=1)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:   # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)
        final_score = None

    print(f"[summary] best length = {best_n}, best bound ≈ {best_score:.6f}")
    return best_seq.tolist()
