def propose_candidate(seed=0,budget_s=1000):

    t0=time.time()
    deadline = t0 + budget_s - 5

    # Stage 1: small scale seeds search
    n_small = 1500
    seeds = generate_seeds(n_small)

    stage_deadline = t0 + budget_s * 0.3
    best_seq = None; best_score = inf
    for name, seq in seeds:

        per_seed = max(2.0, (stage_deadline - time.time())/(len(seeds)+1))
        refined = adam(seq, time.time()+per_seed)

    # Stage 2: upscale to medium
    n_mid = 3000
    best_seq = upscale(best_seq, n_mid)
    # Stage 2 refinement
    best_seq = adam(best_seq, deadline - small_margin) # where small_margin leftover for possible final step

    # Stage 3: optional final tweak: add small uniform dithering and re-run short refinement
    final_seq = final_refinement(best_seq, deadline)
    return final_seq.tolist()

    v = np.zeros_like(seq)

    best_seq = None
    best_score = float('inf')
    it = 0
    # compute fft length
    fft_len = 1 << ((2*n-1).bit_length())
    while time.time() < deadline:
        # compute current seq and sum normalization
        seq = np.exp(log_seq)
        sum_seq = seq.sum()
        if sum_seq == 0:
            seq = np.full(n, 1.0/n)
            log_seq = np.log(seq+eps)
        else:
            seq /= sum_seq
        # convolution FFT
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa*fa, fft_len)[:2*n-1]
        M = conv.max()
        tolerance = 1e-12
        active = np.where(conv >= M - tolerance)[0]
        # subgradient
        g = np.zeros_like(seq)
        # compute contributions in vectorized way if active size small.
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi+1)
            j_range = k - i_range

        t += 1

        v = beta2*v + (1-beta2)*(g*g)
        m_hat = m/(1-beta1**t)
        v_hat = v/(1-beta2**t)
        # compute decayed learning rate
        elapsed = time.time() - start

        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step
        # clip log
        np.clip(log_seq, -30, 30, out=log_seq)
        # evaluate
        score = _fast_evaluation(seq)

            best_seq = seq.copy()

        seq = np.exp(log_seq)

"""
Propose a low‑score step‑function.

Algorithm
---------
1. **Seed creation** – generate a handful of diverse probability vectors
   (Beta‑laws, sin‑power, inverse‑sqrt, random Dirichlet and, if
   available, the previously discovered `height_sequence_1`).  All seeds
   are normalised to sum = 1.

2. **Small‑scale refinement** – each seed is optimised on a modest
   length (≈1800) by an Adam‑style exponentiated‑gradient descent on the
   simplex.  The update runs in log‑space, uses a decaying learning rate
   and the exact sub‑gradient of the ‑max‑operator (the active set of the
   self‑convolution).

3. **Upscaling** – the best small‑scale vector is linearly interpolated to
   the final length (5000) and re‑normalised.

4. **Long final refinement** – the full‑size vector is further optimised
   with the same Adam routine for the remaining time budget.  Because
   the majority of the time is spent here we can perform many thousands
   of FFT‑based iterations, which steadily lowers the metric.

The code is deliberately time‑aware: the total budget (`budget_s`) is split
between the exploratory stage and the final stage, guaranteeing that the
function returns before the harness timeout.

All helper functions are defined at top level; the only external call is
the provided `evaluate_sequence`, which the harness will invoke on the
returned list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  Returns the metric 2·n·max(conv(a,a)) / (sum(a)^2).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    n = seq.size
    if n == 0:
        return float("inf")
    # next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Seed generators
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the interior of (0,1)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p (p may be negative)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _inverse_sqrt_sequence(length: int) -> np.ndarray:
    """a_i ∝ 1 / sqrt(x·(1‑x)) – the continuous arcsine shape."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = 1.0 / np.sqrt(xs * (1.0 - xs) + 1e-16)
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random point from Dirichlet(concentration,…,concentration)."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Adam‑style exponentiated‑gradient descent on the simplex.
# ----------------------------------------------------------------------
def _adam_refine(
    seq: np.ndarray,
    deadline: float,
    init_lr: float = 0.20,
    final_lr: float = 0.02,
) -> np.ndarray:
    """
    Refine `seq` (non‑negative, sum = 1) using a multiplicative Adam update.
    The learning rate decays linearly from ``init_lr`` to ``final_lr``.
    Returns the best vector seen before ``deadline``.
    """
    n = seq.size
    eps = 1e-15
    log_seq = np.log(seq + eps)

    beta1 = 0.9
    beta2 = 0.999
    m = np.zeros_like(seq)
    v = np.zeros_like(seq)
    t = 0

    fft_len = 1 << ((2 * n - 1).bit_length())

    best_seq = None
    best_score = float("inf")
    it = 0
    start = time.time()

    while time.time() < deadline:
        # ------- obtain a normalised probability vector -------
        seq = np.exp(log_seq)
        sum_seq = seq.sum()
        if sum_seq == 0.0:
            seq = np.full(n, 1.0 / n)
            log_seq = np.log(seq + eps)
            sum_seq = 1.0
        else:
            seq /= sum_seq

        # ------- self‑convolution and active set -------
        fa = np.fft.rfft(seq, fft_len)
        conv = np.fft.irfft(fa * fa, fft_len)[: 2 * n - 1]
        M = conv.max()
        tol = 1e-12
        active = np.where(conv >= M - tol)[0]

        # ------- sub‑gradient of the max‑operator -------
        g = np.zeros_like(seq)
        for k in active:
            lo = max(0, k - (n - 1))
            hi = min(k, n - 1)
            i_range = np.arange(lo, hi + 1, dtype=int)
            j_range = k - i_range
            g[i_range] += seq[j_range]

        # ------- Adam moment updates -------
        t += 1
        m = beta1 * m + (1.0 - beta1) * g
        v = beta2 * v + (1.0 - beta2) * (g * g)

        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # ------- decaying learning rate -------
        elapsed = time.time() - start
        total = deadline - start
        lr = final_lr + (init_lr - final_lr) * max(0.0, 1.0 - elapsed / total)

        # ------- multiplicative (log‑space) step -------
        step = lr * m_hat / (np.sqrt(v_hat) + 1e-8)
        log_seq -= step
        np.clip(log_seq, -30.0, 30.0, out=log_seq)

        # ------- keep the best ----------
        score = _fast_evaluation(seq)
        if score < best_score - 1e-12:
            best_score = score
            best_seq = seq.copy()
            if it % 200 == 0:
                print(f"[Adam] iter {it:6d}  best {best_score:.6f}")

        # ------- occasional jitter to escape stalls ----------
        if np.random.rand() < 0.002:
            idx = np.random.randint(n)
            seq[idx] = max(0.0, seq[idx] + np.random.randn() * 1e-3)
            seq /= seq.sum()
            log_seq = np.log(seq + eps)

        it += 1

    if best_seq is None:
        # fallback – last normalised point
        seq = np.exp(log_seq)
        seq /= seq.sum()
        best_seq = seq

    return best_seq

    np.random.seed(seed)

    deadline = start_time + budget_s - 5.0

    # --------------------------------------------------------------

    n_big = 5000

    # --------------------------------------------------------------
