def propose_candidate(seed=0, budget_s=1000):
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5

    # determine length n. Perhaps use variable: start with n=2000, later increase?
    n = 4000  # as previous.

    #init a: 1/sqrt law normalized
    idx = np.arange(1, n+1)
    a = 1/np.sqrt(idx)
    a = a / a.sum()

    best_a = a.copy()
    best_score = _objective(a)

    # parameters
    max_iters = 1000000  # effectively large, but break due to time.
    eta0 = 0.5
    eps0 = 0.1  # initial smoothing scale relative to max conv
    eps_decay = 0.999 # factor per iteration or slowly

    no_improve = 0
    patience = 2000

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # compute conv via FFT
        conv = _conv_fft(a)
        max_conv = conv.max()
        # Compute smoothing epsilon decreasing
        eps = eps0 * max(1e-6, 0.98**(t // 1000))  # or multiply by factor per iter
        # compute softmax weights
        # subtract max for stability
        z = (conv - max_conv) / eps
        # clip exponent to avoid overflow
        # exponent may be large negative; we can cap at -50
        # Actually we want stable; set lower bound for exponent.
        z = np.clip(z, -50, None)
        w = np.exp(z)
        w_sum = w.sum()
        if w_sum == 0:
            w = np.ones_like(w) / w.size
        else:
            w /= w_sum

        # compute gradient: g = 2 * correlation a with w
        # use np.correlate
        corr = np.correlate(a, w, mode='full')
        # correlation length = n + (2n-1) -1 = 3n-2
        # take slice offset = len(w)-1 = 2n-2
        offset = len(w) - 1
        g = 2.0 * corr[offset: offset + n]

        # exponentiated gradient update
        eta = eta0 * (1.0 - t / max_iters)  # decreasing.

        log_a = np.log(a + 1e-15) - eta * g
        log_a -= np.max(log_a)  # shift
        a_new = np.exp(log_a)
        a_new /= a_new.sum()

        # track improvements
        cur_score = _objective(a_new)
        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_new.copy()
            no_improve = 0
            print(f"[iter {t}] new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # jitter
                jitter = np.random.randn(n) * 1e-4
                a = np.clip(best_a + jitter, 0, None)
                a = _project_to_simplex(a)
                no_improve = 0
                continue

        a = a_new

    # final evaluation
    try:
        final_score = evaluate_sequence(best_a.tolist())
        print("[final] evaluate_sequence reports O={:.6f}".format(final_score))
    except Exception as e:
        print("[final] evaluate_sequence call failed:", e)
        final_score = None

    return best_a.tolist()

    conv = _conv_fft(a)
    # compute smooth max
    max_val = conv.max()
    # avoid overflow using shift
    shifted = (conv - max_val) / eps
    shifted = np.clip(shifted, -50, None)
    exps = np.exp(shifted)

"""
Propose a high‑quality step‑function sequence.

The objective to minimise is

    O = 2 * n * max(conv(a, a)) / (sum(a)**2)

which is scale‑invariant (the denominator simply fixes the normalisation).
We therefore optimise on the probability simplex Σ a_i = 1, a_i ≥ 0.

Algorithm
---------
1.  Choose a length `n`.  If a previous construction (`height_sequence_1`)
    exists we reuse its length, otherwise we pick a power‑of‑two‑friendly
    size (≈ 5000).

2.  Build a few sensible starts:
        * the previous best (if any)
        * the classic 1/√‑law  a_i ∝ 1/√i
        * a random simplex point

3.  For each start run a *smooth* projected gradient descent:
        * the convolution a*a is obtained by a real FFT.
        * the “max” is replaced by a smooth approximation
              f_ε(a) = ε·log Σ_i exp(conv_i/ε)
          (the soft‑max).  For ε → 0 this tends to the true maximum.
        * The gradient of f_ε is
              ∇f_ε = 2 * correlation(a, w) ,
          where w_i = exp(conv_i/ε)/Σ_j exp(conv_j/ε) are the soft‑max
          weights.  Correlation is computed by a second FFT.
        * A simple additive step a←a−α·∇f_ε is taken and the result is
          projected back onto the simplex.
        * ε is slowly decreased (annealing) and the step size α follows a
          1/√t schedule.  When progress stalls a tiny random jitter is added.

   This scheme uses the full information of *all* convolution entries
   (through the soft‑max) instead of only the active constraints,
   typically yielding a better descent direction than the pure sub‑gradient
   method used previously.

4.  Keep the best sequence seen (according to the exact bound
   `evaluate_sequence`).  The whole search stops after the allotted time
   (≈ 1000 s).

The implementation is fully vectorised, uses only NumPy and the standard
library, and respects all contest rules (no I/O, no closures, no lambdas).
"""

import time
import numpy as np

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """
    Euclidean projection of a vector onto the probability simplex
    {x ≥ 0, Σx = 1}.
    """
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
def _objective(a: np.ndarray) -> float:
    """
    Exact bound 2·n·max(conv(a,a)) for a simplex‑normalised vector.
    """
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = np.fft.rfft(a, n=size)
    conv = np.fft.irfft(A * A, n=size).real[: 2 * n - 1]
    return float(2.0 * n * conv.max())


# ----------------------------------------------------------------------
def _smooth_optimize(a0: np.ndarray,
                     deadline: float,
                     max_iters: int = 20000) -> tuple[np.ndarray, float]:
    """
    Smooth (soft‑max) projected gradient descent on the simplex.
    Returns the best vector found and its exact objective value.
    """
    a = a0.copy()
    best_a = a.copy()
    best_score = _objective(a)

    n = a.size
    # a size large enough for the linear correlation a⋆w (length 3n‑2)
    size = 1 << ((3 * n - 2).bit_length())

    eps = None                     # smoothing parameter
    eps_decay = 0.95               # multiplicative decay every `eps_step`
    eps_step = 500
    alpha0 = 0.4                   # initial step size (will decay as 1/√t)
    patience = max_iters // 30
    no_improve = 0

    for t in range(max_iters):
        if time.time() >= deadline:
            break

        # ---- FFT of the current candidate -------------------------------------------------
        A = np.fft.rfft(a, n=size)

        # ---- linear convolution a * a ------------------------------------------------------
        conv = np.fft.irfft(A * A, n=size).real[: 2 * n - 1]

        # ---- smoothing parameter -----------------------------------------------------------
        if eps is None:
            spread = conv.max() - conv.min()
            eps = max(spread * 0.2, 1e-8)      # initialise a modest smoothing scale
        if (t % eps_step) == 0 and t > 0:
            eps = max(eps * eps_decay, 1e-12)  # slowly anneal

        # ---- soft‑max weights ---------------------------------------------------------------
        max_c = conv.max()
        shifted = (conv - max_c) / eps                     # ≤ 0
        shifted = np.clip(shifted, -50, 0)                # safety against under‑flow
        w_exp = np.exp(shifted)
        w = w_exp / w_exp.sum()                          # shape (2n‑1,)

        # ---- gradient of the smoothed max -------------------------------------------------
        # correlation a ⋆ w  (linear, not circular)
        W = np.fft.rfft(w, n=size)
        corr_full = np.fft.irfft(A * np.conj(W), n=size).real[: 3 * n - 2]
        offset = len(w) - 1                               # = 2n‑2
        grad = 2.0 * corr_full[offset: offset + n]       # shape (n,)

        # ---- step size (1/√t decay) --------------------------------------------------------
        alpha = alpha0 / np.sqrt(t + 1.0)

        # ---- projected additive update -----------------------------------------------------
        a_next = a - alpha * grad
        a_next = _project_to_simplex(a_next)

        # ---- exact objective (for logging / early stopping) --------------------------------
        cur_score = 2.0 * n * conv.max()                 # same as _objective(a)

        if cur_score < best_score - 1e-12:
            best_score = cur_score
            best_a = a_next.copy()
            no_improve = 0
            print(f"[smooth] iter {t:5d}  new best O={best_score:.6f}")
        else:
            no_improve += 1
            if no_improve >= patience:
                # small random perturbation to escape flat plateaus
                jitter = np.random.randn(n) * 1e-4
                a = best_a + jitter
                a = np.clip(a, 0.0, None)
                a = _project_to_simplex(a)
                no_improve = 0
                continue

        a = a_next

    return best_a, best_score

def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs):
    """
    Search for a non‑negative step‑function that minimises the bound
    used by `evaluate_sequence`.  Returns the sequence as a plain list.
    """
    np.random.seed(seed)
    deadline = time.time() + budget_s - 0.5   # small safety margin

    # ---- choose a working length -------------------------------------------------
    if "height_sequence_1" in globals():
        prev = np.asarray(height_sequence_1, dtype=float)
        n = prev.size
        prev = prev / prev.sum()
    else:
        # a power‑of‑two‑friendly default length – enough resolution but still fast
        n = 5000
        prev = None

    # ---- build a small pool of starts -------------------------------------------
    starts = []

    if prev is not None:
        starts.append(prev)

    # 1/√ law (theoretically good)
    idx = np.arange(1, n + 1, dtype=float)
    a_sqrt = 1.0 / np.sqrt(idx)
    a_sqrt /= a_sqrt.sum()
    starts.append(a_sqrt)

    # random simplex point
    a_rand = np.random.rand(n)
    a_rand = _project_to_simplex(a_rand)
    starts.append(a_rand)

    best_seq = None
    best_score = float("inf")

    # ---- allocation of iterations ------------------------------------------------
    # Rough budget: give each start an equal share of the remaining time.
    remaining = deadline - time.time()
    iters_per_start = 25000                     # generous, but will be cut by time

    for i, a_start in enumerate(starts):
        if time.time() >= deadline:
            break

        init_score = _objective(a_start)
        print(f"[init {i}] start O={init_score:.6f}")

        a_opt, score_opt = _smooth_optimize(
            a_start,
            deadline=deadline,
            max_iters=iters_per_start,
        )

        print(f"[result {i}] smoothed O={score_opt:.6f}")

        if score_opt < best_score:
            best_score = score_opt
            best_seq = a_opt.copy()

    # ---- fallback (should never be needed) ---------------------------------------
    if best_seq is None:
        best_seq = a_sqrt.copy()
        best_score = _objective(best_seq)

    # ---- final exact evaluation ---------------------------------------------------
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:                     # pragma: no cover
        print("[final] evaluate_sequence call failed:", exc)
        final_score = None

    return best_seq.tolist()
