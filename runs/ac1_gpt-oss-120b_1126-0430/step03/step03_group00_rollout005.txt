"""
Propose a non‑negative step‑function sequence that minimizes

        O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The algorithm is a hybrid of population‑based simulated annealing,
projected sub‑gradient descent and a short Adam polishing phase.
It works entirely with numpy/FFT – no external solvers are required.

Key ideas
---------
* Work on the simplex (sum = 1, a_i ≥ 0) via Euclidean projection.
* Keep a small population (≈10) of candidates; each iteration mutates
  one parent by a random “mass‑transfer”, Gaussian jitter or a
  crossover between two parents.  Acceptance follows a Metropolis rule
  with a slowly cooling temperature.
* Every few thousand iterations a deterministic sub‑gradient step on the
  current maximal convolution entry is performed on the global best.
* After the time budget is exhausted the best sequence is polished with
  a few hundred Adam steps on a log‑sum‑exp surrogate (the same surrogate
  used in the reference solution) – this usually squeezes out the last
  few percent of the objective.
* All heavy work (convolution, gradient) is done by FFTs of length a
  power‑of‑two (≈3·n) which is very fast for n≈5 000.

The routine returns the best sequence found as a plain Python list.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x ≥ 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project vector v onto the simplex."""
    n = v.shape[0]
    # sort descending
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # all non‑positive – return uniform distribution
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# FFT‑based full self‑convolution (length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Return a*a (full convolution) using an FFT of length L."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of the maximal convolution entry (as used in the reference code)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Return the sub‑gradient of max(conv) w.r.t. a."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)  # possibly several maximisers
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


# ----------------------------------------------------------------------
# Smooth surrogate gradient (log‑sum‑exp)
def _smooth_gradient_lse(a: np.ndarray, conv: np.ndarray,
                        alpha: float, L: int) -> np.ndarray:
    """
    Gradient of (2·n/α)·log Σ_k exp(α·conv_k).
    The formula reduces to 2·corr(w, a) where w_k ∝ exp(α·conv_k).
    """
    scaled = alpha * conv
    max_scaled = np.max(scaled)
    exp_scaled = np.exp(scaled - max_scaled)          # stability
    w = exp_scaled / exp_scaled.sum()

    # correlation w ⋆ a   (g_i = 2·∑_j w_{i+j} a_j)
    a_rev = a[::-1]
    fft_w = np.fft.rfft(w, n=L)
    fft_a_rev = np.fft.rfft(a_rev, n=L)
    corr_rev = np.fft.irfft(fft_w * fft_a_rev, n=L)
    n_len = a.size
    grad = 2.0 * corr_rev[n_len - 1 : n_len - 1 + n_len]
    return grad

        total_iters += 1

        # ----------------------------------------------------------
        # Periodic deterministic sub‑gradient step on the global best
        if total_iters - last_subgrad >= subgrad_interval:
            # recompute conv for the best sequence
            conv_best = _conv_fft(best_seq, L)
            g = _subgradient_max_conv(best_seq, conv_best)
            # Simple Polyak step: step = (f - f_best) / ||g||^2
            cur_obj = 2.0 * n * conv_best.max()
            step_size = (cur_obj - best_score) / (np.linalg.norm(g) ** 2 + 1e-12)
            step_size = max(step_size, 0.0)          # safety
            # avoid too tiny steps
            step_size = max(step_size, 1e-4) * 0.9
            best_seq = best_seq - step_size * g
            best_seq = _project_to_simplex(best_seq)
            # recompute objective after sub‑grad step
            best_score = _objective_simplex(best_seq, n, L)
            # replace the worst individual by the improved best (keeps diversity)
            worst_idx = np.argmax([obj for (_, obj) in population])
            population[worst_idx] = (best_seq.copy(), best_score)
            last_subgrad = total_iters
            print(f"[{time.time():.1f}] sub‑grad step improved O={best_score:.6f}")

    # --------------------------------------------------------------
    # Final polishing with Adam on the LSE surrogate ----------------
    time_left = deadline - time.time()
    if time_left > 5.0:
        # Allocate the remaining budget (but not more than ~60 s)
        polish_time = min(55.0, time_left)
        start_polish = time.time()
        a = best_seq.copy()

        # Adam hyper‑parameters
        beta1 = 0.9
        beta2 = 0.999
        eps = 1e-12
        m = np.zeros_like(a)
        v = np.zeros_like(a)
        lr0 = 0.6

        # surrogate schedule
        alpha_start = 20.0
        alpha_end = 12000.0
        max_iter = 150_000

        it = 0
        while time.time() - start_polish < polish_time and it < max_iter:
            conv = _conv_fft(a, L)
            cur_obj = 2.0 * n * conv.max()
            if cur_obj < best_score:
                best_score, best_seq = cur_obj, a.copy()
                print(f"[polish] iter {it} new best O={best_score:.6f}")

            # linear anneal of α
            alpha = alpha_start + (alpha_end - alpha_start) * (it / max_iter)

            grad = _smooth_gradient_lse(a, conv, alpha, L)

            # Adam update
            m = beta1 * m + (1.0 - beta1) * grad
            v = beta2 * v + (1.0 - beta2) * (grad * grad)
            m_hat = m / (1.0 - beta1 ** (it + 1))
            v_hat = v / (1.0 - beta2 ** (it + 1))
            lr = lr0 / math.sqrt(it + 1)

            a_new = a - lr * m_hat / (np.sqrt(v_hat) + eps)

            # tiny jitter occasionally
            if np.random.rand() < 0.001:
                a_new += np.random.randn(n) * 1e-5

            a_new = _project_to_simplex(a_new)
            a = a_new
            it += 1

    # --------------------------------------------------------------
    # Final sanity check with the provided evaluator (if present)
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as e:
        print("[final] evaluate_sequence failed:", e)

    return best_seq.tolist()
