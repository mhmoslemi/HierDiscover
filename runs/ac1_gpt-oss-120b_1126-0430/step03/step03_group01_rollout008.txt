"""
Propose a step‑function (non‑negative sequence) that minimizes

    2 * n * max_k (a * a)[k]   (sum(a) = 1)

The algorithm works on the probability simplex by parametrising the
distribution with logits `z` (softmax → a).  At every iteration the
gradient of a *smooth* approximation of the max‑convolution is computed.
The smooth max is the log‑sum‑exp of the self‑convolution, which yields
a dense (sub‑)gradient that steers the mass away from the current
largest autocorrelation entries.  An Adam optimiser with a decreasing
step‑size and a temperature schedule (τ grows from a small to a large
value) is used.  The support size is gradually enlarged (up to 20 000)
by linear interpolation of the best distribution found so far.
Random perturbations and occasional restarts help escape plateaus.
The best candidate (according to the cheap proxy) is finally evaluated
with the provided ``evaluate_sequence`` and returned.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Helper utilities (top‑level, no closures)
# ----------------------------------------------------------------------


def _softmax(z: np.ndarray) -> np.ndarray:
    """Stable soft‑max returning a probability vector that sums to 1."""
    z_max = np.max(z)
    e = np.exp(z - z_max)
    return e / e.sum()


def _logits_from_distribution(a: np.ndarray, eps: float = 1e-16) -> np.ndarray:
    """Log‑it representation that yields `a` after a soft‑max."""
    return np.log(a + eps)


def _next_pow2(x: int) -> int:
    """Smallest power‑of‑two ≥ x (useful for FFT lengths)."""
    if x <= 1:
        return 1
    return 1 << ((x - 1).bit_length())


def _interpolate_distribution(a: np.ndarray, new_n: int) -> np.ndarray:
    """Linear interpolation of a probability vector to a new length."""
    old_n = a.shape[0]
    if old_n == new_n:
        return a.copy()
    old_x = np.linspace(0.0, 1.0, old_n)
    new_x = np.linspace(0.0, 1.0, new_n)
    new_a = np.interp(new_x, old_x, a)
    new_a = np.maximum(new_a, 0.0)
    s = new_a.sum()
    if s == 0.0:
        new_a[:] = 1.0 / new_n
    else:
        new_a /= s
    return new_a


def _smooth_max_gradient(a: np.ndarray, tau: float):
    """
    Gradient of the smooth max (log‑sum‑exp) of the self‑convolution a*a.
    Returns (grad_a, conv), where `conv` is the full convolution (length 2n‑1)
    and `grad_a` is ∂/∂a of the smooth max.
    """
    n = a.shape[0]

    # ----- self‑convolution (full) ------------------------------------
    Lc = _next_pow2(2 * n - 1)
    fa = np.fft.rfft(a, Lc)
    conv = np.fft.irfft(fa * fa, Lc)[: 2 * n - 1]

    # ----- soft‑max weights over convolution entries ------------------
    c_max = conv.max()
    # shift to avoid overflow, then apply temperature τ
    exp_arg = tau * (conv - c_max)
    # clip extreme negative values for numerical safety (optional)
    w = np.exp(exp_arg)
    w_sum = w.sum()
    if w_sum == 0.0 or not np.isfinite(w_sum):
        # fallback: concentrate all weight on the maximal entry
        w = np.zeros_like(conv)
        w[np.argmax(conv)] = 1.0
        w_sum = 1.0
    p = w / w_sum  # probability distribution over the 2n‑1 offsets

    # ----- gradient w.r.t. a -----------------------------------------
    # ∂/∂a_j (smooth max) = 2 * Σ_k p_k * a_{k‑j}
    a_rev = a[::-1]                     # reversed a
    Lg = _next_pow2(3 * n - 2)          # length for a_rev * p
    fa_rev = np.fft.rfft(a_rev, Lg)
    fp = np.fft.rfft(p, Lg)
    conv_rev = np.fft.irfft(fa_rev * fp, Lg)[: 3 * n - 2]

    # slice corresponds to indices n‑1 … n‑1+n‑1 (length n)
    grad = 2.0 * conv_rev[n - 1 : n - 1 + n]
    return grad, conv

        last_improve = 0
        report_every = 2000

        while time.time() < time_limit:
            it += 1

            # ----- current distribution ---------------------------------
            a = _softmax(z)

            # ----- temperature for this iteration ----------------------
            tau = tau_start + (tau_end - tau_start) * min(1.0, it / tau_ramp_iters)

            # ----- gradient of smooth‑max -------------------------------
            grad_a, conv = _smooth_max_gradient(a, tau)
            cheap_score = 2.0 * n * conv.max()

            # ----- keep the best seen so far -----------------------------
            if cheap_score < best_score:
                best_score = cheap_score
                best_seq = a.copy()
                last_improve = it
                if it % report_every == 0:
                    print(f"[n={n} it={it:06d}] NEW BEST cheap = {cheap_score:.6f}")

            # ----- mirror‑descent gradient on the logits ----------------
            mu = np.dot(grad_a, a)
            grad_z = a * (grad_a - mu)

            # ----- Adam momentum updates --------------------------------
            m = beta1 * m + (1.0 - beta1) * grad_z
            v = beta2 * v + (1.0 - beta2) * (grad_z ** 2)
            m_hat = m / (1.0 - beta1 ** it)
            v_hat = v / (1.0 - beta2 ** it)

            # ----- step direction ---------------------------------------
            step_dir = - m_hat / (np.sqrt(v_hat) + eps_adam)

            # ----- decreasing step‑size --------------------------------
            alpha_t = base_alpha / (math.sqrt(it) + 1.0)

            # ----- update logits ----------------------------------------
            z_new = z + alpha_t * step_dir
            # clamp to avoid overflow in the soft‑max exponent
            z_new = np.clip(z_new, -50.0, 50.0)

            # ----- simple monotonicity check (optional) ---------------
            # we evaluate the cheap score for the trial step; if it worsens
            # we perform a few halvings of the step size.
            accepted = False
            trial_alpha = alpha_t
            for _ in range(4):
                a_trial = _softmax(z + trial_alpha * step_dir)
                _, conv_trial = _smooth_max_gradient(a_trial, tau)
                trial_score = 2.0 * n * conv_trial.max()
                if trial_score <= cheap_score:
                    z = z + trial_alpha * step_dir
                    accepted = True
                    break
                trial_alpha *= 0.5
            if not accepted:
                # no descent found – still move a little
                z = z_new

            # ----- occasional random perturbations ----------------------
            if it % 5000 == 0 and np.random.rand() < 0.25:
                z += 0.02 * np.random.randn(n)
                m *= 0.5
                v *= 0.5

            # ----- early stop if stagnation after many iterations ------
            if it - last_improve > 150000:
                # stagnation, break out early
                break

        return best_seq, best_score

    # ---------------------------------------------------------------
    # 2. Progressive enlargement of the support (if time permits)
    # ---------------------------------------------------------------
    current_n = a0.shape[0]
    current_seq = a0
    best_overall_seq = a0.copy()
    best_overall_score = float("inf")
    last_enlarge = 0

    while time.time() < deadline:
        remaining = deadline - time.time()
        # allocate roughly half the remaining time to the current size,
        # leaving room for a possible enlargement.
        time_for_stage = max(5.0, remaining * 0.5)

        seq, cheap = _optimise_on_support(current_seq, current_n, time.time() + time_for_stage)

        # update the global best
        if cheap < best_overall_score:
            best_overall_score = cheap
            best_overall_seq = seq.copy()
            print(f"[GLOBAL] New best cheap = {cheap:.6f} with n = {current_n}")

        # decide whether to enlarge
        if current_n >= n_max:
            # already at maximum size – continue optimizing a bit more
            if time.time() + 5.0 > deadline:
                break
            # a short extra optimisation phase before exiting
            continue

        # enlarge only if we have comfortable time left
        if time.time() + 10.0 > deadline:
            break

        # enlarge support (linear interpolation)
        new_n = min(n_max, int(current_n * 1.5) + 1)
        print(f"[Enlarge] Support {current_n} → {new_n}")
        # Use the best sequence of the just‑finished stage as seed
        current_seq = _interpolate_distribution(best_overall_seq, new_n)
        current_n = new_n
        # reset counters for the new problem
        last_enlarge = 0

    # ---------------------------------------------------------------
    # 3. Final exact evaluation (uses the provided evaluate_sequence)
    # ---------------------------------------------------------------
    final_seq = best_overall_seq.tolist()
    try:
        final_score = evaluate_sequence(final_seq)
        print(f"Final exact score = {final_score:.6f}")
    except Exception as exc:   # defensive – should never happen
        final_score = float("inf")
        print(f"Error during final evaluation: {exc}")

    return final_seq
