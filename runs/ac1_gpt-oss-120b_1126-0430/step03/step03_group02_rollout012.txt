"""
propose_candidate
-----------------

Search for a non‑negative step‑function that minimises the bound

    O = 2·n·max(conv(a,a)) / (sum(a)**2)

which is exactly the value returned by the supplied ``evaluate_sequence``.
The search proceeds in two stages:

1. **Low‑dimensional parametric sweep** – we consider sequences of the
   form

        a_i ∝ (i + θ)^{-p} ,   i = 1 … n

   where the exponent *p* (≈ ½) and the shift *θ* are varied on a coarse
   grid.  For each candidate the exact objective (using FFT) is computed
   and the best parameters are stored.

2. **Fine‑grained mirror‑descent (Adam) on the simplex** – the best
   candidate from stage 1 is used as a warm‑start.  We optimise the
   sub‑gradient of the max‑convolution in the log‑domain (i.e.
   entropic mirror‑descent) with an Adam style adaptive step‑size.
   The iterate is always projected onto the probability simplex via the
   soft‑max representation, guaranteeing feasibility.

Both stages are time‑capped (default 1000 s).  The final sequence is
returned as a plain list of floats.
"""

import time
import numpy as np
from numpy.fft import fft, ifft

# ----------------------------------------------------------------------
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex {x≥0, Σx=1}."""
    n = v.shape[0]
    if n == 0:
        return v
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho = np.where(u - cssv / np.arange(1, n + 1) > 0)[0]
    if rho.size == 0:
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    return np.maximum(v - theta, 0.0)


def _objective(a: np.ndarray) -> float:
    """Return 2·n·max(conv(a,a)) for a already normalised (Σa=1)."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = fft(a, n=size)
    conv = ifft(A * A).real[: 2 * n - 1]
    return float(2.0 * n * np.max(conv))


def _conv_fft(a: np.ndarray) -> np.ndarray:
    """Linear convolution a*a using a power‑of‑two FFT."""
    n = a.size
    size = 1 << ((2 * n - 2).bit_length())
    A = fft(a, n=size)
    conv = ifft(A * A).real[: 2 * n - 1]
    return conv


def _subgradient_max_conv(a: np.ndarray) -> np.ndarray:
    """
    Sub‑gradient of max(conv(a,a)) w.r.t. a (on the simplex).
    The gradient of the max‐operator is the average of the gradients at
    all indices attaining the maximum (within a tiny tolerance).
    """
    n = a.size
    conv = _conv_fft(a)
    max_val = conv.max()
    ks = np.where(np.abs(conv - max_val) < 1e-14)[0]
    g = np.zeros_like(a)
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        # each term a_i a_j appears twice (i↔j) except when i=j,
        # but the formula below correctly accounts for both.
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size > 0:
        g /= ks.size
    return g


def _softmax(s: np.ndarray) -> np.ndarray:
    """Stable soft‑max (produces a probability vector)."""
    s = s - np.max(s)  # shift for numerical stability
    e = np.exp(s)
    return e / e.sum()


def _adam_optimize(a0: np.ndarray,
                   max_iters: int,
                   lr: float,
                   beta1: float,
                   beta2: float,
                   eps: float,
                   deadline: float) -> tuple[np.ndarray, float]:
    """
    Mirror‑descent on the simplex using Adam updates in log‑space.
    Returns the best sequence found and its objective value.
    """
    n = a0.size
    # initialise log‑parameters (soft‑max representation)
    s = np.log(a0 + 1e-15)
    m = np.zeros_like(s)
    v = np.zeros_like(s)

    best_a = a0.copy()
    best_val = _objective(best_a)

    for t in range(1, max_iters + 1):
        if time.time() >= deadline:
            break

        a = _softmax(s)

        # sub‑gradient of max‑convolution
        g = _subgradient_max_conv(a)

        # convert to gradient w.r.t. soft‑max logits
        mean_g = np.dot(a, g)
        grad_s = a * (g - mean_g)  # shape (n,)

        # Adam moments
        m = beta1 * m + (1.0 - beta1) * grad_s
        v = beta2 * v + (1.0 - beta2) * (grad_s ** 2)

        # bias‑correction
        m_hat = m / (1.0 - beta1 ** t)
        v_hat = v / (1.0 - beta2 ** t)

        # update log‑parameters
        s -= lr * m_hat / (np.sqrt(v_hat) + eps)

        # evaluate
        cur_val = _objective(_softmax(s))
        if cur_val < best_val - 1e-12:
            best_val = cur_val
            best_a = _softmax(s).copy()
            # occasionally print progress
            if t % 500 == 0:
                print(f"[adam] iter {t:5d}  new best O={best_val:.6f}")

    return best_a, best_val
