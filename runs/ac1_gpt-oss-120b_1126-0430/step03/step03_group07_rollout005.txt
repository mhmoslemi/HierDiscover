```
def project_simplex(v):
    # v: numpy array
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(n) + 1
    cond = u - cssv / ind > 0
    rho = ind[cond][-1]
    theta = cssv[cond][-1] / rho
    w = np.maximum(v - theta, 0)
    return w
```

```
def softmax_weights(conv, alpha):
```

Let's write the final answer with code inside triple backticks.assistantfinal**Solution Overview**

The metric to minimise is  

\[
J(a)=\frac{2n\;\max\limits_{k}(a*a)_k}{\bigl(\sum_i a_i\bigr)^2},
\qquad a_i\ge 0 .
\]

Because the metric is *scale‑invariant* we can normalise the vector to a probability
distribution (∑ a = 1) and only minimise the peak of the self‑convolution.
The programme below therefore works on the simplex.

The optimisation strategy is a **projected Adam‑style gradient descent** on a
*smoothed* version of the max‑operator:

*   For a given vector `a` we compute the full convolution `c = a * a`.
*   A temperature parameter `α` turns the peak into a smooth soft‑max  
    \(w_k = \frac{e^{αc_k}}{\sum_j e^{αc_j}}\).  
    For `α → ∞` the weighted sum \(\sum_k w_k c_k\) converges to the true
    maximum.
*   The gradient of the weighted sum w.r.t. `a` can be derived analytically and
    is obtained by a second convolution:
    \[
    \nabla_a = 2\;a\star w ,
    \qquad
    (a\star w)_i = \sum_{j} a_j\,w_{i+j}\; .
    \]
    Both convolutions are evaluated with the FFT ( O(n log n) ).
*   After the Adam update the vector is **projected onto the probability
    simplex** (Duchi et al. 2008) so that the constraints `a_i ≥ 0, ∑a_i = 1`
    are always satisfied.
*   The temperature `α` is increased **exponentially** from a small value
    (`α≈0.5`) to a large value (`α≈200`) during the run – this gives a smooth
    start and focuses on the true peak later on.
*   Several diverse seeds (Beta‑distributions, sin‑power windows,
    Dirichlet draws, a triangular shape and the DPSS (Slepian) window) are
    generated.  Each seed receives a time slice, is refined with the optimiser
    on a *small* size (`n≈2000`) and the best result is kept.
*   The best small‑size vector is up‑sampled to the final length (`n=5000`) by
    linear interpolation and refined again for the remaining time budget.

The whole procedure runs completely inside the allotted **1000 s** and returns
the best vector found.  All numbers are non‑negative, the sum is normalised to
1 (the wrapper `evaluate_sequence` will later rescale if needed) and the code
uses only NumPy, SciPy and the standard library.

```python
"""
Propose a low‑score step‑function.

Algorithm
---------
1. Generate a diverse set of seed probability vectors:
   * symmetric / asymmetric Beta densities,
   * sin ^p windows,
   * random Dirichlet draws,
   * a simple triangular shape,
   * a DPSS (Slepian) window (always non‑negative after clipping).

2. For every seed a short “soft‑max Adam” optimisation is performed on a
   modest size (n≈2000).  The objective is a smoothed version of the
   peak‑convolution; the temperature α grows from 0.5 to 200 during the run.
   After each Adam step the vector is projected onto the probability simplex.

3. The best seed is interpolated to the final length (n=5000) and a final
   optimisation (same method, remaining time) is carried out.

The optimiser works entirely with FFT‑based convolutions, is scale‑invariant,
and respects the non‑negativity / unit‑sum constraints.  The resulting
sequence is returned as a plain Python list of floats.
"""

import time
import math
import numpy as np
from scipy.signal.windows import dpss

# ----------------------------------------------------------------------
# Helper: projection onto the probability simplex (Duchi et al. 2008)
# ----------------------------------------------------------------------
def _project_simplex(v: np.ndarray) -> np.ndarray:
    """Return the Euclidean projection of v onto {x >= 0, sum(x)=1}."""
    n = v.size
    # sort v in descending order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    ind = np.arange(1, n + 1)
    # find the last index where the condition holds
    cond = u - cssv / ind > 0
    if not np.any(cond):
        # all entries are non‑positive → uniform distribution
        return np.full(n, 1.0 / n)
    rho = ind[cond][-1] - 1        # zero‑based index
    theta = cssv[cond][-1] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Fast metric (identical to the supplied evaluate_sequence but without checks)
# ----------------------------------------------------------------------
def _fast_metric(a: np.ndarray) -> float:
    """2·n·max(conv(a,a)) / (sum(a)^2)  for a with sum≈1."""
    n = a.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    s = a.sum()
    if s < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (s * s))


# ----------------------------------------------------------------------
# Full (a*a) convolution via FFT (returns length 2n‑1)
# ----------------------------------------------------------------------
def _full_convolution(a: np.ndarray) -> np.ndarray:
    n = a.size
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(a, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    return conv


# ----------------------------------------------------------------------
# Gradient of the soft‑max weighted sum of the convolution
# ----------------------------------------------------------------------
def _softmax_gradient(a: np.ndarray, conv: np.ndarray, alpha: float) -> np.ndarray:
    """
    Compute ∇_a  Σ_k w_k conv_k ,  where w_k ∝ exp(alpha·conv_k).
    Returns a vector of length n.
    """
    n = a.size
    # soft‑max weights
    z = conv * alpha
    z_max = np.max(z)
    exp_z = np.exp(z - z_max)          # stability
    w = exp_z / exp_z.sum()            # shape (2n‑1,)

    # gradient = 2 * (a ⋆ w)   where (a ⋆ w)_i = Σ_j a_j w_{i+j}
    # This is obtained by convolving a with the reversed w.
    w_rev = w[::-1]
    L = 1 << ((n + w_rev.size - 1).bit_length())
    fa = np.fft.rfft(a, L)
    fw = np.fft.rfft(w_rev, L)
    conv2 = np.fft.irfft(fa * fw, L)[: n + w_rev.size - 1]   # length 3n‑2

    offset = w_rev.size - 1          # = 2n‑2
    # grad_i = 2 * conv2[offset - i]   for i = 0..n‑1
    grad = 2.0 * conv2[offset - np.arange(n)]
    return grad


# ----------------------------------------------------------------------
# Adam optimiser with simplex projection and exponentially increasing α.
# ----------------------------------------------------------------------
def _adam_optimize(
    a0: np.ndarray,
    deadline: float,
    alpha_min: float = 0.5,
    alpha_max: float = 200.0,
) -> np.ndarray:
    """
    Refine a probability vector a0 (∑a0=1) using Adam and a soft‑max
    approximation of the max‑convolution.
    The routine stops when time.time() reaches ``deadline`` and returns the
    best vector (lowest true metric) observed.
    """
    a = a0.copy()
    n = a.size
    a = _project_simplex(a)

    # Adam state
    m = np.zeros(n)
    v = np.zeros(n)
    beta1 = 0.9
    beta2 = 0.999
    eps = 1e-8
    lr = 0.05                # base learning rate for the additive update

    best = a.copy()
    best_score = _fast_metric(best)

    start = time.time()
    it = 0
    while time.time() < deadline:
        it += 1
        # --- current convolution ---
        conv = _full_convolution(a)

        # --- temperature schedule (exponential growth) ---
        fraction = (time.time() - start) / (deadline - start)
        alpha = alpha_min * (alpha_max / alpha_min) ** fraction

        # --- gradient of the smoothed objective ---
        grad = _softmax_gradient(a, conv, alpha)

        # --- Adam moment updates ---
        m = beta1 * m + (1.0 - beta1) * grad
        v = beta2 * v + (1.0 - beta2) * (grad * grad)
        m_hat = m / (1.0 - beta1 ** it)
        v_hat = v / (1.0 - beta2 ** it)

        # --- additive Adam step ---
        step = lr * m_hat / (np.sqrt(v_hat) + eps)
        a -= step

        # --- project back onto the simplex ---
        a = _project_simplex(a)

        # occasional tiny jitter to escape plateaus
        if np.random.rand() < 0.001:
            j = np.random.randint(n)
            a[j] = max(0.0, a[j] + np.random.randn() * 1e-4)
            a = _project_simplex(a)

        # --- evaluate true metric and keep the best ---
        if it % 200 == 0:
            cur_score = _fast_metric(a)
            if cur_score < best_score - 1e-12:
                best_score = cur_score
                best = a.copy()
                print(f"[Adam] iter {it:6d}   best {best_score:.6f}   α={alpha:.2f}")

    # final check (in case the very last iterate is the best)
    final_score = _fast_metric(a)
    if final_score < best_score:
        best = a.copy()
        best_score = final_score

    return best


# ----------------------------------------------------------------------
# Seed generators (all return a probability vector of the requested length)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    # avoid singularities at the boundaries (xs in (0,1))
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq = np.nan_to_num(seq, nan=0.0, posinf=0.0, neginf=0.0)
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.nan_to_num(vals, nan=0.0, posinf=0.0, neginf=0.0)
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _triangular_sequence(length: int) -> np.ndarray:
    i = np.arange(length, dtype=float)
    seq = (i + 1) * (length - i)
    seq = np.maximum(seq, 0.0)
    seq /= seq.sum()
    return seq


def _dpss_sequence(length: int, nw: float = 3.0) -> np.ndarray:
    # DPSS (Slepian) windows are real and mostly non‑negative;
    # we clip any tiny negative wiggles to zero.
    seq = dpss(length, nw, Kmax=1, sym=True)[0]
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq

# Main entry point required by the evaluation harness
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Return a step‑function (list of non‑negative floats) whose
    evaluation metric is as small as possible.
    The optimisation runs for at most ``budget_s`` seconds.
    """
    np.random.seed(seed)
    start_time = time.time()
    # keep a small safety margin for the harness
    deadline = start_time + budget_s - 5.0

    # ------------------------------------------------------------------
    # 1. Parameters
    # ------------------------------------------------------------------
    n_small = 2000          # exploratory size (fast FFTs)
    n_big = 5000            # final size required by the challenge

    # ------------------------------------------------------------------
    # 2. Build a pool of seeds (name, vector)
    # ------------------------------------------------------------------
    seeds = []

    # classic arcsine (Beta(0.5,0.5))
    seeds.append(("arcsine", _beta_sequence(0.5, 0.5, n_small)))

    # symmetric betas around 0.5
    for a in [0.35, 0.40, 0.45, 0.55, 0.60, 0.65, 0.80, 1.00]:
        seeds.append((f"beta_sym_{a:.2f}", _beta_sequence(a, a, n_small)))

    # a few asymmetric betas
    for a, b in [(0.3, 0.7), (0.7, 0.3), (0.4, 0.9), (0.9, 0.4)]:
        seeds.append((f"beta_{a:.2f}_{b:.2f}", _beta_sequence(a, b, n_small)))

    # sin‑power windows (p may be negative)
    for p in [-0.6, -0.5, -0.4, -0.3, -0.2, 0.0, 0.2]:
        seeds.append((f"sin_pow_{p:.1f}", _sin_power_sequence(p, n_small)))

    # random Dirichlet draws (concentration <1 → spiky)
    for i in range(4):
        seeds.append((f"rand_dir_{i}", _random_dirichlet(n_small, concentration=0.4)))

    # a simple triangular shape
    seeds.append(("triangular", _triangular_sequence(n_small)))

    # DPSS (Slepian) window – known for low autocorrelation sidelobes
    seeds.append(("dpss", _dpss_sequence(n_small, nw=3.0)))

    # ------------------------------------------------------------------
    # 3. Run a short Adam refinement on each seed (time‑sliced)
    # ------------------------------------------------------------------
    best_vec = None
    best_score = float("inf")
    remaining = list(seeds)

    while remaining and time.time() < deadline:
        # allocate a fair slice of the remaining time
        time_left = deadline - time.time()
        per_seed = max(5.0, time_left / (len(remaining) + 1))
        name, vec = remaining.pop(0)
        vec = np.maximum(vec, 0.0)
        vec = _project_simplex(vec)

        # optimise this seed
        seed_deadline = time.time() + per_seed
        refined = _adam_optimize(vec, seed_deadline)
        score = _fast_metric(refined)

        print(f"[seed {name}]  score {score:.6f}")

        if score < best_score:
            best_score = score
            best_vec = refined.copy()
            print(f"*** new global best = {best_score:.6f}")

    # ------------------------------------------------------------------
    # 4. Upscale the best small‑size vector to the final length
    # ------------------------------------------------------------------
    if best_vec is None:
        best_vec = np.full(n_small, 1.0 / n_small)

    # linear interpolation (preserves monotonicity and shape)
    x_small = np.arange(n_small, dtype=float)
    x_big = np.linspace(0, n_small - 1, n_big)
    upscaled = np.interp(x_big, x_small, best_vec)

    upscaled = np.maximum(upscaled, 0.0)
    upscaled = _project_simplex(upscaled)

    # ------------------------------------------------------------------
    # 5. Final refinement on the full size (remaining budget)
    # ------------------------------------------------------------------
    if time.time() < deadline:
        print("[final] starting refinement on full length...")
        final_vec = _adam_optimize(upscaled, deadline)
        final_score = _fast_metric(final_vec)
        print(f"[final] best score {final_score:.6f}")
        best_vec = final_vec
        best_score = final_score

    # ------------------------------------------------------------------
    # 6. Return the result as a plain Python list
    # ------------------------------------------------------------------
    print(f"[summary] final metric {best_score:.6f} with length {best_vec.size}")
    return best_vec.tolist()
```
