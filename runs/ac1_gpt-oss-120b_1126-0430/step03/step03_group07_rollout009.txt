def project_simplex(v):
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1
    rho = np.where(u > cssv / np.arange(1, len(v)+1))[0][-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0)
    return w

def _smooth_max_gradient(seq, beta):
    n = seq.size
    # compute conv_self
    m_len = 1 << ((2*n - 1).bit_length())
    fa = np.fft.rfft(seq, m_len)
    conv = np.fft.irfft(fa * fa, m_len)[:2*n-1]
    max_val = conv.max()
    # compute softmax weights
    # compute exponent = beta*(conv - max_val) <= 0
    exp_vals = np.exp(beta * (conv - max_val))
    sum_exp = exp_vals.sum()
    if sum_exp == 0:
        w = np.full_like(conv, 1.0/(conv.size))
    else:
        w = exp_vals / sum_exp
    # reverse w
    w_rev = w[::-1]
    m_len2 = 1 << ((n + (2*n-1) - 1).bit_length())
    # compute conv of seq and w_rev
    fa_seq = np.fft.rfft(seq, m_len2)
    fa_w = np.fft.rfft(w_rev, m_len2)
    conv_grad_full = np.fft.irfft(fa_seq * fa_w, m_len2)
    # conv_grad_full length = m_len2 >= n + (2n-1)-1 = 3n-2
    # take indices 2n-2 - i
    grad = 2.0 * conv_grad_full[ (2*n-2) - np.arange(n) ]
    # Return both max_val (for metric) and gradient
    return max_val, grad

def _project_simplex(v):
    # Based on algorithm by Duchi et al.
    n = v.shape[0]
    u = np.sort(v)[::-1]

"""
Propose a step‑function (non‑negative vector) that minimizes

    2·n·max(conv(a,a)) / (sum(a)²)

using a two‑phase strategy:

1️⃣  Generate a large, diverse set of seed distributions (Beta,
    sinusoidal powers, random Dirichlet, simple linear ramps, …)
    directly at the target length (≈5000).  Keep the best few seeds
    according to a fast FFT‑based evaluation.

2️⃣  Refine each promising seed with a *projected gradient descent* on
    the simplex.  The objective (the maximal entry of the self‑
    convolution) is smoothed with a soft‑max (temperature β) so that a
    well‑behaved gradient can be obtained via FFTs.  The update is a
    simple step followed by an exact Euclidean projection onto the
    probability simplex.  β is increased slowly, turning the smooth
    problem into the original non‑smooth one.

The whole search runs for the allotted time (default 1000 s) and
returns the best vector found.  All numbers are non‑negative and the
evaluation harness will call the provided ``evaluate_sequence`` function
on the result.
"""

import time
import math
import numpy as np

# ----------------------------------------------------------------------
# Fast evaluation (identical to the supplied evaluate_sequence but without
# the security checks).  Returns a float (never np.inf for valid input).
# ----------------------------------------------------------------------
def _fast_evaluation(seq: np.ndarray) -> float:
    """Metric 2·n·max(conv(seq,seq)) / (sum(seq)²)."""
    n = seq.size
    if n == 0:
        return float("inf")
    # FFT length: next power of two ≥ 2·n‑1
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]
    max_b = conv.max()
    sum_a = seq.sum()
    if sum_a < 0.01:
        return float("inf")
    return float(2.0 * n * max_b / (sum_a * sum_a))


# ----------------------------------------------------------------------
# Simplex projection (Euclidean) – Duchi et al. (2008)
# ----------------------------------------------------------------------
def _proj_simplex(v: np.ndarray) -> np.ndarray:
    """Project the vector v onto the probability simplex (sum=1, v≥0)."""
    if v.ndim != 1:
        raise ValueError("v must be a 1‑D array")
    n = v.shape[0]
    # sort in decreasing order
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    ind = np.arange(1, n + 1)
    # Find the last index where u_i > (cumulative_sum_i - 1) / i
    rho = np.where(u > cssv / ind)[0]
    if rho.size == 0:
        # all entries are zero – return uniform vector
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1.0)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Seed generators (all produce a probability vector of the given length)
# ----------------------------------------------------------------------
def _beta_sequence(alpha: float, beta: float, length: int) -> np.ndarray:
    """Beta(α,β) density sampled on the mid‑points of the grid."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    a = np.power(xs, alpha - 1.0, out=np.empty_like(xs))
    b = np.power(1.0 - xs, beta - 1.0, out=np.empty_like(xs))
    seq = a * b
    seq[np.isinf(seq)] = 0.0
    seq = np.maximum(seq, 0.0)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _sin_power_sequence(p: float, length: int) -> np.ndarray:
    """a_i ∝ sin(π·(i+0.5)/length)^p (p may be negative)."""
    xs = (np.arange(length, dtype=float) + 0.5) / length
    vals = np.sin(np.pi * xs) ** p
    vals = np.maximum(vals, 0.0)
    if vals.sum() == 0.0:
        vals = np.full(length, 1.0 / length)
    else:
        vals /= vals.sum()
    return vals


def _random_dirichlet(length: int, concentration: float = 0.5) -> np.ndarray:
    """Draw a random Dirichlet point with the given concentration."""
    seq = np.random.gamma(concentration, 1.0, size=length)
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


def _linear_ramp(length: int, slope: float = 1.0) -> np.ndarray:
    """Simple linear ramp from 0 to 1 (then normalised)."""
    xs = np.linspace(0, 1, length) ** slope
    seq = xs
    if seq.sum() == 0.0:
        seq = np.full(length, 1.0 / length)
    else:
        seq /= seq.sum()
    return seq


# ----------------------------------------------------------------------
# Smoothed max‑gradient of the self‑convolution.
# ----------------------------------------------------------------------
def _smooth_max_and_grad(seq: np.ndarray, beta: float) -> tuple[float, np.ndarray]:
    """
    Returns (max_conv, gradient) where the gradient is the derivative of
    the soft‑max approximation of max(conv(seq,seq)) with temperature β.
    The gradient is computed efficiently with FFTs (O(n log n)).
    """
    n = seq.size
    # ----- convolution of seq with itself -----
    m = 1 << ((2 * n - 1).bit_length())
    fa = np.fft.rfft(seq, m)
    conv = np.fft.irfft(fa * fa, m)[: 2 * n - 1]

    max_val = conv.max()

    # ----- soft‑max weights over the convolution entries -----
    # exponent ≤ 0, safe from overflow
    exp_vals = np.exp(beta * (conv - max_val))
    w = exp_vals / exp_vals.sum()          # w has length 2n‑1 and sums to 1

    # ----- gradient = 2 * conv(seq , w_rev) evaluated at appropriate indices -----
    w_rev = w[::-1]                         # reverse order
    m2 = 1 << ((n + (2 * n - 1) - 1).bit_length())   # length ≥ 3n‑2
    fa_seq = np.fft.rfft(seq, m2)
    fa_w = np.fft.rfft(w_rev, m2)
    conv_grad_full = np.fft.irfft(fa_seq * fa_w, m2)   # length m2

    # Extract indices 2n‑2 - i   (see analysis)
    idx = (2 * n - 2) - np.arange(n, dtype=int)
    grad = 2.0 * conv_grad_full[idx]
    return max_val, grad


# ----------------------------------------------------------------------
# Projected gradient descent on the simplex (smooth max).
# ----------------------------------------------------------------------
def _pgd_smooth(seq: np.ndarray, deadline: float,
                beta_start: float = 4.0,
                beta_factor: float = 1.025,
                beta_max: float = 200.0,
                lr0: float = 0.12,
                restart_prob: float = 0.001) -> np.ndarray:
    """
    Minimise max(conv(seq,seq)) using a smooth surrogate.
    Returns the best vector seen before ``deadline``.
    """
    n = seq.size
    best = seq.copy()
    best_score = _fast_evaluation(best)

    # optimisation state
    beta = beta_start
    t = 0
    lr = lr0

    while time.time() < deadline:
        t += 1

        # compute smoothed max and its gradient
        max_val, grad = _smooth_max_and_grad(seq, beta)

        # simple back‑tracking line search (at most 5 halvings)
        step_accepted = False
        for _ in range(5):
            # take a step and project onto simplex
            cand = _proj_simplex(seq - lr * grad)

            cand_score = _fast_evaluation(cand)
            if cand_score <= best_score * (1.0 + 1e-12):
                # improvement or (very) equal – accept
                seq = cand
                if cand_score < best_score - 1e-12:
                    best_score = cand_score
                    best = cand.copy()
                step_accepted = True
                break
            # otherwise shrink step size
            lr *= 0.5

        if not step_accepted:
            # restore original step size for next iteration
            lr = lr0

        # gradually warm‑up β (more smoothing → easier optimisation)
        beta = min(beta * beta_factor, beta_max)

        # occasional random jitter to escape plateaus
        if np.random.rand() < restart_prob:
            jitter = np.random.randn(n) * 1e-4
            seq = _proj_simplex(seq + jitter)

        # occasional progress output
        if t % 1000 == 0:
            print(f"[PGD] iter {t:6d}  best {best_score:.6f}  β={beta:.1f}")

    return best


# ----------------------------------------------------------------------
# Main entry point required by the evaluation harness.
# ----------------------------------------------------------------------
def propose_candidate(seed: int = 42, budget_s: int = 1000, **kwargs) -> list[float]:
    """
    Produce a low‑score step‑function.

    Strategy
    --------
    1.  Create a large collection of diversified seed vectors (Beta,
        sin‑power, Dirichlet, linear ramps, maybe the historic best).
    2.  Evaluate them quickly; keep the K best (K≈7).
    3.  Refine each of those K seeds with a projected‑gradient descent
        that smooths the “max” operator with a soft‑max (parameter β)
        and uses an exact projection onto the probability simplex.
    4.  Return the best vector discovered.
    """
    np.random.seed(seed)
    start = time.time()
    deadline = start + budget_s - 5.0                 # safety margin

    # ------------------------------------------------------------------
    # Target length – 5000 works well for the original challenge.
    # ------------------------------------------------------------------
    n = 5000

    # ------------------------------------------------------------------
    # 1️⃣  Build a diverse set of seeds
    # ------------------------------------------------------------------
    seeds = []

    # a) classic arcsine (Beta(0.5,0.5))
    seeds.append(_beta_sequence(0.5, 0.5, n))

    # b) symmetric Beta with a fine grid of α values
    for a in np.linspace(0.15, 1.2, 13):
        seeds.append(_beta_sequence(a, a, n))

    # c) a handful of asymmetric Beta shapes
    for a, b in [(0.25, 0.75), (0.75, 0.25), (0.2, 0.9), (0.9, 0.2)]:
        seeds.append(_beta_sequence(a, b, n))

    # d) sinusoidal power families (both concave and convex)
    for p in [-0.8, -0.6, -0.4, -0.2, 0.0, 0.2, 0.4]:
        seeds.append(_sin_power_sequence(p, n))

    # e) random Dirichlet draws with various concentrations
    for conc in [0.2, 0.4, 0.7]:
        for _ in range(6):
            seeds.append(_random_dirichlet(n, concentration=conc))

    # f) simple linear ramps (different exponents)
    for slope in [0.5, 1.0, 2.0, 3.0]:
        seeds.append(_linear_ramp(n, slope=slope))

    # g) reuse the historic best if it exists in the global namespace
    try:
        historic = globals()["height_sequence_1"]
        if isinstance(historic, list) and len(historic) == n:
            seeds.append(np.array(historic, dtype=float))
        elif isinstance(historic, list):
            # resize via simple interpolation
            x_old = np.arange(len(historic), dtype=float)
            x_new = np.linspace(0, len(historic) - 1, n)
            hist_arr = np.interp(x_new, x_old, historic)
            hist_arr = np.maximum(hist_arr, 0.0)
            hist_arr /= hist_arr.sum()
            seeds.append(hist_arr)
    except Exception:
        pass

    # ------------------------------------------------------------------
    # 2️⃣  Quick evaluation of every seed, keep the K best
    # ------------------------------------------------------------------
    K = 7
    best_seeds = []   # list of (score, vector)
    for idx, s in enumerate(seeds):
        # ensure a proper probability vector
        s = np.maximum(s, 0.0)
        if s.sum() == 0.0:
            s = np.full(n, 1.0 / n)
        else:
            s = s / s.sum()
        score = _fast_evaluation(s)
        best_seeds.append((score, s))
        print(f"[seed {idx:03d}]  score {score:.6f}")

    # keep only the K lowest‑score seeds
    best_seeds.sort(key=lambda x: x[0])
    best_seeds = best_seeds[:K]

    # ------------------------------------------------------------------
    # 3️⃣  Refine each promising seed with projected gradient descent
    # ------------------------------------------------------------------
    global_best = None
    global_best_score = float("inf")

    # allocate roughly equal time to each refinement run
    remaining = deadline - time.time()
    per_seed_deadline = remaining / (len(best_seeds) + 1e-9)

    for i, (seed_score, seed_vec) in enumerate(best_seeds):
        print(f"\n[refine #{i}]  start score {seed_score:.6f}")
        split_deadline = time.time() + per_seed_deadline
        refined = _pgd_smooth(seed_vec, split_deadline)
        refined_score = _fast_evaluation(refined)
        print(f"[refine #{i}]  refined score {refined_score:.6f}")

        if refined_score < global_best_score:
            global_best_score = refined_score
            global_best = refined.copy()
            print(f"*** new global best = {global_best_score:.6f}")

    # ------------------------------------------------------------------
    # 4️⃣  Final optional polishing (use any leftover budget)
    # ------------------------------------------------------------------
    if time.time() < deadline and global_best is not None:
        print("\n[final polish]  remaining budget, one more refinement pass...")
        final_vec = _pgd_smooth(global_best, deadline)
        final_score = _fast_evaluation(final_vec)
        print(f"[final polish]  final score {final_score:.6f}")
        if final_score < global_best_score:
            global_best = final_vec
            global_best_score = final_score

    # ------------------------------------------------------------------
    # 5️⃣  Return result as a plain Python list
    # ------------------------------------------------------------------
    if global_best is None:
        # fallback – uniform distribution (should never happen)
        global_best = np.full(n, 1.0 / n)

    print(f"\n[summary] final metric {global_best_score:.6f} with length {global_best.size}")
    return global_best.tolist()
