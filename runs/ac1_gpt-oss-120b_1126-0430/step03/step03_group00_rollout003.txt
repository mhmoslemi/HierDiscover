"""
Propose a non‑negative step‑function (probability vector) that minimises

    O(a) = 2·n·max(conv(a,a)) / (sum(a)^2)

The implementation consists of three stages:

1️⃣  **Parametric seeding** – generate a few simple families (uniform,
    symmetric exponential, symmetric Gaussian, left‑hand exponential) for
    several lengths (800 … 6000).  Keep the best few candidates.

2️⃣  **Deterministic sub‑gradient descent** – starting from the best seed,
    repeatedly move the current vector opposite to the sub‑gradient of the
    maximal convolution entry.  The step size is adaptively limited so
    that the vector stays in the probability simplex.  Occasionally a tiny
    jitter is added and very small entries are pruned.

3️⃣  **Stochastic refinement** – a Metropolis‑style mass‑transfer local
    search (move a random fraction of probability mass from a “high‑gradient”
    index to a “low‑gradient” one).  If the move improves the exact objective
    it is accepted, otherwise it may be accepted with a temperature‑controlled
    probability.  Sub‑gradient nudges are interleaved to accelerate convergence.

The whole routine respects a 1000 s wall‑clock budget (a small safety margin
is kept).  The best sequence found is finally checked with the provided
`evaluate_sequence` function and returned as a plain Python list.
"""

import time
import math
import numpy as np


# ----------------------------------------------------------------------
# Helper: Euclidean projection onto the probability simplex {x >= 0, sum(x)=1}
def _project_to_simplex(v: np.ndarray) -> np.ndarray:
    """Project a vector onto the probability simplex."""
    n = v.shape[0]
    u = np.sort(v)[::-1]
    cssv = np.cumsum(u) - 1.0
    rho_arr = u - cssv / np.arange(1, n + 1)
    rho = np.where(rho_arr > 0)[0]
    if rho.size == 0:
        # All entries are non‑positive – return the uniform distribution.
        return np.full_like(v, 1.0 / n)
    rho = rho[-1]
    theta = cssv[rho] / (rho + 1)
    w = np.maximum(v - theta, 0.0)
    return w


# ----------------------------------------------------------------------
# Full self‑convolution using FFT (returns length 2·n‑1)
def _conv_fft(a: np.ndarray, L: int) -> np.ndarray:
    """Convolution a*a via FFT of length L (L must be >= 2·n‑1)."""
    fft_a = np.fft.rfft(a, n=L)
    conv_full = np.fft.irfft(fft_a * fft_a, n=L)
    return conv_full[: 2 * a.size - 1]


# ----------------------------------------------------------------------
# Exact objective for a simplex‑normalised vector (sum == 1)
def _objective_simplex(a: np.ndarray, n: int, L: int) -> float:
    """Return 2·n·max(conv(a,a)) (a is assumed to sum to 1)."""
    conv = _conv_fft(a, L)
    return 2.0 * n * conv.max()


# ----------------------------------------------------------------------
# Sub‑gradient of max(conv) w.r.t. a (averaged over all maximisers)
def _subgradient_max_conv(a: np.ndarray, conv: np.ndarray) -> np.ndarray:
    """Sub‑gradient of the maximal entry of the self‑convolution."""
    n = a.size
    max_val = conv.max()
    ks = np.flatnonzero(conv == max_val)            # indices attaining the max
    g = np.zeros_like(a)
    # For each maximising shift k, add contribution 2·a_j to all i that pair with it.
    for k in ks:
        i_start = max(0, k - (n - 1))
        i_end = min(n - 1, k)
        i_idx = np.arange(i_start, i_end + 1, dtype=int)
        j_idx = k - i_idx
        g[i_idx] += 2.0 * a[j_idx]
    if ks.size:
        g /= ks.size
    return g

                if new_obj < cur_obj:
                    a, conv, cur_obj = a_new, conv_new, new_obj
                    no_improve = 0
                    if cur_obj < best_obj:
                        best_obj, best_seq = cur_obj, a.copy()
                        print(f"[{time.time():.1f}] New global best O={best_obj:.6f} (restart={restart_counter}, it={it_loc})")
                else:
                    # Metropolis acceptance with temperature T
                    prob = math.exp((cur_obj - new_obj) / max(T, 1e-12))
                    if np.random.rand() < prob:
                        a, conv, cur_obj = a_new, conv_new, new_obj
                        no_improve = 0
                    else:
                        no_improve += 1
            else:
                no_improve += 1

            # ---- occasional sub‑gradient nudge (helps when the max is flat) ----
            if it_loc % 9 == 0:
                g = _subgradient_max_conv(a, conv)
                eta = 0.07 * (1.0 - it_loc / max_local_iter)
                a = a - eta * g
                a = _project_to_simplex(a)
                conv = _conv_fft(a, L)
                cur_obj = 2.0 * n * conv.max()
                if cur_obj < best_obj:
                    best_obj, best_seq = cur_obj, a.copy()
                    print(f"[{time.time():.1f}] Sub‑grad improved O={best_obj:.6f}")

            # temperature decay & loop bookkeeping
            T *= 0.9998
            it_loc += 1

            if no_improve >= no_improve_limit:
                break

        restart_counter += 1
        if time.time() >= deadline:
            break

    # ------------------------------------------------------------------
    # Final sanity check with the official evaluator
    try:
        final_score = evaluate_sequence(best_seq.tolist())
        print(f"[final] evaluate_sequence reports O={final_score:.6f}")
    except Exception as exc:
        print("[final] evaluate_sequence failed:", exc)

    # Return the best sequence found (plain Python list)
    return best_seq.tolist()
